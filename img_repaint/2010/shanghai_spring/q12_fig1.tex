  gflowterm/.style={draw, shape=rectangle, rounded corners=1.5mm, minimum width=26mm, minimum height=10mm, inner xsep=3mm, inner ysep=1.5mm, align=center, font=\normalsize, execute at begin node={\setlength{\parindent}{0pt}}},
  gflowio/.style={draw, trapezium, trapezium left angle=72, trapezium right angle=108, minimum width=34mm, minimum height=10mm, inner sep=1.5mm, align=center, font=\normalsize, execute at begin node={\setlength{\parindent}{0pt}}},
  gflowdec/.style={draw, diamond, aspect=2.2, minimum width=42mm, minimum height=20mm, inner sep=1mm, align=center, font=\normalsize, execute at begin node={\setlength{\parindent}{0pt}}},
  gflowarr/.style={->, >=Stealth, line width=0.6pt}
}
\begin{tikzpicture}[gflowarr, x=1mm, y=1mm]
\parindent=0pt
\node[gflowterm] (start) at (0,0) {开始};
\node[gflowproc, minimum width=62mm] (init) at (0,-20) {$a\gets\sqrt{5},b\gets\sqrt{7},n\gets 1$};
\coordinate (J) at (0,-34);
\node[gflowdec] (dec) at (0,-52) {$n(b-a)>1$};
\node[gflowproc, minimum width=30mm] (m) at (0,-74) {$m\gets[na]$};
\node[gflowproc, minimum width=42mm] (r) at (0,-94) {$r\gets(m+1)/n$};
\node[gflowio] (out) at (0,-114) {输出 $r$};

\node[gflowproc, minimum width=30mm] (inc) at (48,-52) {$n\gets n+1$};
\draw (start.south) -- (init.north);
\draw[->] (init.south) -- (J) -- (dec.north);

\draw[->] (dec.south) -- node[midway, right=1mm] {是} (m.north);

\draw[->] (out.south) -- (stop.north);
