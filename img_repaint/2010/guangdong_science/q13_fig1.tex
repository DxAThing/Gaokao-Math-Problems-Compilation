\tikzset{
  gflowproc/.style={draw, rectangle, minimum width=34mm, minimum height=9mm, inner sep=1.5mm, align=center, font=\normalsize, execute at begin node={\setlength{\parindent}{0pt}}},
  gflowterm/.style={draw, shape=rectangle, rounded corners=1.5mm, minimum width=26mm, minimum height=10mm, inner xsep=3mm, inner ysep=1.5mm, align=center, font=\normalsize, execute at begin node={\setlength{\parindent}{0pt}}},
  gflowio/.style={draw, trapezium, trapezium left angle=72, trapezium right angle=108, minimum width=62mm, minimum height=10mm, inner sep=1.5mm, align=center, font=\normalsize, execute at begin node={\setlength{\parindent}{0pt}}},
  gflowdec/.style={draw, diamond, aspect=2.4, minimum width=36mm, minimum height=18mm, inner sep=1mm, align=center, font=\normalsize, execute at begin node={\setlength{\parindent}{0pt}}},
  gflowarr/.style={->, >=Stealth, line width=0.6pt}
}
\begin{tikzpicture}[gflowarr, x=1mm, y=1mm]
\parindent=0pt
\node[gflowterm] (start) at (0,0) {开始};
\node[gflowio] (input) at (0,-20) {输入 $n,x_{1},x_{2},\cdots,x_{n}$};
\node[gflowproc, minimum width=52mm] (init) at (0,-40) {$s_{1}=0,s_{2}=0,i=1$};
\coordinate (J) at (0,-54);
\node[gflowdec] (dec) at (0,-102) {$i\leqslant n$};
\node[gflowio, minimum width=34mm] (out) at (0,-128) {输出 $s$};
\node[gflowterm] (stop) at (0,-148) {结束};
\node[gflowproc, minimum width=34mm, minimum height=14mm] (acc) at (62,-102) {$s_{1}=s_{1}+x_{i}$\\$s_{2}=s_{2}+x_{i}^{2}$};
\node[gflowproc, minimum width=44mm, minimum height=16mm] (avg) at (62,-78) {$s=\dfrac{1}{i}\left(s_{2}-\dfrac{1}{i}s_{1}^{2}\right)$};
\node[gflowproc, minimum width=26mm] (inc) at (62,-54) {$i=i+1$};
\draw[->] (start.south) -- (input.north);
\draw[->] (input.south) -- (init.north);
% One continuous shaft; the return joins its interior, without an arrow at J.
\draw[->] (init.south) -- (dec.north);
\draw[->] (dec.south) -- node[midway, right=2mm, fill=white, inner sep=1.5pt] {否} (out.north);
\draw[->] (out.south) -- (stop.north);
\draw[->] (dec.east) -- node[pos=0.2, above=2mm, fill=white, inner sep=1.5pt] {是} (acc.west);
\draw[->] (acc.north) -- (avg.south);
\draw[->] (avg.north) -- (inc.south);
% Lower the decision and side chain: increment aligns with the original junction.
\draw[->] (inc.west) -- (J);
\end{tikzpicture}
