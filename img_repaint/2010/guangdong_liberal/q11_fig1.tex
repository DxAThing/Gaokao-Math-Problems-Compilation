\tikzset{
  gflowproc/.style={draw, rectangle, minimum width=34mm, minimum height=9mm, inner sep=1.5mm, align=center, font=\normalsize, execute at begin node={\setlength{\parindent}{0pt}}},
  gflowterm/.style={draw, shape=rectangle, rounded corners=1.5mm, minimum width=26mm, minimum height=10mm, inner xsep=3mm, inner ysep=1.5mm, align=center, font=\normalsize, execute at begin node={\setlength{\parindent}{0pt}}},
  gflowio/.style={draw, trapezium, trapezium left angle=72, trapezium right angle=108, minimum width=52mm, minimum height=10mm, inner sep=1.5mm, align=center, font=\normalsize, execute at begin node={\setlength{\parindent}{0pt}}},
  gflowdec/.style={draw, diamond, aspect=2.4, minimum width=34mm, minimum height=18mm, inner sep=1mm, align=center, font=\normalsize, execute at begin node={\setlength{\parindent}{0pt}}},
  gflowarr/.style={->, >=Stealth, line width=0.6pt}
}
\begin{tikzpicture}[gflowarr, x=1mm, y=1mm]
\parindent=0pt
% main column x=0
\node[gflowterm] (start) at (0,0) {开始};
\node[gflowio] (input) at (0,-20) {输入 $x_{1},\cdots,x_{4}$};
\node[gflowproc, minimum width=38mm] (init) at (0,-40) {$s_{1}=0,i=1$};
\coordinate (J) at (0,-52);
\node[gflowdec] (dec) at (0,-98) {$i\leqslant 4$};
\node[gflowio, minimum width=34mm] (out) at (0,-122) {输出 $s$};
\node[gflowterm] (stop) at (0,-142) {结束};
% right column x=45mm
\node[gflowproc, minimum width=32mm] (acc) at (45,-98) {$s_{1}=s_{1}+x_{i}$};
\node[gflowproc, minimum width=26mm] (avg) at (45,-76) {$s=\dfrac{1}{i}s_{1}$};
\node[gflowproc, minimum width=26mm] (inc) at (45,-52) {$i=i+1$};
% main verticals
\draw[->] (start.south) -- (input.north);
\draw[->] (input.south) -- (init.north);
\draw[->] (init.south) -- (J) -- (dec.north);

\draw[->] (dec.south) -- node[midway, right=2mm, fill=white, inner sep=1.5pt] {否} (out.north);
\draw[->] (out.south) -- (stop.north);
% yes branch: long horizontal with label clearance
\draw[->] (dec.east) -- node[midway, above=2mm, fill=white, inner sep=1.5pt] {是} (acc.west);
% right vertical chain
\draw[->] (acc.north) -- (avg.south);
\draw[->] (avg.north) -- (inc.south);
% Align the return frame with the original junction.
\draw[->] (inc.west) -- (J);
\end{tikzpicture}
