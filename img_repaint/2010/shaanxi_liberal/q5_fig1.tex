% Source PDF: source/普通高考/2010/2010陕西文.pdf, page 1, question 5.
% Elements: rounded start/end terminals, input/output parallelograms, two
% initialization/update rectangles, one intentionally blank process rectangle,
% one decision diamond, and the no-branch loop. All lines are solid; no dashed
% or hidden segments are present.
% Node -> directed-edge list: start -> input x_1,...,x_10 -> (n=1,S=1)
% -> blank process -> (n>=10); yes -> output S -> finish; no -> (n=n+1)
% -> shared entry immediately before the blank process. The blank rectangle is
% preserved as blank because the source asks for its missing expression.
\begin{tikzpicture}[
  >=Stealth,
  x=1cm,y=1cm,
  line width=0.45pt,
  line cap=round,line join=round,
  font=\normalsize,
  flow/.style={draw,anchor=center,align=center,inner sep=0pt,
    execute at begin node={\setlength{\parindent}{0pt}}},
  every node/.style={anchor=center,align=center,
    execute at begin node={\setlength{\parindent}{0pt}}},
  terminal/.style={flow,rounded corners=.18cm,minimum width=1.40cm,minimum height=.74cm},
  process/.style={flow,minimum height=.76cm},
  io/.style={flow,shape=trapezium,trapezium left angle=72,trapezium right angle=108,
    minimum width=2.25cm,minimum height=.80cm},
  decision/.style={flow,diamond,aspect=2.8,minimum width=3.20cm,minimum height=1.05cm,inner sep=0pt},
  label/.style={inner sep=0pt,anchor=center,align=center,
    execute at begin node={\setlength{\parindent}{0pt}}},
]
  % Safety margin keeps both side branches and the rounded finish in the bbox.
  \path[use as bounding box] (-4.50,-8.35) rectangle (4.00,0.42);

  \node[terminal] (start) at (0,0) {开始};
  \node[io,minimum width=3.30cm] (input) at (0,-1.20) {输入 $x_1,x_2,\cdots,x_{10}$};
  \node[process,minimum width=2.90cm] (init) at (0,-2.48) {$n=1,S=1$};
  \node[process,minimum width=2.45cm] (blank) at (0,-3.88) {};
  \node[process,minimum width=2.35cm] (increment) at (-2.90,-3.88) {$n=n+1$};
  \node[decision] (test) at (0,-5.12) {$n\geq 10$};
  \node[io,minimum width=2.35cm] (output) at (0,-6.45) {输出 $S$};
  \node[terminal] (finish) at (0,-7.75) {结束};

  \draw[->] (start.south) -- (input.north);
  \draw[->] (input.south) -- (init.north);
  % Keep the shared stem 2.5 mm clear of the initialization box.
  \coordinate (merge-level) at (0,-3.18);
  \draw (init.south) -- (merge-level);
  \draw[->] (merge-level) -- (blank.north);
  \draw[->] (blank.south) -- (test.north);
  \draw[->] (test.south) -- node[right=2pt] {是} (output.north);
  \draw[->] (output.south) -- (finish.north);

  % False branch goes left to n=n+1, then returns right into the shared entry.
  \coordinate (no-left) at (-2.90,-5.12);
  \coordinate (loop-top) at (-2.90,-3.18);
  \draw (test.west) -- (no-left);
  \draw[->] (no-left) -- (increment.south);
  \draw (increment.north) -- (loop-top);
  \draw[->] (loop-top) -- (merge-level);
  \node[label,anchor=east] at (-1.20,-4.82) {否};
\end{tikzpicture}
