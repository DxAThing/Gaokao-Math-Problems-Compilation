% Five-pyramid P-ABCDE schematic (2010 Shandong science Q19).
% PA vertical dashed; E on dashed AD; solid PB,PC,BC,CD.
\begin{tikzpicture}[x=1cm,y=1cm,line width=.6pt,font=\normalsize,>=Stealth]
  \coordinate (P) at (0,5.2);
  \coordinate (A) at (0,1.2);
  \coordinate (B) at (-3.6,-1.2);
  \coordinate (C) at (1,-1.2);
  \coordinate (D) at (4.6,-1.2);
  \coordinate (E) at (2.3,1.2);
  % solid visible edges
  \draw[solid] (P) -- (B);
  \draw[solid] (P) -- (C);
  \draw[solid] (B) -- (C);
  \draw[solid] (C) -- (D);
  % dashed hidden edges
  \draw[dashed] (P) -- (A);
  \draw[solid] (P) -- (D);
\draw[dashed] (P)--(E);
  \draw[dashed] (A) -- (B);
  \draw[dashed] (A) -- (C);
  \draw[dashed] (A) -- (E);
  \draw[dashed] (E) -- (D);
  % labels offset from strokes
  \node[above=2pt] at (P) {$P$};
  \node[left=3pt] at (A) {$A$};
  \node[below left=2pt] at (B) {$B$};
  \node[below right=2pt] at (C) {$C$};
  \node[right=3pt] at (D) {$D$};
  \node[above=2pt,xshift=0.00bp,yshift=2.00bp] at (E) {$E$};
\end{tikzpicture}
