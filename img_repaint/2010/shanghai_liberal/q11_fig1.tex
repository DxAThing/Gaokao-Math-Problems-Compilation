\begin{tikzpicture}[

  io/.style={flowbox,shape=trapezium,trapezium left angle=72,

  \node[terminal] (start) at (0,11.20) {开始};
  \node[flowbox,minimum width=4.35cm] (init) at (0,9.72)

  \draw[flowarrow] (start.south) -- (init.north);
  \draw[flowarrow] (init.south) -- (report.north);
  \draw (report.south) -- (loopright);
  \draw[flowarrow] (loopright) -- (test.north);
  \draw[flowarrow] (test.south) -- node[right=3pt] {是} (inc.north);
  \draw[flowarrow] (inc.south) -- (input.north);
  \draw[flowarrow] (input.south) -- (update.north);

  \draw (update.west) -- (-3.38,1.18) -- (loopjoin);
