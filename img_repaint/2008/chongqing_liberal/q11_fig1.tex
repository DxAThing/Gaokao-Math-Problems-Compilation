  \fill[white] (\bx,\ey) -- (\fx,\fy) -- (\gx,\gy) -- (\ex,\ey) -- cycle;
  \fill[gray!15] (\ex,\by) -- (\hx,\hy) -- (\gx,\gy) -- (\ex,\ey) -- cycle;
  \draw (\bx,\by) rectangle (\ex,\ey);
  \draw (\bx,\ey) -- (\fx,\fy) -- (\gx,\gy) -- (\ex,\ey);
  \draw (\ex,\by) -- (\hx,\hy) -- (\gx,\gy);
}
