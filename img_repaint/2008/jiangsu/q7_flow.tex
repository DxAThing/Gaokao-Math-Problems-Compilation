\begin{tikzpicture}[x=1cm,y=1cm,>=Stealth,line width=0.6pt,
  start/.style={draw, shape=rectangle, rounded corners=1.5mm, minimum width=26mm, minimum height=10mm, inner xsep=3mm, inner ysep=1.5mm, align=center, execute at begin node={\setlength{\parindent}{0pt}}},
  proc/.style={draw, rectangle, minimum width=2.6cm, minimum height=0.9cm, align=center, inner sep=4pt, execute at begin node={\setlength{\parindent}{0pt}}},
  wide/.style={draw, rectangle, minimum width=5.4cm, minimum height=0.95cm, align=center, inner sep=4pt, execute at begin node={\setlength{\parindent}{0pt}}},
  loopbox/.style={draw, rectangle, minimum width=2.8cm, minimum height=0.9cm, align=center, inner sep=4pt, execute at begin node={\setlength{\parindent}{0pt}}},
  io/.style={draw, trapezium, trapezium left angle=75, trapezium right angle=105, minimum width=4.2cm, minimum height=0.95cm, align=center, inner sep=4pt, execute at begin node={\setlength{\parindent}{0pt}}},
  dec/.style={draw, diamond, aspect=2.2, minimum width=4.0cm, minimum height=1.5cm, align=center, inner sep=2pt, execute at begin node={\setlength{\parindent}{0pt}}},
  lbl/.style={font=\normalsize, inner sep=2pt}]
\node[start] (start) at (0,0) {开始};
\node[proc] (pS0) at (0,-1.8) {$S\leftarrow 0$};
\node[proc] (pi1) at (0,-3.3) {$i\leftarrow 1$};
\coordinate (J1) at (0,-4.4);
\node[io] (ioIn) at (0,-5.5) {输入 $G_{i},F_{i}$};
\node[wide] (pS) at (0,-7.1) {$S\leftarrow S+G_{i}\times F_{i}$};
\node[dec] (dec) at (0,-9.0) {$i\geqslant 5$};
\node[loopbox] (pLoop) at (-4.6,-7.1) {$i\leftarrow i+1$};
\node[io] (ioOut) at (0,-11.3) {输出 $S$};
\node[start] (end) at (0,-13.1) {结束};
% trunk (continuous through loop junction J1)
\draw[->] (start.south) -- (pS0.north);
\draw[->] (pS0.south) -- (pi1.north);
\draw[->] (pi1.south) -- (ioIn.north);
\draw[->] (ioIn.south) -- (pS.north);
\draw[->] (pS.south) -- (dec.north);
\draw[->] (dec.south) -- (ioOut.north);
\node[lbl,right=2.2mm] at ($(dec.south)!0.4!(ioOut.north)$) {Y};
\draw[->] (ioOut.south) -- (end.north);
% N branch: diamond west to loop box west via outer left channel
\draw[->] (dec.west) -| (pLoop.south);
\node[lbl,above=2.2mm,xshift=0.0005mm,yshift=-0.0516mm] at ($(dec.west)+(-5mm,0)$) {N};
% loop return: loop box north up then right into trunk junction J1
\draw[->] (pLoop.north) -- (-4.6,-4.4) -- (J1);
\end{tikzpicture}
