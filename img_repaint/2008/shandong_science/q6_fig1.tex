% 三视图: 上圆下矩(正/侧) + 圆形俯视图. 实线为主, 无虚线.
% 正(主)视图: 矩形宽2高3, 上方圆直径2与矩形顶边相切; 右侧竖向尺寸2(圆径)/3(矩高), 下方横向尺寸2(矩宽).
% 侧(左)视图: 与正视图同形同尺寸, 无尺寸标注.
% 俯视图: 圆直径2, 与正视矩宽对正.
% 点修: 仅平移俯视图至右侧横向并排, 投影边线/尺寸/虚实未改; 图名基线对齐.
\begin{tikzpicture}[line width=.6pt,font=\normalsize,x=1cm,y=1cm,>=Stealth]
% 正(主)视图
\draw (0,4) rectangle (2,7);
\draw (1,8) circle (1);
% 正视图尺寸: 圆径 2, 矩高 3, 矩宽 2
\draw[<->] (2.35,7) -- (2.35,9);
\node[right] at (2.45,8) {2};
\draw[<->] (2.35,4) -- (2.35,7);
\node[right] at (2.45,5.5) {3};
\draw[<->] (0,3.6) -- (2,3.6);
\node[below] at (1,3.6) {2};
\node at (1,2.5) {正视图};
% 侧(左)视图(与正视图高平齐、宽相等)
\draw (4,4) rectangle (6,7);
\draw (5,8) circle (1);
\node at (5,2.5) {侧视图（左）};
% 俯视图(直径2, 平移至右侧与正侧横向并排, 中心高度居中)
\draw (9,5.5) circle (1);
\node at (9,2.5) {俯视图};
\end{tikzpicture}
