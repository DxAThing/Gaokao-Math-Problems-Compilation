% Node-edge list: 开始 -> 输入 p -> n=1,S=0 -> S<p?;
% 是 -> S=S+1/(2^n) -> n=n+1 -> loop-merge -> n=1,S=0;
% 否 -> 输出 n -> 结束. All borders/connectors are solid.
% Every node clears paragraph/list state and is centered; no artificial spacing,
% shift, or coordinate labels are used.

\begin{tikzpicture}[
    >=Stealth,
    x=1cm,
    y=0.88cm,
    line width=0.45pt,
    line cap=round,
    line join=round,
    font=\normalsize,
    flowtext/.style={anchor=center,align=center,
        execute at begin node={\everypar{}\leftskip=0pt\rightskip=0pt\setlength{\parindent}{0pt}}},
    flowbox/.style={draw,flowtext,minimum height=0.68cm,
        inner xsep=5pt,inner ysep=3pt},
    terminal/.style={flowbox,rounded corners=0.16cm,minimum width=1.35cm},
    process/.style={flowbox,minimum width=2.55cm},
    decision/.style={flowbox,diamond,aspect=2.9,minimum width=3.75cm,
        minimum height=1.05cm},
    io/.style={flowbox,trapezium,trapezium left angle=72,
        trapezium right angle=108,minimum height=0.78cm},
]
    \path[use as bounding box] (-3.85,0.15) rectangle (4.50,10.20);

    \node[terminal] (start) at (0,9.75) {开始};
    \node[io,minimum width=1.65cm] (input) at (0,8.45) {输入 $p$};
    \node[process,minimum width=3.20cm] (init) at (0,7.10) {$n=1,S=0$};
    \node[decision] (test) at (0,5.45) {$S<p?$};
    \node[process,minimum width=3.05cm,minimum height=1.00cm] (update)
        at (0,3.90) {$S=S+\dfrac{1}{2^n}$};
    \node[process,minimum width=2.35cm] (increment) at (0,2.40)
        {$n=n+1$};
    \node[io,minimum width=2.55cm] (output) at (3.05,4.35) {输出 $n$};
    \node[terminal] (finish) at (3.05,2.90) {结束};

    \draw[->] (start.south) -- (input.north);
    \draw[->] (input.south) -- (init.north);
    \draw[->] (init.south) -- (test.north);
    \draw[->] (test.south) -- node[right] {是} (update.north);
    \draw[->] (update.south) -- (increment.north);

    % Negative branch exits to output and does not share the loop return.
    \draw[->] (test.east) -- node[pos=0.18,above right] {否} (3.05,5.45)
        -- (3.05,5.15) -- (output.north);
    \draw[->] (output.south) -- (finish.north);

    % Loop return enters the main stem immediately above n=1,S=0.
    % Keep the return segment at least 2.5 mm above the init box.
    \coordinate (loop-merge) at (0,7.85);
    \draw (increment.south) -- (0,1.75) -- (-3.20,1.75)
        -- (-3.20,7.85);
    % Return path merges into the main stem; the arrowhead marks the
    % leftward flow before the shared stem enters the initialization node.
    \draw[->] (-3.20,7.85) -- (loop-merge);
    \draw (loop-merge) -- (init.north);
\end{tikzpicture}
