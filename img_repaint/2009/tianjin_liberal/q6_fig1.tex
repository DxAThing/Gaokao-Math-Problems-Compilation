% Source PDF: source/普通高考/2009/2009天津文.pdf, page 1 (question 6).
% Node-edge list: 开始 -> S=0,i=1 -> S=S+i^2 -> i=i+1 -> i>4?;
% 否 loops back to the connector above S=S+i^2, while 是 -> 输出 S -> 结束.
% All borders and connectors are solid and directed; branch labels are beside exits.
\begin{tikzpicture}[
    >=Stealth,
    x=1cm,
    y=0.86cm,
    line width=0.45pt,
    line cap=round,
    line join=round,
    flowtext/.style={anchor=center,align=center,
        execute at begin node={\setlength{\parindent}{0pt}}},
    every node/.style={flowtext,font=\normalsize},
    flowbox/.style={draw,flowtext,minimum height=0.68cm,inner xsep=4pt,inner ysep=2pt},
    terminal/.style={flowbox,rounded corners=0.18cm,minimum width=1.30cm},
    process/.style={flowbox,minimum width=2.85cm},
    decision/.style={draw,diamond,flowtext,aspect=2.8,minimum width=3.45cm,
        minimum height=1.00cm,inner sep=0pt},
    io/.style={draw,trapezium,flowtext,trapezium left angle=72,
        trapezium right angle=108,minimum width=2.15cm,minimum height=0.74cm,
        inner xsep=4pt,inner ysep=2pt},
    branchlabel/.style={flowtext,inner sep=0pt},
]
    \path[use as bounding box] (-3.55,-0.55) rectangle (3.55,10.95);

    \node[terminal] (start) at (0,10.45) {开始};
    \node[process,minimum width=3.05cm] (init) at (0,9.10) {$S=0,\ i=1$};
    \node[process,minimum width=3.25cm] (update) at (0,7.65) {$S=S+i^2$};
    \node[process,minimum width=2.40cm] (inc) at (0,6.20) {$i=i+1$};
    \node[decision] (check) at (0,4.72) {$i>4?$};
    \node[io,minimum width=2.10cm] (output) at (0,3.05) {输出 $S$};
    \node[terminal] (finish) at (0,1.70) {结束};

    \draw[->] (start.south) -- (init.north);
    \draw[->] (init.south) -- (update.north);
    \draw[->] (update.south) -- (inc.north);
    \draw[->] (inc.south) -- (check.north);
    \draw[->] (check.south) -- node[branchlabel,right=2pt] {是} (output.north);
    \draw[->] (output.south) -- (finish.north);

    % The negative branch returns to the single stem immediately above S=S+i^2.
    \draw (check.east) -- node[branchlabel,above=2pt] {否} (3.05,4.72)
        -- (3.05,8.35);
    \draw[->] (3.05,8.35) -- (0,8.35);
\end{tikzpicture}
