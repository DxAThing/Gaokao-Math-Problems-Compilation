% Source PDF: source/普通高考/2009/2009天津理.pdf, page 1 (question 5).
% Node-edge list: 开始 -> S=0,i=1 -> T=3i-1 -> S=S+T -> i=i+1 -> i>5?;
% 否 loops back to the connector above T=3i-1, while 是 -> 输出 S -> 结束.
% All borders and connectors are solid and directed; branch labels are beside exits.
\begin{tikzpicture}[
    >=Stealth,
    x=1cm,
    y=0.86cm,
    line width=0.45pt,
    line cap=round,
    line join=round,
    flowtext/.style={anchor=center,align=center,
        execute at begin node={\setlength{\parindent}{0pt}}},
    every node/.style={flowtext,font=\normalsize},
    flowbox/.style={draw,flowtext,minimum height=0.68cm,inner xsep=4pt,inner ysep=2pt},
    terminal/.style={flowbox,rounded corners=0.18cm,minimum width=1.30cm},
    process/.style={flowbox,minimum width=2.85cm},
    decision/.style={draw,diamond,flowtext,aspect=2.8,minimum width=3.55cm,
        minimum height=1.00cm,inner sep=0pt},
    io/.style={draw,trapezium,flowtext,trapezium left angle=72,
        trapezium right angle=108,minimum width=2.20cm,minimum height=0.74cm,
        inner xsep=4pt,inner ysep=2pt},
    branchlabel/.style={flowtext,inner sep=0pt},
]
    \path[use as bounding box] (-3.65,-0.55) rectangle (3.65,12.15);

    \node[terminal] (start) at (0,11.65) {开始};
    \node[process,minimum width=3.05cm] (init) at (0,10.30) {$S=0,\ i=1$};
    \node[process,minimum width=2.55cm] (term) at (0,8.85) {$T=3i-1$};
    \node[process,minimum width=3.05cm] (update) at (0,7.40) {$S=S+T$};
    \node[process,minimum width=2.40cm] (inc) at (0,5.95) {$i=i+1$};
    \node[decision] (check) at (0,4.47) {$i>5?$};
    \node[io,minimum width=2.20cm] (output) at (0,2.80) {输出 $S$};
    \node[terminal] (finish) at (0,1.45) {结束};

    \draw[->] (start.south) -- (init.north);
    \draw[->] (init.south) -- (term.north);
    \draw[->] (term.south) -- (update.north);
    \draw[->] (update.south) -- (inc.north);
    \draw[->] (inc.south) -- (check.north);
    \draw[->] (check.south) -- node[branchlabel,right=2pt] {是} (output.north);
    \draw[->] (output.south) -- (finish.north);

    % The negative branch returns to the single stem immediately above T=3i-1.
    \draw (check.east) -- node[branchlabel,above=2pt] {否} (3.15,4.47)
        -- (3.15,9.55);
    \draw[->] (3.15,9.55) -- (0,9.55);
\end{tikzpicture}
