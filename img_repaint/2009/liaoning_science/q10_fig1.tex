\begin{tikzpicture}[
  >=Stealth,
  x=1cm,y=0.82cm,
  line width=0.45pt,line cap=round,line join=round,
  font=\normalsize,
  flowtext/.style={anchor=center,align=center,inner sep=0pt,
    execute at begin node={\everypar{}\setlength{\parindent}{0pt}\noindent}},
  every node/.style={flowtext},
  flowbox/.style={flowtext,draw,minimum height=0.70cm,inner xsep=4pt,inner ysep=2pt},
  terminal/.style={flowbox,rounded corners=0.18cm,minimum width=1.40cm,minimum height=0.78cm},
  io/.style={flowbox,shape=trapezium,trapezium left angle=72,trapezium right angle=108,
    minimum height=0.84cm},

]
  \path[use as bounding box] (-4.60,0.10) rectangle (5.20,13.00);
  \node[terminal] (start) at (0,12.55) {开始};
  \node[io,minimum width=5.80cm] (input) at (0,11.25)
    {输入 $N$，$a_{1}$，$a_{2}$，$\cdots$，$a_{N}$};
  \node[flowbox,minimum width=4.70cm] (init) at (0,9.85)
    {$k=1$，$S=0$，$T=0$};
  \node[flowbox,minimum width=1.95cm] (assign) at (0,8.40) {$A=a_{k}$};
  \node[decision] (testA) at (0,6.90) {};

  \node[flowbox,minimum width=2.70cm] (addS) at (2.85,5.35) {$S=S+A$};
  \node[decision] (testK) at (0,3.95) {$k<N$};
  \node[flowbox,minimum width=2.80cm] (blank) at (0,2.50) {};
  \node[io,minimum width=3.20cm] (output) at (0,1.18) {输出 $S$，$V$};
  \node[terminal] (finish) at (0,0.02) {结束};
  \node[flowbox,minimum width=2.85cm] (inc) at (4.00,7.05) {$k=k+1$};

  \draw[->] (start.south) -- (input.north);
  \draw[->] (input.south) -- (init.north);
  \draw[->] (init.south) -- (assign.north);
  \draw[->] (assign.south) -- (testA.north);
  \draw[->] (testA.west) -- node[branchlabel,above=2pt] {否} (addT.north);

  \draw (addT.south) -- (-2.85,4.65) -- (0,4.65);
  \draw (addS.south) -- (2.85,4.65) -- (0,4.65);
  \draw[->] (0,4.65) -- (testK.north);
  \draw[->] (testK.south) -- node[branchlabel,right=3pt] {否} (blank.north);
  \draw[->] (blank.south) -- (output.north);
  \draw[->] (output.south) -- (finish.north);
  \draw (testK.east) -- node[branchlabel,above=2pt] {是} (4.00,3.95);
  \draw[->] (4.00,3.95) -- (4.00,6.65) -- (inc.south);
  \draw[->] (inc.north) -- (4.00,9.10) -- (0,9.10);
