% Topology: 开始→输入→k=1,S=0,T=0→A=a_k→空白判断；否→T=T+A；是→S=S+A；两支汇合→k<N?；是→k=k+1→回到 A=a_k；否→空白处理→输出S,V→结束。
\begin{tikzpicture}[
  >=Stealth,
  x=1cm,y=0.82cm,
  line width=0.45pt,line cap=round,line join=round,
  font=\normalsize,
  flowtext/.style={anchor=center,align=center,inner sep=0pt,
    execute at begin node={\everypar{}\setlength{\parindent}{0pt}\noindent}},
  every node/.style={flowtext},
  flowbox/.style={flowtext,draw,minimum height=0.70cm,inner xsep=4pt,inner ysep=2pt},
  terminal/.style={flowbox,rounded corners=0.18cm,minimum width=1.35cm,minimum height=0.76cm},
  io/.style={flowbox,shape=trapezium,trapezium left angle=72,trapezium right angle=108,
    minimum height=0.80cm},
  decision/.style={flowbox,diamond,aspect=2.9,minimum width=3.90cm,minimum height=1.00cm,inner sep=1pt},
  branchlabel/.style={flowtext,inner sep=0pt},
]
  \path[use as bounding box] (-4.25,-0.65) rectangle (5.75,13.10);
  \node[terminal] (start) at (0,12.55) {开始};
  \node[io,minimum width=5.15cm] (input) at (0,11.20)
    {输入 $N$，$a_{1}$，$a_{2}$，$\cdots$，$a_{N}$};
  \node[flowbox,minimum width=4.20cm] (init) at (0,9.90)
    {$k=1$，$S=0$，$T=0$};
  \node[flowbox,minimum width=1.80cm] (assign) at (0,8.20) {$A=a_{k}$};
  \node[decision] (testA) at (0,6.80) {};
  \node[flowbox,minimum width=2.35cm] (addT) at (-2.55,5.45) {$T=T+A$};
  \node[flowbox,minimum width=2.35cm] (addS) at (2.55,5.45) {$S=S+A$};
  \node[decision] (testK) at (0,3.85) {$k<N$};
  \node[flowbox,minimum width=2.25cm] (blank) at (0,2.45) {};
  \node[io,minimum width=2.70cm] (output) at (0,1.15) {输出 $S$，$V$};
  \node[terminal] (finish) at (0,-0.20) {结束};
  \node[flowbox,minimum width=2.45cm] (inc) at (4.35,7.10) {$k=k+1$};
  \coordinate (assignin) at (0,8.95);

  \draw[->] (start.south) -- (input.north);
  \draw[->] (input.south) -- (init.north);
  \draw (init.south) -- (assignin);
  \draw[->] (assign.south) -- (testA.north);
  \draw[->] (testA.west) -- node[pos=0.25,branchlabel,above=4pt] {否}
    (-2.55,6.80) -- (addT.north);
  \draw[->] (testA.east) -- node[pos=0.25,branchlabel,above=5pt] {是}
    (2.55,6.80) -- (addS.north);
  \draw (addT.south) -- (-2.55,4.80) -- (0,4.80);
  \draw (addS.south) -- (2.55,4.80) -- (0,4.80);
  \draw[->] (0,4.80) -- (testK.north);
  \draw[->] (testK.south) -- node[branchlabel,right=3pt] {否} (blank.north);
  \draw[->] (blank.south) -- (output.north);
  \draw[->] (output.south) -- (finish.north);
  \draw (testK.east) -- node[branchlabel,above=5pt] {是} (4.35,3.85);
  \draw[->] (4.35,3.85) -- (4.35,6.75);
  \draw (inc.north) -- (4.35,9.10) -- (0,9.10) -- (assignin);
  \draw[->] (assignin) -- (assign.north);
\end{tikzpicture}
