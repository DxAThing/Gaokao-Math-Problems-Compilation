% Triangular pyramid on O-xyz; base right angle at O, OT vertical.
\begin{tikzpicture}[x=1cm,y=1cm,line width=.6pt,font=\normalsize,>=Stealth]
  \coordinate (O) at (0,0);
  \coordinate (X) at (-1.8,0);
  \coordinate (Y) at (-0.9,-1.5);
  \coordinate (T) at (0,3.6);
  % axes (OX segment hidden dashed; OY/OT coincide with solid edges)
  \draw[dashed] (O) -- (X);
  \draw[->] (X) -- (-3.4,0) node[below, yshift=-2pt] {$x$};
  \draw[->] (O) -- (-1.2,-2.0) node[below, yshift=-2pt] {$y$};
  \draw[->] (O) -- (0,4.6) node[right, xshift=3pt] {$z$};
  % tetrahedron solid edges
  \draw (O) -- (Y);
  \draw (O) -- (T);
  \draw (X) -- (Y);
  \draw (X) -- (T);
  \draw (Y) -- (T);
  % labels with clearance
  \node[right, xshift=3pt, yshift=-2pt] at (O) {$O$};
  \node at (-1.05,0.55) {$3$};
  \node[right, xshift=4pt] at (-0.45,-0.75) {$4$};
  \node[right, xshift=4pt] at (0,1.8) {$4$};
\end{tikzpicture}
