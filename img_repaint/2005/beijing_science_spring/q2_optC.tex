% 2005 春季北京理 q2 选项 C：错误选项，图象为 y=log_2 x 的右支（未取绝对值，x<1 时落在 x 轴下方）。
\begin{tikzpicture}[x=1.2cm,y=1.2cm,line width=.6pt,>={Stealth[length=1.5mm,width=1.3mm]},font=\normalsize]
  \draw[->] (-3.0,0) -- (3.0,0) node[below,inner sep=0pt,xshift=3pt,yshift=-1pt] {$x$};
  \draw[->] (0,-3.0) -- (0,3.0) node[above left,inner sep=1pt] {$y$};
  \draw[domain=0.2:2.5,samples=200] plot (\x,{ln(\x)/ln(2)});
  \node[below left,inner sep=0pt,xshift=-2pt,yshift=-1pt] at (0,0) {$O$};
  \node[below,inner sep=0pt,xshift=5pt,yshift=-2pt] at (1,0) {$1$};
\end{tikzpicture}
