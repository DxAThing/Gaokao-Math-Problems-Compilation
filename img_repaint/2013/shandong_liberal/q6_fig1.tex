\begin{tikzpicture}[x=1mm,y=1mm,line width=0.5pt,>=Stealth,
  every node/.append style={execute at begin node={\setlength{\parindent}{0pt}}},
  term/.style={draw,shape=rectangle,rounded corners=1.5mm,align=center,inner xsep=3mm,inner ysep=1.5mm,minimum width=26mm,minimum height=10mm},

  \node[io] (inA) at (0,-21) {输入 $a$};

  \draw[->] (pR.west) -- (J1);
  \draw[->] (d1.south) -- node[midway,right=2mm,lab] {否} (d2.north);
  \draw[->] (d2.west) -- node[pos=0.18,above=2mm,lab] {是} (-44,-75) -- (pL.south);
  \draw[->] (pL.east) -- (J2);
  \draw[->] (d2.south) -- node[midway,right=2mm,lab] {否} (outA.north);
  \draw[->] (outA.south) -- (end.north);
