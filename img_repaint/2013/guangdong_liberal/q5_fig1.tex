% Node-edge list: 开始 -> 输入 n -> i=1,s=1 -> i<=n;
% 是 -> s=s+(i-1) -> i=i+1 -> loop-merge -> i<=n;
% 否 -> 输出 s -> 结束. All borders/connectors are solid.
% Every node is locally zero-indented and centered; no manual spacing or shifts.

\begin{tikzpicture}[
    >=Stealth,
    x=1cm,
    y=0.90cm,
    line width=0.45pt,
    line cap=round,
    line join=round,
    font=\normalsize,
    flowtext/.style={anchor=center, align=center,
    execute at begin node={\setlength{\parindent}{0pt}}},
    flowbox/.style={draw, flowtext, minimum height=0.68cm,
        inner xsep=4pt, inner ysep=2pt},
    terminal/.style={flowbox, rounded corners=0.16cm, minimum width=1.35cm},
    process/.style={flowbox, minimum width=2.00cm},
    io/.style={flowbox, trapezium, trapezium left angle=72,
        trapezium right angle=108, minimum height=0.78cm},
    decision/.style={flowbox, diamond, aspect=3.0, minimum width=4.15cm,
        minimum height=1.05cm},
]
    \path[use as bounding box] (-4.55,-0.50) rectangle (4.35,10.30);

    \node[terminal] (start) at (0,9.95) {开始};
    \node[io, minimum width=3.40cm] (input) at (0,8.65) {输入 $n$};
    \node[process, minimum width=3.15cm] (init) at (0,7.25) {$i=1,s=1$};
    \node[decision] (test) at (0,5.55) {$i\leqslant n$};
    \node[process, minimum width=3.45cm, minimum height=1.05cm] (update)
        at (0,3.95) {$s=s+(i-1)$};
    \node[process, minimum width=2.55cm] (increment) at (0,2.50)
        {$i=i+1$};

    \node[io, minimum width=2.45cm] (output) at (3.25,4.75) {输出 $s$};
    \node[terminal] (finish) at (3.25,3.25) {结束};

    \draw[->] (start.south) -- (input.north);
    \draw[->] (input.south) -- (init.north);

    % Initialization and loop return share one connector above the test.
    \coordinate (loop-merge) at (0,6.55);
    \draw (init.south) -- (loop-merge);
    \draw[->] (loop-merge) -- (test.north);

    \draw[->] (test.south) -- node[left] {是} (update.north);
    \draw[->] (update.south) -- (increment.north);

    % Negative branch is independent from the loop and enters output directly.
    \draw[->] (test.east) -- node[above] {否} (3.25,5.55)
        -- (3.25,5.20) -- (output.north);
    \draw[->] (output.south) -- (finish.north);

    % Loop return: left and up, then a single arrow into the shared stem.
    \draw (increment.south) -- (0,1.95) -- (-3.35,1.95)
        -- (-3.35,6.55);
    \draw[->] (-3.35,6.55) -- (loop-merge);
\end{tikzpicture}
