\begin{tikzpicture}[x=1mm,y=1mm,line width=0.5pt,>=Stealth,
  every node/.append style={execute at begin node={\setlength{\parindent}{0pt}}},
  term/.style={draw,shape=rectangle,rounded corners=1.5mm,align=center,inner xsep=3mm,inner ysep=1.5mm,minimum width=26mm,minimum height=10mm},
  io/.style={draw,trapezium,trapezium left angle=75,trapezium right angle=105,align=center,inner sep=1.8mm,minimum width=28mm,minimum height=8mm},
  proc/.style={draw,rectangle,align=center,inner sep=1.8mm,minimum width=24mm,minimum height=8mm},
  dec/.style={draw,diamond,aspect=2,align=center,inner sep=1.8mm},
  lab/.style={inner sep=0.6mm}]
  \node[term] (start) at (0,0) {开始};
  \node[io] (inX) at (0,-21) {输入 $x$};
  \node[dec,minimum width=52mm] (d1) at (0,-46) {$x$ 为偶数?};
  \node[proc] (y1) at (52,-46) {$y=1$};
  \node[dec,minimum width=62mm] (d2) at (0,-82) {$x$ 能被 $3$ 整除?};
  \node[proc] (y2) at (-52,-82) {$y=2$};
  \node[proc,minimum width=22mm] (y3) at (0,-110) {$y=3$};
  \coordinate (JM) at (0,-121);
  \node[io] (outY) at (0,-132) {输出 $y$};
  \node[term] (end) at (0,-153) {结束};
  \draw[->] (start.south) -- (inX.north);
  \draw[->] (inX.south) -- (d1.north);
  \draw[->] (d1.east) -- node[midway,above=2mm,lab] {否} (y1.west);
  \draw[->] (d1.south) -- node[midway,right=2mm,lab] {是} (d2.north);
  \draw[->] (d2.west) -- node[midway,above=2mm,lab] {否} (y2.east);
  \draw[->] (d2.south) -- node[midway,right=2mm,lab] {是} (y3.north);
  \draw[->] (y3.south) -- (outY.north);
  \draw[-] (y1.south) -- (52,-121) -- (JM);
  \draw[-] (y2.south) -- (-52,-121) -- (JM);
  \draw[->] (outY.south) -- (end.north);
\end{tikzpicture}
