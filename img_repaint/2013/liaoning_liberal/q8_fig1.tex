% Source: source/普通高考/2013/2013辽宁文.pdf, page 1, question 8.
% The source PDF contains vector drawing commands (pdfimages -list: no embedded images).
% Directed edges: start -> input -> initialization -> test;
% test --是--> update -> increment -> test (left return);
% test --否--> output -> finish. No update-to-output edge is present.
\begin{tikzpicture}[
  >=Stealth,
  flow/.style={draw, line width=0.55pt, -{Stealth[length=2.1pt,width=1.8pt]}},
  every node/.style={align=center, inner sep=0pt,
    execute at begin node={\setlength{\parindent}{0pt}}},
  box/.style={draw, rectangle, minimum height=0.65cm,
    inner xsep=0.15cm, inner ysep=0.08cm, align=center,
    execute at begin node={\setlength{\parindent}{0pt}}},
  terminal/.style={box, rounded corners=0.18cm, minimum width=1.15cm},
  io/.style={box, trapezium, trapezium left angle=72, trapezium right angle=108,
    minimum width=1.85cm},
  decision/.style={draw, diamond, aspect=2.8, minimum width=2.55cm,
    minimum height=0.95cm, inner xsep=0.10cm, inner ysep=0.06cm,
    align=center, execute at begin node={\setlength{\parindent}{0pt}}},
]
  \path[use as bounding box] (-2.95,0) rectangle (4.55,10.35);

  \node[terminal] (start) at (0,9.85) {开始};
  \node[io, minimum width=1.80cm] (input) at (0,8.45) {输入 $n$};
  \node[box, minimum width=3.25cm] (init) at (0,7.00) {$S=0,i=2$};
  \node[decision] (test) at (0,5.50) {$i\leq n$};
  \node[box, minimum width=4.55cm, minimum height=1.05cm] (update) at (0,3.55)
    {$S=S+\dfrac{1}{i^2-1}$};
  \node[box, minimum width=2.55cm] (increment) at (0,1.65) {$i=i+2$};
  \node[io, minimum width=1.95cm] (output) at (3.60,4.25) {输出 $S$};
  \node[terminal] (finish) at (3.60,2.65) {结束};
  \coordinate (loopjoin) at (0,5.98);

  \draw[flow] (start) -- (input);
  \draw[flow] (input) -- (init);
  \draw[-] (init) -- (loopjoin);
  \draw[flow] (loopjoin) -- (test.north);
  \draw[flow] (test.south) -- node[right, inner sep=1pt] {是} (update.north);
  \draw[flow] (update) -- (increment);
  \draw[flow] (test.east) -- node[above, inner sep=1pt] {否} (3.60,5.50)
    -- (3.60,4.68) -- (output.north);
  \draw[flow] (output) -- (finish);
  \draw[flow] (increment.south) -- (0,0.55) -- (-2.45,0.55)
    -- (-2.45,5.98) -- (loopjoin);
\end{tikzpicture}
