% Source: source/普通高考/2007/2007新课标文(海南,宁夏).pdf, page 1, question 5.
% Complete node -> directed-edge list:
%   开始 -> k=1 -> S=0 -> k<=50?;
%   是 -> S=S+2k -> k=k+1 -> left return into the decision;
%   否 -> 输出 S -> 结束.
% The return edge is independent of the output branch and enters the west side
% of the decision exactly once, as in the source PDF.

\begin{tikzpicture}[
  >=Stealth,
  x=1cm,
  y=0.82cm,
  line width=0.45pt,
  line cap=round,
  line join=round,
  font=\normalsize,
  flowbox/.style={draw,anchor=center,align=center,
    execute at begin node={\setlength{\parindent}{0pt}\noindent},
    minimum height=0.68cm,inner xsep=4pt,inner ysep=2.5pt},
  terminal/.style={flowbox,rounded corners=0.18cm,minimum width=1.35cm,
    minimum height=0.72cm},
  process/.style={flowbox},
  decision/.style={flowbox,diamond,aspect=2.9,minimum width=3.55cm,
    minimum height=1.0cm},
  io/.style={flowbox,trapezium,trapezium left angle=72,
    trapezium right angle=108,minimum width=2.25cm,minimum height=0.78cm},
  branchlabel/.style={anchor=center,align=center,inner sep=0pt,
    execute at begin node={\setlength{\parindent}{0pt}\noindent}},
]
  \node[terminal] (start) at (0,9.00) {开始};
  \node[process,minimum width=1.65cm] (initK) at (0,7.65) {$k=1$};
  \node[process,minimum width=1.65cm] (initS) at (0,6.35) {$S=0$};
  \node[decision] (test) at (0,4.95) {$k\leq 50?$};
  \node[process,minimum width=2.65cm] (updateS) at (0,3.35)
    {$S=S+2k$};
  \node[process,minimum width=2.45cm] (increment) at (0,1.80)
    {$k=k+1$};
  \node[io,minimum width=2.15cm] (output) at (3.20,3.35) {输出 $S$};
  \node[terminal] (finish) at (3.20,1.80) {结束};

  \draw[->] (start.south) -- (initK.north);
  \draw[->] (initK.south) -- (initS.north);
  \draw[->] (initS.south) -- (test.north);

  \draw[->] (test.south) -- node[branchlabel,right=2pt] {是} (updateS.north);
  \draw[->] (updateS.south) -- (increment.north);
  \coordinate (return-left) at (-2.35,1.45);
  \coordinate (return-entry) at (-2.35,4.95);
  \draw (increment.south) -- (return-left) -- (return-entry);
  \draw[->] (return-entry) -- (test.west);

  \draw (test.east) -- node[branchlabel,above=2pt] {否} (3.20,4.95);
  \draw[->] (3.20,4.95) -- (3.20,3.74) -- (output.north);
  \draw[->] (output.south) -- (finish.north);
\end{tikzpicture}
