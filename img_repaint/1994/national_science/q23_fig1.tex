% 1994 national science Q23: right triangular prism A1B1C1-ABC, D midpoint of AC
\begin{tikzpicture}[x=1cm,y=1cm,font=\normalsize,line width=.6pt,>=Stealth,
  solid edge/.style={solid,line width=.6pt},
  hidden edge/.style={dashed,line width=.6pt},
  ptlabel/.style={font=\normalsize,inner sep=2pt}]
\coordinate (A1) at (0,4);
\coordinate (A) at (4.0,4.0);
\coordinate (B1) at (-0.7,0);
\coordinate (B) at (3.0,0);
\coordinate (C1) at (1.2,1.6);
\coordinate (C) at (5.2,1.6);
\coordinate (D) at (4.6,2.8);
% prism outline: top, left, bottom solid
\draw[solid edge] (A1)--(A);
\draw[solid edge] (A1)--(B1);
\draw[solid edge] (B1)--(B);
% bottom triangle edges
\draw[solid edge] (A)--(B);
\draw[solid edge] (B)--(C);
\draw[solid edge] (A)--(D)--(C);
% median BD
\draw[solid edge] (B)--(D);
% diagonal AB1 on side face
\draw[solid edge] (A)--(B1);
% hidden: laterals and top triangle interior
\draw[hidden edge] (A1)--(C1);
\draw[hidden edge] (B1)--(C1);
\draw[hidden edge] (B)--(C1);
\draw[hidden edge] (C1)--(C);
\draw[hidden edge] (C1)--(D);
\foreach \p in {A1,A,B1,B,C1,C,D}{\fill (\p) circle (1.2pt);}
\node[ptlabel,above left,xshift=-4pt,yshift=3pt] at (A1) {$A_{1}$};
\node[ptlabel,above right,xshift=4pt,yshift=3pt] at (A) {$A$};
\node[ptlabel,below left,xshift=-4pt,yshift=-3pt] at (B1) {$B_{1}$};
\node[ptlabel,below,xshift=1pt,yshift=-4pt] at (B) {$B$};
\node[ptlabel,left,xshift=-5pt,yshift=1pt] at (C1) {$C_{1}$};
\node[ptlabel,right,xshift=5pt] at (C) {$C$};
\node[ptlabel,right,xshift=4pt,yshift=1pt] at (D) {$D$};
\end{tikzpicture}
