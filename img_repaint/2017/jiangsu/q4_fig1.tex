\tikzset{
  flowstart/.style={draw, shape=rectangle, rounded corners=1.5mm, minimum width=26mm, minimum height=10mm, inner xsep=3mm, inner ysep=1.5mm, align=center, fill=white, execute at begin node={\parindent=0pt}},
  flowio/.style={draw, trapezium, trapezium left angle=75, trapezium right angle=105, minimum width=34mm, minimum height=11mm, inner sep=2mm, align=center, fill=white, execute at begin node={\parindent=0pt}},
  flowproc/.style={draw, rectangle, minimum width=26mm, minimum height=10mm, inner sep=2mm, align=center, fill=white, execute at begin node={\parindent=0pt}},
  flowprocl/.style={draw, rectangle, minimum width=30mm, minimum height=10mm, inner sep=2mm, align=center, fill=white, execute at begin node={\parindent=0pt}},
  flowprocr/.style={draw, rectangle, minimum width=46mm, minimum height=10mm, inner sep=2mm, align=center, fill=white, execute at begin node={\parindent=0pt}},
  flowdec/.style={draw, diamond, aspect=2.4, minimum width=48mm, minimum height=22mm, inner sep=1mm, align=center, fill=white, execute at begin node={\parindent=0pt}},
  flowarr/.style={thick, ->, >=Stealth},
  flowplain/.style={thick},
  flowlbl/.style={font=\normalsize, inner sep=1.5pt, execute at begin node={\parindent=0pt}}
}
\begin{tikzpicture}[x=1mm, y=1mm]
\node[flowstart] (start) at (0,0) {开始};
\node[flowio] (input) at (0,-26) {输入~$x$};
\node[flowdec] (cond) at (0,-51) {$x\geqslant 1?$};
\node[flowprocl] (yl) at (-44,-78) {$y\leftarrow 2^{x}$};
\node[flowprocr] (yr) at (44,-78) {$y\leftarrow 2+\log_{2}x$};
\node[flowio] (outp) at (0,-108) {输出~$y$};
\node[flowstart] (endn) at (0,-130) {结束};
\coordinate (J) at (0,-95);
\draw[flowarr] (start.south) -- (input.north);
\draw[flowarr] (input.south) -- (cond.north);
\draw[flowarr] (cond.west) -- node[pos=0.25, above=2mm, flowlbl] {Y} (-44,-51) -- (yl.north);
\draw[flowarr] (cond.east) -- node[pos=0.25, above=2mm, flowlbl] {N} (44,-51) -- (yr.north);
\draw[flowplain] (yl.south) |- (J);
\draw[flowplain] (yr.south) |- (J);
\draw[flowarr] (J) -- (outp.north);
\draw[flowarr] (outp.south) -- (endn.north);
\end{tikzpicture}
