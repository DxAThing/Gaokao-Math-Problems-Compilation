% Elements: rounded start/end terminals, input/output parallelograms, process boxes, decision diamond,
% directed connectors, and the left loop-back/right exit branches.
% Relations / directed edges: 开始→输入 a→S=0,K=1→K≤6; 是→S=S+a·K→a=-a→K=K+1→(left loop)→K≤6;
% 否→(right exit)→输出 S→结束.  The flowchart is schematic; coordinates preserve topology and relative layout.
% solid segments: all box outlines and connector lines/arrows; dashed segments: none.
% labels: node text centered; 是 is beside the downward true branch and 否 beside the right exit branch.

\begin{tikzpicture}[
    >=Stealth,
    line width=0.45pt,
    line cap=round,
    line join=round,
    font=\normalsize,
    flowbox/.style={draw, minimum height=0.52cm, inner xsep=3pt, inner ysep=1.2pt,
        align=center, text centered, execute at begin node={\setlength{\parindent}{0pt}}},
    terminal/.style={flowbox, rounded corners=2.5pt, minimum width=0.95cm},
    process/.style={flowbox},
    io/.style={flowbox, trapezium, trapezium left angle=72, trapezium right angle=108},
    flowarrow/.style={-Stealth, line width=0.45pt},
]
    % The vertical spacing follows the source chart; x coordinates keep its narrow center column.
    \node[terminal] (start) at (0,0) {开始};
    \node[io, minimum width=1.35cm] (input) at (0,-1.05) {输入 $a$};
    \node[process, minimum width=2.40cm] (init) at (0,-2.10) {$S=0,K=1$};
    \node[draw, diamond, aspect=3.0, minimum width=2.72cm, minimum height=0.82cm,
        inner sep=0pt, align=center, text centered,
        execute at begin node={\setlength{\parindent}{0pt}}] (test) at (0,-3.15) {$K\leq 6$};
    \node[process, minimum width=2.45cm] (sum) at (0,-4.30) {$S=S+a\cdot K$};
    \node[process, minimum width=1.35cm] (neg) at (0,-5.30) {$a=-a$};
    \node[process, minimum width=2.00cm] (inc) at (0,-6.30) {$K=K+1$};
    \node[io, minimum width=1.40cm] (output) at (0,-7.40) {输出 $S$};
    \node[terminal] (finish) at (0,-8.50) {结束};
    \coordinate (merge) at (0,-2.50);

    \draw[flowarrow] (start) -- (input);
    \draw[flowarrow] (input) -- (init);
    \draw (init.south) -- (merge);
    \draw[flowarrow] (merge) -- (test.north);
    \draw[flowarrow] (test) -- node[right=2pt] {是} (sum);
    \draw[flowarrow] (sum) -- (neg);
    \draw[flowarrow] (neg) -- (inc);
    \draw[flowarrow] (output) -- (finish);

    % False branch exits right, turns down, then returns left to the output arrow.
    \coordinate (rightbranch) at (2.40,-3.15);
    \coordinate (rightlevel) at (2.40,-7.10);
    \draw (test.east) -- node[above=2pt] {否} (rightbranch)
        -- (rightlevel) -- (0,-7.10);
    \draw[flowarrow] (0,-7.10) -- (output.north);

    % True loop: after increment, turn left and rise to the decision entry.
    \coordinate (leftbranch) at (-2.05,-6.30);
    % Place the loop entry below the initialization box (>=2.5 mm clear).
    \coordinate (leftlevel) at (-2.05,-2.50);
    \draw (inc.west) -- (leftbranch) -- (leftlevel);
    \draw[flowarrow] (leftlevel) -- (merge);

    % Preserve the source crop's modest white border around the inked chart.
    \path[use as bounding box] (-2.55,0.30) rectangle (3.10,-8.90);
\end{tikzpicture}
