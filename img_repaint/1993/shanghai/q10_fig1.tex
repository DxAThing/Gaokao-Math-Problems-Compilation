% 1993 Shanghai Q10: cube ABCD-A1B1C1D1 with section B-D-C1
\begin{tikzpicture}[x=1cm,y=1cm,font=\normalsize,line width=.6pt,>=Stealth,
  solid edge/.style={solid,line width=.6pt},
  hidden edge/.style={dashed,line width=.6pt},
  ptlabel/.style={font=\normalsize,inner sep=2pt}]
\coordinate (A) at (0,0);
\coordinate (B) at (3,0);
\coordinate (C) at (4.0,1.0);
\coordinate (D) at (1.0,1.0);
\coordinate (Aup) at (0,2.6);
\coordinate (B1) at (3,2.6);
\coordinate (C1) at (4.0,3.6);
\coordinate (D1) at (1.0,3.6);
% front face
\draw[solid edge] (A)--(B)--(B1)--(Aup)--cycle;
% back face
\draw[solid edge] (C1)--(D1);
\draw[solid edge] (Aup)--(D1);
\draw[solid edge] (B1)--(C1);
\draw[hidden edge] (D)--(C);
\draw[hidden edge] (D)--(D1);
\draw[solid edge] (C)--(C1);
% depth
\draw[hidden edge] (A)--(D);
\draw[solid edge] (B)--(C);
% section B-D-C1
\draw[hidden edge] (B)--(D);
\draw[hidden edge] (D)--(C1);
\draw[solid edge] (B)--(C1);
\foreach \p in {A,B,C,D,Aup,B1,C1,D1}{\fill (\p) circle (1.2pt);}
\node[ptlabel,below left,xshift=-4pt,yshift=-3pt] at (A) {$A$};
\node[ptlabel,below right,xshift=4pt,yshift=-3pt] at (B) {$B$};
\node[ptlabel,right,xshift=5pt,yshift=-1pt] at (C) {$C$};
\node[ptlabel,left,xshift=-4pt,yshift=1pt] at (D) {$D$};
\node[ptlabel,left,xshift=-5pt] at (Aup) {$A_{1}$};
\node[ptlabel,right,xshift=2pt,yshift=-5pt] at (B1) {$B_{1}$};
\node[ptlabel,above right,xshift=3pt,yshift=2pt] at (C1) {$C_{1}$};
\node[ptlabel,above left,xshift=-3pt,yshift=3pt] at (D1) {$D_{1}$};
\end{tikzpicture}
