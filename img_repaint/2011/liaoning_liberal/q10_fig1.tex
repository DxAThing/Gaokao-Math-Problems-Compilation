% Program topology: 开始 -> 输入 n -> s=0,t=1,k=1,p=1 -> k<n; 是 -> p=s+t
% -> s=t,t=p -> k=k+1 -> left return to the shared stem above the test;
% 否 -> shared output p -> 结束. All connectors are solid directed edges.

\begin{tikzpicture}[
    >=Stealth,
    x=1cm,
    y=0.84cm,
    line width=0.45pt,
    line cap=round,
    line join=round,
    font=\normalsize,
    flowtext/.style={anchor=center, align=center,
        execute at begin node={\setlength{\parindent}{0pt}}},
    every node/.style={flowtext},
    flowbox/.style={draw, flowtext, minimum height=0.68cm,
        inner xsep=5pt, inner ysep=3pt},
    terminal/.style={flowbox, rounded corners=0.16cm, minimum width=1.35cm},
    process/.style={flowbox, minimum width=2.25cm},
    decision/.style={flowbox, diamond, aspect=3.0, minimum width=3.55cm,
        minimum height=1.04cm},
    io/.style={flowbox, trapezium, trapezium left angle=72,
        trapezium right angle=108, minimum height=0.78cm},
    branchlabel/.style={flowtext, inner sep=0pt},
]
    \path[use as bounding box] (-3.55,-1.25) rectangle (4.45,10.45);

    \node[terminal] (start) at (0,10.10) {开始};
    \node[io, minimum width=1.85cm] (input) at (0,8.92) {输入 $n$};
    \node[process, minimum width=4.15cm] (init) at (0,7.65)
        {$s=0,t=1,k=1,p=1$};
    \node[decision] (test) at (0,5.95) {$k<n$};
    \node[process, minimum width=2.25cm] (add) at (0,4.50) {$p=s+t$};
    \node[process, minimum width=2.80cm] (swap) at (0,3.22)
        {$s=t,t=p$};
    \node[process, minimum width=2.35cm] (increment) at (0,1.95)
        {$k=k+1$};
    \node[io, minimum width=2.05cm] (output) at (0,0.75) {输出 $p$};
    \node[terminal] (finish) at (0,-0.55) {结束};

    \draw[->] (start.south) -- (input.north);
    \draw[->] (input.south) -- (init.north);
    \draw (init.south) -- (0,6.80);
    \draw[->] (0,6.80) -- (test.north);

    \draw[->] (test.south) -- node[branchlabel, right=2pt] {是} (add.north);
    \draw[->] (add.south) -- (swap.north);
    \draw[->] (swap.south) -- (increment.north);
    \draw (test.east) -- node[branchlabel, above=2pt] {否} (3.25,5.95) -- (3.25,1.38) -- (0,1.38);
    \draw[->] (0,1.38) -- (output.north);
    \draw[->] (output.south) -- (finish.north);

    % Return from k update: one left-pointing arrow enters the shared stem;
    % the stem carries the only downward arrow into the decision.
    \draw[->] (increment.west) -- (-3.05,1.95) -- (-3.05,6.80) -- (0,6.80);
\end{tikzpicture}
