% 2011_liaoning_science_q6_fig1: 垂直主干 开始->输入n->初始化->diamond k<n -是->p=s+t->s=t,t=p->k=k+1 回路; 否经右侧通道至 输出p
\begin{tikzpicture}[>=Stealth, line width=0.6pt,
  startstop/.style={shape=rectangle, rounded corners=1.5mm, draw, minimum width=26mm, minimum height=10mm, inner xsep=3mm, inner ysep=1.5mm, align=center},
  proc/.style={rectangle, draw, minimum height=8mm, inner sep=2mm, align=center},
  dec/.style={diamond, draw, aspect=2.8, inner sep=1.5mm, align=center},
  io/.style={trapezium, trapezium left angle=75, trapezium right angle=105, draw, minimum height=8mm, inner sep=2mm, align=center},
  lbl/.style={fill=white, inner sep=1pt}
]
{\parindent=0pt
\node[startstop] (start) at (0,0) {开始};
\node[io, minimum width=26mm] (input) at (0,-1.7) {输入 $n$};
\node[proc, minimum width=62mm] (init) at (0,-3.4) {$s=0,t=1,k=1,p=1$};
\node[dec, minimum width=40mm] (cond) at (0,-5.4) {$k<n$};
\node[proc, minimum width=28mm] (p1) at (0,-7.2) {$p=s+t$};
\node[proc, minimum width=32mm] (p2) at (0,-8.9) {$s=t,t=p$};
\node[proc, minimum width=28mm] (inc) at (0,-10.6) {$k=k+1$};
\node[io, minimum width=26mm] (out) at (0,-12.5) {输出 $p$};
\node[startstop] (stop) at (0,-14.2) {结束};
\coordinate (Jtop) at (0,-4.35);
\draw[->] (start.south) -- (input.north);
\draw[->] (input.south) -- (init.north);
\draw[->] (init.south) -- (Jtop) -- (cond.north);

\draw[->] (cond.south) -- (p1.north) node[midway, right=2mm, lbl] {是};
\draw[->] (p1.south) -- (p2.north);
\draw[->] (p2.south) -- (inc.north);
\draw[->] (out.south) -- (stop.north);
% 否分支: cond east -> 右通道 -> 下 -> 进入 out north (单路径, 避开中部各框)
\draw[->] (cond.east) -- ++(14mm,0) -- ++(0,-61mm) -| (out.north);
\node[lbl, above=2mm] at ([xshift=5mm]cond.east) {否};
% 回路: inc west -> 左通道 -> 上 -> 进入 Jtop (箭头朝东, 与主干南向不同向)
\draw[->] (inc.west) -- ++(-14mm,0) |- (Jtop);
}
\end{tikzpicture}
