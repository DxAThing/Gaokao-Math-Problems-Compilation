\tikzset{
  gflowproc/.style={draw, rectangle, minimum width=34mm, minimum height=9mm, inner sep=1.5mm, align=center, font=\normalsize, execute at begin node={\setlength{\parindent}{0pt}}},
  gflowterm/.style={draw, shape=rectangle, rounded corners=1.5mm, minimum width=26mm, minimum height=10mm, inner xsep=3mm, inner ysep=1.5mm, align=center, font=\normalsize, execute at begin node={\setlength{\parindent}{0pt}}},
  gflowio/.style={draw, trapezium, trapezium left angle=72, trapezium right angle=108, minimum width=36mm, minimum height=10mm, inner sep=1.5mm, align=center, font=\normalsize, execute at begin node={\setlength{\parindent}{0pt}}},
  gflowdec/.style={draw, diamond, aspect=2.4, minimum width=34mm, minimum height=18mm, inner sep=1mm, align=center, font=\normalsize, execute at begin node={\setlength{\parindent}{0pt}}},
  gflowarr/.style={->, >=Stealth, line width=0.6pt}
}
\begin{tikzpicture}[gflowarr, x=1mm, y=1mm]
\parindent=0pt
\node[gflowterm] (start) at (0,0) {开始};
\node[gflowio] (input) at (0,-20) {输入 $N$};
\node[gflowproc, minimum width=38mm] (init) at (0,-40) {$k=1,p=1$};
\coordinate (J) at (0,-52);
\node[gflowproc, minimum width=30mm] (mul) at (0,-64) {$p=p\cdot k$};
\node[gflowdec] (dec) at (0,-86) {$k<N$};
\node[gflowio] (out) at (0,-108) {输出 $p$};
\node[gflowterm] (stop) at (0,-128) {结束};
\node[gflowproc, minimum width=30mm] (inc) at (42,-64) {$k=k+1$};
\draw (start.south) -- (input.north);
\draw (input.south) -- (init.north);
\draw[->] (init.south) -- (J) -- (mul.north);

\draw[->] (mul.south) -- (dec.north);
\draw[->] (dec.south) -- node[midway, right=1mm] {否} (out.north);
\draw[->] (out.south) -- (stop.north);
% yes branch: east then up into k box south (horizontal first, then vertical; no overshoot)
\draw[->] (dec.east) -| (inc.south);
\node[above=1mm] at ([xshift=5mm]dec.east) {是};
% Exit upward into the clear junction level, then turn left once.
\draw[->] (inc.north) |- (J);
\end{tikzpicture}
