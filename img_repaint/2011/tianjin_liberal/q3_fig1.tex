\begin{tikzpicture}[x=1cm, y=1cm, >=Stealth, line width=0.6pt]
\tikzset{
  flowchart node/.style={align=center, execute at begin node={\parindent=0pt}},
  startstop/.style={flowchart node, draw, shape=rectangle, rounded corners=1.5mm, minimum width=26mm, minimum height=10mm, inner xsep=3mm, inner ysep=1.5mm},
  process/.style={flowchart node, draw, rectangle, minimum height=0.95cm, inner sep=5pt},
  decision/.style={flowchart node, draw, diamond, aspect=2.4, inner sep=4pt},
  io/.style={flowchart node, draw, trapezium, trapezium left angle=70, trapezium right angle=110, inner sep=5pt, minimum height=0.95cm}
}
\node[startstop] (start) at (0,0) {开始};
\node[io, minimum width=2.4cm] (input) at (0,-1.4) {输入 $x$};
\node[decision, minimum width=3.4cm, minimum height=1.6cm] (cond) at (0,-3.5) {$|x|>3?$};
\node[process, minimum width=2.6cm] (upd) at (0,-5.5) {$x=|x-3|$};
\node[process, minimum width=2.2cm] (comp) at (0,-7.3) {$y=2^{x}$};
\node[io, minimum width=2.4cm] (output) at (0,-8.9) {输出 $y$};
\node[startstop] (end) at (0,-10.5) {结束};
\coordinate (Jtop) at (0,-2.3);
\coordinate (Jmid) at (0,-6.4);
\coordinate (retW) at (-3.4,-5.5);
\coordinate (retWN) at (-3.4,-2.3);
\coordinate (noE) at (3.4,-3.5);
\coordinate (noES) at (3.4,-6.4);
\draw[->] (start.south) -- (input.north);
\draw[->] (input.south) -- (cond.north);
\draw[->] (cond.south) -- (upd.north) node[midway, right=1.0mm] {是};
\draw[->] (upd.south) -- (comp.north);
\draw[->] (comp.south) -- (output.north);
\draw[->] (output.south) -- (end.north);
% west loop back around decision box (2 bends)
\draw[->] (upd.west) -- (retW) -- (retWN) -- (Jtop);
% east 否 branch around update box (2 bends)
\draw[->] (cond.east) -- (noE) -- (noES) -- (Jmid);
\node[above=1.0mm] at (2.2,-3.5) {否};
\end{tikzpicture}
