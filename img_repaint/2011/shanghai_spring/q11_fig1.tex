% Source PDF: source/春季高考/2011/2011春季上海.pdf, page 1, question 11.
% The chart is vector artwork; the original q11 PNG is retained for comparison.
% Node-edge list: 开始 -> i=0,s=66 -> s<=0;
% s<=0 --是--> 输出 i -> 结束; s<=0 --否--> s=s-10 -> i=i+1
% -> loop-merge -> s<=0. All borders/connectors are solid; no dashed lines.
% Each node is locally zero-indented and centered; the loop return is a single
% left-pointing arrow into the shared stem above the decision.

\begin{tikzpicture}[
    >=Stealth,
    x=1cm,
    y=0.86cm,
    line width=0.45pt,
    line cap=round,
    line join=round,
    font=\normalsize,
    flowtext/.style={anchor=center, align=center,
        execute at begin node={\setlength{\parindent}{0pt}}},
    flowbox/.style={draw, flowtext, minimum height=0.70cm,
        inner xsep=5pt, inner ysep=3pt},
    terminal/.style={flowbox, rounded corners=0.16cm, minimum width=1.40cm},
    process/.style={flowbox, minimum width=2.55cm},
    decision/.style={flowbox, diamond, aspect=2.9, minimum width=3.65cm,
        minimum height=1.05cm},
    io/.style={flowbox, trapezium, trapezium left angle=72,
        trapezium right angle=108, minimum height=0.80cm},
]
    \path[use as bounding box] (-3.80,0.25) rectangle (4.75,8.95);

    \node[terminal] (start) at (0,8.55) {开始};
    \node[process, minimum width=3.55cm] (init) at (0,7.15)
        {$i=0,s=66$};
    \node[decision] (test) at (0,4.95) {$s\leqslant 0$};
    \node[io, minimum width=2.65cm] (output) at (0,2.95) {输出 $i$};
    \node[terminal] (finish) at (0,1.35) {结束};
    \node[process, minimum width=2.85cm] (decrement) at (3.25,4.95)
        {$s=s-10$};
    \node[process, minimum width=2.85cm] (increment) at (3.25,6.15)
        {$i=i+1$};

    \draw[->] (start.south) -- (init.north);
    % Initial stem and loop return share one connector above the decision.
    \coordinate (loop-merge) at (0,6.15);
    \draw (init.south) -- (loop-merge);
    \draw[->] (loop-merge) -- (test.north);

    % Decision branches: yes terminates; no updates s and i before looping.
    \draw[->] (test.south) -- (output.north);
    \draw[->] (output.south) -- (finish.north);
    \draw[->] (test.east) -- (decrement.west);
    \draw[->] (decrement.north) -- (increment.south);
    \draw[->] (increment.west) -- (loop-merge);

    \node[flowtext, anchor=south, inner sep=1pt] at (1.30,5.10) {否};
    \node[flowtext, anchor=east, inner sep=1pt] at (-0.55,4.05) {是};
\end{tikzpicture}
