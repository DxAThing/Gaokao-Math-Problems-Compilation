% Source PDF: source/普通高考/2011/2011江西文.pdf, page 1, question 13.
% The program diagram is vector line art in the source PDF (pdfimages: no embedded images).
% Node-edge list: 开始 -> s=0,n=1 -> s=(s+n)n -> n=n+1 -> s>3?;
% 是 -> 输出s -> 结束; 否 -> independent loop back to the single entry before s=(s+n)n.
% The loop joins the existing horizontal entry; no second center arrow or output connection is drawn.

\begin{tikzpicture}[
  x=1cm,y=1cm,baseline,>=Stealth,
  line width=0.45pt,line cap=butt,line join=miter,
  every node/.style={anchor=center,align=center,font=\normalsize,
    execute at begin node={\setlength{\parindent}{0pt}\noindent}},
  flowbox/.style={draw,anchor=center,align=center,inner xsep=4pt,inner ysep=2.5pt,
    execute at begin node={\setlength{\parindent}{0pt}\noindent}},
  terminal/.style={flowbox,rounded corners=0.16cm,minimum height=0.68cm},
  process/.style={flowbox,minimum height=0.68cm},
  decision/.style={flowbox,diamond,aspect=2.8,minimum height=0.92cm,inner xsep=4pt,inner ysep=2pt},
  io/.style={flowbox,trapezium,trapezium left angle=72,trapezium right angle=108,
    minimum height=0.74cm}
]
  \node[terminal,minimum width=1.35cm] (start) at (0,0) {开始};
  \node[process,minimum width=2.45cm] (init) at (2.15,0) {$s=0,n=1$};
  \node[process,minimum width=2.75cm] (update) at (5.05,0) {$s=(s+n)n$};
  \node[process,minimum width=2.25cm] (increment) at (7.70,0) {$n=n+1$};
  \node[decision,minimum width=2.75cm] (test) at (10.45,0) {$s>3$};
  \node[io,minimum width=2.15cm] (output) at (13.05,0) {输出 $s$};
  \node[terminal,minimum width=1.35cm] (finish) at (15.25,0) {结束};

  \draw[->] (start.east) -- (init.west);
  \draw[->] (init.east) -- (update.west);
  \draw[->] (update.east) -- (increment.west);
  \draw[->] (increment.east) -- (test.west);
  \draw[->] (test.east) -- (output.west);
  \draw[->] (output.east) -- (finish.west);

  % 否 returns below the chain and enters the same update connector.
  \coordinate (merge) at (3.55,0);
  \coordinate (loopdown) at (10.45,-1.45);
  \coordinate (loopleft) at (3.55,-1.45);
  \draw[->] (test.south) -- (loopdown) -- (loopleft) -- (merge);
  \node[anchor=south,inner sep=1pt] at (11.75,0.12) {是};
  \node[anchor=north,inner sep=1pt] at (10.45,-0.52) {否};
\end{tikzpicture}
