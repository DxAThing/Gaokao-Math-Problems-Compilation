% 两矩形: 长宽分别相等的正视图与俯视图, 横向并排, 长对正宽相等.
% 两矩形全等空白矩形, 全部实线, 无尺寸、无虚线、无内部线条.
% 点修: 俯视图矩形平移至右侧横排 + 正(主)视图改名为正视图, 未改边线/尺寸.
\begin{tikzpicture}[line width=.6pt,font=\normalsize,x=1cm,y=1cm]
% 正视图矩形
\draw (0,3.0) rectangle (3,4.8);
\node at (1.5,2.5) {正视图};
% 俯视图矩形(与正视图同宽同长, 平移至右侧横排)
\draw (4.6,3.0) rectangle (7.6,4.8);
\node at (6.1,2.5) {俯视图};
\end{tikzpicture}
