\tikzset{
  gflowproc/.style={draw, rectangle, minimum width=30mm, minimum height=9mm, inner sep=1.5mm, align=center, font=\normalsize, execute at begin node={\setlength{\parindent}{0pt}}},
  gflowterm/.style={draw, shape=rectangle, rounded corners=1.5mm, minimum width=26mm, minimum height=10mm, inner xsep=3mm, inner ysep=1.5mm, align=center, font=\normalsize, execute at begin node={\setlength{\parindent}{0pt}}},
  gflowio/.style={draw, trapezium, trapezium left angle=72, trapezium right angle=108, minimum width=40mm, minimum height=10mm, inner sep=1.5mm, align=center, font=\normalsize, execute at begin node={\setlength{\parindent}{0pt}}},
  gflowdec/.style={draw, diamond, aspect=2.6, minimum width=56mm, minimum height=20mm, inner sep=1mm, align=center, font=\normalsize, execute at begin node={\setlength{\parindent}{0pt}}},
  gflowarr/.style={->, >=Stealth, line width=0.6pt}
}
\begin{tikzpicture}[gflowarr, x=1mm, y=1mm]
\parindent=0pt
% main column x=0
\node[gflowterm] (start) at (0,0) {开始};
\node[gflowio, minimum width=72mm] (input) at (0,-20) {输入非负整数 $l,m,n$};
\node[gflowdec] (dec1) at (0,-44) {$l^{2}+m^{2}+n^{2}=0$};
\node[gflowproc, minimum width=58mm] (calc) at (0,-70) {$y=70l+21m+15n$};
\coordinate (J1) at (0,-84);
\node[gflowdec, minimum width=48mm] (dec2) at (0,-102) {$y>105$};
\coordinate (Jout) at (0,-120);
\node[gflowio, minimum width=32mm] (out) at (0,-132) {输出 $y$};
\node[gflowterm] (stop) at (0,-152) {结束};
% left box
\node[gflowproc, minimum width=24mm] (y105) at (-52,-70) {$y=105$};
% right box
\node[gflowproc, minimum width=32mm] (sub) at (52,-84) {$y=y-105$};
% dec1 south (no) to calc
\draw (start.south) -- (input.north);
\draw (input.south) -- (dec1.north);
\draw[->] (dec1.south) -- node[midway, right=1mm] {否} (calc.north);
\draw[->] (calc.south) -- (J1) -- (dec2.north);

% dec1 west (yes): west 12mm then down into y105 north
\draw[->] (dec1.west) -| (y105.north);
\node[above=1mm] at ([xshift=-5mm]dec1.west) {是};
% left outer channel down to output junction
\coordinate (lw) at (-64,-70);
\coordinate (lj) at (-64,-120);
\draw (y105.south) -- ++(0,-4) -| (lj);
\draw (lj) -- (Jout);
% dec2 south (no) to output junction: plain wire into Jout, single trunk arrow below (no same-line duplicate arrows)
\draw (dec2.south) -- node[midway, right=1mm] {否} (Jout);
\draw[->] (Jout) -- (out.north);
\draw[->] (out.south) -- (stop.north);
% dec2 east (yes): east then up into sub south (orthogonal, single bend)
\draw[->] (dec2.east) -| (sub.south);
\node[above=1mm] at ([xshift=5mm]dec2.east) {是};
% return: sub west to J1 via offset channel (no overshoot of J1)
\coordinate (sw) at (28,-84);
\draw (sub.west) -- (sw);
\draw[->] (sw) -- (J1);
\end{tikzpicture}
