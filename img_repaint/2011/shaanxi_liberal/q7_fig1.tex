\tikzset{
  gflowproc/.style={draw, rectangle, minimum width=34mm, minimum height=11mm, inner sep=1.5mm, align=center, font=\normalsize, execute at begin node={\setlength{\parindent}{0pt}}},
  gflowterm/.style={draw, shape=rectangle, rounded corners=1.5mm, minimum width=26mm, minimum height=10mm, inner xsep=3mm, inner ysep=1.5mm, align=center, font=\normalsize, execute at begin node={\setlength{\parindent}{0pt}}},
  gflowio/.style={draw, trapezium, trapezium left angle=72, trapezium right angle=108, minimum width=44mm, minimum height=10mm, inner sep=1.5mm, align=center, font=\normalsize, execute at begin node={\setlength{\parindent}{0pt}}},
  gflowdec/.style={draw, diamond, aspect=2.6, minimum width=62mm, minimum height=20mm, inner sep=1mm, align=center, font=\normalsize, execute at begin node={\setlength{\parindent}{0pt}}},
  gflowarr/.style={->, >=Stealth, line width=0.6pt}
}
\begin{tikzpicture}[gflowarr, x=1mm, y=1mm]
\parindent=0pt
\node[gflowterm] (start) at (0,0) {开始};
\node[gflowio] (input) at (0,-20) {输入 $x_{1},x_{2},x_{3}$};
\node[gflowdec] (dec) at (0,-44) {$|x_{1}-x_{2}|<|x_{2}-x_{3}|$};
\node[gflowproc, minimum width=32mm, minimum height=13mm] (py) at (0,-70) {$p=\dfrac{x_{1}+x_{2}}{2}$};
\node[gflowproc, minimum width=32mm, minimum height=13mm] (pn) at (48,-70) {$p=\dfrac{x_{2}+x_{3}}{2}$};
\coordinate (J) at (0,-86);
\node[gflowio, minimum width=30mm] (out) at (0,-100) {输出 $p$};
\node[gflowterm] (stop) at (0,-120) {结束};
\draw (start.south) -- (input.north);
\draw (input.south) -- (dec.north);
% yes: straight down with side label
\draw[->] (dec.south) -- node[midway, right=1mm] {是} (py.north);
% no: east 14mm then down into right box north (single bend, label above long segment)
\draw[->] (dec.east) -| (pn.north);
\node[above=1mm] at ([xshift=5mm]dec.east) {否};
% merge: single trunk py->out with one arrow; right branch T-joins trunk interior (no coincident wire endpoints)
\draw[->] (py.south) -- (out.north);
\draw (pn.south) |- (J);
\draw[->] (out.south) -- (stop.north);
\end{tikzpicture}
