  fzstart/.style={shape=rectangle, rounded corners=1.5mm, draw=black, line width=0.6pt, minimum width=26mm, minimum height=10mm, inner xsep=3mm, inner ysep=1.5mm, align=center, execute at begin node={\parindent=0pt}},
  fzio/.style={trapezium, trapezium left angle=70, trapezium right angle=110, draw=black, line width=0.6pt, inner sep=1.5mm, align=center, execute at begin node={\parindent=0pt}},

% A=A+B / B=A+B / k=k+1 / 输出 B / 结束. 共8节点.
% 有向边: 开始->输入, 输入->菱形(途经汇合J1), 菱形南->A(是),
% A->B, B->k, k西->回环->J1(箭头在J1),
% 菱形东->右下外侧->输出(否, 整条完整路径, 仅末端箭头), 输出->结束. 共8边.
\node[fzstart] (start) at (0,0) {开始};
\node[fzio, minimum width=92mm, minimum height=10mm] (input) at (0,-1.7) {输入 $n=3$, $A=1$, $B=2$, $k=1$};
\coordinate (J1) at (0,-3.05);
\node[fzdia, minimum width=46mm, minimum height=20mm] (dia) at (0,-4.5) {$k\leqslant n$};
\node[fzproc, minimum width=34mm, minimum height=9mm] (procA) at (0,-6.9) {$A=A+B$};
\node[fzproc, minimum width=34mm, minimum height=9mm] (procB) at (0,-8.5) {$B=A+B$};
\node[fzproc, minimum width=34mm, minimum height=9mm] (prock) at (0,-10.0) {$k=k+1$};
\node[fzio, minimum width=30mm, minimum height=10mm] (output) at (0,-12.05) {输出 $B$};
\node[fzstart] (endnode) at (0,-13.75) {结束};
\draw[fzedge] (start.south) -- (input.north);
\draw[fzedge] (input.south) -- (dia.north);
\draw[fzedge] (dia.south) -- node[fzlab, right=1.5mm]{是} (procA.north);
\draw[fzedge] (procA.south) -- (procB.north);
\draw[fzedge] (procB.south) -- (prock.north);
\draw[fzedge] (prock.west) -- (-4.3,-10.0) -- (-4.3,-3.05) -- (J1);
\draw[fzedge] (dia.east) -- node[fzlab, above=1.5mm, pos=0.25]{否} (4.3,-4.5) -- (4.3,-11.0) -- (0,-11.0) -- (output.north);
\draw[fzedge] (output.south) -- (endnode.north);
