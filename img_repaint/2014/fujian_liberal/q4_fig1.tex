    terminal/.style={flowbox, rounded corners=0.14cm, minimum width=1.25cm},

    \draw[->] (start.south) -- (init.north);
    \draw[->] (init.south) -- (test.north);
    \draw[->] (test.south) -- node[right=2pt] {否} (output.north);
    \draw[->] (output.south) -- (finish.north);

    % 是 branch: increment n, then return from the box's left edge to the

    \draw[->] (test.east) -- node[below=2pt] {是} (2.55,5.45) -- (2.55,6.06) -- (increment.south);
