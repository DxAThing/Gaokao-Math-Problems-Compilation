  flowstart/.style={shape=rectangle, draw, rounded corners=1.5mm, align=center, inner xsep=3mm, inner ysep=1.5mm, minimum width=26mm, minimum height=10mm},
  flowio/.style={trapezium, draw, trapezium left angle=75, trapezium right angle=105, align=center, inner sep=3mm, minimum width=34mm, minimum height=10mm},

  \node[flowio, below=12mm of start] (in1) {输入 $x,y$};
  \node[flowdec, below=14mm of in1] (dec) {\begin{tabular}{@{}c@{}}$x\geqslant 0,y\geqslant 0,$\\$x+y\leqslant 1?$\end{tabular}};

  \node[flowio, below=14mm of merge] (out1) {输出 $S$};
  \node[flowstart, below=12mm of out1] (end1) {结束};

  \draw[flowarrow] (in1.south) -- (dec.north);
  \draw[flowarrow] (dec.west) -- node[above, yshift=1.5mm, pos=0.18] {是} (pl.north |- dec.west) -- (pl.north);
  \draw[flowarrow] (dec.east) -- node[above, yshift=1.5mm, pos=0.18] {否} (pr.north |- dec.east) -- (pr.north);
  \draw[flowarrow,-] (pl.south) -- (pl.south |- merge) -- (merge);
  \draw[flowarrow,-] (pr.south) -- (pr.south |- merge) -- (merge);
  \draw[flowarrow] (merge) -- (out1.north);
  \draw[flowarrow] (out1.south) -- (end1.north);
