% Source PDF: source/普通高考/2014/2014重庆理.pdf, page 1; vector flowchart.
% Grid evidence: tmp/review_2014_chongqing_science/grid/page-1-grid100.jpg and
%   q5_original_final.png (no-grid crop from the source page).
% Elements: rounded start/end terminals; initialization rectangle k=9,s=1;
%   blank decision diamond (the question asks for its condition); right-side
%   update rectangles s=s*k/(k+1), k=k-1; output parallelogram; connectors and labels.
% Complete node -> directed-edge list:
%   start -> init -> test; test --是--> update_s -> decrement -> main entry above test;
%   test --否--> output -> finish. The loop return has one left-pointing arrow
%   into the main vertical stem; no edge joins the update branch to output.
% All borders/connectors are solid; no dashed segments or extra edges.

\begin{tikzpicture}[
  >=Stealth,
  line width=0.45pt,
  line join=round,
  line cap=round,
  font=\normalsize,
  flowtext/.style={anchor=center, align=center, inner sep=0pt,
    execute at begin node={\setlength{\parindent}{0pt}}},
  every node/.style={flowtext},
  flowbox/.style={flowtext, draw, minimum height=0.68cm,
    inner xsep=3pt, inner ysep=2pt},
  terminal/.style={flowbox, rounded corners=0.18cm, minimum width=1.45cm},
  process/.style={flowbox},
  decision/.style={flowbox, diamond, aspect=2.7, minimum width=3.50cm,
    minimum height=0.95cm},
  io/.style={flowbox, trapezium, trapezium left angle=72,
    trapezium right angle=72, minimum width=2.35cm, minimum height=0.76cm},
]
  \node[terminal] (start) at (0,9.55) {开始};
  \node[process, minimum width=2.65cm] (init) at (0,8.05) {$k=9,s=1$};
  \node[decision] (test) at (0,5.40) {};

  \node[process, minimum width=3.25cm, minimum height=0.95cm]
    (update_s) at (3.85,5.40) {$s=s\cdot\dfrac{k}{k+1}$};
  \node[process, minimum width=2.35cm] (decrement) at (3.85,7.05)
    {$k=k-1$};
  \node[io] (output) at (0,3.55) {输出 $k$};
  \node[terminal] (finish) at (0,2.10) {结束};

  % Main trunk and false branch.
  \draw[->] (start.south) -- (init.north);
  \draw[->] (init.south) -- (test.north);
  \draw[->] (test.south) -- node[flowtext, right] {否} (output.north);
  \draw[->] (output.south) -- (finish.north);

  % True branch and loop return. The horizontal return arrow merges above test;
  % it remains separate from the downward main-trunk arrow into the diamond.
  \draw[->] (test.east) -- node[flowtext, above] {是} (update_s.west);
  \draw[->] (update_s.north) -- (decrement.south);
  \draw[->] (decrement.north) -- (3.85,7.60) -- (0,7.60);
\end{tikzpicture}
