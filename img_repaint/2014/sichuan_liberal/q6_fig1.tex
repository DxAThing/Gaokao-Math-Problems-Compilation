% Source PDF: source/普通高考/2014/2014四川文.pdf, page 1, question 6.
% pdfimages -list found no embedded images; the source program flowchart is vector artwork.
% Node -> directed-edge list: 开始 -> 输入 x,y -> test;
% test --是--> S=2x+y -> merge -> 输出 S -> 结束;
% test --否--> S=1 -> merge.  The two branch outputs share only the merge
% connector before 输出 S; there is no extra or missing directed edge.
% All borders and connectors are solid; node text is centered with local parindent=0pt.

\begin{tikzpicture}[
    >=Stealth,
    x=1cm,
    y=1cm,
    line width=0.45pt,
    line cap=round,
    line join=round,
    font=\normalsize,
    flowtext/.style={anchor=center, align=center,
        execute at begin node={\setlength{\parindent}{0pt}}},
    flowbox/.style={flowtext, draw, minimum height=0.68cm,
        inner xsep=4pt, inner ysep=2pt},
    terminal/.style={flowbox, rounded corners=2pt, minimum width=1.35cm},
    io/.style={flowbox, trapezium, trapezium left angle=72, trapezium right angle=108},
    process/.style={flowbox, minimum width=2.55cm},
    decision/.style={flowbox, diamond, aspect=2.7, inner sep=0pt,
        minimum width=4.75cm, minimum height=1.25cm},
    branchlabel/.style={flowtext, inner sep=0pt},
]
    \node[terminal] (start) at (0,8.65) {开始};
    \node[io, minimum width=2.50cm] (input) at (0,7.25) {输入 $x,y$};
    \node[decision] (test) at (0,5.65) {$x\geq0,y\geq0,$\\$x+y\leq1?$};
    \node[process, minimum width=2.20cm] (yes) at (-2.65,4.05) {$S=2x+y$};
    \node[process, minimum width=2.20cm] (no) at (2.65,4.05) {$S=1$};
    \node[io, minimum width=1.80cm] (output) at (0,2.55) {输出 $S$};
    \node[terminal] (finish) at (0,1.15) {结束};

    \draw[->] (start.south) -- (input.north);
    \draw[->] (input.south) -- (test.north);

    % The two decision exits run outward and then downward into their own
    % process boxes; labels sit beside the exits, not inside the diamond.
    \draw[->] (test.west) -- (-2.65,5.65) -- (-2.65,4.55) -- (yes.north);
    \draw[->] (test.east) -- (2.65,5.65) -- (2.65,4.55) -- (no.north);
    \node[branchlabel] at (-2.70,5.92) {是};
    \node[branchlabel] at (2.70,5.92) {否};

    % Both result boxes feed one shared horizontal merge, then the independent
    % downward arrow enters 输出 S.
    \draw (yes.south) -- (-2.65,3.45) -- (0,3.45);
    \draw (no.south) -- (2.65,3.45) -- (0,3.45);
    \draw[->] (0,3.45) -- (output.north);
    \draw[->] (output.south) -- (finish.north);
\end{tikzpicture}
