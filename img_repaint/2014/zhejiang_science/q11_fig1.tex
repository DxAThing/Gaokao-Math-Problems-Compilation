% Nodes: 开始, 输入 n, S=0,i=1, S=2S+i, i=i+1, S>n?, 输出 i, 结束.
% Directed edges: 开始→输入 n→S=0,i=1→S=2S+i→i=i+1→S>n?;
% S>n? 是→输出 i→结束; S>n? 否→right loop→horizontal left arrow→the
% central connector before S=2S+i. No other connectors are present.
% All node text is explicitly zero-indented and centered; all segments are solid.

\begin{tikzpicture}[
    >=stealth,
    x=1cm,
    y=0.88cm,
    line width=0.45pt,
    line cap=round,
    line join=round,
    font=\normalsize,
    every node/.style={anchor=center, align=center,
        execute at begin node={\setlength{\parindent}{0pt}}},
    flowbox/.style={draw, anchor=center, minimum height=0.68cm, inner xsep=4pt,
        inner ysep=2pt, align=center,
        execute at begin node={\setlength{\parindent}{0pt}}},
    terminal/.style={flowbox, rounded corners=0.16cm, minimum width=1.35cm},
    io/.style={flowbox, trapezium, trapezium left angle=70, trapezium right angle=110,
        minimum width=1.85cm, minimum height=0.76cm},
    process/.style={flowbox, minimum width=2.65cm},
    decision/.style={flowbox, diamond, aspect=3.0, minimum width=3.25cm,
        minimum height=1.05cm},
]
    \path[use as bounding box] (-1.85,-0.42) rectangle (3.20,9.95);

    \node[terminal] (start) at (0,9.45) {开始};
    \node[io] (input) at (0,8.16) {输入 $n$};
    \node[process, minimum width=3.05cm] (init) at (0,6.86) {$S=0,i=1$};
    \node[process, minimum width=2.95cm] (update) at (0,5.52) {$S=2S+i$};
    \node[process, minimum width=2.45cm] (inc) at (0,4.14) {$i=i+1$};
    \node[decision] (check) at (0,2.72) {$S>n?$};
    \node[io, minimum width=1.75cm] (output) at (0,1.30) {输出 $i$};
    \node[terminal] (finish) at (0,0.02) {结束};

    % Main chain and the terminating branch.
    \draw[->] (start.south) -- (input.north);
    \draw[->] (input.south) -- (init.north);
    \draw[->] (init.south) -- (update.north);
    \draw[->] (update.south) -- (inc.north);
    \draw[->] (inc.south) -- (check.north);
    % The source places 是 on the left of the independent downward arrow.
    \draw[->] (check.south) -- node[left=2pt] {是} (output.north);
    \draw[->] (output.south) -- (finish.north);

    % S>n? 的否 branch returns by a left-pointing arrow to the update connector.
    \coordinate (loopRight) at (2.55,2.72);
    \coordinate (loopTop) at (2.55,6.14);
    \coordinate (loopJoin) at (0,6.14);
    \draw (check.east) -- node[above=2pt] {否} (loopRight) -- (loopTop);
    \draw[->] (loopTop) -- (loopJoin);
\end{tikzpicture}
