% Node-edge list: 开始 -> T=1,i=1 -> T=T/i -> i=i+1 -> i>5?;
% 是 -> 输出 T -> 结束; 否 -> right loop-back to the connector above T=T/i.
% Solid segments only; no other edges or dashed segments.

\begin{tikzpicture}[
  x=1cm,y=1cm,baseline,>=Stealth,
  line width=0.45pt,line cap=butt,line join=miter,
  every node/.style={anchor=center,align=center,font=\normalsize,
    execute at begin node={\setlength{\parindent}{0pt}\noindent}},
  flowbox/.style={draw,anchor=center,align=center,inner xsep=4pt,inner ysep=2.5pt,
    execute at begin node={\setlength{\parindent}{0pt}\noindent}},
  terminal/.style={flowbox,rounded corners=0.16cm,minimum width=1.35cm,minimum height=0.68cm},
  process/.style={flowbox,minimum height=0.68cm},
  decision/.style={flowbox,diamond,aspect=2.8,minimum height=0.92cm,inner xsep=4pt,inner ysep=2pt},
  io/.style={flowbox,trapezium,trapezium left angle=72,trapezium right angle=108,
    minimum height=0.74cm}
]
  \node[terminal] (start) at (0,9.55) {开始};
  \node[process,minimum width=2.85cm] (init) at (0,8.10) {$T=1,i=1$};
  \node[process,minimum width=1.95cm] (divide) at (0,6.80) {$T=\dfrac{T}{i}$};
  \node[process,minimum width=2.25cm] (increment) at (0,5.25) {$i=i+1$};
  \node[decision,minimum width=2.75cm] (test) at (0,3.80) {$i>5?$};
  \node[io,minimum width=2.15cm] (output) at (0,2.10) {输出 $T$};
  \node[terminal] (finish) at (0,0.60) {结束};

  % Main chain, with a single merge stem above the division box.
  \draw[->] (start.south) -- (init.north);
  \coordinate (merge) at (0,7.50);
  \draw (init.south) -- (merge);
  \draw[->] (merge) -- (divide.north);
  \draw[->] (divide.south) -- (increment.north);
  \draw[->] (increment.south) -- (test.north);
  \draw[->] (test.south) -- (output.north);
  \draw[->] (output.south) -- (finish.north);

  % 否 branch loops right to the shared connector; output remains separate.
  \draw (test.east) -- (2.65,3.80) -- (2.65,7.50);
  \draw[->] (2.65,7.50) -- (merge);

  \node[anchor=south west,inner xsep=1pt,inner ysep=1pt] at (0.28,2.95) {是};
  \node[anchor=south west,inner xsep=1pt,inner ysep=1pt] at (1.30,4.05) {否};
\end{tikzpicture}
