% Source PDF: source/普通高考/2012/2012湖南文.pdf, page 1; vector program flowchart.
% Grid evidence: tmp/review_2012_hunan_liberal/grid/page-1-grid100.jpg and
%   q14_original_clean2.png (no-grid crop from the source page).
% Elements: rounded start/end terminals; input rectangle; initialization
% rectangle i=1; update rectangle x=x-1; decision diamond x<1?; increment
% rectangle i=i+1; output parallelogram 输出 i; 是/否 labels.
% Complete node -> directed-edge list from the source:
%   start -> input -> init -> decrement -> test; test --是--> output -> finish;
%   test --否--> increment -> return to the main connector above decrement.
% The return arrow, main downward arrows, and output branch are independent;
% all connectors are solid and there are no other edges. Node text is centered
% with local zero paragraph indentation and compact padding.

\begin{tikzpicture}[
  >=Stealth,
  x=1cm, y=1.05cm,
  line width=0.45pt,
  line cap=round,
  line join=round,
  font=\normalsize,
  flowtext/.style={anchor=center, align=center, inner sep=0pt,
    execute at begin node={\setlength{\parindent}{0pt}}},
  every node/.style={flowtext},
  flowbox/.style={flowtext, draw, minimum height=0.68cm,
    inner xsep=3pt, inner ysep=2pt},
  terminal/.style={flowbox, rounded corners=0.18cm, minimum width=1.45cm},
  io/.style={flowbox, trapezium, trapezium left angle=72,
    trapezium right angle=72, minimum width=2.35cm, minimum height=0.76cm},
  decision/.style={flowbox, diamond, aspect=2.7, minimum width=3.55cm,
    minimum height=1.00cm},
]
  \path[use as bounding box] (-2.10,-0.45) rectangle (4.45,9.35);
  \node[terminal] (start) at (0,8.95) {开始};
  \node[flowbox, minimum width=2.20cm] (input) at (0,7.50) {输入 $x$};
  \node[flowbox, minimum width=1.75cm] (init) at (0,6.05) {$i=1$};
  \node[flowbox, minimum width=3.05cm] (decrement) at (0,4.55)
    {$x=x-1$};
  \node[decision] (test) at (0,3.00) {$x<1?$};
  \node[io, minimum width=2.25cm] (output) at (0,1.45) {输出 $i$};
  \node[terminal] (finish) at (0,0.05) {结束};
  \node[flowbox, minimum width=2.35cm] (increment) at (3.10,4.55)
    {$i=i+1$};

  % Main trunk and terminating branch.
  \draw[->] (start.south) -- (input.north);
  \draw[->] (input.south) -- (init.north);
  \draw[->] (init.south) -- (decrement.north);
  \draw[->] (decrement.south) -- (test.north);
  \draw[->] (test.south) -- node[flowtext, right] {是} (output.north);
  \draw[->] (output.south) -- (finish.north);

  % 否 branch increments i, then returns to the connector above x=x-1.
  \draw[->] (test.east) -- node[flowtext, above] {否}
    (3.10,3.00) -- (increment.south);
  % Lower the return rail so it clears the initialization box while still
  % entering the main stem at an interior point.
  \draw[->] (increment.north) -- (3.10,5.25) -- (0,5.25);
\end{tikzpicture}
