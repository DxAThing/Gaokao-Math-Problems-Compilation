% Elements: arrowed axes; x ticks 13.55, ..., 14.75; y ticks 0,.40,.50,.60,.65,1.30,1.55;

    % columns/vertical offsets so 0.40--0.65 stay distinct at page scale.
    yticklabels={,,,,,,},
    scaled y ticks=false,
    yticklabel style={/pgf/number format/fixed, /pgf/number format/precision=2},
    tick align=outside,
    tick label style={font=\small},
    yticklabel style={font=\scriptsize, /pgf/number format/fixed, /pgf/number format/precision=2},

    \node[anchor=south west,align=center,font=\small,inner sep=0pt]

      at (axis cs:6.15,0.09) {全球年平均气温/$^{\circ}$C};
