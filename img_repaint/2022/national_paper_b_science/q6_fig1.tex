\begin{tikzpicture}[x=1cm, y=1cm]
\tikzset{
  fzstart/.style={shape=rectangle, rounded corners=1.5mm, draw=black, line width=0.6pt, minimum width=26mm, minimum height=10mm, inner xsep=3mm, inner ysep=1.5mm, align=center, execute at begin node={\parindent=0pt}},
  fzio/.style={trapezium, trapezium left angle=70, trapezium right angle=110, draw=black, line width=0.6pt, inner sep=1.5mm, align=center, execute at begin node={\parindent=0pt}},
  fzproc/.style={rectangle, draw=black, line width=0.6pt, inner sep=1.5mm, align=center, execute at begin node={\parindent=0pt}},
  fzdia/.style={diamond, draw=black, line width=0.6pt, inner sep=1.5mm, align=center, execute at begin node={\parindent=0pt}},
  fzedge/.style={draw=black, line width=0.6pt, -{Stealth[length=2.4mm]}},
  fzlab/.style={font=\normalsize, inner sep=1mm}
}
% 节点清单: 开始 / 输入 a=1,b=1,n=1 / b=b+2a / a=b-a,n=n+1 /
% 菱形 |b^2/a^2-2|<0.01 / 输出 n / 结束. 共7节点.
% 有向边: 开始->输入, 输入->处理1(途经汇合J1), 处理1->处理2, 处理2->菱形,
% 菱形西->回环->J1(否, 箭头在J1), 菱形南->输出(是), 输出->结束. 共7边.
\node[fzstart] (start) at (0,0) {开始};
\node[fzio, minimum width=58mm, minimum height=10mm] (input) at (0,-1.8) {输入 $a=1$, $b=1$, $n=1$};
\coordinate (J1) at (0,-3.3);
\node[fzproc, minimum width=32mm, minimum height=9mm] (proc1) at (0,-4.5) {$b=b+2a$};
\node[fzproc, minimum width=48mm, minimum height=9mm] (proc2) at (0,-5.9) {$a=b-a$, $n=n+1$};
\node[fzdia, minimum width=62mm, minimum height=32mm] (dia) at (0,-9.0) {$\left|\dfrac{b^{2}}{a^{2}}-2\right|<0.01$};
\node[fzio, minimum width=30mm, minimum height=10mm] (output) at (0,-12.2) {输出 $n$};
\node[fzstart] (endnode) at (0,-13.9) {结束};
\draw[fzedge] (start.south) -- (input.north);
\draw[fzedge] (input.south) -- (proc1.north);
\draw[fzedge] (proc1.south) -- (proc2.north);
\draw[fzedge] (proc2.south) -- (dia.north);
\draw[fzedge] (dia.west) -- node[fzlab, above=1.5mm, pos=0.25]{否} (-5.2,-9.0) -- (-5.2,-3.3) -- (J1);
\draw[fzedge] (dia.south) -- node[fzlab, right=1.5mm, pos=0.45]{是} (output.north);
\draw[fzedge] (output.south) -- (endnode.north);
\end{tikzpicture}
