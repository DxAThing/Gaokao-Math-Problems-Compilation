% Complete node -> directed-edge list from the source:
%   start -> input -> update -> test; test --是--> output -> finish;
%   test --否--> n-increment -> input/update entry.  The left return arrow
%   joins the unique input-to-update vertical stem; no loop-to-output edge exists.
% solid segments: all node outlines and connectors; dashed segments: none.
% Labels are centered in nodes; 是/否 sit beside their outgoing branches.

\begin{tikzpicture}[
  >=Stealth,
  x=1cm,y=1cm,
  line width=0.50pt,
  line join=round,
  line cap=round,
  flowtext/.style={font=\normalsize,anchor=center,align=center,
                   execute at begin node={\setlength{\parindent}{0pt}}},
  flowbox/.style={flowtext,draw,inner xsep=4pt,inner ysep=2pt},
  terminator/.style={flowbox,rounded corners=0.18cm,
                     minimum width=1.22cm,minimum height=0.72cm},
  io/.style={flowbox,trapezium,trapezium left angle=78,trapezium right angle=78,
             minimum height=0.72cm},
  process/.style={flowbox},
  decision/.style={flowbox,diamond,aspect=4.0,inner xsep=4pt,inner ysep=2pt},
]
  \path[use as bounding box] (0,0) rectangle (8,9);

  \node[terminator] (start) at (5.50,8.22) {开始};
  \node[io,minimum width=3.20cm] (input) at (5.50,6.88) {输入 \(x,y,n\)};
  \node[process,minimum width=4.73cm,minimum height=1.20cm] (update)
      at (5.50,5.29) {\(x=x+\frac{n-1}{2},y=ny\)};
  \node[process,minimum width=2.23cm,minimum height=0.63cm] (increment)
      at (1.42,5.29) {\(n=n+1\)};
  \node[decision,minimum width=4.92cm,minimum height=1.22cm] (test)
      at (5.50,3.45) {\(x^2+y^2\geq36\)};
  \node[io,minimum width=2.66cm] (output) at (5.50,1.85) {输出 \(x,y\)};
  \node[terminator] (finish) at (5.50,0.49) {结束};

  % Main path and the true branch.
  \draw[->] (start.south) -- (input.north);
  \draw[->] (input.south) -- (update.north);
  \draw[->] (update.south) -- (test.north);
  \draw[->] (test.south) -- (output.north);
  \draw[->] (output.south) -- (finish.north);

  % False branch: test -> n increment, then back to the shared input/update stem.
  \draw[->] (test.west) -- (1.42,3.45) -- (increment.south);
  \draw[->] (increment.north) -- (1.42,6.20) -- (5.50,6.20);

  \node[flowtext,inner sep=0pt,anchor=north] at (2.83,3.38) {否};
  \node[flowtext,inner sep=0pt,anchor=west] at (4.86,2.78) {是};
\end{tikzpicture}
