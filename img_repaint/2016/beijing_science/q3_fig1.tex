% Original PDF: source/普通高考/2016/2016北京理.pdf, page 1.
% pdfimages -list found no embedded bitmap; the flowchart is vector/text artwork.
% Grid evidence: tmp/redraw_flowchart_2016_beijing_science/grid/q3_block2_grid.jpg;
% comparison source is the no-grid crop q3_original.png from that page.
% Elements: rounded start/end nodes, input/output parallelograms, three rectangles,
% one decision diamond, solid directed connectors, and 是/否 labels.
% Directed edges from the source: start->input->init->update->test;
% test --是--> output->finish; test --否--> increment->update (loop return).
% solid segments: all box borders, trunk arrows, loop branch, and arrowheads;
% dashed segments: none. Labels are centered in boxes with zero paragraph indent.
\begin{tikzpicture}[
  >=stealth,
  line width=0.5pt,
  line join=round,
  line cap=round,
  font=\large,
  flowtext/.style={anchor=center,align=center,inner sep=0pt,
    execute at begin node={\setlength{\parindent}{0pt}}},
  every node/.style={flowtext},
  flowbox/.style={draw,line width=0.5pt,minimum height=0.68cm,
    inner xsep=4pt,inner ysep=2pt},
  io/.style={flowbox,shape=trapezium,
    trapezium left angle=75,trapezium right angle=105,
    minimum height=0.78cm},
  flowarrow/.style={->,line width=0.5pt},
]
  % Main trunk nodes.
  \node[flowbox,rounded corners=0.18cm,minimum width=1.35cm]
    (start) at (0,8.0) {开始};
  \node[io,minimum width=2.55cm] (input) at (0,6.75) {输入 $a$};
  \node[flowbox,minimum width=3.15cm] (init) at (0,5.48) {$k=0,b=a$};
  \node[flowbox,minimum width=2.45cm,minimum height=1.05cm]
    (update) at (0,3.95) {$a=\displaystyle\frac{-1}{1+a}$};
  \node[flowbox,shape=diamond,aspect=2.7,minimum width=4.05cm,
    minimum height=0.95cm] (test) at (0,2.35) {$a=b$};
  \node[io,minimum width=2.55cm] (output) at (0,0.90) {输出 $k$};
  \node[flowbox,rounded corners=0.18cm,minimum width=1.35cm]
    (finish) at (0,-0.40) {结束};

  % No branch increments k before returning to the update node.
  \node[flowbox,minimum width=2.25cm] (increment) at (2.85,3.95)
    {$k=k+1$};

  % Main vertical path and the yes branch.
  \draw[flowarrow] (start.south) -- (input.north);
  \draw[flowarrow] (input.south) -- (init.north);
  \draw[flowarrow] (init.south) -- (update.north);
  \draw[flowarrow] (update.south) -- (test.north);
  \draw[flowarrow] (test.south) -- node[flowtext, right] {是} (output.north);
  \draw[flowarrow] (0,0.51) -- (finish.north);

  % No branch: test -> increment -> return arrow into the trunk above update.
  \draw[flowarrow] (test.east) -- node[flowtext, above] {否} (2.85,2.35) -- (2.85,3.56);
  \draw[flowarrow] (increment.north) -- (2.85,4.82) -- (0,4.82);
\end{tikzpicture}
