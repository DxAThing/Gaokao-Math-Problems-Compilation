\begin{tikzpicture}[x=1cm,y=1cm,>=Stealth]
\tikzset{
  flowStart/.style={rectangle, rounded corners=1.5mm, draw, line width=0.6pt, minimum width=26mm, minimum height=10mm, inner xsep=3mm, inner ysep=1.5mm, align=center, execute at begin node={\setlength{\parindent}{0pt}\centering}},
  flowIO/.style={trapezium, trapezium left angle=75, trapezium right angle=105, draw, line width=0.6pt, minimum width=30mm, minimum height=10mm, inner sep=2mm, align=center, execute at begin node={\setlength{\parindent}{0pt}\centering}},
  flowProc/.style={rectangle, draw, line width=0.6pt, minimum width=28mm, minimum height=10mm, inner sep=2mm, align=center, execute at begin node={\setlength{\parindent}{0pt}\centering}},
  flowDec/.style={diamond, aspect=2.4, draw, line width=0.6pt, minimum width=36mm, minimum height=15mm, inner sep=1mm, align=center, execute at begin node={\setlength{\parindent}{0pt}\centering}},
  flowEdge/.style={->, >=Stealth, line width=0.6pt},
  flowPlain/.style={-, line width=0.6pt},
  flowLabel/.style={font=\normalsize, inner sep=1pt}
}
\node[flowStart] (start) at (2.0,0) {开始};
\node[flowIO] (input) at (2.0,-1.9) {输入 $a,b$};
\coordinate (J) at (2.0,-3.3);
\node[flowDec] (dec1) at (2.0,-4.5) {$a\ne b$};
\node[flowDec] (dec2) at (-0.5,-6.9) {$a>b$};
\node[flowIO] (out) at (4.8,-6.9) {输出 $a$};
\node[flowProc] (abox) at (-2.9,-9.1) {$a=a-b$};
\node[flowProc] (bbox) at (1.8,-9.1) {$b=b-a$};
\coordinate (C) at (-0.5,-10.4);
\node[flowStart] (end) at (4.8,-8.9) {结束};
\draw[flowEdge] (start.south) -- (input.north);
\draw[flowEdge] (input.south) -- (dec1.north);
\draw[flowEdge] (dec1.west) -| node[flowLabel, above=2mm, pos=0.2] {是} (dec2.north);
\draw[flowEdge] (dec1.east) -| node[flowLabel, above=2mm, pos=0.2] {否} (out.north);
\draw[flowEdge] (dec2.west) -| node[flowLabel, above=2mm, pos=0.2] {是} (abox.north);
\draw[flowEdge] (dec2.east) -| node[flowLabel, above=2mm, pos=0.2] {否} (bbox.north);
\draw[flowEdge] (out.south) -- (end.north);
% labels attached to first horizontal segments above (see draws above)
% 底合流：两更新框无箭头汇入真实共享点C，共享主干C至J单独绘制并仅在J收箭头
\draw[flowPlain] (abox.south) |- (C);
\draw[flowPlain] (bbox.south) |- (C);
\draw[flowEdge] (C) -- (-0.5,-11.0) -- (-5.4,-11.0) -- (-5.4,-3.3) -- (J);
\end{tikzpicture}
