% nodes: 开始 / n=1,T=1 / n<3 / T=T+∫x^n dx / n=n+1 / 输出T / 结束 (7)
% edges: 开始->init; init->diamond; diamond->Tbox(是,下); Tbox->nupd; nupd->J(回环);
%   diamond->输出T(否,右); 输出T->结束
\begin{tikzpicture}[>=Stealth, line width=0.6pt,
  every node/.append style={execute at begin node={\setlength{\parindent}{0pt}}},
  startstop/.style={draw, shape=rectangle, rounded corners=1.5mm, minimum width=26mm, minimum height=10mm, inner xsep=3mm, inner ysep=1.5mm, align=center},
  proc/.style={draw, rectangle, minimum width=3.4cm, minimum height=1.0cm, align=center, inner sep=5pt},
  io/.style={draw, trapezium, trapezium left angle=70, trapezium right angle=110, minimum width=3.4cm, minimum height=1.0cm, align=center, inner sep=5pt},
  dec/.style={draw, diamond, minimum width=3.6cm, minimum height=1.8cm, align=center, inner sep=4pt}]
  \node[startstop] (START) at (0,0) {开始};
  \node[proc, minimum width=3.8cm] (INIT) at (0,-2.0) {$n=1,T=1$};
  \node[dec] (DIA) at (0,-4.4) {$n<3$};
  \node[proc, minimum width=5.2cm, minimum height=1.3cm] (TBOX) at (0,-6.7) {$T=T+\int_{0}^{1}x^{n}\mathrm{d}x$};
  \node[proc] (NUP) at (0,-8.9) {$n=n+1$};
  \node[io] (OUT) at (5.2,-6.7) {输出 $T$};
  \node[startstop] (END) at (5.2,-8.9) {结束};
  \coordinate (J) at (0,-3.0);
  \draw[->] (START.south) -- (INIT.north);
  \draw[->] (INIT.south) -- (DIA.north);
  \draw[->] (DIA.south) -- node[midway, right=3pt] {是} (TBOX.north);
  \draw[->] (TBOX.south) -- (NUP.north);
  \draw[->] (DIA.east) -- node[above=3pt, pos=0.15] {否} (5.2,-4.4) -- (OUT.north);
  \draw[->] (OUT.south) -- (END.north);
  \draw[->] (NUP.west) -- (-3.8,-8.9) -- (-3.8,-3.0) -- (J);
\end{tikzpicture}
