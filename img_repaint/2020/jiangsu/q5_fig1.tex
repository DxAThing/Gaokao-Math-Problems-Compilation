\begin{tikzpicture}[>=Stealth, line width=0.6pt,
  every node/.append style={execute at begin node={\parindent=0pt}},
  start/.style={rectangle, draw, rounded corners=1.5mm, minimum width=26mm, minimum height=10mm, inner xsep=3mm, inner ysep=1.5mm, align=center},
  proc/.style={rectangle, draw, minimum width=30mm, minimum height=11mm, inner sep=4pt, align=center},
  dec/.style={diamond, draw, aspect=2.4, inner sep=4pt, align=center},
  io/.style={trapezium, draw, trapezium left angle=70, trapezium right angle=110, minimum width=26mm, minimum height=11mm, inner sep=4pt, align=center}
]
\node[start] (start) at (0,0) {开始};
\node[io] (inX) at (0,-2.0) {输入$x$};
\node[dec] (dec) at (0,-4.1) {$x>0?$};
\node[proc] (pY) at (-3.9,-6.8) {$y\gets 2^{x}$};
\node[proc] (pN) at (3.9,-6.8) {$y\gets x+1$};

\node[proc] (pOut) at (0,-10.4) {输出$y$};

\draw[->] (inX.south) -- (dec.north);
\draw[->] (dec.west) -- node[pos=0.2,above=0.7mm] {Y} (-3.9,-4.1) -- (pY.north);
\draw[->] (dec.east) -- node[pos=0.2,above=0.7mm] {N} (3.9,-4.1) -- (pN.north);
\draw (pY.south) -- (-3.9,-9.2) -- (Jm);
\draw (pN.south) -- (3.9,-9.2) -- (Jm);
\draw[->] (Jm) -- (pOut.north);
\draw[->] (pOut.south) -- (end.north);
