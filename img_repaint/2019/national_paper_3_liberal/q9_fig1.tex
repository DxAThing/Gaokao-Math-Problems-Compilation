% elements: start/end rounded terminals; input/output parallelograms; process boxes
%           x=1,s=0, s=s+x, x=x/2; decision diamond x<epsilon; loop-back arrow;
%           Chinese labels 开始、输入 epsilon、输出 s、结束、是、否.
% relations: the main flow is top-to-bottom; 否 returns from the decision to the
%            s=s+x step, while 是 continues to 输出 s and then 结束.
% node -> directed edge list from the source: 开始->输入ε, 输入ε->(x=1,s=0),
% (x=1,s=0)->(s=s+x), (s=s+x)->(x=x/2), (x=x/2)->(x<ε),
% (x<ε)-否->(s=s+x), (x<ε)-是->输出s, 输出s->结束.
% solid segments: all box outlines and connectors; dashed segments: none.
% labels: formula labels centered in boxes; 是 below the decision and 否 at the
%         upper-right loop branch, with arrows preserving the original topology.
\begin{tikzpicture}[x=1cm,y=1cm,>=stealth,line width=0.55pt,
  line cap=round,line join=round,
  every node/.append style={anchor=center,align=center,
    execute at begin node=\setlength{\parindent}{0pt}}]
  % Main vertical flow.
  \node[draw,rounded corners=0.18cm,minimum width=1.25cm,minimum height=0.68cm,inner xsep=4pt,inner ysep=2pt] (start) at (0,10.8) {开始};
  \node[draw,shape=trapezium,trapezium left angle=72,trapezium right angle=108,
    minimum width=0.95cm,minimum height=0.76cm,inner xsep=4pt,inner ysep=2pt] (input) at (0,9.58) {输入 $\varepsilon$};
  \node[draw,minimum width=2.55cm,minimum height=0.72cm,inner xsep=4pt,inner ysep=2pt] (init) at (0,8.36) {$x=1,s=0$};
  \node[draw,minimum width=2.12cm,minimum height=0.72cm,inner xsep=4pt,inner ysep=2pt] (sum) at (0,7.18) {$s=s+x$};
  \node[draw,minimum width=1.55cm,minimum height=1.02cm,inner xsep=4pt,inner ysep=2pt] (half) at (0,5.88) {$x=\dfrac{x}{2}$};
  \node[draw,diamond,aspect=2.7,minimum width=2.62cm,minimum height=0.74cm,inner xsep=4pt,inner ysep=2pt] (test) at (0,4.53) {$x<\varepsilon$};
  \node[draw,shape=trapezium,trapezium left angle=72,trapezium right angle=108,
    minimum width=0.95cm,minimum height=0.76cm,inner xsep=4pt,inner ysep=2pt] (output) at (0,3.28) {输出 $s$};
  \node[draw,rounded corners=0.18cm,minimum width=1.25cm,minimum height=0.68cm,inner xsep=4pt,inner ysep=2pt] (finish) at (0,2.08) {结束};

  \draw[->] (start.south) -- (input.north);
  \draw[->] (input.south) -- (init.north);
  \draw[->] (init.south) -- (sum.north);
  \draw[->] (sum.south) -- (half.north);
  \draw[->] (half.south) -- (test.north);
  \draw[->] (test.south) -- (output.north);
  \draw[->] (output.south) -- (finish.north);

  % 否 branch loops back to the accumulation step.
  \draw[->] (test.east) -- (2.02,4.53) -- (2.02,7.72) -- (0.00,7.72);
  \node[anchor=west,inner sep=0pt] at (1.56,5.08) {否};
  \node[anchor=west,inner sep=0pt] at (0.26,3.98) {是};
\end{tikzpicture}
