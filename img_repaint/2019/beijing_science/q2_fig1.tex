  flowbox/.style={draw,align=center,anchor=center,inner sep=1pt,

  % Start and initialisation nodes.
  \node[draw,rounded corners=5pt,minimum width=1.05cm,minimum height=0.60cm,
        inner sep=1pt] (start) at (0,7.25) {开始};
  \node[flowbox,minimum width=2.35cm,minimum height=0.62cm]
        (init) at (0,6.00) {$k=1,s=1$};

  % Update and increment nodes.
  \node[flowbox,minimum width=2.05cm,minimum height=1.15cm]
        (update) at (0,4.32) {$s=\dfrac{2s^2}{3s-2}$};
  \node[flowbox,minimum width=1.85cm,minimum height=0.62cm]
        (inc) at (2.35,4.32) {$k=k+1$};

  \node[flowbox,shape=diamond,aspect=2.9,minimum width=2.90cm,

  \node[io,minimum width=1.75cm] (output) at (0,1.43) {输出 $s$};
  \node[draw,rounded corners=5pt,minimum width=1.05cm,minimum height=0.60cm,
        inner sep=1pt] (end) at (0,0.15) {结束};

  % Main downward path.
  \draw[->] (start.south) -- (init.north);
  \draw (init.south) -- (0,5.25);
  \draw[->] (0,5.25) -- (update.north);
  \draw[->] (update.south) -- (decision.north);
  \draw[->] (decision.south) -- node[right] {是} (output.north);
  \draw[->] (output.south) -- (end.north);

  % No branch: diamond -> increment -> return to update.
  \draw (decision.east) -- (2.35,3.03);
  \draw[->] (2.35,3.03) -- (inc.south);
  \node[anchor=east] at (2.15,3.20) {否};
  % The return arrow turns above the update box and terminates at the visible
  % trunk junction; the trunk's own arrow continues into update.
  \draw (inc.north) -- (2.35,5.25) -- (0,5.25);
