\section{单选题}

\begin{problem}
直线 \( y = x + 1 \) 与圆 \( x^{2} + y^{2} = 1 \) 的位置关系是（\quad）

\choices
    {相切}
    {相交但直线不过圆心}
    {直线过圆心}
    {相离}
\end{problem}

\begin{problem}

    {\(2 + \i\)}
    {\(-2 - \i\)}
    {\(-2 + \i\)}
\end{problem}

\begin{problem}

\begin{problem}

\begin{problem}

\section{解答题}

\begin{problem}
设函数 \(f(x) = \sin \left(\frac{\pi}{4}x - \frac{\pi}{6}\right) - 2\cos^{2} \frac{\pi}{8}x + 1\).

\begin{enumerate}
\item 求 \(f(x)\) 的最小正周期；
\item 若函数 \(y = g(x)\) 与 \(y = f(x)\) 的图象关于直线 \(x = 1\) 对称，求当 \(x \in \left[0, \frac{4}{3}\right]\) 时 \(y = g(x)\) 的最大值.
\end{enumerate}
\end{problem}

\begin{problem}
某单位为绿化环境，移栽了甲、乙两种大树各 2 株. 设甲、乙两种大树移栽的成活率分别为 \(\frac{2}{3}\) 和 \(\frac{1}{2}\)，且各株大树是否成活互不影响. 求移栽的 4 株大树中：

\begin{enumerate}
\item 两种大树各成活 1 株的概率；
\item 成活的株数 \(\xi\) 的分布列与期望.
\end{enumerate}
\end{problem}

\begin{problem}
设函数 \(f(x) = ax^{2} + bx + k\ (k > 0)\) 在 \(x = 0\) 处取得极值，且曲线 \(y = f(x)\) 在点 \((1, f(1))\) 处的切线垂直于直线 \(x + 2y + 1 = 0\).

\begin{enumerate}
\item 求 \(a, b\) 的值；
\item 若函数 \(g(x) = \frac{\e^{x}}{f(x)}\)，讨论 \(g(x)\) 的单调性.
\end{enumerate}
\end{problem}

\begin{problem}
设 \(m\) 个不全相等的正数 \(a_1, a_2, \cdots, a_m\ (m \geqslant 7)\) 依次围成一个圆圈.

\begin{enumerate}
    \item 若 \(m = 2009\)，且 \(a_1, a_2, \cdots, a_{1005}\) 是公差为 \(d\) 的等差数列，而 \(a_1, a_{2009}, a_{2008}, \cdots, a_{1006}\) 是公比为 \(q\) 的等比数列；数列 \(a_1, a_2, \cdots, a_m\) 的前 \(n\) 项和 \(S_n\)（\(n \leqslant m\)）满足：\(S_3 = 15\)，\(S_{2009} = S_{2007} + 12a_1\)，求通项 \(a_n\)（\(n \leqslant m\)）；
    \item 若每个数 \(a_n\)（\(n \leqslant m\)）是其左右相邻两数平方的等比中项，求证：\(a_1 + \cdots + a_6 + a_7^{2} + \cdots + a_m^{2} > m a_1 a_2 \cdots a_m\).
\end{enumerate}
\end{problem}
