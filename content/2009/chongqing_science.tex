\section{单选题}

\begin{problem}
直线 \( y = x + 1 \) 与圆 \( x^{2} + y^{2} = 1 \) 的位置关系是（\quad）

\choices
    {相切}
    {相交但直线不过圆心}
    {直线过圆心}
    {相离}

\begin{answer}
B
\end{answer}

\begin{solution}
将直线化为一般式 \(x-y+1=0\). 原点到该直线的距离为
\[
\frac{|1|}{\sqrt{1^{2}+(-1)^{2}}}=\frac{1}{\sqrt{2}}<1
\]
所以直线与圆有两个交点；又原点不满足直线方程，故直线不过圆心，选 B.
\end{solution}
\end{problem}

\begin{problem}
已知复数 \(z\) 的实部为 \(-1\)，虚部为 \(2\)，则 \(\frac{5\i}{z}=\)（\quad）

\choices
    {\(2 - \i\)}
    {\(2 + \i\)}
    {\(-2 - \i\)}
    {\(-2 + \i\)}

\begin{answer}
A
\end{answer}

\begin{solution}
由实部和虚部可知 \(z=-1+2\i\). 分母有理化得
\[
\frac{5\i}{z}=\frac{5\i(-1-2\i)}{(-1)^{2}+2^{2}}
=\frac{10-5\i}{5}=2-\i
\]
因此选 A.
\end{solution}
\end{problem}

\begin{problem}
\(\left(x^{2} + \frac{2}{x}\right)^{8}\) 的展开式中 \(x^{4}\) 的系数是（\quad）

\choices
    {16}
    {70}
    {560}
    {1120}

\begin{answer}
D
\end{answer}

\begin{solution}
展开式的通项为从 8 个因式中选 \(k\) 个取 \(2/x\)，其余取 \(x^{2}\)，即
\[
\mathrm C_{8}^{k}(x^{2})^{8-k}\left(\frac{2}{x}\right)^{k}
=\mathrm C_{8}^{k}2^{k}x^{16-3k}
\]
要得到 \(x^{4}\)，必须有 \(16-3k=4\)，故 \(k=4\). 所求系数为
\[
\mathrm C_{8}^{4}2^{4}=70\times16=1120
\]
因此选 D.
\end{solution}
\end{problem}

\begin{problem}
已知 \(|\bs{a}| = 1\)，\(|\bs{b}| = 6\)，\(\bs{a} \cdot (\bs{b} - \bs{a}) = 2\)，则向量 \(\bs{a}\) 与 \(\bs{b}\) 的夹角是（\quad）

\choices
    {\(\frac{\pi}{6}\)}
    {\(\frac{\pi}{4}\)}
    {\(\frac{\pi}{3}\)}
    {\(\frac{\pi}{2}\)}

\begin{answer}
C
\end{answer}

\begin{solution}
由条件
\[
\bs a\cdot(\bs b-\bs a)=\bs a\cdot\bs b-|\bs a|^{2}=2
\]
得 \(\bs a\cdot\bs b=3\). 设 \(\bs a\) 与 \(\bs b\) 的夹角为 \(\theta\)，则
\[
\cos\theta=\frac{\bs a\cdot\bs b}{|\bs a||\bs b|}
=\frac{3}{1\times6}=\frac12
\]
由于 \(0\leqslant\theta\leqslant\pi\)，所以 \(\theta=\pi/3\)，选 C.
\end{solution}
\end{problem}

\begin{problem}
不等式 \( |x + 3| - |x - 1| \leqslant a^{2} - 3a \) 对任意实数 \(x\) 恒成立，则实数 \(a\) 的取值范围为（\quad）

\choices
    {\((-\infty, -1] \cup [4, +\infty)\)}
    {\((-\infty, -2] \cup [5, +\infty)\)}
    {\([1,2]\)}
    {\((-\infty, 1] \cup [2, +\infty)\)}

\begin{answer}
A
\end{answer}

\begin{solution}
分段计算左端：
\[
|x+3|-|x-1|=
\begin{cases}
-4,&x\leqslant-3,\\
2x+2,&-3\leqslant x\leqslant1,\\
4,&x\geqslant1.
\end{cases}
\]
因此其最大值为 \(4\). 要使不等式对任意实数 \(x\) 恒成立，充要条件是
\[
a^{2}-3a\geqslant4
\]
即
\[
(a-4)(a+1)\geqslant0
\]
所以 \(a\leqslant-1\) 或 \(a\geqslant4\)，选 A.
\end{solution}
\end{problem}

\begin{problem}
锅中煮有芝麻馅汤圆 6 个，花生馅汤圆 5 个，豆沙馅汤圆 4 个，这三种汤圆的外部特征完全相同. 从中任意舀取 4 个汤圆，则每种汤圆都至少取到 1 个的概率为（\quad）

\choices
    {\(\frac{8}{91}\)}
    {\(\frac{25}{91}\)}
    {\(\frac{48}{91}\)}
    {\(\frac{60}{91}\)}

\begin{answer}
C
\end{answer}

\begin{solution}
4 个汤圆从 15 个中任取，样本数为 \(\mathrm C_{15}^{4}\). 每种馅至少取到 1 个时，取法只能是数量分配 \((2,1,1)\). 因而有利取法数为
\[
 \mathrm C_{6}^{2}\mathrm C_{5}^{1}\mathrm C_{4}^{1}
 +\mathrm C_{6}^{1}\mathrm C_{5}^{2}\mathrm C_{4}^{1}
 +\mathrm C_{6}^{1}\mathrm C_{5}^{1}\mathrm C_{4}^{2}
 =300+240+180=720
\]
所求概率为
\[
\frac{720}{\mathrm C_{15}^{4}}=\frac{720}{1365}=\frac{48}{91}
\]
故选 C.
\end{solution}
\end{problem}

\begin{problem}
设 \(\triangle ABC\) 的三个内角 \(A\)，\(B\)，\(C\)，向量 \(\bs{m} = (\sqrt{3} \sin A, \sin B)\)，\(\bs{n} = (\cos B, \sqrt{3} \cos A)\)，若 \(\bs{m} \cdot \bs{n} = 1 + \cos (A + B)\)，则 \(C =\)（\quad）

\choices
    {\(\frac{\pi}{6}\)}
    {\(\frac{\pi}{3}\)}
    {\(\frac{2\pi}{3}\)}
    {\(\frac{5\pi}{6}\)}

\begin{answer}
C
\end{answer}

\begin{solution}
由向量内积得
\[
\bs m\cdot\bs n
=\sqrt{3}\sin A\cos B+\sqrt{3}\sin B\cos A
=\sqrt{3}\sin(A+B)
\]
设 \(\theta=A+B=\pi-C\)，则 \(0<\theta<\pi\)，且题设化为
\[
\sqrt{3}\sin\theta=1+\cos\theta
\]
用半角公式，因 \(0<\theta<\pi\)，有 \(\sin(\theta/2)>0\) 且 \(\cos(\theta/2)>0\)，从而
\[
2\sqrt{3}\sin\frac\theta2\cos\frac\theta2
=2\cos^{2}\frac\theta2
\]
故 \(\tan(\theta/2)=1/\sqrt{3}\). 于是 \(\theta=\pi/3\)，所以
\[
C=\pi-\theta=\frac{2\pi}{3}
\]
故选 C.
\end{solution}
\end{problem}

\begin{problem}
已知 \(\displaystyle\lim_{x\to\infty}\left(\frac{2x^2}{x + 1} -ax - b\right) = 2\)，其中 \(a\)，\(b\in \R\)，则 \(a - b\) 的值为（\quad）

\choices
    {\(-6\)}
    {\(-2\)}
    {2}
    {6}

\begin{answer}
D
\end{answer}

\begin{solution}
作除法可得
\[
\frac{2x^{2}}{x+1}=2x-2+\frac{2}{x+1}
\]
因此
\[
\frac{2x^{2}}{x+1}-ax-b
=(2-a)x-(2+b)+\frac{2}{x+1}
\]
极限为有限值，必须有 \(a=2\). 此时极限为 \(-2-b\)，由题设
\[
-2-b=2
\]
得 \(b=-4\). 所以 \(a-b=2-(-4)=6\)，选 D.
\end{solution}
\end{problem}

\begin{problem}
已知二面角 \(\alpha - l - \beta\) 的大小为 \(50^{\circ}\)，\(P\) 为空间中任意一点，则过点 \(P\) 且与平面 \(\alpha\) 和平面 \(\beta\) 所成的角都是 \(25^{\circ}\) 的直线的条数为（\quad）

\choices
    {2}
    {3}
    {4}
    {5}

\begin{answer}
B
\end{answer}

\begin{solution}
以交线 \(l\) 为 \(z\) 轴，在垂直于 \(l\) 的截面内取平面 \(\alpha\) 的方向为方位角 \(0\)，平面 \(\beta\) 的方向为方位角 \(50^{\circ}\). 设所求直线的一个单位方向向量为
\(\bs u=(\rho\cos\psi,\rho\sin\psi,\eta)\)，\(\rho\geqslant0\)，\(\rho^{2}+\eta^{2}=1\)
其中 \(\psi\) 按 \(180^{\circ}\) 计. 直线与平面的夹角等于方向向量与平面法向量夹角的余角，因此
\[
\sin25^{\circ}=\rho|\sin\psi|
\quad\text{且}\quad
\sin25^{\circ}=\rho|\sin(\psi-50^{\circ})|
\]
于是
\[
|\sin\psi|=|\sin(\psi-50^{\circ})|
\]
从而 \(\psi=25^{\circ}\) 或 \(\psi=115^{\circ}\)（按 \(180^{\circ}\) 计）.

当 \(\psi=25^{\circ}\) 时，\(\rho=1\)，所以 \(\eta=0\)，得到 1 条直线；当 \(\psi=115^{\circ}\) 时，
\[
\rho=\frac{\sin25^{\circ}}{\sin115^{\circ}}
=\frac{\sin25^{\circ}}{\sin65^{\circ}}<1
\]
此时 \(\eta=\pm\sqrt{1-\rho^{2}}\)，得到 2 条不同的直线. 因此直线共有 \(1+2=3\) 条，选 B.
\end{solution}
\end{problem}

\begin{problem}
已知以 \(T = 4\) 为周期的函数 \(f(x) = \begin{cases}
m\sqrt{1 - x^{2}}, & x \in (-1, 1],\\
1 - |x - 2|, & x \in (1, 3].
\end{cases}\) 其中 \(m > 0\). 若方程 \(3f(x) = x\) 恰有 5 个实数解，则 \(m\) 的取值范围为（\quad）

\choices
    {\(\left(\frac{\sqrt{15}}{3}, \frac{8}{3}\right)\)}
    {\(\left(\frac{\sqrt{15}}{3}, \sqrt{7}\right)\)}
    {\(\left(\frac{4}{3}, \frac{8}{3}\right)\)}
    {\(\left(\frac{4}{3}, \sqrt{7}\right)\)}

\begin{answer}
B
\end{answer}

\begin{solution}
函数以 4 为周期，先考察一个基本周期 \((-1,3]\). 在 \([0,1]\) 上，方程为
\[
3m\sqrt{1-x^{2}}=x
\]
由于 \(f(x)\geqslant0\)，方程的解必须满足 \(x\geqslant0\)，且 \(x=0\) 不满足方程，所以不必考察负半轴.
左端随 \(x\) 增大而减小、右端递增，且两端在 \(x=0\) 与 \(x=1\) 的大小关系相反，故恰有 1 根.

在 \((1,3]\) 上，\(f(x)=1-|x-2|\). 当 \(1<x\leqslant2\) 时，方程为 \(3(x-1)=x\)，根为 \(x=3/2\)；当 \(2\leqslant x\leqslant3\) 时，方程为 \(3(3-x)=x\)，根为 \(x=9/4\). 所以基本周期内已有 3 根.

对 \(k\geqslant1\)，在区间 \((4k-1,4k+1]\) 上令 \(x=4k+t\)，则第一段方程化为
\(m=\frac{4k+t}{3\sqrt{1-t^{2}}}\)，\(-1<t<1\).
记
\[
h_k(t)=\frac{4k+t}{3\sqrt{1-t^{2}}}
\]
其导数为
\[
h_k'(t)=\frac{1+4kt}{3(1-t^{2})^{3/2}}
\]
所以 \(h_k\) 在 \((-1,-1/(4k))\) 上递减，在 \((-1/(4k),1)\) 上递增，并且两端趋于 \(+\infty\). 唯一最小值为
\[
h_k\left(-\frac1{4k}\right)=\frac{\sqrt{16k^{2}-1}}{3}
\]
因此第 \(k\) 个平移后的第一段有 2 个根当且仅当 \(m>\sqrt{16k^{2}-1}/3\)，有 1 个根当 \(m\) 等于该值，否则没有根.

当 \(k=1\) 时，产生 2 个根的条件是 \(m>\sqrt{15}/3\)；当 \(k=2\) 时，产生根的临界值是
\[
m=\frac{\sqrt{16\cdot2^{2}-1}}{3}=\sqrt7
\]
由于上述临界值随 \(k\) 增大而增大，若 \(m<\sqrt7\)，只有 \(k=1\) 的第一段可能产生根；若 \(m\geqslant\sqrt7\)，则至少还会产生第 \(k=2\) 段的根. 第二段中 \(3f(x)\leqslant3\)，而 \(x>4k+1\geqslant5\)，不可能有根. 因此总根数恰为 \(3+2=5\) 当且仅当
\[
\frac{\sqrt{15}}{3}<m<\sqrt7
\]
故选 B.
\end{solution}
\end{problem}

\begin{problem}
若 \(A = \{x \in \R \mid |x| < 3\}\)，\(B = \{x \in \R \mid 2^{x} > 1\}\)，则 \(A \cap B =\)\fillinblank{}.

\begin{answer}
\((0,3)\)
\end{answer}

\begin{solution}
由 \(|x|<3\) 得 \(-3<x<3\)，所以 \(A=(-3,3)\). 又因为底数 \(2>1\)，
\[
2^x>1=2^0\quad\Longleftrightarrow\quad x>0
\]
故 \(B=(0,+\infty)\). 因此
\[
A\cap B=(0,3)
\]
\end{solution}
\end{problem}

\begin{problem}
若 \(f(x) = \frac{1}{2^{x} - 1} + a\) 是奇函数，则 \(a =\)\fillinblank{}.

\begin{answer}
\(\frac{1}{2}\)
\end{answer}

\begin{solution}
函数定义域为所有 \(x\ne0\) 的实数，且对任意 \(x\ne0\)，
\[
f(-x)=\frac{1}{2^{-x}-1}+a
=\frac{2^x}{1-2^x}+a
\]
所以
\[
f(x)+f(-x)
=\frac{1}{2^x-1}+\frac{2^x}{1-2^x}+2a
=-1+2a
\]
函数为奇函数当且仅当上式恒为 0，故 \(a=1/2\).
\end{solution}
\end{problem}

\begin{problem}
将 4 名大学生分配到 3 个乡镇去当村官，每个乡镇至少一名，则不同的分配方案有\fillinblank{} 种.（用数字作答）

\begin{answer}
36
\end{answer}

\begin{solution}
4 名大学生分到 3 个有标号乡镇且每个乡镇至少 1 人，人数分配只能是 \((2,1,1)\). 先选出人数为 2 的乡镇，有 3 种；再从 4 人中选 2 人去该乡镇，有 \(\mathrm C_{4}^{2}\) 种；余下 2 人分别去另外两个乡镇，有 \(2!\) 种. 因此方案数为
\[
3\mathrm C_{4}^{2}2!=3\times6\times2=36
\]
\end{solution}
\end{problem}

\begin{problem}
设 \(a_1 = 2\)，\(a_{n+1} = \frac{2}{a_n + 1}\)，\(b_n = \left|\frac{a_n + 2}{a_n - 1}\right|\)，\(n \in \N^{*}\)，则数列 \(\{b_n\}\) 的通项 \(b_n =\)\fillinblank{}.

\begin{answer}
\(2^{n+1}\)
\end{answer}

\begin{solution}
由递推式
\(a_{n+1}-1=\frac{1-a_n}{a_n+1}=-\frac{a_n-1}{a_n+1}\)，\(a_{n+1}+2=\frac{2(a_n+2)}{a_n+1}\).
因此
\[
b_{n+1}=\left|\frac{a_{n+1}+2}{a_{n+1}-1}\right|
=2\left|\frac{a_n+2}{a_n-1}\right|=2b_n
\]
又 \(b_1=|(2+2)/(2-1)|=4=2^2\)，所以
\[
b_n=b_1\cdot2^{n-1}=2^{n+1}
\]
递推所得各项为正，且若某项等于 1，则由递推反推会得到 \(a_1=1\)，与 \(a_1=2\) 矛盾，故上述分式始终有意义.
\end{solution}
\end{problem}

\begin{problem}
已知双曲线 \(\frac{x^{2}}{a^{2}} - \frac{y^{2}}{b^{2}} = 1\)（\(a > 0\)，\(b > 0\)）的左，右焦点分别为 \(F_{1}(-c, 0)\)，\(F_{2}(c, 0)\). 若双曲线上存在一点 \(P\) 使 \(\frac{\sin \angle PF_{1}F_{2}}{\sin \angle PF_{2}F_{1}} = \frac{a}{c}\)，则该双曲线的离心率的取值范围为\fillinblank{}.

\begin{answer}
\((1,\sqrt{2}+1)\)
\end{answer}

\begin{solution}
在三角形 \(PF_1F_2\) 中，由正弦定理
\[
\frac{\sin\angle PF_1F_2}{\sin\angle PF_2F_1}
=\frac{PF_2}{PF_1}=\frac{a}{c}=\frac1e<1
\]
故 \(PF_2<PF_1\)，点 \(P\) 在双曲线的右支，并且双曲线定义给出
\[
PF_1-PF_2=2a
\]
令 \(PF_1=r_1\)，\(PF_2=r_2\)，则 \(r_2=r_1/e\)，从而
\(r_1=\frac{2ae}{e-1}\)，\(r_2=\frac{2a}{e-1}\).
三角形存在的必要条件是 \(r_1+r_2>F_1F_2=2c=2ae\)，于是
\[
\frac{2a(e+1)}{e-1}>2ae
\quad\Longleftrightarrow\quad
e^2-2e-1<0
\]
结合双曲线离心率 \(e>1\)，得
\[
1<e<1+\sqrt2
\]
反之，在此范围内，除
\[
r_1+r_2>2c
\]
外，还有
\(r_2+2c-r_1=2c-2a=2a(e-1)>0\)，\(r_1+2c-r_2=2a+2c>0\).
所以三边 \(r_1\)，\(r_2\)，\(2c\) 满足三角形不等式，且 \(r_1-r_2=2a\)，可构成右支上一点 \(P\)，故范围正是
\[
(1,1+\sqrt2)
\]
\end{solution}
\end{problem}

\section{解答题}

\begin{problem}
设函数 \(f(x) = \sin \left(\frac{\pi}{4}x - \frac{\pi}{6}\right) - 2\cos^{2} \frac{\pi}{8}x + 1\).

\begin{enumerate}
\item 求 \(f(x)\) 的最小正周期；
\item 若函数 \(y = g(x)\) 与 \(y = f(x)\) 的图象关于直线 \(x = 1\) 对称，求当 \(x \in \left[0, \frac{4}{3}\right]\) 时 \(y = g(x)\) 的最大值.
\end{enumerate}

\begin{solution}
\begin{enumerate}
\item 由 \(2\cos^{2}u=1+\cos2u\)，有
\[
f(x)=\sin\left(\frac{\pi x}{4}-\frac{\pi}{6}\right)-\cos\frac{\pi x}{4}
=\sqrt{3}\sin\left(\frac{\pi x}{4}-\frac{\pi}{3}\right)
\]
正弦函数的自变量系数为 \(\pi/4\)，故最小正周期为
\[
T=\frac{2\pi}{\pi/4}=8
\]
\item 关于直线 \(x=1\) 对称时，点 \((x,g(x))\) 对应于 \((2-x,f(2-x))\)，所以
\[
g(x)=f(2-x)=\sqrt{3}\sin\left(\frac{\pi(2-x)}4-\frac\pi3\right)
=\sqrt{3}\cos\left(\frac{\pi x}{4}+\frac\pi3\right)
\]
当 \(0\leqslant x\leqslant4/3\) 时，
\[
\frac\pi3\leqslant\frac{\pi x}{4}+\frac\pi3\leqslant\frac{2\pi}3
\]
余弦函数在该区间的最大值为 \(1/2\)，故 \(g(x)\) 的最大值为 \(\sqrt3/2\).
\end{enumerate}
\end{solution}
\end{problem}

\begin{problem}
某单位为绿化环境，移栽了甲，乙两种大树各 2 株. 设甲，乙两种大树移栽的成活率分别为 \(\frac{2}{3}\) 和 \(\frac{1}{2}\)，且各株大树是否成活互不影响. 求移栽的 4 株大树中：

\begin{enumerate}
\item 两种大树各成活 1 株的概率；
\item 成活的株数 \(\xi\) 的分布列与期望.
\end{enumerate}

\begin{solution}
\begin{enumerate}
\item 设甲树成活数为 \(X\)，乙树成活数为 \(Y\). 则
\(P(X=0)=\frac19\)，\(P(X=1)=\frac49\)，\(P(X=2)=\frac49\)
\(P(Y=0)=\frac14\)，\(P(Y=1)=\frac12\)，\(P(Y=2)=\frac14\).
两类树的成活相互独立，因此
\[
P(X=1,Y=1)=\frac49\times\frac12=\frac29
\]
\item \(\xi=X+Y\). 由 \(X\) 与 \(Y\) 相互独立，逐项计算得
\[
P(\xi=0)=P(X=0)P(Y=0)=\frac19\times\frac14=\frac1{36}
\]
再有
\[
P(\xi=1)=P(X=0)P(Y=1)+P(X=1)P(Y=0)
=\frac19\times\frac12+\frac49\times\frac14=\frac16
\]
以及
\[
\begin{aligned}
P(\xi=2)&=P(X=0)P(Y=2)+P(X=1)P(Y=1)+P(X=2)P(Y=0)\\
&=\frac19\times\frac14+\frac49\times\frac12+\frac49\times\frac14=\frac{13}{36}.
\end{aligned}
\]
对 \(\xi=3\)，\(4\) 分别有
\[
P(\xi=3)=P(X=1)P(Y=2)+P(X=2)P(Y=1)
=\frac49\times\frac14+\frac49\times\frac12=\frac13
\]
\[
P(\xi=4)=P(X=2)P(Y=2)=\frac49\times\frac14=\frac19
\]
所以分布列为
\begin{center}
\begin{tabular}{c|ccccc}
\(\xi\)&\(0\)&\(1\)&\(2\)&\(3\)&\(4\)\\
\hline
\(P\)&\(\frac1{36}\)&\(\frac16\)&\(\frac{13}{36}\)&\(\frac13\)&\(\frac19\)
\end{tabular}
\end{center}
于是
\[
E(\xi)=0\cdot\frac1{36}+1\cdot\frac16+2\cdot\frac{13}{36}
+3\cdot\frac13+4\cdot\frac19=\frac73
\]
\end{enumerate}
\end{solution}
\end{problem}

\begin{problem}
设函数 \(f(x) = ax^{2} + bx + k\)（\(k > 0\)）在 \(x = 0\) 处取得极值，且曲线 \(y = f(x)\) 在点 \((1, f(1))\) 处的切线垂直于直线 \(x + 2y + 1 = 0\).

\begin{enumerate}
\item 求 \(a\)，\(b\) 的值；
\item 若函数 \(g(x) = \frac{\e^{x}}{f(x)}\)，讨论 \(g(x)\) 的单调性.
\end{enumerate}

\begin{solution}
\begin{enumerate}
\item 函数在 \(x=0\) 处取得极值，故 \(f'(0)=b=0\). 直线 \(x+2y+1=0\) 的斜率为 \(-1/2\)，其垂线斜率为 \(2\). 因而
\[
f'(1)=2a+b=2
\]
结合 \(b=0\) 得 \(a=1\).
\item 此时 \(f(x)=x^2+k>0\)，且
\[
g'(x)=\frac{\e^x(x^2+k)-\e^x(2x)}{(x^2+k)^2}
=\frac{\e^x\bigl((x-1)^2+k-1\bigr)}{(x^2+k)^2}
\]
分母和 \(\e^x\) 始终为正，只需考察 \((x-1)^2+k-1\).

当 \(k\geqslant1\) 时，\((x-1)^2+k-1\geqslant0\)，故 \(g(x)\) 在 \(\R\) 上单调递增；当 \(0<k<1\) 时，令
\(x_1=1-\sqrt{1-k}\)，\(x_2=1+\sqrt{1-k}\)
则导数在 \((-\infty,x_1)\) 和 \((x_2,+\infty)\) 为正，在 \((x_1,x_2)\) 为负. 因此 \(g\) 在前两个区间上分别递增，在中间区间上递减.
\end{enumerate}
\end{solution}
\end{problem}

\begin{solution}

\item 由平面 \(CSD\perp\) 平面 \(ABCD\)，且两平面的交线为 \(CD\)，又 \(AD\perp CD\)，得 \(AD\perp\) 平面 \(CSD\). 因而
\[
AS^{2}=AD^{2}+DS^{2}
\]
从而 \(DS=\sqrt{3-1}=\sqrt2\). 在直角三角形 \(CSD\) 中，\(CS\perp DS\)，故

CD^{2}=CS^{2}+DS^{2}=4+2=6
\]

\(C=(0,0,0)\)，\(D=(\sqrt6,0,0)\)，\(A=(\sqrt6,1,0)\).
令平面 \(CSD\) 为 \(y=0\)，由 \(CS=2\)，\(DS=\sqrt2\) 解得

S=\left(\frac{2\sqrt6}{3},0,\frac{2\sqrt3}{3}\right)
\]
设 \(B=(0,t,0)\). 由四边形顶点顺序取 \(t>0\). 因 \(E\) 为 \(BS\) 的中点，

E=\left(\frac{\sqrt6}{3},\frac t2,\frac{\sqrt3}{3}\right)

由 \(CE=\sqrt2\) 得

1+\frac{t^2}{4}=2
\]
故 \(t=2\)，即 \(B=(0,2,0)\). 平面 \(BCS\) 的方程为
\[
x-\sqrt2z=0
\]
所以点 \(A\) 到该平面的距离为
\[
\frac{|\sqrt6|}{\sqrt{1+2}}=\sqrt2
\]
\item 平面 \(ACD\) 为 \(z=0\)，可取其法向量 \(\bs n_1=(0,0,1)\). 又
\(\overrightarrow{CD}=(\sqrt6,0,0)\)，\(\overrightarrow{CE}=\left(\frac{\sqrt6}{3},1,\frac{\sqrt3}{3}\right)\)
故平面 \(ECD\) 的法向量可取

\bs n_2=\overrightarrow{CD}\times\overrightarrow{CE}
=\left(0,-\sqrt2,\sqrt6\right)
\]
二面角取锐角 \(\theta\)，则
\[
\cos\theta=\frac{|\bs n_1\cdot\bs n_2|}{|\bs n_1||\bs n_2|}
=\frac{\sqrt6}{\sqrt{2+6}}=\frac{\sqrt3}{2}
\]
因此 \(\theta=\pi/6\).
\end{enumerate}
\end{solution}

\begin{problem}
已知以原点 \(O\) 为中心的椭圆的一条准线方程为 \(x = \frac{4\sqrt{3}}{3}\)，离心率 \(e = \frac{\sqrt{3}}{2}\)，\(M\) 是椭圆上的动点.

\begin{enumerate}
\item 若点 \(C\)，\(D\) 的坐标分别是 \((0, -\sqrt{3})\)，\((0, \sqrt{3})\)，求 \(|MC| \cdot |MD|\) 的最大值；
\item 如图，点 \(A\) 的坐标为 \((1,0)\)，\(B\) 是圆 \(x^{2} + y^{2} = 1\) 上的点，\(N\) 是点 \(M\) 在 \(x\) 轴上的射影，点 \(Q\) 满足条件：\(\overrightarrow{OQ} = \overrightarrow{OM} + \overrightarrow{ON}\)，\(\overrightarrow{QA} \cdot \overrightarrow{BA} = 0\). 求线段 \(QB\) 的中点 \(P\) 的轨迹方程.
\end{enumerate}

\bitmapfigure[max width=0.38\linewidth]{img/2009/chongqing_science/q20_fig1.png}

\begin{solution}
\begin{enumerate}
\item 因准线方程为 \(x=4\sqrt{3}/3\)，椭圆长轴在 \(x\) 轴上，且准线到原点的距离为 \(a/e\). 由 \(e=\sqrt{3}/2\)，有
\(\frac{a}{e}=\frac{4\sqrt3}{3}\)，\(a=e\cdot\frac{4\sqrt3}{3}=2\)，\(c=ae=\sqrt3\).
又 \(b^2=a^2-c^2=1\)，故椭圆方程为
\[
\frac{x^2}{4}+y^2=1
\]
设 \(M=(u,v)\)，则 \(u^2/4+v^2=1\)，且
\(|MC|^2=u^2+(v+\sqrt3)^2\)，\(|MD|^2=u^2+(v-\sqrt3)^2\).
于是
\[
|MC|^2|MD|^2=(u^2+v^2+3)^2-12v^2
=(7-3v^2)^2-12v^2
=9v^4-54v^2+49
\]
令 \(z=v^2\)，则 \(0\leqslant z\leqslant1\)，且
\[
9z^2-54z+49
\]
的导数为 \(18z-54<0\)，故其在 \([0,1]\) 上严格递减. 因 \(|MC|\cdot|MD|\geqslant0\)，其最大值在 \(z=0\) 时取得. 此时 \(M=(2,0)\) 或 \(M=(-2,0)\)，从而
\[
|MC|\cdot|MD|=\sqrt{49}=7
\]
\item 设 \(M=(u,v)\). 因 \(N=(u,0)\) 且 \(\overrightarrow{OQ}=\overrightarrow{OM}+\overrightarrow{ON}\)，有 \(Q=(2u,v)\). 由 \(u^2/4+v^2=1\)，令 \(Q=(X,Y)\)，则
\[
\frac{X^2}{16}+Y^2=1
\]
设 \(P=(x,y)\). 由于 \(P\) 是 \(QB\) 的中点，\(B=2P-Q\). 又 \(A=(1,0)\)，条件 \(\overrightarrow{QA}\cdot\overrightarrow{BA}=0\) 给出
\[
(1-X)(1-2x+X)+Y(2y-Y)=0
\]
即
\[
2(xX+yY)=X^2+Y^2+2x-1
\]
另一方面，\(|B|=1\)，所以
\[
1=|2P-Q|^2=4(x^2+y^2)-4(xX+yY)+X^2+Y^2
\]
代入上式可得
\[
X^2+Y^2=4\left[\left(x-\frac12\right)^2+y^2\right]
\]
取参数 \(\lambda=\cos\theta\)，\(\mu=\sin\theta\)，则 \(Q=(4\lambda,\mu)\)，\(\lambda^2+\mu^2=1\). 因而 \(P\) 必须满足
\[
\left(x-\frac12\right)^2+y^2=\frac{1+15\lambda^2}{4}
\]
\[
(4\lambda-1)x+\mu y=\frac{15\lambda^2}{2}
\]
令
\(K=(4\lambda-1)^2+\mu^2=15\lambda^2-8\lambda+2\)，\(D=\frac{15\lambda^2-4\lambda+1}{2}\).
由于 \(\lambda^2+\mu^2=1\)，有 \(K>0\). 上面两式分别是以 \((1/2,0)\) 为圆心的圆和一条直线. 圆心到直线的有向距离为 \(D/\sqrt K\)，并且
\[
\frac{1+15\lambda^2}{4}-\frac{D^2}{K}=\frac{\mu^2}{4K}
\]
所以轨迹的参数方程可写为
\[
\begin{aligned}
x&=\frac12+\frac{D(4\lambda-1)}{K}-\varepsilon\frac{\mu^2}{2K},\\
y&=\frac{D\mu}{K}+\varepsilon\frac{\mu(4\lambda-1)}{2K},\\
\lambda&=\cos\theta,\quad \mu=\sin\theta,\quad \theta\in[0,2\pi),\\
&\varepsilon\in\{1,-1\}.
\end{aligned}
\]
对任意上述参数，取 \(Q=(4\lambda,\mu)\)、\(B=2P-Q\)，则 \(|B|=1\) 且 \(\overrightarrow{QA}\cdot\overrightarrow{BA}=0\)；反之原题中的 \(M\)，\(N\)，\(Q\)，\(B\)，\(P\) 必满足上述两式. 因此该参数方程给出的曲线正是所求轨迹.
\end{enumerate}
\end{solution}
\end{problem}

\begin{problem}
设 \(m\) 个不全相等的正数 \(a_1\)，\(a_2\)，\(\ldots\)，\(a_m\)（\(m\geqslant 7\)）依次围成一个圆圈.

\begin{enumerate}
\item 若 \(m=2009\)，且 \(a_1\)，\(a_2\)，\(\ldots\)，\(a_{1005}\) 是公差为 \(d\) 的等差数列，而 \(a_1\)，\(a_{2009}\)，\(a_{2008}\)，\(\ldots\)，\(a_{1006}\) 是公比为 \(q\) 的等比数列；数列 \(a_1\)，\(a_2\)，\(\ldots\)，\(a_m\) 的前 \(n\) 项和 \(S_n\)（\(n\leqslant m\)）满足：\(S_3=15\)，\(S_{2009}=S_{2007}+12a_1\)，求通项 \(a_n\)（\(n\leqslant m\)）.
\item 若每个数 \(a_n\)（\(n\leqslant m\)）是其左右相邻两数平方的等比中项，求证：\(a_1+\cdots+a_6+a_7^2+\cdots+a_m^2>ma_1a_2\cdots a_m\).
\end{enumerate}

\begin{solution}
\begin{enumerate}
\item 由正数条件 \(q>0\). 等比数列的排列顺序给出
\(a_{2009}=a_1q\)，\(a_{2008}=a_1q^2\).
又
\[
S_{2009}-S_{2007}=a_{2008}+a_{2009}=12a_1
\]
所以 \(q^2+q=12\). 结合 \(q>0\)，得 \(q=3\).

令 \(a_1=t\). 前 \(1005\) 项为等差数列，且
\[
S_3=a_1+(a_1+d)+(a_1+2d)=3a_1+3d=15
\]
故 \(d=5-t\). 因而
\(a_n=t+(n-1)(5-t)\)，\(1\leqslant n\leqslant1005\).
后面的等比数列按题目给出的次序为
\(a_n=t\,3^{2010-n}\)，\(1006\leqslant n\leqslant2009\).
为了保证等差部分各项为正，只需两端均为正，即
\(t>0\)，\(t+1004(5-t)=5020-1003t>0\).
所以题面条件得到的是一族数列
\[
0<t<\frac{5020}{1003},\qquad
a_n=\begin{cases}
t+(n-1)(5-t),&1\leqslant n\leqslant1005,\\
t\,3^{2010-n},&1006\leqslant n\leqslant2009.
\end{cases}
\]
题面没有给出 \(d=q\) 或其他联系 \(d\) 与 \(q\) 的条件，因此不能唯一确定 \(t\). 例如额外补充题面未出现的 \(d=q\) 时才得到 \(t=2\)，\(d=3\)，这不能作为原题的必然结论.

\item 由于各项为正，“\(a_n\) 是左右相邻两数平方的等比中项”表示
\[
a_n^2=a_{n-1}^{\,2}a_{n+1}^{\,2}
\]
其中下标按圆圈循环. 于是
\(a_n=a_{n-1}a_{n+1}\)，\(a_{n+1}=\frac{a_n}{a_{n-1}}\).
逐次使用后一关系，得到
\(a_{r+2}=\frac{a_{r+1}}{a_r}\)，\(a_{r+3}=\frac1{a_r}\)，\(a_{r+4}=\frac1{a_{r+1}}\)，\(a_{r+5}=\frac{a_r}{a_{r+1}}\)，\(a_{r+6}=a_r\).
所以数列具有六项周期.

若 \(m=6k+p\)，其中 \(0\leqslant p<6\)，圆周闭合又使数列具有 \(p\) 项周期. 若 \(p=1\) 或 \(p=5\)，所有项相等，设为 \(c\)，递推式给出 \(c=c^2\)，因正数而 \(c=1\). 若 \(p=2\) 或 \(p=4\)，奇数项均为 \(u\)、偶数项均为 \(v\)，递推式给出 \(u=v^2\)、\(v=u^2\)，从而 \(u=v=1\). 若 \(p=3\)，设三个周期值为 \(u\)，\(v\)，\(w\)，则
\(u=wv\)，\(v=uw\)，\(w=uv\).
由前两式得 \(w=1\)，再由第三式得 \(u=v=1\). 这些情形都与“不全相等”矛盾，故 \(m=6k\).

将递推式对 \(n=1\)，\(\ldots\)，\(m\) 相乘，设 \(P=a_1a_2\cdots a_m\)，则
\[
P=P^2
\]
因 \(P>0\)，故 \(P=1\). 由
\(a_2=a_1a_3\)，\(a_3=a_2a_4\)，\(a_4=a_3a_5\)，\(a_5=a_4a_6\)
依次得到
\(a_3=\frac{a_2}{a_1}\)，\(a_4=\frac1{a_1}\)，\(a_5=\frac1{a_2}\)，\(a_6=\frac{a_1}{a_2}\).
因此
\(a_1+a_4\geqslant2\)，\(a_2+a_5\geqslant2\)，\(a_3+a_6\geqslant2\).
三式同时取等号会导致 \(a_1=a_2=1\)，进而六项全为 \(1\)，再由六项周期与“不全相等”矛盾，所以
\[
a_1+\cdots+a_6>6
\]
同样由平方的均值不等式，
\(a_1^2+a_4^2\geqslant2\)，\(a_2^2+a_5^2\geqslant2\)，\(a_3^2+a_6^2\geqslant2\)
故 \(a_1^2+\cdots+a_6^2\geqslant6\). 因 \(m=6k\) 且六项周期，
\[
a_7^2+\cdots+a_m^2=(k-1)(a_1^2+\cdots+a_6^2)\geqslant6(k-1)
\]
于是
\[
a_1+\cdots+a_6+a_7^2+\cdots+a_m^2
>6+6(k-1)=6k=m=mP=ma_1a_2\cdots a_m
\]
证毕.
\end{enumerate}
\end{solution}
\end{problem}
