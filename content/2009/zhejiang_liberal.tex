\section{单选题}

\begin{problem}
设 \(U = \R\)，\(A = \{x \mid x > 0\}\)，\(B = \{x \mid x > 1\}\)，则 \(A \cap \complement_U B =\)（\quad）

\choices
    {\(\{x\mid 0\leqslant x < 1\}\)}
    {\(\{x\mid 0 < x\leqslant 1\}\)}
    {\( \{x \mid x < 0\} \)}
    {\( \{x \mid x > 1\} \)}
\end{problem}

\begin{problem}
“x > 0”是“x ≠ 0”的（\quad）

\choices
    {充分而不必要条件}
    {必要而不充分条件}
    {充分必要条件}
    {既不充分也不必要条件}
\end{problem}

\begin{problem}

    {\(-1 + \i\)}
    {\(1 - \i\)}
    {\(-1 - \i\)}
\end{problem}

\begin{problem}
设 \(\alpha, \beta\) 是两个不同的平面，\(l\) 是一条直线. 以下命题正确的是（\quad）

\choices
    {若 \(l \perp \alpha, \alpha \perp \beta\)，则 \(l \subset \beta\)}
    {若 \(l // \alpha, \alpha // \beta\)，则 \(l \subset \beta\)}
    {若 \(l \perp \alpha, \alpha // \beta\)，则 \(l \perp \beta\)}
    {若 \(l // \alpha, \alpha \perp \beta\)，则 \(l \perp \beta\)}
\end{problem}

\begin{problem}

    {\(\left(\frac{7}{3}, \frac{7}{9}\right)\)}
    {\(\left(-\frac{7}{9}, -\frac{7}{3}\right)\)}
\end{problem}

\begin{problem}

\begin{enumerate}[label={（\arabic*）},itemsep=0.2em]
\item 求 \( \triangle ABC \) 的面积；
\item 若 \(c = 1\)，求 \(a\) 的值.
\end{enumerate}
\end{problem}

\begin{problem}

\begin{enumerate}[label={（\arabic*）},itemsep=0.2em]
\item 若函数 \( f(x) \) 的图象过原点，且在原点处的切线斜率是 \(-3\). 求 \( a, b \) 的值；
\item 若函数 \( f(x) \) 在区间 \((-1, 1)\) 上不单调，求 \( a \) 的取值范围.
\end{enumerate}
\end{problem}
