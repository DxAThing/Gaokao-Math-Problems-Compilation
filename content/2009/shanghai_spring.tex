\begin{problem}

\bitmapfigure[float=inline,max width=0.38\linewidth]{img/2009/shanghai_spring/q16_fig1.png}
\end{problem}

\begin{solution}
在 \(\mathrm{Rt}\triangle AA_1C\) 中，\(AC=AA_1\tan\angle AA_1C=4\sqrt{3}\cdot\frac{\sqrt{3}}{3}=4\).
作 \(B_1H\perp\) 平面 \(ABC\)，垂足为 \(H\)，则 \(\angle B_1BH=\frac{\pi}{3}\).
在 \(\mathrm{Rt}\triangle B_1BH\) 中，\(B_1H=BB_1\sin\angle B_1BH=AA_1\sin\frac{\pi}{3}=4\sqrt{3}\cdot\frac{\sqrt{3}}{2}=6\).

于是 \(V=S_{\triangle ABC}\cdot B_1H=\frac12\times 4\times 4\times 6=48\).
\end{solution}

\begin{problem}

已知数列 \( \left\{a_{n}\right\} \) 的前 \(n\) 项和为 \( S_{n} \)， \( a_{1}=1 \)，且 \( 3a_{n+1}+2S_{n}=3 \)（ \( n \) 为正整数）.

\begin{enumerate}
\item 求数列 \( \{a_{n}\} \) 的通项公式；
\item 记 \( S = a_1 + a_2 + \cdots + a_n + \cdots \). 若对任意正整数 \( n \)， \( kS \leq S_n \) 恒成立，求实数 \( k \) 的最大值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由 \(3a_{n+1}+2S_n=3\) 和当 \(n\ge2\) 时的 \(3a_n+2S_{n-1}=3\)，相减得 \(3a_{n+1}-3a_n+2a_n=0\)，故 \(\frac{a_{n+1}}{a_n}=\frac13\)（\(n\ge2\)）. 又 \(a_1=1\)，\(3a_2+2a_1=3\)，解得 \(a_2=\frac13\). 故 \(\{a_n\}\) 是首项为 \(1\)，公比为 \(\frac13\) 的等比数列，于是 \(a_n=\left(\frac13\right)^{n-1}\)（\(n\) 为正整数）.

\item 由（1）知 \(S=\frac{a_1}{1-q}=\frac{1}{1-\frac13}=\frac32\). 又 \(S_n=\frac{a_1(1-q^n)}{1-q}=\frac{1-\left(\frac13\right)^n}{1-\frac13}=\frac32\left[1-\left(\frac13\right)^n\right]\). 由题意可知，对于任意的正整数 \(n\)，恒有 \(\frac32k\le \frac32\left[1-\left(\frac13\right)^n\right]\)，解得 \(k\le 1-\left(\frac13\right)^n\). 数列 \(\left\{1-\left(\frac13\right)^n\right\}\) 单调递增，故当 \(n=1\) 时，数列中的最小项为 \(\frac23\)，从而 \(k\le\frac23\)，即实数 \(k\) 的最大值为 \(\frac23\).
\end{enumerate}
\end{solution}

\begin{problem}

我国计划发射火星探测器，该探测器的运行轨道是以火星（其半径 \(R=34\) 百公里）的中心 \(F\) 为一个焦点的椭圆. 如图，已知探测器的近火星点（轨道上离火星表面最近的点）\(A\) 到火星表面的距离为\(8\) 百公里，远火星点（轨道上离火星表面最远的点）\(B\) 到火星表面的距离为\(800\) 百公里. 假定探测器由近火星点 \(A\) 第一次逆时针运行到与轨道中心 \(O\) 的距离为\(\sqrt{ab}\) 百公里时进行变轨，其中 \(a\)，\(b\) 分别为椭圆的长半轴，短半轴的长，求此时探测器与火星表面的距离（精确到\(1\) 百公里）.

\bitmapfigure[max width=0.5\linewidth]{img/2009/shanghai_spring/q18_fig1.png}
\end{problem}

\begin{solution}
设所求轨道方程为 \(\dfrac{x^{2}}{a^{2}}+\dfrac{y^{2}}{b^{2}}=1\)（\(a>b>0\)），\(c=\sqrt{a^{2}-b^{2}}\).
由 \(a+c=800+34\)，\(a-c=8+34\)，得 \(a=438\)，\(c=396\).
于是 \(b^{2}=a^{2}-c^{2}=35028\).
所以所求轨道方程为 \(\dfrac{x^{2}}{191844}+\dfrac{y^{2}}{35028}=1\).
设变轨时，探测器位于 \(P(x_0,y_0)\)，则
\(x_0^2+y_0^2=ab=81975.1\)，\(\frac{x_0^2}{191844}+\frac{y_0^2}{35028}=1\).
解得 \(x_0=239.7\)，\(y_0=156.7\)（由题意）.
所以探测器在变轨时与火星表面的距离为
\(\sqrt{(x_0-c)^2+y_0^2}-R\approx 187.3\).
答：探测器在变轨时与火星表面的距离约为\(187\) 百公里.
\end{solution}

\begin{problem}

如图，在直角坐标系 \(xOy\) 中，有一组对角线长为 \( a_{n} \) 的正方形 \( A_{n}B_{n}C_{n}D_{n} \) （ \(n=1\)，\(2\)，\(\cdots\)）， 其对角线 \( B_n D_n \) 依次放置在 \( x \) 轴上（相邻顶点重合）. 设 \( \{a_n\} \) 是首项为 \( a \)，公差为 \( d \)（\(d > 0\)）的等差数列，点 \(B_1\) 的坐标为 \( (d, 0) \).

\begin{enumerate}
\item 当 \(a = 8\)，\(d = 4\) 时，证明：顶点 \(A_{1}\)， \(A_{2}\)， \(A_{3}\) 不在同一条直线上；
\item 在（1）的条件下，证明：所有顶点 \( A_{n} \) 均落在抛物线 \( y^{2}=2x \) 上；
\item 为使所有顶点 \( A_{n} \) 均落在抛物线 \( y^{2}=2px \) （\(p>0\)）上，求 \(a\) 与 \(d\) 之间所应满足的关系式.
\end{enumerate}

\bitmapfigure[max width=0.54\linewidth]{img/2009/shanghai_spring/q19_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由题意可知，\(A_1(8,4)\)，\(A_2(18,6)\)，\(A_3(32,8)\). 故 \(k_{A_1A_2}=\dfrac{6-4}{18-8}=\dfrac15\)，\(k_{A_2A_3}=\dfrac{8-6}{32-18}=\dfrac17\). 因为 \(k_{A_1A_2}\ne k_{A_2A_3}\)，故顶点 \(A_1\)，\(A_2\)，\(A_3\) 不在同一条直线上.

\item 由题意可知，顶点 \(A_n\) 的横坐标 \(x_n=d+a_1+a_2+\cdots+a_{n-1}+\frac12a_n=2(n+1)^2\)，纵坐标 \(y_n=\frac12a_n=2(n+1)\). 对任意正整数 \(n\)，点 \(A_n(x_n,y_n)\) 的坐标满足方程 \(y^2=2x\)，故所有顶点 \(A_n\) 均落在抛物线 \(y^2=2x\) 上.

\item
【法一】由题意可知，顶点 \(A_n\) 的横，纵坐标分别是 \(x_n=d+\frac12a+(n-1)a+\frac12(n-1)^2d\)，\(y_n=\frac12[a+(n-1)d]\). 消去 \(n-1\)，可得 \(x_n=\frac{2}{d}y_n^2+d+\frac{a(d-a)}{2d}\). 为使得所有顶点 \(A_n\) 均落在抛物线 \(y^2=2px\)（\(p>0\)）上，则有
\[
\begin{cases}
\dfrac d2=2p,\\
d+\dfrac{a(d-a)}{2d}=0.
\end{cases}
\]
解之，得 \(d=4p\)，\(a=8p\). 故 \(a\)，\(d\) 所应满足的关系式是：\(a=2d\).

【法二】点 \(A_1(x_1,y_1)\) 的坐标为
\[
\begin{cases}
x_1=d+\dfrac12a,\\
y_1=\dfrac12a.
\end{cases}
\]
点 \(A_1(x_1,y_1)\) 在抛物线 \(y^2=2px\) 上，故 \(p=\frac{y_1^2}{2x_1}=\frac{a^2}{4(2d+a)}\). 又点 \(A_2(x_2,y_2)\) 的坐标为
\[
\begin{cases}
x_2=\dfrac32a+\dfrac32d,\\
y_2=\dfrac12(a+d).
\end{cases}
\]
把点 \(A_2(x_2,y_2)\) 代入抛物线方程，解得 \(a=2d\). 因此 \(p=\dfrac d4\)，故抛物线方程为 \(y^2=\dfrac d2x\). 又
\[
\begin{cases}
x_n=d+\dfrac12a+(n-1)a+\dfrac12(n-1)^2d=\dfrac{(n+1)^2}{2}d,\\
y_n=\dfrac12[a+(n-1)d]=\dfrac{n+1}{2}d.
\end{cases}
\]
故所有顶点 \(A_n(x_n,y_n)\) 落在抛物线 \(y^2=\dfrac d2x\) 上. 故 \(a\)，\(d\) 所应满足的关系式是：\(a=2d\).
\end{enumerate}
\end{solution}

\begin{problem}

设函数 \(f_{n}(\theta)=\sin^{n}\theta+(-1)^{n}\cos^{n}\theta\)，\(0\leq\theta\leq\frac{\pi}{4}\)，其中 \(n\) 为正整数.

\begin{enumerate}
\item 判断函数 \( f_{1}(\theta) \)， \( f_{3}(\theta) \) 的单调性，并就 \( f_{1}(\theta) \) 的情形证明你的结论；
\item 证明： \( 2f_{6}(\theta)-f_{4}(\theta)=\left(\cos^{4}\theta-\sin^{4}\theta\right)\left(\cos^{2}\theta-\sin^{2}\theta\right) \)；
\item 对于任意给定的正整数 \( n \)，求函数 \( f_{n}(\theta) \) 的最大值和最小值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item \(f_1(\theta)\)，\(f_3(\theta)\) 在 \(\left[0,\frac{\pi}{4}\right]\) 上均为单调递增的函数.
对于函数 \(f_1(\theta)=\sin\theta-\cos\theta\)，设 \(\theta_1<\theta_2\)，\(\theta_1\)，\(\theta_2\in\left[0,\frac{\pi}{4}\right]\)，则
\(f_1(\theta_1)-f_1(\theta_2)=(\sin\theta_1-\sin\theta_2)+(\cos\theta_2-\cos\theta_1)\).
因为 \(\sin\theta_1<\sin\theta_2\)，\(\cos\theta_2<\cos\theta_1\)，故 \(f_1(\theta_1)<f_1(\theta_2)\)，于是 \(f_1(\theta)\) 在 \(\left[0,\frac{\pi}{4}\right]\) 上单调递增.

\item 原式左边
\[
2f_6(\theta)-f_4(\theta)=2(\sin^6\theta+\cos^6\theta)-(\sin^4\theta+\cos^4\theta).
\]
又
\[
2(\sin^6\theta+\cos^6\theta)-(\sin^4\theta+\cos^4\theta)
=2(1-3\sin^2\theta\cos^2\theta)-(1-2\sin^2\theta\cos^2\theta).
\]
于是 \(2f_6(\theta)-f_4(\theta)=1-\sin^22\theta=\cos^22\theta\).
原式右边 \((\cos^4\theta-\sin^4\theta)(\cos^2\theta-\sin^2\theta)=(\cos^2\theta-\sin^2\theta)^2=\cos^22\theta\). 故 \(2f_6(\theta)-f_4(\theta)=\left(\cos^4\theta-\sin^4\theta\right)\left(\cos^2\theta-\sin^2\theta\right)\).

\item 当 \(n=1\) 时，\(f_1(\theta)\) 在 \(\left[0,\frac{\pi}{4}\right]\) 上单调递增，故最大值为 \(f_1\left(\frac{\pi}{4}\right)=0\)，最小值为 \(f_1(0)=-1\).

当 \(n=2\) 时，\(f_2(\theta)=1\)，故最大，最小值均为 1.

当 \(n=3\) 时，\(f_3(\theta)\) 在 \(\left[0,\frac{\pi}{4}\right]\) 上单调递增，故最大值为 \(f_3\left(\frac{\pi}{4}\right)=0\)，最小值为 \(f_3(0)=-1\).

当 \(n=4\) 时，\(f_4(\theta)=1-\frac12\sin^22\theta\) 在 \(\left[0,\frac{\pi}{4}\right]\) 上单调递减，故最大值为 \(f_4(0)=1\)，最小值为 \(f_4\left(\frac{\pi}{4}\right)=\frac12\).

下面讨论正整数 \(n\ge5\) 的情形：

当 \(n\) 为奇数时，对任意 \(\theta_1\)，\(\theta_2\in\left[0,\frac{\pi}{4}\right]\) 且 \(\theta_1<\theta_2\)，有
\(f_n(\theta_1)-f_n(\theta_2)=(\sin^n\theta_1-\sin^n\theta_2)+(\cos^n\theta_2-\cos^n\theta_1)\)，
且 \(0\le\sin\theta_1<\sin\theta_2<1\)，\(0<\cos\theta_2<\cos\theta_1\le1\)，从而 \(\sin^n\theta_1<\sin^n\theta_2\)，\(\cos^n\theta_2<\cos^n\theta_1\)，故 \(f_n(\theta_1)<f_n(\theta_2)\).
所以 \(f_n(\theta)\) 在 \(\left[0,\frac{\pi}{4}\right]\) 上为单调递增，则最大值为 \(f_n\left(\frac{\pi}{4}\right)=0\)，最小值为 \(f_n(0)=-1\).

当 \(n\) 为偶数时，一方面有 \(f_n(\theta)=\sin^n\theta+\cos^n\theta\le \sin^2\theta+\cos^2\theta=1=f_n(0)\).
另一方面，由于对任意正整数 \(l\ge2\)，有
\(2f_{2l}(\theta)-f_{2l-2}(\theta)=(\cos^{2l-2}\theta-\sin^{2l-2}\theta)(\cos^2\theta-\sin^2\theta)\ge0\)，
故 \(f_{2l}(\theta)\ge \frac12f_{2l-2}(\theta)\). 迭代得
\(f_n(\theta)\ge \frac1{2^{\frac n2-1}}f_2(\theta)=2\sqrt{\left(\frac12\right)^n}\).
因此函数 \(f_n(\theta)\) 的最大值为 \(f_n(0)=1\)，最小值为 \(2\sqrt{\left(\frac12\right)^n}\).

综上所述，当 \(n\) 为奇数时，函数 \(f_n(\theta)\) 的最大值为 \(0\)，最小值为 \(-1\)；当 \(n\) 为偶数时，函数 \(f_n(\theta)\) 的最大值为 \(1\)，最小值为 \(2\sqrt{\left(\frac12\right)^n}\).
\end{enumerate}
\end{solution}
