\begin{problem}
有四个关于三角函数的命题：\(p_{1}\)：\(\exists x \in \R\)，\(\sin^{2}\frac{x}{2} + \cos^{2}\frac{x}{2} = \frac{1}{2}.\)；\(p_{2}\)：\(\exists x\)，\(y \in \R\)，\(\sin(x - y) = \sin x - \sin y\)；\(p_{3}\)：\(\forall x \in [0,\pi]\)，\(\sqrt{\frac{1 - \cos 2x}{2}} = \sin x\)；\(p_{4}\)：\(\sin x = \cos y \Rightarrow x + y = \frac{\pi}{2}.\) 其中假命题的是（\quad）

\choices
    {\(p_1, p_4\)}
    {\(p_{2}\)，\(p_{4}\)}
    {\(p_1\)，\(p_3\)}
    {\(p_2\)，\(p_3\)}
\end{problem}

\begin{problem}

\begin{problem}

\end{enumerate}
\end{problem}
