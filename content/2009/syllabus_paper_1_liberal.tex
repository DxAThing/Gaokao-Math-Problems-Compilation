\section{单选题}

\begin{problem}
\(\sin 585^{\circ}\) 的值为（\quad）

\choices
    {\(-\frac{\sqrt{2}}{2}\)}
    {\(\frac{\sqrt{2}}{2}\)}
    {\(-\frac{\sqrt{3}}{2}\)}
    {\(\frac{\sqrt{3}}{2}\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
因为 \(585^{\circ}=360^{\circ}+225^{\circ}\)，所以 \(\sin 585^{\circ}=\sin 225^{\circ}=-\frac{\sqrt{2}}{2}\)，故选 A.
\end{solution}

\begin{problem}
设集合 \(A = \{4, 5, 7, 9\}\)，\(B = \{3, 4, 7, 8, 9\}\)，全集 \(U = A \cup B\)，则集合 \(\complement_U (A \cap B)\) 中的元素共有（\quad）

\choices
    {\(3\text{ 个}\)}
    {\(4\text{ 个}\)}
    {\(5\text{ 个}\)}
    {\(6\text{ 个}\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
由 \(U=A\cup B=\{3,4,5,7,8,9\}\)，\(A\cap B=\{4,7,9\}\)，得 \(\complement_U(A\cap B)=\{3,5,8\}\)，其中有 \(3\) 个元素，故选 A.
\end{solution}

\begin{problem}
不等式 \(\left|\frac{x + 1}{x - 1}\right| < 1\) 的解集为（\quad）

\choices
    {\(\{x\mid 0 < x < 1\} \cup \{x\mid x > 1\}\)}
    {\(\{x\mid 0 < x < 1\}\)}
    {\(\{x\mid -1 < x < 0\}\)}
    {\(\{x\mid x < 0\}\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
由定义域知 \(x\neq1\)，且两边均为非负数，因此原不等式等价于 \((x+1)^2<(x-1)^2\). 展开并化简得 \(4x<0\)，所以解集为 \(\{x\mid x<0\}\)，故选 D.
\end{solution}

\begin{problem}
已知 \(\tan \alpha = 4\)，\(\cot \beta = \frac{1}{3}\)，则 \(\tan (\alpha + \beta) =\)（\quad）

\choices
    {\(\frac{7}{11}\)}
    {\(-\frac{7}{11}\)}
    {\(\frac{7}{13}\)}
    {\(-\frac{7}{13}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
由 \(\cot\beta=\frac{1}{3}\) 得 \(\tan\beta=3\). 又 \(1-\tan\alpha\tan\beta=1-12=-11\neq0\)，故
\[
\tan(\alpha+\beta)=\frac{\tan\alpha+\tan\beta}{1-\tan\alpha\tan\beta}=\frac{4+3}{1-4\times3}=-\frac{7}{11}
\]
因此选 B.
\end{solution}

\begin{problem}
设双曲线 \(\frac{x^2}{a^2} - \frac{y^2}{b^2} = 1\)（\(a > 0\)，\(b > 0\)）的渐近线与抛物线 \(y = x^2 + 1\) 相切，则该双曲线的离心率等于（\quad）

\choices
    {\(\sqrt{3}\)}
    {\(2\)}
    {\(\sqrt{5}\)}
    {\(\sqrt{6}\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
设双曲线渐近线的斜率绝对值为 \(k=\frac{b}{a}>0\)，则两条渐近线为 \(y=kx\) 和 \(y=-kx\). 将它们分别与抛物线 \(y=x^2+1\) 联立，得到方程 \(x^2-kx+1=0\) 和 \(x^2+kx+1=0\). 两个方程的判别式都为 \(k^2-4\)，所以渐近线与抛物线相切的条件是 \(k^2-4=0\). 由 \(k>0\) 得 \(k=2\)，从而
\[
e=\sqrt{1+\frac{b^2}{a^2}}=\sqrt{1+k^2}=\sqrt{5}
\]
故选 C.
\end{solution}

\begin{problem}
已知函数 \( f(x) \) 的反函数为 \( g(x) = 1 + 2\lg x \)（\(x > 0\)），则 \( f(1) + g(1) = \)（\quad）

\choices
    {\(0\)}
    {\(1\)}
    {\(2\)}
    {\(4\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
因为 \(g\) 是 \(f\) 的反函数，所以 \(f(1)\) 等于满足 \(g(x)=1\) 的 \(x\). 由 \(1+2\lg x=1\) 得 \(\lg x=0\)，从而 \(x=1\)，即 \(f(1)=1\). 又 \(g(1)=1+2\lg1=1\)，所以 \(f(1)+g(1)=2\)，故选 C.
\end{solution}

\begin{problem}
甲组有 \(5\text{ 名}\)男同学，\(3\text{ 名}\)女同学；乙组有 \(6\text{ 名}\)男同学，\(2\text{ 名}\)女同学. 若从甲，乙两组中各选出 \(2\text{ 名}\)同学，则选出的 \(4\text{ 人}\)中恰有 \(1\text{ 名}\)女同学的不同选法共有（\quad）

\choices
    {\(150\text{ 种}\)}
    {\(180\text{ 种}\)}
    {\(300\text{ 种}\)}
    {\(345\text{ 种}\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
恰有 \(1\) 名女同学时，分两种情况：甲组选出 \(1\) 名女生和 \(1\) 名男生、乙组选出 \(2\) 名男生，或甲组选出 \(2\) 名男生、乙组选出 \(1\) 名女生和 \(1\) 名男生. 因此不同选法共有
\[
\mathrm C_3^1\mathrm C_5^1\mathrm C_6^2+\mathrm C_5^2\mathrm C_2^1\mathrm C_6^1=3\times5\times15+10\times2\times6=345
\]
故选 D.
\end{solution}

\begin{problem}
设非零向量 \(\bs{a}\)，\(\bs{b}\)，\(\bs{c}\) 满足 \(|\bs{a}| = |\bs{b}| = |\bs{c}|\)，\(\bs{a} + \bs{b} = \bs{c}\)，则 \(\langle \bs{a}, \bs{b} \rangle =\)（\quad）

\choices
    {\(150^{\circ}\)}
    {\(120^{\circ}\)}
    {\(60^{\circ}\)}
    {\(30^{\circ}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
设 \(\theta=\langle\bs{a},\bs{b}\rangle\)，并记 \(|\bs{a}|=|\bs{b}|=|\bs{c}|=r\)，其中 \(r>0\). 由 \(\bs{c}=\bs{a}+\bs{b}\) 得
\[
r^2=|\bs{c}|^2=|\bs{a}+\bs{b}|^2=r^2+r^2+2\bs{a}\cdot\bs{b}
\]
所以 \(\bs{a}\cdot\bs{b}=-\frac{r^2}{2}\)，从而 \(\cos\theta=\frac{\bs{a}\cdot\bs{b}}{|\bs{a}||\bs{b}|}=-\frac{1}{2}\). 因此 \(\theta=120^{\circ}\)，故选 B.
\end{solution}

\begin{problem}
已知三棱柱 \(ABC - A_{1}B_{1}C_{1}\) 的侧棱与底面边长都相等，\(A_{1}\) 在底面 \(ABC\) 上的射影为 \(BC\) 的中点，则异面直线 \(AB\) 与 \(CC_{1}\) 所成的角的余弦值为（\quad）

\choices
    {\(\frac{\sqrt{3}}{4}\)}
    {\(\frac{\sqrt{5}}{4}\)}
    {\(\frac{\sqrt{7}}{4}\)}
    {\(\frac{3}{4}\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
设三棱柱的底面边长为 \(a\)，取底面为 \(z=0\)，令 \(BC\) 的中点为原点，并取
\(B=\left(-\frac{a}{2},0,0\right)\)，\(C=\left(\frac{a}{2},0,0\right)\)，\(A=\left(0,\frac{\sqrt{3}a}{2},0\right)\). 设 \(A_{1}\) 的竖直坐标为 \(h\)，由题设中 \(A_{1}\) 在底面上的射影为原点，得
\[
\overrightarrow{AA_{1}}=\left(0,-\frac{\sqrt{3}a}{2},h\right)
\]
侧棱长等于底面边长 \(a\)，所以 \(\frac{3a^2}{4}+h^2=a^2\)，从而 \(h=\frac{a}{2}\). 又 \(\overrightarrow{CC_{1}}=\overrightarrow{AA_{1}}\)，而 \(\overrightarrow{AB}=\left(-\frac{a}{2},-\frac{\sqrt{3}a}{2},0\right)\)，故
\[
\cos\angle\left(AB,CC_{1}\right)=\frac{\overrightarrow{AB}\cdot\overrightarrow{CC_{1}}}{|\overrightarrow{AB}||\overrightarrow{CC_{1}}|}
=\frac{3a^2/4}{a\cdot a}=\frac{3}{4}
\]
故选 D.
\end{solution}

\begin{problem}
如果函数 \( y = 3\cos (2x + \varphi) \) 的图象关于点 \( \left(\frac{4\pi}{3},0\right) \) 中心对称，那么 \( |\varphi| \) 的最小值为（\quad）

\choices
    {\(\frac{\pi}{6}\)}
    {\(\frac{\pi}{4}\)}
    {\(\frac{\pi}{3}\)}
    {\(\frac{\pi}{2}\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
设 \(x_0=\frac{4\pi}{3}\). 图象关于 \((x_0,0)\) 中心对称，等价于对任意 \(t\)，有 \(y(x_0+t)+y(x_0-t)=0\). 由余弦和公式，
\[
y(x_0+t)+y(x_0-t)=6\cos(2x_0+\varphi)\cos 2t
\]
因此必须有 \(\cos(2x_0+\varphi)=0\)，即 \(2x_0+\varphi=\frac{\pi}{2}+k\pi\)，其中 \(k\in\Z\). 代入 \(x_0=\frac{4\pi}{3}\) 得
\[
\varphi=\frac{\pi}{2}+k\pi-\frac{8\pi}{3}
\]
取 \(k=2\) 时 \(\varphi=-\frac{\pi}{6}\)，而其他整数 \(k\) 所得值与此相差整数个 \(\pi\)，故 \(|\varphi|\) 的最小值为 \(\frac{\pi}{6}\)，选 A.
\end{solution}

\begin{problem}
已知二面角 \(\alpha - l - \beta\) 为 \(60^{\circ}\)，动点 \(P\)，\(Q\) 分别在面 \(\alpha\)，\(\beta\) 内，\(P\) 到 \(\beta\) 的距离为 \(\sqrt{3}\)，\(Q\) 到 \(\alpha\) 的距离为 \(2\sqrt{3}\)，则 \(P\)，\(Q\) 两点之间距离的最小值为（\quad）

\choices
    {\(\sqrt{2}\)}
    {\(2\)}
    {\(2\sqrt{3}\)}
    {\(4\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
取 \(l\) 为 \(x\) 轴，令面 \(\beta\) 为 \(z=0\)，面 \(\alpha\) 的方程为 \(z=\sqrt{3}y\). 若 \(P=(x_0,y_0,\sqrt{3}y_0)\)，则 \(P\) 到面 \(\beta\) 的距离为 \(\sqrt{3}|y_0|\)，由题设得 \(|y_0|=1\). 因此可写成 \(P=(x_0,\varepsilon,\sqrt{3}\varepsilon)\)，其中 \(\varepsilon\in\{1,-1\}\). 若 \(Q=(x,y,0)\) 在面 \(\beta\) 上，则 \(Q\) 到面 \(\alpha\) 的距离为 \(\frac{\sqrt{3}}{2}|y|\)，由题设得 \(|y|=4\)，故可写成 \(Q=(x,4\delta,0)\)，其中 \(\delta\in\{1,-1\}\). 于是
\[
|PQ|^2=(x-x_0)^2+(4\delta-\varepsilon)^2+3
\]
当 \(x=x_0\) 时，距离关于 \(x\) 取得最小值. 若 \(\delta=\varepsilon\)，则 \(|4\delta-\varepsilon|=3\)；若 \(\delta=-\varepsilon\)，则 \(|4\delta-\varepsilon|=5\). 所以
\[
|PQ|_{\min}=\sqrt{3^2+3}=2\sqrt{3}
\]
故选 C.
\end{solution}

\begin{problem}
已知椭圆 \(C: \frac{x^2}{2} + y^2 = 1\) 的右焦点为 \(F\)，右准线 \(l\)，点 \(A \in l\)，线段 \(AF\) 交 \(C\) 于点 \(B\). 若 \(\overrightarrow{FA} = 3\overrightarrow{FB}\)，则 \(|\overrightarrow{AF}| =\)（\quad）

\choices
    {\(\sqrt{2}\)}
    {\(2\)}
    {\(\sqrt{3}\)}
    {\(3\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
椭圆的长半轴为 \(\sqrt{2}\)，短半轴为 \(1\)，故半焦距为 \(c=1\)，右焦点为 \(F=(1,0)\)，右准线为 \(x=2\). 设 \(A=(2,t)\). 由 \(\overrightarrow{FA}=3\overrightarrow{FB}\)，得
\[
B=F+\frac{1}{3}(A-F)=\left(\frac{4}{3},\frac{t}{3}\right)
\]
因为 \(B\) 在椭圆上，所以
\[
\frac{(4/3)^2}{2}+\left(\frac{t}{3}\right)^2=1
\]
即 \(\frac{8}{9}+\frac{t^2}{9}=1\)，从而 \(t^2=1\). 因此
\[
|AF|=\sqrt{(2-1)^2+t^2}=\sqrt{2}
\]
故选 A.
\end{solution}

\section{填空题}

\begin{problem}
\((x - y)^{10}\) 的展开式中，\(x^{7}y^{3}\) 的系数与 \(x^{3}y^{7}\) 的系数之和等于\fillinblank{}.
\end{problem}

\begin{answer}
\(-240\)
\end{answer}

\begin{solution}
由二项式定理，\((x-y)^{10}\) 中 \(x^7y^3\) 的系数为 \((-1)^3\mathrm C_{10}^{3}=-120\)，而 \(x^3y^7\) 的系数为 \((-1)^7\mathrm C_{10}^{7}=-120\). 所以两系数之和为 \(-120-120=-240\).
\end{solution}

\begin{problem}
设等差数列 \(\{a_{n}\}\) 的前 \(n\) 项和为 \(S_{n}\)，若 \(S_{9} = 72\)，则 \(a_{2} + a_{4} + a_{9} = \)\fillinblank{}.
\end{problem}

\begin{answer}
\(24\)
\end{answer}

\begin{solution}
设等差数列的公差为 \(d\). 等差数列的前 \(9\) 项和等于首、末两项平均数乘以项数，所以 \(S_9=9a_5\). 由 \(S_9=72\) 得 \(a_5=8\). 又
\[
a_2+a_4+a_9=(a_1+d)+(a_1+3d)+(a_1+8d)=3(a_1+4d)=3a_5
\]
因此 \(a_2+a_4+a_9=3\times8=24\).
\end{solution}

\begin{problem}
已知 \(OA\) 为球 \(O\) 的半径，过 \(OA\) 的中点 \(M\) 且垂直于 \(OA\) 的平面截球面得到圆 \(M\)，若圆 \(M\) 的面积为 \(3\pi\)，则球 \(O\) 的表面积等于\fillinblank{}.
\end{problem}

\begin{answer}
\(16\pi\)
\end{answer}

\begin{solution}
设球 \(O\) 的半径为 \(R\). 因为 \(M\) 是 \(OA\) 的中点，所以球心 \(O\) 到截面平面的距离为 \(\frac{R}{2}\). 设截面圆半径为 \(r\)，由截面圆半径、球半径和球心到截面的距离组成的直角三角形，有
\[
r^2=R^2-\left(\frac{R}{2}\right)^2=\frac{3R^2}{4}
\]
又截面圆面积为 \(3\pi\)，即 \(\pi r^2=3\pi\)，所以 \(r^2=3\). 代入上式得 \(\frac{3R^2}{4}=3\)，从而 \(R^2=4\). 因此球的表面积为 \(4\pi R^2=16\pi\).
\end{solution}

\begin{problem}
若直线 \(m\) 被两平行线 \(l_{1}: x - y + 1 = 0\) 与 \(l_{2}: x - y + 3 = 0\) 所截得的线段的长为 \(2\sqrt{2}\)，则 \(m\) 的倾斜角可以是：

\begin{circlelist}
\item \(15^{\circ}\)；
\item \(30^{\circ}\)；
\item \(45^{\circ}\)；
\item \(60^{\circ}\)；
\item \(75^{\circ}\).
\end{circlelist}

其中正确答案的序号是\fillinblank{}.
\end{problem}

\begin{answer}
\circled{1}\circled{5}
\end{answer}

\begin{solution}
两平行线之间的距离为
\[
d=\frac{|3-1|}{\sqrt{1^2+(-1)^2}}=\sqrt{2}
\]
设直线 \(m\) 的倾斜角为 \(\alpha\). 两直线 \(l_1\)，\(l_2\) 的倾斜角均为 \(45^{\circ}\)，所以截线段长度满足
\[
\frac{d}{|\sin(\alpha-45^{\circ})|}=2\sqrt{2}
\]
由此得 \(|\sin(\alpha-45^{\circ})|=\frac{1}{2}\). 在 \(0^{\circ}\leqslant\alpha<180^{\circ}\) 内，给出的五个角中只有 \(\alpha=15^{\circ}\) 和 \(\alpha=75^{\circ}\) 满足条件. 因此正确答案的序号是 \(\circled{1}\circled{5}\).
\end{solution}

\section{解答题}

\begin{problem}
设等差数列 \(\{a_{n}\}\) 的前 \(n\) 项和为 \(S_{n}\)，公比是正数的等比数列 \(\{b_{n}\}\) 的前 \(n\) 项和为 \(T_{n}\)，已知 \(a_{1} = 1\)，\(b_{1} = 3\)，\(a_{3} + b_{3} = 17\)，\(T_{3} - S_{3} = 12\)，求 \(\{a_{n}\}\)，\(\{b_{n}\}\) 的通项公式.
\end{problem}

\begin{solution}
设等差数列 \(\{a_n\}\) 的公差为 \(d\)，等比数列 \(\{b_n\}\) 的公比为 \(q\)，由题设 \(q>0\). 由 \(a_1=1\)，\(b_1=3\)，得
\(a_3=1+2d\)，\(b_3=3q^2\)，\(S_3=3(1+d)\)，\(T_3=3(1+q+q^2)\).
将 \(a_3+b_3=17\) 代入，得 \(2d+3q^2=16\). 将 \(T_3-S_3=12\) 代入，得
\[
3(1+q+q^2)-3(1+d)=12
\]
即 \(d=q+q^2-4\). 代回前一个方程，得到
\[
2(q+q^2-4)+3q^2=16
\]
整理得 \(5q^2+2q-24=0\)，所以 \(q=2\) 或 \(q=-\frac{12}{5}\). 由于 \(q>0\)，只能取 \(q=2\)，于是 \(d=2\). 因此
\(a_n=a_1+(n-1)d=2n-1\)，\(b_n=b_1q^{n-1}=3\cdot2^{n-1}\).
\end{solution}

\begin{problem}
在 \(\triangle ABC\) 中，内角 \(A\)，\(B\)，\(C\) 的对边长分别为 \(a\)，\(b\)，\(c\)，已知 \(a^{2} - c^{2} = 2b\)，且 \(\sin B = 4\cos A \sin C\)，求 \(b\).
\end{problem}

\begin{solution}
设三角形外接圆半径为 \(R\). 由正弦定理，\(\sin B=\frac{b}{2R}\)，\(\sin C=\frac{c}{2R}\)，所以题设等式化为 \(b=4c\cos A\). 又由余弦定理，
\[
\cos A=\frac{b^2+c^2-a^2}{2bc}
\]
由 \(a^2-c^2=2b\) 得 \(b^2+c^2-a^2=b^2-2b=b(b-2)\)，从而
\[
b=4c\cdot\frac{b(b-2)}{2bc}=2(b-2)
\]
这里 \(b>0\)、\(c>0\)，所以前面的约分合法. 解得 \(b=4\)，这也是三角形边长所要求的唯一正值.
\end{solution}

\begin{problem}
如图，四棱锥 \(S - ABCD\) 中，底面 \(ABCD\) 为矩形，\(SD \perp\) 底面 \(ABCD\)，\(AD = \sqrt{2}\)，\(DC = SD = 2\)，点 \(M\) 在侧棱 \(SC\) 上，\(\angle ABM = 60^{\circ}\).

\begin{enumerate}
\item 证明：\(M\) 是侧棱 \(SC\) 的中点；
\item 求二面角 \(S - AM - B\) 的大小.
\end{enumerate}

\bitmapfigure[float=inline,max width=0.42\linewidth]{img/2009/syllabus_paper_1_liberal/q20_fig1.png}
\end{problem}

\begin{solution}
取空间直角坐标系，使 \(D=(0,0,0)\)，\(C=(2,0,0)\)，\(A=(0,\sqrt{2},0)\)，\(B=(2,\sqrt{2},0)\)，\(S=(0,0,2)\). 设 \(M\) 将 \(SC\) 按参数 \(t\) 分割，则
写成 \(M=(2t,0,2-2t)\)，其中 \(0\leqslant t\leqslant1\).
于是
\(\overrightarrow{BA}=(-2,0,0)\)，\(\overrightarrow{BM}=(2t-2,-\sqrt{2},2-2t)\).
由 \(\angle ABM=60^{\circ}\)，有
\[
\frac{\overrightarrow{BA}\cdot\overrightarrow{BM}}{|\overrightarrow{BA}||\overrightarrow{BM}|}
=\frac{1}{2}
\]
计算得
\[
\frac{4(1-t)}{2\sqrt{8(1-t)^2+2}}=\frac{1}{2}
\]
即 \(4(1-t)=\sqrt{8(1-t)^2+2}\). 由于 \(0\leqslant t\leqslant1\)，两边均非负，平方并化简得 \(8(1-t)^2=2\)，所以 \(1-t=\frac{1}{2}\)，即 \(t=\frac{1}{2}\). 因此 \(M\) 是 \(SC\) 的中点.

此时 \(M=(1,0,1)\). 设二面角 \(S-AM-B\) 的大小为 \(\theta\)，取公共棱 \(AM\) 的方向向量 \(\bs{u}=\overrightarrow{AM}=(1,-\sqrt{2},1)\)，则 \(|\bs{u}|^2=4\). 在平面 \(SAM\) 和平面 \(BAM\) 内，分别取指向 \(S\) 和 \(B\) 一侧且垂直于 \(AM\) 的向量

\(\bs{p}=\overrightarrow{AS}-\bs{u}=(-1,0,1)\)，\(\bs{q}=\overrightarrow{AB}-\frac{1}{2}\bs{u}=\left(\frac{3}{2},\frac{\sqrt{2}}{2},-\frac{1}{2}\right)\).

其中 \(\overrightarrow{AS}=(0,-\sqrt{2},2)\)，\(\overrightarrow{AB}=(2,0,0)\)，且
\(\frac{\overrightarrow{AS}\cdot\bs{u}}{|\bs{u}|^2}=1\)，\(\frac{\overrightarrow{AB}\cdot\bs{u}}{|\bs{u}|^2}=\frac{1}{2}\).
因而 \(\bs{p}\cdot\bs{q}=-2\)，\(|\bs{p}|=\sqrt{2}\)，\(|\bs{q}|=\sqrt{3}\)，所以
\[
\cos\theta=\frac{\bs{p}\cdot\bs{q}}{|\bs{p}||\bs{q}|}
=-\frac{2}{\sqrt{2}\sqrt{3}}=-\frac{\sqrt{6}}{3}
\]
因此 \(\theta=\arccos\left(-\frac{\sqrt{6}}{3}\right)\).
\end{solution}

\begin{problem}
甲，乙二人进行一次围棋比赛，约定先胜\(3\text{ 局}\)者获得这次比赛的胜利. 比赛结束，假设在一局中，甲获胜的概率为\(0.6\)，乙获胜的概率为\(0.4\)，各局比赛结果相互独立，已知前\(2\text{ 局}\)中，甲，乙各胜\(1\text{ 局}\).

\begin{enumerate}
\item 求再赛\(2\text{ 局}\)结束这次比赛的概率；
\item 求甲获得这次比赛胜利的概率.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 再赛 \(2\) 局后比赛结束，当且仅当甲连胜 \(2\) 局或乙连胜 \(2\) 局. 因此所求概率为 \(0.6^2+0.4^2=0.36+0.16=0.52\).
\item 甲获得比赛胜利有两类互斥情形：一是甲在再赛的前 \(2\) 局中连胜，概率为 \(0.6^2\)；二是再赛 \(2\) 局一胜一负，概率为 \(2\times0.6\times0.4\)，此时双方均胜 \(2\) 局，甲在下一局获胜的概率为 \(0.6\). 所以甲获胜的概率为
\[
0.6^2+\left(2\times0.6\times0.4\right)\times0.6=0.36+0.288=0.648
\]
\end{enumerate}
\end{solution}

\begin{problem}
已知函数 \( f(x) = x^4 - 3x^2 + 6 \).

\begin{enumerate}
\item 讨论 \( f(x) \) 的单调性；
\item 设点 \(P\) 在曲线 \(y = f(x)\) 上，若该曲线在点 \(P\) 处的切线 \(l\) 通过坐标原点，求 \(l\) 的方程.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item \(f'(x)=4x^3-6x=2x(2x^2-3)\). 令 \(a=\sqrt{\frac{3}{2}}\)，则临界点为 \(-a\)，\(0\)，\(a\). 根据 \(x\) 与 \(2x^2-3\) 的符号，\(f'(x)\) 在各区间的符号依次为负、正、负、正. 因此 \(f(x)\) 在 \(\left(-\infty,-\sqrt{\frac{3}{2}}\right)\) 上单调递减，在 \(\left(-\sqrt{\frac{3}{2}},0\right)\) 上单调递增，在 \(\left(0,\sqrt{\frac{3}{2}}\right)\) 上单调递减，在 \(\left(\sqrt{\frac{3}{2}},+\infty\right)\) 上单调递增.

\item 设 \(P=(x_0,f(x_0))\). 曲线在 \(P\) 处的切线方程为 \(y-f(x_0)=f'(x_0)(x-x_0)\). 因为切线经过原点，所以 \(f(x_0)=x_0f'(x_0)\)，即
\[
x_0^4-3x_0^2+6=x_0(4x_0^3-6x_0)
\]
整理得 \(3(x_0^2-2)(x_0^2+1)=0\). 由于 \(x_0\) 为实数，故 \(x_0=\pm\sqrt{2}\). 此时 \(f(x_0)=4\)，且 \(f'(x_0)=2x_0\)，所以相应切线为 \(y=2\sqrt{2}x\) 或 \(y=-2\sqrt{2}x\).
\end{enumerate}
\end{solution}

\begin{problem}
如图，已知抛物线 \(E: y^{2} = x\) 与圆 \(M: (x - 4)^{2} + y^{2} = r^{2}\)（\(r > 0\)）相交于 \(A\)，\(B\)，\(C\)，\(D\) 四个点.

\begin{enumerate}
\item 求 \(r\) 的取值范围；
\item 当四边形 \(ABCD\) 的面积最大时，求对角线 \(AC\)，\(BD\) 的交点 \(P\) 坐标.
\end{enumerate}

\FigureLayoutDeclare{img/2009/syllabus_paper_1_liberal/q23_fig1.png}{6.5cm}{13bp 10bp 13bp 10bp}
\bitmapfigure[max width=0.42\linewidth]{img/2009/syllabus_paper_1_liberal/q23_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由抛物线方程 \(y^2=x\)，将 \(y^2\) 代入圆方程得
\[
(x-4)^2+x=r^2
\]
即 \(x^2-7x+16-r^2=0\). 抛物线与圆有四个交点，当且仅当这个关于 \(x\) 的方程有两个不相等的正根. 其根的和为 \(7\)，根的积为 \(16-r^2\)，判别式为
\[
\Delta=49-4(16-r^2)=4r^2-15
\]
两个根不相等要求 \(4r^2-15>0\)，两个根均为正要求 \(16-r^2>0\). 结合 \(r>0\)，得到
\[
\frac{\sqrt{15}}{2}<r<4
\]
\item 设两个正根为 \(x_1\)，\(x_2\)，且 \(0<x_1<x_2\)，令 \(a=\sqrt{x_1}\)，\(b=\sqrt{x_2}\). 四个交点可表示为 \(A=(a^2,a)\)，\(B=(a^2,-a)\)，\(C=(b^2,-b)\)，\(D=(b^2,b)\). 四边形的两条平行边 \(AB\)，\(CD\) 长度分别为 \(2a\)，\(2b\)，两平行边之间的距离为 \(b^2-a^2\)，所以面积为
\[
S=\frac{2a+2b}{2}(b^2-a^2)=(a+b)^2(b-a)
\]
由韦达定理，\(a^2+b^2=x_1+x_2=7\). 令 \(u=ab\)，则 \(0<u<\frac{7}{2}\)，且
\[
S(u)=(7+2u)\sqrt{7-2u}
\]
求导得
\[
S'(u)=\frac{7-6u}{\sqrt{7-2u}}
\]
所以当 \(0<u<\frac{7}{6}\) 时 \(S'(u)>0\)，当 \(\frac{7}{6}<u<\frac{7}{2}\) 时 \(S'(u)<0\)，面积最大时 \(u=\frac{7}{6}\).

由图形关于 \(x\) 轴对称，交点 \(P\) 在 \(x\) 轴上. 直线 \(AC\) 上的点写成
\[
(a^2,a)+\lambda\bigl(b^2-a^2,-a-b\bigr)
\]
令其纵坐标为 \(0\)，得 \(\lambda=\frac{a}{a+b}\)，从而横坐标为
\[
a^2+\frac{a}{a+b}(b^2-a^2)=a^2+a(b-a)=ab=u
\]
因此对角线 \(AC\)，\(BD\) 的交点为 \(P=\left(\frac{7}{6},0\right)\).
\end{enumerate}
\end{solution}
