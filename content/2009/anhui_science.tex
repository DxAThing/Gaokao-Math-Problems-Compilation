\begin{problem}
在 \(\triangle ABC\) 中，\(\sin (C - A) = 1\)，\(\sin B = \frac{1}{3}\).

\begin{enumerate}
\item 求 \(\sin A\) 的值；
\item 设 \(AC = \sqrt{6}\)，求 \(\triangle ABC\) 的面积.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 因为 \(A\)，\(B\)，\(C\) 是三角形的内角，所以 \(C-A\in(-\pi,\pi)\). 由 \(\sin(C-A)=1\)，得 \(C-A=\frac{\pi}{2}\)，即 \(C=A+\frac{\pi}{2}\). 从而 \(B=\pi-A-C=\frac{\pi}{2}-2A\)，且 \(0<A<\frac{\pi}{4}\). 因此
\[
\frac{1}{3}=\sin B=\sin\left(\frac{\pi}{2}-2A\right)=\cos 2A.
\]
于是
\[
\sin^{2}A=\frac{1-\cos 2A}{2}=\frac{1}{3}.
\]
因为 \(A\) 是三角形内角，\(\sin A>0\)，所以 \(\sin A=\frac{\sqrt{3}}{3}\).
\item 设 \(b=AC=\sqrt{6}\)，\(c=AB\). 由正弦定理，
\[
\frac{c}{\sin C}=\frac{b}{\sin B}.
\]
又 \(\sin C=\sin\left(A+\frac{\pi}{2}\right)=\cos A=\sqrt{1-\sin^{2}A}=\sqrt{\frac{2}{3}}\)，所以
\[
c=\sqrt{6}\cdot\frac{\sqrt{2/3}}{1/3}=6.
\]
故
\[
S_{\triangle ABC}=\frac{1}{2}AC\cdot AB\cdot\sin A
=\frac{1}{2}\cdot\sqrt{6}\cdot6\cdot\frac{\sqrt{3}}{3}=3\sqrt{2}.
\]
\end{enumerate}
\end{solution}

\begin{problem}
某地有 \(A\)，\(B\)，\(C\)，\(D\) 四人先后感染了甲型 \(\mathrm{H1N1}\) 流感. 其中只有 \(A\) 到过疫区. \(B\) 肯定是受 \(A\) 感染的. 对于 \(C\)，因为难以断定他是受 \(A\) 还是受 \(B\) 感染的，于是假定他受 \(A\) 和受 \(B\) 感染的概率都是 \( \frac{1}{2} \). 同样也假定 \(D\) 受 \(A\)，\(B\) 和 \(C\) 感染的概率都是 \( \frac{1}{3} \). 在这种假定之下，\(B\)，\(C\)，\(D\) 中直接受 \(A\) 感染的人数 \(X\) 就是一个随机变量. 写出 \(X\) 的分布列（不要求写出计算过程），并求 \(X\) 的均值（即数学期望）.
\end{problem}

\begin{solution}
由于 \(B\) 肯定直接受 \(A\) 感染，先计入人数 \(1\). 设 \(I_C\) 表示事件“\(C\) 直接受 \(A\) 感染”，\(I_D\) 表示事件“\(D\) 直接受 \(A\) 感染”，则
\[
X=1+I_C+I_D.
\]
根据题意，\(P(I_C)=\frac{1}{2}\)，并且无论 \(C\) 由谁感染，\(D\) 直接受 \(A\) 感染的条件概率都是 \(\frac{1}{3}\). 因此
\[
P(X=1)=P(I_C=0,I_D=0)=\frac{1}{2}\cdot\frac{2}{3}=\frac{1}{3},
\]
\[
P(X=3)=P(I_C=1,I_D=1)=\frac{1}{2}\cdot\frac{1}{3}=\frac{1}{6}.
\]
当 \(X=2\) 时，有两种互斥情形：\(C\) 直接受 \(A\) 感染而 \(D\) 不直接受 \(A\) 感染，或 \(C\) 不直接受 \(A\) 感染而 \(D\) 直接受 \(A\) 感染. 所以
\[
P(X=2)=\frac{1}{2}\cdot\frac{2}{3}+\frac{1}{2}\cdot\frac{1}{3}=\frac{1}{2}.
\]
于是，\(X\) 的分布列由
\[
P(X=1)=\frac{1}{3},\qquad P(X=2)=\frac{1}{2},\qquad P(X=3)=\frac{1}{6}
\]
给出，且
\[
E(X)=1\cdot\frac{1}{3}+2\cdot\frac{1}{2}+3\cdot\frac{1}{6}=\frac{11}{6}.
\]
\end{solution}

如图，四棱锥 \(F - ABCD\) 的底面 \(ABCD\) 是菱形，其对角线 \(AC = 2\)，\(BD = \sqrt{2}\)，\(AE\)，\(CF\) 都与平面 \(ABCD\) 垂直，\(AE = 1\)，\(CF = 2\).

\bitmapfigure[max width=0.36\linewidth]{img/2009/anhui_science/q18_fig1.png}

\begin{solution}
\begin{enumerate}
\item 设 \(AC\) 与 \(BD\) 交于 \(O\)，过 \(O\) 作 \(OG\perp AF\)，垂足为 \(G\)，连接 \(BG\)，\(DG\). 因为菱形的对角线互相垂直，所以 \(BD\perp AC\). 又 \(CF\perp\) 平面 \(ABCD\)，从而 \(BD\perp CF\)，故 \(BD\perp\) 平面 \(ACF\)，进而 \(BD\perp AF\). 因为 \(BD\) 与 \(OG\) 是平面 \(BGD\) 内相交的两条直线，且 \(AF\perp BD\)，\(AF\perp OG\)，所以 \(AF\perp\) 平面 \(BGD\). 因此 \(\angle BGD\) 是二面角 \(B-AF-D\) 的平面角.

在直角三角形 \(AOF\) 中，\(AO=\frac{AC}{2}=1\)，\(CF\perp AC\)，且 \(AF=\sqrt{AC^{2}+CF^{2}}=2\sqrt{2}\). 由
\[
S_{\triangle AOF}=\frac{1}{2}AO\cdot CF=1=\frac{1}{2}AF\cdot OG,
\]
得 \(OG=\frac{1}{\sqrt{2}}\). 又 \(OB=OD=\frac{BD}{2}=\frac{1}{\sqrt{2}}\)，且 \(OB\perp OG\)，\(OD\perp OG\). 所以直角三角形 \(BOG\)，\(DOG\) 都是等腰直角三角形，\(\angle BGO=\angle OGD=\frac{\pi}{4}\). 因而
\[
\angle BGD=\angle BGO+\angle OGD=\frac{\pi}{2}.
\]
所以二面角 \(B-AF-D\) 的大小为 \(\frac{\pi}{2}\).
\item 连接 \(EB\)，\(EC\)，\(ED\)，设直线 \(AF\) 与直线 \(CE\) 交于 \(H\). 两个四棱锥的公共部分为四棱锥 \(H-ABCD\). 在平面 \(ACEF\) 内建立直角坐标系，以 \(AC\) 为横轴、过 \(A\) 且垂直于底面的直线为纵轴，则可取 \(A(0,0)\)，\(C(2,0)\)，\(E(0,1)\)，\(F(2,2)\). 于是直线 \(AF\) 的方程为 \(y=x\)，直线 \(CE\) 的方程为 \(y=1-\frac{x}{2}\)，联立得
\[
x=1-\frac{x}{2},\qquad x=\frac{2}{3},\qquad y=\frac{2}{3}.
\]
所以点 \(H\) 到平面 \(ABCD\) 的距离为 \(\frac{2}{3}\). 又
\[
S_{ABCD}=\frac{1}{2}AC\cdot BD=\frac{1}{2}\cdot2\cdot\sqrt{2}=\sqrt{2}.
\]
故公共部分的体积为
\[
V=\frac{1}{3}S_{ABCD}\cdot\frac{2}{3}=\frac{2\sqrt{2}}{9}.
\]
\end{enumerate}
\end{solution}

\begin{problem}
已知函数 \( f(x) = x - \frac{2}{x} + a(2 - \ln x)\)，\(a > 0\). 讨论 \( f(x) \) 的单调性.
\end{problem}

\begin{solution}
函数的定义域为 \(x>0\)，且
\[
f'(x)=1+\frac{2}{x^{2}}-\frac{a}{x}=\frac{x^{2}-ax+2}{x^{2}}.
\]

当 \(0<a<2\sqrt{2}\) 时，二次式 \(x^{2}-ax+2\) 的判别式为 \(a^{2}-8<0\)，且二次项系数为正，所以对一切 \(x>0\) 都有 \(f'(x)>0\). 因此 \(f(x)\) 在 \((0,+\infty)\) 上单调递增.

当 \(a=2\sqrt{2}\) 时，
\[
f'(x)=\frac{(x-\sqrt{2})^{2}}{x^{2}}\geqslant0.
\]
除 \(x=\sqrt{2}\) 外均有 \(f'(x)>0\)，所以 \(f(x)\) 仍在 \((0,+\infty)\) 上单调递增.

当 \(a>2\sqrt{2}\) 时，令
\[
x_{1}=\frac{a-\sqrt{a^{2}-8}}{2},\qquad x_{2}=\frac{a+\sqrt{a^{2}-8}}{2}.
\]
因为 \(x_{1}x_{2}=2>0\)，\(x_{1}+x_{2}=a>0\)，所以 \(0<x_{1}<x_{2}\). 此时
\[
x^{2}-ax+2=(x-x_{1})(x-x_{2}).
\]
由于 \(x^{2}>0\)，可得 \(f'(x)>0\) 在 \((0,x_{1})\) 和 \((x_{2},+\infty)\) 上成立，\(f'(x)<0\) 在 \((x_{1},x_{2})\) 上成立. 因而 \(f(x)\) 在 \((0,x_{1})\)，\((x_{2},+\infty)\) 上单调递增，在 \((x_{1},x_{2})\) 上单调递减.
\end{solution}

\begin{problem}
点 \(P(x_0, y_0)\) 在椭圆 \(\frac{x^2}{a^2} + \frac{y^2}{b^2} = 1\) （\(a > b > 0\)）上，\(x_0 = a \cos \beta\)，\(y_0 = b \sin \beta\)，\(0 < \beta < \frac{\pi}{2}\)，直线 \(l_2\) 与直线 \(l_1:\ \frac{x_0}{a^2} x + \frac{y_0}{b^2} y = 1\) 垂直，\(O\) 为坐标原点，直线 \(OP\) 的倾斜角为 \(\alpha\)，直线 \(l_2\) 的倾斜角为 \(\gamma\).

\begin{enumerate}
\item 证明：点 \(P\) 是椭圆 \(\frac{x^2}{a^2} + \frac{y^2}{b^2} = 1\) 与直线 \(l_1\) 的唯一交点；
\item 证明： \(\tan \alpha\)，\(\tan \beta\)，\(\tan \gamma\) 构成等比数列.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 任取椭圆上的点 \(Q(x,y)\)，令 \(u=\frac{x}{a}\)，\(v=\frac{y}{b}\). 则 \(u^{2}+v^{2}=1\). 又因为 \(x_{0}=a\cos\beta\)，\(y_{0}=b\sin\beta\)，所以直线 \(l_{1}\) 的方程可写为
\[
u\cos\beta+v\sin\beta=1.
\]
由柯西不等式，
\[
u\cos\beta+v\sin\beta
\leqslant\sqrt{u^{2}+v^{2}}\sqrt{\cos^{2}\beta+\sin^{2}\beta}=1.
\]
若点 \(Q\) 在直线 \(l_{1}\) 上，上式取等号. 由于两个向量 \((u,v)\)，\((\cos\beta,\sin\beta)\) 都是单位向量，等号成立当且仅当
\[
(u,v)=(\cos\beta,\sin\beta).
\]
于是 \(x=a\cos\beta=x_{0}\)，\(y=b\sin\beta=y_{0}\)，即 \(Q=P\). 因此点 \(P\) 是椭圆与直线 \(l_{1}\) 的唯一交点.
\item 因为 \(0<\beta<\frac{\pi}{2}\)，所以 \(x_{0}>0\)，\(y_{0}>0\). 直线 \(OP\) 的斜率为
\[
\tan\alpha=\frac{y_{0}}{x_{0}}=\frac{b}{a}\tan\beta.
\]
直线 \(l_{1}\) 的斜率为 \(-\frac{x_{0}/a^{2}}{y_{0}/b^{2}}\)，所以与它垂直的直线 \(l_{2}\) 的斜率为
\[
\tan\gamma=\frac{a^{2}y_{0}}{b^{2}x_{0}}=\frac{a}{b}\tan\beta.
\]
从而
\[
\tan\alpha\tan\gamma=\left(\frac{b}{a}\tan\beta\right)\left(\frac{a}{b}\tan\beta\right)=\tan^{2}\beta.
\]
因此 \(\tan\alpha\)，\(\tan\beta\)，\(\tan\gamma\) 构成等比数列.
\end{enumerate}
\end{solution}

\begin{problem}
首项为正数的数列 \(\{a_n\}\) 满足 \(a_{n + 1} = \frac{1}{4} (a_n^2 + 3)\)，\(n\in \N_+\).

\begin{enumerate}
\item 证明：若 \(a_1\) 为奇数，则对一切 \(n \geqslant 2\)，\(a_n\) 都是奇数；
\item 若对一切 \(n \in \N_+\) 都有 \(a_{n+1} > a_n\)，求 \(a_1\) 的取值范围.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 若 \(a_n\) 为奇数，则存在整数 \(k\)，使得 \(a_n^{2}=8k+1\). 由递推关系式，
\[
a_{n+1}=\frac{a_n^{2}+3}{4}=\frac{8k+4}{4}=2k+1,
\]
所以 \(a_{n+1}\) 也是奇数. 由数学归纳法，若 \(a_1\) 为奇数，则对一切 \(n\geqslant2\)，\(a_n\) 都是奇数.
\item 由递推关系式，
\[
a_{n+1}-a_n=\frac{a_n^{2}-4a_n+3}{4}=\frac{(a_n-1)(a_n-3)}{4}.
\]
因此，\(a_{n+1}>a_n\) 等价于 \(a_n<1\) 或 \(a_n>3\).

若对一切正整数 \(n\) 都有 \(a_{n+1}>a_n\)，特别地，\(a_1>0\) 且 \(a_1\notin[1,3]\)，所以必须有
\[
a_1\in(0,1)\cup(3,+\infty).
\]

反之，若 \(0<a_1<1\)，则当 \(0<a_n<1\) 时，
\[
a_{n+1}-a_n>0,
\qquad
0<a_{n+1}=\frac{a_n^{2}+3}{4}<1.
\]
由归纳法，所有 \(a_n\) 都属于 \((0,1)\)，且 \(a_{n+1}>a_n\).

若 \(a_1>3\)，则当 \(a_n>3\) 时，
\[
a_{n+1}-a_n>0,
\qquad
a_{n+1}=\frac{a_n^{2}+3}{4}>\frac{9+3}{4}=3.
\]
同样由归纳法，所有 \(a_n\) 都大于 \(3\)，且 \(a_{n+1}>a_n\). 因此 \(a_1\) 的取值范围为
\[
\boxed{(0,1)\cup(3,+\infty)}.
\]
\end{enumerate}
\end{solution}
