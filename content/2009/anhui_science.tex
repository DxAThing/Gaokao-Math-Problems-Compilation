\begin{problem}

\item 求 \(\sin A\) 的值；
\item 设 \(AC = \sqrt{6}\)，求 \(\triangle ABC\) 的面积.
\end{enumerate}

\begin{problem}
某地有 \(A\)、\(B\)、\(C\)、\(D\) 四人先后感染了甲型 \(\mathrm{H1N1}\) 流感. 其中只有 \(A\) 到过疫区. \(B\) 肯定是受 \(A\) 感染的. 对于 \(C\)，因为难以断定他是受 \(A\) 还是受 \(B\) 感染的，于是假定他受 \(A\) 和受 \(B\) 感染的概率都是 \( \frac{1}{2} \). 同样也假定 \(D\) 受 \(A\)、\(B\) 和 \(C\) 感染的概率都是 \( \frac{1}{2} \). 在这种假定之下，\(B\)、\(C\)、\(D\) 中直接受 \(A\) 感染的人数 \(X\) 就是一个随机变量. 写出 \(X\) 的分布列(不要求写出计算过程)，并求 \(X\) 的均值(即数学期望).
\end{problem}

\begin{problem}
点 \(P(x_0, y_0)\) 在椭圆 \(\frac{x^2}{a^2} + \frac{y^2}{b^2} = 1\ (a > b > 0)\) 上，\(x_0 = a \cos \beta, y_0 = b \sin \beta, 0 < \beta < \frac{\pi}{2}\)，直线 \(l_1\) 与直线 \(\frac{x_0}{a^2} x + \frac{y_0}{b^2} y = 1\) 垂直，\(O\) 为坐标原点，直线 \(OP\) 的倾斜角为 \(\alpha\)，直线 \(l_2\) 的倾斜角为 \(\gamma\).

\begin{enumerate}
\item 证明：点 \(P\) 是椭圆 \(\frac{x^2}{a^2} + \frac{y^2}{b^2} = 1\) 与直线 \(l_1\) 的唯一交点；
\item 证明: \(\tan \alpha, \tan \beta, \tan \gamma\) 构成等比数列.
\end{enumerate}
\end{problem}

\begin{problem}

\begin{enumerate}
\item 证明：若 \(a_1\) 为奇数，则对一切 \(n \geqslant 2\)，\(a_n\) 都是奇数；
\item 若对一切 \(n \in \mathbf{N}_+\) 都有 \(a_{n+1} > a_n\)，求 \(a_1\) 的取值范围.
\end{enumerate}
\end{problem}
