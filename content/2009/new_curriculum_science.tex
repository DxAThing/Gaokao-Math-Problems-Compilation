\section{单选题}

\begin{problem}
已知集合 \(A = \{1, 3, 5, 7, 9\}\)，\(B = \{0, 3, 6, 9, 12\}\)，则 \(A \cap \complement_{\N} B =\)（\quad）

\choices
    {\(\{1, 5, 7\}\)}
    {\(\{3, 5, 7\}\)}
    {\(\{1, 3, 9\}\)}
    {\(\{1, 2, 3\}\)}
\end{problem}

\begin{answer}
A.
\end{answer}

\begin{solution}
全集为自然数集，因此
\(\complement_{\N}B=\{n\in\N\mid n\notin B\}\). 在集合 \(A\) 的元素中，\(3\)，\(9\) 属于 \(B\)，而 \(1\)，\(5\)，\(7\) 不属于 \(B\)，所以
\(A\cap\complement_{\N}B=\{1,5,7\}\).
故选 A.
\end{solution}

\begin{problem}
复数 \(\frac{3 + 2\i}{2 - 3\i} - \frac{3 - 2\i}{2 + 3\i} =\)（\quad）

\choices
    {\(0\)}
    {\(2\)}
    {\(-2\i\)}
    {\(2\)}
\end{problem}

\begin{answer}
\(2\i\)（原卷 D 项误印为 \(2\)）.
\end{answer}

\begin{solution}
有
\[
\frac{3+2\i}{2-3\i}=\frac{(3+2\i)(2+3\i)}{(2-3\i)(2+3\i)}=\frac{13\i}{13}=\i.
\]

同理，
\[
\frac{3-2\i}{2+3\i}=\frac{(3-2\i)(2-3\i)}{(2+3\i)(2-3\i)}=\frac{-13\i}{13}=-\i.
\]

所以原式为 \(\i-(-\i)=2\i\). 因此原卷选项 D 的印刷内容与运算结果不一致，按数学结果答案为 \(2\i\).
\end{solution}

对变量 \(x\)，\(y\) 有观测数据 \((x_{i}, y_{i})\)（\(i = 1, 2, \dots, 10\)），得散点图 1；对变量 \(u\)，\(v\) 有观测数据 \((u_{i}, v_{i})\)（\(i = 1, 2, \dots, 10\)），得散点图 2. 由这两个散点图可以判断（\quad）

    {变量 \(x\) 与 \(y\) 负相关，\(u\) 与 \(v\) 正相关}

\begin{problem}
双曲线 \(\frac{x^2}{4} - \frac{y^2}{12} = 1\) 的焦点到渐近线的距离为（\quad）

\choices
    {\(2 \sqrt{3}\)}
    {\(2\)}
    {\(\sqrt{3}\)}
    {\(1\)}
\end{problem}

\begin{answer}
A.
\end{answer}

\begin{solution}
双曲线的标准方程中 \(a^2=4\)，\(b^2=12\)，因此
\(a=2\)，\(b=2\sqrt3\)，且
\(c=\sqrt{a^2+b^2}=4\). 其焦点为 \((4,0)\) 和 \((-4,0)\)，渐近线方程为
\(y=\pm\dfrac ba x=\pm\sqrt3x\).

以右焦点 \((4,0)\) 为例，点到渐近线 \(\sqrt3x-y=0\) 的距离为
\[
\frac{|\sqrt3\cdot4-0|}{\sqrt{(\sqrt3)^2+(-1)^2}}
=\frac{4\sqrt3}{2}=2\sqrt3.
\]
左焦点到相应渐近线的距离相同，所以所求距离为 \(2\sqrt3\)，故选 A.
\end{solution}

\begin{problem}
有四个关于三角函数的命题：\(p_1\)：\(\exists x \in \R\)，\(\sin^2 \frac{x}{2} + \cos^2 \frac{x}{2} = \frac{1}{2}\) ；\(p_2\)：\(\exists x,y \in \R\)，\(\sin (x - y) = \sin x - \sin y\) ；\(p_3\)：\(\forall x \in \left[0,\pi \right]\)，\(\sqrt{\frac{1 - \cos 2x}{2}} = \sin x\) ；\(p_4\)：\(\sin x = \cos y \Rightarrow x + y = \frac{\pi}{2}\). 其中假命题的是（\quad）

\choices
    {\(p_1, p_4\)}
    {\(p_2\)，\(p_4\)}
    {\(p_1\)，\(p_3\)}
    {\(p_2\)，\(p_3\)}
\end{problem}

\begin{answer}
A.
\end{answer}

\begin{solution}
对命题 \(p_1\)，由同角基本关系
\(\sin^2\dfrac{x}{2}+\cos^2\dfrac{x}{2}=1\)，不可能等于 \(\dfrac12\)，所以 \(p_1\) 是假命题.

对命题 \(p_2\)，取 \(x=y=0\)，则
\(\sin(x-y)=0=\sin x-\sin y\)，所以存在满足条件的 \(x,y\)，\(p_2\) 是真命题.

对命题 \(p_3\)，有
\[
\sqrt{\frac{1-\cos2x}{2}}=\sqrt{\sin^2x}=|\sin x|.
\]
当 \(x\in[0,\pi]\) 时，\(\sin x\geqslant0\)，故 \(|\sin x|=\sin x\)，所以 \(p_3\) 是真命题.

对命题 \(p_4\)，取 \(x=0\)，\(y=-\dfrac\pi2\)，则
\(\sin x=0=\cos y\)，但 \(x+y=-\dfrac\pi2\neq\dfrac\pi2\)，所以 \(p_4\) 是假命题.

因此假命题为 \(p_1,p_4\)，故选 A.
\end{solution}

\begin{problem}
设 \(x\)，\(y\) 满足 \(\begin{cases}
2x + y \geqslant 4,\\
x - y \geqslant -1,\\
x - 2y \leqslant 2,
\end{cases}\) 则 \(z = x + y\)（\quad）

\choices
    {有最小值 \(2\)，最大值 \(3\)}
    {有最小值 \(2\)，无最大值}
    {有最大值 \(3\)，无最小值}
    {既无最小值，也无最大值}
\end{problem}

\begin{answer}
B.
\end{answer}

\begin{solution}
由约束条件可得
\[
y\geqslant4-2x,\qquad y\leqslant x+1,\qquad y\geqslant\frac{x}{2}-1.
\]
要使可行的 \(y\) 存在，必须有 \(4-2x\leqslant x+1\)，即 \(x\geqslant1\). 当 \(1\leqslant x\leqslant2\) 时，\(4-2x\geqslant\dfrac{x}{2}-1\)，所以
\(z=x+y\geqslant x+(4-2x)=4-x\geqslant2\)，并且在 \((x,y)=(2,0)\) 时取到 \(z=2\). 当 \(x\geqslant2\) 时，
\(z\geqslant x+\left(\dfrac{x}{2}-1\right)=\dfrac32x-1\geqslant2\)，故 \(z\) 的最小值为 \(2\).

取 \(x\geqslant2\)，\(y=\dfrac{x}{2}-1\)，则三条约束均满足，而
\(z=\dfrac32x-1\) 可随 \(x\) 增大而无限增大，所以 \(z\) 无最大值.
故选 B.
\end{solution}

\begin{problem}
等比数列 \(\{a_{n}\}\) 的前 \(n\) 项和为 \(S_{n}\)，且 \(4a_{1}\)，\(2a_{2}\)，\(a_{3}\) 成等差数列. 若 \(a_{1} = 1\)，则 \(S_{4} =\)（\quad）

\choices
    {\(7\)}
    {\(8\)}
    {\(15\)}
    {\(16\)}
\end{problem}

\begin{answer}
C.
\end{answer}

\begin{solution}
设等比数列的公比为 \(q\). 由 \(a_1=1\)，有
\(a_2=q\)，\(a_3=q^2\). 由于 \(4a_1\)，\(2a_2\)，\(a_3\) 成等差数列，中间项的两倍等于两边项之和，因此
\[
2(2q)=4+q^2,
\]
即
\[
q^2-4q+4=(q-2)^2=0.
\]
所以 \(q=2\)，从而
\(S_4=1+2+2^2+2^3=15\).
故选 C.
\end{solution}

\begin{problem}
如图，正方体 \(ABCD - A_{1}B_{1}C_{1}D_{1}\) 的棱长为 \(1\)，线段 \(B_{1}D_{1}\) 上有两个动点 \(E\)，\(F\)，且 \(EF = \frac{\sqrt{2}}{2}\)，则下列结论中错误的是（\quad）

\bitmapfigure[max width=0.42\linewidth]{img/2009/new_curriculum_science/q8_fig1.png}

\choices
    {\(AC \perp BE\)}
    {\(EF \parallel\) 平面 \(ABCD\)}
    {三棱锥 \(A - BEF\) 的体积为定值}
    {异面直线 \(AE\)，\(BF\) 所成的角为定值}
\end{problem}

\begin{answer}
D.
\end{answer}

\begin{solution}
建立空间直角坐标系，令
\(A(0,0,0)\)，\(B(1,0,0)\)，\(C(1,1,0)\)，\(D(0,1,0)\)，并令
\(A_1(0,0,1)\)，\(B_1(1,0,1)\)，\(C_1(1,1,1)\)，\(D_1(0,1,1)\). 设
\[
E=(1-t,t,1),\qquad F=(1-s,s,1),\qquad 0\leqslant s,t\leqslant1.
\]
由 \(EF=\dfrac{\sqrt2}{2}\) 得
\(\sqrt2|s-t|=\dfrac{\sqrt2}{2}\)，即 \(|s-t|=\dfrac12\).

有
\[
\overrightarrow{AC}=(1,1,0),\qquad \overrightarrow{BE}=(-t,t,1),
\]
所以 \(\overrightarrow{AC}\cdot\overrightarrow{BE}=0\)，即 \(AC\perp BE\). 直线 \(EF\) 的方向向量为 \((s-t,t-s,0)\)，它平行于平面 \(ABCD\)，故结论 B 正确.

四面体 \(A-BEF\) 的体积为
\[
V=\frac16\left|\det\begin{pmatrix}
1&0&0\\
1-t&t&1\\
1-s&s&1
\end{pmatrix}\right|
=\frac{|t-s|}{6}=\frac1{12},
\]
所以结论 C 也正确.

最后考察结论 D. 当 \(t=0,s=\dfrac12\) 时，可取两异面直线的方向向量
\(\overrightarrow{AE}=(1,0,1)\)，\(\overrightarrow{BF}=\left(-\dfrac12,\dfrac12,1\right)\)，两向量夹角的余弦为
\(\dfrac{1}{2\sqrt3}\). 当 \(t=\dfrac12,s=1\) 时，方向向量可取
\(\overrightarrow{AE}=\left(\dfrac12,\dfrac12,1\right)\)，\(\overrightarrow{BF}=(0,1,1)\)，夹角的余弦为 \(\dfrac{\sqrt3}{2}\). 两者不相等，故该角不是定值.
因此错误的是 D.
\end{solution}

\begin{problem}
已知 \(O\)，\(N\)，\(P\) 在 \(\triangle ABC\) 所在平面内，且 \(\left|\overrightarrow{OA}\right| = \left|\overrightarrow{OB}\right| = \left|\overrightarrow{OC}\right|\)，\(\overrightarrow{NA} + \overrightarrow{NB} + \overrightarrow{NC} = \mathbf{0}\)，\(\overrightarrow{PA} \cdot \overrightarrow{PB} = \overrightarrow{PB} \cdot \overrightarrow{PC} = \overrightarrow{PC} \cdot \overrightarrow{PA}\)，则点 \(O\)，\(N\)，\(P\) 依次是 \(\triangle ABC\) 的（\quad）

\choices
    {重心，外心，垂心}
    {重心，外心，内心}
    {外心，重心，垂心}
    {外心，垂心，内心}
\end{problem}

\begin{answer}
C.
\end{answer}

\begin{solution}
由 \(|\overrightarrow{OA}|=|\overrightarrow{OB}|=|\overrightarrow{OC}|\)，点 \(O\) 到三角形 \(ABC\) 的三个顶点距离相等，所以 \(O\) 是三角形 \(ABC\) 的外心.

设点 \(A,B,C,N\) 的位置向量分别为 \(\mathbf a,\mathbf b,\mathbf c,\mathbf n\). 由
\[
\overrightarrow{NA}+\overrightarrow{NB}+\overrightarrow{NC}=\mathbf0
\]
得
\[
(\mathbf a-\mathbf n)+(\mathbf b-\mathbf n)+(\mathbf c-\mathbf n)=\mathbf0,
\]
即 \(\mathbf n=\dfrac{\mathbf a+\mathbf b+\mathbf c}{3}\)，所以 \(N\) 是三角形的重心.

令
\(\mathbf u=\overrightarrow{PA}\)，\(\mathbf v=\overrightarrow{PB}\)，\(\mathbf w=\overrightarrow{PC}\). 题设给出
\(\mathbf u\cdot\mathbf v=\mathbf v\cdot\mathbf w=\mathbf w\cdot\mathbf u\). 于是
\[
\overrightarrow{AP}\cdot\overrightarrow{BC}=(-\mathbf u)\cdot(\mathbf w-\mathbf v)=0,
\]
所以 \(AP\perp BC\). 同理可得 \(BP\perp CA\)，\(CP\perp AB\)，故 \(P\) 是三角形的垂心.

因此 \(O,N,P\) 依次是外心、重心、垂心，故选 C.
\end{solution}

\begin{problem}
一个棱锥的三视图如图，则该棱锥的全面积（单位：\(\mathrm{cm}^2\)）为（\quad）

\bitmapfigure[max width=0.48\linewidth]{img/2009/new_curriculum_science/q11_fig1.png}

\choices
    {\(48 + 12\sqrt{2}\)}
    {\(48 + 24\sqrt{2}\)}
    {\(36 + 12\sqrt{2}\)}
    {\(36 + 24\sqrt{2}\)}
\end{problem}

\begin{answer}
A.
\end{answer}

\begin{solution}
由三视图可知，该棱锥的底面是直角边长均为 \(6\mathrm{~cm}\) 的等腰直角三角形，顶点在底面上的射影为底面斜边的中点，棱锥的高为 \(4\mathrm{~cm}\).

底面斜边长为 \(6\sqrt2\mathrm{~cm}\)，底面积为
\[
S_\text{底}=\frac12\times6\times6=18\mathrm{~cm}^2.
\]
设顶点在底面的射影为 \(O\). 由于 \(O\) 是斜边中点，\(O\) 到两条直角边的距离均为 \(3\mathrm{~cm}\). 所以以任一条直角边为底的侧面，其斜高为
\[
\sqrt{4^2+3^2}=5\mathrm{~cm},
\]
两个这样的侧面面积之和为
\[
2\times\frac12\times6\times5=30\mathrm{~cm}^2.
\]
对于以斜边为底的侧面，因 \(O\) 在斜边上且 \(SO\perp\) 底面，\(SO=4\mathrm{~cm}\) 就是该侧面的高，其面积为
\[
\frac12\times6\sqrt2\times4=12\sqrt2\mathrm{~cm}^2.
\]
故棱锥的全面积为
\[
18+30+12\sqrt2=48+12\sqrt2\mathrm{~cm}^2.
\]
故选 A.
\end{solution}

\begin{problem}
用 \(\min \{a, b, c\}\) 表示 \(a\)，\(b\)，\(c\) 三个数中的最小值. 设 \(f(x) = \min \{2^x, x + 2, 10 - x\}\)（\(x \geqslant 0\)），则 \(f(x)\) 的最大值为（\quad）

\choices
    {\(4\)}
    {\(5\)}
    {\(6\)}
    {\(7\)}
\end{problem}

\begin{answer}
C.
\end{answer}

\begin{solution}
对任意 \(x\geqslant0\)，都有
\[
f(x)\leqslant\min\{x+2,10-x\}.
\]
而两个一次函数 \(x+2\) 与 \(10-x\) 的交点满足
\(x+2=10-x\)，解得 \(x=4\)，交点函数值为 \(6\). 因此
\(\min\{x+2,10-x\}\leqslant6\)，从而 \(f(x)\leqslant6\).

当 \(x=4\) 时，\(2^x=16\)，\(x+2=6\)，\(10-x=6\)，所以
\(f(4)=\min\{16,6,6\}=6\). 上界可以取到，故 \(f(x)\) 的最大值为 \(6\)，选 C.
\end{solution}

\begin{problem}
设已知抛物线 \(C\) 的顶点在坐标原点，焦点为 \(F(1,0)\)，直线 \(l\) 与抛物线 \(C\) 相交于 \(A\)，\(B\) 两点. 若 \(AB\) 的中点为 \((2,2)\)，则直线 \(l\) 的方程为\fillinblank{}.
\end{problem}

\begin{answer}
\(y=x\).
\end{answer}

\begin{solution}
抛物线的焦点为 \(F(1,0)\)，顶点为原点，所以其方程为
\(y^2=4x\). 设直线与抛物线的两个交点对应的参数分别为 \(t_1,t_2\). 抛物线的参数表示为
\[
(x,y)=(t^2,2t).
\]
由弦中点为 \((2,2)\)，得
\[
\frac{t_1^2+t_2^2}{2}=2,\qquad \frac{2t_1+2t_2}{2}=2,
\]
即
\[
t_1^2+t_2^2=4,\qquad t_1+t_2=2.
\]
于是
\(4=(t_1+t_2)^2-2t_1t_2=4-2t_1t_2\)，所以 \(t_1t_2=0\). 因而两交点参数为 \(0,2\)，对应的交点为 \((0,0)\)，\((4,4)\). 过这两点的直线方程为
\(y=x\).
\end{solution}

\begin{problem}
已知函数 \( y = \sin (\omega x + \varphi) \) （\(\omega > 0, -\pi \leqslant \varphi < \pi\)）的图象如图所示，则 \(\varphi = \)\fillinblank{}.

\bitmapfigure[max width=0.40\linewidth]{img/2009/new_curriculum_science/q14_fig1.png}
\end{problem}

\begin{answer}
\(\dfrac{9\pi}{10}\).
\end{answer}

\begin{solution}
由图可知，函数在 \(x=\dfrac{3\pi}{4}\) 处取得最小值，在 \(x=2\pi\) 处取得相邻的最大值. 因此从最小值到相邻最大值的横坐标距离为半个周期，即
\[
\frac{T}{2}=2\pi-\frac{3\pi}{4}=\frac{5\pi}{4},
\]
从而 \(T=\dfrac{5\pi}{2}\)，又 \(T=\dfrac{2\pi}{\omega}\)，所以
\(\omega=\dfrac45\).

在 \(x=2\pi\) 处函数取得最大值，故
\[
2\pi\omega+\varphi=\frac\pi2+2k\pi,\qquad k\in\mathbb Z.
\]
代入 \(\omega=\dfrac45\)，得
\[
\varphi=\frac\pi2-\frac{8\pi}{5}+2k\pi=-\frac{11\pi}{10}+2k\pi.
\]
结合 \(-\pi\leqslant\varphi<\pi\)，取 \(k=1\)，得到
\(\varphi=\dfrac{9\pi}{10}\).
\end{solution}

\begin{problem}
\(7\)名志愿者中安排\(6\)人在周六，周日两天参加社区公益活动. 若每天安排\(3\)人，则不同的安排方案共有\fillinblank{} 种.（用数字作答）
\end{problem}

\begin{answer}
\(140\).
\end{answer}

\begin{solution}
先从 \(7\) 名志愿者中选出周六参加活动的 \(3\) 人，再从余下的 \(4\) 人中选出周日参加活动的 \(3\) 人，剩下 \(1\) 人不参加. 因此不同安排方案数为
\[
\mathrm C_7^3\mathrm C_4^3=35\times4=140.
\]
\end{solution}

\begin{problem}
等差数列 \(\{a_{n}\}\) 前 \(n\) 项和为 \(S_{n}\). 已知 \(a_{m-1} + a_{m+1} - a_{m}^{2} = 0\)，\(S_{2m-1} = 38\)，则 \(m =\)\fillinblank{}.
\end{problem}

\begin{answer}
\(10\).
\end{answer}

\begin{solution}
设等差数列的公差为 \(d\). 则
\(a_{m-1}+a_{m+1}=2a_m\)，所以题设条件化为
\[
2a_m-a_m^2=0,
\]
即 \(a_m=0\) 或 \(a_m=2\).

等差数列中关于第 \(m\) 项对称的各项和均为 \(2a_m\)，因此
\[
S_{2m-1}=(2m-1)a_m=38.
\]
由于右边不为零，\(a_m\neq0\)，故 \(a_m=2\). 于是
\[
2(2m-1)=38,
\]
解得 \(m=10\).
\end{solution}

\begin{problem}
为了测量两山顶 \(M\)，\(N\) 间的距离，飞机沿水平方向在 \(A\)，\(B\) 两点进行测量. \(A\)，\(B\)，\(M\)，\(N\) 在同一个铅垂平面内（如示意图），飞机能够测量的数据有俯角和 \(A\)，\(B\) 间的距离，请设计一个方案，包括：

\begin{enumerate}
\item 指出需要测量的数据（用字母表示，并在图中标出）；
\item 用文字和公式写出计算 \(M\)，\(N\) 间距离的步骤.
\end{enumerate}

\bitmapfigure[max width=0.42\linewidth]{img/2009/new_curriculum_science/q17_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 记从 \(A\) 观察山顶 \(M,N\) 的俯角分别为 \(\alpha_1,\beta_1\)，从 \(B\) 观察山顶 \(M,N\) 的俯角分别为 \(\alpha_2,\beta_2\)，并测量 \(AB=d\). 将这些角和线段 \(AB\) 标在图中.
\item 在 \(\triangle ABM\) 中，\(\angle MAB=\alpha_1\)，\(\angle ABM=\alpha_2\)，所以 \(\angle AMB=\pi-\alpha_1-\alpha_2\). 由正弦定理，
\(AM=\dfrac{d\sin\alpha_2}{\sin(\alpha_1+\alpha_2)}\).

在 \(\triangle ABN\) 中，\(\angle BAN=\beta_1\)，且 \(\angle ABN=\pi-\beta_2\)，所以 \(\angle ANB=\beta_2-\beta_1\). 由正弦定理，
\(AN=\dfrac{d\sin\beta_2}{\sin(\beta_2-\beta_1)}\).

由图中两条视线在点 \(A\) 所成的角为 \(\alpha_1-\beta_1\)，再由余弦定理，
\(MN=\sqrt{AM^2+AN^2-2AM\cdot AN\cos(\alpha_1-\beta_1)}\).
\end{enumerate}
\end{solution}

\begin{problem}
如图，四棱锥 \(S - ABCD\) 的底面是正方形，每条侧棱的长都是底面边长的 \(\sqrt{2}\) 倍，\(P\) 为侧棱 \(SD\) 上的点.

\begin{enumerate}
\item 求证：\(AC \perp SD\)；
\item 若 \(SD \perp\) 平面 \(PAC\)，求二面角 \(P - AC - D\) 的大小；
\item 在上一问的条件下，侧棱 \(SC\) 上是否存在一点 \(E\)，使得 \(BE \parallel\) 平面 \(PAC\). 若存在，求 \(SE:EC\) 的值；若不存在，试说明理由.
\end{enumerate}

\bitmapfigure[max width=0.34\linewidth]{img/2009/new_curriculum_science/q19_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设正方形边长为 \(a\)，取空间直角坐标系，使
\(A(0,0,0)\)，\(B(a,0,0)\)，\(C(a,a,0)\)，\(D(0,a,0)\).

由于 \(SA=SB=SC=SD\)，点 \(S\) 到正方形四个顶点的距离相等，所以 \(S\) 在正方形四个顶点的垂直平分面交线上；该交线经过正方形中心且垂直于底面. 因而可取 \(S\) 在正方形中心的正上方. 由 \(SA=a\sqrt{2}\) 及正方形中心到顶点的距离为 \(\dfrac{a\sqrt{2}}{2}\)，得
\(S\left(\dfrac{a}{2},\dfrac{a}{2},\dfrac{a\sqrt{6}}{2}\right)\). 于是
\(\overrightarrow{AC}=(a,a,0)\)，\(\overrightarrow{SD}=\left(-\dfrac{a}{2},\dfrac{a}{2},-\dfrac{a\sqrt{6}}{2}\right)\)，从而
\(\overrightarrow{AC}\cdot\overrightarrow{SD}=-\dfrac{a^2}{2}+\dfrac{a^2}{2}=0\). 因此 \(AC\perp SD\).
\item 设 \(P=S+t\overrightarrow{SD}\)，其中 \(0\leqslant t\leqslant1\). 由 \(SD\perp\) 平面 \(PAC\)，知 \(SD\perp AP\). 计算得
\(\overrightarrow{AP}=\left(\dfrac{a(1-t)}{2},\dfrac{a(1+t)}{2},\dfrac{a\sqrt{6}(1-t)}{2}\right)\).

因此
\(\overrightarrow{SD}\cdot\overrightarrow{AP}=a^2\left(2t-\dfrac{3}{2}\right)=0\)，解得 \(t=\dfrac{3}{4}\). 取 \(a=1\) 计算二面角不影响角度，此时
\(\overrightarrow{AC}=(1,1,0)\)，\(\overrightarrow{AP}=\left(\dfrac18,\dfrac78,\dfrac{\sqrt6}{8}\right)\).

平面 \(PAC\) 的一个法向量为
\(\bs{n}=\overrightarrow{AC}\times\overrightarrow{AP}=\left(\dfrac{\sqrt6}{8},-\dfrac{\sqrt6}{8},\dfrac34\right)\)，而平面 \(DAC\) 的法向量可取 \(\bs{k}=(0,0,1)\). 故二面角 \(P-AC-D\) 的余弦为
\(\dfrac{|\bs{n}\cdot\bs{k}|}{|\bs{n}||\bs{k}|}=\dfrac{3/4}{\sqrt3/2}=\dfrac{\sqrt3}{2}\)，所以二面角的大小为 \(30^{\circ}\).
\item 仍取 \(a=1\). 设 \(E=S+u\overrightarrow{SC}\)，其中 \(0\leqslant u\leqslant1\)，则
\(E\left(\dfrac{1+u}{2},\dfrac{1+u}{2},\dfrac{\sqrt6(1-u)}{2}\right)\).

要使 \(BE\parallel\) 平面 \(PAC\)，只需使 \(\overrightarrow{BE}\cdot\bs{n}=0\). 代入
\(\overrightarrow{BE}=\left(\dfrac{u-1}{2},\dfrac{u+1}{2},\dfrac{\sqrt6(1-u)}{2}\right)\)，得
\(\overrightarrow{BE}\cdot\bs{n}=-\dfrac{\sqrt6}{8}+\dfrac{3\sqrt6}{8}(1-u)=0\)，解得 \(u=\dfrac23\).

所以存在这样的点 \(E\)，且 \(SE:EC=u:(1-u)=2:1\).
\end{enumerate}
\end{solution}

\begin{problem}
已知椭圆 \(C\) 的中心为直角坐标系 \(xOy\) 的原点，焦点在 \(x\) 轴上，它的一个顶点到两个焦点的距离分别是 \(7\) 和 \(1\).

\begin{enumerate}
\item 求椭圆 \(C\) 的方程；
\item 若 \(P\) 为椭圆 \(C\) 上的动点，\(M\) 为过 \(P\) 且垂直于 \(x\) 轴的直线上的点，\(\frac{|OP|}{|OM|} = \lambda\)，求点 \(M\) 的轨迹方程，并说明轨迹是什么曲线.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设椭圆的半长轴、半焦距分别为 \(a,c\). 由一个顶点到两个焦点的距离为 \(a+c\) 和 \(a-c\)，得
\(a+c=7\)，\(a-c=1\)，从而 \(a=4\)，\(c=3\). 因此 \(b^2=a^2-c^2=16-9=7\)，椭圆 \(C\) 的方程为
\(\dfrac{x^2}{16}+\dfrac{y^2}{7}=1\).
\item 设 \(P=(x,y_0)\)，\(M=(x,y)\). 由 \(P\) 在椭圆上，得
\(y_0^2=7-\dfrac{7x^2}{16}\)，且 \(-4\leqslant x\leqslant4\). 由于 \(PM\perp x\) 轴，\(P\) 与 \(M\) 的横坐标相同，条件 \(\dfrac{|OP|}{|OM|}=\lambda\) 给出
\[
x^2+y_0^2=\lambda^2(x^2+y^2).
\]

代入 \(y_0^2=7-\dfrac{7x^2}{16}\)，并整理得
\[
(16\lambda^2-9)x^2+16\lambda^2y^2=112,\qquad -4\leqslant x\leqslant4.
\]
由于 \(|OP|>0\)，必须有 \(\lambda>0\). 于是分情况讨论：

当 \(\lambda=\dfrac34\) 时，轨迹方程为 \(y^2=\dfrac{112}{9}\)，并且 \(-4\leqslant x\leqslant4\)，所以轨迹是两条平行于 \(x\) 轴的线段.

当 \(0<\lambda<\dfrac34\) 时，方程可写为
\[
16\lambda^2y^2-(9-16\lambda^2)x^2=112,\qquad -4\leqslant x\leqslant4,
\]
所以轨迹是中心在原点、实轴在 \(y\) 轴上的双曲线被 \(-4\leqslant x\leqslant4\) 截得的部分.

当 \(\dfrac34<\lambda<1\) 时，轨迹方程为椭圆
\[
\frac{x^2}{\dfrac{112}{16\lambda^2-9}}+\frac{y^2}{\dfrac{112}{16\lambda^2}}=1,
\qquad -4\leqslant x\leqslant4,
\]
即该椭圆被 \(-4\leqslant x\leqslant4\) 截得的部分.

当 \(\lambda\geqslant1\) 时，轨迹是完整的椭圆
\[
\frac{x^2}{\dfrac{112}{16\lambda^2-9}}+\frac{y^2}{\dfrac{112}{16\lambda^2}}=1.
\]
其中 \(\lambda=1\) 时该椭圆为原椭圆 \(\dfrac{x^2}{16}+\dfrac{y^2}{7}=1\).
\end{enumerate}
\end{solution}

\begin{problem}
已知函数 \( f(x) = (x^3 + 3x^2 + ax + b)\e^{-x} \).

\begin{enumerate}
\item 若 \(a = b = -3\)，求 \(f(x)\) 的单调区间；
\item 若 \(f(x)\) 在 \((-\infty, \alpha)\)，\((2,\beta)\) 单调增加，在 \((\alpha, 2)\)，\((\beta, +\infty)\) 单调减少，证明：\(\beta - \alpha < 6\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 当 \(a=b=-3\) 时，
\(f'(x)=\bigl(3x^2+6x-3-(x^3+3x^2-3x-3)\bigr)\e^{-x}=x(9-x^2)\e^{-x}\).

因为 \(\e^{-x}>0\)，所以 \(f'(x)\) 的符号由 \(x(9-x^2)=x(3-x)(3+x)\) 决定. 于是 \(f'(x)>0\) 的区间为 \((-\infty,-3)\) 和 \((0,3)\)，\(f'(x)<0\) 的区间为 \((-3,0)\) 和 \((3,+\infty)\). 因此，\(f(x)\) 在 \((-\infty,-3)\)、\((0,3)\) 上单调增加，在 \((-3,0)\)、\((3,+\infty)\) 上单调减少.
\item 对一般的 \(a,b\)，
\(f'(x)=\bigl[-x^3+(6-a)x+(a-b)\bigr]\e^{-x}\).

题设的单调性变化说明 \(\alpha<2<\beta\)，且 \(x=\alpha,2,\beta\) 都是导函数的零点. 因 \(\e^{-x}>0\)，有
\(-x^3+(6-a)x+(a-b)=-(x-\alpha)(x-2)(x-\beta)\).

比较两边的二次项系数，得 \(\alpha+2+\beta=0\)，即 \(\alpha+\beta=-2\). 由于 \(\beta>2\)，所以 \(\alpha=-2-\beta<-4\)，从而
\(\beta-\alpha=-2-2\alpha>6\)，这与题面要求证明的 \(\beta-\alpha<6\) 正好相反.

例如取 \(\alpha=-5\)，\(\beta=3\)，并取 \(a=-13\)，\(b=17\)，则
\(f'(x)=-(x+5)(x-2)(x-3)\e^{-x}\).

在 \((-\infty,-5)\) 和 \((2,3)\) 上，三次因子的符号为负，所以 \(f'(x)>0\)；在 \((-5,2)\) 和 \((3,+\infty)\) 上，三次因子的符号为正，所以 \(f'(x)<0\). 因此 \(f(x)\) 在 \((-\infty,-5)\)、\((2,3)\) 上单调增加，在 \((-5,2)\)、\((3,+\infty)\) 上单调减少，完全满足题设单调性条件，但 \(\beta-\alpha=8>6\). 因此按原题面，第二问的结论不成立，无法给出正确证明.
\end{enumerate}
\end{solution}

\begin{problem}
如图，已知 \(\triangle ABC\) 的两条角平分线 \(AD\) 和 \(CE\) 相交于 \(H\)，\(\angle B = 60^{\circ}\)，\(F\) 在 \(AC\) 上，且 \(AE = AF\).

\begin{enumerate}
\item 证明：\(B\)，\(D\)，\(H\)，\(E\) 四点共圆；
\item 证明：\(CE\) 平分 \(\angle DEF\).
\end{enumerate}

\bitmapfigure[max width=0.34\linewidth]{img/2009/new_curriculum_science/q22_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设 \(\angle A=2\alpha\)，则 \(\angle C=120^{\circ}-2\alpha\). 记 \(BC=a\)，\(CA=b\)，\(AB=c\). 由角平分线定理，
\(BD=\dfrac{ac}{b+c}\)，\(BE=\dfrac{ac}{a+b}\)，所以
\(\dfrac{BD}{BE}=\dfrac{a+b}{b+c}\).

由正弦定理，\(a:b:c=\sin2\alpha:\sin60^{\circ}:\sin(120^{\circ}-2\alpha)\)，从而
\(\dfrac{a+b}{b+c}=\dfrac{\sin2\alpha+\sin60^{\circ}}{\sin60^{\circ}+\sin(120^{\circ}-2\alpha)}=\dfrac{\sin(30^{\circ}+\alpha)}{\cos\alpha}\).

在 \(\triangle BDE\) 中，\(\angle DBE=60^{\circ}\). 设 \(\angle BDE=\theta\)，则 \(\angle BED=120^{\circ}-\theta\). 由正弦定理及上面的边比，
\[
\frac{\sin(120^{\circ}-\theta)}{\sin\theta}=\frac{\sin(30^{\circ}+\alpha)}{\cos\alpha}.
\]

由于 \(0<\alpha<60^{\circ}\)，且
\(2\dfrac{\sin(30^{\circ}+\alpha)}{\cos\alpha}-1=\sqrt3\tan\alpha\)，上式等价于
\(\tan\theta=\cot\alpha\). 又 \(0<\theta<120^{\circ}\)，且 \(\cot\alpha>0\)，所以 \(0<\theta<90^{\circ}\)，故 \(\theta=90^{\circ}-\alpha\)，从而 \(\angle BED=30^{\circ}+\alpha\).

又因为 \(H\) 是三角形的内心，且 \(E,H,C\) 共线，所以
\(\angle BHE=180^{\circ}-\angle BHC=180^{\circ}-(90^{\circ}+\alpha)=90^{\circ}-\alpha\).
因此 \(\angle BDE=\angle BHE\)，四点 \(B,D,H,E\) 共圆.
\item 在 \(\triangle AEF\) 中，\(AE=AF\)，且 \(\angle EAF=\angle A=2\alpha\)，所以
\(\angle AEF=90^{\circ}-\alpha\). 由于 \(A,E,B\) 共线，得
\(\angle BEF=180^{\circ}-\angle AEF=90^{\circ}+\alpha\).

在 \(\triangle BCE\) 中，\(CE\) 是角平分线，故
\(\angle BCE=60^{\circ}-\alpha\)，从而
\(\angle BEC=180^{\circ}-60^{\circ}-(60^{\circ}-\alpha)=60^{\circ}+\alpha\). 于是
\(\angle CEF=\angle BEF-\angle BEC=30^{\circ}\).

由第（1）问，\(\angle BED=30^{\circ}+\alpha\)，所以
\(\angle DEC=\angle BEC-\angle BED=30^{\circ}\). 故 \(\angle DEC=\angle CEF\)，即 \(CE\) 平分 \(\angle DEF\).
\end{enumerate}
\end{solution}

\begin{problem}
已知曲线 \(C_{1}: \begin{cases}
x = -4 + \cos t,\\
y = 3 + \sin t,
\end{cases}\)（\(t\) 为参数），\(C_{2}: \begin{cases}

\end{solution}
