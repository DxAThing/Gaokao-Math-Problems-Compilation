\section{单选题}

\begin{problem}
已知集合 \(M = \{x \mid -3 < x \leqslant 5\}\)，\(N = \{x \mid -5 < x < 5\}\)，则 \(M \cap N =\)（\quad）

\choices
    {\(\{x\mid -5 < x < 5\}\)}
    {\(\{x\mid -3 < x < 5\}\)}
    {\(\{x\mid -5 < x \leqslant 5\}\)}
    {\(\{x\mid -3 < x \leqslant 5\}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
集合的交集中的元素必须同时属于两个集合，因此
\(M\cap N=\{x\mid -3<x<5\}\).
所以选 B.
\end{solution}

\begin{problem}
已知复数 \(z = 1 - 2\i\)，那么 \(\frac{1}{\overline{z}} =\)（\quad）

\choices
    {\(\frac{\sqrt{5}}{5} + \frac{2\sqrt{5}}{5}\i\)}
    {\(\frac{\sqrt{5}}{5} -\frac{2\sqrt{5}}{5}\i\)}
    {\(\frac{1}{5} + \frac{2}{5}\i\)}
    {\(\frac{1}{5} -\frac{2}{5}\i\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
由 \(z=1-2\i\) 得 \(\overline{z}=1+2\i\)，所以
\[
\frac{1}{\overline{z}}=\frac{1}{1+2\i}=\frac{1-2\i}{(1+2\i)(1-2\i)}=\frac{1-2\i}{5}=\frac{1}{5}-\frac{2}{5}\i
\]
所以选 D.
\end{solution}

\begin{problem}
平面向量 \(\bs{a}\) 与 \(\bs{b}\) 的夹角为 \(60^{\circ}\)，\(\bs{a} = (2,0)\)，\(|\bs{b}| = 1\)，则 \(|\bs{a} + 2\bs{b}| =\)（\quad）

\choices
    {\(\sqrt{3}\)}
    {\(2\sqrt{3}\)}
    {\(4\)}
    {\(12\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
由向量数量积公式，\(\bs{a}\cdot\bs{b}=|\bs{a}|\,|\bs{b}|\cos60^{\circ}=2\cdot1\cdot\frac12=1\). 因此
\[
|\bs{a}+2\bs{b}|^2=|\bs{a}|^2+4|\bs{b}|^2+4\bs{a}\cdot\bs{b}=4+4+4=12
\]
由于模长为正，\(|\bs{a}+2\bs{b}|=2\sqrt{3}\). 所以选 B.
\end{solution}

\begin{problem}
已知圆 \(C\) 与直线 \(x - y = 0\) 及 \(x - y - 4 = 0\) 都相切，圆心在直线 \(x + y = 0\) 上，则圆 \(C\) 的方程为（\quad）

\choices
    {\((x + 1)^2 +(y - 1)^2 = 2\)}
    {\((x - 1)^2 +(y + 1)^2 = 2\)}
    {\((x - 1)^2 +(y - 1)^2 = 2\)}
    {\((x + 1)^2 +(y + 1)^2 = 2\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
两条平行切线 \(x-y=0\) 与 \(x-y-4=0\) 的距离为 \(\frac{4}{\sqrt2}=2\sqrt2\)，所以圆的半径为 \(\sqrt2\). 圆心到两条切线距离相等，故圆心在中线 \(x-y-2=0\) 上；又圆心在 \(x+y=0\) 上. 联立得圆心为 \((1,-1)\)，从而圆的方程为
\[
(x-1)^2+(y+1)^2=2
\]
所以选 B.
\end{solution}

\begin{problem}
从 \(5\) 名男医生，\(4\) 名女医生中选 \(3\) 名医生组成一个医疗小分队，要求其中男，女医生都有，则不同的组队方案共有（\quad）

\choices
    {\(70\) 种}
    {\(80\) 种}
    {\(100\) 种}
    {\(140\) 种}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
按男医生人数分类，三人中有男有女的方案数为
\[
\mathrm C_5^1\mathrm C_4^2+\mathrm C_5^2\mathrm C_4^1=5\cdot6+10\cdot4=70
\]
所以选 A.
\end{solution}

\begin{problem}
设等比数列 \(\{a_{n}\}\) 的前 \(n\) 项和为 \(S_{n}\)，若 \(\frac{S_{6}}{S_{3}} = 3\)，则 \(\frac{S_{9}}{S_{6}} =\)（\quad）

\choices
    {\(2\)}
    {\( \frac{7}{3} \)}
    {\( \frac{8}{3} \)}
    {\(3\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
设等比数列的公比为 \(q\). 前三项和不为零，且
\[
S_6=S_3+q^3S_3
\]
由 \(S_6/S_3=3\) 得 \(1+q^3=3\)，即 \(q^3=2\). 又
\[
S_9=S_6+q^6S_3=(3+4)S_3=7S_3
\]
所以 \(\frac{S_9}{S_6}=\frac73\). 因此选 B.
\end{solution}

\begin{problem}
曲线 \(y = \frac{x}{x - 2}\) 在点 \((1,-1)\) 处的切线方程为（\quad）

\choices
    {\(y = x - 2\)}
    {\(y = -3x + 2\)}
    {\(y = 2x - 3\)}
    {\(y = -2x + 1\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
函数 \(y=\frac{x}{x-2}\) 的导数为
\[
y'=\frac{(x-2)-x}{(x-2)^2}=-\frac{2}{(x-2)^2}
\]
在 \(x=1\) 处的切线斜率为 \(-2\)，故切线为
\[
y+1=-2(x-1)
\]
即 \(y=-2x+1\). 所以选 D.
\end{solution}

\begin{problem}
已知函数 \(f(x) = A\cos (\omega x + \varphi)\) 的图象如图所示，\(f\left(\frac{\pi}{2}\right) = -\frac{2}{3}\)，则 \(f(0) =\)（\quad）

\bitmapfigure[max width=9.4cm]{img/2009/liaoning_science/q8_fig1.png}

\choices
    {\(-\frac{2}{3}\)}
    {\( \frac{2}{3} \)}
    {\(-\frac{1}{2}\)}
    {\(\frac{1}{2}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
由图像可知，\(x=\frac{7\pi}{12}\) 和 \(x=\frac{11\pi}{12}\) 是相邻零点，零点间距为 \(\frac\pi3\)，所以半个周期为 \(\frac\pi3\)，从而 \(\omega=3\). 在 \(x=\frac{7\pi}{12}\) 处函数由负值向正值穿过横轴，可写成
\[
f(x)=A\sin\left(3x-\frac{7\pi}{4}\right)
\]
代入 \(f\left(\frac\pi2\right)=-\frac23\)，得
\[
-A\sin\frac\pi4=-\frac23
\]
所以 \(A=\frac{2\sqrt2}{3}\). 因此
\[
f(0)=A\sin\left(-\frac{7\pi}{4}\right)=\frac{2\sqrt2}{3}\cdot\frac{\sqrt2}{2}=\frac23
\]
所以选 B.
\end{solution}

\begin{problem}
已知偶函数 \( f(x) \) 在区间 \([0, +\infty)\) 单调增加，则满足 \( f(2x - 1) < f\left(\frac{1}{3}\right) \) 的 \( x \) 取值范围是（\quad）

\choices
    {\(\left(\frac{1}{3}, \frac{2}{3}\right)\)}
    {\(\left[\frac{1}{3}, \frac{2}{3}\right)\)}
    {\(\left(\frac{1}{2}, \frac{2}{3}\right)\)}
    {\(\left[\frac{1}{2}, \frac{2}{3}\right)\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
由于 \(f\) 为偶函数，\(f(2x-1)=f(|2x-1|)\). 又 \(f\) 在 \([0,+\infty)\) 上单调增加，所以
\[
f(2x-1)<f\left(\frac13\right)\quad\Longleftrightarrow\quad |2x-1|<\frac13
\]
解得
\[
-\frac13<2x-1<\frac13\quad\Longleftrightarrow\quad \frac13<x<\frac23
\]
所以选 A.
\end{solution}

\begin{problem}
某店一个月的收入和支出总共记录了 \(N\) 个数据 \(a_1\)，\(a_2\)，\(\dots\)，\(a_N\)，其中收入记为正数，支出记为负数. 该店用下边的程序框图计算月总收入 \(S\) 和月净盈利 \(V\)，那么在图中空白的判断框和处理框中，应分别填入下列四个选项中的（\quad）

\texfigure[max width=0.60\linewidth,width=0.60\linewidth]{img_repaint/2009/liaoning_science/q10_fig1.tex}

\choices
    {\(A > 0\)，\(V = S - T\)}
    {\(A < 0\)，\(V = S - T\)}
    {\(A > 0\)，\(V = S + T\)}
    {\(A < 0\)，\(V = S + T\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
收入记为正数，因此当 \(A>0\) 时应把 \(A\) 累加到收入总额 \(S\)，判断框应为 \(A>0\). 支出记为负数，故 \(T\) 是所有支出数据之和，月净盈利为收入加上负的支出，即 \(V=S+T\). 所以选 C.
\end{solution}

\begin{problem}
正六棱锥 \(P - ABCDEF\) 中，\(G\) 为 \(PB\) 的中点，则三棱锥 \(D - GAC\) 与三棱锥 \(P - GAC\) 体积之比为（\quad）

\choices
    {\(1: 1\)}
    {\(1:2\)}
    {\(2: 1\)}
    {\(3:2\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
取正六边形外接圆半径为 \(1\)，设棱锥高为 \(h\)，建立直角坐标系，使底面在 \(z=0\) 上，令 \(A=(1,0,0)\)，\(B=\left(\frac12,\frac{\sqrt3}{2},0\right)\)，\(C=\left(-\frac12,\frac{\sqrt3}{2},0\right)\)，\(D=(-1,0,0)\)，\(P=(0,0,h)\). 则 \(G\) 为 \(PB\) 的中点，故 \(G=\left(\frac14,\frac{\sqrt3}{4},\frac h2\right)\). 由三点 \(G\)，\(A\)，\(C\) 可得平面 \(GAC\) 的方程为 \(x+\sqrt3y=1\). 点 \(D\) 到该平面的距离为 \(\frac{|-1-1|}{\sqrt{1+3}}=1\)，点 \(P\) 到该平面的距离为 \(\frac{|0-1|}{\sqrt{1+3}}=\frac12\).
两个三棱锥的底面同为 \(\triangle GAC\)，所以体积比等于高之比，即
\[
V_{D-GAC}:V_{P-GAC}=1:\frac12=2:1
\]
所以选 C.
\end{solution}

\begin{problem}
若 \(x_{1}\) 满足 \(2x + 2^{x} = 5\)，\(x_{2}\) 满足 \(2x + 2\log_{2}(x - 1) = 5\)，则 \(x_{1} + x_{2} =\)（\quad）

\choices
    {\( \frac{5}{2} \)}
    {\(3\)}
    {\( \frac{7}{2} \)}
    {\(4\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
令 \(h(x)=2x+2^x\)，则
\[
h'(x)=2+2^x\ln2>0
\]
所以 \(h(x)\) 在 \(\R\) 上严格增加，故 \(x_1\) 唯一. 令
\[
y=2^{x_1-1}=\frac52-x_1>0
\]
则 \(\log_2y=x_1-1\)，从而
\[
y+\log_2y=\left(\frac52-x_1\right)+(x_1-1)=\frac32
\]
而 \(x_2\) 满足的方程等价于
\[
(x_2-1)+\log_2(x_2-1)=\frac32
\]
函数 \(y+\log_2y\) 在 \((0,+\infty)\) 上严格增加，因此其正根唯一，故 \(x_2-1=y\). 所以
\[
x_1+x_2=x_1+1+y=x_1+1+\frac52-x_1=\frac72
\]
所以选 C.
\end{solution}

\section{填空题}

\begin{problem}
某企业有 \(3\) 个分厂生产同一种电子产品，第一，二，三分厂的产量之比为 \(1:2:1\)，用分层抽样方法（每个分厂的产品为一层）从 \(3\) 个分厂生产的电子产品中共取 \(100\) 件作使用寿命的测试，由所得的测试结果算得从第一，二，三分厂取出的产品的使用寿命的平均值分别为 \(980 \mathrm{~h}\)，\(1020 \mathrm{~h}\)，\(1032 \mathrm{~h}\)，则抽取的 \(100\) 件产品的使用寿命的平均值为\fillinblank{}\(\mathrm{h}\).
\end{problem}

\begin{answer}
\(1013\)
\end{answer}

\begin{solution}
产量之比为 \(1:2:1\)，分层抽样也按此比例抽取，所以三厂分别抽取 \(25\)、\(50\)、\(25\) 件. 样本平均寿命为加权平均数
\[
\frac{25\cdot980+50\cdot1020+25\cdot1032}{100}=1013
\]
因此填 \(1013\).
\end{solution}

\begin{problem}
等差数列 \(\{a_{n}\}\) 的前 \(n\) 项和为 \(S_{n}\)，且 \(6S_{5} - 5S_{3} = 5\)，则 \(a_{4} = \)\fillinblank{}.
\end{problem}

\begin{answer}
\(\frac{1}{3}\)
\end{answer}

\begin{solution}
设等差数列首项为 \(a_1\)，公差为 \(d\). 由等差数列前 \(n\) 项和公式，\(S_5=5a_3\)，\(S_3=3a_2\). 因而
\[
6S_5-5S_3=30a_3-15a_2=15(2a_3-a_2)=15a_4
\]
由题设 \(15a_4=5\)，所以 \(a_4=\frac13\).
\end{solution}

\begin{problem}
设某几何体的三视图如下（尺寸的长度单位为 \(\mathrm{m}\)）. 则该几何体的体积为\fillinblank{}\(\mathrm{m}^{3}\).

\bitmapfigure[max width=0.56\linewidth]{img/2009/liaoning_science/q15_fig1.png}
\end{problem}

\begin{answer}
\(4\)
\end{answer}

\begin{solution}
由三视图可知，该几何体是一个三棱锥. 取俯视图所示的三角形为底面，它的底和高分别为 \(4\mathrm{~m}\) 与 \(3\mathrm{~m}\)，故底面积为
\[
S=\frac12\cdot4\cdot3=6\mathrm{~m}^2
\]
由正视图和侧视图可知，三棱锥的高为 \(2\mathrm{~m}\). 因此体积为
\[
V=\frac13Sh=\frac13\cdot6\cdot2=4\mathrm{~m}^3
\]
所以填 \(4\).
\end{solution}

\begin{problem}
已知 \(F\) 是双曲线 \(\frac{x^2}{4} - \frac{y^2}{12} = 1\) 的左焦点，\(A(1,4)\)，\(P\) 是双曲线右支上的动点，则 \(|PF| + |PA|\) 的最小值为\fillinblank{}.
\end{problem}

\begin{answer}
\(9\)
\end{answer}

\begin{solution}
双曲线的焦距参数为 \(c=\sqrt{4+12}=4\)，左、右焦点分别为 \(F=(-4,0)\) 和 \(F_1=(4,0)\). 设 \(P=(x,y)\) 在右支上，则 \(x\geqslant2\)，且由双曲线方程得 \(y^2=3x^2-12\). 于是
\[
|PF_1|^2=(x-4)^2+y^2=4(x-1)^2
\]
所以 \(|PF_1|=2x-2\)，并且由双曲线定义 \(|PF|=|PF_1|+4\). 因此
\[
|PF|+|PA|=2x+2+|PA|
\]
下面证明 \(|PA|+2x\geqslant7\). 当 \(x\geqslant\frac72\) 时显然成立. 当 \(2\leqslant x<\frac72\) 时，\(7-2x>0\)，且
\[
|PA|^2-(7-2x)^2=(x-1)^2+(y-4)^2-(7-2x)^2=2(13x-4y-22)
\]
另一方面，利用 \(y^2=3x^2-12\)，有
\[
(13x-22)^2-(4y)^2=121\left(x-\frac{286}{121}\right)^2\geqslant0
\]
由于 \(x\geqslant2\)，\(13x-22>0\)，故 \(13x-22\geqslant4|y|\geqslant4y\). 从而 \(|PA|^2\geqslant(7-2x)^2\)，即 \(|PA|+2x\geqslant7\). 等号在 \(P=\left(\frac{286}{121},\frac{264}{121}\right)\) 时取得，所以
\[
|PF|+|PA|\geqslant9
\]
故最小值为 \(9\).
\end{solution}

\section{解答题}

\begin{problem}
如图，\(A\)，\(B\)，\(C\)，\(D\) 都在同一个与水平面垂直的平面内，\(B\)，\(D\) 为两岛上的两座灯塔的塔顶. 测量船于水面 \(A\) 处测得 \(B\) 点和 \(D\) 点的仰角分别为 \(75^{\circ}\)，\(30^{\circ}\)，于水面 \(C\) 处测得 \(B\) 点和 \(D\) 点的仰角均为 \(60^{\circ}\)，\(AC = 0.1 \mathrm{~km}\). 试探究图中 \(B\)，\(D\) 间距离与另外哪两点间距离相等，然后求 \(B\)，\(D\) 的距离.（计算结果精确到 \(0.01 \mathrm{~km}\)，\(\sqrt{2} \approx 1.414\)，\(\sqrt{6} \approx 2.449\)）

\bitmapfigure[max width=0.48\linewidth]{img/2009/liaoning_science/q17_fig1.png}
\end{problem}

\begin{solution}
设 \(AC=l=0.1\mathrm{~km}\). 由仰角可知，在三角形 \(ABC\) 中，\(\angle BAC=105^{\circ}\)，\(\angle BCA=60^{\circ}\)，所以 \(\angle ABC=15^{\circ}\). 在三角形 \(ACD\) 中，\(\angle CAD=30^{\circ}\)，\(\angle ACD=120^{\circ}\)，所以 \(\angle ADC=30^{\circ}\). 因此
\[
AB^2=l^2\frac{\sin^260^{\circ}}{\sin^215^{\circ}}=l^2(6+3\sqrt3)
\]
且由正弦定理 \(BC=l\frac{\sin75^{\circ}}{\sin15^{\circ}}=l(2+\sqrt3)\)，\(CD=l\). 又 \(\angle BCD=60^{\circ}\)，所以由余弦定理
\[
BD^2=BC^2+CD^2-2BC\cdot CD\cos60^{\circ}=l^2(6+3\sqrt3)=AB^2
\]
长度均为正，故 \(BD=BA\). 于是
\[
BD=AB=l\frac{\sin60^{\circ}}{\sin15^{\circ}}=0.1\cdot\frac{3\sqrt2+\sqrt6}{2}\approx0.33\mathrm{~km}
\]
\end{solution}

\begin{problem}
如图，已知两个正方形 \(ABCD\) 和 \(DCEF\) 不在同一平面内，\(M\)，\(N\) 分别为 \(AB\)，\(DF\) 的中点.
\begin{enumerate}
\item 若平面 \(ABCD\) \(\perp\) 平面 \(DCEF\)，求直线 \(MN\) 与平面 \(DCEF\) 所成角的正弦值；
\item 用反证法证明：直线 \(ME\) 与 \(BN\) 是两条异面直线.
\end{enumerate}

\bitmapfigure[max width=0.46\linewidth]{img/2009/liaoning_science/q18_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设正方形边长为 \(a\)，取 \(D=(a,0,0)\)，\(C=(0,0,0)\)，\(B=(0,0,a)\)，\(A=(a,0,a)\)，并令第二个正方形所在平面为 \(z=0\)，则 \(F=(a,a,0)\). 因此 \(M=\left(\frac a2,0,a\right)\)，\(N=\left(a,\frac a2,0\right)\).
直线 \(MN\) 的方向向量可取 \(\left(\frac a2,\frac a2,-a\right)\)，其长度为 \(a\sqrt{\frac32}\). 与平面 \(DCEF\) 所成角的正弦等于方向向量的竖直分量绝对值与向量长度之比，所以
\[
\sin\theta=\frac{a}{a\sqrt{3/2}}=\frac{\sqrt6}{3}
\]
\item 反设直线 \(ME\) 与 \(BN\) 共面. 设正方形边长为 \(a>0\)，以 \(C\) 为原点，取单位向量 \(\bs{e}\)，\(\bs{u}\)，\(\bs{v}\) 分别沿 \(CD\)，\(CB\)，\(CE\) 方向，则 \(\bs{e}\)，\(\bs{u}\)，\(\bs{v}\) 线性无关. 于是
\(C=0\)，\(D=a\bs{e}\)，\(B=a\bs{u}\)，\(A=a(\bs{e}+\bs{u})\)，\(E=a\bs{v}\)，\(F=a(\bs{e}+\bs{v})\). 因而 \(M=a\left(\frac12\bs{e}+\bs{u}\right)\)，\(N=a\left(\bs{e}+\frac12\bs{v}\right)\).
两直线的方向向量分别为 \(-\frac12\bs{e}-\bs{u}+\bs{v}\) 和 \(\bs{e}-\bs{u}+\frac12\bs{v}\)，它们不成比例，所以不平行. 共面且不平行则必相交. 若交点对应参数 \(t\)，\(s\)，则
\[
M+t(E-M)=B+s(N-B)
\]
比较 \(\bs{e}\)，\(\bs{u}\)，\(\bs{v}\) 的系数，得到
\[
\begin{cases}
\dfrac{1-t}{2}=s,\\
1-t=1-s,\\
t=\dfrac{s}{2}
\end{cases}
\]
后两式给出 \(t=s\) 与 \(t=s/2\)，所以 \(s=t=0\)，代入第一式却得 \(1/2=0\)，矛盾. 因此两直线不共面；结合它们不平行，直线 \(ME\) 与 \(BN\) 是两条异面直线.
\end{enumerate}
\end{solution}

\begin{problem}
某人向一目标射击 \(4\) 次，每次击中目标的概率为 \(\frac{1}{3}\). 该目标分为 \(3\) 个不同的部分，第一，二，三部分面积之比为 \(1:3:6\). 击中目标时，击中任何一部分的概率与其面积成正比.
\begin{enumerate}
\item 设 \(X\) 表示目标被击中的次数，求 \(X\) 的分布列；
\item 若目标被击中 \(2\) 次，\(A\) 表示事件“第一部分至少被击中 \(1\) 次或第二部分被击中 \(2\) 次”，求 \(P(A)\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 每次击中目标的概率为 \(p=\frac13\)，所以 \(X\sim B\left(4,\frac13\right)\). 于是
\[
P(X=k)=\mathrm C_4^k\left(\frac13\right)^k\left(\frac23\right)^{4-k}
\]
其中 \(k=0\)，\(1\)，\(2\)，\(3\)，\(4\).
因此分布列为
\begin{center}
\begin{tabular}{c|ccccc}
\(X\)&\(0\)&\(1\)&\(2\)&\(3\)&\(4\)\\ \hline
\(P(X=k)\)&\(\frac{16}{81}\)&\(\frac{32}{81}\)&\(\frac{8}{27}\)&\(\frac{8}{81}\)&\(\frac{1}{81}\)
\end{tabular}
\end{center}
\item 在已击中目标的条件下，击中第一、二、三部分的概率分别为 \(\frac1{10}\)，\(\frac3{10}\)，\(\frac6{10}\). 事件 \(A\) 中“第一部分至少被击中一次”和“第二部分被击中两次”互斥，故
\[
P(A)=\left[1-\left(\frac9{10}\right)^2\right]+\left(\frac3{10}\right)^2=\frac{19}{100}+\frac9{100}=0.28
\]
\end{enumerate}
\end{solution}

\begin{problem}
已知椭圆 \(C\) 过点 \(A\left(1, \frac{3}{2}\right)\)，两个焦点为 \((-1,0)\)，\((1,0)\).
\begin{enumerate}
\item 求椭圆 \(C\) 的方程；
\item \(E\)，\(F\) 是椭圆 \(C\) 上的两个动点. 如果直线 \(AE\) 的斜率与 \(AF\) 的斜率互为相反数，证明直线 \(EF\) 的斜率为定值，并求出这个定值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设椭圆方程为 \(\frac{x^2}{a^2}+\frac{y^2}{b^2}=1\)，则 \(a^2-b^2=1\). 代入 \(A\left(1,\frac32\right)\) 得
\[
\frac1{a^2}+\frac{9}{4b^2}=1
\]
令 \(a^2=u\)，则 \(b^2=u-1\)，从而
\[
\frac1u+\frac{9}{4(u-1)}=1\quad\Longleftrightarrow\quad 4u^2-17u+4=0
\]
解得 \(u=4\) 或 \(u=\frac14\). 因为 \(a^2>c^2=1\)，只能取 \(a^2=4\)，故 \(b^2=3\)，椭圆方程为
\[
\frac{x^2}{4}+\frac{y^2}{3}=1
\]
\item 设直线 \(AE\) 的斜率为 \(m\)，则直线 \(AF\) 的斜率为 \(-m\). 在斜率为 \(m\) 的直线上令 \(x=1+u\)，\(y=\frac32+mu\).
代入椭圆方程，因 \(u=0\) 对应点 \(A\)，另一个交点对应
\[
u_m=-\frac{\frac12+m}{\frac14+\frac{m^2}{3}}=-\frac{6(1+2m)}{3+4m^2}
\]
同理，点 \(F\) 对应
\[
u_{-m}=-\frac{6(1-2m)}{3+4m^2}
\]
由于直线 \(EF\) 已定义且 \(E\)，\(F\) 为不同点，\(m\ne0\)，并且
\[
u_m-u_{-m}=-\frac{24m}{3+4m^2}\ne0
\]
故
\[
\frac{y_E-y_F}{x_E-x_F}=\frac{m u_m-(-m)u_{-m}}{u_m-u_{-m}}=m\frac{u_m+u_{-m}}{u_m-u_{-m}}=\frac12
\]
所以直线 \(EF\) 的斜率为定值 \(\frac12\).
\end{enumerate}
\end{solution}

\begin{problem}
已知函数 \(f(x) = \frac{1}{2}x^2 - ax + (a - 1)\ln x\)，\(a > 1\).
\begin{enumerate}
\item 讨论函数 \(f(x)\) 的单调性；
\item 证明：若 \(a < 5\)，则对于任意 \(x_{1}\)，\(x_{2} \in (0, +\infty)\)，\(x_{1} \neq x_{2}\)，有 \(\dfrac{f(x_{1}) - f(x_{2})}{x_{1} - x_{2}} > -1\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 函数定义域为 \((0,+\infty)\)，且
\[
f'(x)=x-a+\frac{a-1}{x}=\frac{(x-1)(x-a+1)}{x}
\]
当 \(1<a<2\) 时，\(0<a-1<1\)，故 \(f'(x)\) 在 \((0,a-1)\)、\((1,+\infty)\) 上为正，在 \((a-1,1)\) 上为负. 因此 \(f(x)\) 在 \((0,a-1)\)、\((1,+\infty)\) 上单调增加，在 \((a-1,1)\) 上单调减少.

当 \(a=2\) 时，\(f'(x)=\frac{(x-1)^2}{x}\geqslant0\)，故 \(f(x)\) 在 \((0,+\infty)\) 上单调增加.

当 \(a>2\) 时，\(1<a-1\)，故 \(f'(x)\) 在 \((0,1)\)、\((a-1,+\infty)\) 上为正，在 \((1,a-1)\) 上为负. 因此 \(f(x)\) 在 \((0,1)\)、\((a-1,+\infty)\) 上单调增加，在 \((1,a-1)\) 上单调减少.
\item 令 \(g(x)=f(x)+x=\frac12x^2-(a-1)x+(a-1)\ln x\). 当 \(1<a<5\) 时，
\[
g'(x)=\frac{x^2-(a-1)x+(a-1)}{x}
\]
分子是关于 \(x\) 的二次式，其判别式为
\[
\Delta=(a-1)^2-4(a-1)=(a-1)(a-5)<0
\]
因二次式首项系数为正，分子恒正；又 \(x>0\)，所以 \(g'(x)>0\)，即 \(g(x)\) 在 \((0,+\infty)\) 上严格增加. 因此对任意不同的 \(x_1\)，\(x_2>0\)，
\[
\frac{f(x_1)-f(x_2)}{x_1-x_2}+1=\frac{g(x_1)-g(x_2)}{x_1-x_2}>0
\]
从而
\[
\frac{f(x_1)-f(x_2)}{x_1-x_2}>-1
\]
\end{enumerate}
\end{solution}

\section{选做题}

\begin{problem}
已知 \(\triangle ABC\) 中，\(AB = AC\)，\(D\) 是 \(\triangle ABC\) 外接圆劣弧 \(\widehat{AC}\) 上的点（不与点 \(A\)，\(C\) 重合），延长 \(BD\) 至 \(E\).

\begin{enumerate}
\item 求证：\(AD\) 的延长线平分 \(\angle CDE\)；
\item 若 \(\angle BAC = 30^{\circ}\)，\(\triangle ABC\) 中 \(BC\) 边上的高为 \(2 + \sqrt{3}\)，求 \(\triangle ABC\) 外接圆的面积.
\end{enumerate}

\bitmapfigure[max width=0.34\linewidth]{img/2009/liaoning_science/q22_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设 \(A'\) 为 \(AD\) 的延长线上满足 \(D\) 在 \(A\) 与 \(A'\) 之间的点. 按圆周上的顺序记弧为 \(\widehat{AD}\)，\(\widehat{DC}\)，\(\widehat{CB}\)，\(\widehat{BA}\). 因为 \(AB=AC\)，对应的弧相等，所以
\[
m(\widehat{BA})=m(\widehat{AD})+m(\widehat{DC})
\]
由圆周角定理，
\[
\angle CDA=\frac12\left[m(\widehat{CB})+m(\widehat{BA})\right]
\]
又因为 \(B\)，\(D\)，\(E\) 共线且 \(DE\) 是 \(DB\) 的延长线，三角形 \(ABD\) 的外角定理给出
\[
\angle ADE=\angle DAB+\angle ABD=\frac12\left[m(\widehat{DC})+m(\widehat{CB})+m(\widehat{AD})\right]
\]
利用上述弧的相等关系，得 \(\angle ADE=\angle CDA\). 由于 \(DA'\) 与 \(DA\) 为反向射线，有 \(\angle CDA'=180^{\circ}-\angle CDA\)，\(\angle A'DE=180^{\circ}-\angle ADE\)，故 \(\angle CDA'=\angle A'DE\). 因而 \(AD\) 的延长线平分 \(\angle CDE\).
\item 设从 \(A\) 向 \(BC\) 作高，垂足为 \(H\). 因为 \(AB=AC\)，所以 \(AH\) 也是中线和角平分线，\(AH=2+\sqrt3\)，\(\angle BAH=15^{\circ}\). 在直角三角形 \(ABH\) 中，
\[
BH=AH\tan15^{\circ}=(2+\sqrt3)(2-\sqrt3)=1
\]
所以 \(BC=2\). 由正弦定理 \(BC=2R\sin30^{\circ}=R\)，故外接圆半径 \(R=2\)，面积为 \(\pi R^2=4\pi\).
\end{enumerate}
\end{solution}

\begin{problem}
在直角坐标系 \(xOy\) 中，以 \(O\) 为极点，\(x\) 正半轴为极轴建立极坐标系，曲线 \(C\) 的极坐标方程为 \(\rho \cos \left(\theta - \frac{\pi}{3}\right) = 1\)，\(M\)，\(N\) 分别为 \(C\) 与 \(x\) 轴，\(y\) 轴的交点.
\begin{enumerate}
\item 写出 \(C\) 的直角坐标方程，并求 \(M\)，\(N\) 的极坐标；
\item 设 \(MN\) 的中点为 \(P\)，求直线 \(OP\) 的极坐标方程.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由 \(x=\rho\cos\theta\)、\(y=\rho\sin\theta\)，有
\[
\rho\cos\left(\theta-\frac\pi3\right)=\frac{x+\sqrt3y}{2}=1
\]
所以曲线的直角坐标方程为 \(x+\sqrt3y=2\). 与 \(x\) 轴交于 \(M=(2,0)\)，与 \(y\) 轴交于 \(N=\left(0,\frac{2\sqrt3}{3}\right)\)，故
\(M\left(2,0\right)\)，\(N\left(\frac{2\sqrt3}{3},\frac\pi2\right)\) 为它们的极坐标.
\item 中点为 \(P=\left(1,\frac1{\sqrt3}\right)\). 因此 \(OP\) 的斜率为 \(\frac1{\sqrt3}=\tan\frac\pi6\)，其极角为 \(\frac\pi6\). 直线包含两个方向，故极坐标方程为 \(\theta=\frac\pi6\)，其中 \(\rho\in\R\).
\end{enumerate}
\end{solution}

\begin{problem}
设函数 \(f(x) = |x - 1| + |x - a|\).
\begin{enumerate}
\item 若 \(a = -1\)，解不等式 \(f(x) \geqslant 3\)；
\item 如果 \(\forall x \in \R\)，\(f(x) \geqslant 2\)，求 \(a\) 的取值范围.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 当 \(a=-1\) 时，\(f(x)=|x-1|+|x+1|\). 分三段讨论：
\[
f(x)=
\begin{cases}
-2x,&x\leqslant-1,\\
2,&-1\leqslant x\leqslant1,\\
2x,&x\geqslant1
\end{cases}
\]
因此 \(f(x)\geqslant3\) 等价于 \(x\leqslant-\frac32\) 或 \(x\geqslant\frac32\)，解集为
\[
\left(-\infty,-\frac32\right]\cup\left[\frac32,+\infty\right)
\]
\item 对任意 \(x\)，由三角不等式
\[
f(x)=|x-1|+|x-a|\geqslant|(x-1)-(x-a)|=|a-1|
\]
当 \(x\) 位于 \(a\) 与 \(1\) 之间时等号成立，所以 \(f(x)\) 的最小值为 \(|a-1|\). 题设等价于 \(|a-1|\geqslant2\)，即
\[
a\leqslant-1\quad\text{或}\quad a\geqslant3
\]
故 \(a\in(-\infty,-1]\cup[3,+\infty)\).
\end{enumerate}
\end{solution}
