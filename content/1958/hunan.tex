\chapter{1958年湖南卷}

\begin{problem}
不用除法证明 \((2+x)^{2n-1}+x^{2n-1}+x+1\) 能被 \(x+1\) 整除（\(n\) 为正整数）.
\end{problem}

\begin{solution}
设 \(f(x)=(2+x)^{2n-1}+x^{2n-1}+x+1\)，则
\[
f(-1)=(2-1)^{2n-1}+(-1)^{2n-1}-1+1=1-1-1+1=0
\]
根据多项式 \(f(x)\) 能被 \(x-\alpha\) 整除之重要条件 \(f(\alpha)=0\)，故原式能被 \(x+1\) 整除.
\end{solution}

\begin{problem}
不解方程 \(6x^{2}-x-2=0\)，求作一个二次方程，使它的根是原方程的根的倒数.
\end{problem}

\begin{solution}
设原方程之两个根为 \(x_1\)，\(x_2\)，由根与系数的关系得
\[
\begin{cases}
x_1+x_2=\frac{1}{6},\\
x_1x_2=-\frac{1}{3}.
\end{cases}
\]
则
\[
\begin{cases}
\frac{1}{x_1}+\frac{1}{x_2}=\frac{x_2+x_1}{x_1x_2}=\frac{\frac{1}{6}}{-\frac{1}{3}}=-\frac{1}{2},\\
\frac{1}{x_1}\cdot\frac{1}{x_2}=\frac{1}{x_1x_2}=\frac{1}{-\frac{1}{3}}=-3.
\end{cases}
\]
故所求方程为 \(x^2+\frac{1}{2}x-3=0\)，或 \(2x^2+x-6=0\).
\end{solution}

\begin{problem}
求证 \(\sin 75^{\circ}=\frac{1}{4}\left( \sqrt{6}+\sqrt{2} \right)\).
\end{problem}

\begin{solution}
\begin{align*}
\sin 75^{\circ}
&= \sin \left( 45^{\circ}+30^{\circ} \right) 
= \sin 45^{\circ}\cos 30^{\circ}+\cos 45^{\circ}\sin 30^{\circ} \\
&= \frac{\sqrt{2}}{2}\cdot\frac{\sqrt{3}}{2}+\frac{\sqrt{2}}{2}\cdot\frac{1}{2} 
= \frac{\sqrt{6}}{4}+\frac{\sqrt{2}}{4} \\
&= \frac{1}{4}\left( \sqrt{6}+\sqrt{2} \right)
\end{align*}
\end{solution}

\begin{problem}
证明：设三角形的三个角成等差数列，最大角与最小角相差 \(30^{\circ}\)，则其三边之比顺次等于 \(2\sqrt{2}\)，\(2\sqrt{3}\)，\(\sqrt{6}+\sqrt{2}\) 之比.
\end{problem}

\begin{solution}
由题知三角形的最大角与最小角相差 \(30^{\circ}\)，且三个角成等差数列.
设三角形的一个角为 \(a\)，则另两个角必为 \(a-15^{\circ}\) 与 \(a+15^{\circ}\).
由三角形内角和定理得 \(a+\left( a-15^{\circ} \right)+\left( a+15^{\circ} \right)=180^{\circ}\)，
故 \(3a=180^{\circ}\)，于是 \(a=60^{\circ}\)，\(a-15^{\circ}=45^{\circ}\)，\(a+15^{\circ}=75^{\circ}\).
\bitmapfigure[max width=0.42\linewidth]{img/1958/hunan/q4_solution_fig1.png}
第 \(3\) 题已证得 \(\sin 75^{\circ}=\frac{1}{4}\left( \sqrt{6}+\sqrt{2} \right)\).
由正弦定理，\(\frac{a}{\sin A}=\frac{b}{\sin B}=\frac{c}{\sin C}\)，故
\begin{align*}
a:b:c
&=\sin 45^{\circ}:\sin 60^{\circ}:\sin 75^{\circ}\\
&=\frac{\sqrt{2}}{2}:\frac{\sqrt{3}}{2}:\frac{\sqrt{6}+\sqrt{2}}{4}\\
&=2\sqrt{2}:2\sqrt{3}:\left( \sqrt{6}+\sqrt{2} \right)
\end{align*}
\end{solution}

\begin{problem}
一矩形内接于圆，它的面积是 \(48\text{平方厘米}\)，圆半径是 \(5\text{厘米}\)，求矩形的两邻边.
\end{problem}

\begin{solution}
\bitmapfigure[max width=0.42\linewidth]{img/1958/hunan/q5_solution_fig1.png}
如图，矩形 \(ABCD\) 内接于 \(\odot O\)，连接 \(AC\)，则 \(AC=10\text{厘米}\)，为 \(\odot O\) 的直径.
由题意知 \(AB\cdot BC=48\)，又 \(AB^{2}+BC^{2}=AC^{2}=100\).
由 \(\left( AB^{2}+BC^{2} \right)+\left( AB\cdot BC \right)\times 2\) 得 \(AB^{2}+2AB\cdot BC+BC^{2}=100+2\times 48\)，
即 \(\left( AB+BC \right)^{2}=196\)，
故 \(AB+BC=14\)（只取正值）.
联立 \(AB\cdot BC=48\) 与 \(AB+BC=14\) 解之得 \(AB=6\text{厘米}\)（或 \(8\text{厘米}\)），\(BC=8\text{厘米}\)（或 \(6\text{厘米}\)）.
\end{solution}

\begin{problem}
设 \(AD\)，\(BE\) 是由圆 \(O\) 的直径 \(AB\) 两端点所引的切线，\(DE\) 是过圆上任意一点 \(F\) 的切线，此切线与 \(AD\)，\(BE\) 交于 \(D\) 及 \(E\)，求证：\(AB\cdot BE=OF^{2}\).
\end{problem}

\begin{solution}
\bitmapfigure[max width=0.42\linewidth]{img/1958/hunan/q6_solution_fig1.png}
如图，连接 \(OD\)，\(OE\).
因为 \(AD\)，\(BE\)，\(DE\) 均与 \(\odot O\) 相切，
所以 \(\angle OAD=\angle OFE=\angle OBE=90^{\circ}\).
于是 \(\angle OBE+\angle OFE=180^{\circ}\)，
故 \(B\)，\(O\)，\(F\)，\(E\) 四点共圆（对角互补的四边形四顶点共圆）.
故 \(\angle AOF=\angle BEF\)（圆内接四边形外角等于内对角）.
又 \(\angle BEO=\angle FEO\)，\(\angle AOD=\angle FOD\)，
故 \(\angle AOD=\angle BEO\).
故直角三角形 \(BOE\) 相似于直角三角形 \(DAO\)，则 \(\frac{BE}{AO}=\frac{BO}{AD}\).
又 \(AO=BO=FO\)，即 \(AD\cdot BE=AO\cdot BO=OF^{2}\).
\end{solution}

\begin{problem}
正六棱锥底面的边长是 \(4\mathrm{cm}\)，侧面与底面所成二面角等于 \(45^{\circ}\)，求这棱锥体的体积.
\end{problem}

\begin{solution}
\bitmapfigure[max width=0.42\linewidth]{img/1958/hunan/q7_solution_fig1.png}
如图，正六棱锥 \(P-ABCDEF\) 的底面的边长
\[
AB=BC=CD=DE=EF=FA=4\mathrm{cm}
\]
高 \(PO\perp\) 底面 \(ABCDEF\)，\(O\) 为底面中心，在底面过 \(O\) 作 \(OM\perp AB\) 交 \(AB\) 于 \(M\)，连接 \(OA\)，\(OB\)，\(PM\)，则 \(PM\perp AB\)（三垂线定理）.
显然 \(\triangle AOB\) 为等边三角形，即 \(OA=OB=AB=4\mathrm{cm}\)，\(AM=MB=2\mathrm{cm}\).
在直角三角形 \(BOM\) 中，
\[
OM=\sqrt{OB^{2}-MB^{2}}=\sqrt{4^{2}-2^{2}}=\sqrt{12}=2\sqrt{3}\mathrm{cm}
\]
因为 \(PM\perp AB\)，\(OM\perp AB\)，
所以侧面与底面所成二面角之平面角 \(\angle PMO=45^{\circ}\).
在直角三角形 \(POM\) 中，
\[
\angle OPM=90^{\circ}-\angle PMO=90^{\circ}-45^{\circ}=45^{\circ}=\angle PMO
\]
故 \(PO=OM=2\sqrt{3}\mathrm{cm}\).
六棱锥的底面积
\[
S=6\times \frac{1}{2}AB\cdot OM=3\times 4\times 2\sqrt{3}=24\sqrt{3}\mathrm{cm}^{2}
\]
故六棱锥的体积 \(V=\frac{1}{3}S\cdot PO=\frac{1}{3}\times 24\sqrt{3}\times 2\sqrt{3}=48\mathrm{cm}^{3}\).
\end{solution}

\begin{problem}
证明二面角的平分面内任意一点到这二面角的两个面的距离相等.
\end{problem}

\begin{solution}
\bitmapfigure[max width=0.42\linewidth]{img/1958/hunan/q8_solution_fig1.png}
已知：平面 \(P\) 平分由平面 \(M\) 和平面 \(N\) 组成的二面角，\(A\) 为平面 \(P\) 上任意一点. \(AD\perp\) 平面 \(M\)，\(AE\perp\) 平面 \(N\).

求证：\(AD=AE\).

在平面 \(P\) 内过 \(A\) 作 \(AF\perp BC\) 交 \(BC\) 于 \(F\)，连接 \(DF\) 和 \(EF\)，则 \(DF\perp BC\)，\(EF\perp BC\)（三垂线定理），而 \(\angle DFE\) 为二面角的平面角.
因平面 \(P\) 平分平面 \(M\) 和平面 \(N\) 组成的二面角，故在平面 \(ADFE\) 内，\(\angle DFA=\angle EFA\)（两个二面角相等，则它们的平面角也相等）.
又 \(AD\perp\) 平面 \(M\)，\(AE\perp\) 平面 \(N\)（已知），
故 \(AD\perp DF\)，\(AE\perp EF\)（平面的垂线必垂直于平面上过垂足的任一直线）.
故 \(AD=DE\)（角平分线上的任一点到两边的距离相等）.
\end{solution}
