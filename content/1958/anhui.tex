\chapter{1958年安徽卷}

\begin{problem}
\begin{enumerate}[label=（\alph*）、]
\item 已知方程 \(x^{2}-2x+q=0\) 两根的差的平方等于 \(16\)，求 \(q\) 的值.
\item 计算 \(\log_{\frac{3}{2}}\frac{9}{4}+\left( 0.25 \right)^{-\frac{3}{2}}+\log_{2}\sin 135^{\circ}\).
\item 证明 \(\sin \left( \alpha+\beta \right)<\sin \alpha+\sin \beta\)（\(\alpha\)，\(\beta\) 都是锐角）.
\item 说明函数 \(y=x^{2}-4\) 的图像和函数 \(y=x^{2}\) 的图像对于坐标轴的位置有什么区别.
\item 夹在两个平行平面间的两条线段，它们的长的和是 \(28\text{寸}\)，它们在同一个平面上的射影是 \(9\text{寸}\) 和 \(5\text{寸}\)，求这两条线段的长.
\end{enumerate}

\end{problem}

\begin{solution}
（1）因为一元二次方程 \(x^{2}-2x+q=0\) 的两根差的平方等于判别式，所以 \(4-4q=16\)，得 \(q=-3\).

（2）因为 \(\log_{\frac{3}{2}}\frac{9}{4}=2\)，\(\left( 0.25 \right)^{-\frac{3}{2}}=8\)，\(\log_{2}\sin 135^{\circ}=-\frac{1}{2}\)，所以原式 \(=2+8-\frac{1}{2}=\frac{19}{2}\).

（3）因为 \(\alpha\)，\(\beta\) 都是锐角，所以 \(0<\sin \alpha,\sin \beta,\cos \alpha,\cos \beta<1\). 因而 \(\sin \left( \alpha+\beta \right)=\sin \alpha\cos \beta+\cos \alpha\sin \beta<\sin \alpha+\sin \beta\).

（4）\(y=x^{2}-4\) 的图像是 \(y=x^{2}\) 的图像向下平移 \(4\) 个单位，所以对称轴仍是 \(y\) 轴，顶点由 \(\left( 0,0 \right)\) 变为 \(\left( 0,-4 \right)\)，并与 \(x\) 轴交于 \(\left( -2,0 \right)\)，\(\left( 2,0 \right)\).

（5）设两条线段长分别为 \(x\) 和 \(y\)（\(x>y\)），则 \(x+y=28\)，且 \(x^{2}-9^{2}=y^{2}-5^{2}\). 所以 \(\left( x-y \right)\left( x+y \right)=56\)，得 \(x-y=2\)，故 \(x=15\)，\(y=13\).
\end{solution}

\begin{problem}
\begin{enumerate}[label=（\alph*）、]
\item 解方程组 \(\begin{cases}\log_{3}\left( 2x+8 \right)-\log_{3}y=1,\\ \sqrt{2x+14}-\sqrt{2y+7}=0.\end{cases}\)
\item 解方程 \(\left( \sqrt{2+\sqrt{3}} \right)^{x}+\left( \sqrt{2-\sqrt{3}} \right)^{x}=4\).
\end{enumerate}

\end{problem}

\begin{solution}
（1）由 \(\log_{3}\left( 2x+8 \right)-\log_{3}y=1\) 得 \(\log_{3}\frac{2x+8}{y}=1\)，所以 \(2x+8=3y\). 又由 \(\sqrt{2x+14}=\sqrt{2y+7}\) 得 \(2x+14=2y+7\)，即 \(x-y=-\frac{7}{2}\). 联立解得 \(y=1\)，\(x=-\frac{5}{2}\).

（2）设 \(t=\left( \sqrt{2+\sqrt{3}} \right)^{x}\)，则 \(\left( \sqrt{2-\sqrt{3}} \right)^{x}=t^{-1}\). 原方程化为 \(t+\frac{1}{t}=4\)，即 \(t^{2}-4t+1=0\)，解得 \(t=2\pm\sqrt{3}\). 又 \(\left( \sqrt{2+\sqrt{3}} \right)^{2}=2+\sqrt{3}\)，\(\left( \sqrt{2+\sqrt{3}} \right)^{-2}=2-\sqrt{3}\)，所以 \(x=2\) 或 \(x=-2\).
\end{solution}

\begin{problem}
设 \(\sin \alpha\) 为 \(\sin \theta\) 与 \(\cos \theta\) 的等差中项，\(\sin \beta\) 为 \(\sin \theta\) 与 \(\cos \theta\) 的等比中项，求证：\(2\cos 2\alpha=\cos 2\beta\).

\end{problem}

\begin{solution}
由题意，\(\sin \alpha=\frac{\sin \theta+\cos \theta}{2}\)，\(\sin \beta=\sqrt{\sin \theta\cos \theta}\). 所以 \(\sin^{2}\alpha=\frac{\left( \sin \theta+\cos \theta \right)^{2}}{4}=\frac{1+\sin 2\theta}{4}\)，从而 \(\cos 2\alpha=1-2\sin^{2}\alpha=\frac{1-\sin 2\theta}{2}\). 又 \(\sin^{2}\beta=\sin \theta\cos \theta=\frac{1}{2}\sin 2\theta\)，所以 \(\cos 2\beta=1-2\sin^{2}\beta=1-\sin 2\theta\). 因而 \(2\cos 2\alpha=\cos 2\beta\).
\end{solution}

\begin{problem}
已知三棱锥的底面积为 \(S\)，高为 \(H\)，过这个三棱锥的三个侧棱的中点作一截面，求过这个三棱锥的底面上任一点为顶点，以所作截面为底的新的三棱锥的体积.

\end{problem}

\begin{solution}
过三个侧棱的中点作的截面与底面平行，并且相似比为 \(1:2\)，所以截面面积为 \(\frac{S}{4}\). 该截面到原底面的距离为 \(\frac{H}{2}\)，故以该截面为底、底面上一点为顶点的新三棱锥体积为 \(V=\frac{1}{3}\cdot\frac{S}{4}\cdot\frac{H}{2}=\frac{SH}{24}\).
\end{solution}

\begin{problem}
在内接于圆的四边形 \(ABCD\) 中，对应边 \(AB\)、\(DC\) 的延长线交于 \(P\)，\(AD\)、\(BC\) 的延长线交于 \(Q\)，求证：

\begin{enumerate}[label=（\alph*）、]
\item \(PB\cdot CA=PC\cdot BD\).
\item 角 \(P\) 及角 \(Q\) 的平分线互相垂直.
\end{enumerate}

\end{problem}

\begin{solution}
（1）由 \(A,B,P\) 共线得 \(\angle PAC=\angle BAC\)，由 \(D,C,P\) 共线得 \(\angle PDB=\angle BDC\). 又因为 \(A,B,C,D\) 共圆，所以 \(\angle BAC=\angle BDC\)，从而 \(\angle PAC=\angle PDB\). 同时，由 \(D,C,P\) 共线得 \(\angle PCA=\angle DCA\)，由 \(A,B,P\) 共线得 \(\angle PBD=\angle DBA\). 又因为 \(A,B,C,D\) 共圆，所以 \(\angle DCA=\angle DBA\)，从而 \(\angle PCA=\angle PBD\). 因而 \(\triangle PAC\sim\triangle PDB\)，所以 \(\frac{CA}{BD}=\frac{PC}{PB}\)，即 \(PB\cdot CA=PC\cdot BD\).

（2）设 \(P\) 与 \(Q\) 的平分线分别为 \(\ell_{P}\) 和 \(\ell_{Q}\). 由圆周角定理，\(\angle P=\frac{1}{2}\left( \widehat{AD}+\widehat{BC} \right)\)，\(\angle Q=\frac{1}{2}\left( \widehat{AB}+\widehat{CD} \right)\)，所以 \(\angle P+\angle Q=180^{\circ}\). 因而 \(\ell_{P}\) 与 \(\ell_{Q}\) 所成的角为 \(\frac{1}{2}\left( \angle P+\angle Q \right)=90^{\circ}\)，故它们互相垂直.
\end{solution}
