\begin{problem}
\(\lg(x+7)-\lg(x+1)=\lg(x+3)\).
\end{problem}

\begin{solution}
\((x+7)=(x+1)(x+3)\)，\(x^{2}+3x-4=0\)，\(x_{1}=1\).
\(x_{2}=-4\)（不合题意舍去），所以 \(x=1\) 是方程的解.
\end{solution}

\begin{problem}
\(\sin 10x=\frac{1}{2}\)，求 \(x\).
\end{problem}

\begin{solution}
\(10x=\frac{\pi}{6}+2k\pi\)，\(x=\lg\left( \frac{\pi}{6}+2k\pi \right)\).（\(k=0\)，\(1\)，\(2\)，\(\cdots\)）
\(10x=\frac{5\pi}{6}+2k\pi\)，\(x=\lg\left( \frac{5\pi}{6}+2k\pi \right)\).
所以 \(x=\frac{\pi}{6}+2k\pi\)（\(k=0\)，\(\pm1\)，\(\pm2\)，\(\cdots\)）为原方程的解.
\end{solution}

\begin{problem}
\(\tan\frac{\theta}{2}=t\)，求证 \(\tan\theta=\frac{2t}{1-t^{2}}\)，\(\sin\theta=\frac{2t}{1+t^{2}}\)，\(\cos\theta=\frac{1-t^{2}}{1+t^{2}}\).
\end{problem}

\begin{solution}
\(\tan\theta=\dfrac{2\tan\frac{\theta}{2}}{1-\tan^{2}\frac{\theta}{2}}=\dfrac{2t}{1-t^{2}}\)，
\(\sin\theta=2\sin\frac{\theta}{2}\cos\frac{\theta}{2}=2\tan\frac{\theta}{2}\cos^{2}\frac{\theta}{2}
=\dfrac{2\tan\frac{\theta}{2}}{1+\tan^{2}\frac{\theta}{2}}=\dfrac{2t}{1+t^{2}}\)，
\(\cos\theta=\dfrac{\sin\theta}{\tan\theta}=\dfrac{1-t^{2}}{1+t^{2}}\).
\end{solution}

\begin{problem}
解方程 \(\sin x+\sqrt{3}\cos x=2\).
\end{problem}

\begin{solution}
\(2\left( \sin x\cdot\cos\frac{\pi}{3}+\sin\frac{\pi}{3}\cdot\cos x \right)=2\)，\(\sin\left( x+\frac{\pi}{3} \right)=1\)，\(x+\frac{\pi}{3}=\frac{\pi}{2}+2k\pi\).
所以 \(x=\frac{\pi}{6}+2k\pi\)（\(k=0\)，\(\pm1\)，\(\pm2\)，\(\cdots\)）为原方程的解.
\end{solution}

\begin{problem}
已知正六棱锥底边长为 \(a\)，高为 \(h\)，求侧面积.
\end{problem}

\begin{solution}
\(AB=\sqrt{h^{2}+a^{2}}\).
\(AH=\sqrt{h^{2}+a^{2}-\frac{a^{2}}{4}}=\sqrt{h^{2}+\frac{3}{4}a^{2}}\)，\(S_{\triangle ABC}=\frac{a}{2}\sqrt{h^{2}+\frac{3}{4}a^{2}}\).
故侧面积 \(S=6\cdot S_{\triangle ABC}=3a\sqrt{h^{2}+\frac{3}{4}a^{2}}\).

\end{solution}

\begin{problem}
已知长方体三度的和 \(a+b+c=13\)，对角线为 \(\sqrt{65}\)，求全面积.
\end{problem}

\begin{solution}
\(a^{2}+b^{2}+c^{2}=65\)，\(a+b+c=13\). 所以长方体的全面积为
\[
2(ab+bc+ac)=(a+b+c)^{2}-(a^{2}+b^{2}+c^{2})=169-65=104
\]
\end{solution}

\begin{problem}
已知：\(AB\) 为直径，\(DC\) 为弦，\(ABCD\) 内接于 \(\odot O\). 求证：

\begin{enumerate}
\item \(AC\cdot DB=AB\cdot DC+AD\cdot CB\).

\item \(AB\) 为方程 \(x^{3}-(a^{2}+b^{2}+c^{2})x-2abc=0\) 的根.
\end{enumerate}
\end{problem}

\begin{solution}
（1）作 \(\angle ADE=\angle BDC\).
由 \(\triangle ADE\sim\triangle BDC\)，得 \(AD\cdot CB=AE\cdot BD\).
由 \(\triangle CDE\sim\triangle BDA\)，得 \(AB\cdot DC=EC\cdot BD\).
相加得 \(AB\cdot DC+AD\cdot CB=(AE+EC)BD=AC\cdot DB\).

\bitmapfigure[max width=0.34\linewidth]{img/1958/jiangsu/q5_solution_fig1.png}

（2）由（1）知 \(AB\) 是直径.
\[
\begin{aligned}
AB\cdot b+c\cdot a&=\sqrt{AB^{2}-a^{2}}\cdot\sqrt{AB^{2}-c^{2}},\\
AB^{2}\cdot b^{2}+c^{2}\cdot a^{2}+2AB\cdot abc&=(AB^{2}-a^{2})(AB^{2}-c^{2}),\\
AB^{3}-(a^{2}+b^{2}+c^{2})AB-2abc&=0.
\end{aligned}
\]
所以 \(AB\) 是 \(x^{3}-(a^{2}+b^{2}+c^{2})x-2abc=0\) 的根.
\end{solution}
