\chapter{1958年江苏卷}

\begin{problem}
（1）\(\mathrm{Lg}(X+7)-\mathrm{Lg}(X+1)=\mathrm{Lg}(X+3)\).

（2）\(\sin 10X=\frac{1}{2}\)，求 \(X\).

（3）\(\mathrm{tg}\frac{\theta}{2}=t\)，求证 \(\mathrm{tg}\theta=\frac{2t}{1-t^{2}}\)，\(\sin\theta=\frac{2t}{1+t^{2}}\)，\(\cos\theta=\frac{1-t^{2}}{1+t^{2}}\).

\end{problem}

\begin{solution}
（1）解：\((X+7)=(X+1)(X+3)\)，\(X^{2}+3X-4=0\)，\(X_{1}=1\).

\(X_{2}=-4\)（不合题意舍去），所以 \(X=1\) 是方程的解. 

（2）解：\(10X=\frac{\pi}{6}+2K\pi\)，\(X=\frac{1}{10}\left( \frac{\pi}{6}+2K\pi \right)\)

\(10X=\frac{5\pi}{6}+2K\pi\)，\(X=\frac{1}{10}\left( \frac{5\pi}{6}+2K\pi \right)\)

（\(K=0,1,2,\cdots\)）

（3）证明：
\[
\mathrm{tg}\theta=\frac{2\mathrm{tg}\frac{\theta}{2}}{1-\mathrm{tg}^{2}\frac{\theta}{2}}=\frac{2t}{1-t^{2}}
\]

\[
\sin\theta=2\sin\frac{\theta}{2}\cdot\cos\frac{\theta}{2}=2\mathrm{tg}\frac{\theta}{2}\cdot\cos^{2}\frac{\theta}{2}
\]

\[
=\frac{2\mathrm{tg}\frac{\theta}{2}}{1+\mathrm{tg}^{2}\frac{\theta}{2}}=\frac{2t}{1+t^{2}}
\]

\[
\cos\theta=\frac{\sin\theta}{\mathrm{tg}\theta}=\frac{1-t^{2}}{1+t^{2}}
\]
\end{solution}

\begin{problem}
解方程 \(\sin X+\sqrt{3}\cos X=2\).

\end{problem}

\begin{solution}
解：

\[
2\left( \sin X\cdot\cos\frac{\pi}{3}+\sin\frac{\pi}{3}\cdot\cos X \right)=2
\]

\[
\sin\left( X+\frac{\pi}{3} \right)=1
\]

所以

\[
X+\frac{\pi}{3}=\frac{\pi}{2}+2K\pi
\]

所以 \(X=\frac{\pi}{6}+2K\pi\)（\(K=0,\pm1,\pm2,\cdots\)）为原方程的解.
\end{solution}

\begin{problem}
已知正六棱锥底边长为 \(a\)，高为 \(h\)，求侧面积.

\end{problem}

\begin{solution}
解：\(AB=\sqrt{h^{2}+a^{2}}\). 

\bitmapfigure[max width=0.33\linewidth,max totalheight=4.8cm]{img/1958/jiangsu/q3_solution_fig1.png}

\[
AH=\sqrt{h^{2}+a^{2}-\frac{a^{2}}{4}}=\sqrt{h^{2}+\frac{3}{4}a^{2}}
\]

\[
S_{\triangle ABC}=\frac{a}{2}\sqrt{h^{2}+\frac{3}{4}a^{2}}
\]

故侧面积 \(S=6\cdot S_{\triangle ABC}=3a\sqrt{h^{2}+\frac{3}{4}a^{2}}\).
\end{solution}

\begin{problem}
已知长方体三度的和 \(a+b+c=13\)，对角线为 \(\sqrt{65}\)，求全面积.

\end{problem}

\begin{solution}
解：\(a^{2}+b^{2}+c^{2}=65\)，\(a+b+c=13\). 所以长方体的全面积为：

\[
2(ab+bc+ac)=(a+b+c)^{2}-(a^{2}+b^{2}+c^{2})=169-65=104
\]
\end{solution}

\begin{problem}
已知：\(AB\)为直径，\(DC\)为弦，\(ABCD\)内接于 \(\odot O\). 求证：

（1）\(AC\cdot DB=AB\cdot DC+AD\cdot CB\).

（2）\(AB\)为方程 \(X^{3}-(a^{2}b^{2}+c^{2})X-2abc=0\) 的根.

\end{problem}

\begin{solution}
证明：

（1）作 \(\angle ADE=\angle BDC\).

由 \(\triangle ADE\sim\triangle BDC\)，得 \(AD\cdot CB=AE\cdot BD\).

由 \(\triangle CDE\sim\triangle BDA\)，得 \(AB\cdot DC=EC\cdot BD\).

相加得 \(AB\cdot DC+AD\cdot CB=(AE+EC)BD=AC\cdot DB\).

\bitmapfigure[max width=0.34\linewidth,max totalheight=4.8cm]{img/1958/jiangsu/q5_solution_fig1.png}

（2）由（1）知 \(AB\) 是直径.

\[
AB\cdot b+c\cdot a=\sqrt{AB^{2}-a^{2}}\cdot\sqrt{AB^{2}-c^{2}}
\]

\[
AB^{2}\cdot b^{2}+c^{2}\cdot a^{2}+2AB\cdot abc=(AB^{2}-a^{2})(AB^{2}-c^{2})
\]

\[
AB^{3}-(a^{2}+b^{2}+c^{2})AB-2abc=0
\]

所以 \(AB\) 是 \(X^{3}-(a^{2}+b^{2}+c^{2})X-2abc=0\) 的根.
\end{solution}
