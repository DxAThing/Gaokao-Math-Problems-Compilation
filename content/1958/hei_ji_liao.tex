\chapter{1958年黑吉辽卷}

\begin{problem}
已知：圆 \(O\) 内直交的二弦 \(AB\)，\(CD\) 相交于 \(P\). 求证：\(PA^2+PB^2+PC^2+PD^2\) 为一定值.

\bitmapfigure[max width=0.33\linewidth,max totalheight=4.8cm]{img/1958/hei_ji_liao/q1_fig1.png}

\end{problem}

\begin{solution}
连接 \( AC \)，则由勾股定理知：
\(AC^2=PC^2+PA^2\) 并且
\(BD^2=PB^2+PD^2\).
因此，\(AC^2+BD^2=PA^2+PB^2+PC^2+PD^2\).
作直径 \(CE\)，连 \(AE\)，\(BD\). 又在 \(\triangle AEC\) 中及 \(\triangle PBD\) 中，\( \angle CAB=\angle CDB \)，\( \angle CAB+\angle BAE=\angle CDB+\angle PBD=90^\circ \).
所以 \( \angle BAE=\angle ABD \)，即 \(\widehat{AED}=\widehat{EDB}\). 而 \(\widehat{AED}-\widehat{ED}=\widehat{BDE}-\widehat{ED}\)，即 \(\widehat{AE}=\widehat{BD}\)，所以 \(AE=BD\).
在直角 \(\triangle AEC\) 中，\(EC^2=AC^2+AE^2=AC^2+BD^2\). 因此 \(CE^2=PA^2+PB^2+PC^2+PD^2\). 其定值为圆的直径的平方.
\end{solution}

\begin{problem}
工地上有沙堆，其下底面为矩形，长，宽各为 \(a\text{米}\)，\(b\text{米}\)，其各侧面都与地平面夹成 \(45^\circ\) 角，上，下底距离为 \(h\text{米}\)，求沙堆体积.

\bitmapfigure[max width=0.34\linewidth,max totalheight=4.8cm]{img/1958/hei_ji_liao/q2_fig1.png}

\end{problem}

\begin{solution}
如图所示，上底面的长，宽各为 \(a-2h\)，\(b-2h\)，\(S_1=ab\)，\(S_2=\left( a-2h \right)\left( b-2h \right)\).
\(V=\frac13 h\left( S_1+\sqrt{S_1S_2}+S_2 \right)=\frac13 h\left[ ab+\sqrt{ab\left( a-2h \right)\left( b-2h \right)}+\left( a-2h \right)\left( b-2h \right) \right]\).
\end{solution}

\begin{problem}
求根式：\(\sqrt[3]{a}\) 与 \(\sqrt{5-a}\) 的乘积.

\end{problem}

\begin{solution}
\(\sqrt[3]{a}\cdot\sqrt{5-a}=\sqrt[6]{a^2}\cdot\sqrt[6]{\left( 5-a \right)^3}=\sqrt[6]{a^2\left( 5-a \right)^3}\).
\end{solution}

\begin{problem}
解方程：\(x^{1+\lg x}=20^{\log_{20}100}\).

\end{problem}

\begin{solution}
\(x^{1+\lg x}=100\)，两端取对数，得 \(\left( 1+\lg x \right)\lg x=2\).
所以 \(\lg^2x+\lg x-2=0\)，\(\left( \lg x+2 \right)\left( \lg x-1 \right)=0\).
所以 \(\lg x=1\)，\(\lg x=-2\). 故 \(x_1=10\)，\(x_2=\frac{1}{100}\)（对数定义）.
检验后知：\(10\)，\(\frac{1}{100}\) 都是原方程的根.
\end{solution}

\begin{problem}
解方程组：\(\begin{cases}
2x-5z=10-6a,\\
x-2y+3z=2a+1,\\
5x-6y+9z=23-10a.
\end{cases}\)
若 \(x\) 为某直角三角形的斜边，\(y\) 及 \(z\) 为直角边，试确定 \(a\) 的值.

\end{problem}

\begin{solution}
由第二式乘 \(3\) 得（4）\(3x-6y+9z=6a+3\).
由第三式减去上式，得（5）\(2x=23-10a-6a-3\).
所以 \(x=10-8a\).
把 \(x=10-8a\) 代入第一式，得（6）\(2\left( 10-8a \right)-5z=10-6a\).
所以 \(z=2-2a\).
把 \(x=10-8a\)，\(z=2-2a\) 代入第二式，得 \(10-8a-2y+3\left( 2-2a \right)=2a+1\)，\(y=\frac{15-16a}{2}\).
因为 \(x\) 为某直角三角形的斜边，\(y\) 及 \(z\) 为直角边，所以 \(x^2=y^2+z^2\). 即 \(\left( 10-8a \right)^2=\left( \frac{15-16a}{2} \right)^2+\left( 2-2a \right)^2\).
解方程 \( 16a^2+8\times16a-159=0 \)，得 \(a=\frac{-16\pm\sqrt{415}}{4}\).
\end{solution}

\begin{problem}
已知：\(\alpha\)，\(\beta\) 皆为锐角，求证：\(\sin \left( \alpha+\beta \right) \leq \sin \alpha+\sin \beta\).

\end{problem}

\begin{solution}
\(\sin \left( \alpha+\beta \right)=\sin \alpha\cdot\cos \beta+\cos \alpha\cdot\sin \beta\).
因为 \(\alpha\)，\(\beta\) 皆为锐角，所以 \(\sin \alpha<1\)，\(\sin \beta<1\)，\(\cos \alpha<1\)，\(\cos \beta<1\).
所以 \(\sin \alpha\cdot\cos \beta<\sin \alpha\)，\(\cos \alpha\cdot\sin \beta<\sin \beta\).
所以 \(\sin \left( \alpha+\beta \right)<\sin \alpha+\sin \beta\)，从而 \(\sin \left( \alpha+\beta \right)\leq \sin \alpha+\sin \beta\).
\end{solution}

\begin{problem}
已知，正方形 \(ABCD\) 边长为 \(a\)，连接相邻各边中点，得一正方形 \(A_1B_1C_1D_1\)，再连接 \(A_1B_1C_1D_1\) 相邻各边中点又得一个正方形 \(A_2B_2C_2D_2\)，如此相继下去，可得无数多个正方形. 求这些正方形面积的总和.

\end{problem}

\begin{solution}
设 \(n\) 个正方形面积总和为 \(S_n\).
\(S_1=a^2\)，\(S_2=\left( \frac{\sqrt2}{2} \right)^2a^2=\frac{a^2}{2}\)，\(S_3=\left( \frac{a}{2} \right)^2=\frac{a^2}{4}\).
\(\frac{S_2}{S_1}=q=\frac{\frac{a^2}{2}}{a^2}=\frac12\)
\(S_n=\frac{a^2}{1-\frac12}=2a^2\).
\end{solution}

\begin{problem}
解不等式：\(2\cos^2 x-\left( 2+\sqrt3 \right)\cos x+\sqrt3<0\).

\end{problem}

\begin{solution}
原式可变为 \(\left( \cos x-1 \right)\left( 2\cos x-\sqrt3 \right)<0\).
由 \(\begin{cases}
\cos x-1<0,\\
2\cos x-\sqrt3>0
\end{cases}\)
得 \(\begin{cases}
\cos x<1,\\
\cos x>\frac{\sqrt3}{2}
\end{cases}\)
即 \(\frac{\sqrt3}{2}<\cos x<1\).
由 \(\begin{cases}
\cos x-1>0,\\
2\cos x-\sqrt3<0
\end{cases}\)
得 \(\begin{cases}
\cos x>1,\\
\cos x<\frac{\sqrt3}{2}
\end{cases}\)
矛盾，无解.
当\(\cos x=\frac{\sqrt3}{2}\) 时，\(x=2k\pi\pm\frac{\pi}{6}\).
所以 \(2k\pi-\frac{\pi}{6}<x<2k\pi+\frac{\pi}{6}\)（\(k\) 为整数），且 \(x\neq 2k\pi\).
\end{solution}
