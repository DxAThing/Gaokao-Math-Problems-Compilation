\chapter{1958年湖北卷}

\begin{problem}
设有数列 \(1+2x+3x^{2}+4x^{3}+\cdots\).

\begin{enumerate}
\item 写出这个数列的第 \(n\) 项；
\item 写出这个数列的前 \(n\) 项和的公式；
\item 问这个和的公式当 \(x=1\) 时能否成立? 如能成立，说明理由；如不能成立，怎样求和?
\end{enumerate}

\end{problem}

\begin{solution}
解：（1）这个数列的通项公式为 \(a_n=nx^{n-1}\).

（2）由求和公式
\[
s_n=1+2x+3x^{2}+4x^{3}+\cdots+(n-1)x^{n-2}+nx^{n-1}+\cdots \tag{1}
\]
上述两边同乘 \(x\) 得：
\[
x s_n=x+2x^{2}+3x^{3}+4x^{4}+\cdots+(n-2)x^{n-2}+(n-1)x^{n-1}+nx^{n}+\cdots \tag{2}
\]
由（1）减（2）得
\[
s_n-x s_n=(1+x+x^{2}+x^{3}+\cdots+x^{n-2}+x^{n-1})-nx^{n}
\]
上式右边括号中是一个等比数列，首项为 \(1\)，公比为 \(x\)，末项为 \(x^{n-1}\)，由等比数列求和公式得：
\[
(1-x)s_n=\frac{1-x(x^{n-1})}{1-x}-nx^{n}
\]
\[
s_n=\frac{1-x^{n}}{(1-x)^{2}}-\frac{nx^{n}}{1-x}
\]
（3）当 \(x=1\) 时，以上求和公式中分母为 0，公式无意义，由（2）式得：
\[
S_n=1+2+3+4+\cdots+(n-2)+(n-1)+n
\]
这是一个等差数列，首项为 \(1\)，公差为 \(1\)，末项为 \(n\)，由等差数列求和公式得：
\[
S_n=\frac{n(n+1)}{2}
\]
\end{solution}

\begin{problem}
设 \(A,B,C\) 是 \(\triangle ABC\) 的三个内角，求证：\(\tan A+\tan B+\tan C=\tan A\cdot\tan B\cdot\tan C\).

\end{problem}

\begin{solution}
证明：因为 \(A+B+C=\pi\)，因此 \(C=\pi-(A+B)\). 故
\[
\tan C=\tan\left[\pi-(A+B)\right]=-\,\tan(A+B)
=-\left(\frac{\tan A+\tan B}{-\tan A\cdot\tan B}\right)
\]
即得
\[
\tan C\left(1-\tan A\cdot\tan B\right)=-(\tan A+\tan B)
\]
展开移项得
\[
\tan A+\tan B+\tan C=\tan A\cdot\tan B\cdot\tan C
\]
\end{solution}

\begin{problem}
解方程 \((\mathrm{Log}_x\sqrt{5})^{2}+\mathrm{Log}_x5\sqrt{5}+\frac{5}{4}=0\).

\end{problem}

\begin{solution}
解：\((\mathrm{Log}_x\sqrt{5})^{2}+\mathrm{Log}_x5\sqrt{5}+\frac{5}{4}=0\).

令 \(\mathrm{Log}_x\sqrt{5}=y\)，则
\[
\mathrm{Log}_x5\sqrt{5}=\mathrm{Log}_x(\sqrt{5})^{3}=3y
\]
于是原方程变为
\[
y^{2}+3y+\frac{5}{4}=0
\]
\[
4y^{2}+12y+5=0
\]
\[
(2y+1)(2y+5)=0
\]
故 \(y=-\frac{1}{2}\) 或 \(y=-\frac{5}{2}\).

于是所求的解是 \(\mathrm{Log}_x\sqrt{5}=-\frac{1}{2}\) 和 \(\mathrm{Log}_x\sqrt{5}=-\frac{5}{2}\)，即 \(x=\frac{1}{5}\) 和 \(x=\frac{1}{\sqrt[5]{5}}\).
\end{solution}

\begin{problem}
在 \(\triangle ABC\) 中，\(AD\) 是 \(BC\) 的中线，而且 \(\angle B\) 和 \(\angle CAD\) 互为余角，问 \(\triangle ABC\) 是怎样的三角形.

\end{problem}

\begin{solution}
（1）几何解法：

\bitmapfigure[max width=0.35\linewidth,max totalheight=4.8cm]{img/1958/hubei/q4_fig1.png}

延长 \(AD\) 交 \(\triangle ABC\) 的外接圆于 \(E\)，连接 \(BE\)，则 \(\angle CAD=\angle CBE\).
但 \(\angle CAD+\angle ABC=90^\circ\)，所以 \(\angle CBE+\angle ABC=90^\circ\).
因而 \(AE\) 为圆之直径，即圆心 \(O\) 在中线 \(AD\) 上.

（1）若圆心与中点 \(D\) 重合，则 \(BC\) 为圆之直径，即圆心 \(O\) 在中线 \(AD\) 上，此时三角形 \(ABC\) 是直角三角形.

（2）若圆心与中点 \(D\) 不重合，则 \(AD\perp BC\)，故 \(AD\) 为 \(BC\) 之中垂线，因此 \(AB\) 等于 \(AC\)，这时 \(\triangle ABC\) 是等腰三角形.

（2）三角解法：

设 \(D\) 为 \(BC\) 边的中点，则 \(BD=DC\).

\bitmapfigure[max width=0.34\linewidth,max totalheight=4.8cm]{img/1958/hubei/q4_fig2.png}

因为 \(\angle B+\angle CAD=90^\circ\)，所以 \(\angle C+\angle BAD=90^\circ\).
因而 \(\sin\angle CAD=\cos B\)，\(\sin\angle BAD=\cos C\).
在 \(\triangle ABC\) 与 \(\triangle ACD\) 中，依正弦定律分别有
\begin{align*}
\frac{AD}{\sin B}&=\frac{BD}{\sin\angle BAD},\\
\frac{AD}{\sin C}&=\frac{DC}{\sin\angle CAB}
\end{align*}
由于 \(BD=DC\)，因此 \(\sin B\cdot\sin\angle CAD=\sin C\cdot\sin\angle BAD\).
即 \(\sin B\cdot\cos B=\sin C\cos C\).
因此 \(\sin 2B=\sin 2C\).
于是 \(2\angle B=2\angle C\) 或 \(180^\circ-2\angle B=2\angle C\).
即得 \(\angle B=\angle C\) 或 \(\angle B+\angle C=90^\circ\).
即 \(\triangle ABC\) 为等腰三角形，或直角三角形.
\end{solution}

\begin{problem}
在直角三角形所在平面外取一点 \(P\)，而且 \(P\) 点和三角形三个顶点的距离都是相等的，证明，连接 \(P\) 点和三角形斜边中点的直线垂直于这个三角形所在的平面.

\end{problem}

\begin{solution}
证明：如左图，\(\triangle ABC\) 是直角三角形，\(\angle B=90^\circ\).

\bitmapfigure[max width=0.40\linewidth,max totalheight=4.8cm]{img/1958/hubei/q5_fig1.png}

现在过 \(P\) 点作一直线与三角形所在平面垂直，并与 \(\triangle ABC\) 所在的平面交于 \(D\) 点.

又因直角三角形斜边上的中点到各顶点的距离相等，故 \(D\) 点即 \(AC\) 中点.
\end{solution}

\begin{problem}
比较 \(\left(\frac{21}{20}\right)^{100}\) 与 \(100\) 的大小.

\end{problem}

\begin{solution}
解：因为 \(\mathrm{Lg}100=2\)，
\[
\mathrm{Lg}\left(\frac{21}{20}\right)^{100}=100(\mathrm{Lg}21-\mathrm{Lg}20)=100\times0.0212=2.12
\]
所以
\[
\left(\frac{21}{20}\right)^{100}>100
\]
\end{solution}

\begin{problem}
如 \(\alpha^{2}+b^{2}=7ab\)，求证 \(\mathrm{Lg}\left(\frac{\alpha+b}{3}\right)=\frac{1}{2}\left(\mathrm{Lg}\alpha+\mathrm{Lg}b\right)\).

\end{problem}

\begin{solution}
证：\(\alpha^{2}+2ab+b^{2}=9ab\)，即 \((\alpha+b)^{2}=9ab\).
即 \((\alpha+b)=3\sqrt{\alpha b}\)（因为 \(\alpha>0\)，\(b>0\)，因此取正号）.
即 \(\frac{\alpha+b}{3}=\sqrt{\alpha b}\).
取对数 \(\mathrm{Lg}\frac{\alpha+b}{3}=\frac{1}{2}\mathrm{Lg}(\alpha b)\)，即 \(\mathrm{Lg}\left(\frac{\alpha+b}{3}\right)=\frac{1}{2}\left(\mathrm{Lg}\alpha+\mathrm{Lg}b\right)\).
\end{solution}

\begin{problem}
已知方程式 \(x^{3}+px^{2}+qx+\gamma=0\) 的三根 \(\alpha,b,c\) 成调合级数，求 \(\alpha,b,c\).

\end{problem}

\begin{solution}
解：已知：
\[
\alpha+b+c=-p \tag{1}
\]
\[
\alpha b+bc+c\alpha=q \tag{2}
\]
\[
\alpha bc=-\gamma \tag{3}
\]
又
\[
\frac{1}{\alpha}-\frac{1}{b}=\frac{1}{b}-\frac{1}{c} \tag{4}
\]
从（4），（3）知
\[
b=\frac{2c\alpha}{\alpha+c}=\frac{q-c\alpha}{\alpha+c}
\]
所以 \(2c\alpha=q-c\alpha\).

即 \(q=3c\alpha\). 代入（3）
\[
\frac{bq}{3}=-\gamma
\]
即 \(b=-\frac{3\gamma}{q}\).

\[
\alpha+c=-p-b=\frac{3\gamma-qp}{q}
\]
但由（4），\(c=\frac{q}{3\alpha}\) 代入
\[
3q\alpha^{2}-3(pq-3\gamma)+q^{2}=0
\]
解得 \(\alpha\) 代入 \(c=\frac{q}{3\alpha}\) 即得 \(c\)
\end{solution}
