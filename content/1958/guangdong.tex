\chapter{1958年广东卷}

\section{解答题}

\begin{problem}
求二项式 \(\left(3+2x\right)^{5}\) 展开式中 \(x^{3}\) 的系数.
\end{problem}

\begin{solution}
解 \(x^{3}\) 项系数为 \(\mathrm C_{5}^{3}\cdot 3^{2}\cdot 2^{3}=\frac{5\times 4}{1\times 2}\cdot 3^{2}\cdot 2^{3}=720\).
\end{solution}

\begin{problem}
求证 \(\cos x\cdot\cos 2x\cdot\cos 4x=\frac{\sin 8x}{8\sin x}\)（\(x\neq k\pi\)，\(k\) 为任意整数）.
\end{problem}

\begin{solution}
证明 因为 \(x\neq k\pi\)（\(k\) 为任意整数），故右式恒有意义.
\[
\begin{gathered}
\frac{\sin 8x}{8\sin x}
=\frac{2\sin 4x\cdot\cos 4x}{8\sin x}
=\frac{4\sin 2x\cdot\cos 2x\cdot\cos 4x}{8\sin x}\\
{}=\frac{8\sin x\cdot\cos x\cdot\cos 2x\cdot\cos 4x}{8\sin x}
=\cos x\cdot\cos 2x\cdot\cos 4x
\end{gathered}
\]
故右边 \(=\) 左边.
\end{solution}

\begin{problem}
\(AB\)，\(AC\) 为 \(\odot O\) 的两弦，\(D\) 为 \(\widehat{AB}\) 的中点，\(E\) 为 \(\widehat{AC}\) 的中点，连 \(DE\)，交 \(AB\) 于 \(M\)，交 \(AC\) 于 \(N\). 求证：\(AM=AN\).
\end{problem}

\begin{solution}
证 如图，\(\angle 1\) 为 \(\frac{1}{2}\left(\widehat{AE}+\widehat{BD}\right)\) 所度.
\solutionfigure{\bitmapfigure[max width=0.34\linewidth]{img/1958/guangdong/q3_solution_fig1.png}}
\(\angle 2\) 为 \(\frac{1}{2}\left(\widehat{AD}+\widehat{EC}\right)\) 所度.
但 \(\widehat{AE}=\widehat{EC}\)，\(\widehat{AD}=\widehat{BD}\).
所以 \(\angle 1=\angle 2\)，\(AM=AN\).
\end{solution}

\begin{problem}
求证：正四面体相对的两棱互相垂直.
\end{problem}

\begin{solution}
证 正四面体 \(ABCD\)（如图）顶点 \(A\) 在底面内的射影是底面正三角形 \(BCD\) 的中心 \(O\). 因此棱 \(AD\) 在底面内的射影为 \(DO\).

而 \(DO\perp BC\)，依三垂线定理，\(AD\perp BC\). 同理 \(AC\perp BD\)，\(AB\perp CD\).
\end{solution}


\begin{problem}
解方程 \(\sin x=\sqrt{3}\cos x\).
\end{problem}

\begin{solution}
解 以 \(2\) 除各项得 \(\frac{1}{2}\sin x-\frac{\sqrt{3}}{2}\cos x=0\). 即 \(\sin x\cos\frac{\pi}{3}-\cos x\sin\frac{\pi}{3}=0\). 故 \(\sin\left(x-\frac{\pi}{3}\right)=0\)，\(x-\frac{\pi}{3}=n\pi\)，其中 \(n\) 为整数. 故 \(x=n\pi+\frac{\pi}{3}\).
\end{solution}

\begin{problem}
解方程组 \(\begin{cases}
\sqrt{x+\frac{1}{y}}+\sqrt{x+2y-1}=4\\
2x+2y+\frac{1}{y}=9.
\end{cases}\)
\end{problem}

\begin{solution}
解 设 \(\sqrt{x+\frac{1}{y}}=X\)，\(\sqrt{x+2y-1}=Y\). 并且 \(X\)，\(Y\) 非负数，则有 \(X+Y=4\)，\(\left(x+\frac{1}{y}\right)+\left(x+2y-1\right)=8\). 后式可写为 \(X^{2}+Y^{2}=8\). 对 \(X+Y=4\) 两边平方，得 \(X^{2}+2XY+Y^{2}=16\). 由 \(2\left(X^{2}+Y^{2}\right)-\left(X^{2}+2XY+Y^{2}\right)=0\)，得 \(\left(X-Y\right)^{2}=0\). 所以 \(X=Y\)，代入 \(X+Y=4\)，得 \(2X=4\). 所以 \(X=Y=2\). 即 \(\sqrt{x+\frac{1}{y}}=2\)，\(\sqrt{x+2y-1}=2\). 平方之：\(x+\frac{1}{y}=4\)，\(x+2y-1=4\). 两式相减得 \(2y-\frac{1}{y}-1=0\)，即 \(2y^{2}-y-1=0\). 故 \(\left(y-1\right)\left(2y+1\right)=0\). 因此 \(y_{1}=1\)，\(y_{2}=-\frac{1}{2}\). 代入 \(x+\frac{1}{y}=4\)：\(x_{1}=3\)，\(x_{2}=6\). 经检验，\(x_{1}=3\)，\(y_{1}=1\)；\(x_{2}=6\)，\(y_{2}=-\frac{1}{2}\) 是原方程组的解.
\end{solution}

\begin{problem}
设有两个同心圆，半径各为 \(R\)，\(r\)（\(R>r\)）. 今由圆心 \(O\) 作半径交大圆于 \(A\)，交小圆于 \(A'\)，由 \(A\) 作直线 \(AD\) 垂直于大圆直径 \(BC\)，并交 \(BC\) 于 \(D\)，作 \(A'E\perp AD\)，并交 \(AD\) 于 \(E\)，已知：\(\angle OAD=\alpha\). 求 \(OE\) 的长.
\end{problem}

\begin{solution}
解 如右图，在直角 \(\triangle OAD\) 中，\(OD=R\sin\alpha\)，\(AD=R\cos\alpha\).
\solutionfigure{\bitmapfigure[max width=0.32\linewidth]{img/1958/guangdong/q7_solution_fig1.png}}
在直角 \(\triangle A'AE\) 中，\(AE=\left(R-r\right)\cos\alpha\)，\(ED=AD-AE=R\cos\alpha-\left(R-r\right)\cos\alpha=r\cos\alpha\).
（或作 \(A'D'\perp OD\)，则有 \(ED=A'D'=r\cos\alpha\)）
在直角 \(\triangle OED\) 中，\(OE=\sqrt{OD^{2}+ED^{2}}=\sqrt{R^{2}\sin^{2}\alpha+r^{2}\cos^{2}\alpha}\).
\end{solution}

\begin{problem}
已知 \(\triangle ABC\)，求作圆过 \(A\) 及 \(AB\) 中点 \(M\)，并与 \(BC\) 相切.
\end{problem}

\begin{solution}
作法 延长 \(AB\) 至 \(D\)，使 \(BD=\dfrac{AB}{2}=BM\). 作 \(AB\) 与 \(BM\) 的比例中项 \(x\). 在 \(BC\) 上截取 \(BT=x\). 过 \(A\)，\(M\)，\(T\) 作圆 \(O\) 即为所求.
\solutionfigure{\bitmapfigure[max width=0.42\linewidth]{img/1958/guangdong/q8_solution_fig1.png}}

证明 由圆幂定理，设由圆外一点，作一切线与一割线，则此切线为割线及其圆外部分之比例中项.

今由作图，\(BT^{2}=BE^{2}=DB\cdot BA=BM\cdot AB\).
所以 \(T\) 为所求作的圆的切点，而此圆过 \(A\)，\(M\)，\(T\)，即为所求.
\end{solution}

\begin{problem}
已知一直角三角形的斜边长为 \(2\)，斜边上的高为 \(\frac{\sqrt{3}}{2}\)，求证：此三角形的两个锐角是下列三角方程 \(\sin^{2}x-\frac{1+\sqrt{3}}{2}\sin x+\frac{\sqrt{3}}{4}=0\) 的根.
\end{problem}

\begin{solution}
解 设 \(\angle A=x\).
\solutionfigure{\bitmapfigure[max width=0.34\linewidth]{img/1958/guangdong/q9_solution_fig1.png}}
则 \(a=2\sin x\)，\(b=\frac{\sqrt{3}}{2\sin x}\). 依勾股定理，\(c^{2}=a^{2}+b^{2}\)，故得 \(4\sin^{2}x+\frac{3}{4\sin^{2}x}=4\). 左边配方，\(\left(2\sin x+\frac{\sqrt{3}}{2\sin x}\right)^{2}=4+2\sqrt{3}\). 注意到 \(x\) 为锐角，故 \(2\sin x+\frac{\sqrt{3}}{2\sin x}>0\)，又右边 \(4+2\sqrt{3}=1+2\sqrt{3}+3=\left(1+\sqrt{3}\right)^{2}\). 故将方程两边开方，得 \(2\sin x+\frac{\sqrt{3}}{2\sin x}=1+\sqrt{3}\). 以 \(\frac{1}{2}\sin x\) 乘之并移项，得 \(\sin^{2}x-\frac{1+\sqrt{3}}{2}\sin x+\frac{\sqrt{3}}{4}=0\). 若设 \(\angle B=x\)，则由 \(c^{2}=a^{2}+b^{2}\) 亦得前式. 故两锐角均满足上述方程.
\end{solution}
