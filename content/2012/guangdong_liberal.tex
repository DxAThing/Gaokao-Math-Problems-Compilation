\begin{problem}
设 \(0 < a < 1\)，集合 \(A = \{x \in \R \mid x > 0\}\)，\(B = \{x \in \R \mid 2x^2 - 3(1 + a)x + 6a > 0\}\)，\(D = A \cap B\).
\begin{enumerate}
\item 求集合 \(D\) （用区间表示）；
\item 求函数 \( f(x) = 2x^{3} - 3(1 + a)x^{2} + 6ax \) 在 \(D\) 内的极值点.
\end{enumerate}
\end{problem}

\begin{solution}
短答案：
\begin{enumerate}
\item 当 \(\frac{1}{3} < a < 1\) 时，\(D = (0,+\infty)\)；当 \(a = \frac{1}{3}\) 时，\(D = (0,1) \cup (1,+\infty)\)；当 \(0 < a < \frac{1}{3}\) 时，\(D = (0,x_{1}) \cup (x_{2},+\infty)\)，其中
\[
x_{1} = \frac{3(1+a)-\sqrt{3(a-3)(3a-1)}}{4},\qquad
x_{2} = \frac{3(1+a)+\sqrt{3(a-3)(3a-1)}}{4}.
\]
\item 当 \(0 < a \leqslant \frac{1}{3}\) 时，极值点为 \(a\)；当 \(\frac{1}{3} < a < 1\) 时，极值点为 \(a\) 和 \(1\).
\end{enumerate}
\end{solution}
