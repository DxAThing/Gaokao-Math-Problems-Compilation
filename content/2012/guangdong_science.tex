\section{单选题}

\begin{problem}
设 \(\i\) 为虚数单位，则复数 \(\frac{5 - 6\i}{\i}=\)（\quad）

\choices
    {\(6 + 5\i\)}
    {\(6 - 5\i\)}
    {\(-6 + 5\i\)}
    {\(-6 - 5\i\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
由 \(\i^2=-1\) 得 \(\dfrac{1}{\i}=-\i\)，所以
\(\dfrac{5-6\i}{\i}=(5-6\i)(-\i)=-5\i+6\i^2=-6-5\i\)，故选 D.
\end{solution}

\begin{problem}
设集合 \(U = \{1, 2, 3, 4, 5, 6\}\)，\(M = \{1, 2, 4\}\)，则 \(\complement_U M =\)（\quad）

\choices
    {\(U\)}
    {\(\{1, 3, 5\}\)}
    {\(\{3, 5, 6\}\)}
    {\(\{2, 4, 6\}\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
补集 \(\complement_U M\) 由全集 \(U\) 中不属于 \(M\) 的元素组成. 由 \(U=\{1,2,3,4,5,6\}\) 和 \(M=\{1,2,4\}\)，删去 \(1\)，\(2\)，\(4\) 后得到 \(\complement_U M=\{3,5,6\}\)，故选 C.
\end{solution}

\begin{problem}
若向量 \(\overrightarrow{BA} = (2, 3)\)，\(\overrightarrow{CA} = (4, 7)\)，则 \(\overrightarrow{BC} =\)（\quad）

\choices
    {\((-2, -4)\)}
    {\((3, 4)\)}
    {\((6, 10)\)}
    {\((-6, -10)\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
由三角形法则，\(\overrightarrow{BC}=\overrightarrow{BA}+\overrightarrow{AC}=\overrightarrow{BA}-\overrightarrow{CA}\). 因此
\(\overrightarrow{BC}=(2,3)-(4,7)=(-2,-4)\)，故选 A.
\end{solution}

\begin{problem}
下列函数中，在区间 \((0,+\infty)\) 上为增函数的是（\quad）

\choices
    {\(y = \ln (x + 2)\)}
    {\(y = -\sqrt{x + 1}\)}
    {\(y = \left(\frac{1}{2}\right)^x\)}
    {\(y = x + \frac{1}{x}\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
在 \((0,+\infty)\) 上，\(\ln(x+2)\) 的导数为 \(\dfrac{1}{x+2}>0\)，所以选项 A 中的函数为增函数. 选项 B 的导数 \(-\dfrac{1}{2\sqrt{x+1}}<0\)，选项 C 的导数 \(\left(\dfrac{1}{2}\right)^x\ln\dfrac{1}{2}<0\)，选项 D 的导数 \(1-\dfrac{1}{x^2}\) 在 \(x=1\) 两侧符号不同，均不能在整个区间上为增函数，故选 A.
\end{solution}

\begin{problem}
已知变量 \(x\)，\(y\) 满足约束条件 \(\begin{cases}
y \le 2,\\
x + y \ge 1,\\
x - y \le 1,
\end{cases}\) 则 \(z = 3x + y\) 的最大值为（\quad）

\choices
    {\(12\)}
    {\(11\)}
    {\(3\)}
    {\(-1\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
由 \(x+y\geqslant1\) 和 \(x-y\leqslant1\) 得 \(1-y\leqslant x\leqslant y+1\)，从而 \(y\geqslant0\). 又有 \(y\leqslant2\)，所以可行域的顶点为 \((1,0)\)、\((-1,2)\)、\((3,2)\). 在这些顶点处，\(z=3x+y\) 的值分别为 \(3\)、\(-1\)、\(11\)，故最大值为 \(11\)，选 B.
\end{solution}

某几何体的三视图如图所示，则它的体积为（\quad）

\bitmapfigure[max width=0.74\linewidth]{img/2012/guangdong_science/q6_fig1.png}

\choices
    {\(12\pi\)}
    {\(45\pi\)}
    {\(57\pi\)}
    {\(81\pi\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
由三视图可知，该几何体由底面半径为 \(3\)、高为 \(5\) 的圆柱和底面半径为 \(3\) 的圆锥组成. 圆锥的母线长为 \(5\)，故其高为 \(\sqrt{5^2-3^2}=4\). 所以体积为
\(\pi\cdot3^2\cdot5+\dfrac{1}{3}\pi\cdot3^2\cdot4=45\pi+12\pi=57\pi\)，故选 C.
\end{solution}

\begin{problem}
从个位数与十位数之和为奇数的两位数中任取一个，其个位数为 \(0\) 的概率是（\quad）

\choices
    {\(\frac{4}{9}\)}
    {\(\frac{1}{3}\)}
    {\(\frac{2}{9}\)}
    {\(\frac{1}{9}\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
设十位数字为 \(a\)，个位数字为 \(b\). 当 \(a\) 为奇数、\(b\) 为偶数时有 \(5\times5=25\) 个两位数；当 \(a\) 为非零偶数、\(b\) 为奇数时有 \(4\times5=20\) 个. 因此总数为 \(45\). 个位数字为 \(0\) 且数位和为奇数时，十位数字必须为奇数，共有 \(5\) 个，所求概率为 \(\dfrac{5}{45}=\dfrac{1}{9}\)，故选 D.
\end{solution}

\begin{problem}
对任意两个非零的平面向量 \(\bs{\alpha}\) 和 \(\bs{\beta}\)，定义 \(\bs{\alpha} \circ \bs{\beta} = \frac{\bs{\alpha} \cdot \bs{\beta}}{\bs{\beta} \cdot \bs{\beta}}\). 若平面向量 \(\bs{a}\)，\(\bs{b}\) 满足 \(|\bs{a}| \ge |\bs{b}| > 0\)，\(\bs{a}\) 与 \(\bs{b}\) 的夹角 \(\theta \in \left( 0, \frac{\pi}{4} \right)\)，且 \(\bs{a} \circ \bs{b}\) 和 \(\bs{b} \circ \bs{a}\) 都在集合 \(\left\{ \frac{n}{2} \mid n \in \Z \right\}\) 中，则 \(\bs{a} \circ \bs{b} =\)（\quad）

\choices
    {\(\frac{1}{2}\)}
    {\(1\)}
    {\(\frac{3}{2}\)}
    {\(\frac{5}{2}\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
记 \(p=\bs{a}\circ\bs{b}\)，\(q=\bs{b}\circ\bs{a}\). 设 \(r=\dfrac{|\bs{a}|}{|\bs{b}|}\geqslant1\)，则由夹角条件
\(p=r\cos\theta\)，\(q=\dfrac{\cos\theta}{r}\). 因为 \(0<\theta<\dfrac{\pi}{4}\)，所以 \(0<q<1\)，而 \(q\) 是 \(\dfrac{1}{2}\) 的整数倍，故 \(q=\dfrac{1}{2}\). 又 \(pq=\cos^2\theta\)，从而
\(\dfrac{1}{2}<pq<1\)，即 \(1<p<2\). 同样由于 \(p\) 是 \(\dfrac{1}{2}\) 的整数倍，得到 \(p=\dfrac{3}{2}\)，故选 C.
\end{solution}

\section{填空题}

\begin{problem}
不等式 \(|x + 2| - |x| \leqslant 1\) 的解集为\fillinblank{}.
\end{problem}

\begin{answer}
\(\left(-\infty,-\frac{1}{2}\right]\)
\end{answer}

\begin{solution}
按 \(x\) 与 \(-2,0\) 的大小规律讨论. 当 \(x<-2\) 时，\(|x+2|-|x|=-2\leqslant1\)；当 \(-2\leqslant x<0\) 时，\(|x+2|-|x|=2x+2\leqslant1\)，故 \(x\leqslant-\dfrac{1}{2}\)；当 \(x\geqslant0\) 时，\(|x+2|-|x|=2>1\). 合并得解集为 \(\left(-\infty,-\dfrac{1}{2}\right]\).
\end{solution}

\begin{problem}
\(\left(x^{2} + \frac{1}{x}\right)^{6}\) 的展开式中 \(x^{3}\) 的系数为\fillinblank{}. （用数字作答）
\end{problem}

\begin{answer}
20
\end{answer}

\begin{solution}
展开式的通项为 \(\mathrm C_6^k(x^2)^k\left(\dfrac{1}{x}\right)^{6-k}=\mathrm C_6^k x^{3k-6}\). 要得到 \(x^3\)，需满足 \(3k-6=3\)，解得 \(k=3\)，所以所求系数为 \(\mathrm C_6^3=20\).
\end{solution}

\begin{problem}
已知递增的等差数列 \(\{a_{n}\}\) 满足 \(a_{1} = 1\)，\(a_{3} = a_{2}^{2} - 4\)，则 \(a_{n} = \)\fillinblank{}.
\end{problem}

\begin{answer}
\(2n-1\)
\end{answer}

\begin{solution}
设等差数列的公差为 \(d\). 由递增性知 \(d>0\)，且 \(a_2=1+d\)，\(a_3=1+2d\). 代入条件得
\(1+2d=(1+d)^2-4\)，即 \(d^2=4\). 结合 \(d>0\) 得 \(d=2\)，故 \(a_n=a_1+(n-1)d=1+2(n-1)=2n-1\).
\end{solution}

\begin{problem}
曲线 \( y = x^{3} - x + 3 \) 在点 \( (1,3) \) 处的切线方程为\fillinblank{}.
\end{problem}

\begin{answer}
\(y=2x+1\)
\end{answer}

\begin{solution}
令 \(f(x)=x^3-x+3\)，则 \(f'(x)=3x^2-1\)，所以在 \(x=1\) 处的切线斜率为 \(f'(1)=2\). 利用点斜式，切线方程为 \(y-3=2(x-1)\)，即 \(y=2x+1\).
\end{solution}

\begin{problem}
在平面直角坐标系 \(xOy\) 中，曲线 \(C_1\) 和 \(C_2\) 的参数方程分别为 \(\begin{cases}
x = t,\\
y = \sqrt{t},
\end{cases}\) （\(t\) 为参数） 和 \(\begin{cases}
x = \sqrt{2}\cos \theta,\\
y = \sqrt{2}\sin \theta,
\end{cases}\) （\(\theta\) 为参数），则曲线 \(C_1\) 和 \(C_2\) 的交点坐标为\fillinblank{}.
\end{problem}

\begin{answer}
\((1,1)\)
\end{answer}

\begin{solution}
由 \(C_1\) 的参数方程得 \(x=t\geqslant0\)，且 \(y=\sqrt{t}\geqslant0\)，从而 \(x=y^2\). 由 \(C_2\) 的参数方程得 \(x^2+y^2=2\). 代入 \(x=y^2\)，得 \(y^4+y^2-2=0\)，即 \((y^2-1)(y^2+2)=0\). 因为 \(y\geqslant0\)，所以 \(y=1\)，进而 \(x=1\)，交点为 \((1,1)\).
\end{solution}

\begin{problem}
已知函数 \( f(x) = 2\cos \left( \omega x + \frac{\pi}{6} \right) \) （其中 \(\omega > 0\)，\(x \in \R\) 的最小正周期为 \( 10\pi \)）.
\begin{enumerate}
\item 求 \( \omega \) 的值；
\item 设 \(\alpha\)，\(\beta \in \left[0, \frac{\pi}{2}\right]\)，\( f\left( 5\alpha + \frac{5}{3}\pi \right) = -\frac{6}{5} \)，\( f\left( 5\beta - \frac{5}{6}\pi \right) = \frac{16}{17} \)，求 \( \cos (\alpha + \beta) \) 的值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 函数 \(f(x)=2\cos\left(\omega x+\dfrac{\pi}{6}\right)\) 的最小正周期为 \(\dfrac{2\pi}{\omega}\). 由题意，\(\dfrac{2\pi}{\omega}=10\pi\)，且 \(\omega>0\)，所以 \(\omega=\dfrac{1}{5}\).
\item 由 \(\omega=\dfrac{1}{5}\)，得
\(f\left(5\alpha+\dfrac{5\pi}{3}\right)=2\cos\left(\alpha+\dfrac{\pi}{2}\right)=-2\sin\alpha\). 因此 \(\sin\alpha=\dfrac{3}{5}\). 因为 \(\alpha\in\left[0,\dfrac{\pi}{2}\right]\)，所以 \(\cos\alpha=\dfrac{4}{5}\). 同理，
\(f\left(5\beta-\dfrac{5\pi}{6}\right)=2\cos\beta=\dfrac{16}{17}\)，从而 \(\cos\beta=\dfrac{8}{17}\). 由 \(\beta\in\left[0,\dfrac{\pi}{2}\right]\)，得 \(\sin\beta=\dfrac{15}{17}\). 所以
\(\cos(\alpha+\beta)=\cos\alpha\cos\beta-\sin\alpha\sin\beta=\dfrac{4}{5}\cdot\dfrac{8}{17}-\dfrac{3}{5}\cdot\dfrac{15}{17}=-\dfrac{13}{85}\).
\end{enumerate}
\end{solution}

\begin{problem}
如图所示，在四棱锥 \(P - ABCD\) 中，底面 \(ABCD\) 为矩形，\(PA \perp\) 平面 \(ABCD\)，点 \(E\) 在线段 \(PC\) 上，\(PC \perp\) 平面 \(BDE\).
\begin{enumerate}
\item 证明： \(BD \perp\) 平面 \(PAC\)；
\item 若 \(PA = 1\)，\(AD = 2\)，求二面角 \(B - PC - A\) 的正切值.
\end{enumerate}

\bitmapfigure[max width=0.46\linewidth]{img/2012/guangdong_science/q18_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 因为 \(PA\perp\) 平面 \(ABCD\)，且 \(BD\) 在平面 \(ABCD\) 内，所以 \(PA\perp BD\). 又因为 \(PC\perp\) 平面 \(BDE\)，且 \(BD\) 在平面 \(BDE\) 内，所以 \(PC\perp BD\). 直线 \(PA\)、\(PC\) 相交于 \(P\)，且都在平面 \(PAC\) 内，因此 \(BD\perp\) 平面 \(PAC\).
\item 设 \(AC\) 与 \(BD\) 的交点为 \(O\). 由（1）知 \(BD\perp AC\)，而矩形的两条对角线互相垂直时该矩形为正方形，所以 \(AB=AD=2\)，从而 \(AC=2\sqrt{2}\)，\(BO=OC=\sqrt{2}\). 由于 \(PA\perp\) 平面 \(ABCD\)，所以 \(PA\perp AC\)，在直角三角形 \(PAC\) 中
\(PC=\sqrt{PA^2+AC^2}=3\).

在直角三角形 \(EOC\) 和 \(PAC\) 中，\(\angle ECO=\angle PCA\)，所以两三角形相似，进而 \(\dfrac{OE}{OC}=\dfrac{PA}{PC}=\dfrac{1}{3}\)，即 \(OE=\dfrac{\sqrt{2}}{3}\). 于是
\(\tan\angle BEO=\dfrac{BO}{OE}=\dfrac{\sqrt{2}}{\sqrt{2}/3}=3\)，所以二面角的正切值为 \(3\).
\end{enumerate}
\end{solution}

\begin{problem}
设数列 \(\{a_{n}\}\) 的前 \(n\) 项和为 \(S_{n}\)，满足 \(2S_{n} = a_{n + 1} - 2^{n + 1} + 1\)，\(n \in \mathbf{N}^{*}\)，且 \(a_{1}\)，\(a_{2} + 5\)，\(a_{3}\) 成等差数列.
\begin{enumerate}
\item 求 \(a_{1}\) 的值；
\item 求数列 \(\{a_{n}\}\) 的通项公式；
\item 证明：对一切正整数 \(n\)，有 \(\frac{1}{a_1} + \frac{1}{a_2} + \frac{1}{a_3} + \cdots + \frac{1}{a_n} < \frac{3}{2}\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 令 \(n=1\)，由 \(2S_n=a_{n+1}-2^{n+1}+1\) 得 \(2a_1=a_2-3\)，即 \(a_2=2a_1+3\). 令 \(n=2\)，得 \(2(a_1+a_2)=a_3-7\)，所以 \(a_3=2a_1+2a_2+7=6a_1+13\). 又因为 \(a_1\)、\(a_2+5\)、\(a_3\) 成等差数列，所以 \(2(a_2+5)=a_1+a_3\). 代入前两式，得 \(4a_1+16=7a_1+13\)，从而 \(a_1=1\).
\item 将已知关系分别取 \(n+1\) 和 \(n\) 后相减，得
\(2a_{n+1}=a_{n+2}-a_{n+1}-2^{n+1}\)，即 \(a_{n+2}=3a_{n+1}+2^{n+1}\). 令下标整体减去 \(1\)，得到 \(a_{n+1}=3a_n+2^n\). 由（1）知 \(a_1=1\)，且 \(a_2=2a_1+3=5\)，该递推式对 \(n=1\) 也成立. 于是
\(a_{n+1}+2^{n+1}=3(a_n+2^n)\). 由于 \(a_1+2=3\)，所以 \(a_n+2^n=3^n\)，故 \(a_n=3^n-2^n\).
\item 对任意正整数 \(k\)，有 \(a_k=3^k-2^k\)，且
\(a_k-3^{k-1}=2\left(3^{k-1}-2^{k-1}\right)\geqslant0\)，所以 \(a_k\geqslant3^{k-1}>0\). 因此
\(\displaystyle\sum_{k=1}^{n}\frac{1}{a_k}\leqslant\sum_{k=1}^{n}\frac{1}{3^{k-1}}=\frac{1-(1/3)^n}{1-1/3}<\frac{3}{2}\)，命题得证.
\end{enumerate}
\end{solution}

\begin{problem}
在平面直角坐标系 \(xOy\) 中，已知椭圆 \(C: \frac{x^{2}}{a^{2}} + \frac{y^{2}}{b^{2}} = 1\) （\(a > b > 0\)）的离心率 \(e = \sqrt{\frac{2}{3}}\)，且椭圆 \(C\) 上的点到点 \(Q(0,2)\) 的距离的最大值为 \(3\).
\begin{enumerate}
\item 求椭圆 \(C\) 的方程；
\item 在椭圆 \(C\) 上，是否存在点 \(M(m, n)\)，使得直线 \(l: mx + ny = 1\) 与圆 \(O: x^2 + y^2 = 1\) 相交于不同的两点 \(A\)，\(B\)，且 \(\triangle OAB\) 的面积最大？若存在，求出点 \(M\) 的坐标及相对应的 \(\triangle OAB\) 的面积；若不存在，请说明理由.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设椭圆的半焦距为 \(c\). 由离心率条件 \(e=\dfrac{c}{a}=\sqrt{\dfrac{2}{3}}\)，以及 \(c^2=a^2-b^2\)，得 \(a^2=3b^2\)，所以椭圆方程可写为 \(x^2+3y^2=3b^2\).

设椭圆上一点为 \((x,y)\)，则 \(-b\leqslant y\leqslant b\)，且 \(x^2=3b^2-3y^2\). 该点到 \(Q(0,2)\) 的距离记为 \(d\)，则
\(d^2=x^2+(y-2)^2=3b^2+6-2(y+1)^2\).

当 \(b\geqslant1\) 时，区间 \([-b,b]\) 含有 \(-1\)，故 \(d_{\max}^2=3b^2+6\). 由 \(d_{\max}=3\) 得 \(b=1\). 当 \(0<b<1\) 时，\(y+1>0\)，且其最小值在 \(y=-b\) 处取得，故 \(d_{\max}^2=(b+2)^2\)，由 \(d_{\max}=3\) 得 \(b=1\)，与 \(b<1\) 矛盾. 因此 \(b=1\)，\(a^2=3\)，椭圆方程为 \(\dfrac{x^2}{3}+y^2=1\).
\item 设圆心 \(O\) 到直线 \(l:mx+ny=1\) 的距离为 \(d_0\). 因为直线与单位圆交于不同两点，所以 \(0<d_0<1\)，且 \(d_0=\dfrac{1}{\sqrt{m^2+n^2}}\). 弦长为 \(AB=2\sqrt{1-d_0^2}\)，于是
\(S_{\triangle OAB}=\dfrac{1}{2}AB\cdot d_0=d_0\sqrt{1-d_0^2}\leqslant\dfrac{1}{2}\)，等号当且仅当 \(d_0^2=\dfrac{1}{2}\). 所以面积最大时 \(m^2+n^2=2\). 又点 \(M(m,n)\) 在椭圆上，故
\[
\begin{cases}
m^2+n^2=2,\\
\dfrac{m^2}{3}+n^2=1.
\end{cases}
\]
解得 \(m^2=\dfrac{3}{2}\)，\(n^2=\dfrac{1}{2}\). 因此存在四个点
\(M\left(\dfrac{\sqrt{6}}{2},\dfrac{\sqrt{2}}{2}\right)\)、\(M\left(-\dfrac{\sqrt{6}}{2},\dfrac{\sqrt{2}}{2}\right)\)、\(M\left(\dfrac{\sqrt{6}}{2},-\dfrac{\sqrt{2}}{2}\right)\)、\(M\left(-\dfrac{\sqrt{6}}{2},-\dfrac{\sqrt{2}}{2}\right)\)，相应的面积为 \(\dfrac{1}{2}\).
\end{enumerate}
\end{solution}

\begin{problem}
设 \(a < 1\)，集合 \(A = \{x \in \R \mid x > 0\}\)，\(B = \{x \in \R \mid 2x^2 - 3(1 + a)x + 6a > 0\}\)，\(D = A \cap B\).
\begin{enumerate}
\item 求集合 \(D\) （用区间表示）；
\item 求函数 \( f(x) = 2x^3 - 3(1 + a)x^2 + 6ax \) 在 \(D\) 内的极值点.
\end{enumerate}
\end{problem}

\begin{solution}
令 \(h(x)=2x^2-3(1+a)x+6a\). 其判别式为
\(\Delta=9(1+a)^2-48a=(3a-1)(3a-9)\). 当 \(a\leqslant\dfrac{1}{3}\) 时，令
\(x_{1}=\dfrac{3+3a-\sqrt{\Delta}}{4}\)，
\(x_{2}=\dfrac{3+3a+\sqrt{\Delta}}{4}\)，其中 \(x_1\leqslant x_2\)，且仅当 \(a=\dfrac{1}{3}\) 时等号成立.
\begin{enumerate}
\item
当 \(\dfrac{1}{3}<a<1\) 时，\(\Delta<0\)，且二次函数 \(h(x)\) 的开口向上，所以 \(h(x)>0\) 对一切实数 \(x\) 成立. 当 \(a=\dfrac{1}{3}\) 时，\(h(x)=2(x-1)^2\).

当 \(a<\dfrac{1}{3}\) 时，\(\Delta>0\)，且 \(h(0)=6a\)、\(h(1)=3a-1\)、\(x_1x_2=3a\). 若 \(0<a<\dfrac{1}{3}\)，则 \(h(0)>0\)、\(h(1)<0\)，从而 \(0<x_1<1<x_2\)；又 \(h(a)=a(3-a)>0\) 且 \(a<1\)，所以 \(a<x_1\). 若 \(a\leqslant0\)，则 \(x_1x_2\leqslant0\)，从而 \(x_1\leqslant0<x_2\)；再由 \(h(1)<0\) 得 \(x_2>1\). 因此
\(D=\begin{cases}
(0,+\infty),&\dfrac{1}{3}<a<1,\\
(0,1)\cup(1,+\infty),&a=\dfrac{1}{3},\\
(0,x_{1})\cup(x_{2},+\infty),&0<a<\dfrac{1}{3},\\
(x_{2},+\infty),&a\leqslant0.
\end{cases}\)
其中，当 \(0<a<\dfrac{1}{3}\) 时有 \(0<a<x_1<1<x_2\)；当 \(a\leqslant0\) 时有 \(x_1\leqslant0<1<x_2\).
\item 函数导数为 \(f'(x)=6x^2-6(1+a)x+6a=6(x-1)(x-a)\). 由于 \(a<1\)，函数在 \((-\infty,a)\)、\((1,+\infty)\) 上递增，在 \((a,1)\) 上递减.

当 \(\dfrac{1}{3}<a<1\) 时，\(a\)，\(1\in D\)，所以极大值点为 \(a\)，极小值点为 \(1\)；当 \(a=\dfrac{1}{3}\) 时，\(a\in D\) 而 \(1\notin D\)，所以极大值点为 \(\dfrac{1}{3}\)；当 \(0<a<\dfrac{1}{3}\) 时，\(a\in(0,x_1)\subset D\) 而 \(1\notin D\)，所以极大值点为 \(a\)；当 \(a\leqslant0\) 时，\(D=(x_2,+\infty)\subset(1,+\infty)\)，函数在 \(D\) 上递增，故无极值点.
\end{enumerate}
\end{solution}
