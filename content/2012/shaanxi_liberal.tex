\section{单选题}

\begin{problem}
集合 \(M = \{x \mid \lg x > 0\}\)，\(N = \{x \mid x^2 \leqslant 4\}\)，则 \(M \cap N =\)（\quad）

\choices
    {\((1,2)\)}
    {\([1,2)\)}
    {\((1,2]\)}
    {\([1,2]\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
由 \(\lg x>0\)，得 \(x>1\)，所以 \(M=(1,+\infty)\). 由 \(x^2\leqslant4\)，得 \(-2\leqslant x\leqslant2\)，所以 \(N=[-2,2]\). 因此 \(M\cap N=(1,2]\)，故选 C.
\end{solution}

\begin{problem}
下列函数中，既是奇函数又是增函数的为（\quad）

\choices
    {\(y = x + 1\)}
    {\(y = -x^3\)}
    {\(y = \frac{1}{x}\)}
    {\(y = x|x|\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
函数 \(y=x+1\) 不满足奇函数条件，因为 \(f(-x)=-x+1\ne-f(x)\). 函数 \(y=-x^3\) 满足奇函数条件，但其导数为 \(f'(x)=-3x^2\leqslant0\)，且在 \(\R\) 上严格递减，因而不是增函数. 函数 \(y=\frac{1}{x}\) 也满足奇函数条件，但在定义域上 \(1<2\) 而 \(f(1)=1>\frac{1}{2}=f(2)\)，所以不是增函数. 对于 \(y=x|x|\)，有 \(f(-x)=-f(x)\)，所以它是奇函数；当 \(x<0\) 时 \(f(x)=-x^2\)，导数 \(-2x>0\)，当 \(x>0\) 时 \(f(x)=x^2\)，导数 \(2x>0\)，且 \(f(0)=0\). 若 \(x_1<0\leqslant x_2\)，则 \(f(x_1)<0\leqslant f(x_2)\)，故该函数在整个 \(\R\) 上递增. 因此选 D.
\end{solution}

\begin{problem}
设 \(a\)，\(b \in \R\)，\(\i\) 是虚数单位，则“\(ab = 0\)”是“复数 \(a + \frac{b}{\i}\) 为纯虚数”的（\quad）

\choices
    {充分不必要条件}
    {必要不充分条件}
    {充分必要条件}
    {既不充分也不必要条件}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
因为 \(\frac{1}{\i}=-\i\)，所以复数 \(a+\frac{b}{\i}=a-b\i\). 若它为纯虚数，则实部 \(a=0\)，从而必有 \(ab=0\)，所以 \(ab=0\) 是“复数 \(a+\frac{b}{\i}\) 为纯虚数”的必要条件. 但 \(ab=0\) 时也可能有 \(b=0\)、\(a\ne0\)，此时复数为非零实数，不是纯虚数，因此不是充分条件，故选 B.
\end{solution}

\begin{problem}
已知圆 \(C: x^2 + y^2 - 4x = 0\)，\(l\) 是过点 \(P(3,0)\) 的直线，则（\quad）

\choices
    {\(l\) 与 \(C\) 相交}
    {\(l\) 与 \(C\) 相切}
    {\(l\) 与 \(C\) 相离}
    {以上三个选项均有可能}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
将圆的方程配方，得 \((x-2)^2+y^2=4\)，所以圆 \(C\) 的圆心为 \(O(2,0)\)，半径为 \(2\). 点 \(P(3,0)\) 到圆心的距离为 \(OP=1<2\)，即点 \(P\) 在圆内. 过圆内一点的任意直线都与圆相交于两个点，因此 \(l\) 与 \(C\) 相交，故选 A.
\end{solution}

\begin{problem}
设向量 \(\bs{a} = (1, \cos \theta)\) 与 \(\bs{b} = (-1, 2 \cos \theta)\) 垂直，则 \(\cos 2\theta\) 等于（\quad）

\begin{problem}
设函数 \(f(x) = \frac{2}{x} + \ln x\)，则（\quad）

\choices
    {\(x = \frac{1}{2}\) 为 \(f(x)\) 的极大值点}
    {\(x = \frac{1}{2}\) 为 \(f(x)\) 的极小值点}
    {\(x = 2\) 为 \(f(x)\) 的极大值点}
    {\(x = 2\) 为 \(f(x)\) 的极小值点}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
函数的定义域为 \(x>0\)，且 \(f'(x)=-\frac{2}{x^2}+\frac{1}{x}=\frac{x-2}{x^2}\). 因为 \(x^2>0\)，所以当 \(0<x<2\) 时 \(f'(x)<0\)，当 \(x>2\) 时 \(f'(x)>0\). 因此 \(f(x)\) 在 \((0,2)\) 上递减，在 \((2,+\infty)\) 上递增，故 \(x=2\) 为极小值点，选 D.
\end{solution}

\begin{problem}
小王从甲地到乙地的往返时速分别为 \(a\) 和 \(b\)（\(a < b\)），其全程的平均时速为 \(v\)，则（\quad）

\choices
    {\(a < v < \sqrt{ab}\)}
    {\(v = \sqrt{ab}\)}
    {\(\sqrt{ab} < v < \frac{a + b}{2}\)}
    {\(v = \frac{a + b}{2}\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
设甲、乙两地之间的单程路程为 \(s\). 往返总路程为 \(2s\)，总用时为 \(\frac{s}{a}+\frac{s}{b}\)，所以平均时速 \(v=\frac{2s}{\frac{s}{a}+\frac{s}{b}}=\frac{2ab}{a+b}\). 由于 \(a<b\)，有 \(v>a\Longleftrightarrow2b>a+b\)，所以 \(v>a\). 又由算术平均数大于几何平均数，\(a+b>2\sqrt{ab}\)，从而 \(v<\sqrt{ab}\). 因此 \(a<v<\sqrt{ab}\)，故选 A.
\end{solution}

\begin{problem}
设函数 \(f(x) = \begin{cases}
\sqrt{x}, & x \geqslant 0,\\
\left(\frac{1}{2}\right)^x, & x < 0.
\end{cases}\) 则 \(f\left(f(-4)\right) = \)\fillinblank{}.
\end{problem}

\begin{answer}
4
\end{answer}

\begin{solution}
因为 \(-4<0\)，所以 \(f(-4)=\left(\frac{1}{2}\right)^{-4}=2^4=16\). 又因为 \(16\geqslant0\)，所以 \(f\left(f(-4)\right)=f(16)=\sqrt{16}=4\).
\end{solution}

\begin{problem}
观察下列不等式：\(1 + \frac{1}{2^2} < \frac{3}{2}\)，\(1 + \frac{1}{2^2} + \frac{1}{3^2} < \frac{5}{3}\)，\(1 + \frac{1}{2^2} + \frac{1}{3^2} + \frac{1}{4^2} < \frac{7}{4}\). 照此规律，第五个不等式为\fillinblank{}.
\end{problem}

\begin{answer}
\(1 + \frac{1}{2^2} + \frac{1}{3^2} + \frac{1}{4^2} + \frac{1}{5^2} + \frac{1}{6^2} < \frac{11}{6}\)
\end{answer}

\begin{solution}
观察已给出的三个不等式，左边分别含有从 \(\frac{1}{2^2}\) 到 \(\frac{1}{n^2}\) 的项，右边分别为 \(\frac{2n-1}{n}\)，其中 \(n=2\)，\(3\)，\(4\). 第五个不等式对应 \(n=6\)，因此为 \(1+\frac{1}{2^2}+\frac{1}{3^2}+\frac{1}{4^2}+\frac{1}{5^2}+\frac{1}{6^2}<\frac{2\times6-1}{6}=\frac{11}{6}\).
\end{solution}

\begin{problem}
在 \(\triangle ABC\) 中，角 \(A\)，\(B\)，\(C\) 所对应的边长分别为 \(a\)，\(b\)，\(c\)，若 \(a = 2\)，\(B = \frac{\pi}{6}\)，\(c = 2\sqrt{3}\)，则 \(b = \)\fillinblank{}.
\end{problem}

\begin{answer}
2
\end{answer}

\begin{solution}
在三角形 \(ABC\) 中，角 \(B\) 的两边为 \(a\) 和 \(c\)，由余弦定理 \(b^2=a^2+c^2-2ac\cos B\). 代入 \(a=2\)，\(c=2\sqrt3\)，\(B=\frac{\pi}{6}\)，得 \(b^2=2^2+(2\sqrt3)^2-2\cdot2\cdot2\sqrt3\cdot\cos\frac{\pi}{6}=4+12-12=4\). 因为边长为正，所以 \(b=2\).
\end{solution}

\begin{problem}[A]
若存在实数 \(x\) 使 \( |x - a| + |x - 1| \leqslant 3 \) 成立，则实数 \(a\) 的取值范围是\fillinblank{}.
\end{problem}

\begin{answer}
\([-2,4]\)
\end{answer}

\begin{solution}
对任意实数 \(x\)，由三角不等式有 \(|x-a|+|x-1|\geqslant|a-1|\)，且当 \(x\) 位于 \(a\) 与 \(1\) 之间时等号成立. 因此存在实数 \(x\) 使题设不等式成立，当且仅当 \(|a-1|\leqslant3\). 解得 \(-3\leqslant a-1\leqslant3\)，即 \(-2\leqslant a\leqslant4\)，所以 \(a\) 的取值范围为 \([-2,4]\).
\end{solution}

\begin{problem}
已知等比数列 \(\{a_n\}\) 的公比为 \(q = -\frac{1}{2}\).

\begin{enumerate}
\item 若 \(a_3 = \frac{1}{4}\)，求数列 \(\{a_n\}\) 的前 \(n\) 项和；
\item 证明：对任意 \(k \in \N_+\)，\(a_k\)，\(a_{k+2}\)，\(a_{k+1}\) 成等差数列.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由等比数列的通项公式，\(a_3=a_1q^2\). 代入 \(a_3=\frac{1}{4}\) 和 \(q=-\frac{1}{2}\)，得 \(a_1\left(-\frac{1}{2}\right)^2=\frac{1}{4}\)，所以 \(a_1=1\). 因为 \(q\ne1\)，数列的前 \(n\) 项和为
\(S_n=\dfrac{a_1\left(1-q^n\right)}{1-q}=\dfrac{1-\left(-\frac{1}{2}\right)^n}{1+\frac{1}{2}}=\dfrac{2+\left(-\frac{1}{2}\right)^{n-1}}{3}\).

\item 由等比数列的通项公式，\(a_{k+1}=a_kq=-\frac{1}{2}a_k\)，\(a_{k+2}=a_kq^2=\frac{1}{4}a_k\). 因此 \(2a_{k+2}=\frac{1}{2}a_k\)，而 \(a_k+a_{k+1}=a_k-\frac{1}{2}a_k=\frac{1}{2}a_k\)，从而 \(2a_{k+2}=a_k+a_{k+1}\). 所以 \(a_k\)，\(a_{k+2}\)，\(a_{k+1}\) 成等差数列.
\end{enumerate}
\end{solution}

\begin{problem}
函数 \(f(x) = A \sin \left(\omega x - \frac{\pi}{6}\right) + 1\)（\(A > 0\)，\(\omega > 0\)）的最大值为 \(3\)，其图象相邻两条对称轴之间的距离为 \(\frac{\pi}{2}\).

\begin{enumerate}
\item 求函数 \(f(x)\) 的解析式；
\item 设 \(\alpha \in \left(0, \frac{\pi}{2}\right)\)，\(f\left(\frac{\alpha}{2}\right) = 2\)，求 \(\alpha\) 的值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 因为 \(A>0\)，函数的最大值为 \(1+A\). 由最大值为 \(3\)，得 \(1+A=3\)，所以 \(A=2\). 正弦函数相邻两条对称轴之间的距离为半个周期，即 \(\frac{\pi}{\omega}\). 由题意 \(\frac{\pi}{\omega}=\frac{\pi}{2}\)，得 \(\omega=2\). 因此 \(f(x)=2\sin\left(2x-\frac{\pi}{6}\right)+1\).

\item 将 \(x=\frac{\alpha}{2}\) 代入上式，得 \(2\sin\left(\alpha-\frac{\pi}{6}\right)+1=2\)，即 \(\sin\left(\alpha-\frac{\pi}{6}\right)=\frac{1}{2}\). 由 \(0<\alpha<\frac{\pi}{2}\)，有 \(-\frac{\pi}{6}<\alpha-\frac{\pi}{6}<\frac{\pi}{3}\). 在此区间内，正弦值为 \(\frac{1}{2}\) 的角只有 \(\frac{\pi}{6}\)，所以 \(\alpha-\frac{\pi}{6}=\frac{\pi}{6}\)，解得 \(\alpha=\frac{\pi}{3}\).
\end{enumerate}
\end{solution}

\begin{problem}
直三棱柱 \(ABC - A_1B_1C_1\) 中，\(AB = AA_1\)，\(\angle CAB = \frac{\pi}{2}\).

\begin{enumerate}
\item 证明：\(CB_1 \perp BA_1\)；
\item 已知 \(AB = 2\)，\(BC = \sqrt{5}\)，求三棱锥 \(C_1 - ABA_1\) 的体积.

\begin{minipage}{\linewidth}

\begin{solution}
\begin{enumerate}
\item 建立空间直角坐标系，使点 \(A\) 为原点，\(x\) 轴、\(y\) 轴、\(z\) 轴分别沿 \(AB\)、\(AC\)、\(AA_1\) 方向. 记 \(AB=AA_1=m\)，\(AC=n\). 由于三棱柱是直三棱柱且 \(\angle CAB=\frac{\pi}{2}\)，可设

\end{solution}

\begin{problem}
假设甲乙两种品牌的同类产品在某地区市场上销售量相等，为了解他们的使用寿命，现从两种品牌的产品中分别随机抽取 \(100\text{ 个}\) 进行测试，结果统计如下：

\begin{enumerate}
\item 估计甲品牌产品寿命小于 \(200\text{ 小时}\) 的概率；
\item 这两种品牌产品中，某个产品已使用了 \(200\text{ 小时}\)，试估计该产品是甲品牌的概率.
\end{enumerate}

\begin{minipage}{\linewidth}
\centering
\bitmapfigure[max width=0.72\linewidth]{img/2012/shaanxi_liberal/q19_fig1.png}
\end{minipage}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由甲品牌的频率分布直方图，寿命小于 \(200\text{ 小时}\) 的样本数为 \(5+20=25\). 因此用样本频率估计，甲品牌产品寿命小于 \(200\text{ 小时}\) 的概率为 \(\frac{25}{100}=\frac{1}{4}\).

\item 甲品牌产品寿命不少于 \(200\text{ 小时}\) 的样本数为 \(40+25+10=75\)，乙品牌产品寿命不少于 \(200\text{ 小时}\) 的样本数为 \(30+25+15=70\). 两种品牌的销售量相等，所以在已知产品来自两种品牌的前提下，产品属于甲品牌且寿命不少于 \(200\text{ 小时}\) 的样本数与两品牌寿命不少于 \(200\text{ 小时}\) 的样本总数之比，可估计所求概率. 因而所求概率为 \(\frac{75}{75+70}=\frac{15}{29}\).
\end{enumerate}
\end{solution}

\begin{problem}
已知椭圆 \(C_1: \frac{x^2}{4} + y^2 = 1\)，椭圆 \(C_2\) 以 \(C_1\) 的长轴为短轴，且与 \(C_1\) 有相同的离心率.

\begin{enumerate}
\item 求椭圆 \(C_2\) 的方程；
\item 设 \(O\) 为坐标原点，点 \(A\)，\(B\) 分别在椭圆 \(C_1\) 和 \(C_2\) 上，\(\overrightarrow{OB} = 2\overrightarrow{OA}\)，求直线 \(AB\) 的方程.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 椭圆 \(C_1\) 的长半轴为 \(2\)，短半轴为 \(1\)，焦半距为 \(\sqrt{2^2-1^2}=\sqrt{3}\)，所以其离心率为 \(e=\frac{\sqrt{3}}{2}\). 椭圆 \(C_2\) 以 \(C_1\) 的长轴为短轴，因此 \(C_2\) 的短半轴为 \(b_2=2\). 设 \(C_2\) 的长半轴和焦半距分别为 \(a_2\) 和 \(c_2\)，则
\(\frac{c_2}{a_2}=\frac{\sqrt{3}}{2}\)，且 \(c_2^2=a_2^2-b_2^2=a_2^2-4\). 于是 \(\frac{3}{4}a_2^2=a_2^2-4\)，解得 \(a_2=4\). 因为短轴在 \(x\) 轴上，故椭圆 \(C_2\) 的方程为 \(\frac{x^2}{4}+\frac{y^2}{16}=1\).

\item 设 \(A=(x,y)\)，由 \(\overrightarrow{OB}=2\overrightarrow{OA}\)，得 \(B=(2x,2y)\). 将 \(A\) 代入 \(C_1\) 的方程，将 \(B\) 代入 \(C_2\) 的方程，分别得到 \(\frac{x^2}{4}+y^2=1\) 和 \(\frac{(2x)^2}{4}+\frac{(2y)^2}{16}=1\)，即 \(x^2+\frac{y^2}{4}=1\). 将第一式乘以 \(4\)，得到 \(x^2+4y^2=4\)，将第二式乘以 \(4\)，得到 \(4x^2+y^2=4\). 两式相减，得 \(x^2=y^2\). 再代回 \(x^2+4y^2=4\)，得 \(5x^2=4\)，即 \(x^2=y^2=\frac{4}{5}\). 因此 \(y=x\) 或 \(y=-x\). 直线 \(AB\) 经过原点 \(O\)，所以其方程为 \(y=x\) 或 \(y=-x\).
\end{enumerate}
\end{solution}

\begin{problem}
设函数 \(f_n(x) = x^n + bx + c\)（\(n \in \N_+\)，\(b\)，\(c \in \R\)）.

\begin{enumerate}
\item 设 \(n \geqslant 2\)，\(b = 1\)，\(c = -1\)，证明：\(f_n(x)\) 在区间 \(\left(\frac{1}{2}, 1\right)\) 内存在唯一零点；
\item 设 \(n\) 为偶数，\(|f_n(-1)| \leqslant 1\)，\(|f_n(1)| \leqslant 1\)，求 \(b + 3c\) 的最小值和最大值；
\item 设 \(n = 2\)，若对任意 \(x_1\)，\(x_2 \in [-1, 1]\)，有 \(|f_2(x_1) - f_2(x_2)| \leqslant 4\)，求 \(b\) 的取值范围.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 当 \(b=1\)，\(c=-1\) 时，\(f_n(x)=x^n+x-1\). 当 \(n\geqslant2\) 时，\(f_n\left(\frac{1}{2}\right)=\frac{1}{2^n}-\frac{1}{2}\leqslant\frac{1}{4}-\frac{1}{2}<0\)，而 \(f_n(1)=1>0\). 函数 \(f_n(x)\) 在 \(\left[\frac{1}{2},1\right]\) 上连续，由零点存在定理，函数在 \(\left(\frac{1}{2},1\right)\) 内至少存在一个零点. 又因为 \(f_n'(x)=nx^{n-1}+1>0\) 在 \(\left(\frac{1}{2},1\right)\) 内成立，所以 \(f_n(x)\) 在该区间内单调递增，零点至多有一个. 因此，\(f_n(x)\) 在 \(\left(\frac{1}{2},1\right)\) 内存在唯一零点.

\item 因为 \(n\) 为偶数，所以 \(f_n(-1)=1-b+c\)，\(f_n(1)=1+b+c\). 令 \(u=c-b\)，\(v=b+c\). 由 \(\lvert f_n(-1)\rvert\leqslant1\) 和 \(\lvert f_n(1)\rvert\leqslant1\)，分别得到 \(-2\leqslant u\leqslant0\) 和 \(-2\leqslant v\leqslant0\). 由 \(b=\frac{v-u}{2}\)，\(c=\frac{u+v}{2}\)，有 \(b+3c=u+2v\). 因此 \(-6\leqslant u+2v\leqslant0\). 当 \(u=v=-2\) 时，\(b=0\)，\(c=-2\)，取得最小值 \(-6\)；当 \(u=v=0\) 时，\(b=0\)，\(c=0\)，取得最大值 \(0\). 所以 \(b+3c\) 的最小值为 \(-6\)，最大值为 \(0\).

\item 令 \(g_b(x)=x^2+bx\). 因为 \(f_2(x)=g_b(x)+c\)，所以 \(f_2(x_1)-f_2(x_2)=g_b(x_1)-g_b(x_2)\)，常数 \(c\) 不影响函数值的最大差值. 对任意 \(x_1\in[-1,1]\)，\(x_2\in[-1,1]\)，差值绝对值的最大值等于 \(g_b\) 在 \([-1,1]\) 上的最大值与最小值之差，因此题设等价于该函数值域长度不超过 \(4\). 此外，\(g_{-b}(x)=x^2-bx=g_b(-x)\)，而区间 \([-1,1]\) 关于原点对称，所以参数为 \(b\) 与参数为 \(-b\) 时的函数值域长度相同. 只需先设 \(b\geqslant0\) 讨论.

当 \(0\leqslant b\leqslant2\) 时，\(g_b'(x)=2x+b\) 的零点为 \(x_0=-\frac{b}{2}\in[-1,0]\)，故函数最小值为 \(g_b(x_0)=-\frac{b^2}{4}\). 又因 \(b\geqslant0\)，有 \(g_b(1)=1+b\geqslant1-b=g_b(-1)\)，所以函数最大值为 \(1+b\). 函数值域长度为 \(1+b+\frac{b^2}{4}\). 题设要求该长度不超过 \(4\)，即 \(1+b+\frac{b^2}{4}\leqslant4\)，等价于 \((b-2)(b+6)\leqslant0\)，在本段范围内恰好要求 \(b\leqslant2\).

当 \(b>2\) 时，对任意 \(x\in[-1,1]\)，有 \(g_b'(x)=2x+b\geqslant b-2>0\)，所以 \(g_b(x)\) 在 \([-1,1]\) 上递增，函数值域长度为 \(g_b(1)-g_b(-1)=2b>4\)，不满足题设. 因而在 \(b\geqslant0\) 时，必须且只需 \(0\leqslant b\leqslant2\). 利用关于 \(b\) 与 \(-b\) 的对称性，得到 \(b\) 的取值范围为 \([-2,2]\).
\end{enumerate}
\end{solution}
