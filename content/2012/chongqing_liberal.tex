\section{单选题}

\begin{problem}
命题“若 \(p\) 则 \(q\)”的逆命题是（\quad）

\choices
    {若 \(q\) 则 \(p\)}
    {若 \(\lnot p\) 则 \(\lnot q\)}
    {若 \(\lnot q\) 则 \(\lnot p\)}
    {若 \(p\) 则 \(\lnot q\)}
\end{problem}

\begin{problem}
不等式 \(\frac{x-1}{x+2}<0\) 的解集为（\quad）

\choices
    {\((1,+\infty)\)}
    {\((-\infty,-2)\)}
    {\((-2,1)\)}
    {\((-\infty,-2)\cup(1,+\infty)\)}
\end{problem}

\begin{problem}
设 \(A, B\) 为直线 \(y=x\) 与圆 \(x^2+y^2=1\) 的两个交点，则 \(|AB|=\)（\quad）

    {\(1\)}
    {\(\sqrt{2}\)}
    {\(\sqrt{3}\)}
    {\(2\)}

\begin{problem}
\((1-3x)^5\) 的展开式中 \(x^3\) 的系数为（\quad）

\choices

    {\(-90\)}
    {\(90\)}
    {\(270\)}

\begin{problem}
设 \(x\in\R\)，向量 \(a=(x,1)\)，\(b=(1,-2)\)，且 \(a\perp b\)，则 \(|a+b|=\)（\quad）

\section{解答题}

\begin{problem}
已知 \(\{a_{n}\}\) 为等差数列，且 \(a_{1}+a_{3}=8\)，\(a_{2}+a_{4}=12\).

\begin{enumerate}
\item 求 \(\{a_{n}\}\) 的通项公式；
\item 记 \(\{a_{n}\}\) 的前 \(n\) 项和为 \(\{S_{n}\}\)，若 \(a_{1},a_{k},a_{k+2}\) 成等比数列，求正整数 \(k\) 的值.
\end{enumerate}
\end{problem}

\begin{problem}
已知函数 \(f(x)=ax^3+bx+c\) 在点 \(x=2\) 处取得极值 \(c-16\).

\begin{enumerate}
\item 求 \(a,b\) 的值；
\item 若 \(f(x)\) 有极大值 28，求 \(f(x)\) 在 \([-3,3]\) 上的最小值.
\end{enumerate}
\end{problem}

\begin{problem}
甲、乙两人轮流投篮，每人每次投一球. 约定甲先投且先投中者获胜，一直有人获胜或每人都已投球 3 次时投篮结束. 设甲每次投篮投中的概率为 \(\frac{1}{3}\)，乙每次投篮投中的概率为 \(\frac{1}{2}\)，且各次投篮互不影响.

\begin{enumerate}
\item 求乙获胜的概率；
\item 求投篮结束时乙只投了 2 个球的概率.
\end{enumerate}
\end{problem}

\begin{problem}
设函数 \(f(x)=A\sin(\omega x+\varphi)\)（其中 \(A>0\)，\(\omega>0\)，\(-\pi<\varphi\leqslant \pi\)）在 \(x=\frac{\pi}{6}\) 处取得最大值 2，其图象与 \(x\) 轴的相邻两个交点的距离为 \(\frac{\pi}{2}\).

\begin{enumerate}
\item 求 \(f(x)\) 的解析式；
\item 求函数 \(g(x)=\frac{6\cos^{4}x-\sin^{2}x-1}{f\left(x+\frac{\pi}{6}\right)}\) 的值域.
\end{enumerate}
\end{problem}
