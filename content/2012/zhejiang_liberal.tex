\begin{problem}

\begin{enumerate}
\item 求角 \(B\) 的大小；
\item 若 \(b = 3\)，\( \sin C = 2 \sin A \)，求 \(a, c\) 的值.
\end{enumerate}
\end{problem}

\item 求数列 \(\{a_{n} \cdot b_{n}\}\) 的前 \(n\) 项和 \(T_{n}\).
\end{enumerate}
\end{problem}

\end{problem}
