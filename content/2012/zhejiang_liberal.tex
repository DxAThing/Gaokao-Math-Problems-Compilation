\begin{problem}
已知 \(a \in \R\)，函数 \(f(x) = 4x^3 - 2ax + a\).

\begin{enumerate}
\item 求 \(f(x)\) 的单调区间；
\item 证明：当 \(0 \leqslant x \leqslant 1\) 时，\(f(x) + |2 - a| > 0\).
\end{enumerate}
\end{problem}

\begin{solution}
短答案：
\begin{enumerate}
\item 当 \(a\leqslant 0\) 时，增区间为 \((-\infty,+\infty)\)；当 \(a>0\) 时，增区间为 \(\left(-\infty,-\sqrt{\frac{a}{6}}\right]\) 和 \(\left[\sqrt{\frac{a}{6}},+\infty\right)\)，减区间为 \(\left[-\sqrt{\frac{a}{6}},\sqrt{\frac{a}{6}}\right]\).
\item 结论为 \(f(x)+|2-a|>0\).
\end{enumerate}
\end{solution}
