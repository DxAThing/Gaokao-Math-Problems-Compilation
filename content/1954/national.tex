\begin{problem}
化简 \(\left[\left(a^{-\frac{3}{2}}\cdot b^{2}\right)^{-1}\cdot\left(a\cdot b^{-3}\right)^{\frac{1}{2}}\cdot\left(b^{\frac{1}{2}}\right)^{7}\right]^{\frac{1}{3}}\).

\end{problem}

\begin{solution}
\[
\begin{gathered}
\text{原式}=\left(a^{\frac{3}{2}}b^{-2}a^{\frac{1}{2}}b^{-\frac{3}{2}}b^{\frac{7}{2}}\right)^{\frac{1}{3}}=\left(a^{\frac{3}{2}+\frac{1}{2}}b^{-2-\frac{3}{2}+\frac{7}{2}}\right)^{\frac{1}{3}},\\
\text{原式}=\left(a^{2}\cdot b^{0}\right)^{\frac{1}{3}}=a^{\frac{2}{3}}
\end{gathered}
\]
\end{solution}

\begin{problem}
解 \(\frac{1}{6}\log x=\frac{1}{3}\log a+2\log b+\log c\).

\end{problem}

\begin{solution}
原式可化为 \(\log x^{\frac{1}{6}}=\log a^{\frac{1}{3}}b^{2}c\)，于是有 \(x^{\frac{1}{6}}=a^{\frac{1}{3}}b^{2}c\)，所以 \(x=\left(a^{\frac{1}{3}}b^{2}c\right)^{6}=a^{2}b^{12}c^{6}\).
\end{solution}

\begin{problem}
用二项式定理计算 \(\left(3.02\right)^{4}\)，使误差小于 \(\frac{1}{1000}\).

\end{problem}

\begin{solution}
由 \(\left(3.02\right)^{4}=\left(3+0.02\right)^{4}\) 展开得
\[
\begin{aligned}
\left(3.02\right)^{4}
&=3^{4}+4\times 3^{3}\times 0.02+6\times 3^{2}\times \left(0.02\right)^{2} \\
&\quad+4\times 3\times \left(0.02\right)^{3}+\left(0.02\right)^{4}
\end{aligned}
\]
可知第4项的值已小于 \(0.01\)，所以，计算可到第3项为止，其误差必小于千分之一.
\[
\begin{gathered}
\left(3.02\right)^{4}=3^{4}+4\times 3^{3}\times 0.02+6\times 3^{2}\times \left(0.02\right)^{2},\\
\left(3.02\right)^{4}=81+2.16+0.0216=83.182
\end{gathered}
\]
\end{solution}

\begin{problem}
直角三角形弦上半圆的面积等于勾上半圆与股上半圆面积之和，试证明之.

\end{problem}

\begin{solution}
设直角三角形的勾为 \(a\)，股为 \(b\)，弦为 \(c\)，则有 \(c^{2}=a^{2}+b^{2}\).
所以弦上半圆的面积
\[
\begin{gathered}
\frac{1}{2}\pi\left(\frac{c}{2}\right)^{2}=\frac{1}{2}\pi\cdot\frac{c^{2}}{4}=\frac{1}{2}\pi\cdot\frac{a^{2}+b^{2}}{4},\\
\frac{1}{2}\pi\left(\frac{c}{2}\right)^{2}=\frac{1}{2}\pi\left(\frac{a}{2}\right)^{2}+\frac{1}{2}\pi\left(\frac{b}{2}\right)^{2}
\end{gathered}
\]
即弦上半圆面积等于勾上半圆面积与股上半圆的面积.
\end{solution}

\begin{problem}
已知球的半径为 \(r\)，求内接正方体的体积.

\end{problem}

\begin{solution}
内接正方体的中心即该球的球心. 正方体过中心的对角线为该球的直径，故其长为 \(2r\). 若设正方体的边长为 \(a\)，则有 \(3a^{2}=4r^{2}\)，所以 \(a=\frac{2}{3}\sqrt{3}r\).
所以内接正方体的体积 \(V=a^{3}=\left(\frac{2}{3}\sqrt{3}r\right)^{3}=\frac{8}{9}\sqrt{3}r^{3}\).
\end{solution}

\begin{problem}
已知三角形的一边之长为 \(a\)，两邻角为 \(\beta\) 及 \(\gamma\)，求计算边长 \(b\) 的计算公式.

\end{problem}

\begin{solution}
由正弦定理可知 \(\frac{a}{\sin\left[180^{\circ}-\left(\beta+\gamma\right)\right]}=\frac{b}{\sin\beta}\)，所以 \(b=\frac{a\sin\beta}{\sin\left[180^{\circ}-\left(\beta+\gamma\right)\right]}=\frac{a\sin\beta}{\sin\left(\beta+\gamma\right)}\).
\end{solution}

\begin{problem}
描绘 \(y=3x^{2}-7x-1\) 之图象，并按下列条件分别求 \(x\) 的值的范围：

\begin{enumerate}
\item \(y>0\)；
\item \(y<0\).
\end{enumerate}

\end{problem}

\begin{solution}
将原方程变形，可得 \(\left(x-\frac{7}{6}\right)^{2}=\frac{1}{3}\left(y+\frac{61}{12}\right)\).
于是，抛物线顶点为 \(\left(\frac{7}{6},-\frac{61}{12}\right)\)（如图）.
抛物线与 \(x\) 轴的交点为：\(M\left(\frac{7-\sqrt{61}}{6},0\right)\)，\(N\left(\frac{7+\sqrt{61}}{6},0\right)\).

当 \(y>0\) 时，\(x\) 的取值范围为：
\(\left(-\infty,\frac{7-\sqrt{61}}{6}\right)\cup\left(\frac{7+\sqrt{61}}{6},+\infty\right)\).

当 \(y<0\) 时，\(x\) 的取值范围为：
\(\left(\frac{7-\sqrt{61}}{6},\frac{7+\sqrt{61}}{6}\right)\).

\begin{problem}
假设两圆互相外切，求证用连心线段为直径所作的圆必与前两圆的外公切线相切.

\end{problem}

\begin{solution}
设 \(\odot O_{1}\) 及 \(\odot O_{2}\) 为互相外切的二圆，其中一外公切线为 \(A_{1}A_{2}\)，切点 \(A_{1}\) 及 \(A_{2}\)（如图），令点 \(O\) 为连心线 \(O_{1}O_{2}\) 的中点，过 \(O\) 作 \(OA\perp A_{1}A_{2}\).
因 \(OA=\frac{1}{2}\left(O_{1}A_{1}+O_{2}A_{2}\right)=\frac{1}{2}O_{1}O_{2}\)，因此，以 \(O_{1}O_{2}\) 为直径，即以 \(O\) 为圆心，\(OA\) 为半径的圆必与直线 \(A_{1}A_{2}\) 相切. 同理可证，此圆必切于 \(\odot O_{1}\) 及 \(\odot O_{2}\) 的另外一条外公切线.

\end{solution}

\begin{problem}
解 \(\frac{1+\tan x}{1-\tan x}=1+\sin 2x\)，求 \(x\) 之通值.

\end{problem}

\begin{solution}
由原方程得 \(\frac{\cos x+\sin x}{\cos x-\sin x}=\left(\cos x+\sin x\right)^{2}\)，即 \(\cos x+\sin x=\left(\cos x+\sin x\right)^{2}\left(\cos x-\sin x\right)\). 整理为
\[
\left(\cos x+\sin x\right)\left(1-\cos^{2}x+\sin^{2}x\right)=2\left(\cos x+\sin x\right)\cdot\sin^{2}x=0
\]
所以 \(\cos x+\sin x=0\)，或 \(\sin^{2}x=0\).

由方程 \(\cos x+\sin x=0\) 得，\(\tan x=-1\). 因此 \(x=k\pi-\frac{\pi}{4}\)（\(k\) 为整数）.

由方程 \(\sin^{2}x=0\)，得 \(x=k\pi\)（\(k\) 为整数）.

由检验可知，\(x=k\pi-\frac{\pi}{4}\)，\(x=k\pi\)（\(k\) 为整数）均为方程的通解.
\end{solution}

\begin{problem}
有一直圆锥，全面积为 \(a\)；与之同底同高之直圆柱全面积为 \(a'\). 求该圆锥高与母线之比.

\end{problem}

\begin{solution}
设直圆锥的高为 \(h\)，底面半径为 \(R\)，母线长为 \(l\)，则
\(\frac{a}{a'}=\frac{\pi R\left(R+l\right)}{2\pi R\left(R+h\right)}=\frac{R+l}{2\left(R+h\right)}\).
所以 \(2a\left(R+h\right)=a'\left(R+l\right)\). 由 \(R=\sqrt{l^{2}-h^{2}}\)，代入可得
\(2a\left(\sqrt{l^{2}-h^{2}}+h\right)=a'\left(\sqrt{l^{2}-h^{2}}+l\right)\).
即 \(\left(2a-a'\right)\sqrt{l^{2}-h^{2}}=a'l-2ah\). 两边同除以 \(l\)，可得
\(\left(2a-a'\right)\sqrt{1-\left(\frac{h}{l}\right)^{2}}=a'-2a\cdot\frac{h}{l}\).
等式两边平方：
\(\left(2a-a'\right)^{2}\left(1-\left(\frac{h}{l}\right)^{2}\right)=a'^{2}-4aa'\cdot\frac{h}{l}+4a^{2}\left(\frac{h}{l}\right)^{2}\).
\(\left(8a^{2}-4aa'+a'^{2}\right)\left(\frac{h}{l}\right)^{2}-4aa'\cdot\frac{h}{l}+\left(4aa'-4a^{2}\right)=0\).
这是一个关于 \(\frac{h}{l}\) 的一元二次方程的判别式
\[
\begin{gathered}
\Delta=\left(-4aa'\right)^{2}-4\left(8a^{2}-4aa'+a'^{2}\right)\left(4aa'-4a^{2}\right),\\
\Delta=16a\left(2a-a'\right)^{3}>0
\end{gathered}
\]
因此该一元二次方程有两个实根，解得
\[
\begin{gathered}
\frac{h}{l}=\frac{4aa'\pm\sqrt{16a\left(2a-a'\right)^{3}}}{2\left(8a^{2}-4aa'+a'^{2}\right)},\\
\frac{h}{l}=\frac{2aa'\pm 2\left(2a-a'\right)\sqrt{a\left(2a-a'\right)}}{4a^{2}+\left(2a-a'\right)^{2}}
\end{gathered}
\]
即为圆锥的高与母线的比.
\end{solution}
