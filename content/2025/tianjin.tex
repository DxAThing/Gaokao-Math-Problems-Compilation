\section{单选题}

\begin{problem}
已知集合 \(\mathcal U=\{1,2,3,4,5\}\)，\(A=\{1,3\}\)，\(B=\{2,3,5\}\)，则 \(\mathcal U\setminus(A\cup B)=\)（\quad）

\choices
    {\(\{1,2,3,4\}\)}
    {\(\{2,3,4\}\)}
    {\(\{2,4\}\)}
    {\(\{4\}\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
由定义可得 \(A\cup B=\{1,2,3,5\}\)，故 \(\mathcal U\setminus (A\cup B)=\{4\}\)，故选：D.
\end{solution}

\begin{problem}
设 \(x\in\R\)，则“\(x=0\)”是“\(\sin 2x=0\)”的（\quad）

\choices
    {充分不必要条件}
    {必要不充分条件}
    {充分必要条件}
    {既不充分也不必要条件}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
因为 \(x=0\Rightarrow \sin 2x=\sin 0=0\)，所以是充分条件；但 \(x=\pi\) 时 \(\sin2x=0\) 而 \(x\ne0\)，所以不是必要条件. 故选：A.
\end{solution}

\begin{problem}
已知函数 \(y=f(x)\) 的图象如下，则 \(f(x)\) 的解析式可能为（\quad）

\bitmapfigure[max width=0.42\linewidth]{img/2025/tianjin/q3_fig1.png}

\choices
    {\(f(x)=\frac{x}{1-|x|}\)}
    {\(f(x)=\frac{x}{|x|-1}\)}
    {\(f(x)=\frac{|x|}{1-x^2}\)}
    {\(f(x)=\frac{|x|}{x^2-1}\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
图象关于 \(y\) 轴对称，所以函数 \(f(x)\) 为偶函数. 选项 A 中 \(f(-x)=\frac{-x}{1-|-x|}=-\frac{x}{1-|x|}=-f(x)\)，选项 B 中 \(f(-x)=\frac{-x}{|-x|-1}=-\frac{x}{|x|-1}=-f(x)\).
二者都为奇函数，因此排除 A，B.

当 \(0<x<1\) 时，
\(\frac{|x|}{1-x^2}>0\)，\(\frac{|x|}{x^2-1}<0\)，
与图像在 \((0,1)\) 内为负相符，故 C 错误. 故 D 正确.
\end{solution}

\begin{problem}
若 \(m\) 为直线，\(\alpha\)，\(\beta\) 为两个平面，则下列结论中正确的是（\quad）

\choices
    {若 \(m\parallel \alpha\)，\(n\subset \alpha\)，则 \(m\parallel n\)}
    {若 \(m\perp\alpha\)，\(m\perp\beta\)，则 \(\alpha\perp\beta\)}
    {若 \(m\parallel\alpha\)，\(m\perp\beta\)，则 \(\alpha\perp\beta\)}
    {若 \(m\subset\alpha\)，\(\alpha\perp\beta\)，则 \(m\perp\beta\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
A：在平行关系下，\(m\) 可与 \(n\) 平行或异面，故错.

B：给出两平面都含垂直于同一平面的同一条直线，无法推出两平面垂直，故错.

C：因为 \(m\parallel\alpha\)，所以在平面 \(\alpha\) 内存在直线 \(n\) 使 \(n\parallel m\). 又 \(m\perp\beta\)，所以 \(n\perp\beta\). 平面 \(\alpha\) 内存在一条直线垂直于平面 \(\beta\)，故 \(\alpha\perp\beta\).

D：\(m\subset\alpha\) 时 \(m\) 可与 \(\beta\) 的位置任意，故错. 故选 C.
\end{solution}

\begin{problem}
下列说法中错误的是（\quad）

\choices
    {若 \(X\sim N(\mu,\sigma^2)\)，则 \(P(X\le\mu-\sigma)=P(X\ge\mu+\sigma)\)}
    {若 \(X\sim N(1,2^2)\)，\(Y\sim N(2,2^2)\)，则 \(P(X<1)<P(Y<2)\)}
    {\(|r|\) 越接近 1，相关性越强}
    {\(|r|\) 越接近 0，相关性越弱}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
对称性给出 A 正确；正态分布标准化后 B 变为 \(P(X<1)=P(Y<2)=0.5\)，故 B 错误；C，D 与相关系数定义一致，故正确. 故选 B.
\end{solution}

\begin{problem}
\(S_n=-n^2+8n\)，则数列 \(\{|a_n|\}\) 的前 12 项和为（\quad）

\choices
    {112}
    {48}
    {80}
    {64}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
由 \(S_n=-n^2+8n\)，得 \(a_n=S_n-S_{n-1}=-2n+9\). 当 \(n\le4\) 时 \(a_n\ge0\)，当 \(n\ge5\) 时 \(a_n\le0\). 故
\[
\sum_{k=1}^{12}|a_k|=\sum_{k=1}^4 a_k-\sum_{k=5}^{12}a_k=2S_4-S_{12}=80
\]
故选：C.
\end{solution}

\begin{problem}
函数 \(f(x)=0.3^x-\sqrt{x}\) 的零点所在区间是（\quad）

\choices
    {\((0,0.3)\)}
    {\((0.3,0.5)\)}
    {\((0.5,1)\)}
    {\((1,2)\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
因为底数 \(0.3\in(0,1)\)，所以函数 \(y=0.3^x\) 在 \(\R\) 上单调递减；函数 \(y=\sqrt{x}\) 在 \([0,+\infty)\) 上单调递增.
当 \(x_1<x_2\) 且 \(x_1,x_2\in[0,+\infty)\) 时，
\(0.3^{x_1}>0.3^{x_2}\)，\(\sqrt{x_1}<\sqrt{x_2}\)，
从而
\(
f(x_1)=0.3^{x_1}-\sqrt{x_1}>0.3^{x_2}-\sqrt{x_2}=f(x_2)
\)
所以 \(f(x)=0.3^x-\sqrt{x}\) 在定义域上单调递减. 又
\(f(0)=1>0\)，\(f(0.3)=0.3^{0.3}-0.3^{0.5}>0\)，\(f(0.5)=0.3^{0.5}-0.5^{0.5}<0\).
函数 \(f(x)=0.3^x-\sqrt{x}\) 在 \([0.3,0.5]\) 上连续，且 \(f(0.3)>0\)，\(f(0.5)<0\)，所以由零点存在性定理，存在 \(x_0\in(0.3,0.5)\) 使 \(f(x_0)=0\). 又因为 \(f(x)\) 在定义域上单调递减，所以零点只能落在该区间内. 故选 B.
\end{solution}

\begin{problem}
\(f(x)=\sin(\omega x+\varphi)\)（\(\omega>0\)，\(-\pi<\varphi<\pi\)），在 \(\left[-\frac{5\pi}{12},\frac{\pi}{12}\right]\) 上单调递增，且 \(x=\frac{\pi}{12}\) 是它的一条对称轴，\((\frac{\pi}{3},0)\) 是它的一个对称中心.

当 \(x\in[0,\frac{\pi}{2}]\) 时，\(f(x)\) 的最小值为（\quad）

\choices
    {\(-\frac{\sqrt3}{2}\)}
    {\(-\frac12\)}
    {1}
    {0}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
设 \(f(x)\) 的最小正周期为 \(T\)，则 \(T=\frac{2\pi}{\omega}\). 由 \(x=\frac{\pi}{12}\) 是它的一条对称轴，可得
\(\omega\cdot \frac{\pi}{12}+\varphi=\frac{\pi}{2}+k\pi\)（\(k\in\Z\)）. 由 \(\left(\frac{\pi}{3},0\right)\) 是它的一个对称中心，可得
\(\omega\cdot \frac{\pi}{3}+\varphi=m\pi\)（\(m\in\Z\)）. 两式相减，得
\(\frac{\omega\pi}{4}=m\pi-\frac{\pi}{2}-k\pi\)，
所以
\(\omega=4n+2\)（\(n\in\Z\)）.
又函数在 \(\left[-\frac{5\pi}{12},\frac{\pi}{12}\right]\) 上单调递增，该区间长度为 \(\frac{\pi}{2}\). 要使正弦函数在这段区间上保持单调递增，必须有
\(\omega\cdot \frac{\pi}{2}\le \pi\)，
即 \(0<\omega\le 2\). 因此 \(\omega=2\).
代入上面的条件，可得 \(\frac{\pi}{6}+\varphi=\frac{\pi}{2}+k\pi\)（\(k\in\Z\)），且 \(\frac{2\pi}{3}+\varphi=m\pi\)（\(m\in\Z\)）. 于是
\(
\varphi=\frac{\pi}{3}+k\pi=m\pi-\frac{2\pi}{3}
\)
再由 \(-\pi<\varphi<\pi\)，取 \(k=0\)，\(m=1\)，得
\(
\varphi=\frac{\pi}{3}
\)
故
\(
f(x)=\sin\left(2x+\frac{\pi}{3}\right)
\)
当 \(x\in\left[0,\frac{\pi}{2}\right]\) 时，
\(2x+\frac{\pi}{3}\in\left[\frac{\pi}{3},\frac{4\pi}{3}\right]\).
因此
\(
f(x)_{\min}=\sin\frac{4\pi}{3}=-\frac{\sqrt3}{2}
\)
故选 A.
\end{solution}

\begin{problem}
双曲线 \(\frac{x^2}{a^2}-\frac{y^2}{b^2}=1\)\((a>0,b>0)\) 的左，右焦点分别为 \(F_1\)，\(F_2\)，以右焦点 \(F_2\) 为焦点的抛物线 \(y^2=2px\)\((p>0)\) 与双曲线在第一象限交于 \(P\)，若 \(|PF_1|+|PF_2|=3|F_1F_2|\)，则双曲线离心率 \(e=\)（\quad）

\choices
    {2}
    {5}
    {\(\frac{\sqrt2+1}{2}\)}
    {\(\frac{\sqrt5+1}{2}\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
设 \(F_2=(c,0)\)，双曲线半焦距为 \(c\)，则 \(P=(x_0,y_0)\) 满足 \(|PF_1|=3c+a\)，\(|PF_2|=3c-a\). 由抛物线定义可得 \(|PA|=|PF_2|=3c-a\)（\(A\) 为 \(F_1\) 的垂足）. 进一步代入到 \(y_0^2=4cx_0\) 与双曲线方程可得 \(2c^2-3ac-2a^2=0\)，故 \(c=2a\)，所以 \(e=\frac{c}{a}=2\).
\end{solution}

\begin{problem}
小桐一周跑 \(2\) 次，一次 \(5\) 圈或 \(6\) 圈. 第一次跑 \(5\) 圈或 \(6\) 圈概率各为 \(0.5\)；若第一次跑 \(5\) 圈，则第二次分别跑 \(5\) 圈，\(6\) 圈的概率为 \(0.4\)，\(0.6\)；若第一次跑 \(6\) 圈，则第二次分别跑 \(5\) 圈，\(6\) 圈的概率为 \(0.6\)，\(0.4\).
则一周跑 \(11\) 圈的概率为 \(\fillinblank{}\)；若一周至少跑 \(11\) 圈为达标，则连续跑 \(4\) 周，记合格周数为 \(X\)，则 \(E(X)=\fillinblank{}\).
\end{problem}

\begin{answer}
0.6；3.2
\end{answer}

\begin{solution}
设 \(B\) 为第一次跑 5 圈事件，\(C\) 为第二次跑 5 圈事件，\(A\) 为一周跑 11 圈事件. 则
\(P(A)=P(B)P(C^\mathrm c\mid B)+P(B^\mathrm c)P(C\mid B^\mathrm c)=0.5\times 0.6+0.5\times 0.6=0.6\).
达标事件为一周至少跑 \(11\) 圈，即 \(P(D)=P(A)+P(B^\mathrm c)P(C^\mathrm c\mid B^\mathrm c)=0.6+0.5\times 0.4=0.8\).
故 \(X\sim B(4,0.8)\)，\(E(X)=4\times 0.8=3.2\).
\end{solution}

\begin{problem}
在 \(\triangle ABC\) 中，角 \(A\)，\(B\)，\(C\) 对边分别为 \(a\)，\(b\)，\(c\). 已知 \(a\sin B=\sqrt3\,b\cos A\)，\(c-2b=1\)，\(a=\sqrt7\).
\begin{enumerate}
    \item 求 \(A\)；
    \item 求 \(c\)；
    \item 求 \(\sin(A+2B)\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由正弦定理 \(\frac{a}{\sin A}=\frac{b}{\sin B}\)，可得 \(a\sin B=b\sin A\).
又已知 \(a\sin B=\sqrt3\,b\cos A\)，所以 \(b\sin A=\sqrt3\,b\cos A\). 因为 \(b>0\)，且若 \(\cos A=0\)，则 \(\sin A=\pm1\)，与 \(\sin A=\sqrt3\cos A\) 矛盾，所以 \(\cos A\ne0\). 故
\(
\tan A=\sqrt3
\)
又 \(A\in(0,\pi)\)，所以
\(
A=\frac{\pi}{3}
\)

\item 由 \(A=\frac{\pi}{3}\) 得 \(\cos A=\frac{1}{2}\). 由余弦定理 \(a^2=b^2+c^2-2bc\cos A\)，再由 \(c-2b=1\)，得 \(c=2b+1\). 代入 \(a=\sqrt7\)，得
\(
7=b^2+(2b+1)^2-b(2b+1)
\)
化简得
\(
b^2+b-2=0
\)
解得 \(b=1\) 或 \(b=-2\). 因为 \(b>0\)，所以 \(b=1\)，从而
\(
c=3
\)

\item 由余弦定理 \(b^2=a^2+c^2-2ac\cos B\)，代入 \(a=\sqrt7\)，\(b=1\)，\(c=3\)，得
\(
1=7+9-6\sqrt7\cos B
\)
所以
\(
\cos B=\frac{5\sqrt7}{14}
\)
于是
\[
\sin B=\sqrt{1-\cos^2 B}=\sqrt{1-\frac{175}{196}}=\frac{\sqrt{21}}{14}
\]
从而
\(\cos 2B=2\cos^2 B-1=2\times \frac{175}{196}-1=\frac{11}{14}\)，\(\sin 2B=2\sin B\cos B=2\times \frac{\sqrt{21}}{14}\times \frac{5\sqrt7}{14}=\frac{5\sqrt3}{14}\).
因此
\[
\sin(A+2B)=\sin A\cos 2B+\cos A\sin 2B=\frac{\sqrt3}{2}\cdot\frac{11}{14}+\frac12\cdot\frac{5\sqrt3}{14}=\frac{4\sqrt3}{7}
\]
\end{enumerate}
\end{solution}

\begin{problem}
正方体 \(ABCD-A_1B_1C_1D_1\) 的棱长为 4，\(E\)，\(F\) 分别为 \(A_1D_1\)，\(C_1B_1\) 中点，且 \(CG=3GC_1\).

\begin{enumerate}
\item 求证：\(GF\perp\) 平面 \(FBE\)；
\item 求平面 \(FBE\) 与平面 \(EBG\) 的夹角余弦；
\item 求四面体 \(D-FBE\) 的体积.
\end{enumerate}

\bitmapfigure[max width=0.55\linewidth]{img/2025/tianjin/q17_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 如题意可用坐标法. 取
\(D(0,0,0)\)，\(A(4,0,0)\)，\(B(4,4,0)\)，\(C(0,4,0)\)，\(D_1(0,0,4)\)，\(A_1(4,0,4)\)，\(B_1(4,4,4)\)，\(C_1(0,4,4)\)，
则
\(E(2,0,4)\)，\(F(2,4,4)\)，\(G(0,4,3)\).
于是
\(\overrightarrow{EF}=(0,4,0)\)，\(\overrightarrow{EB}=(2,4,-4)\)，\(\overrightarrow{FG}=(-2,0,-1)\).
设平面 \(FBE\) 的一个法向量为 \(\bs{n}=(x,y,z)\)，则
\(\overrightarrow{EF}\cdot \bs{n}=4y=0\)，\(\overrightarrow{EB}\cdot \bs{n}=2x+4y-4z=0\).
取 \(x=2\)，可得 \(y=0\)，\(z=1\)，故可取 \(\bs{n}=(2,0,1)\). 而 \(\overrightarrow{FG}=-(2,0,1)\)，所以 \(FG\perp\) 平面 \(FBE\).

\item 由（1）知平面 \(FBE\) 的一个法向量可取 \(\bs{n}_1=(2,0,1)\). 又
\(\overrightarrow{EG}=(-2,4,-1)\)，\(\overrightarrow{BG}=(-4,0,3)\).
设平面 \(EBG\) 的一个法向量为 \(\bs{n}_2=(x,y,z)\)，则
\(\overrightarrow{EG}\cdot \bs{n}_2=-2x+4y-z=0\)，\(\overrightarrow{BG}\cdot \bs{n}_2=-4x+3z=0\).
取 \(x=6\)，可得 \(y=5\)，\(z=8\)，故可取 \(\bs{n}_2=(6,5,8)\). 于是
\(
\cos\angle(FBE,EBG)=\frac{|\bs{n}_1\cdot\bs{n}_2|}{|\bs{n}_1|\,|\bs{n}_2|}=\frac{4}{5}
\)

\item 由
\(\overrightarrow{EF}=(0,4,0)\)，\(\overrightarrow{BF}=(-2,0,4)\)，
得
\(
\overrightarrow{EF}\cdot \overrightarrow{BF}=0\times(-2)+4\times 0+0\times 4=0
\)
所以 \(EF\perp BF\)，因此 \(S_{\triangle BFE}=\frac12|EF|\cdot|BF|=\frac12\times 4\times \sqrt{(-2)^2+4^2}=4\sqrt5\).
又 \(\overrightarrow{DE}=(2,0,4)\)，故点 \(D\) 到平面 \(FBE\) 的距离为
\(
d=\frac{|\overrightarrow{DE}\cdot\overrightarrow{FG}|}{|\overrightarrow{FG}|}=\frac{8}{\sqrt5}
\)
所以
\[
V_{D-BFE}=\frac13\cdot d\cdot S_{\triangle BFE}=\frac13\cdot \frac{8}{\sqrt5}\cdot 4\sqrt5=\frac{32}{3}
\]
\end{enumerate}
\end{solution}

\begin{problem}
已知椭圆 \(\frac{x^2}{a^2}+\frac{y^2}{b^2}=1\)\((a>b>0)\) 的左焦点为 \(F\)，右顶点为 \(A\)，\(P\) 为 \(x=a\) 上一点，且直线 \(PF\) 的斜率为 \(\frac13\)，\(\triangle PFA\) 的面积为 \(\frac32\)，离心率为 \(\frac12\).
\begin{enumerate}
\item 求椭圆方程；
\item 过 \(P\) 的直线与椭圆有唯一交点 \(B\)（异于 \(A\)），证明 \(PF\) 平分 \(\angle AFB\).
\end{enumerate}

\bitmapfigure[max width=0.42\linewidth]{img/2025/tianjin/q18_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设椭圆的半焦距为 \(c\)，则左焦点为 \(F(-c,0)\)，右顶点为 \(A(a,0)\). 由离心率 \(e=\frac{1}{2}\)，得 \(\frac{c}{a}=\frac{1}{2}\)，所以 \(a=2c\).
因为 \(P\) 在直线 \(x=a\) 上，设 \(P(a,m)\). 又直线 \(PF\) 的斜率为 \(\frac{1}{3}\)，所以 \(\frac{m-0}{a-(-c)}=\frac{1}{3}\). 代入 \(a=2c\)，得 \(\frac{m}{3c}=\frac{1}{3}\)，所以 \(m=c\). 因此
\(
P(a,m)=\left(2c,c\right)
\)
由题意 \(S_{\triangle PFA}=\frac{3}{2}\)，又 \(|AF|=a+c=3c\)，点 \(P\) 到 \(x\) 轴的距离为 \(c\)，所以
\[
S_{\triangle PFA}=\frac{1}{2}\cdot 3c\cdot c=\frac{3c^2}{2}=\frac{3}{2}
\]
故
\(
c=1
\)
于是 \(a=2\)，\(b^2=a^2-c^2=4-1=3\)，所以椭圆方程为
\(
\frac{x^2}{4}+\frac{y^2}{3}=1
\)

\item 由（1）知
\(P(2,1)\)，\(F(-1,0)\)，\(A(2,0)\).
易知直线 \(PB\) 的斜率存在，设过点 \(P\) 的直线方程为
\(\ell:y=kx+m\).
因为点 \(P(2,1)\) 在直线上，所以 \(1=2k+m\)，即 \(m=1-2k\).
联立 \(y=kx+m\) 与 \(\frac{x^2}{4}+\frac{y^2}{3}=1\)，消去 \(y\)，得
\((3+4k^2)x^2+8kmx+4m^2-12=0\).
因为直线 \(\ell\) 与椭圆只有一个公共点，所以判别式为 0，即
\(
(8km)^2-4(3+4k^2)(4m^2-12)=0
\)
化简得
\(
4k^2-m^2+3=0
\)
再代入 \(m=1-2k\)，得
\(
4k^2-(1-2k)^2+3=0
\)
解得
\(k=-\frac{1}{2}\)，\(m=2\).
故
\(
\ell:y=-\frac12x+2
\)
联立椭圆方程可得
\(
B\left(1,\frac{3}{2}\right)
\)
此时
\(\overrightarrow{FB}=\left(2,\frac32\right)\)，\(\overrightarrow{FP}=(3,1)\)，\(\overrightarrow{FA}=(3,0)\).
于是
\(\cos\angle BFP=\frac{\overrightarrow{FB}\cdot\overrightarrow{FP}}{|\overrightarrow{FB}|\,|\overrightarrow{FP}|} =\frac{3\sqrt{10}}{10}\)，
且
\(\cos\angle PFA=\frac{\overrightarrow{FA}\cdot\overrightarrow{FP}}{|\overrightarrow{FA}|\,|\overrightarrow{FP}|} =\frac{3\sqrt{10}}{10}\).
因为 \(\angle BFP\)，\(\angle PFA\in \left(0,\frac{\pi}{2}\right)\)，所以 \(\angle BFP=\angle PFA\)，故 \(PF\) 平分 \(\angle AFB\).
\end{enumerate}
\end{solution}

\begin{problem}
已知数列 \(\{a_n\}\) 为等差数列，\(\{b_n\}\) 为等比数列，\(a_1=b_1=2\)，\(a_2=b_2+1\)，\(a_3=b_3\).
\begin{enumerate}
\item 求 \(\{a_n\}\)，\(\{b_n\}\) 的通项；
\item \(\forall n\in\N^*\)，\(I=\{0,1\}\)，定义 \(T_n=\{p_1a_1b_1+p_2a_2b_2+\cdots+p_na_nb_n\mid p_i\in I,i=1,\dots,n\}\).
\begin{enumerate}
\item 求证：对任意 \(t\in T_n\)，有 \(t<a_{n+1}b_{n+1}\)；
\item 求 \(T_n\) 的所有元素之和.
\end{enumerate}
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设等差数列 \(\{a_n\}\) 的公差为 \(d\)，等比数列 \(\{b_n\}\) 的公比为 \(q\). 由题意得 \(2+d=2q+1\) 且 \(2+2d=2q^2\). 解得
\(d=3\)，\(q=2\).
所以
\(a_n=2+3(n-1)=3n-1\)，\(b_n=2\cdot 2^{n-1}=2^n\).

\item
\begin{enumerate}
\item 由（1）可得
\(a_nb_n=(3n-1)2^n\).
对任意 \(t\in T_n\)，都有
\(t\le a_1b_1+a_2b_2+\cdots+a_nb_n\).
设
\(S_n=a_1b_1+a_2b_2+\cdots+a_nb_n\).
则
\(S_n=2\times 2+5\times 2^2+\cdots +(3n-1)2^n\).
于是
\(2S_n=2\times 2^2+5\times 2^3+\cdots +(3n-1)2^{n+1}\).
两式相减，得
\(
-S_n=4+3\left(2^2+2^3+\cdots +2^n\right)-(3n-1)2^{n+1}
\)
因为
\(2^2+2^3+\cdots +2^n=2^{n+1}-4\)，
所以
\(
-S_n=4+3(2^{n+1}-4)-(3n-1)2^{n+1}=-8+(4-3n)2^{n+1}
\)
故
\(
S_n=8+(3n-4)2^{n+1}
\)
又
\(a_{n+1}b_{n+1}=(3n+2)2^{n+1}>8+(3n-4)2^{n+1}\)，
即
\(a_{n+1}b_{n+1}-S_n=6\cdot 2^{n+1}-8>0\).
因此对任意 \(t\in T_n\)，都有
\(t<a_{n+1}b_{n+1}\).

\item 在 \(T_n\) 的所有元素中，每个 \(a_ib_i\) 对应的系数 \(p_i\) 取 \(0\) 或 \(1\)，并且其余 \(n-1\) 个系数可以任意取值，所以每个 \(a_ib_i\) 在所有元素之和中都恰好出现 \(2^{n-1}\) 次.
因此 \(T_n\) 的所有元素之和为
\(2^{n-1}\sum_{i=1}^n a_ib_i=2^{n-1}S_n=2^{n-1}\left[8+(3n-4)2^{n+1}\right]\).
\end{enumerate}
\end{enumerate}
\end{solution}

\begin{problem}
已知 \(f(x)=ax-(\ln x)^2\).
\begin{enumerate}
\item 当 \(a=1\) 时，求 \(f(x)\) 在点 \((1,f(1))\) 处的切线方程；
\item 当 \(f(x)\) 有三个零点 \(x_1<x_2<x_3\) 时，
\begin{enumerate}
    \item 求 \(a\) 的取值范围；
    \item 证明 \((\ln x_2-\ln x_1)\cdot \ln x_3<\frac{4\e}{\e-1}\).
\end{enumerate}
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 当 \(a=1\) 时，\(f(x)=x-(\ln x)^2\)，则 \(f'(x)=1-\frac{2\ln x}{x}\).
所以 \(f'(1)=1\)，且 \(f(1)=1\)，故切线为 \(y=x\).

\item
\begin{enumerate}
\item 令 \(f(x)=0\)，得 \(a=\frac{(\ln x)^2}{x}\). 设 \(g(x)=\frac{(\ln x)^2}{x}\)（\(x>0\)），则 \(g'(x)=\frac{\ln x(2-\ln x)}{x^2}\).
由 \(g'(x)=0\) 得 \(x=1\) 或 \(x=\e^2\)，且
\(g(1)=0\)，\(g(\e^2)=\frac{4}{\e^2}\).
当 \(0<x<1\) 时，\(g'(x)<0\)；当 \(1<x<\e^2\) 时，\(g'(x)>0\)；当 \(x>\e^2\) 时，\(g'(x)<0\).
又当 \(x\to 0^+\) 时，\(g(x)\to +\infty\)；当 \(x\to +\infty\) 时，\(g(x)\to 0\).
\solutionfigure{\bitmapfigure[max width=0.55\linewidth]{img/2025/tianjin/q20_solution_fig1.png}}
因此方程 \(a=g(x)\) 在 \((0,+\infty)\) 内有 3 个根，当且仅当
\(0<a<\frac{4}{\e^2}\).

\item 由（1）中 \(g(x)\) 的单调性以及方程 \(g(x)=a\) 在 \((0,+\infty)\) 内有三个根，可得
\(
0<x_1<1<x_2<\e^2<x_3
\)
设 \(\ln x_1=t_1\)，\(\ln x_2=t_2\)，\(\ln x_3=t_3\)，则
\(
t_1<0<t_2<2<t_3
\)
且
\(ae^{t_1}=t_1^2\)，\(ae^{t_2}=t_2^2\)，\(ae^{t_3}=t_3^2\).
由后两个等式可得
\(\ln a+t_2=2\ln t_2\)，\(\ln a+t_3=2\ln t_3\).
两式相减，得
\(
t_3-t_2=2(\ln t_3-\ln t_2)
\)
由对数平均不等式，
\(\frac{t_3-t_2}{\ln t_3-\ln t_2}>\sqrt{t_2t_3}\)，
所以 \(t_2t_3<4\). 于是要证
\((\ln x_2-\ln x_1)\ln x_3<\frac{4\e}{\e-1}\)，
只需证
\(-t_1t_3\le \frac{4}{\e-1}\).
又因为 \(t_1<0\)，且 \(t_1^2=ae^{t_1}<a\)，故 \(-t_1<\sqrt{a}\). 再由 \(ae^{t_3}=t_3^2\) 得
\(\sqrt{a}\,t_3=\frac{t_3^2}{\e^{t_3/2}}\).
于是只需证明
\(\frac{t_3^2}{\e^{t_3/2}}\le \frac{4}{\e-1}\).
设 \(\varphi(t)=\frac{t^2}{\e^{t/2}}\)（\(t>2\)），则
\(\varphi'(t)=\frac{t(4-t)}{2\e^{t/2}}\).
所以当 \(2<t<4\) 时，\(\varphi'(t)>0\)；当 \(t>4\) 时，\(\varphi'(t)<0\). 故 \(\varphi(t)\) 在 \(t=4\) 处取得最大值
\(\varphi(4)=\frac{16}{\e^2}\).
而
\(\frac{16}{\e^2}<\frac{4}{\e-1}\)，
因为这等价于
\(4(\e-1)<\e^2\)，
即
\((\e-2)^2>0\).
从而
\(
-t_1t_3<\sqrt{a}\,t_3=\frac{t_3^2}{\e^{t_3/2}}\le \frac{16}{\e^2}<\frac{4}{\e-1}
\)
故
\(
(\ln x_2-\ln x_1)\cdot \ln x_3<\frac{4\e}{\e-1}
\)
\end{enumerate}
\end{enumerate}
\end{solution}
