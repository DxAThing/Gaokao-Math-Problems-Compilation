\section{单选题}

\begin{problem}
设全集 \(I = \{0, 1, 2, 3, 4\}\)，集合 \(A = \{0, 1, 2, 3\}\)，集合 \(B = \{2, 3, 4\}\)，则 \(\overline{A} \cup \overline{B} =\)（\quad）

\choices
    {\(\{0\} \)}
    {\(\{0,1\}\)}
    {\(\{0,1,4\}\)}
    {\(\{0,1,2,3,4\}\)}

\end{problem}

    {\((0, +\infty)\)}
    {\((0,2)\)}
    {\((1, +\infty)\)}
    {\((0,1)\)}

\begin{problem}

    {\(g(x) = x, h(x) = \lg (10^{x} + 10^{-x} + 2)\)}
