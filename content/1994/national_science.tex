\begin{problem}
已知 \(z = 1 + \i\).

\begin{enumerate}
\item 设 \(\omega = z^2 + 3\overline{z} - 4\)，求 \(\omega\) 的三角形式；
\item 如果 \(\frac{z^{2}+az+b}{z^{2}-z+1}=1-\i\)，求实数 \(a\)，\(b\) 的值.
\end{enumerate}

\end{problem}

\begin{problem}
设 \(\{a_{n}\}\) 是正数组成的数列，其前 \(n\) 项和为 \(S_{n}\)，且对于所有的自然数 \(n\)，\(a_{n}\) 与 \(2\) 的等差中项等于 \(S_{n}\) 与 \(2\) 的等比中项.

\begin{enumerate}
\item 写出数列 \(\{a_{n}\}\) 的前 \(3\) 项；
\item 求数列 \(\{a_{n}\}\) 的通项公式（写出推证过程）；
\item 令 \(b_{n} = \frac{1}{2}\left(\frac{a_{n}}{a_{n+1}} + \frac{a_{n+1}}{a_{n}}\right)\)（\(n \in \mathbb{N}\)），求 \(\lim\limits_{n \to \infty}(b_{1} + b_{2} + \cdots + b_{n} - n)\).
\end{enumerate}
\end{problem}
