\section{解答题}

\begin{problem}
已知 \(\sin\alpha=\frac{3}{5}\)，\(\alpha\in\left(\frac{\pi}{2},\pi\right)\)，\(\tan(\pi-\beta)=\frac{1}{2}\)，求 \(\tan(\alpha-2\beta) \) 的值.

\end{problem}

\begin{solution}
因为 \(\sin \alpha = \frac{3}{5}\)，\(\alpha \in \left(\frac{\pi}{2}, \pi\right)\)，所以 \(\cos \alpha = -\frac{4}{5}\)，\(\tan \alpha = -\frac{3}{4}\)。
又因为 \(\tan(\pi - \beta) = \frac{1}{2}\)，所以 \(\tan \beta = -\frac{1}{2}\)，从而
\[
\tan 2\beta=\frac{2\tan \beta}{1 - \tan^2 \beta}=\frac{2 \cdot \left(-\frac{1}{2}\right)}{1 - \left(-\frac{1}{2}\right)^2}=-\frac{4}{3}
\]
于是
\[
\tan(\alpha - 2\beta)=\frac{\tan \alpha - \tan 2\beta}{1 + \tan \alpha \tan 2\beta}=\frac{-\frac{3}{4} + \frac{4}{3}}{1 + \left(-\frac{3}{4}\right)\left(-\frac{4}{3}\right)}=\frac{7}{24}
\]
\end{solution}

\begin{problem}
已知空间三个点 \(P(-2,0,2)\)，\(Q(-1,1,2)\) 和 \(R(-3,0,4)\). 设 \(\bs{a} = \overrightarrow{PQ}\)，\(\bs{b} = \overrightarrow{PR}\).

\begin{enumerate}
\item 求 \(\bs{a} \) 与 \(\bs{b} \) 的夹角 \(\theta \)（用反三角函数表示）；
\item 试确定实数 \(k\)，使 \(k \bs{a} + \bs{b}\) 与 \(k \bs{a} - 2\bs{b}\) 互相垂直.
\end{enumerate}

\end{problem}

\begin{solution}
（1）\(\bs{a} = \overrightarrow{PQ} = (1,1,0)\)，\(\bs{b} = \overrightarrow{PR} = (-1,0,2)\). 所以 \(\left|\bs{a}\right|=\sqrt{2}\)，\(\left|\bs{b}\right|=\sqrt{5}\)，
\[
\cos \theta=\frac{\bs{a}\cdot \bs{b}}{|\bs{a}||\bs{b}|}=\frac{-1}{\sqrt{10}}=-\frac{\sqrt{10}}{10}
\]
因此 \(\theta = \arccos\left(-\frac{\sqrt{10}}{10}\right)\).

（2）\((k\bs{a}+\bs{b})\cdot(k\bs{a}-2\bs{b}) = \bigl(k-1,k,2\bigr)\cdot \bigl(k+2,k,-4\bigr) = 2k^2+k-10\)。由两向量互相垂直得 \(2k^2+k-10=0\)，解得 \(k=-\frac{5}{2}\) 或 \(k=2\).
\end{solution}

\begin{problem}
如图，在梯形 \(ABCD\) 中，\(AD // BC\)，\(\angle ABC = \frac{\pi}{2}\)，\(AB = a\)，\(AD = 3a\)，且 \(\angle ADC = \arcsin \frac{\sqrt{5}}{5}\)，又 \(PA \perp\) 平面 \(ABCD\)，\(PA = a\). 求：

\begin{enumerate}
\item 二面角 \(P - CD - A\) 的大小（用反三角函数表示）；
\item 点 \(A\) 到平面 \(PBC\) 的距离.
\end{enumerate}

\bitmapfigure[max width=0.58\linewidth,max totalheight=4.8cm]{img/1994/shanghai_liberal/q23_fig1.png}

\end{problem}

\begin{solution}
（1）在平面 \(ABCD\) 内，过 \(A\) 作 \(AE \perp CD\)，垂足为 \(E\)，连接 \(PE\).
因为 \(PA \perp\) 平面 \(ABCD\)，所以由三垂线定理知 \(PE \perp CD\)，从而 \(\angle PEA\) 是二面角 \(P - CD - A\) 的平面角.
在直角三角形 \(AED\) 中，\(AD = 3a\)，\(\angle ADE = \arcsin \frac{\sqrt{5}}{5}\)，所以 \(AE = AD \sin \angle ADE = \dfrac{3\sqrt{5}}{5}a\)。
在直角三角形 \(PAE\) 中，\(\tan \angle PEA = \dfrac{PA}{AE} = \dfrac{\sqrt{5}}{3}\)，
所以二面角 \(P - CD - A\) 的大小为 \(\arctan \frac{\sqrt{5}}{3}\).

（2）在平面 \(PAB\) 内，过 \(A\) 作 \(AH \perp PB\)，垂足为 \(H\).
因为 \(PA \perp\) 平面 \(ABCD\)，所以 \(PA \perp EC\)，又 \(AB \perp BC\)，所以 \(BC \perp\) 平面 \(PAB\)，因此 \(BC \perp AH\)，从而 \(AH \perp\) 平面 \(PBC\). 故 \(AH\) 的长即为点 \(A\) 到平面 \(PBC\) 的距离.
在等腰直角三角形 \(PAB\) 中，\(AH = \frac{\sqrt{2}}{2}a\). 所以，点 \(A\) 到平面 \(PBC\) 的距离为 \(\frac{\sqrt{2}}{2}a\).
\end{solution}

\begin{problem}
如图，抛物线 \(y = 3x^{2}\) 和 \(y = -x^{2} + 4ax\ (a > 0)\) 交于原点 \(O\) 和点 \(A\).

\begin{enumerate}
\item 求这两条抛物线所围成的图形的面积；
\item 过点 \(A\) 作抛物线 \(y = -x^{2} + 4ax\) 的切线，该切线与抛物线 \(y = 3x^{2}\) 交于另一点 \(P\)，求线段 \(AP\) 的垂直平分线的方程.
\end{enumerate}

\bitmapfigure[max width=0.42\linewidth,max totalheight=4.8cm]{img/1994/shanghai_liberal/q24_fig1.png}

\end{problem}

\begin{solution}
（1）由题设可得
\[
\begin{cases}
y = 3x^2,\\
y = -x^2 + 4ax.
\end{cases}
\]
解得交点 \(A(a,3a^2)\). 所以
\[
S=\displaystyle\int_0^a \bigl(-x^2 + 4ax - 3x^2\bigr)\,dx=\left(-\frac{4}{3}x^3 + 2ax^2\right)\Big|_0^a=\frac{2}{3}a^3
\]

（2）由 \(y = -x^2 + 4ax\) 得 \(y' = -2x + 4a\)，故点 \(A\) 处切线斜率为 \(2a\)，切线方程为 \(y - 3a^2 = 2a(x - a)\)，即 \(y = 2ax + a^2\)。
联立
\[
\begin{cases}
y = 2ax + a^2,\\
y = 3x^2,
\end{cases}
\]
得 \(x = -\dfrac{a}{3}\)，\(y = \dfrac{a^2}{3}\)。故点 \(P\) 的坐标为 \(\left(-\dfrac{a}{3}, \dfrac{a^2}{3}\right)\)。

设 \(AP\) 的中点为 \(M\)，则 \(M\left(\dfrac{a}{3}, \dfrac{5a^2}{3}\right)\)。
因为 \(AP\) 的斜率为 \(2a\)，所以其垂直平分线的斜率为 \(-\dfrac{1}{2a}\)。于是垂直平分线方程为 \(y - \dfrac{5a^2}{3} = -\dfrac{1}{2a}\left(x - \dfrac{a}{3}\right)\)，即 \(y = -\dfrac{1}{2a}x + \dfrac{5a^2}{3} + \dfrac{1}{6}\)。
\end{solution}

\begin{problem}
设 \(\{a_{n}\}\) 是公差为 \(d\) 的等差数列，\(a_{3} + a_{5} = 2\)，\(S_{20} = a_{1} + a_{2} + \cdots + a_{20} = 150\)，又 \(b_{n} = 2^{a_{n} - 2a_{n+1}}\left(n \in \N\right)\).

\begin{enumerate}
\item 求 \(a_{1}\)，\(d\) 的值；
\item 求证 \(\left|b_{n}\right|\) 是等比数列，并求 \(b_{4}\)；
\item 设 \(k\) 为某个自然数，且满足 \(\lim\limits_{n\to\infty}\left(b_k b_{k+1} + b_{k+1} b_{k+2} + \cdots + b_n b_{n+1}\right) = \frac{1}{96}\)，求 \(k\) 的值.
\end{enumerate}

\end{problem}

\begin{solution}
（1）由题设
\[
\begin{cases}
a_3 + a_5 = 2,\\
S_{20} = \dfrac{(a_1 + a_{20})\cdot 20}{2} = 150.
\end{cases}
\]
又 \(a_3 = a_1 + 2d\)，\(a_5 = a_1 + 4d\)，\(a_{20} = a_1 + 19d\)，所以
\[
\begin{cases}
2a_1 + 6d = 2,\\
10(2a_1 + 19d) = 150.
\end{cases}
\]
解得 \(a_1 = -2\)，\(d = 1\).

（2）由 \(b_n = 2^{a_n - 2a_{n+1}}\)，\(b_{n+1} = 2^{a_{n+1} - 2a_{n+2}}\)，
\(\dfrac{b_{n+1}}{b_n} = 2^{3a_{n+1} - 2a_{n+2} - a_n}\).
而 \(\{a_n\}\) 是等差数列，\(a_{n+1} = a_n + d\)，\(a_{n+2} = a_n + 2d\)，所以 \(3a_{n+1} - 2a_{n+2} - a_n = -d = -1\)。
故 \(\dfrac{b_{n+1}}{b_n} = 2^{-1} = \dfrac{1}{2}\)，所以 \(\{b_n\}\) 是等比数列，且 \(b_1 = 1\)，于是 \(b_n = \left(\dfrac{1}{2}\right)^{n-1}\)。

（3）由（2）得 \(b_k b_{k+1} = \left(\dfrac{1}{2}\right)^{2k-1}\)。
因此 \(b_k b_{k+1} + b_{k+1}b_{k+2} + \cdots\) 是首项为 \(\left(\dfrac12\right)^{2k-1}\)，公比为 \(\left(\dfrac12\right)^2\) 的无穷等比数列的和，故
\[
\lim\limits_{n\to\infty}\left(b_k b_{k+1} + b_{k+1}b_{k+2} + \cdots + b_n b_{n+1}\right)=\frac{\left(\frac12\right)^{2k-1}}{1-\left(\frac12\right)^2}=\frac{8}{3 \cdot 2^{2k}}
\]
由题设等于 \(\dfrac{1}{96}\)，所以 \(\dfrac{8}{3 \cdot 2^{2k}} = \dfrac{1}{96}\)，解得 \(2^{2k} = 2^8\)，因此 \(k = 4\).
\end{solution}

\begin{problem}
设 \(P(1, t)\) 是直角坐标系内的点（\(t > 0\)）. 将向量 \(\overrightarrow{OP}\) 绕原点 \(O\) 按逆时针方向旋转 \(\frac{\pi}{2}\)，再把它的模变为原来的两倍，得到向量 \(\overrightarrow{OQ}\)，以 \(OP\)，\(OQ\) 为邻边作矩形 \(OPRQ\).
\begin{enumerate}
\item 求点 \(Q\)，\(R\) 的坐标；
\item 求矩形 \(OPRQ\) 在第一象限部分的面积 \(S(t)\)；
\item 确定函数 \(S(t)\) 的单调区间，并加以证明.
\end{enumerate}

\end{problem}

\begin{solution}
（1）\(\overrightarrow{OP} = (1,t)\). 将它绕原点逆时针旋转 \(\frac{\pi}{2}\) 后，再把模变为原来的两倍，得到 \(\overrightarrow{OQ} = (-2t,2)\)。
所以 \(Q(-2t,2)\). 又 \(\overrightarrow{OR} = \overrightarrow{OP} + \overrightarrow{OQ} = (-2t+1, t+2)\)，所以 \(R(-2t+1,t+2)\).

\bitmapfigure[max width=0.64\linewidth,max totalheight=4.8cm]{img/1994/shanghai_liberal/q26_solution_fig1.png}
\bitmapfigure[max width=0.64\linewidth,max totalheight=4.8cm]{img/1994/shanghai_liberal/q26_solution_fig2.png}

（2）当 \(-2t+1>0\)，即 \(0<t<\frac12\) 时，点 \(R\) 在第一象限内，此时 \(S(t)\) 为四边形 \(OPRQ\) 在第一象限部分的面积. 直线 \(QR\) 的方程为 \(y-2=t(x+2t)\). 令 \(x=0\)，得 \(y=2t^2+2\)，设交点为 \(K(0,2t^2+2)\). 因而
\[
S(t)=S_{OPRQ}-S_{OKQ}=2(1+t^2)-\frac12(2t^2+2)\cdot 2t=2(1-t+t^2-t^3)
\]

当 \(-2t+1\le0\)，即 \(t\ge\frac12\) 时，点 \(R\) 在 \(y\) 轴上或第二象限内，此时 \(S(t)\) 为 \(\triangle OPL\) 的面积. 直线 \(PR\) 的方程为 \(y-t=-\dfrac{1}{t}(x-1)\).
令 \(x=0\)，得 \(y=t+\dfrac1t\)，设交点为 \(L\!\left(0,t+\dfrac1t\right)\). 所以 \(S(t)=\dfrac12\left(t+\dfrac1t\right)\).

因此
\[
S(t)=
\begin{cases}
2(1-t+t^2-t^3), & 0<t<\frac12,\\[0.2em]
\dfrac12\left(t+\dfrac1t\right), & t\ge\frac12.
\end{cases}
\]

（3）当 \(0<t_1<t_2<\frac12\) 时，
\(S(t_1)-S(t_2)=2(t_2-t_1)\bigl[1-(t_1+t_2)+(t_1^2+t_1t_2+t_2^2)\bigr]>0\)，
所以 \(S(t)\) 在 \((0,\frac12)\) 内是减函数.

当 \(\frac12\le t_1<t_2\le1\) 时，
\(S(t_1)-S(t_2)=\dfrac12(t_1-t_2)\left(1-\dfrac{1}{t_1t_2}\right)>0\)，
所以 \(S(t)\) 在 \([\frac12,1]\) 内也是减函数.

又
\(S\!\left(\dfrac12\right)=2\left(1-\dfrac12+\dfrac14-\dfrac18\right)=\dfrac54\).
对于 \(0<t_1<\frac12\le t_2\le1\)，有 \(S(t_1)>\dfrac54\ge S(t_2)\)，所以 \(S(t)\) 在 \((0,1]\) 内是减函数.

当 \(1\le t_1<t_2\) 时，
\(S(t_1)-S(t_2)=\dfrac12(t_1-t_2)\left(1-\dfrac{1}{t_1t_2}\right)<0\)，
所以 \(S(t)\) 在 \([1,+\infty)\) 内是增函数.

因此，\(S(t)\) 的单调区间分别为 \((0,1]\) 和 \([1,+\infty)\).
\end{solution}
