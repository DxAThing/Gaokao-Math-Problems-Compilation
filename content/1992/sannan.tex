\chapter{1992年三南卷}
\section{单选题}

\begin{problem}
设函数 \(z=\i^2+\sqrt{3}\i\)，那么 \(\arg z\) 是（\quad）

\choices
    {\(\frac{5}{6}\pi\)}
    {\(\frac{\pi}{3}\)}
    {\(\frac{2}{3}\pi\)}
    {\(-\frac{4}{3}\pi\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
由 \(\i^2=-1\)，得
\[
z=-1+\sqrt{3}\i
\]
它的实部为 \(-1\)，虚部为 \(\sqrt{3}>0\)，所以 \(z\) 位于第二象限. 设主值辐角为 \(\theta\)，则 \(\tan\theta=-\sqrt{3}\)，且 \(\theta\in(\frac{\pi}{2},\pi)\)，从而
\[
\theta=\frac{2\pi}{3}
\]
因此 \(\arg z=\frac{2\pi}{3}\)，故选 C.
\end{solution}

\begin{problem}
如果等边圆柱（即底面直径与母线相等的圆柱）的体积是 \(16\pi\mathrm{~cm}^3\)，那么它的底面半径等于（\quad）

\choices
    {\(4\sqrt[3]{2}\mathrm{~cm}\)}
    {\(4\mathrm{~cm}\)}
    {\(2\sqrt[3]{2}\mathrm{~cm}\)}
    {\(2\mathrm{~cm}\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
设底面半径为 \(r>0\)，高为 \(h\). 由等边圆柱的定义，底面直径等于母线，而圆柱的母线就是高，所以 \(h=2r\). 由圆柱体积公式，
\[
16\pi=\pi r^2h=2\pi r^3
\]
约去 \(2\pi\) 得 \(r^3=8\). 因为 \(r>0\)，所以 \(r=2\mathrm{~cm}\)，故选 D.
\end{solution}

\begin{problem}
\(\frac{\arcsin \frac{\sqrt{3}}{2}-\arccos \left( -\frac{1}{2} \right)}{\arctan \left( -\sqrt{3} \right)}\) 的值等于（\quad）

\choices
    {\(1\)}
    {\(0\)}
    {\(-\frac{2}{5}\)}
    {\(\frac{6}{5}\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
由反三角函数的主值范围 \(\arcsin u\in[-\frac\pi2,\frac\pi2]\)、\(\arccos u\in[0,\pi]\)、\(\arctan u\in(-\frac\pi2,\frac\pi2)\)，有
\(\arcsin\frac{\sqrt{3}}2=\frac\pi3\)，\(\arccos\left(-\frac12\right)=\frac{2\pi}3\)，\(\arctan(-\sqrt{3})=-\frac\pi3\).
所以分子为 \(\frac\pi3-\frac{2\pi}3=-\frac\pi3\)，分母为 \(-\frac\pi3\)，原式为
\[
\frac{-\frac\pi3}{-\frac\pi3}=1
\]
故选 A.
\end{solution}

\begin{problem}
函数 \(y=\log_{\frac{1}{2}}(1-x)\)（\(x<1\)）的反函数是（\quad）

\choices
    {\(y=1+2^{-x}\)（\(x\in\R\)）}
    {\(y=1-2^{-x}\)（\(x\in\R\)）}
    {\(y=1+2^x\)（\(x\in\R\)）}
    {\(y=1-2^x\)（\(x\in\R\)）}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
原函数的定义域为 \(x<1\)，此时 \(1-x>0\). 由于底数 \(\frac12\) 大于 \(0\) 且不等于 \(1\)，当 \(x\) 遍历 \((-\infty,1)\) 时，\(y\) 的值域为 \(\R\)，所以反函数的定义域为 \(\R\). 交换原函数中的 \(x\) 与 \(y\)，得
\[
x=\log_{\frac12}(1-y)
\]
由对数定义，
\[
\left(\frac12\right)^x=1-y
\]
于是
\(y=1-\left(\frac12\right)^x=1-2^{-x}\)，\(x\in\R\).
故选 B.
\end{solution}

\begin{problem}
在长方体 \(ABCD-A_1B_1C_1D_1\) 中，如果 \(AB=BC=a\)，\(AA_1=2a\)，那么点 \(A\) 到直线 \(A_1C\) 的距离等于（\quad）

\choices
    {\(\frac{2\sqrt{6}}{3}a\)}
    {\(\frac{3\sqrt{6}}{2}a\)}
    {\(\frac{2\sqrt{3}}{3}a\)}
    {\(\frac{\sqrt{6}}{3}a\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
取 \(A\) 为原点，令 \(AB\)、\(BC\)、\(AA_1\) 分别沿三个坐标轴正向，则
\(A_1=(0,0,2a)\)，\(C=(a,a,0)\).
设直线 \(A_1C\) 上的点为 \(P\)，用参数 \(t\) 表示为
\[
P=A_1+t(C-A_1)=(ta,ta,2a(1-t))
\]
于是
\[
AP^2=a^2\bigl(t^2+t^2+4(1-t)^2\bigr)
=a^2\left(6\left(t-\frac23\right)^2+\frac43\right)
\]
当 \(t=\frac23\) 时，\(AP^2\) 取得最小值 \(\frac43a^2\)，所以点 \(A\) 到直线 \(A_1C\) 的距离为
\[
d=\sqrt{\frac43a^2}=\frac{2\sqrt{3}}3a
\]
故选 C.
\end{solution}

\begin{problem}
函数 \(y=\sin x\cos x+\sqrt{3}\cos^2 x-\frac{\sqrt{3}}{2}\) 的最小正周期等于（\quad）

\choices
    {\(\pi\)}
    {\(2\pi\)}
    {\(\frac{\pi}{4}\)}
    {\(\frac{\pi}{2}\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
利用二倍角公式 \(\sin x\cos x=\frac12\sin 2x\) 和 \(\cos^2x=\frac{1+\cos 2x}{2}\)，原函数化为
\[
y=\frac12\sin2x+\frac{\sqrt{3}}{2}(1+\cos2x)-\frac{\sqrt{3}}2
=\frac12\sin2x+\frac{\sqrt{3}}2\cos2x
=\sin\left(2x+\frac{\pi}{3}\right)
\]
最后一步利用了 \(\sin(u+\frac\pi3)=\frac12\sin u+\frac{\sqrt3}{2}\cos u\). 因为正弦函数的最小正周期是 \(2\pi\)，故满足 \(2T=2\pi\) 的最小正周期为 \(T=\pi\)，故选 A.
\end{solution}

\begin{problem}
有一个椭圆，它的极坐标方程是（\quad）

\choices
    {\(\rho=\frac{5}{\sqrt{3}-2\cos\theta}\)}
    {\(\rho=\frac{5}{\sqrt{3}-\sqrt{3}\cos\theta}\)}
    {\(\rho=\frac{2-\sqrt{3}\cos\theta}{5}\)}
    {\(\rho=\frac{5}{2-\sqrt{3}\cos\theta}\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
以焦点为极点的圆锥曲线极坐标方程可写成
\[
\rho=\frac{p}{1-e\cos\theta}
\]
其中离心率满足：\(0<e<1\) 时为椭圆，\(e=1\) 时为抛物线，\(e>1\) 时为双曲线. 逐项比较，选项 A 中 \(e=\frac{2}{\sqrt3}>1\)，表示双曲线；选项 B 中 \(e=1\)，表示抛物线；选项 C 不是上述以 \(\rho\) 为分母的圆锥曲线标准形式. 选项 D 可化为
\[
\rho=\frac{5/2}{1-\frac{\sqrt{3}}2\cos\theta}
\]
其离心率为 \(e=\frac{\sqrt3}{2}\)，满足 \(0<e<1\)，所以表示椭圆，故选 D.
\end{solution}

\begin{problem}
不等式 \(\left| \sqrt{x-2}-3 \right|<1\) 的解集是（\quad）

\choices
    {\(\{x\mid 5<x<16\}\)}
    {\(\{x\mid 6<x<18\}\)}
    {\(\{x\mid 7<x<20\}\)}
    {\(\{x\mid 8<x<22\}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
根式有意义要求 \(x-2\geqslant0\)，即 \(x\geqslant2\). 在此定义域内，
\[
\left|\sqrt{x-2}-3\right|<1
\Longleftrightarrow 2<\sqrt{x-2}<4
\]
两端均为正数，可以平方而不改变不等号方向，得
\[
4<x-2<16
\]
即 \(6<x<18\). 这个区间已自动满足 \(x\geqslant2\)，所以解集为 \(\{x\mid6<x<18\}\)，故选 B.
\end{solution}

\begin{problem}
设等差数列 \(\{a_n\}\) 的公差是 \(d\)，如果它的前 \(n\) 项和 \(S_n=-n^2\)，那么（\quad）

\choices
    {\(a_n=2n-1\)，\(d=-2\)}
    {\(a_n=2n-1\)，\(d=2\)}
    {\(a_n=-2n+1\)，\(d=-2\)}
    {\(a_n=-2n+1\)，\(d=2\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
令 \(S_0=0\). 由 \(a_n=S_n-S_{n-1}\)，有
\[
a_n=S_n-S_{n-1}=-n^2+(n-1)^2=-2n+1
\]
因此
\[
d=a_{n+1}-a_n=\bigl[-2(n+1)+1\bigr]-(-2n+1)=-2
\]
所以 \(a_n=-2n+1\)、\(d=-2\)，故选 C.
\end{solution}

\begin{problem}
方程 \(\cos 2x=3\cos x+1\) 的解集是（\quad）

\choices
    {\(\left\{x\mid x=2k\pi\pm\frac{2}{3}\pi,\ k\in\Z\right\}\)}
    {\(\left\{x\mid x=k\pi\pm\frac{1}{3}\pi,\ k\in\Z\right\}\)}
    {\(\left\{x\mid x=k\pi\pm\frac{2}{3}\pi,\ k\in\Z\right\}\)}
    {\(\left\{x\mid x=2k\pi\pm\frac{1}{3}\pi,\ k\in\Z\right\}\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
令 \(u=\cos x\)，则 \(-1\leqslant u\leqslant1\). 由 \(\cos 2x=2\cos^2x-1\)，原方程变为
\(2u^2-1=3u+1\)，\(2u^2-3u-2=0\).
因式分解得
\[
(2u+1)(u-2)=0
\]
根 \(u=2\) 不符合 \(-1\leqslant u\leqslant1\)，故只能取 \(u=-\frac12\). 于是
\(x=2k\pi\pm\frac{2}{3}\pi\)，\(k\in\Z\).
这些值都满足 \(\cos x=-\frac12\)，故解集为选项 A，故选 A.
\end{solution}

\begin{problem}
有一条半径是 \(2\) 的弧，其度数是 \(60^\circ\)，它绕经过弧的中点的直径旋转得到一个球冠，那么这个球冠的面积是（\quad）

\choices
    {\(4(2-\sqrt{3})\pi\)}
    {\(2(2-\sqrt{3}\pi)\)}
    {\(4\sqrt{3}\pi\)}
    {\(2\sqrt{3}\pi\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
旋转所得球面的半径等于原弧所在圆的半径，因此 \(R=2\). 原弧的圆心角为 \(60^\circ\)，而旋转轴是经过弧中点的直径，所以弧的两个端点与旋转轴的夹角都是 \(30^\circ\). 球冠的高为
\[
h=R-R\cos30^\circ=2-\sqrt{3}
\]
球冠面积公式为 \(2\pi Rh\)，故
\[
S=2\pi\cdot2(2-\sqrt3)=4(2-\sqrt3)\pi
\]
故选 A.
\end{solution}

\begin{problem}
某小组共有 \(10\) 名学生，其中女生 \(3\) 名. 现选举 \(2\) 名代表，至少有 \(1\) 名女生当选的不同的选法共有（\quad）

\choices
    {\(27\) 种}
    {\(48\) 种}
    {\(21\) 种}
    {\(24\) 种}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
按当选女生人数分类，两个代表中恰有一名女生时，先从三名女生中选一名，再从七名男生中选一名，选法数为
\[
\mathrm C_3^1\mathrm C_7^1=3\times7=21
\]
两个代表都是女生时，选法数为
\[
\mathrm C_3^2=3
\]
这两类互不相交且正好覆盖“至少有一名女生”的情形，所以总数为 \(21+3=24\)，故选 D.
\end{solution}

\begin{problem}
设全集 \(U=\R\)，集合 \(M=\left\{x\mid \sqrt{x^2}>2\right\}\)，\(N=\left\{x\mid \log_x 7>\log_3 7\right\}\)，那么 \(M\cap\overline{N}=\)（\quad）

\choices
    {\(\{x\mid x<-2\}\)}
    {\(\{x\mid x<-2\text{\ 或\ }x\geqslant 3\}\)}
    {\(\{x\mid x\geqslant 3\}\)}
    {\(\{x\mid -2\leqslant x<3\}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
因为 \(\sqrt{x^2}=|x|\)，所以
\[
M=\{x\mid |x|>2\}=\{x\mid x<-2\text{ 或 }x>2\}
\]
对集合 \(N\)，对数的定义域要求 \(x>0\) 且 \(x\ne1\). 当 \(0<x<1\) 时，\(\ln x<0\)，所以 \(\log_x7=\frac{\ln7}{\ln x}<0\)，而 \(\log_37>0\)，不满足不等式. 当 \(x>1\) 时，\(\ln x>0\)，且
\[
\log_x7>\log_37\Longleftrightarrow \frac{\ln7}{\ln x}>\frac{\ln7}{\ln3}
\Longleftrightarrow \ln x<\ln3
\Longleftrightarrow 1<x<3
\]
因此 \(N=(1,3)\). 全集为 \(\R\)，所以
\[
\overline N=(-\infty,1]\cup[3,+\infty)
\]
与 \(M\) 相交时，\(x<-2\) 全部保留；在 \(x>2\) 中只有 \(x\geqslant3\) 不属于 \(N\). 故
\[
M\cap\overline N=(-\infty,-2)\cup[3,+\infty)
\]
即 \(\{x\mid x<-2\text{ 或 }x\geqslant3\}\)，故选 B.
\end{solution}

\begin{problem}
设 \(\{a_n\}\) 是由正数组成的等比数列，公比 \(q=2\)，且 \(a_1a_2a_3\cdots a_{30}=2^{30}\)，那么 \(a_3a_6a_9\cdots a_{30}\) 等于（\quad）

\choices
    {\(2^{10}\)}
    {\(2^{20}\)}
    {\(2^{16}\)}
    {\(2^{15}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
设首项为 \(a_1=A>0\)，则由公比 \(q=2\) 有 \(a_n=A2^{n-1}\). 于是
\[
A^{30}2^{0+1+\cdots+29}=2^{30}
\]
因为 \(0+1+\cdots+29=\frac{29\cdot30}{2}=435\)，所以
\[
A^{30}=2^{30-435}=2^{-405}
\]
又因 \(A>0\)，故 \(A=2^{-405/30}=2^{-27/2}\). 所求乘积含十项，因此
\[
a_3a_6\cdots a_{30}=A^{10}2^{(3-1)+(6-1)+\cdots+(30-1)}
\]
其中
\[
(3-1)+(6-1)+\cdots+(30-1)=(3+6+\cdots+30)-10=3\cdot\frac{10\cdot11}{2}-10=155
\]
所以
\[
a_3a_6\cdots a_{30}=2^{-135}2^{155}=2^{20}
\]
故选 B.
\end{solution}

\begin{problem}
设 \(\triangle ABC\) 不是直角三角形，\(A\) 和 \(B\) 是它的两个内角，那么（\quad）

\choices
    {“\(A<B\)”是“\(\tan A<\tan B\)”的充分条件，但不是必要条件}
    {“\(A<B\)”是“\(\tan A<\tan B\)”的必要条件，但不是充分条件}
    {“\(A<B\)”是“\(\tan A<\tan B\)”的充分必要条件}
    {“\(A<B\)”不是“\(\tan A<\tan B\)”的充分条件，也不是必要条件}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
要判断“\(A<B\)”是否为“\(\tan A<\tan B\)”的充分或必要条件，分别构造反例即可.

取 \(A=30^\circ\)、\(B=120^\circ\)，则第三个内角为 \(30^\circ\)，确实构成非直角三角形，并且 \(A<B\). 但是
\[
\tan A=\frac{\sqrt3}{3}>-\sqrt3=\tan B
\]
所以 \(A<B\) 不能推出 \(\tan A<\tan B\)，不是充分条件.

再取 \(A=120^\circ\)、\(B=30^\circ\)，第三个内角为 \(30^\circ\)，仍是非直角三角形. 此时
\[
\tan A=-\sqrt3<\frac{\sqrt3}{3}=\tan B
\]
但 \(A<B\) 不成立，所以 \(A<B\) 也不是必要条件. 故选 D.
\end{solution}

\begin{problem}
对于定义域是 \(\R\) 的任何奇函数 \(f(x)\)，都有（\quad）

\choices
    {\(f(x)-f(-x)>0\)（\(x\in\R\)）}
    {\(f(x)-f(-x)\leqslant 0\)（\(x\in\R\)）}
    {\(f(x)f(-x)\leqslant 0\)（\(x\in\R\)）}
    {\(f(x)f(-x)>0\)（\(x\in\R\)）}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
因为 \(f\) 是奇函数，对任意 \(x\in\R\) 都有 \(f(-x)=-f(x)\)，于是
\[
f(x)f(-x)=-f(x)^2\leqslant0
\]
其中等号允许在 \(f(x)=0\) 时成立，所以选项 C 对所有 \(x\in\R\) 都成立. 故选 C.
\end{solution}

\begin{problem}
如果双曲线的两条渐近线的方程是 \(y=\pm\frac{3}{2}x\)，焦点坐标是 \((-\sqrt{26},0)\) 和 \((\sqrt{26},0)\)，那么它的两条准线之间的距离是（\quad）

\choices
    {\(\frac{8}{13}\sqrt{26}\)}
    {\(\frac{4}{13}\sqrt{26}\)}
    {\(\frac{18}{13}\sqrt{26}\)}
    {\(\frac{9}{13}\sqrt{26}\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
由焦点在 \(x\) 轴上且渐近线关于两坐标轴对称，设双曲线方程为
\(\frac{x^2}{a^2}-\frac{y^2}{b^2}=1\)，\(a>0\)，\(\ b>0\).
其渐近线为 \(y=\pm\frac ba x\)，因此
\(\frac{b}{a}=\frac32\)，\(b^2=\frac94a^2\).
焦点横坐标为 \(c=\sqrt{26}\)，且双曲线满足 \(c^2=a^2+b^2\)，所以
\(\frac{13}{4}a^2=26\)，\(a^2=8\).
离心率为 \(e=\frac ca\)，故两条准线分别为
\[
x=\pm\frac ae=\pm\frac{a^2}{c}
\]
所以两准线之间的距离为
\[
\frac{2a^2}{c}=\frac{16}{\sqrt{26}}=\frac{8\sqrt{26}}{13}
\]
故选 A.
\end{solution}

\section{填空题}

\begin{problem}
\(\tan \frac{\pi}{8}=\)\fillinblank{}.
\end{problem}

\begin{answer}
\(\sqrt{2}-1\)
\end{answer}

\begin{solution}
因为 \(\frac\pi8\in(0,\frac\pi2)\)，半角公式取正号，故
\[
\tan\frac{\pi}{8}=\sqrt{\frac{1-\cos\frac{\pi}{4}}{1+\cos\frac{\pi}{4}}}
=\sqrt{\frac{1-\frac{\sqrt{2}}2}{1+\frac{\sqrt{2}}2}}
 =\sqrt{\frac{2-\sqrt2}{2+\sqrt2}}
\]
将根式内的分式分子分母同乘 \(2-\sqrt2\)，得
\[
\frac{2-\sqrt2}{2+\sqrt2}
=\frac{(2-\sqrt2)^2}{4-2}=3-2\sqrt2=(\sqrt2-1)^2
\]
由于 \(\sqrt2-1>0\)，所以 \(\tan\frac\pi8=\sqrt2-1\).
\end{solution}

\begin{problem}
设直线的参数方程是 \(\begin{cases}
x=2+\frac{1}{2}t,\\
y=3+\frac{\sqrt{3}}{2}t,
\end{cases}\) 那么它的斜截式方程是\fillinblank{}.
\end{problem}

\begin{answer}
\(y=\sqrt{3}x+3-2\sqrt{3}\)
\end{answer}

\begin{solution}
由参数方程的第一式，因 \(\frac12\ne0\)，可解得
\[
t=2(x-2)
\]
将其代入第二式，得
\[
y=3+\frac{\sqrt{3}}2\cdot2(x-2)=\sqrt{3}x+3-2\sqrt{3}
\]
由于 \(t\) 可取任意实数，参数方程上的所有点都满足上述直线方程，且反过来直线上每个 \(x\) 都对应 \(t=2(x-2)\). 所以斜截式方程为 \(y=\sqrt{3}x+3-2\sqrt{3}\).
\end{solution}

\begin{problem}
如果三角形的顶点分别是 \(O(0,0)\)，\(A(0,15)\)，\(B(-8,0)\)，那么它的内切圆方程是\fillinblank{}.
\end{problem}

\begin{answer}
\((x+3)^2+(y-3)^2=9\)
\end{answer}

\begin{solution}
先求三边长：
\(OA=15\)，\(OB=8\)，\(AB=\sqrt{8^2+15^2}=17\).
所以三角形是以 \(O\) 为直角顶点的直角三角形. 其内切圆半径为
\[
r=\frac{OA+OB-AB}{2}=\frac{15+8-17}{2}=3
\]
再按内心坐标公式，内心坐标是以各顶点为权、权分别取对边长：
\[
I=\frac{17O+8A+15B}{17+8+15}=(-3,3)
\]
这也与内心到两条直角边所在坐标轴的距离相等相符. 因此内切圆方程为
\[
(x+3)^2+(y-3)^2=3^2
\]
即 \((x+3)^2+(y-3)^2=9\).
\end{solution}

\begin{problem}
\(\displaystyle\lim_{n\to\infty}\left[ \frac{1}{1\cdot 4}+\frac{1}{4\cdot 7}+\frac{1}{7\cdot 10}+\cdots+\frac{1}{(3n-2)(3n+1)} \right]=\)\fillinblank{}.
\end{problem}

\begin{answer}
\(\frac13\)
\end{answer}

\begin{solution}
对任意正整数 \(k\)，作部分分式分解：
\[
\frac{1}{(3k-2)(3k+1)}=\frac13\left(\frac{1}{3k-2}-\frac{1}{3k+1}\right)
\]
因此前 \(n\) 项和为
\[
S_n=\frac13\left(1-\frac14+\frac14-\frac17+\cdots+\frac{1}{3n-2}-\frac{1}{3n+1}\right)=\frac13\left(1-\frac{1}{3n+1}\right)
\]
因为 \(\frac{1}{3n+1}\to0\)，所以
\[
\lim_{n\to\infty}S_n=\frac13
\]
\end{solution}

\begin{problem}
\(91^{92}\) 除以 \(100\) 的余数是\fillinblank{}.
\end{problem}

\begin{answer}
81
\end{answer}

\begin{solution}
写成 \(91=1-10\). 由二项式定理，
\[
91^{92}=(1-10)^{92}\equiv1-92\times10\pmod {100}
\]
因为从二次项开始每一项都含有因子 \(10^2=100\)，所以模 \(100\) 时均为零. 于是
\[
91^{92}\equiv1-920=-919\equiv81\pmod {100}
\]
由于余数取 \(0\) 到 \(99\) 之间的整数，余数为 \(81\).
\end{solution}

\begin{problem}
已知三棱锥 \(A-BCD\) 的体积是 \(V\)，棱 \(BC\) 的长是 \(a\)，面 \(ABC\) 和面 \(DBC\) 的面积分别是 \(S_1\) 和 \(S_2\). 设面 \(ABC\) 和面 \(DBC\) 所成的二面角是 \(\alpha\)，那么 \(\sin \alpha=\)\fillinblank{}.
\end{problem}

\begin{answer}
\(\frac{3aV}{2S_1S_2}\)
\end{answer}

\begin{solution}
设在平面 \(ABC\) 内从 \(A\) 向 \(BC\) 作垂线，垂足为 \(E\)，记 \(AE=h_1\)；在平面 \(DBC\) 内从 \(D\) 向 \(BC\) 作垂线，垂足为 \(F\)，记 \(DF=h_2\). 由三角形面积公式，
\(h_1=\frac{2S_1}{a}\)，\(h_2=\frac{2S_2}{a}\).
过 \(E\) 作一个垂直于 \(BC\) 的截面，记为 \(\gamma\). 平面 \(\gamma\) 与平面 \(ABC\) 的交线是 \(AE\)，与平面 \(DBC\) 的交线是过 \(E\) 的直线 \(g\)，两交线的夹角就是二面角 \(\alpha\). 设 \(H\) 为点 \(A\) 到平面 \(DBC\) 的垂足，则点 \(A\) 到平面 \(DBC\) 的距离等于点 \(A\) 到直线 \(g\) 的距离，为
\[
AH=AE\sin\alpha=h_1\sin\alpha
\]
以平面 \(DBC\) 为底面，三棱锥的体积为
\[
V=\frac13S_2\cdot AH=\frac13\cdot\frac12ah_2\cdot h_1\sin\alpha
=\frac16ah_1h_2\sin\alpha
\]
代入 \(h_1=\frac{2S_1}{a}\)、\(h_2=\frac{2S_2}{a}\)，得
\[
V=\frac{2S_1S_2}{3a}\sin\alpha
\]
从而
\[
\sin\alpha=\frac{3aV}{2S_1S_2}
\]
\end{solution}

\section{解答题}

\begin{problem}
已知关于 \(x\) 的方程 \(2a^{2x-2}-7a^{x-1}+3=0\) 有一个根是 \(2\)，求 \(a\) 的值和方程其余的根.
\end{problem}

\begin{solution}
指数方程的底数须满足 \(a>0\) 且 \(a\ne1\). 令
\[
t=a^{x-1}>0
\]
则 \(a^{2x-2}=t^2\)，原方程化为
\[
2t^2-7t+3=0
\]
即
\[
(2t-1)(t-3)=0
\]
所以 \(t=\frac12\) 或 \(t=3\). 已知 \(x=2\) 是一个根，此时 \(t=a^{2-1}=a\)，因此
\[
a=\frac12\quad\text{或}\quad a=3
\]

当 \(a=\frac12\) 时，\(t=\frac12\) 已给出根 \(x=2\)，另一个根对应 \(t=3\)，满足
\[
\left(\frac12\right)^{x-1}=3
\]
所以 \(x-1=\log_{\frac12}3\)，即
\[
x=1+\log_{\frac12}3
\]

当 \(a=3\) 时，\(t=3\) 已给出根 \(x=2\)，另一个根对应 \(t=\frac12\)，满足
\[
3^{x-1}=\frac12
\]
所以 \(x-1=\log_3\frac12=-\log_3 2\)，即
\[
x=1-\log_3 2
\]
\end{solution}

\begin{problem}
已知平面 \(\alpha\) 和不在这个平面内的直线 \(a\) 都垂直于平面 \(\beta\). 求证： \(a\parallel \alpha\).
\end{problem}

\begin{solution}
设 \(l=\alpha\cap\beta\). 在平面 \(\alpha\) 内取一点 \(P\)，过 \(P\) 作直线 \(m\subset\alpha\)，使 \(m\perp l\). 由于两平面 \(\alpha\)、\(\beta\) 垂直，平面 \(\alpha\) 内垂直于交线 \(l\) 的直线垂直于平面 \(\beta\)，所以 \(m\perp\beta\). 题设又给出 \(a\perp\beta\)，两条垂直于同一平面的直线互相平行，故
\[
a\parallel m
\]

由于 \(m\subset\alpha\)，上式说明直线 \(a\) 的方向平行于平面 \(\alpha\). 若 \(a\) 与 \(\alpha\) 相交于 \(Q\)，则在平面 \(\alpha\) 内过 \(Q\) 且平行于 \(m\) 的直线唯一；而 \(a\) 也过 \(Q\) 且平行于 \(m\)，故 \(a\) 必在平面 \(\alpha\) 内，这与题设“直线 \(a\) 不在平面 \(\alpha\) 内”矛盾. 因此 \(a\) 不与 \(\alpha\) 相交，且方向平行于 \(\alpha\)，从而
\[
a\parallel\alpha
\]
\end{solution}

\begin{problem}
证明不等式： \(1+\frac{1}{\sqrt{2}}+\frac{1}{\sqrt{3}}+\cdots+\frac{1}{\sqrt{n}}<2\sqrt{n}\)（\(n\in \N^*\)）.
\end{problem}

\begin{solution}
对任意正整数 \(k\)，有
\[
2\left(\sqrt{k}-\sqrt{k-1}\right)=\frac{2}{\sqrt{k}+\sqrt{k-1}}>\frac{1}{\sqrt{k}}
\]
其中最后一个严格不等式等价于
\[
2\sqrt{k}>\sqrt{k}+\sqrt{k-1}
\Longleftrightarrow \sqrt{k}>\sqrt{k-1}
\]
对正整数 \(k\) 成立. 于是对 \(k=1\)，\(2\)，\(\ldots\)，\(n\) 逐项相加，得到
\[
1+\frac{1}{\sqrt{2}}+\cdots+\frac{1}{\sqrt{n}}
<2\sum_{k=1}^{n}\left(\sqrt{k}-\sqrt{k-1}\right)=2\sqrt{n}
\]
这里的和是望远镜和，右端的中间项全部消去，且 \(\sqrt0=0\). 所以不等式得证.
\end{solution}

\begin{problem}
设抛物线经过两点 \((-1,6)\) 和 \((-1,-2)\)，对称轴与 \(x\) 轴平行，开口向右；直线 \(y=2x+7\) 被抛物线截得的线段的长是 \(4\sqrt{10}\)，求抛物线的方程.
\end{problem}

\begin{solution}
设抛物线方程为
\((y-k)^2=4p(x-h)\)，\(p>0\).
它的对称轴为 \(y=k\). 因为两个已知点的横坐标相同，且纵坐标为 \(6\)、\(-2\)，它们关于对称轴对称，所以
\[
k=\frac{6+(-2)}2=2
\]
将点 \((-1,6)\) 代入，得
\(16=4p(-1-h)\)，\(p(-1-h)=4\).
因此
\[
h=-1-\frac4p
\]

直线 \(y=2x+7\) 与抛物线的交点横坐标满足
\[
(2x+7-2)^2=4p(x-h)
\]
，即
\[
4x^2+(20-4p)x+25+4ph=0
\]
设两个交点的横坐标为 \(x_1\)，\(x_2\). 由于交点在斜率为 \(2\) 的直线上，横坐标之差对应的线段长度为
\[
\sqrt{1+2^2}|x_1-x_2|=\sqrt{5}|x_1-x_2|=4\sqrt{10}
\]
所以
\[
|x_1-x_2|=4\sqrt2
\]
对二次方程 \(Ax^2+Bx+C=0\)，两根之差的平方为 \(\frac{B^2-4AC}{A^2}\). 在本题中
\[
|x_1-x_2|^2
=\frac{(20-4p)^2-16(25+4ph)}{16}
=p(p-10-4h)
\]
与 \(|x_1-x_2|^2=32\) 联立，得
\[
p(p-10-4h)=32
\]
代入 \(h=-1-\frac4p\)，得
\(p^2-6p+16=32\)，\(p^2-6p-16=0\).
解得 \(p=8\) 或 \(p=-2\)，由 \(p>0\) 取 \(p=8\)，进而 \(h=-1-\frac48=-\frac32\). 因此抛物线方程为
\[
(y-2)^2=32\left(x+\frac32\right)
\]
即
\[
(y-2)^2=32x+48
\]
\end{solution}

\begin{problem}
求同时满足下列两个条件的所有复数 \(z\)：
\begin{circlelist}
\item \(z+\frac{10}{z}\) 是实数，且 \(1<z+\frac{10}{z}\leqslant 6\)；
\item \(z\) 的实部和虚部都是整数.
\end{circlelist}
\end{problem}

\begin{solution}
设
\(z=x+y\i\)，\(x\)，\(y\in\Z\)，\(z\ne0\).
则
\[
z+\frac{10}{z}=x+y\i+\frac{10(x-y\i)}{x^2+y^2}
\]
其虚部为
\[
y\left(1-\frac{10}{x^2+y^2}\right)
\]
该式为零，所以分两种情况讨论.

当 \(y=0\) 时，\(x\ne0\)，且
\[
1<x+\frac{10}{x}\leqslant6
\]
若 \(x<0\)，则 \(x+\frac{10}{x}<0\)，不可能满足左侧不等式. 若 \(x>0\)，由基本不等式
\[
x+\frac{10}{x}\geqslant2\sqrt{10}>6
\]
也不满足右侧不等式. 因此 \(y=0\) 时无解.

另一种情况是 \(x^2+y^2=10\). 由于 \(x\)，\(y\) 为整数，平方和为 \(10=1+9\)，所以
\[
(x,y)=(\pm1,\pm3)\quad\text{或}\quad(\pm3,\pm1)
\]
在这种情况下，
\[
z+\frac{10}{z}=x+y\i+\frac{10(x-y\i)}{10}=2x
\]
不等式变为 \(1<2x\leqslant6\)，即 \(\frac12<x\leqslant3\). 结合 \(x\in\{\pm1,\pm3\}\)，只能取 \(x=1\) 或 \(x=3\). 分别配上对应的 \(y=\pm3\) 或 \(y=\pm1\)，所以所有复数为
\(z=1+3\i\)，\(1-3\i\)，\(3+\i\)，\(3-\i\).
\end{solution}
