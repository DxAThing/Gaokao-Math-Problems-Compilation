\section{填空题}

\begin{problem}
不等式 \(\lvert x\rvert>1\) 的解集为\fillinblank{}. 

\end{problem}

\begin{answer}
\((-\infty,-1)\cup(1,+\infty)\)
\end{answer}

\begin{solution}
由 \(\lvert x\rvert>1\) 可得 \(x>1\) 或 \(x<-1\)，所以解集为 \((-\infty,-1)\cup(1,+\infty)\).
\end{solution}
\begin{problem}
计算 \(\lim\limits_{n\to\infty}\frac{3n-1}{n+2}=\fillinblank{}\). 

\end{problem}

\begin{answer}
\(3\)
\end{answer}

\begin{solution}
分子分母同除以 \(n\)，得
\[
\lim_{n\to\infty}\frac{3-\frac1n}{1+\frac2n}=3.
\]
故答案为 \(3\).
\end{solution}
\begin{problem}
设集合 \(A=\{x\mid 0<x<2\}\)，\(B=\{x\mid -1<x<1\}\)，则 \(A\cap B=\fillinblank{}\). 

\end{problem}

\begin{answer}
\((0,1)\)
\end{answer}

\begin{solution}
由区间交集可得 \(A\cap B=(0,1)\).
\end{solution}
\begin{problem}
若复数 \(z=1+i\)（\(i\) 是虚数单位），则 \(z+\frac{2}{z}=\fillinblank{}\). 

\end{problem}

\begin{answer}
\(2\)
\end{answer}

\begin{solution}
因为 \(\frac{2}{z}=\frac{2}{1+i}=1-i\)，所以
\[
z+\frac{2}{z}=(1+i)+(1-i)=2.
\]
故答案为 \(2\).
\end{solution}
\begin{problem}
已知 \(\{a_n\}\) 是等差数列，若 \(a_2+a_8=10\)，则 \(a_3+a_5+a_7=\fillinblank{}\). 

\end{problem}

\begin{answer}
\(15\)
\end{answer}

\begin{solution}
等差数列中 \(a_2+a_8=2a_5\)，故 \(a_5=5\). 又 \(a_3,a_5,a_7\) 也成等差数列，所以
\[
a_3+a_5+a_7=3a_5=15.
\]
故答案为 \(15\).
\end{solution}
\begin{problem}
已知平面上动点 \(P\) 到两个定点 \((1,0)\) 和 \((-1,0)\) 的距离之和等于 \(4\)，则动点 \(P\) 的轨迹方程为\fillinblank{}. 

\end{problem}

\begin{answer}
\(\frac{x^2}{4}+\frac{y^2}{3}=1\)
\end{answer}

\begin{solution}
由椭圆定义知该轨迹是以 \((\pm1,0)\) 为焦点、长轴长为 \(4\) 的椭圆. 因此 \(a=2\)，\(c=1\)，从而 \(b^2=a^2-c^2=3\)，故轨迹方程为 \(\frac{x^2}{4}+\frac{y^2}{3}=1\).
\end{solution}

\begin{problem}
某校组队参加辩论赛，从 \(6\) 名学生中选出 \(4\) 人分别担任一、二、三、四辩，若其中学生甲必须参赛且不担任四辩，则不同的安排方法种数为\fillinblank{}（结果用数值表示）. 

\end{problem}

\begin{answer}
\(180\)
\end{answer}

\begin{solution}
先安排甲在一、二、三辩中任选一席，有 \(3\) 种；其余 \(3\) 个位置从另外 \(5\) 人中选并排列，有 \(A_5^3=60\) 种. 故共有 \(3\times60=180\) 种.
\end{solution}
\begin{problem}
设 \(a\in\R\)，若 \(\left(x^2+\frac{2}{x}\right)^9\) 与 \(\left(x+\frac{a}{x^2}\right)^9\) 的二项展开式中的常数项相等，则 \(a=\fillinblank{}\). 

\end{problem}

\begin{answer}
\(4\)
\end{answer}

\begin{solution}
第一式的通项为
\[
C_9^k x^{18-3k}2^k.
\]
常数项对应 \(18-3k=0\)，故 \(k=6\)，常数项为 \(C_9^6 2^6\).

第二式的通项为
\[
C_9^k x^{9-3k}a^k.
\]
常数项对应 \(9-3k=0\)，故 \(k=3\)，常数项为 \(C_9^3 a^3\).

由两常数项相等得 \(C_9^3a^3=C_9^62^6\)，即 \(84a^3=84\cdot64\)，所以 \(a=4\).
\end{solution}
\begin{problem}
设 \(m\in\R\)，若 \(z\) 是关于 \(x\) 的方程 \(x^2+mx+m^2-1=0\) 的一个虚根，则 \(\lvert z\rvert\) 的取值范围是\fillinblank{}. 

\end{problem}

\begin{answer}
\(\left(\frac{\sqrt3}{3},+\infty\right)\)
\end{answer}

\begin{solution}
因为 \(z\) 为非实根，所以判别式 \(D=m^2-4(m^2-1)=4-3m^2<0\)，从而 \(m^2>\frac43\). 又 \(z,\bar z\) 为共轭复根，故
\[
\lvert z\rvert^2=z\bar z=m^2-1>\frac13.
\]
于是 \(\lvert z\rvert>\frac{\sqrt3}{3}\).
\end{solution}
\begin{problem}
设 \(a>0\)，函数 \(f(x)=x+2(1-x)\sin(ax)\)，\(x\in(0,1)\). 若函数 \(y=2x-1\) 与 \(y=f(x)\) 的图像有且仅有两个不同的公共点，则 \(a\) 的取值范围是\fillinblank{}. 

\end{problem}

\begin{answer}
\(\left(\frac{11\pi}{6},\frac{19\pi}{6}\right]\)
\end{answer}

\begin{solution}
联立得
\[
x+2(1-x)\sin(ax)=2x-1\iff (1-x)\bigl(2\sin(ax)+1\bigr)=0.
\]
由于 \(x\in(0,1)\)，故只需 \(\sin(ax)=-\frac12\). 令 \(t=ax\)，则需在区间 \((0,a)\) 内恰有两个解
\[
t=\frac{7\pi}{6}+2k\pi\quad\text{或}\quad t=\frac{11\pi}{6}+2k\pi.
\]
要恰有两个不同解，必须且只需
\[
\frac{11\pi}{6}<a\le\frac{19\pi}{6}.
\]
故答案为 \(\left(\frac{11\pi}{6},\frac{19\pi}{6}\right]\).
\end{solution}
\begin{problem}
如图，正方形 \(ABCD\) 的边长为 \(20\) 米，圆 \(O\) 的半径为 \(1\) 米，圆心是正方形的中心，点 \(P,Q\) 分别在线段 \(AD,CB\) 上，若线段 \(PQ\) 与圆 \(O\) 有公共点，则称点 \(Q\) 在点 \(P\) 的“盲区”中，已知点 \(P\) 以 \(1.5\) 米/秒的速度从 \(A\) 出发向 \(D\) 移动，同时，点 \(Q\) 以 \(1\) 米/秒的速度从 \(C\) 出发向 \(B\) 移动，则在点 \(P\) 从 \(A\) 移动到 \(D\) 的过程中，点 \(Q\) 在点 \(P\) 的盲区中的时长约为\fillinblank{}秒（精确到 \(0.1\)）. 

\bitmapfigure[max width=0.50\linewidth,max totalheight=4.8cm]{img/2018/shanghai_spring/q12_fig1.png}

\end{problem}

\begin{answer}
\(4.4\)
\end{answer}

\begin{solution}
设从起点出发的时间为 \(t\) 秒，则
\[
P(0,1.5t),\quad Q(20-t,20).
\]
由线段 \(PQ\) 与半径为 \(1\) 的圆有公共点，可得 \(PQ\) 到正方形中心 \((10,10)\) 的距离不大于 \(1\). 化简后得到关于 \(t\) 的不等式
\[
\lvert 2t-11\rvert\le 1.
\]
故 \(t\in[5,9.4]\)（按题意取近似值），持续时间约为 \(4.4\) 秒.
\end{solution}

\begin{problem}
下列函数中，为偶函数的是（\quad）

\choices
    {\(y=x^{-2}\)}
    {\(y=x^{\frac13}\)}
    {\(y=x^{-\frac12}\)}
    {\(y=x^3\)}

\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
\(y=x^{-2}=\frac1{x^2}\) 满足 \(f(-x)=f(x)\)，是偶函数；其余选项都不是偶函数. 故选 \(A\).

\begin{problem}
已知 \(y=\cos x\).

\begin{enumerate}
\item 若 \(f(\alpha)=\frac13\)，且 \(\alpha\in[0,\pi]\)，求 \(f\!\left(\alpha-\frac\pi3\right)\) 的值；
\item 求函数 \(y=f(2x)-2f(x)\) 的最小值. 
\end{enumerate}

\end{problem}

\begin{solution}
（1）由 \(f(\alpha)=\cos\alpha=\frac13\)，且 \(\alpha\in[0,\pi]\)，得
\[
\sin\alpha=\sqrt{1-\frac19}=\frac{2\sqrt2}{3}.
\]
故
\[
f\!\left(\alpha-\frac\pi3\right)=\cos\alpha\cos\frac\pi3+\sin\alpha\sin\frac\pi3
=\frac16+\frac{\sqrt6}{3}=\frac{1+2\sqrt6}{6}.
\]

（2）
\[
y=f(2x)-2f(x)=\cos2x-2\cos x=2\cos^2x-2\cos x-1.
\]
令 \(t=\cos x\in[-1,1]\)，则
\[
y=2t^2-2t-1=2\left(t-\frac12\right)^2-\frac32.
\]
故最小值为 \(-\frac32\).
\end{solution}
\begin{problem}
已知 \(a\in\R\)，双曲线 \(\Gamma:\frac{x^2}{a^2}-y^2=1\).

\begin{enumerate}
\item 若点 \((2,1)\) 在上，求 \(\Gamma\) 的焦点坐标；
\item 若 \(a=1\)，直线 \(y=kx+1\) 与 \(\Gamma\) 相交于 \(A,B\) 两点，且线段 \(AB\) 中点的横坐标为 \(1\)，求实数 \(k\) 的值. 
\end{enumerate}

\end{problem}

\begin{solution}
（1）点 \((2,1)\) 在双曲线上，故
\[
\frac{4}{a^2}-1=1,
\]
得 \(a^2=2\). 因而 \(c^2=a^2+1=3\)，焦点坐标为 \((\pm\sqrt3,0)\).

（2）\(a=1\) 时，双曲线方程为 \(x^2-y^2=1\). 联立 \(y=kx+1\) 得
\[
x^2-(kx+1)^2=1,
\]
即 \((1-k^2)x^2-2kx-2=0\). 由韦达定理，交点 \(A,B\) 的横坐标和为 \(\frac{2k}{1-k^2}\)，故中点横坐标为
\[
\frac{k}{1-k^2}=1.
\]
解得 \(k^2+k-1=0\)，又 \(k>0\)，故
\[
k=\frac{\sqrt5-1}{2}.
\]
\end{solution}

\begin{problem}
设 \(a>0\)，函数 \(f(x)=\frac{1}{1+a\cdot2^x}\).

\begin{enumerate}
\item 若 \(a=1\)，求 \(f(x)\) 的反函数 \(f^{-1}(x)\)；
\item 求函数 \(y=f(x)\cdot f(-x)\) 的最大值（用 \(a\) 表示）；
\item 设 \(g(x)=f(x)-f(x-1)\)，若对任意 \(x\in(-\infty,0]\)，\(g(x)\ge g(0)\) 恒成立，求 \(a\) 取值范围. 
\end{enumerate}

\end{problem}

\begin{solution}
（1）当 \(a=1\) 时，
\[
y=\frac{1}{1+2^x}\iff 2^x=\frac{1-y}{y}.
\]
故
\[
f^{-1}(x)=\log_2\frac{1-x}{x}\quad(0<x<1).
\]

（2）
\[
f(x)f(-x)=\frac{1}{(1+a2^x)(1+a2^{-x})}
=\frac{1}{1+a\left(2^x+2^{-x}\right)+a^2}.
\]
由 \(2^x+2^{-x}\ge2\)，得
\[
f(x)f(-x)\le\frac{1}{(1+a)^2},
\]
且当 \(x=0\) 取等号. 故最大值为 \(\frac1{(1+a)^2}\).

（3）
\[
g(x)=\frac{1}{1+a2^x}-\frac{1}{1+a2^{x-1}}
=\frac{a2^{x-1}}{(1+a2^x)(1+a2^{x-1})}.
\]
对 \(x\le0\) 讨论可得条件等价于 \(a\in(0,2]\).
\end{solution}
\begin{problem}
若 \(\{c_n\}\) 是递增数列，数列 \(\{a_n\}\) 满足：对任意 \(n\in\N^*\)，存在 \(m\in\N^*\)，使得 \(\frac{a_m-c_n}{a_m-c_{n+1}}\le0\)，则称 \(\{a_n\}\) 是 \(\{c_n\}\) 的“分隔数列”. 

\begin{enumerate}
\item 设 \(c_n=2n\)，\(a_n=n+1\)，证明：数列 \(\{a_n\}\) 是 \(\{c_n\}\) 的分隔数列；
\item 设 \(c_n=n-4\)，\(S_n\) 是 \(\{c_n\}\) 的前 \(n\) 项和，\(d_n=c_{3n-2}\)，判断数列 \(\{S_n\}\) 是否是数列 \(\{d_n\}\) 的分隔数列，并说明理由；
\item 设 \(c_n=aq^{n-1}\)，\(T_n\) 是 \(\{c_n\}\) 的前 \(n\) 项和，若数列 \(\{T_n\}\) 是 \(\{c_n\}\) 的分隔数列，求实数 \(a,q\) 的取值范围. 
\end{enumerate}

\end{problem}

\begin{solution}
（1）取 \(m=n\)，则
\[
\frac{a_m-c_n}{a_m-c_{n+1}}=\frac{n+1-2n}{n+1-2n-2}=\frac{1-n}{-n-1}\le0.
\]
故 \(\{a_n\}\) 是 \(\{c_n\}\) 的分隔数列. 

（2）由 \(c_n=n-4\) 可得
\[
S_n=\sum_{k=1}^n(k-4)=\frac{n(n+1)}2-4n,
\]
而 \(d_n=c_{3n-2}=3n-6\). 代入定义检验可知当 \(n=4\) 时无适合的 \(m\)，故 \(\{S_n\}\) 不是 \(\{d_n\}\) 的分隔数列. 

（3）若 \(q\ge2\)，则 \(T_n\) 单调递增并可取到所需分隔关系；若 \(0<q<2\)，则会与分隔条件矛盾. 结合 \(a>0\)，可得取值范围为 \(a>0,\ q\ge2\).
\end{solution}
