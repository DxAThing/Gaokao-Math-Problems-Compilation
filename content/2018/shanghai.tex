\section{填空题}

\begin{problem}
行列式 \(\begin{vmatrix}4 & 1\\2 & 5\end{vmatrix}\) 的值为\fillinblank{}.

\end{problem}

\begin{answer}
\(18\)
\end{answer}

\begin{solution}
直接计算得 \(\begin{vmatrix}4 & 1\\2 & 5\end{vmatrix}=4\times5-2\times1=18\). 故答案为 \(18\).
\end{solution}

\begin{problem}
双曲线 \(\frac{x^2}{4}-y^2=1\) 的渐近线方程为\fillinblank{}.

\end{problem}

\begin{answer}
\(y=\pm\frac12x\)
\end{answer}

\begin{solution}
由双曲线 \(\frac{x^2}{4}-y^2=1\) 可知 \(a=2\)，\(b=1\). 其渐近线方程为 \(y=\pm\frac{b}{a}x=\pm\frac12x\). 故答案为 \(y=\pm\frac12x\).
\end{solution}

\begin{problem}
在 \((1+x)^7\) 的二项展开式中，\(x^2\) 项的系数为\fillinblank{}. （结果用数值表示）

\end{problem}

\begin{answer}
\(21\)
\end{answer}

\begin{solution}
二项式 \((1+x)^7\) 的通项为 \(T_{r+1}=C_7^r x^r\). 令 \(r=2\)，得 \(x^2\) 项的系数为 \(C_7^2=21\). 故答案为 \(21\).
\end{solution}

\begin{problem}
设常数 \(a\in\R\)，函数 \(f(x)=\log_2(x+a)\). 若 \(f(x)\) 的反函数的图像经过点 \((3,1)\)，则 \(a=\)\fillinblank{}.

\end{problem}

\begin{answer}
\(7\)
\end{answer}

\begin{solution}
因为反函数的图像经过点 \((3,1)\)，所以原函数的图像经过点 \((1,3)\). 因此 \(\log_2(1+a)=3\)，解得 \(a=7\). 故答案为 \(7\).
\end{solution}

\begin{problem}
已知复数 \(z\) 满足 \((1+\i)z=1-7\i\)（\(\i\) 是虚数单位），则 \(\lvert z\rvert=\)\fillinblank{}.

\end{problem}

\begin{answer}
\(5\)
\end{answer}

\begin{solution}
由 \((1+\i)z=1-7\i\)，得 \(z=\frac{1-7\i}{1+\i}=\frac{(1-7\i)(1-\i)}{(1+\i)(1-\i)}=-3-4\i\). 所以 \(\lvert z\rvert=\sqrt{(-3)^2+(-4)^2}=5\). 故答案为 \(5\).
\end{solution}

\begin{problem}
记等差数列 \(\{a_n\}\) 的前 \(n\) 项和为 \(S_n\)，若 \(a_3=0\)，\(a_6+a_7=14\)，则 \(S_7=\)\fillinblank{}.

\end{problem}

\begin{answer}
\(14\)
\end{answer}

\begin{solution}
设公差为 \(d\)，首项为 \(a_1\). 由 \(a_3=0\) 得 \(a_1+2d=0\). 又由 \(a_6+a_7=14\) 得 \((a_1+5d)+(a_1+6d)=14\). 解得 \(a_1=-4\)，\(d=2\). 因此 \(S_7=\frac{7}{2}\bigl(2a_1+6d\bigr)=7a_1+21d=-28+42=14\). 故答案为 \(14\).
\end{solution}

\begin{problem}
已知 \(\alpha\in\{-2,-1,-\frac12,1,2,3\}\). 若幂函数 \(f(x)=x^\alpha\) 为奇函数，且在 \((0,+\infty)\) 上递减，则 \(\alpha=\)\fillinblank{}.

\end{problem}

\begin{answer}
\(-1\)
\end{answer}

\begin{solution}
幂函数 \(f(x)=x^\alpha\) 为奇函数，故 \(\alpha\) 必为奇数；又因为它在 \((0,+\infty)\) 上递减，所以 \(\alpha<0\). 在给定取值中只有 \(\alpha=-1\) 符合条件. 故答案为 \(-1\).
\end{solution}

\begin{problem}
在平面直角坐标系中，已知点 \(A(-1,0)\)，\(B(2,0)\)，\(E\)，\(F\) 是 \(y\) 轴上的两个动点，且 \(\lvert\overrightarrow{EF}\rvert=2\)，则 \(\overrightarrow{AE}\cdot\overrightarrow{BF}\) 的最小值为\fillinblank{}.

\end{problem}

\begin{answer}
\(-3\)
\end{answer}

\begin{solution}
设 \(E(0,a)\)，\(F(0,b)\)，则 \(\lvert a-b\rvert=2\)，故可设 \(a=b+2\) 或 \(b=a+2\).

当 \(a=b+2\) 时，\(\overrightarrow{AE}=(1,a)\)，\(\overrightarrow{BF}=(-2,b)\)，所以 \(\overrightarrow{AE}\cdot\overrightarrow{BF}=-2+ab\). 而 \(ab=b(b+2)=b^2+2b=(b+1)^2-1\ge -1\)，故最小值为 \(-3\).

当 \(b=a+2\) 时，\(\overrightarrow{AE}=(1,a)\)，\(\overrightarrow{BF}=(-2,b)\)，所以 \(\overrightarrow{AE}\cdot\overrightarrow{BF}=-2+ab\). 而 \(ab=a(a+2)=a^2+2a=(a+1)^2-1\ge -1\)，故最小值为 \(-3\).

故答案为 \(-3\).
\end{solution}

\begin{problem}
有编号互不相同的五个砝码，其中 \(5\) 克，\(3\) 克，\(1\) 克砝码各一个，\(2\) 克砝码两个. 从中随机选取三个，则这三个砝码的总质量为 \(9\) 克的概率是\fillinblank{}. （结果用最简分数表示）

\end{problem}

\begin{answer}
\(\frac15\)
\end{answer}

\begin{solution}
五个砝码中任选三个，共有 \(C_5^3=10\) 种选法. 总质量为 \(9\) 克的情形只有两种：选 \(5\)，\(3\)，\(1\) 或选 \(5\)，\(2\)，\(2\). 因此所求概率为 \(\frac{2}{10}=\frac15\). 故答案为 \(\frac15\).
\end{solution}

\begin{problem}
设等比数列 \(\{a_n\}\) 的通项公式为 \(a_n=q^{n-1}\)（\(n\in\N^*\)），前 \(n\) 项和为 \(S_n\). 若 \(\lim\limits_{n\to\infty}\frac{S_n}{a_{n+1}}=\frac12\)，则 \(q=\)\fillinblank{}.

\end{problem}

\begin{answer}
\(3\)
\end{answer}

\begin{solution}
由 \(a_n=q^{n-1}\) 可知 \(a_1=1\). 若 \(q\ne1\)，则 \(S_n=\frac{q^n-1}{q-1}\)，所以 \(\frac{S_n}{a_{n+1}}=\frac{q^n-1}{(q-1)q^n}\). 取极限得 \(\lim\limits_{n\to\infty}\frac{S_n}{a_{n+1}}=\frac1{q-1}=\frac12\)，解得 \(q=3\). 故答案为 \(3\).
\end{solution}

\begin{problem}
已知常数 \(a>0\)，函数 \(f(x)=\frac{2^x}{2^x+ax}\) 的图像经过点 \(P\left(p,\frac65\right)\)，\(Q\left(q,-\frac15\right)\). 若 \(2^{p+q}=36pq\)，则 \(a=\)\fillinblank{}.

\end{problem}

\begin{answer}
\(6\)
\end{answer}

\begin{solution}
由 \(P\left(p,\frac65\right)\) 在函数图像上，得 \(\frac{2^p}{2^p+ap}=\frac65\)，即 \(2^p=-6ap\). 由 \(Q\left(q,-\frac15\right)\) 在函数图像上，得 \(\frac{2^q}{2^q+aq}=-\frac15\)，即 \(2^q=-\frac{aq}{6}\). 两式相乘，得 \(2^{p+q}=a^2pq\). 又已知 \(2^{p+q}=36pq\)，故 \(a^2=36\). 因为 \(a>0\)，所以 \(a=6\). 故答案为 \(6\).
\end{solution}

\begin{problem}
已知实数 \(x_1\)，\(x_2\)，\(y_1\)，\(y_2\) 满足：\(x_1^2+y_1^2=1\)，\(x_2^2+y_2^2=1\)，\(x_1x_2+y_1y_2=\frac12\)，则 \(\frac{\lvert x_1+y_1-1\rvert}{\sqrt2}+\frac{\lvert x_2+y_2-1\rvert}{\sqrt2}\) 的最大值为\fillinblank{}.

\end{problem}

\begin{answer}
\(\sqrt2+\sqrt3\)
\end{answer}

\begin{solution}
设 \(A(x_1,y_1)\)，\(B(x_2,y_2)\). 由条件知 \(A\)，\(B\) 在单位圆上，且 \(\cos\angle AOB=\frac12\)，所以 \(\angle AOB=60^\circ\)，故弦 \(AB\) 长为 \(1\).

表达式 \(\frac{\lvert x_1+y_1-1\rvert}{\sqrt2}+\frac{\lvert x_2+y_2-1\rvert}{\sqrt2}\) 就是点 \(A\)，\(B\) 到直线 \(x+y-1=0\) 的距离之和. 要使这两个距离之和最大，弦 \(AB\) 应与直线 \(x+y-1=0\) 平行.

由弦长为 \(1\) 可知，圆心 \(O\) 到弦 \(AB\) 的距离为 \(\sqrt{1-\left(\frac12\right)^2}=\frac{\sqrt3}{2}\). 又 A，B 在第三象限，所以直线 \(AB\) 可写成 \(x+y=-\frac{\sqrt6}{2}\). 于是点 A，B 到直线 \(x+y-1=0\) 的距离都为 \(\frac{1+\frac{\sqrt6}{2}}{\sqrt2}\). 所以所求最大值为 \(2\cdot\frac{1+\frac{\sqrt6}{2}}{\sqrt2}=\sqrt2+\sqrt3\). 故答案为 \(\sqrt2+\sqrt3\).
\end{solution}

\begin{problem}
设 \(P\) 是椭圆 \(\frac{x^2}{5}+\frac{y^2}{3}=1\) 上的动点，则 \(P\) 到该椭圆的两个焦点的距离之和为（\quad）

\choices
    {\(2\sqrt2\)}
    {\(2\sqrt3\)}
    {\(2\sqrt5\)}
    {\(4\sqrt2\)}

\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
椭圆 \(\frac{x^2}{5}+\frac{y^2}{3}=1\) 中，\(a^2=5\)，故 \(a=\sqrt5\). 由椭圆定义，点 \(P\) 到两个焦点的距离之和为 \(2a=2\sqrt5\). 故选 C.
\end{solution}

\begin{problem}
已知 \(a\in\R\)，则“\(a>1\)”是“\(\frac1a<1\)”的（\quad）

\choices
    {充分非必要条件}
    {必要非充分条件}
    {充要条件}
    {既非充分又非必要条件}

\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
若 \(a>1\)，则 \(\frac1a<1\)，所以“\(a>1\)”是“\(\frac1a<1\)”的充分条件. 反过来，\(\frac1a<1\) 还可以由 \(a<0\) 得到，因此它不是必要条件. 故选 A.
\end{solution}

\begin{problem}
已知圆锥的顶点为 \(P\)，底面圆心为 \(O\)，半径为 \(2\).

\begin{enumerate}
\item 设圆锥的母线长为 \(4\)，求圆锥的体积；
\item 设 \(PO=4\)，\(OA\)，\(OB\) 是底面半径，且 \(\angle AOB=90^\circ\)，\(M\) 为线段 \(AB\) 的中点，如图，求异面直线 \(PM\) 与 \(OB\) 所成的角的大小.
\end{enumerate}

\bitmapfigure[max width=6.0cm]{img/2018/shanghai/q17_fig1.png}

\end{problem}

\begin{solution}
（1）设圆锥高为 \(h\). 由母线长为 \(4\)，底面半径为 \(2\)，得 \(h=\sqrt{4^2-2^2}=2\sqrt3\). 因此圆锥体积为 \(V=\frac13\pi\cdot2^2\cdot2\sqrt3=\frac{8\sqrt3}{3}\pi\).

（2）以 \(O\) 为原点，\(OA\) 为 \(x\) 轴，\(OB\) 为 \(y\) 轴，\(OP\) 为 \(z\) 轴，建立空间直角坐标系，则
\(P(0,0,4)\)，\(A(2,0,0)\)，\(B(0,2,0)\)，\(M(1,1,0)\). 于是 \(\overrightarrow{PM}=(1,1,-4)\)，\(\overrightarrow{OB}=(0,2,0)\).
设异面直线 \(PM\) 与 \(OB\) 所成的角为 \(\theta\)，则
\(\cos\theta=\frac{\lvert\overrightarrow{PM}\cdot\overrightarrow{OB}\rvert}{\lvert\overrightarrow{PM}\rvert\,\lvert\overrightarrow{OB}\rvert}=\frac{2}{\sqrt{18}\cdot2}=\frac{\sqrt2}{6}\). 所以 \(\theta=\arccos\frac{\sqrt2}{6}\).
\end{solution}

\begin{problem}
设常数 \(a\in\R\)，函数 \(f(x)=a\sin2x+2\cos^2x\).

\begin{enumerate}
\item 若 \(f(x)\) 为偶函数，求 \(a\) 的值；
\item 若 \(f\!\left(\frac\pi4\right)=\sqrt3+1\)，求方程 \(f(x)=1-\sqrt2\) 在区间 \([-\pi,\pi]\) 上的解.
\end{enumerate}

\end{problem}

\begin{solution}
（1）由 \(f(-x)=-a\sin2x+2\cos^2x\). 因为 \(f(x)\) 为偶函数，所以 \(f(-x)=f(x)\)，即 \(-a\sin2x+2\cos^2x=a\sin2x+2\cos^2x\).
因此 \(a=0\).

（2）由 \(f\!\left(\frac\pi4\right)=\sqrt3+1\)，得 \(a+1=\sqrt3+1\)，所以 \(a=\sqrt3\).

于是 \(f(x)=\sqrt3\sin2x+2\cos^2x=\sqrt3\sin2x+\cos2x+1=2\sin\!\left(2x+\frac\pi6\right)+1\). 由 \(f(x)=1-\sqrt2\)，得 \(2\sin\!\left(2x+\frac\pi6\right)+1=1-\sqrt2\)，即 \(\sin\!\left(2x+\frac\pi6\right)=-\frac{\sqrt2}{2}\). 所以 \(2x+\frac\pi6=-\frac\pi4+2k\pi\)或\(2x+\frac\pi6=\frac{5\pi}{4}+2k\pi\)，\(k\in\Z\). 解得 \(x=-\frac{5\pi}{24}+k\pi\)或\(x=\frac{13\pi}{24}+k\pi\). 在区间 \([-\pi,\pi]\) 上，解为 \(x=-\frac{11\pi}{24}\)，\(-\frac{5\pi}{24}\)，\(\frac{13\pi}{24}\)，\(\frac{19\pi}{24}\).
\end{solution}

\begin{problem}
某群体的人均通勤时间，是指单日内该群体中成员从居住地到工作地的平均用时. 某地上班族 \(S\) 中的成员仅以自驾或公交方式通勤. 分析显示：当 \(S\) 中 \(x\%\)（\(0<x<100\)）的成员自驾时，自驾群体的人均通勤时间 \(f(x)\)（单位：分钟）为当 \(0<x\le30\) 时，\(f(x)=30\)；当 \(30<x<100\) 时，\(f(x)=2x+\frac{1800}{x}-90\). 而公交群体的人均通勤时间不受 \(x\) 影响，恒为 \(40\) 分钟. 试根据上述分析结果回答下列问题：

\begin{enumerate}
\item 当 \(x\) 在什么范围内时，公交群体的人均通勤时间少于自驾群体的人均通勤时间？
\item 求该地上班族 \(S\) 的人均通勤时间 \(g(x)\) 的表达式；讨论 \(g(x)\) 的单调性，并说明其实际意义.
\end{enumerate}

\end{problem}

\begin{solution}
（1）当 \(0<x\le30\) 时，\(f(x)=30<40\)，不合题意.

当公交群体的人均通勤时间少于自驾群体的人均通勤时间时，有 \(f(x)>40\). 所以只需在 \(30<x<100\) 时讨论，此时 \(2x+\frac{1800}{x}-90>40\). 整理得 \(x^2-65x+900>0\)，解得 \(x<20\)或\(x>45\). 结合 \(30<x<100\)，所以 \(x\in(45,100)\).

（2）当 \(0<x\le30\) 时，
\(g(x)=\frac{x}{100}\cdot30+\left(1-\frac{x}{100}\right)\cdot40=40-\frac{x}{10}\).
当 \(30<x<100\) 时，
\(g(x)=\frac{x}{100}\left(2x+\frac{1800}{x}-90\right)+\left(1-\frac{x}{100}\right)40=\frac{x^2}{50}-\frac{13}{10}x+58\).
\examdisplaycases{g(x)=}{
40-\frac{x}{10}, & 0<x\le30,\\
\frac{x^2}{50}-\frac{13}{10}x+58, & 30<x<100
}

当 \(0<x\le30\) 时，
\(g(x)=40-\frac{x}{10}\)，
所以 \(g(x)\) 单调递减.

对第二段求导得
\(g'(x)=\frac{x}{25}-\frac{13}{10}\).
所以当 \(30<x<32.5\) 时，\(g(x)\) 单调递减；当 \(32.5<x<100\) 时，\(g(x)\) 单调递增.

综上，\(g(x)\) 在 \((0,32.5)\) 上单调递减，在 \((32.5,100)\) 上单调递增.
这说明当自驾人数比例小于 \(32.5\%\) 时，人均通勤时间随自驾比例增大而下降；当自驾人数比例大于 \(32.5\%\) 时，人均通勤时间随自驾比例增大而上升；当自驾人数比例为 \(32.5\%\) 时，人均通勤时间最少.
\end{solution}

\begin{problem}
设常数 \(t>2\). 在平面直角坐标系 \(xOy\) 中，已知点 \(F(2,0)\)，直线 \(l:x=t\)，曲线 \(\Gamma:y^2=8x\)（\(0\le x\le t\)，\(y\ge0\)），\(l\) 与 \(x\) 轴交于点 \(A\)，与 \(\Gamma\) 交于点 \(B\). \(P\)，\(Q\) 分别是曲线 \(\Gamma\) 与线段 \(AB\) 上的动点.

\begin{enumerate}
\item 用 \(t\) 表示点 \(B\) 到点 \(F\) 的距离；
\item 设 \(t=3\)，\(\lvert FQ\rvert=2\)，线段 \(OQ\) 的中点在直线 \(FP\) 上，求 \(\triangle AQP\) 的面积；
\item 设 \(t=8\)，是否存在以 \(FP\)，\(FQ\) 为邻边的矩形 \(FPEQ\)，使得点 \(E\) 在 \(\Gamma\) 上？若存在，求点 \(P\) 的坐标；若不存在，说明理由.
\end{enumerate}

\end{problem}

\begin{solution}
\solutionfigure{\bitmapfigure[max width=0.58\linewidth]{img/2018/shanghai/q20_solution_fig1.png}}

（1）【法一】由 \(B\in\Gamma\) 且 \(x=t\)，可得
\(B\bigl(t,\sqrt{8t}\bigr)\). 所以 \(\lvert BF\rvert=\sqrt{(t-2)^2+8t}=t+2\).

【法二】由抛物线的性质可知，
\(\lvert BF\rvert=t+2\).

（2）当 \(t=3\) 时，\(A(3,0)\)，且 \(\lvert FQ\rvert=2\)，又 \(Q\in AB\)，故
 \(Q(3,\sqrt3)\).
设 \(OQ\) 的中点为 \(D\)，则
 \(D\left(\frac32,\frac{\sqrt3}{2}\right)\).
由 \(D\in FP\)，得直线 \(FP\) 的斜率为
 \(\frac{\sqrt3/2-0}{3/2-2}=-\sqrt3\).
故直线 \(FP\) 方程为
 \(y=-\sqrt3(x-2)\).
联立 \(y^2=8x\) 得
 \(3(x-2)^2=8x\)，即 \(3x^2-20x+12=0\).
由于 \(P\in\Gamma\) 且 \(0\le x\le3\)，取 \(x=\frac23\)，从而
 \(P\left(\frac23,\frac{4\sqrt3}{3}\right)\).
于是
 \(S_{\triangle AQP}=\frac12\cdot AQ\cdot\bigl(3-\tfrac23\bigr)=\frac12\cdot\sqrt3\cdot\frac73=\frac{7\sqrt3}{6}\).

（3）存在，设
 \(P\left(\frac{y^2}{8},y\right)\)，\(E\left(\frac{m^2}{8},m\right)\).
则
 \(k_{PF}=\frac{y}{\frac{y^2}{8}-2}=\frac{8y}{y^2-16}\)，\(k_{FQ}=\frac{16-y^2}{8y}\).
直线 \(QF\) 方程为
 \(y=\frac{16-y^2}{8y}(x-2)\).
所以
 \(Q\left(8,\frac{48-3y^2}{4y}\right)\).
根据 \(\overrightarrow{FP}+\overrightarrow{FQ}=\overrightarrow{FE}\)，得
 \(E\left(\frac{y^2}{8}+6,\frac{48+y^2}{4y}\right)\).
因为 \(E\in\Gamma\)，所以
 \(\left(\frac{48+y^2}{4y}\right)^2=8\left(\frac{y^2}{8}+6\right)\)，解得 \(y^2=\frac{16}{5}\).
故存在以 \(FP\)，\(FQ\) 为邻边的矩形 \(FPEQ\)，使得点 \(E\) 在 \(\Gamma\) 上，且
 \(P\left(\frac25,\frac{4\sqrt5}{5}\right)\).
\end{solution}

\begin{problem}
给定无穷数列 \(\{a_n\}\)，若无穷数列 \(\{b_n\}\) 满足：对任意 \(n\in\N^*\)，都有 \(\lvert b_n-a_n\rvert\le1\)，则称 \(\{b_n\}\) 与 \(\{a_n\}\)“接近”.

\begin{enumerate}
\item 设 \(\{a_n\}\) 是首项为 \(1\)，公比为 \(\frac12\) 的等比数列，\(b_n=a_{n+1}+1\)，\(n\in\N^*\). 判断数列 \(\{b_n\}\) 是否与 \(\{a_n\}\) 接近，并说明理由；
\item 设数列 \(\{a_n\}\) 的前四项为：\(a_1=1\)，\(a_2=2\)，\(a_3=4\)，\(a_4=8\)，\(\{b_n\}\) 是一个与 \(\{a_n\}\) 接近的数列，记集合 \(M=\{x\mid x=b_i,\ i=1,2,3,4\}\)，求 \(M\) 中元素的个数 \(m\)；
\item 已知 \(\{a_n\}\) 是公差为 \(d\) 的等差数列. 若存在数列 \(\{b_n\}\) 满足：\(\{b_n\}\) 与 \(\{a_n\}\) 接近，且在 \(b_2-b_1\)，\(b_3-b_2\)，\(\ldots\)，\(b_{201}-b_{200}\) 中至少有 \(100\) 个为正数，求 \(d\) 的取值范围.
\end{enumerate}

\end{problem}

\begin{solution}
（1）\(\{a_n\}\) 的通项为 \(a_n=2^{1-n}\). 又 \(b_n=a_{n+1}+1=2^{-n}+1\). 因此 \(\lvert b_n-a_n\rvert=\left\lvert 1-2^{-n}\right\rvert<1\)，
所以 \(\{b_n\}\) 与 \(\{a_n\}\) 接近.

（2）因为 \(\{b_n\}\) 与 \(\{a_n\}\) 接近，所以
\(a_n-1\le b_n\le a_n+1\).
由已知前四项可得
 \(b_1\in[0,2]\)，\(b_2\in[1,3]\)，\(b_3\in[3,5]\)，\(b_4\in[7,9]\).
因此 \(b_1\)，\(b_2\)，\(b_3\)，\(b_4\) 中，\(b_1\) 与 \(b_2\) 可能相等，\(b_2\) 与 \(b_3\) 可能相等，但 \(b_1\) 与 \(b_3\) 不可能相等，\(b_3\) 与 \(b_4\) 也不可能相等.

所以 \(M\) 中元素的个数 \(m=3\) 或 \(4\).

（3）设
 \(a_n=a_1+(n-1)d\).

若 \(d>0\)，取 \(b_n=a_n\)，则 \(b_{n+1}-b_n=d>0\)，因此满足题意.

若 \(d=0\)，取 \(b_n=a_1-\frac{1}{n}\)，则 \(\lvert b_n-a_n\rvert=\left\lvert a_1-\frac{1}{n}-a_1\right\rvert=\frac{1}{n}<1\)，
且
 \(b_{n+1}-b_n=\frac{1}{n}-\frac{1}{n+1}>0\)，
因此 \(b_2-b_1\)，\(b_3-b_2\)，\(\ldots\)，\(b_{201}-b_{200}\) 都为正数，符合题意.

若 \(-2<d<0\)，取
 \(b_{2n-1}=a_{2n-1}-1\)，\(b_{2n}=a_{2n}+1\)，
则
 \(b_{2n}-b_{2n-1}=d+2>0\)，\(b_{2n+1}-b_{2n}=d-2<0\)，
因此恰有 \(100\) 个正数，符合题意.

若 \(d\le-2\)，因为 \(\lvert b_n-a_n\rvert\le1\)，所以
 \(a_n-1\le b_n\le a_n+1\)，\(a_{n+1}-1\le b_{n+1}\le a_{n+1}+1\).
于是
 \(b_{n+1}-b_n\le (a_{n+1}+1)-(a_n-1)=d+2\le0\)，
故不可能出现正数.

综上，\(d\) 的取值范围为 \((-2,+\infty)\).
\end{solution}
