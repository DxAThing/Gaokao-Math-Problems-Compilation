\section{单选题}

\begin{problem}
\(\i(2+3\i)=\)（\quad）.

\choices
    {\(3-2\i\)}
    {\(3+2\i\)}
    {\(-3-2\i\)}
    {\(-3+2\i\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
\(\i(2+3\i)=2\i+3\i^2=2\i-3=-3+2\i\)，故选 D.
\end{solution}

\begin{problem}
已知集合 \(A=\{1,3,5,7\}\)，\(B=\{2,3,4,5\}\)，则 \(A\cap B=\)（\quad）.

\choices
    {\(\{3\}\)}
    {\(\{5\}\)}
    {\(\{3,5\}\)}
    {\(\{1,2,3,4,5,7\}\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
集合 \(A\) 与集合 \(B\) 的公共元素为 \(3\) 和 \(5\)，所以 \(A\cap B=\{3,5\}\)，故选 C.
\end{solution}

\begin{problem}
函数 \(f(x)=\dfrac{\e^x-\e^{-x}}{x^2}\) 的图象大致为（\quad）.

\choices
    {\choicebitmap[width=0.15\paperwidth]{img/2018/national_paper_2_liberal/q3_optA.png}}
    {\choicebitmap[width=0.15\paperwidth]{img/2018/national_paper_2_liberal/q3_optB.png}}
    {\choicebitmap[width=0.15\paperwidth]{img/2018/national_paper_2_liberal/q3_optC.png}}
    {\choicebitmap[width=0.15\paperwidth]{img/2018/national_paper_2_liberal/q3_optD.png}}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
函数的定义域为 \(x\ne0\)，且 \(f(-x)=-f(x)\)，所以其图象关于原点对称. 当 \(x\to0^+\) 时，\(\e^x-\e^{-x}\sim2x\)，从而 \(f(x)\sim2/x\to+\infty\)；当 \(x\to0^-\) 时，\(f(x)\to-\infty\). 又当 \(x\to+\infty\) 时，\(f(x)\to+\infty\)，当 \(x\to-\infty\) 时，\(f(x)\to-\infty\). 因此右支在第一象限，左支在第三象限，且两支关于原点对称，符合图象 \(B\)，故选 B.
\end{solution}

\begin{problem}
已知向量 \(\bs a\)，\(\bs b\) 满足 \(\lvert \bs a\rvert=1\)，\(\bs a\cdot\bs b=-1\)，则 \(\bs a\cdot(2\bs a-\bs b)=\)（\quad）.

\choices
    {\(4\)}
    {\(3\)}
    {\(2\)}
    {\(0\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
由 \(\lvert\bs a\rvert=1\) 得 \(\bs a\cdot\bs a=1\)，所以
\(\bs a\cdot(2\bs a-\bs b)=2\bs a\cdot\bs a-\bs a\cdot\bs b=2-(-1)=3\)，故选 B.
\end{solution}

\begin{problem}
从 2 名男同学和 3 名女同学中任选 2 人参加社区服务，则选中的 2 人都是女同学的概率为（\quad）.

\choices
    {\(0.6\)}
    {\(0.5\)}
    {\(0.4\)}
    {\(0.3\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
从 \(5\) 名同学中任选 \(2\) 人共有 \(\mathrm C_5^2=10\) 种选法，其中选中的 \(2\) 人都是女同学有 \(\mathrm C_3^2=3\) 种选法. 因此所求概率为 \(\dfrac{\mathrm C_3^2}{\mathrm C_5^2}=\dfrac3{10}=0.3\)，故选 D.
\end{solution}

\begin{problem}
双曲线 \(\dfrac{x^2}{a^2}-\dfrac{y^2}{b^2}=1\)（\(a>0\)，\(b>0\)）的离心率为 \(\sqrt3\)，则其渐近线方程为（\quad）.

\choices
    {\(y=\pm\sqrt2\,x\)}
    {\(y=\pm\sqrt3\,x\)}
    {\(y=\pm\dfrac{\sqrt2}{2}x\)}
    {\(y=\pm\dfrac{\sqrt3}{2}x\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
双曲线的离心率满足 \(e^2=1+\dfrac{b^2}{a^2}\)，所以 \(3=1+\dfrac{b^2}{a^2}\)，即 \(\dfrac ba=\sqrt2\). 其渐近线方程为 \(y=\pm\dfrac ba x=\pm\sqrt2\,x\)，故选 A.
\end{solution}

\begin{problem}
在 \(\triangle ABC\) 中，\(\cos\dfrac{C}{2}=\dfrac{\sqrt5}{5}\)，\(BC=1\)，\(AC=5\)，则 \(AB=\)（\quad）.

\choices
    {\(4\sqrt2\)}
    {\(\sqrt{30}\)}
    {\(\sqrt{29}\)}
    {\(2\sqrt5\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
由半角公式得 \(\cos C=2\cos^2\dfrac C2-1=2\cdot\dfrac15-1=-\dfrac35\). 由余弦定理，
\(AB^2=AC^2+BC^2-2\cdot AC\cdot BC\cos C=25+1-2\cdot5\cdot1\cdot\left(-\dfrac35\right)=32\)，所以 \(AB=4\sqrt2\)，故选 A.
\end{solution}

\begin{problem}
为计算 \(S=1-\dfrac12+\dfrac13-\dfrac14+\cdots+\dfrac1{99}-\dfrac1{100}\)，设计了如图的程序框图，则在空白框中应填入（\quad）.

\choices
    {\(i=i+1\)}
    {\(i=i+2\)}
    {\(i=i+3\)}
    {\(i=i+4\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
程序开始时 \(i=1\)，每次循环依次把 \(1/i\) 和 \(1/(i+1)\) 加入 \(N\) 与 \(T\)，这一次循环对应级数中的相邻两项. 为了依次取到 \(1\)，\(3\)，\(5\)，\(\ldots\)，\(99\)，循环结束后应使 \(i\) 增加 \(2\)，即填入 \(i=i+2\)，故选 B.

\begin{problem}
在正方体 \(ABCD-A_1B_1C_1D_1\) 中，\(E\) 为棱 \(CC_1\) 的中点，则异面直线 \(AE\) 与 \(CD\) 所成角的正切值为（\quad）.

\choices
    {\(\dfrac{\sqrt2}{2}\)}
    {\(\dfrac{\sqrt3}{2}\)}
    {\(\dfrac{\sqrt5}{2}\)}
    {\(\dfrac{\sqrt7}{2}\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
设正方体棱长为 \(1\)，建立直角坐标系，取 \(A=(0,0,0)\)，\(B=(1,0,0)\)，\(C=(1,1,0)\)，\(D=(0,1,0)\)，则 \(E=(1,1,\tfrac12)\). 取 \(CD\) 的方向向量 \(\bs v=(1,0,0)\)，取 \(AE\) 的方向向量 \(\bs u=(1,1,\tfrac12)\). 设两异面直线所成的锐角为 \(\theta\)，则
\(\cos\theta=\dfrac{\lvert\bs u\cdot\bs v\rvert}{\lvert\bs u\rvert\lvert\bs v\rvert}=\dfrac1{\sqrt{1+1+\tfrac14}}=\dfrac23\). 因此 \(\tan\theta=\sqrt{\dfrac1{\cos^2\theta}-1}=\sqrt{\dfrac94-1}=\dfrac{\sqrt5}{2}\)，故选 C.
\end{solution}

\begin{problem}
若 \(f(x)=\cos x-\sin x\) 在 \([0,a]\) 是减函数，则 \(a\) 的最大值是（\quad）.

\choices
    {\(\dfrac{\pi}{4}\)}
    {\(\dfrac{\pi}{2}\)}
    {\(\dfrac{3\pi}{4}\)}
    {\(\pi\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
\(f'(x)=-\sin x-\cos x=-\sqrt2\sin\left(x+\dfrac\pi4\right)\). 要使 \(f(x)\) 在 \([0,a]\) 上为减函数，需要在该区间内有 \(f'(x)\le0\)，即 \(\sin\left(x+\dfrac\pi4\right)\ge0\). 从 \(x=0\) 开始，这一条件首次失效的端点为 \(x+\dfrac\pi4=\pi\)，所以 \(a=\dfrac{3\pi}{4}\)，故选 C.
\end{solution}

\begin{problem}
已知 \(F_1\)，\(F_2\) 是椭圆 \(C\) 的两个焦点，\(P\) 是 \(C\) 上的一点，若 \(PF_1\perp PF_2\)，且 \(\angle PF_2F_1=60^\circ\)，则 \(C\) 的离心率为（\quad）.

\choices
    {\(1-\dfrac{\sqrt3}{2}\)}
    {\(2-\sqrt3\)}
    {\(\dfrac{\sqrt3-1}{2}\)}
    {\(\sqrt3-1\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
在直角三角形 \(PF_1F_2\) 中，\(\angle PF_2F_1=60^\circ\)，因此另一个锐角为 \(30^\circ\). 设椭圆焦距的一半为 \(c\)，则 \(F_1F_2=2c\) 为斜边，从而 \(PF_2=c\)，\(PF_1=\sqrt3c\). 由椭圆定义，\(PF_1+PF_2=2a\)，所以 \(2a=(\sqrt3+1)c\)，进而 \(e=\dfrac ca=\dfrac2{\sqrt3+1}=\sqrt3-1\)，故选 D.
\end{solution}

\begin{problem}
已知 \(f(x)\) 是定义域为 \((-\infty,+\infty)\) 的奇函数，满足 \(f(1-x)=f(1+x)\). 若 \(f(1)=2\)，则 \(f(1)+f(2)+f(3)+\cdots+f(50)=\)（\quad）.

\choices
    {\(-50\)}
    {\(0\)}
    {\(2\)}
    {\(50\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
令题设等式中的变量为 \(x\)，并令 \(u=1-x\)，则 \(f(u)=f(2-u)\). 由奇函数性质，\(f(2-u)=-f(u-2)\)，所以 \(f(u)=-f(u-2)\)，即 \(f(u+2)=-f(u)\). 又由 \(f(1)=2\) 得 \(f(3)=-2\)，并且取题设等式中的 \(x=1\) 得 \(f(0)=f(2)\)，而奇函数满足 \(f(0)=0\)，故 \(f(2)=0\). 于是偶数项均为 \(0\)，奇数项满足 \(f(4k+1)=2\)，\(f(4k+3)=-2\). 在 \(1\) 至 \(50\) 中，形如 \(4k+1\) 的奇数有 \(13\) 个，形如 \(4k+3\) 的奇数有 \(12\) 个，因此原式为 \(13\cdot2-12\cdot2=2\)，故选 C.
\end{solution}

\section{填空题}

\begin{problem}
曲线 \(y=2\ln x\) 在点 \((1,0)\) 处的切线方程为\fillinblank{}.
\end{problem}

\begin{answer}
\(y=2x-2\)
\end{answer}

\begin{solution}
函数 \(y=2\ln x\) 的导数为 \(y'=\dfrac2x\)，所以在 \(x=1\) 处的切线斜率为 \(2\). 由点斜式，切线方程为 \(y-0=2(x-1)\)，即 \(y=2x-2\).
\end{solution}

\begin{problem}
若变量 \(x\)，\(y\) 满足约束条件
\(\begin{cases}
x+2y-5\ge0,\\
x-2y+3\ge0,\\
x-5\le0
\end{cases}\)
则 \(z=x+y\) 的最大值为\fillinblank{}.
\end{problem}

\begin{answer}
9
\end{answer}

\begin{solution}
由约束条件得 \(y\ge\dfrac{5-x}{2}\)，\(y\le\dfrac{x+3}{2}\)，且 \(x\le5\). 因此可行域存在时还需 \(x\ge1\). 对任意固定的 \(x\)，由于目标函数 \(z=x+y\) 随 \(y\) 增大而增大，故在 \(y=\dfrac{x+3}{2}\) 处取得该 \(x\) 对应的最大值，并且
\(z\le x+\dfrac{x+3}{2}=\dfrac32x+\dfrac32\le\dfrac32\cdot5+\dfrac32=9\). 当 \(x=5\)，\(y=4\) 时满足约束条件且 \(z=9\)，所以最大值为 \(9\).
\end{solution}

\begin{problem}
已知 \(\tan\left(\alpha-\dfrac{5\pi}{4}\right)=\dfrac15\)，则 \(\tan\alpha=\fillinblank{}\).
\end{problem}

\begin{answer}
\(\dfrac32\)
\end{answer}

\begin{solution}
设 \(t=\tan\alpha\). 因为 \(\tan\dfrac{5\pi}{4}=1\)，由正切的差角公式得
\(\dfrac{t-1}{1+t}=\dfrac15\)，所以 \(5t-5=1+t\)，解得 \(t=\dfrac32\). 因此 \(\tan\alpha=\dfrac32\).
\end{solution}

\begin{problem}
已知圆锥的顶点为 \(S\)，母线 \(SA\)，\(SB\) 互相垂直，\(SA\) 与圆锥底面所成角为 \(30^\circ\)，若 \(\triangle SAB\) 的面积为 \(8\)，则该圆锥的体积为\fillinblank{}.
\end{problem}

\begin{answer}
\(8\pi\)
\end{answer}

\begin{solution}
圆锥的母线长度相等，所以 \(SA=SB\). 由 \(\triangle SAB\) 的面积为 \(8\)，且 \(SA\perp SB\)，得 \(\dfrac12SA\cdot SB=8\)，从而 \(SA^2=16\)，即 \(SA=SB=4\). 圆锥的高为 \(h=SA\sin30^\circ=2\)，底面半径为 \(r=SA\cos30^\circ=2\sqrt3\). 因此圆锥体积为 \(V=\dfrac13\pi r^2h=\dfrac13\pi\cdot12\cdot2=8\pi\).
\end{solution}

\begin{problem}
记 \(S_n\) 为等差数列 \(\{a_n\}\) 的前 \(n\) 项和，已知 \(a_1=-7\)，\(S_3=-15\).

\begin{enumerate}
    \item 求 \(\{a_n\}\) 的通项公式；
    \item 求 \(S_n\)，并求 \(S_n\) 的最小值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设等差数列 \(\{a_n\}\) 的公差为 \(d\). 由 \(S_3=\dfrac32(a_1+a_3)\) 及 \(a_3=a_1+2d\)，得
\( -15=\dfrac32(-7-7+2d)=3(d-7)\)，所以 \(d=2\). 因此 \(a_n=a_1+(n-1)d=-7+2(n-1)=2n-9\).
\item \(S_n=\dfrac n2(a_1+a_n)=\dfrac n2(-7+2n-9)=n(n-8)=(n-4)^2-16\). 由于 \(n\) 为正整数，当 \(n=4\) 时取得最小值，最小值为 \(-16\).
\end{enumerate}
\end{solution}

\begin{problem}
如图是某地区 2000 年至 2016 年环境基础设施投资额 \(y\)（单位：亿元）的折线图.

为了预测该地区 2018 年的环境基础设施投资额，建立了 \(y\) 与时间变量 \(t\) 的两个线性回归模型.

\begin{circlelist}
\item 根据 2000 年至 2016 年的数据（时间变量 \(t\) 的值依次为 \(1\)，\(2\)，\(\cdots\)，\(17\)）建立模型：\(\hat y=-30.4+13.5t\)；
\item 根据 2010 年至 2016 年的数据（时间变量 \(t\) 的值依次为 \(1\)，\(2\)，\(\cdots\)，\(7\)）建立模型：\(\hat y=99+17.5t\).
\end{circlelist}

\begin{enumerate}
    \item 分别利用这两个模型，求该地区 2018 年的环境基础设施投资额的预测值；
    \item 你认为用哪个模型得到的预测值更可靠？并说明理由.
\end{enumerate}

\begingroup
\setlength{\leftskip}{0pt}%
\bitmapfigure[float=inline,width=0.88\linewidth]{img/2018/national_paper_2_liberal/q18_fig1.png}
\endgroup
\end{problem}

\begin{solution}
\begin{enumerate}
\item 在第一个模型中，2000 年对应 \(t=1\)，所以 2018 年对应 \(t=19\)，预测值为 \(\hat y=-30.4+13.5\times19=226.1\)（亿元）. 在第二个模型中，2010 年对应 \(t=1\)，所以 2018 年对应 \(t=9\)，预测值为 \(\hat y=99+17.5\times9=256.5\)（亿元）.

\end{enumerate}
\end{solution}

\begin{problem}
如图，在三棱锥 \(P-ABC\) 中，\(AB=BC=2\sqrt2\)，\(PA=PB=PC=AC=4\)，\(O\) 为 \(AC\) 的中点.

\begin{enumerate}
    \item 证明：\(PO\perp\) 平面 \(ABC\)；
    \item 若点 \(M\) 在棱 \(BC\) 上，且 \(MC=2MB\)，求点 \(C\) 到平面 \(POM\) 的距离.
\end{enumerate}

\bitmapfigure[float=inline,width=0.42\linewidth]{img/2018/national_paper_2_liberal/q19_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 因为 \(PA=PC\)，且 \(O\) 是 \(AC\) 的中点，所以 \(PO\perp AC\). 又 \(AB=BC=2\sqrt2\)，\(AC=4\)，满足 \(AB^2+BC^2=AC^2\)，故 \(\triangle ABC\) 在 \(B\) 处为直角三角形，且斜边中线 \(BO=\dfrac12AC=2\). 在直角三角形 \(PAO\) 中，\(PO^2=PA^2-AO^2=4^2-2^2=12\). 于是 \(PO^2+BO^2=12+4=16=PB^2\)，所以 \(PO\perp BO\). 直线 \(AC\) 与 \(BO\) 在平面 \(ABC\) 内相交，且 \(PO\) 同时垂直于它们，因此 \(PO\perp\) 平面 \(ABC\).
\item 取 \(O\) 为原点，令 \(AC\) 在 \(x\) 轴上，建立空间直角坐标系，可取 \(A=(-2,0,0)\)，\(C=(2,0,0)\)，\(B=(0,2,0)\)，\(P=(0,0,2\sqrt3)\). 由 \(MC=2MB\) 及 \(BC=2\sqrt2\)，得 \(MB=\dfrac13BC\)，所以 \(M=\dfrac23B+\dfrac13C=\left(\dfrac23,\dfrac43,0\right)\). 平面 \(POM\) 过原点，且其在底面内的方向为 \(OM\)，故其方程为 \(2x-y=0\). 于是点 \(C=(2,0,0)\) 到平面 \(POM\) 的距离为
\(d=\dfrac{\lvert2\cdot2-0\rvert}{\sqrt{2^2+(-1)^2}}=\dfrac4{\sqrt5}=\dfrac{4\sqrt5}{5}\).
\end{enumerate}
\end{solution}

\begin{problem}
设抛物线 \(C:y^2=4x\) 的焦点为 \(F\)，过 \(F\) 且斜率为 \(k\)（\(k>0\)）的直线 \(l\) 与 \(C\) 交于 \(A\)，\(B\) 两点，\(\lvert AB\rvert=8\).

\begin{enumerate}
    \item 求 \(l\) 的方程；
    \item 求过点 \(A\)，\(B\) 且与 \(C\) 的准线相切的圆的方程.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 抛物线 \(y^2=4x\) 的焦点为 \(F=(1,0)\)，所以直线 \(l\) 的方程为 \(y=k(x-1)\)，其中 \(k>0\). 设 \(l\) 与抛物线交点的横坐标为 \(x_1\)，\(x_2\)，代入得 \(k^2(x-1)^2=4x\)，即 \(k^2x^2-(2k^2+4)x+k^2=0\). 因此 \(x_1+x_2=2+\dfrac4{k^2}\)，\(x_1x_2=1\)，从而
\(\lvert x_1-x_2\rvert=\sqrt{(x_1+x_2)^2-4x_1x_2}=\dfrac{4\sqrt{1+k^2}}{k^2}\). 由于两点在斜率为 \(k\) 的直线上，\(\lvert AB\rvert=\lvert x_1-x_2\rvert\sqrt{1+k^2}=\dfrac{4(1+k^2)}{k^2}=8\)，所以 \(k^2=1\). 由 \(k>0\) 得 \(k=1\)，故 \(l:y=x-1\).
\item 由 \(y=x-1\) 与 \(y^2=4x\) 联立，得 \((x-1)^2=4x\)，即 \(x^2-6x+1=0\)，所以两交点可写为 \(A=(3+2\sqrt2,2+2\sqrt2)\)，\(B=(3-2\sqrt2,2-2\sqrt2)\). 弦 \(AB\) 的中点为 \((3,2)\)，其垂直平分线为 \(y=-x+5\). 设所求圆的圆心为 \(D=(m,5-m)\)，半径为 \(r\). 弦长为 \(8\)，故圆心到弦的垂距为 \(\sqrt2\lvert m-3\rvert\)，由勾股定理得 \(r^2=4^2+2(m-3)^2\). 抛物线的准线为 \(x=-1\)，圆与准线相切，所以 \(r=\lvert m+1\rvert\). 于是
\((m+1)^2=16+2(m-3)^2\)，解得 \(m=3\) 或 \(m=11\). 当 \(m=3\) 时，\(D=(3,2)\)，\(r=4\)，圆方程为 \((x-3)^2+(y-2)^2=16\)；当 \(m=11\) 时，\(D=(11,-6)\)，\(r=12\)，圆方程为 \((x-11)^2+(y+6)^2=144\). 因此所求圆有上述两个.
\end{enumerate}
\end{solution}

\begin{problem}
已知函数 \(f(x)=\dfrac13x^3-a(x^2+x+1)\).

\begin{enumerate}
    \item 若 \(a=3\)，求 \(f(x)\) 的单调区间；
    \item 证明：\(f(x)\) 只有一个零点.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 当 \(a=3\) 时，\(f(x)=\dfrac13x^3-3x^2-3x-3\)，所以 \(f'(x)=x^2-6x-3\). 令 \(f'(x)=0\)，得 \(x=3\pm2\sqrt3\). 因为二次函数 \(x^2-6x-3\) 的开口向上，所以 \(f'(x)>0\) 在 \(\left(-\infty,3-2\sqrt3\right)\) 和 \(\left(3+2\sqrt3,+\infty\right)\) 上成立，\(f'(x)<0\) 在 \(\left(3-2\sqrt3,3+2\sqrt3\right)\) 上成立. 因此 \(f(x)\) 的单调递增区间为 \(\left(-\infty,3-2\sqrt3\right)\) 和 \(\left(3+2\sqrt3,+\infty\right)\)，单调递减区间为 \(\left(3-2\sqrt3,3+2\sqrt3\right)\).
\item 因为 \(x^2+x+1=\left(x+\dfrac12\right)^2+\dfrac34>0\)，所以 \(f(x)=0\) 等价于
\(a=\dfrac{x^3}{3(x^2+x+1)}\). 设 \(g(x)=\dfrac{x^3}{3(x^2+x+1)}\)，则
\(g'(x)=\dfrac{x^2(x^2+2x+3)}{3(x^2+x+1)^2}=\dfrac{x^2\left((x+1)^2+2\right)}{3(x^2+x+1)^2}\ge0\)，且只有 \(x=0\) 时 \(g'(x)=0\). 对任意 \(x_1<x_2\)，由导数的连续性，
\(g(x_2)-g(x_1)=\displaystyle\int_{x_1}^{x_2}g'(x)\,\mathrm{d}x>0\)，因为 \(g'(x)\) 在该区间上非负且不可能恒为零，所以 \(g(x)\) 在 \(\R\) 上严格递增. 又
\(g(x)=\dfrac{x}{3}\cdot\dfrac{x^2}{x^2+x+1}\)，从而 \(g(x)\to-\infty\)（\(x\to-\infty\)），\(g(x)\to+\infty\)（\(x\to+\infty\)）. 因此对任意给定的实数 \(a\)，由连续函数的介值性，方程 \(g(x)=a\) 至少有一个实数解；由严格递增性，至多有一个实数解. 所以方程 \(g(x)=a\)，也就是 \(f(x)=0\)，有且只有一个实数解，即 \(f(x)\) 只有一个零点.
\end{enumerate}
\end{solution}

\section{选考题：解答题}

\begin{problem}
在直角坐标系 \(xOy\) 中，曲线 \(C\) 的参数方程为
\(\begin{cases}
x=2\cos\theta,\\
y=4\sin\theta
\end{cases}\)
（\(\theta\) 为参数），直线 \(l\) 的参数方程为
\(\begin{cases}
x=1+t\cos\alpha,\\
y=2+t\sin\alpha
\end{cases}\)
（\(t\) 为参数）.

\begin{enumerate}
    \item 求 \(C\) 和 \(l\) 的直角坐标方程；
    \item 若曲线 \(C\) 截直线 \(l\) 所得线段的中点坐标为 \((1,2)\)，求 \(l\) 的斜率.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由 \(x=2\cos\theta\)，\(y=4\sin\theta\)，得 \(\cos\theta=\dfrac x2\)，\(\sin\theta=\dfrac y4\). 利用 \(\sin^2\theta+\cos^2\theta=1\)，可得曲线 \(C\) 的直角坐标方程为 \(\dfrac{x^2}{4}+\dfrac{y^2}{16}=1\). 由直线 \(l\) 的参数方程得 \(x-1=t\cos\alpha\)，\(y-2=t\sin\alpha\)，消去参数 \(t\) 可得 \(l\) 的方程为 \((x-1)\sin\alpha-(y-2)\cos\alpha=0\).
\item 将直线参数方程代入椭圆方程，记 \(c=\cos\alpha\)，\(s=\sin\alpha\)，得
\(\dfrac{(1+tc)^2}{4}+\dfrac{(2+ts)^2}{16}=1\)，即 \((4c^2+s^2)t^2+(8c+4s)t-8=0\). 若所得弦的中点为 \((1,2)\)，则对应的两个参数根关于 \(t=0\) 对称，故根的和为零，从而 \(8c+4s=0\)，即 \(2\cos\alpha+\sin\alpha=0\). 因 \(\cos\alpha\ne0\)，直线 \(l\) 的斜率为 \(\tan\alpha=-2\).
\end{enumerate}
\end{solution}

\begin{problem}
设函数 \(f(x)=5-\lvert x+a\rvert-\lvert x-2\rvert\).

\begin{enumerate}
    \item 当 \(a=1\) 时，求不等式 \(f(x)\ge0\) 的解集；
    \item 若 \(f(x)\le1\)，求 \(a\) 的取值范围.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 当 \(a=1\) 时，不等式化为 \(\lvert x+1\rvert+\lvert x-2\rvert\le5\). 当 \(x\le-1\) 时，左边为 \(-2x+1\)，解得 \(x\ge-2\)；当 \(-1\le x\le2\) 时，左边恒为 \(3\)，恒满足不等式；当 \(x\ge2\) 时，左边为 \(2x-1\)，解得 \(x\le3\). 所以解集为 \([-2,3]\).
\item \(f(x)\le1\) 等价于 \(\lvert x+a\rvert+\lvert x-2\rvert\ge4\). 由三角不等式，\(\lvert x+a\rvert+\lvert x-2\rvert\ge\lvert a+2\rvert\)，并且当 \(x\) 位于 \(-a\) 与 \(2\) 之间时等号成立，所以左边关于 \(x\) 的最小值为 \(\lvert a+2\rvert\). 因此要使不等式对任意实数 \(x\) 成立，必须且只需 \(\lvert a+2\rvert\ge4\)，即 \(a\le-6\) 或 \(a\ge2\).
\end{enumerate}
\end{solution}
