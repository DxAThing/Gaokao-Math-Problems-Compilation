\section{单选题}

\begin{problem}
若复数 \( z \) 满足 \( 2z+\overline{z}=3-2\i \)，其中 \( \i \) 为虚数单位，则 \( z=\)（\quad）

\choices
    {\( 1+2\i \)}
    {\( 1-2\i \)}
    {\( -1+2\i \)}
    {\( -1-2\i \)}

\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
设 \( z=a+b\i \)，则 \( \overline{z}=a-b\i \). 由
\(2\left( a+b\i \right)+a-b\i =3-2\i\)
得
\(3a=3\)，\(b=-2\).
所以 \( z=1-2\i \). 故选 B.
\end{solution}

\begin{problem}
设集合 \( A=\{y\mid y=2^x,\ x\in\R\} \)，\( B=\{x\mid x^2-1<0\} \)，则 \( A\cup B=\)（\quad）

\choices
    {\( (-1,1) \)}
    {\( (0,1) \)}
    {\( (-1,+\infty) \)}
    {\( (0,+\infty) \)}

\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
因为 \( A=(0,+\infty) \)，\( B=(-1,1) \)，所以
\(A\cup B=(-1,+\infty)\).
故选 C.
\end{solution}

\begin{problem}
某高校调查了 \( 200 \) 名学生每周的自习时间（单位：小时），制成了如图所示的频率分布直方图，其中自习时间的范围是 \( [17.5,30] \)，样本数据分组为 \( [17.5,20) \)，\( [20,22.5) \)，\( [22.5,25) \)，\( [25,27.5) \)，\( [27.5,30] \). 根据直方图，这 \( 200 \) 名学生中每周的自习时间不少于 \( 22.5 \) 小时的人数是（\quad）

\texfigure[max width=5.8cm]{img_repaint/2016/shandong_science/q3_fig1.tex}

\choices
    {\( 56 \)}
    {\( 60 \)}
    {\( 120 \)}
    {\( 140 \)}

\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
自习时间不少于 \( 22.5 \) 小时的频率为
\(\left( 0.16+0.08+0.04 \right)\times 2.5=0.7\).
所以人数为
\(0.7\times 200=140\).
故选 D.
\end{solution}

\begin{problem}
若变量 \(x\)，\(y\) 满足 \( x+y\le 2 \)，\( 2x-3y\le 9 \)，\( x\ge 0 \)，则 \( x^2+y^2 \) 的最大值是（\quad）

\choices
    {\( 4 \)}
    {\( 9 \)}
    {\( 10 \)}
    {\( 12 \)}

\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
作出可行域，可得其顶点为 \( A(0,-3) \)，\( B(3,-1) \)，\( C(0,2) \). 因为
\(OA^2=9\)，\(OB^2=10\)，\(OC^2=4\)，
所以 \( x^2+y^2 \) 的最大值是 \( 10 \). 故选 C.
\end{solution}

\begin{problem}
一个由半球和四棱锥组成的几何体，其三视图如图所示，则该几何体的体积为（\quad）

\bitmapfigure[max width=6.2cm]{img/2016/shandong_science/q5_fig1.png}

\choices
    {\( \dfrac13+\dfrac{\pi}{3} \)}
    {\( \dfrac13+\dfrac{\sqrt2\,\pi}{3} \)}
    {\( \dfrac13+\dfrac{\sqrt2\,\pi}{6} \)}
    {\( 1+\dfrac{\sqrt2\,\pi}{6} \)}

\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
由俯视图可得下部四棱锥的底面是边长为 \( 1 \) 的正方形，由正视图与侧视图可得其高为 \( 1 \)，所以棱锥体积为
\(\frac13\).
上部是一个半球，且其直径等于正方形底面的对角线，所以半径为
\(R=\frac{\sqrt2}{2}\).
半球体积为
\(\frac12\cdot \frac43\pi R^3=\frac{\sqrt2\,\pi}{6}\).
所以该几何体的体积为
\(\frac13+\frac{\sqrt2\,\pi}{6}\).
故选 C.
\end{solution}

\begin{problem}
已知直线 \(a\)，\(b\) 分别在两个不同的平面 \(\alpha\)，\(\beta\) 内，则“直线 \( a \) 和直线 \( b \) 相交”是“平面 \( \alpha \) 和平面 \( \beta \) 相交”的（\quad）

\choices
    {充分不必要条件}
    {必要不充分条件}
    {充要条件}
    {既不充分也不必要条件}

\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
当直线 \( a \) 和直线 \( b \) 相交时，平面 \( \alpha \) 和平面 \( \beta \) 一定相交；反过来，平面 \( \alpha \) 和平面 \( \beta \) 相交时，直线 \( a \) 和直线 \( b \) 不一定相交. 所以“直线 \( a \) 和直线 \( b \) 相交”是“平面 \( \alpha \) 和平面 \( \beta \) 相交”的充分不必要条件. 故选 A.
\end{solution}

\begin{problem}
函数 \( f(x)=\left( \sin x+\cos x \right)\left( \cos x-\sin x \right) \) 的最小正周期是（\quad）

\choices
    {\( \dfrac{\pi}{2} \)}
    {\( \pi \)}
    {\( \dfrac{3\pi}{2} \)}
    {\( 2\pi \)}

\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
因为
\(f(x)=\left( \sin x+\cos x \right)\left( \cos x-\sin x \right)=\cos^2x-\sin^2x=\cos 2x\)，
所以最小正周期是 \( \pi \). 故选 B.
\end{solution}

\begin{problem}
已知非零向量 \(\bs a\)，\(\bs b\) 满足 \( 4|\bs a|=3|\bs b| \)，\( \cos\angle(\bs a,\bs b)=\dfrac13 \). 若 \( \bs b\perp\left( t\bs a+\bs b \right) \)，则实数 \( t \) 的值为（\quad）

\choices
    {\( 4 \)}
    {\( -4 \)}
    {\( \dfrac94 \)}
    {\( -\dfrac94 \)}

\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
由 \( \bs b\cdot\left( t\bs a+\bs b \right)=0 \) 得
\(t\,\bs a\cdot\bs b+|\bs b|^2=0\).
又
\(\bs a\cdot\bs b=|\bs a||\bs b|\cos\angle(\bs a,\bs b)=\frac13|\bs a||\bs b|\).
由 \( 4|\bs a|=3|\bs b| \) 得 \( |\bs b|=\dfrac43|\bs a| \)，代入上式可得
\(t\cdot \frac13|\bs a||\bs b|+|\bs b|^2=0\)
即
\(t\cdot \frac13|\bs a|\cdot \frac43|\bs a|+\left( \frac43|\bs a| \right)^2=0\).
化简得
\(\frac49 t|\bs a|^2+\frac{16}{9}|\bs a|^2=0\)，
所以 \( t=-4 \). 故选 B.
\end{solution}

\begin{problem}
已知函数 \( f(x) \) 的定义域为 \( \R \). 当 \( x<0 \) 时，\( f(x)=x^3-1 \)；当 \( -1\le x\le 1 \) 时，\( f(-x)=-f(x) \)；当 \( x>\dfrac12 \) 时，\( f\!\left( x+\dfrac12 \right)=f\!\left( x-\dfrac12 \right) \). 则 \( f(6)=\)（\quad）

\choices
    {\( -2 \)}
    {\( -1 \)}
    {\( 0 \)}
    {\( 2 \)}

\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
由 \( x>\dfrac12 \) 时 \( f\!\left( x+\dfrac12 \right)=f\!\left( x-\dfrac12 \right) \) 可得，当 \( x>\dfrac12 \) 时，
\(f(x+1)=f(x)\).
所以 \( f(6)=f(1) \). 又当 \( -1\le x\le 1 \) 时，\( f(-x)=-f(x) \)，故
\(f(1)=-f(-1)\).
当 \( x<0 \) 时，\( f(x)=x^3-1 \)，所以
\(f(-1)=-2\).
因此
\(f(6)=f(1)=-f(-1)=2\).
故选 D.
\end{solution}

\begin{problem}
若函数 \( y=f(x) \) 的图象上存在两点，使得函数的图象在这两点处的切线互相垂直，则称 \( y=f(x) \) 具有 T 性质. 下列函数中具有 T 性质的是（\quad）

\choices
    {\( y=\sin x \)}
    {\( y=\ln x \)}
    {\( y=\e^x \)}
    {\( y=x^3 \)}

\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
若函数 \( y=f(x) \) 具有 T 性质，则其导函数在两点处的函数值互为负倒数. 对于 \( y=\sin x \)，其导函数为 \( y'=\cos x \)，可以取到 \( 1 \) 和 \( -1 \)，故具有 T 性质. 对于 \( y=\ln x \)，\( y=\e^x \)，\( y=x^3 \)，其导函数恒大于 \( 0 \)，不可能取到互为负倒数的一对值. 故选 A.
\end{solution}

\begin{problem}
执行如图的程序框图，若输入的 \(a\)，\(b\) 的值分别为 \( 0 \) 和 \( 9 \)，则输出的 \( i \) 的值为\fillinblank{}.

\texfigure[max width=6.0cm]{img_repaint/2016/shandong_science/q11_fig1.tex}

\end{problem}

\begin{answer}
\( 3 \)
\end{answer}

\begin{solution}
输入 \( a=0 \)，\( b=9 \)，\( i=1 \).
第一次执行循环体后，\( a=1 \)，\( b=8 \)，不满足 \( a\ge b \)，所以 \( i=2 \).
第二次执行循环体后，\( a=3 \)，\( b=6 \)，不满足 \( a\ge b \)，所以 \( i=3 \).
第三次执行循环体后，\( a=6 \)，\( b=3 \)，满足 \( a\ge b \)，输出 \( i=3 \).
\end{solution}

\begin{problem}
若 \( \left( ax^2+\dfrac{1}{\sqrt{x}} \right)^5 \) 的展开式中 \( x^5 \) 的系数是 \( -80 \)，则实数 \( a=\fillinblank{} \).

\end{problem}

\begin{answer}
\( -2 \)
\end{answer}

\begin{solution}
展开式的通项为 \(T_{r+1}=\binom{5}{r}\left( ax^2 \right)^{5-r}\left( \frac{1}{\sqrt{x}} \right)^r=\binom{5}{r}a^{5-r}x^{10-\frac52 r}\).
令 \( 10-\dfrac52 r=5 \)，得 \( r=2 \). 所以 \( x^5 \) 的系数为
\(\binom{5}{2}a^3=10a^3\).
由 \( 10a^3=-80 \) 得 \( a=-2 \).
\end{solution}

\begin{problem}
已知双曲线 \( E:\dfrac{x^2}{a^2}-\dfrac{y^2}{b^2}=1 \)（\(a>0\)，\(b>0\)）. 矩形 \(ABCD\) 的四个顶点在 \(E\) 上，\(AB\)，\(CD\) 的中点为 \(E\) 的两个焦点，且 \( 2|AB|=3|BC| \)，则 \(E\) 的离心率是\fillinblank{}.

\end{problem}

\begin{answer}
\( 2 \)
\end{answer}

\begin{solution}
设双曲线焦点为 \( (\pm c,0) \). 由题意可设
\(A\left( -c,\frac{b^2}{a} \right)\)，\(B\left( -c,-\frac{b^2}{a} \right)\)，\(C\left( c,-\frac{b^2}{a} \right)\)，\(D\left( c,\frac{b^2}{a} \right)\).
于是
\(|AB|=\frac{2b^2}{a}\)，\(|BC|=2c\).
由 \( 2|AB|=3|BC| \) 得
\(2\cdot \frac{2b^2}{a}=3\cdot 2c\)，
即
\(2b^2=3ac\).
又 \( b^2=c^2-a^2 \)，且 \( e=\dfrac{c}{a} \)，所以
\(2e^2-3e-2=0\).
解得 \( e=2 \)（舍去负值）.
\end{solution}

\begin{problem}
在 \( [-1,1] \) 上随机地取一个数 \( k \)，则事件“直线 \( y=kx \) 与圆 \( (x-5)^2+y^2=9 \) 相交”发生的概率为\fillinblank{}.

\end{problem}

\begin{answer}
\( \dfrac34 \)
\end{answer}

\begin{solution}
圆 \( (x-5)^2+y^2=9 \) 的圆心为 \( (5,0) \)，半径为 \( 3 \). 圆心到直线 \( y=kx \) 的距离为
\(\frac{5|k|}{\sqrt{1+k^2}}\).
要使直线与圆相交，需
\(\frac{5|k|}{\sqrt{1+k^2}}<3\).
解得
\(-\frac34<k<\frac34\).
因此所求概率为
\(\frac{\frac34-\left( -\frac34 \right)}{1-\left( -1 \right)}=\frac34\).
\end{solution}

\begin{problem}
已知函数 \( f(x)=|x| \)（\( x\le m \)），\( f(x)=x^2-2mx+4m \)（\( x>m \)），其中 \( m>0 \). 若存在实数 \( b \)，使得关于 \( x \) 的方程 \( f(x)=b \) 有三个不同的根，则 \( m \) 的取值范围是\fillinblank{}.

\end{problem}

\begin{answer}
\( (3,+\infty) \)
\end{answer}

\begin{solution}
当 \( x>m \) 时，
\(f(x)=x^2-2mx+4m=\left( x-m \right)^2+4m-m^2>4m-m^2\).
所以要使方程 \( f(x)=b \) 有三个不同的根，必须有
\(4m-m^2<m\)
即
\(m^2-3m>0\).
又 \( m>0 \)，故 \( m>3 \). 因此 \( m \) 的取值范围是 \( (3,+\infty) \).
\end{solution}
\section{解答题}

\begin{problem}
在 \( \triangle ABC \) 中，角 \(A\)，\(B\)，\(C\) 的对边分别为 \(a\)，\(b\)，\(c\)，已知 \( 2\left( \tan A+\tan B \right)=\dfrac{\tan A}{\cos B}+\dfrac{\tan B}{\cos A} \).
\begin{enumerate}
\item 证明：\( a+b=2c \)；
\item 求 \( \cos C \) 的最小值.
\end{enumerate}

\end{problem}

\begin{solution}
（1）由已知得
\(2\left( \frac{\sin A}{\cos A}+\frac{\sin B}{\cos B} \right)=\frac{\sin A}{\cos A\cos B}+\frac{\sin B}{\cos A\cos B}\).
两边同乘以 \( \cos A\cos B \) 得
\(2\left( \sin A\cos B+\cos A\sin B \right)=\sin A+\sin B\).
所以
\(2\sin\left( A+B \right)=\sin A+\sin B\).
即
\(\sin A+\sin B=2\sin C. \circled{1}\)
根据正弦定理，
\(\frac{a}{\sin A}=\frac{b}{\sin B}=\frac{c}{\sin C}=2R\).
所以
\(\sin A=\frac{a}{2R}\)，\(\sin B=\frac{b}{2R}\)，\(\sin C=\frac{c}{2R}\).
将它们代入 \circled{1} 式，得
\(\frac{a}{2R}+\frac{b}{2R}=2\cdot \frac{c}{2R}\).
所以
\(a+b=2c\).

（2）由 \( a+b=2c \)，得
\(\left( a+b \right)^2=a^2+b^2+2ab=4c^2\).
所以
\(a^2+b^2=4c^2-2ab\).
又 \( a^2+b^2\ge 2ab \)，当且仅当 \( a=b \) 时取等号，且 \(a\)，\(b>0\)，所以
\(4c^2\ge 4ab\)，
即
\(\frac{c^2}{ab}\ge 1\).
由余弦定理，
\(\cos C=\frac{a^2+b^2-c^2}{2ab}=\frac{3c^2-2ab}{2ab}=\frac32\cdot \frac{c^2}{ab}-1\ge \frac12\).
所以 \( \cos C \) 的最小值为
\(\frac12\).
\end{solution}

\begin{problem}
在如图所示的圆台中，\(AC\) 是下底面圆 \(O\) 的直径，\(EF\) 是上底面圆 \(O'\) 的直径，\(FB\) 是圆台的一条母线.
\begin{enumerate}
\item 已知 \(G\)，\(H\) 分别为 \(EC\)，\(FB\) 的中点，求证：\(GH\) // 平面 \(ABC\)；
\item 已知 \( EF=FB=\dfrac12 AC=2\sqrt3 \)，\( AB=BC \)，求二面角 \( F-BC-A \) 的余弦值.
\end{enumerate}

\bitmapfigure[max width=5.3cm]{img/2016/shandong_science/q17_fig1.png}

\end{problem}

\begin{solution}
（1）取 \(FC\) 的中点 \(Q\)，连接 \(GQ\)，\(QH\).
因为 \(G\)，\(H\) 分别为 \(EC\)，\(FB\) 的中点，所以
\(GQ//\frac12 EF\)，\(QH//\frac12 BC\).
又 \( EF//BO \)，所以
\(GQ//\frac12 BO\).
因此平面 \(GQH\) // 平面 \(ABC\).
因为 \( GH\subset \) 平面 \(GQH\)，所以
\(GH//\text{平面 }ABC\).

（2）因为 \( AB=BC \)，所以 \( BO\perp AC \).

又 \( OO' \perp \) 平面 \(ABC\)，所以以 \(O\) 为原点，\(OA\) 为 \( x \) 轴，\(OB\) 为 \( y \) 轴，\(OO'\) 为 \( z \) 轴，建立空间直角坐标系.
由 \( \dfrac12 AC=2\sqrt3 \)，得
\(A\left( 2\sqrt3,0,0 \right)\)，\(C\left( -2\sqrt3,0,0 \right)\).
由 \( AB=BC \)，得
\(B\left( 0,2\sqrt3,0 \right)\).
又 \( EF=2\sqrt3 \)，所以
\(O'F=\sqrt3\).
由 \( FB=2\sqrt3 \)，得
\(OO'=3\).
所以
\(O'\left( 0,0,3 \right)\)，\(F\left( 0,\sqrt3,3 \right)\).
从而
\(\overrightarrow{FC}=\left( -2\sqrt3,-\sqrt3,-3 \right)\)，\(\overrightarrow{CB}=\left( 2\sqrt3,2\sqrt3,0 \right)\).
由题意可知，平面 \(ABC\) 的一个法向量为
\(\overrightarrow{OO'}=\left( 0,0,3 \right)\).
设
\(\bs n=\left( x_0,y_0,z_0 \right)\)
为平面 \(FCB\) 的一个法向量，则
\(\bs n\cdot \overrightarrow{FC}=0\)，\(\bs n\cdot \overrightarrow{CB}=0\).
即
\(\begin{cases}
-2\sqrt3x_0-\sqrt3y_0-3z_0=0,\\
2\sqrt3x_0+2\sqrt3y_0=0.
\end{cases}\)
取 \( x_0=1 \)，得
\(\bs n=\left( 1,-1,-\frac{\sqrt3}{3} \right)\).
于是
\(\cos\angle \left( \overrightarrow{OO'},\bs n \right)=\frac{\overrightarrow{OO'}\cdot \bs n}{\left| \overrightarrow{OO'} \right|\cdot \left| \bs n \right|}=-\frac{\sqrt7}{7}\).
因为二面角 \( F-BC-A \) 的平面角是锐角，所以二面角 \( F-BC-A \) 的余弦值为
\(\frac{\sqrt7}{7}\).
\end{solution}

\begin{problem}
已知数列 \( \{a_n\} \) 的前 \( n \) 项和 \( S_n=3n^2+8n \)，\( \{b_n\} \) 是等差数列，且 \( a_n=b_n+b_{n+1} \).
\begin{enumerate}
\item 求数列 \( \{b_n\} \) 的通项公式；
\item 令 \( c_n=\dfrac{\left( a_n+1 \right)^{n+1}}{\left( b_n+2 \right)^n} \)，求数列 \( \{c_n\} \) 的前 \( n \) 项和 \( T_n \).
\end{enumerate}

\end{problem}

\begin{solution}
（1）由
\(S_n=3n^2+8n\)
得
\(a_n=S_n-S_{n-1}=6n+5\)（\(n\ge 2\)），
又 \( a_1=S_1=11 \)，故
\(a_n=6n+5\).
设等差数列 \( \{b_n\} \) 的公差为 \( d \)，则
\(a_n-a_{n-1}=b_{n+1}-b_{n-1}=2d\).
因为
\(a_n-a_{n-1}=6\)，
所以
\(2d=6\)，\(d=3\).
又
\(a_1=b_1+b_2=2b_1+d\)，
故
\(11=2b_1+3\)，
解得
\(b_1=4\).
所以
\(b_n=4+3\left( n-1 \right)=3n+1\).

（2）由
\(a_n+1=6n+6=6\left( n+1 \right)\)，\(b_n+2=3n+3=3\left( n+1 \right)\)
得
\(c_n=\frac{\left[ 6\left( n+1 \right) \right]^{n+1}}{\left[ 3\left( n+1 \right) \right]^n}=6\left( n+1 \right)\cdot 2^n\).
所以
\(T_n=6\left[ 2\cdot 2+3\cdot 2^2+\cdots +\left( n+1 \right)2^n \right]. \circled{1}\)
两边同乘 \( 2 \)，得
\(2T_n=6\left[ 2\cdot 2^2+3\cdot 2^3+\cdots +n\cdot 2^n+\left( n+1 \right)2^{n+1} \right]. \circled{2}\)
\circled{1} 式减去 \circled{2} 式，得
\(-T_n=6\left[ 2\cdot 2+2^2+2^3+\cdots +2^n-\left( n+1 \right)2^{n+1} \right]\).
即
\(-T_n=12+6\cdot \frac{2\left( 1-2^n \right)}{1-2}-6\left( n+1 \right)2^{n+1}\)
化简得
\(-T_n=\left( -6n \right)2^{n+1}=-3n\cdot 2^{n+2}\)
所以
\(T_n=3n\cdot 2^{n+2}\).
\end{solution}

\begin{problem}
甲，乙两人组成“星队”参加猜成语活动，每轮活动由甲，乙各猜一个成语，在一轮活动中，如果两人都猜对，则“星队”得 \( 3 \) 分；如果只有一个人猜对，则“星队”得 \( 1 \) 分；如果两人都没猜对，则“星队”得 \( 0 \) 分. 已知甲每轮猜对的概率是 \( \dfrac34 \)，乙每轮猜对的概率是 \( \dfrac23 \)；每轮活动中甲，乙猜对与否互不影响；各轮结果亦互不影响. 假设“星队”参加两轮活动，求：
\begin{enumerate}
\item “星队”至少猜对 \( 3 \) 个成语的概率；
\item “星队”两轮得分之和为 \(X\) 的分布列和数学期望 \( E(X) \).
\end{enumerate}

\end{problem}

\begin{solution}
（1）“星队”至少猜对 \( 3 \) 个成语包含“甲猜对 \( 1 \) 个，乙猜对 \( 2 \) 个”“甲猜对 \( 2 \) 个，乙猜对 \( 1 \) 个”“甲猜对 \( 2 \) 个，乙猜对 \( 2 \) 个”三个基本事件.
故所求概率
\(P=\binom21\cdot \frac34\cdot \left( 1-\frac34 \right)\cdot \left( \frac23 \right)^2
+\left( \frac34 \right)^2\cdot \binom21\cdot \frac23\cdot \left( 1-\frac23 \right)
+\left( \frac34 \right)^2\cdot \left( \frac23 \right)^2\)
\(=\frac16+\frac14+\frac14\)
\(=\frac23.\)

（2）“星队”两轮得分之和为 \(X\) 可能为 \(0\)，\(1\)，\(2\)，\(3\)，\(4\)，\(6\).
\(P\left( X=0 \right)=\left( 1-\frac34 \right)^2\cdot \left( 1-\frac23 \right)^2=\frac1{144}\)，
\(P\left( X=1 \right)=2\left[ \frac34\cdot \left( 1-\frac34 \right)\cdot \left( 1-\frac23 \right)^2+\left( 1-\frac34 \right)^2\cdot \frac23\cdot \left( 1-\frac23 \right) \right]=\frac{10}{144}\)，
\(P\left( X=2 \right)
=\frac34\cdot \left( 1-\frac23 \right)\cdot \frac34\cdot \left( 1-\frac23 \right)
+\frac34\cdot \left( 1-\frac23 \right)\cdot \left( 1-\frac34 \right)\cdot \frac23\)
\(\quad +\left( 1-\frac34 \right)\cdot \frac23\cdot \frac34\cdot \left( 1-\frac23 \right)
+\left( 1-\frac34 \right)\cdot \frac23\cdot \left( 1-\frac34 \right)\cdot \frac23
=\frac{25}{144}\)，
\(P\left( X=3 \right)=2\cdot \frac34\cdot \frac23\cdot \left( 1-\frac34 \right)\cdot \left( 1-\frac23 \right)=\frac{12}{144}\)，
\(P\left( X=4 \right)=2\left[ \frac34\cdot \left( 1-\frac34 \right)\cdot \left( \frac23 \right)^2+\frac23\cdot \left( 1-\frac23 \right)\cdot \left( \frac34 \right)^2 \right]=\frac{60}{144}\)，
\(P\left( X=6 \right)=\left( \frac34 \right)^2\cdot \left( \frac23 \right)^2=\frac{36}{144}.\)
所以 \(X\) 的分布列为
\(\begin{array}{c|cccccc}
X & 0 & 1 & 2 & 3 & 4 & 6 \\
\hline
P & \dfrac1{144} & \dfrac{10}{144} & \dfrac{25}{144} & \dfrac{12}{144} & \dfrac{60}{144} & \dfrac{36}{144}
\end{array}\)
故
\(E(X)=0\times \frac1{144}+1\times \frac{10}{144}+2\times \frac{25}{144}+3\times \frac{12}{144}+4\times \frac{60}{144}+6\times \frac{36}{144}=\frac{552}{144}=\frac{23}{6}\).
\end{solution}

\begin{problem}
已知 \( f(x)=a\left( x-\ln x \right)+\dfrac{2x-1}{x^2} \)，\( a\in\R \).
\begin{enumerate}
\item 讨论 \( f(x) \) 的单调性；
\item 当 \( a=1 \) 时，证明 \( f(x)>f'(x)+\dfrac32 \) 对于任意的 \( x\in[1,2] \) 成立.
\end{enumerate}

\end{problem}

\begin{solution}
（1）因为 \(f'(x)=a\left( 1-\frac1x \right)-\frac{2}{x^2}+\frac{2}{x^3}=\frac{\left( x-1 \right)\left( ax^2-2 \right)}{x^3}\)（\(x>0\)）.
当 \( a\le 0 \) 时，\( ax^2-2<0 \) 恒成立，所以
\(f(x)\text{ 在 }\left( 0,1 \right)\text{ 上单调递增，在 }\left( 1,+\infty \right)\text{ 上单调递减. }\)

当 \( 0<a<2 \) 时，\( \sqrt{\dfrac{2}{a}}>1 \)，故
\(f(x)\text{ 在 }\left( 0,1 \right)\text{ 和 }\left( \sqrt{\frac{2}{a}},+\infty \right)\text{ 上单调递增，}\)
\(\text{在 }\left( 1,\sqrt{\frac{2}{a}} \right)\text{ 上单调递减. }\)

当 \( a=2 \) 时，
\(f'(x)=\frac{2\left( x-1 \right)^2\left( x+1 \right)}{x^3}\ge 0\)，
所以 \( f(x) \) 在 \( \left( 0,+\infty \right) \) 上单调递增.

当 \( a>2 \) 时，\( \sqrt{\dfrac{2}{a}}<1 \)，故
\(f(x)\text{ 在 }\left( 0,\sqrt{\frac{2}{a}} \right)\text{ 和 }\left( 1,+\infty \right)\text{ 上单调递增，}\)
\(\text{在 }\left( \sqrt{\frac{2}{a}},1 \right)\text{ 上单调递减. }\)

（2）当 \( a=1 \) 时，
\(f(x)=x-\ln x+\frac{2x-1}{x^2}\).
令
\(F(x)=f(x)-f'(x)=x-\ln x-1+\frac{3}{x}+\frac{1}{x^2}-\frac{2}{x^3}\).
再令
\(g(x)=x-\ln x\)，\(h(x)=-1+\frac{3}{x}+\frac{1}{x^2}-\frac{2}{x^3}\)，
则
\(F(x)=g(x)+h(x)\).
因为
\(g'(x)=1-\frac1x=\frac{x-1}{x}\ge 0\)（\(x\in[1,2]\)），
所以
\(g(x)\ge g(1)=1\).
又
\(h'(x)=\frac{-3x^2-2x+6}{x^4}\).
令 \( \varphi(x)=-3x^2-2x+6 \)，则 \( \varphi(x) \) 在 \( [1,2] \) 上单调递减，且
\(\varphi(1)=1>0\)，\(\varphi(2)=-10<0\).
故存在 \( x_0\in(1,2) \)，使得 \( h(x) \) 在 \( [1,x_0] \) 上单调递增，在 \( [x_0,2] \) 上单调递减. 又
\(h(1)=1\)，\(h(2)=\frac12\).
因此
\(h(x)\ge h(2)=\frac12\).
于是
\(F(x)=g(x)+h(x)>g(1)+h(2)=\frac32\).
即
\(f(x)>f'(x)+\frac32\)，\(x\in[1,2]\).
\end{solution}

\begin{problem}
平面直角坐标系 \( xOy \) 中，椭圆 \( C:\dfrac{x^2}{a^2}+\dfrac{y^2}{b^2}=1 \)（\( a>b>0 \)）的离心率是 \( \dfrac{\sqrt3}{2} \)，抛物线 \( E:x^2=2y \) 的焦点 \(F\) 是 \(C\) 的一个顶点.
\begin{enumerate}
\item 求椭圆 \(C\) 的方程；
\item 设 \(P\) 是 \(E\) 上的动点，且位于第一象限，\(E\) 在点 \(P\) 处的切线 \( l \) 与 \(C\) 交于不同的两点 \(A\)，\(B\)，线段 \(AB\) 的中点为 \(D\)，直线 \(OD\) 与过 \(P\) 且垂直于 \( x \) 轴的直线交于点 \(M\).
\begin{enumerate}
\item 求证：点 \(M\) 在定直线上；
\item 直线 \( l \) 与 \( y \) 轴交于点 \(G\)，记 \( \triangle PFG \) 的面积为 \(S_1\)，\( \triangle PDM \) 的面积为 \(S_2\)，求 \( \dfrac{S_1}{S_2} \) 的最大值及取得最大值时点 \(P\) 的坐标.
\end{enumerate}
\end{enumerate}

\bitmapfigure[max width=5.0cm]{img/2016/shandong_science/q21_fig1.png}

\end{problem}

\begin{solution}
（1）抛物线 \( E:x^2=2y \) 的焦点为
\(F\left( 0,\frac12 \right)\).
因为 \(F\) 是椭圆 \(C\) 的一个顶点，所以
\(b=\frac12\).
又离心率
\(e=\frac{c}{a}=\frac{\sqrt3}{2}\)，
且
\(c^2=a^2-b^2\).
所以
\(\frac{3}{4}a^2=a^2-\frac14\).
解得
\(a=1\).
因此椭圆 \(C\) 的方程为
\(x^2+4y^2=1\).

（2）设
\(P\left( x_0,y_0 \right)\)，
则
\(y_0=\frac{x_0^2}{2}\)，\(x_0>0\).
抛物线 \(E\) 在点 \(P\) 处的切线方程为
\(y-y_0=x_0\left( x-x_0 \right)\)，
即
\(y=x_0x-y_0\).

（1）将 \( y=x_0x-y_0 \) 代入椭圆方程 \( x^2+4y^2=1 \)，得
\(\left( 1+4x_0^2 \right)x^2-8x_0y_0x+4y_0^2-1=0\).
设
\(A\left( x_1,y_1 \right)\)，\(B\left( x_2,y_2 \right)\)，
则由韦达定理
\(x_1+x_2=\frac{8x_0y_0}{1+4x_0^2}\).
故中点 \(D\) 的坐标为
\(D\left( \frac{4x_0y_0}{1+4x_0^2},\ \frac{4x_0^2y_0}{1+4x_0^2}-y_0 \right)\).
由 \( y_0=\dfrac{x_0^2}{2} \) 得
\(D\left( \frac{2x_0^3}{1+4x_0^2},\ -\frac{x_0^2}{2\left( 1+4x_0^2 \right)} \right)\).
于是直线 \(OD\) 的方程为
\(y=-\frac{x}{4x_0}\).
过 \(P\) 且垂直于 \( x \) 轴的直线为 \( x=x_0 \)，与 \(OD\) 交于点 \(M\)，所以
\(M\left( x_0,-\frac14 \right)\).
因此点 \(M\) 在定直线
\(y=-\frac14\)
上.

（2）由切线方程令 \( x=0 \)，得
\(G\left( 0,-y_0 \right)\).
所以
\(S_1=\frac12\cdot \left( y_0+\frac12 \right)\cdot x_0=\frac{x_0\left( 1+x_0^2 \right)}{4}\).
又
\(PM=y_0+\frac14=\frac{1+2x_0^2}{4}\)，
且点 \(D\) 到直线 \( x=x_0 \) 的距离为
\(x_0-\frac{2x_0^3}{1+4x_0^2}=\frac{x_0\left( 1+2x_0^2 \right)}{1+4x_0^2}\).
所以 \(S_2=\frac12\cdot \frac{1+2x_0^2}{4}\cdot \frac{x_0\left( 1+2x_0^2 \right)}{1+4x_0^2}=\frac{x_0\left( 1+2x_0^2 \right)^2}{8\left( 1+4x_0^2 \right)}\).
从而
\(\frac{S_1}{S_2}=\frac{2\left( 1+x_0^2 \right)\left( 1+4x_0^2 \right)}{\left( 1+2x_0^2 \right)^2}\).
令 \(t=1+2x_0^2\)（\(t>1\)），则 \(\frac{S_1}{S_2}=\frac{\left( t+1 \right)\left( 2t-1 \right)}{t^2}=2+\frac1t-\frac1{t^2}\).
设
\(\phi(t)=2+\frac1t-\frac1{t^2}\)，
则
\(\phi'(t)=\frac{2-t}{t^3}\).
所以当 \( t=2 \) 时，\( \phi(t) \) 取得最大值
\(\phi(2)=2+\frac12-\frac14=\frac94\).
此时
\(1+2x_0^2=2\)，
即
\(x_0=\frac{\sqrt2}{2}\).
又
\(y_0=\frac{x_0^2}{2}=\frac14\).
因此
\(\frac{S_1}{S_2}\)
的最大值为
\(\frac94\)，
此时
\(P\left( \frac{\sqrt2}{2},\frac14 \right)\).
\end{solution}
