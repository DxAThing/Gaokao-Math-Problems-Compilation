\section{单选题}

\begin{problem}
已知集合 \(A=\{1,2,3,4\}\)，\(B=\{y\mid y=3x-2,\ x\in A\}\)，则 \(A\cap B=\)（\quad）

\choices
    {\(\{1\}\)}
    {\(\{4\}\)}
    {\(\{1,3\}\)}
    {\(\{1,4\}\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
由 \(x\in A\) 得 \(B=\{1,4,7,10\}\)，所以 \(A\cap B=\{1,4\}\)，故选 D.
\end{solution}

\begin{problem}
设变量 \(x\)，\(y\) 满足约束条件
\(\begin{cases}
x-y+2\ge 0,\\
2x+3y-6\ge 0,\\
3x+2y-9\le 0,
\end{cases}\)
则目标函数 \(z=2x+5y\) 的最小值为（\quad）

\choices
    {\(-4\)}
    {\(6\)}
    {\(10\)}
    {\(17\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
三条边界直线分别为 \(y=x+2\)、\(y=2-\dfrac23x\) 和 \(y=\dfrac{9-3x}{2}\). 由三条约束条件可知，可行域的三个顶点为
\((0,2)\)、\((1,3)\) 和 \((3,0)\)：它们分别是前两条边界直线、第一和第三条边界直线、后两条边界直线的交点. 在三个顶点处分别有
\(z=2\cdot0+5\cdot2=10\)、\(z=2\cdot1+5\cdot3=17\)、\(z=2\cdot3+5\cdot0=6\). 因此 \(z\) 的最小值为 \(6\)，故选 B.
\end{solution}

\begin{problem}
在 \(\triangle ABC\) 中，若 \(AB=\sqrt{13}\)，\(BC=3\)，\(\angle C=120^\circ\)，则 \(AC=\)（\quad）

\choices
    {\(1\)}
    {\(2\)}
    {\(3\)}
    {\(4\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
设 \(AC=x\). 由余弦定理，
\(AB^2=AC^2+BC^2-2AC\cdot BC\cos C=x^2+9-6x\cos120^\circ=x^2+3x+9\). 因为 \(AB=\sqrt{13}\)，所以 \(x^2+3x+9=13\)，即 \((x+4)(x-1)=0\). 由边长 \(x>0\)，得 \(x=1\)，故选 A.
\end{solution}

\begin{problem}
设 \(\{a_n\}\) 是首项为正数的等比数列，公比为 \(q\)，则“\(q<0\)”是“对任意的正整数 \(n\)，\(a_{2n-1}+a_{2n}<0\)”的（\quad）

\choices
    {充要条件}
    {充分而不必要条件}
    {必要而不充分条件}
    {既不充分也不必要条件}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
由于 \(a_1>0\)，有
\(a_{2n-1}+a_{2n}=a_1q^{2n-2}(1+q)\). 若题中的不等式对任意正整数 \(n\) 成立，则 \(q\neq0\)，且 \(q^{2n-2}>0\)，所以必须有 \(1+q<0\)，即 \(q<-1\). 反过来，若 \(q<-1\)，上式对任意正整数 \(n\) 都小于 \(0\)，故该不等式成立. 因此该不等式成立的充要条件是 \(q<-1\). 于是 \(q<0\) 是必要而不充分条件，故选 C.
\end{solution}

\begin{problem}
已知双曲线 \(\dfrac{x^2}{4}-\dfrac{y^2}{b^2}=1\)（\(b>0\)），以原点为圆心，双曲线的实半轴长为半径长的圆与双曲线的两条渐近线相交于 \(A\)，\(B\)，\(C\)，\(D\) 四点，四边形 \(ABCD\) 的面积为 \(2b\)，则双曲线的方程为（\quad）

\choices
    {\(\dfrac{x^2}{4}-\dfrac{3y^2}{4}=1\)}
    {\(\dfrac{x^2}{4}-\dfrac{4y^2}{3}=1\)}
    {\(\dfrac{x^2}{4}-\dfrac{y^2}{4}=1\)}
    {\(\dfrac{x^2}{4}-\dfrac{y^2}{12}=1\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
双曲线的两条渐近线为 \(y=\pm\dfrac b2x\). 令 \(m=\dfrac b2>0\)，则渐近线为 \(y=\pm mx\). 圆的方程为 \(x^2+y^2=4\)，设其与 \(y=mx\) 的交点横坐标的绝对值为 \(u\)，则
\(u^2+m^2u^2=4\)，从而 \(u=\dfrac{2}{\sqrt{1+m^2}}\). 四个交点可写为 \((u,mu)\)、\((u,-mu)\)、\((-u,-mu)\)、\((-u,mu)\)，四边形 \(ABCD\) 的面积为
\(4mu^2=\dfrac{16m}{1+m^2}\). 由题设面积为 \(2b=4m\)，得
\(\dfrac{16m}{1+m^2}=4m\). 因为 \(m>0\)，所以 \(m^2=3\)，即 \(b^2=12\). 故双曲线方程为 \(\dfrac{x^2}{4}-\dfrac{y^2}{12}=1\)，选 D.
\end{solution}

\begin{problem}
已知 \(\triangle ABC\) 是边长为 \(1\) 的等边三角形，点 \(D\)，\(E\) 分别是边 \(AB\)，\(BC\) 的中点，连接 \(DE\) 并延长到点 \(F\)，使得 \(DE=2EF\)，则 \(\overrightarrow{AF}\cdot\overrightarrow{BC}\) 的值为（\quad）

\choices
    {\(-\dfrac85\)}
    {\(\dfrac18\)}
    {\(\dfrac14\)}
    {\(\dfrac{11}{8}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
取 \(A(0,0)\)、\(B(1,0)\)、\(C\left(\dfrac12,\dfrac{\sqrt3}{2}\right)\)，则
\(D\left(\dfrac12,0\right)\)、\(E\left(\dfrac34,\dfrac{\sqrt3}{4}\right)\). 由 \(DE=\dfrac12\)，且 \(DE=2EF\)，可知 \(F\) 在 \(DE\) 的延长线上并满足 \(EF=\dfrac14\)，因此
\(F=E+\dfrac12(E-D)=\left(\dfrac78,\dfrac{3\sqrt3}{8}\right)\). 从而
\(\overrightarrow{AF}=\left(\dfrac78,\dfrac{3\sqrt3}{8}\right)\)，\(\overrightarrow{BC}=\left(-\dfrac12,\dfrac{\sqrt3}{2}\right)\)，故
\[\overrightarrow{AF}\cdot\overrightarrow{BC}=-\dfrac7{16}+\dfrac{9}{16}=\dfrac18\]
因此选 B.
\end{solution}

\begin{problem}
已知函数 \(f(x)=\begin{cases}
x^2+(4a-3)x+3a,&x<0,\\
\log_a(x+1)+1,&x\ge 0,
\end{cases}\)（\(a>0\)，且 \(a\neq1\)）在 \(\R\) 上单调递减，且关于 \(x\) 的方程 \(\lvert f(x)\rvert=2-x\) 恰好有两个不相等的实数解，则 \(a\) 的取值范围是（\quad）

\choices
    {\(\left(0,\dfrac23\right]\)}
    {\(\left[\dfrac23,\dfrac34\right]\)}
    {\(\left[\dfrac13,\dfrac23\right]\cup\left\{\dfrac34\right\}\)}
    {\(\left[\dfrac13,\dfrac23\right)\cup\left\{\dfrac34\right\}\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
先由函数单调递减确定 \(a\) 的范围. 当 \(x<0\) 时，左段函数的导数为 \(2x+4a-3\)，要使其在 \((-\infty,0)\) 上单调递减，需有 \(4a-3\leqslant0\). 当 \(x\geqslant0\) 时，\(\log_a(x+1)+1\) 单调递减要求 \(0<a<1\). 又左段在 \(0\) 左侧的函数值趋近于 \(3a\)，而 \(f(0)=1\)，故需 \(3a\geqslant1\). 因而
\[\frac13\leqslant a\leqslant\frac34\]

因 \(\lvert f(x)\rvert\geqslant0\)，方程有解时必须满足 \(x\leqslant2\).

在 \(x<0\) 时，左段函数值大于等于 \(3a\geqslant1\)，所以方程化为
\[x^2+(4a-3)x+3a=2-x\]
即
\[p_a(x)=x^2+(4a-2)x+3a-2=0\]
当 \(\dfrac13\leqslant a<\dfrac23\) 时，\(p_a(0)<0\)，且 \(p_a(x)\to+\infty\)（\(x\to-\infty\)），并且两根乘积 \(3a-2<0\)，故有且仅有一个负根. 当 \(a=\dfrac23\) 时，\(p_a(x)=x\left(x+\dfrac23\right)\)，负根仍只有一个. 当 \(\dfrac23<a<\dfrac34\) 时，
\[\Delta=(4a-2)^2-4(3a-2)=4(4a-3)(a-1)>0\]
且两根之和 \(2-4a<0\)、两根之积 \(3a-2>0\)，故有两个负根. 当 \(a=\dfrac34\) 时，\(p_a(x)=\left(x+\dfrac12\right)^2\)，只有一个负的不同实根.

再考虑 \(x\geqslant0\). 记 \(s=\dfrac1a-1\)，则 \(f(s)=0\)，且 \(s\in\left[\dfrac13,2\right]\). 在 \([0,s]\) 上方程要求
\[h(x)=\log_a(x+1)+x-1=0\]
因 \(0<a<1\)，有 \(h''(x)=-\dfrac1{(x+1)^2\ln a}>0\)，所以 \(h\) 为凸函数；又 \(h(0)=-1\)，\(h(s)=s-2=\dfrac1a-3\leqslant0\)，故该区间除 \(a=\dfrac13\) 时的端点 \(x=2\) 外无解. 在 \([s,2]\) 上方程要求
\[k(x)=x-3-\log_a(x+1)=0\]
有 \(k'(x)=1-\dfrac1{(x+1)\ln a}>0\)，且 \(k(s)=\dfrac1a-3\leqslant0\)，\(k(2)=-1-\log_a3\geqslant0\)，其中等号仅在 \(a=\dfrac13\) 时成立. 因此 \(x\geqslant0\) 时恰有一个解.

综上，\(x<0\) 时解的个数在 \(\left[\dfrac13,\dfrac23\right]\)、\(\left(\dfrac23,\dfrac34\right)\)、\(\left\{\dfrac34\right\}\) 中分别为 \(1\)、\(2\)、\(1\)，而 \(x\geqslant0\) 时均为 \(1\). 恰有两个不相等实数解当且仅当
\[a\in\left[\dfrac13,\dfrac23\right]\cup\left\{\dfrac34\right\}\]
故选 C.
\end{solution}

\section{填空题}

\begin{problem}
已知 \(a\)，\(b\in\R\)，\(\i\) 是虚数单位，若 \((1+\i)(1-b\i)=a\)，则 \(\dfrac{a}{b}\) 的值为\fillinblank{}.
\end{problem}

\begin{answer}
2
\end{answer}

\begin{solution}
展开左边得
\((1+\i)(1-b\i)=1-b\i+\i-b\i^2=(1+b)+(1-b)\i\). 由于结果为实数 \(a\)，虚部必须为零，所以 \(1-b=0\)，即 \(b=1\)，且 \(a=1+b=2\). 因此 \(\dfrac ab=2\).
\end{solution}

\begin{problem}
\(\left(x^2-\dfrac1x\right)^8\) 的展开式中 \(x^7\) 的系数为\fillinblank{}（用数字作答）.
\end{problem}

\begin{answer}
\(-56\)
\end{answer}

\begin{solution}
展开式的通项为
\[\mathrm C_8^k(x^2)^{8-k}\left(-\dfrac1x\right)^k=(-1)^k\mathrm C_8^k x^{16-3k}\]
要得到 \(x^7\)，需满足 \(16-3k=7\)，解得 \(k=3\). 因此 \(x^7\) 的系数为
\((-1)^3\mathrm C_8^3=-56\).
\end{solution}

\begin{problem}
如图，\(AB\) 是圆的直径，弦 \(CD\) 与 \(AB\) 相交于点 \(E\)，\(BE=2AE=2\)，\(BD=ED\)，则线段 \(CE\) 的长为\fillinblank{}.

\begin{problem}
已知 \(f(x)\) 是定义在 \(\R\) 上的偶函数，且在区间 \((-\infty,0)\) 上单调递增. 若实数 \(a\) 满足 \(f\left(2^{\lvert a-1\rvert}\right)>f\left(-\sqrt2\right)\)，则 \(a\) 的取值范围是\fillinblank{}.
\end{problem}

\begin{answer}
\(\dfrac12<a<\dfrac32\)
\end{answer}

\begin{solution}
因为 \(f\) 是偶函数，且在 \((-\infty,0)\) 上单调递增，所以对正数 \(u\)，有
\(f(u)=f(-u)\)，并且 \(f(u)>f(\sqrt2)\) 当且仅当 \(-u>-\sqrt2\)，即 \(u<\sqrt2\). 题设不等式等价于
\[2^{\lvert a-1\rvert}<\sqrt2=2^{1/2}\]
由于底数 \(2>1\)，得到 \(\lvert a-1\rvert<\dfrac12\)，所以
\[\dfrac12<a<\dfrac32\]
\end{solution}

\begin{problem}
设抛物线
\(\begin{cases}
x=2pt^2,\\
y=2pt,
\end{cases}\)（\(t\) 为参数，\(p>0\)）
的焦点为 \(F\)，准线为 \(l\). 过抛物线上一点 \(A\) 作 \(l\) 的垂线，垂足为 \(B\). 设 \(C\left(\dfrac72p,0\right)\)，\(AF\) 与 \(BC\) 相交于点 \(E\). 若 \(\lvert CF\rvert=2\lvert AF\rvert\)，且 \(\triangle ACE\) 的面积为 \(3\sqrt2\)，则 \(p\) 的值为\fillinblank{}.
\end{problem}

\begin{answer}
\(\sqrt{6}\)
\end{answer}

\begin{solution}
由参数方程 \(x=2pt^2\)、\(y=2pt\) 可知抛物线的焦点为 \(F\left(\dfrac p2,0\right)\)，准线为 \(l:x=-\dfrac p2\). 设点 \(A\) 对应的参数为 \(t\)，则
\(A(2pt^2,2pt)\)，\(B\left(-\dfrac p2,2pt\right)\)，\(C\left(\dfrac{7p}{2},0\right)\).
又
\[CF=3p\]
以及
\[AF^2=p^2\left(2t^2-\dfrac12\right)^2+(2pt)^2=p^2\left(2t^2+\dfrac12\right)^2\]
因 \(p>0\)，得 \(AF=p\left(2t^2+\dfrac12\right)\). 由 \(CF=2AF\)，有
\[3p=2p\left(2t^2+\dfrac12\right)\]
从而 \(t^2=\dfrac12\).

取 \(p=1\) 计算交点的相对坐标，再整体乘以 \(p\). 此时 \(A(1,2t)\)、\(B\left(-\dfrac12,2t\right)\)、\(C\left(\dfrac72,0\right)\)、\(F\left(\dfrac12,0\right)\). 设 \(E\) 在 \(AF\) 上满足 \(E=A+s(F-A)\)，在 \(BC\) 上满足 \(E=B+r(C-B)\). 比较纵坐标得 \(s=r\)，再比较横坐标得
\(1-\dfrac s2=-\dfrac12+4s\)，\(s=\dfrac13\).
所以 \(E\left(\dfrac56,\dfrac{4t}{3}\right)\). 因此
\(\overrightarrow{AC}=\left(\dfrac52,-2t\right)\)，\(\overrightarrow{AE}=\left(-\dfrac16,-\dfrac{2t}{3}\right)\).
三角形 \(ACE\) 的面积为
\[S_{\triangle ACE}=\dfrac12\left|\overrightarrow{AC}\times\overrightarrow{AE}\right|=|t|p^2=\dfrac{p^2}{\sqrt2}\]
由题设 \(S_{\triangle ACE}=3\sqrt2\)，得 \(p^2=6\). 因 \(p>0\)，所以 \(p=\sqrt6\).
\end{solution}

\begin{problem}
已知函数 \(f(x)=4\tan x\sin\left(\dfrac\pi2-x\right)\cos\left(x-\dfrac\pi3\right)-\sqrt3\).

\begin{enumerate}
\item 求 \(f(x)\) 的定义域与最小正周期；
\item 讨论 \(f(x)\) 在区间 \(\left[-\dfrac\pi4,\dfrac\pi4\right]\) 上的单调性.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由 \(\tan x\) 的定义域可知，函数的定义域为
\(\mathcal D=\R\mathbin{\backslash}\left\{\dfrac\pi2+k\pi\mid k\in\Z\right\}\).
在此定义域内，\(\sin\left(\dfrac\pi2-x\right)=\cos x\)，所以
\[
f(x)=4\sin x\cos\left(x-\dfrac\pi3\right)-\sqrt3
 =2\sin x\cos x+2\sqrt3\sin^2x-\sqrt3
\]
\[
f(x)=\sin 2x-\sqrt3\cos 2x=2\sin\left(2x-\dfrac\pi3\right)
\]
因此 \(\pi\) 是函数的一个正周期. 由于 \(2\sin\left(2x-\dfrac\pi3\right)\) 的最小正周期为 \(\pi\)，且定义域也以 \(\pi\) 为周期，故 \(f(x)\) 的最小正周期为 \(\pi\).

\item 在 \(\left[-\dfrac\pi4,\dfrac\pi4\right]\) 上，\(\cos x\neq0\)，故该区间包含在函数的定义域内. 由上式得
\(f'(x)=4\cos\left(2x-\dfrac\pi3\right)\).
当 \(x\in\left[-\dfrac\pi4,\dfrac\pi4\right]\) 时，
\(2x-\dfrac\pi3\in\left[-\dfrac{5\pi}{6},\dfrac\pi6\right]\). 当
\(x\in\left[-\dfrac\pi4,-\dfrac\pi{12}\right)\) 时，
\(2x-\dfrac\pi3\in\left[-\dfrac{5\pi}{6},-\dfrac\pi2\right)\)，故
\(f'(x)<0\)；当 \(x\in\left(-\dfrac\pi{12},\dfrac\pi4\right]\) 时，
\(2x-\dfrac\pi3\in\left(-\dfrac\pi2,\dfrac\pi6\right]\)，故 \(f'(x)>0\).
因此，\(f(x)\) 在 \(\left[-\dfrac\pi4,-\dfrac\pi{12}\right]\) 上单调递减，在
\(\left[-\dfrac\pi{12},\dfrac\pi4\right]\) 上单调递增.
\end{enumerate}
\end{solution}

\begin{problem}
某小组共 \(10\) 人，利用假期参加义工活动，已知参加义工活动次数为 \(1\)，\(2\)，\(3\) 的人数分别为 \(3\)，\(3\)，\(4\). 现从这 \(10\) 人中随机选出 \(2\) 人作为该组代表参加座谈会.

\begin{enumerate}
\item 设 \(A\) 为事件“选出的 \(2\) 人参加义工活动次数之和为 \(4\)”，求事件 \(A\) 发生的概率；
\item 设 \(X\) 为选出的 \(2\) 人参加义工活动次数之差的绝对值，求随机变量 \(X\) 的分布列和数学期望.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 从 \(10\) 人中选出 \(2\) 人的选法共有 \(\mathrm C_{10}^2=45\) 种.
事件 \(A\) 包含两种情形：选出的两人次数分别为 \(1\)，\(3\)，有 \(3\cdot4=12\) 种；选出的两人次数都为 \(2\)，有 \(\mathrm C_3^2=3\) 种. 因此
\[
P(A)=\frac{3\cdot4+\mathrm C_3^2}{\mathrm C_{10}^2}=\frac{15}{45}=\frac13
\]

\item 随机变量 \(X\) 的可能取值为 \(0\)，\(1\)，\(2\). 其中
\[
P(X=0)=\frac{\mathrm C_3^2+\mathrm C_3^2+\mathrm C_4^2}{\mathrm C_{10}^2}=\frac{12}{45}=\frac4{15}
\]
\[
P(X=1)=\frac{3\cdot3+3\cdot4}{\mathrm C_{10}^2}=\frac{21}{45}=\frac7{15}
\]
\[
P(X=2)=\frac{3\cdot4}{\mathrm C_{10}^2}=\frac{12}{45}=\frac4{15}
\]
故分布列为
\begin{center}
\begin{tabular}{c|ccc}
\hline
\(X\) & \(0\) & \(1\) & \(2\) \\
\hline
\(P\) & \(\dfrac4{15}\) & \(\dfrac7{15}\) & \(\dfrac4{15}\) \\
\hline
\end{tabular}
\end{center}
从而
\[
E(X)=0\cdot\frac4{15}+1\cdot\frac7{15}+2\cdot\frac4{15}=1
\]
\end{enumerate}
\end{solution}

\begin{problem}
如图，正方形 \(ABCD\) 的中心为 \(O\)，四边形 \(OBEF\) 为矩形，平面 \(OBEF\perp\) 平面 \(ABCD\)，点 \(G\) 为 \(AB\) 的中点，\(AB=BE=2\).

\begin{enumerate}
\item 求证：\(EG\parallel\) 平面 \(ADF\)；
\item 求二面角 \(O-EF-C\) 的正弦值；
\item 设 \(H\) 为线段 \(AF\) 上的点，且 \(AH=\dfrac23HF\)，求直线 \(BH\) 和平面 \(CEF\) 所成角的正弦值.
\end{enumerate}

\bitmapfigure[max width=0.55\linewidth]{img/2016/tianjin_science/q17_fig1.png}
\end{problem}

\begin{solution}
建立如下一直角坐标系：令正方形 \(ABCD\) 在 \(z=0\) 平面内，取
\(A(1,-1,0)\)，\(B(-1,-1,0)\)，\(C(-1,1,0)\)，\(D(1,1,0)\)，\(O(0,0,0)\).
由四边形 \(OBEF\) 为矩形，且平面 \(OBEF\) 垂直于平面 \(ABCD\)，可取
\(E(-1,-1,2)\)，\(F(0,0,2)\). 又 \(G(0,-1,0)\).

\begin{enumerate}
\item 有
\(\overrightarrow{EG}=(1,0,-2)\)，\(\overrightarrow{AD}=(0,2,0)\)，\(\overrightarrow{AF}=(-1,1,2)\).
于是
\(\overrightarrow{AD}\times\overrightarrow{AF}=(4,0,2)\)，\(\overrightarrow{EG}\cdot\left(\overrightarrow{AD}\times\overrightarrow{AF}\right)=4-4=0\).
因此 \(EG\) 的方向向量平行于平面 \(ADF\). 又 \(E\) 不在平面 \(ADF\) 上，故 \(EG\parallel\) 平面 \(ADF\).

\item 取平面 \(OEF\) 和平面 \(CEF\) 的法向量分别为
\(\bs{n}_1=(1,-1,0)\)，\(\bs{n}_2=(1,-1,-1)\).
设二面角 \(O-EF-C\) 的平面角为 \(\theta\)，则
\[
\cos\theta=\frac{|\bs{n}_1\cdot\bs{n}_2|}{|\bs{n}_1||\bs{n}_2|}
 =\frac2{\sqrt2\sqrt3}=\frac{\sqrt6}{3}
\]
所以
\[
\sin\theta=\sqrt{1-\frac{2}{3}}=\frac{\sqrt3}{3}
\]

\item 由 \(AH=\dfrac23HF\)，得 \(AH:HF=2:3\)，从而
\[
H=\frac{3A+2F}{5}=\left(\frac35,-\frac35,\frac45\right)
\]
因此直线 \(BH\) 的一个方向向量为
\[
\overrightarrow{BH}=\left(\frac85,\frac25,\frac45\right)
\]
可取 \(\bs{d}=(4,1,2)\). 设直线 \(BH\) 与平面 \(CEF\) 所成的角为 \(\varphi\)，则
\[
\sin\varphi=\frac{|\bs{d}\cdot\bs{n}_2|}{|\bs{d}||\bs{n}_2|}
 =\frac{|4-1-2|}{\sqrt{21}\sqrt3}=\frac1{\sqrt{63}}=\frac{\sqrt7}{21}
\]
\end{enumerate}
\end{solution}

\begin{problem}
已知 \(\{a_n\}\) 是各项均为正数的等差数列，公差为 \(d\)，对任意的 \(n\in\N^*\)，\(b_n\) 是 \(a_n\) 和 \(a_{n+1}\) 的等比中项.

\begin{enumerate}
\item 设 \(c_n=b_{n+1}^2-b_n^2\)（\(n\in\N^*\)），求证：数列 \(\{c_n\}\) 是等差数列；
\item 设 \(a_1=d\)，\(T_n=\displaystyle\sum_{k=1}^{2n}(-1)^k b_k^2\)（\(n\in\N^*\)），求证：\(\displaystyle\sum_{k=1}^n\dfrac1{T_k}<\dfrac1{2d^2}\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由 \(b_n\) 是 \(a_n\) 和 \(a_{n+1}\) 的等比中项，且各项均为正数，得
\(b_n^2=a_na_{n+1}\). 所以
\[
c_n=b_{n+1}^2-b_n^2=a_{n+1}a_{n+2}-a_na_{n+1}
 =a_{n+1}(a_{n+2}-a_n)=2d a_{n+1}
\]
于是
\[
c_{n+1}-c_n=2d(a_{n+2}-a_{n+1})=2d^2
\]
为常数，故数列 \(\{c_n\}\) 是等差数列.

\item 因为 \(a_1=d\)，且 \(\{a_n\}\) 为等差数列，所以 \(a_n=nd\). 从而
\(b_n^2=a_na_{n+1}=n(n+1)d^2\). 将相邻两项配对，得
\[
\begin{gathered}
T_n=d^2\sum_{k=1}^{2n}(-1)^k k(k+1)
 =d^2\sum_{j=1}^n\left[-(2j-1)(2j)+(2j)(2j+1)\right]\\
 =d^2\sum_{j=1}^n4j=2d^2n(n+1).
\end{gathered}
\]
因此
\[
\sum_{k=1}^n\frac1{T_k}=\frac1{2d^2}\sum_{k=1}^n\frac1{k(k+1)}
 =\frac1{2d^2}\sum_{k=1}^n\left(\frac1k-\frac1{k+1}\right)
 =\frac1{2d^2}\left(1-\frac1{n+1}\right)<\frac1{2d^2}
\]
\end{enumerate}
\end{solution}

\begin{problem}
设椭圆 \(\dfrac{x^2}{a^2}+\dfrac{y^2}{3}=1\)（\(a>\sqrt3\)）的右焦点为 \(F\)，右顶点为 \(A\)，已知 \(\dfrac1{\lvert OF\rvert}+\dfrac1{\lvert OA\rvert}=\dfrac{3e}{\lvert FA\rvert}\)，其中 \(O\) 为原点，\(e\) 为椭圆的离心率.

\begin{enumerate}
\item 求椭圆的方程；
\item 设过点 \(A\) 的直线 \(l\) 与椭圆交于点 \(B\)（\(B\) 不在 \(x\) 轴上），垂直于 \(l\) 的直线与 \(l\) 交于点 \(M\)，与 \(y\) 轴交于点 \(H\). 若 \(BF\perp HF\)，且 \(\angle MOA\le\angle MAO\)，求直线 \(l\) 的斜率的取值范围.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设椭圆的半焦距为 \(c\)，则 \(c=\lvert OF\rvert\)，\(a=\lvert OA\rvert\)，\(\lvert FA\rvert=a-c\)，且 \(e=\dfrac ca\). 由题设
\[
\frac1c+\frac1a=\frac{3c}{a(a-c)}
\]
两边同乘以 \(ac(a-c)\)，得
\[
(a+c)(a-c)=3c^2
\]
即 \(a^2=4c^2\). 又由椭圆的性质 \(a^2=c^2+3\)，可得
\(4c^2=c^2+3\)，\(c=1\)，\(a=2\).
所以椭圆的方程为
\[
\frac{x^2}{4}+\frac{y^2}{3}=1
\]

\item 设直线 \(l\) 的斜率为 \(k\). 由 \(B\) 不在 \(x\) 轴上，知 \(k\neq0\)，且直线 \(l\) 的方程为 \(y=k(x-2)\).
将其代入椭圆方程，得
\[
3x^2+4k^2(x-2)^2=12
\]
其中一个交点的横坐标为 \(2\)，由韦达定理，点 \(B\) 的坐标为
\[
B\left(\frac{8k^2-6}{4k^2+3},-\frac{12k}{4k^2+3}\right)
\]
设 \(H(0,h)\). 由 \(F(1,0)\) 以及 \(BF\perp HF\)，得
\[
\left(\frac{4k^2-9}{4k^2+3},-\frac{12k}{4k^2+3}\right)
\cdot(-1,h)=0
\]
从而
\[
h=\frac{9-4k^2}{12k}
\]
过 \(H\) 且垂直于 \(l\) 的直线斜率为 \(-\dfrac1k\)，其方程为
\(y=h-\dfrac{x}{k}\). 与 \(l\) 联立，得到
\[
M\left(\frac{20k^2+9}{12(k^2+1)},-\frac{k(4k^2+15)}{12(k^2+1)}\right)
\]
因为 \(x_M>0\)，且
\[
\overrightarrow{AM}=-\frac{4k^2+15}{12(k^2+1)}(1,k)
\]
所以 \(\angle MOA\) 与 \(\angle MAO\) 都是锐角，并且
\(\tan\angle MOA=\frac{|k|(4k^2+15)}{20k^2+9}\)，\(\tan\angle MAO=|k|\).
由 \(\angle MOA\leqslant\angle MAO\)，得
\[
\frac{|k|(4k^2+15)}{20k^2+9}\leqslant|k|
\]
由于 \(k\neq0\)，所以 \(4k^2+15\leqslant20k^2+9\)，即 \(k^2\geqslant\dfrac38\).
因此直线 \(l\) 的斜率的取值范围为
\[
\left(-\infty,-\frac{\sqrt6}{4}\right]\cup\left[\frac{\sqrt6}{4},+\infty\right)
\]
\end{enumerate}
\end{solution}

\begin{problem}
设函数 \(f(x)=(x-1)^3-ax-b\)（\(x\in\R\)，其中 \(a\)，\(b\in\R\)）.

\begin{enumerate}
\item 求 \(f(x)\) 的单调区间；
\item 若 \(f(x)\) 存在极值点 \(x_0\)，且 \(f(x_1)=f(x_0)\)，其中 \(x_1\neq x_0\)，求证：\(x_1+2x_0=3\)；
\item 设 \(a>0\)，函数 \(g(x)=\lvert f(x)\rvert\)，求证：\(g(x)\) 在区间 \([0,2]\) 上的最大值不小于 \(\dfrac14\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item
\[
f'(x)=3(x-1)^2-a
\]
当 \(a\leqslant0\) 时，\(f'(x)\geqslant0\)，且函数在整个实数范围内单调递增.
当 \(a>0\) 时，令 \(f'(x)=0\)，得
\(x=1-\sqrt{\dfrac a3}\) 或 \(x=1+\sqrt{\dfrac a3}\). 于是
\[
\begin{cases}
f'(x)>0,&x<1-\sqrt{\dfrac a3}\ \text{或}\ x>1+\sqrt{\dfrac a3},\\
f'(x)<0,&1-\sqrt{\dfrac a3}<x<1+\sqrt{\dfrac a3}.
\end{cases}
\]
故当 \(a>0\) 时，\(f(x)\) 在 \(\left(-\infty,1-\sqrt{\dfrac a3}\right]\) 和 \(\left[1+\sqrt{\dfrac a3},+\infty\right)\) 上单调递增，在 \(\left[1-\sqrt{\dfrac a3},1+\sqrt{\dfrac a3}\right]\) 上单调递减.

\item 因为 \(x_0\) 是极值点，所以 \(a>0\)，且
\(f'(x_0)=0\)，即 \(a=3(x_0-1)^2\). 令 \(u=x_1-1\)，\(v=x_0-1\)，由 \(f(x_1)=f(x_0)\) 得
\[
0=(u^3-v^3)-a(u-v)=(u-v)(u^2+uv+v^2-a)
\]
又因 \(x_1\neq x_0\)，即 \(u\neq v\)，所以
\[
u^2+uv+v^2-a=0
\]
代入 \(a=3v^2\)，得
\[
u^2+uv-2v^2=(u-v)(u+2v)=0
\]
由于 \(u\neq v\)，故 \(u+2v=0\)，即
\(x_1-1+2(x_0-1)=0\)，\(x_1+2x_0=3\).

\item 令 \(t=x-1\)，则当 \(x\in[0,2]\) 时，\(t\in[-1,1]\)，且
\[
f(t+1)=t^3-at-(a+b)
\]
记 \(p(t)=t^3-at\). 常数 \(-(a+b)\) 不影响函数值的最大差，因此只需证明 \(p(t)\) 在 \([-1,1]\) 上的最大值与最小值之差不小于 \(\dfrac12\).

当 \(0<a\leqslant\dfrac34\) 时，
\[
p(1)-p(-1)=2(1-a)\geqslant\frac12
\]
当 \(a\geqslant\dfrac54\) 时，
\[
p(-1)-p(1)=2(a-1)\geqslant\frac12
\]
当 \(\dfrac34<a<\dfrac54\) 时，令 \(r=\sqrt{\dfrac a3}\)，则 \(r\in(0,1)\)，且 \(a=3r^2\).
\(p(-r)=2r^3\)，\(p(r)=-2r^3\).
所以
\[
p(-r)-p(r)=4r^3=4\left(\frac a3\right)^{\frac32}>\frac12
\]
综上，\(f(x)\) 在 \([0,2]\) 上的最大值与最小值之差至少为 \(\dfrac12\). 设
\(M=\max_{x\in[0,2]}g(x)=\max_{x\in[0,2]}|f(x)|\)，则对任意 \(x_1\)，\(x_2\in[0,2]\)，有
\[
|f(x_1)-f(x_2)|\leqslant|f(x_1)|+|f(x_2)|\leqslant2M
\]
取使函数值达到最大差的两点，得 \(\dfrac12\leqslant2M\)，从而
\[
\max_{x\in[0,2]}g(x)=M\geqslant\frac14
\]
\end{enumerate}
\end{solution}
