\begin{problem}
已知全集 \(U=\{1,2,3,4,5,6\}\)，集合 \(P=\{1,3,5\}\)，\(Q=\{1,2,4\}\)，则 \((\complement_U P)\cup Q=\)（\quad）

\choices
    {\(\{1\}\)}
    {\(\{3,5\}\)}
    {\(\{1,2,4,6\}\)}
    {\(\{1,2,3,4,5\}\)}
\end{problem}

\choices
    {\(m\parallel l\)}
    {\(m\parallel n\)}
    {\(n\perp l\)}
    {\(m\perp n\)}
\end{problem}

\begin{problem}

\begin{problem}

\begin{problem}
在 \(\triangle ABC\) 中，内角 \(A\)，\(B\)，\(C\) 所对的边分别为 \(a\)，\(b\)，\(c\)，已知 \(b+c=2a\cos B\).

\begin{enumerate}
\item 证明：\(A=2B\)；
\item 若 \(\cos B=\dfrac23\)，求 \(\cos C\) 的值.
\end{enumerate}
\end{problem}

\begin{problem}
设数列 \(\{a_n\}\) 的前 \(n\) 项和为 \(S_n\)，已知 \(S_2=4\)，\(a_{n+1}=2S_n+1\)，\(n\in\N^*\).

\begin{enumerate}
\item 求通项公式 \(a_n\)；
\item 求数列 \(\{|a_n-n-2|\}\) 的前 \(n\) 项和.
\end{enumerate}
\end{problem}
