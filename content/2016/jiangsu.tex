\begin{problem}

\begin{problem}
已知函数 \(f(x)=a^x+b^x\)（\(a>0\)，\(b>0\)，\(a\ne1\)，\(b\ne1\)）.

\begin{enumerate}
\item 设 \(a=2\)，\(b=\dfrac12\).

\begin{enumerate}
\item 求方程 \(f(x)=2\) 的根；
\item 若对于任意 \(x\in\R\)，不等式 \(f(2x)\ge mf(x)-6\) 恒成立，求实数 \(m\) 的最大值；
\end{enumerate}

\item 若 \(0<a<1\)，\(b>1\)，函数 \(g(x)=f(x)-2\) 有且只有 \(1\) 个零点，求 \(ab\) 的值.
\end{enumerate}
\end{problem}

\begin{problem}
记 \(U=\{1,2,\cdots,100\}\). 对数列 \(\{a_n\}\)（\(n\in\N^*\)）和 \(U\) 的子集 \(T\)，若 \(T=\varnothing\)，定义 \(S_T=0\)；若 \(T=\{t_1,t_2,\cdots,t_k\}\)，定义 \(S_T=a_{t_1}+a_{t_2}+\cdots+a_{t_k}\). 例如：\(T=\{1,3,66\}\) 时，\(S_T=a_1+a_3+a_{66}\). 现设 \(\{a_n\}\)（\(n\in\N^*\)）是公比为 \(3\) 的等比数列，且当 \(T=\{2,4\}\) 时，\(S_T=30\).

\begin{enumerate}
\item 求 \(\{a_n\}\) 的通项公式；
\item 对任意正整数 \(k\)（\(1\le k\le100\)），若 \(T\subseteq\{1,2,\cdots,k\}\)，求证：\(S_T<a_{k+1}\)；
\item 设 \(C\subseteq U\)，\(D\subseteq U\)，\(S_C\ge S_D\)，求证：\(S_C+S_{C\cap D}\ge2S_D\).
\end{enumerate}
\end{problem}
