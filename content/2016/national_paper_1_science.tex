\section{单选题}

\begin{problem}
设集合 \(A=\{x\mid x^2-4x+3<0\}\)，\(B=\{x\mid 2x-3>0\}\)，则 \(A\cap B=\)（\quad）

\choices
    {\(\left(-3,-\dfrac32\right)\)}
    {\(\left(-3,\dfrac32\right)\)}
    {\(\left(1,\dfrac32\right)\)}
    {\(\left(\dfrac32,3\right)\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
由 \(x^2-4x+3=(x-1)(x-3)\)，得 \(A=(1,3)\). 又由 \(2x-3>0\)，得 \(B=(\dfrac32,+\infty)\). 因此
\[
A\cap B=\left(\dfrac32,3\right)
\]
故选 D.
\end{solution}



\begin{problem}
设 \((1+\i)x=1+y\i\)，其中 \(x\)，\(y\) 是实数，则 \(|x+y\i|=\)（\quad）

\choices
    {\(1\)}
    {\(\sqrt2\)}
    {\(\sqrt3\)}
    {\(2\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
由复数相等，\((1+\i)x=x+x\i=1+y\i\)，所以 \(x=1\)，\(y=1\). 因此
\[
|x+y\i|=|1+\i|=\sqrt{1^2+1^2}=\sqrt2
\]
故选 B.
\end{solution}



\begin{problem}
已知等差数列 \(\{a_n\}\) 前 \(9\) 项的和为 \(27\)，\(a_{10}=8\)，则 \(a_{100}=\)（\quad）

\choices
    {\(100\)}
    {\(99\)}
    {\(98\)}
    {\(97\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
设等差数列的公差为 \(d\). 由等差数列中项的性质，
\[
S_9=9a_5=27
\]
故 \(a_5=3\). 又 \(a_{10}=a_5+5d=8\)，所以 \(d=1\). 从而
\[
a_{100}=a_5+95d=3+95=98
\]
故选 C.
\end{solution}



\begin{problem}
某公司的班车在 \(7{:}30\)，\(8{:}00\)，\(8{:}30\) 发车，小明在 \(7{:}50\) 至 \(8{:}30\) 之间到达发车站乘坐班车，且到达发车站的时刻是随机的，则他等车时间不超过 \(10\) 分钟的概率是（\quad）

\choices
    {\(\dfrac13\)}
    {\(\dfrac12\)}
    {\(\dfrac23\)}
    {\(\dfrac34\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
把 \(7{:}50\) 至 \(8{:}30\) 看作长度为 \(40\) 分钟的等可能时间区间. 到达时间在 \(7{:}50\) 至 \(8{:}00\) 时可乘 \(8{:}00\) 的班车，到达时间在 \(8{:}20\) 至 \(8{:}30\) 时可乘 \(8{:}30\) 的班车；这两段有利时间总长为 \(20\) 分钟. 因此
\[
\frac{20}{40}=\frac12
\]
故选 B.
\end{solution}



\begin{problem}
已知方程 \(\dfrac{x^2}{m^2+n}-\dfrac{y^2}{3m^2-n}=1\) 表示双曲线，且该双曲线两焦点间的距离为 \(4\)，则 \(n\) 的取值范围是（\quad）

\choices
    {\((-1,3)\)}
    {\((-1,\sqrt3)\)}
    {\((0,3)\)}
    {\((0,\sqrt3)\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
要使方程表示双曲线，必须有
\(m^2+n>0\)，\(3m^2-n>0\).
此时双曲线的焦半距满足
\[
c^2=(m^2+n)+(3m^2-n)=4m^2
\]
由两焦点间的距离为 \(4\)，得 \(2c=4\)，即 \(c=2\)，所以 \(m^2=1\). 代回两个正性条件，得到
\(1+n>0\)，\(3-n>0\)
即 \(-1<n<3\). 故选 A.
\end{solution}



\begin{problem}
如图，某几何体的三视图是三个半径相等的圆及每个圆中两条相互垂直的半径. 若该几何体的体积是 \(28\pi/3\)，则它的表面积是（\quad）

\bitmapfigure[max width=6.0cm]{img/2016/national_paper_1_science/q6_fig1.png}

\choices
    {\(17\pi\)}
    {\(18\pi\)}
    {\(20\pi\)}
    {\(28\pi\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
由三视图可知，该几何体是半径为 \(r\) 的球去掉一个八分之一球后形成的部分；三个互相垂直的半径对应三个互相垂直的截面. 因此其体积为球体积的 \(\dfrac78\)，所以
\[
\frac78\cdot\frac43\pi r^3=\frac{28\pi}{3}
\]
解得 \(r=2\). 几何体保留的球面面积为
\[
\frac78\cdot4\pi r^2=14\pi
\]
去掉八分之一球后新增三个互相垂直的四分之一圆面，每个面积为 \(\dfrac14\pi r^2=\pi\). 因此表面积为
\[
14\pi+3\pi=17\pi
\]
故选 A.
\end{solution}



\begin{problem}
函数 \(y=2x^2-e^{|x|}\) 在 \([-2,2]\) 的图象大致为（\quad）

\choices
    {\choicebitmap[width=0.15\paperwidth]{img/2016/national_paper_1_science/q7_optA.png}}
    {\choicebitmap[width=0.15\paperwidth]{img/2016/national_paper_1_science/q7_optB.png}}
    {\choicebitmap[width=0.15\paperwidth]{img/2016/national_paper_1_science/q7_optC.png}}
    {\choicebitmap[width=0.15\paperwidth]{img/2016/national_paper_1_science/q7_optD.png}}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
函数为偶函数. 对 \(x\geqslant0\)，有 \(f(x)=2x^2-e^x\).
\(f(0)=-1<0\)，\(f(2)=8-e^2>0\).
又 \(f'(x)=4x-e^x\). 令 \(g(x)=4x-e^x\)，则 \(g'(x)=4-e^x\). 因此 \(g\) 在 \([0,\ln4]\) 上递增，在 \([\ln4,2]\) 上递减；又 \(g(0)=-1<0\)、\(g(\ln4)=4\ln4-4>0\)、\(g(2)=8-e^2>0\)，所以 \(g\) 在 \([0,2]\) 内只有一个零点，函数在正半轴先减后增. 结合偶性、\(f(0)=-1<0\) 和 \(f(2)=8-e^2>0\)，只有选项 D 的图象符合，故选 D.
\end{solution}



\begin{problem}
若 \(a>b>1\)，\(0<c<1\)，则（\quad）

\choices
    {\(a^c<b^c\)}
    {\(ab^c<ba^c\)}
    {\(a\log_b c<b\log_a c\)}
    {\(\log_a c<\log_b c\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
因为 \(0<c<1\)，所以 \(x^c\) 在正数范围内单调递增，故 \(a^c>b^c\)，选项 A 错误. 又因 \(c-1<0\)，有 \(a^{c-1}<b^{c-1}\)，即 \(ab^c>ba^c\)，选项 B 错误.

对选项 C，由 \(\ln c<0\)、\(\ln a>\ln b>0\)，
\[
a\log_b c<b\log_a c
\Longleftrightarrow
\frac{a\ln c}{\ln b}<\frac{b\ln c}{\ln a}
\Longleftrightarrow a\ln a>b\ln b
\]
函数 \(x\ln x\) 在 \(x>1\) 上单调递增，而 \(a>b>1\)，故 C 正确. 对选项 D，由 \(\ln c<0\) 且 \(\ln a>\ln b>0\)，有 \(\log_a c>\log_b c\)，故 D 错误. 因此选 C.
\end{solution}



\begin{problem}
执行下面的程序图，如果输入的 \(x=0\)，\(y=1\)，\(n=1\)，则输出 \(x\)，\(y\) 的值满足（\quad）

\bitmapfigure[max width=0.35\linewidth]{img/2016/national_paper_1_science/q9_fig1.png}

\choices
    {\(y=2x\)}
    {\(y=3x\)}
    {\(y=4x\)}
    {\(y=5x\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
按程序框图逐次计算. 初始 \(x=0\)，\(y=1\)，\(n=1\). 第一次循环后
\(x=0+\frac{1-1}{2}=0\)，\(y=1\cdot1=1\)，\(x^2+y^2=1<36\)
故 \(n\) 变为 \(2\). 第二次循环后
\(x=0+\frac{2-1}{2}=\frac12\)，\(y=2\cdot2=2\)，\(x^2+y^2=\frac{17}{4}<36\)
故 \(n\) 变为 \(3\). 第三次循环后
\(x=\frac12+\frac{3-1}{2}=\frac32\)，\(y=3\cdot2=6\)，\(x^2+y^2=\frac94+36>36\).
程序停止，输出满足
\[
\frac{y}{x}=\frac{6}{3/2}=4
\]
即 \(y=4x\)，故选 C.
\end{solution}



\begin{problem}
以抛物线 \(C\) 的顶点为圆心的圆交 \(C\) 于 \(A\)，\(B\) 两点，交 \(C\) 的准线于 \(D\)，\(E\) 两点. 已知 \(|AB|=4\sqrt2\)，\(|DE|=2\sqrt5\)，则 \(C\) 的焦点到准线的距离为（\quad）

\choices
    {\(2\)}
    {\(4\)}
    {\(6\)}
    {\(8\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
设抛物线为 \(y^2=2px\)，其中 \(p>0\). 此时焦点为 \((p/2,0)\)，准线为 \(x=-p/2\)，所以焦点到准线的距离为 \(p\). 设圆心为原点、半径为 \(r\)，并设 \(A=(x_0,y_0)\)、\(B=(x_0,-y_0)\). 由 \(|AB|=4\sqrt2\)，得 \(y_0=2\sqrt2\)，从而 \(y_0^2=8\). 因 \(A\) 在抛物线上，
\(8=2px_0\)，\(x_0=\frac4p\).
又 \(A\) 在圆上，所以 \(r^2=x_0^2+y_0^2=\dfrac{16}{p^2}+8\). 准线与圆的交点为 \(D\)，\(E\)，弦 \(DE\) 的半长为 \(\sqrt5\)，圆心到准线的距离为 \(p/2\)，故
\[
r^2-\frac{p^2}{4}=5
\]
令 \(u=p^2\)，联立得
\[
\frac{16}{u}+8-\frac{u}{4}=5
\quad\Longleftrightarrow\quad
u^2-12u-64=0
\]
故 \(u=16\) 或 \(u=-4\). 因 \(u>0\)，得 \(p=4\). 所以焦点到准线的距离为 \(4\)，故选 B.
\end{solution}



\begin{problem}
平面 \(\alpha\) 过正方体 \(ABCD-A_1B_1C_1D_1\) 的顶点 \(A\)，\(\alpha\parallel\) 平面 \(CB_1D_1\)，\(\alpha\cap\) 平面 \(ABCD=m\)，\(\alpha\cap\) 平面 \(ABB_1A_1=n\)，则 \(m\)，\(n\) 所成角的正弦值为（\quad）

\choices
    {\(\dfrac{\sqrt3}{2}\)}
    {\(\dfrac{\sqrt2}{2}\)}
    {\(\dfrac{\sqrt3}{3}\)}
    {\(\dfrac13\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
建立空间直角坐标系，令正方体棱长为 \(1\)，取
\(A=(0,0,0)\)，\(B=(1,0,0)\)，\(C=(1,1,0)\)，\(D=(0,1,0)\)，\(B_1=(1,0,1)\)，\(D_1=(0,1,1)\).
平面 \(CB_1D_1\) 的一个法向量为 \((1,1,1)\)，所以过 \(A\) 且平行于它的平面 \(\alpha\) 为 \(x+y+z=0\). 因而
\(m: x+y=0\)，\(n: x+z=0\).
分别取两直线的方向向量 \(\bs u=(1,-1,0)\)、\(\bs v=(1,0,-1)\)，则
\[
\cos\theta=\frac{\bs u\cdot\bs v}{|\bs u||\bs v|}=\frac12
\]
所以
\[
\sin\theta=\sqrt{1-\cos^2\theta}=\frac{\sqrt3}{2}
\]
故选 A.
\end{solution}



\begin{problem}
已知函数 \(f(x)=\sin(\omega x+\varphi)\) （\(\omega>0\)，\(\ |\varphi|\le \pi/2\)），\(x=-\dfrac{\pi}{4}\) 为 \(f(x)\) 的零点，\(x=\dfrac{\pi}{4}\) 为 \(y=f(x)\) 图象的对称轴，且 \(f(x)\) 在 \(\left(\dfrac{\pi}{18},\dfrac{5\pi}{36}\right)\) 上单调，则 \(\omega\) 的最大值为（\quad）

\choices
    {\(11\)}
    {\(9\)}
    {\(7\)}
    {\(5\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
由 \(x=-\dfrac\pi4\) 是零点，\(x=\dfrac\pi4\) 是对称轴，正弦函数从零点到极值点的相位差为奇数个 \(\dfrac\pi2\)，故
\(\omega\cdot\frac\pi2=\frac{(2k+1)\pi}{2}\)，\(k\in\mathbb Z\)
从而 \(\omega=2k+1\)，即 \(\omega\) 是正奇数. 题中单调区间长度为
\[
\frac{5\pi}{36}-\frac\pi{18}=\frac\pi{12}
\]
一个正弦函数的单调区间长度不超过半个周期 \(\dfrac\pi\omega\)，所以 \(\omega\leqslant12\). 由于 \(\omega\) 为正奇数，只需先检验最大的候选值 \(11\). 当 \(\omega=11\) 时，零点条件为 \(-11\pi/4+\varphi=k\pi\). 结合 \(|\varphi|\leqslant\pi/2\)，只能取 \(\varphi=-\pi/4\). 此时给定区间两端的相位分别为
\(11\cdot\frac\pi{18}-\frac\pi4=\frac{13\pi}{36}\)，\(11\cdot\frac{5\pi}{36}-\frac\pi4=\frac{23\pi}{18}\)
其中包含相位 \(\dfrac\pi2\)，函数在区间内先增后减，不单调，故 \(\omega=11\) 不可取.

当 \(\omega=9\) 时，零点条件给出 \(\varphi=\pi/4\)，相位在给定区间内从 \(3\pi/4\) 变到 \(3\pi/2\)，而 \(\cos t<0\)（\(3\pi/4<t<3\pi/2\)），所以函数在该区间上单调递减，\(\omega=9\) 可以取. 因此 \(\omega\) 的最大值为 \(9\)，故选 B.
\end{solution}


\section{填空题}

\begin{problem}
设向量 \(\bs a=(m,1)\)，\(\bs b=(1,2)\)，且 \(|\bs a+\bs b|^2=|\bs a|^2+|\bs b|^2\)，则 \(m=\fillinblank{}\).
\end{problem}

\begin{answer}
\(-2\)
\end{answer}

\begin{solution}
由向量恒等式
\[
|\bs a+\bs b|^2=|\bs a|^2+|\bs b|^2+2\bs a\cdot\bs b
\]
结合题设得 \(\bs a\cdot\bs b=0\). 而
\[
\bs a\cdot\bs b=(m,1)\cdot(1,2)=m+2
\]
所以 \(m=-2\).
\end{solution}



\begin{problem}
\(\left(2x+\sqrt{x}\right)^5\) 的展开式中，\(x^3\) 的系数是\fillinblank{}. （用数字填写答案）
\end{problem}

\begin{answer}
10
\end{answer}

\begin{solution}
展开式通项中取 \(k\) 个 \(\sqrt{x}\) 因子时，所得项为
\[
\mathrm C_5^k(2x)^{5-k}(\sqrt{x})^k
=\mathrm C_5^k2^{5-k}x^{5-k/2}
\]
要得到 \(x^3\)，需 \(5-\dfrac k2=3\)，即 \(k=4\). 因此 \(x^3\) 的系数为
\[
\mathrm C_5^4 2^1=10
\]
\end{solution}



\begin{problem}
设等比数列 \(\{a_n\}\) 满足 \(a_1+a_3=10\)，\(a_2+a_4=5\)，则 \(a_1a_2\cdots a_n\) 的最大值为\fillinblank{}.
\end{problem}

\begin{answer}
64
\end{answer}

\begin{solution}
设等比数列的首项为 \(a_1\)，公比为 \(q\). 由已知，
\(a_1(1+q^2)=10\)，\(a_1q(1+q^2)=5\).
因 \(1+q^2>0\)，两式相除得 \(q=\dfrac12\)，再得 \(a_1=8\). 因而
\[
a_1a_2\cdots a_n=\prod_{k=1}^n8\left(\frac12\right)^{k-1}
=2^{3n-\frac{n(n-1)}2}=2^{\frac{7n-n^2}{2}}
\]
这里 \(n\) 为正整数. 指数 \(\dfrac{7n-n^2}{2}\) 在 \(n=3\)，\(4\) 时均为最大值 \(6\)，所以乘积的最大值为 \(2^6=64\).
\end{solution}



\begin{problem}
某高科技企业生产产品 \(A\) 和产品 \(B\)，需要甲，乙两种新型材料. 生产一件产品 \(A\) 需要甲材料 \(1.5\text{ kg}\)，乙材料 \(1\text{ kg}\)，用 \(5\) 个工时；生产一件产品 \(B\) 需要甲材料 \(0.5\text{ kg}\)，乙材料 \(0.3\text{ kg}\)，用 \(3\) 个工时，生产一件产品 \(A\) 的利润为 \(2100\) 元，生产一件产品 \(B\) 的利润为 \(900\) 元. 该企业现有甲材料 \(150\text{ kg}\)，乙材料 \(90\text{ kg}\)，则在不超过 \(600\) 个工时的条件下，生产产品 \(A\)，产品 \(B\) 的利润之和的最大值为\fillinblank{} 元.
\end{problem}

\begin{answer}
216000
\end{answer}

\begin{solution}
设生产产品 \(A\)，\(B\) 的数量分别为 \(x\)，\(y\)，则 \(x\geqslant0\)、\(y\geqslant0\)，并且资源约束为
\[
\begin{cases}
1.5x+0.5y\leqslant150,\\
x+0.3y\leqslant90,\\
5x+3y\leqslant600.
\end{cases}
\]
分别化为
\[
\begin{cases}
3x+y\leqslant300,\\
10x+3y\leqslant900,\\
5x+3y\leqslant600,\\
x\geqslant0, y\geqslant0.
\end{cases}
\]
利润函数为 \(z=2100x+900y\). 可行域的顶点为
\((0,0)\)，\(\ (0,200)\)，\(\ (60,100)\)，\(\ (90,0)\).
在这些顶点处，利润分别为 \(0\)、\(180000\)、\(216000\)、\(189000\) 元. 因此最大利润为 \(216000\) 元.
\end{solution}


\section{解答题}

\begin{problem}
\(\triangle ABC\) 的内角 \(A\)，\(B\)，\(C\) 的对边分别为 \(a\)，\(b\)，\(c\)，已知 \(2\cos C\left(a\cos B+b\cos A\right)=c\).

\begin{enumerate}
\item 求 \(C\)；
\item 若 \(c=\sqrt7\)，\(\triangle ABC\) 的面积为 \(\dfrac{3\sqrt3}{2}\)，求 \(\triangle ABC\) 的周长.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由正弦定理，设三角形外接圆半径为 \(R\)，则 \(a=2R\sin A\)，\(b=2R\sin B\)，\(c=2R\sin C\)，从而
\[
a\cos B+b\cos A
=2R\sin A\cos B+2R\sin B\cos A
=2R\sin(A+B)
=2R\sin C=c
\]
题设于是化为 \(2c\cos C=c\). 因 \(c>0\)，得 \(\cos C=\dfrac12\). 又 \(0<C<\pi\)，所以 \(C=\dfrac\pi3\).
\item 由面积公式
\[
\frac12ab\sin C=\frac{3\sqrt3}{2}
\]
代入 \(\sin C=\dfrac{\sqrt3}{2}\)，得 \(ab=6\). 又由余弦定理
\[
c^2=a^2+b^2-2ab\cos C
\]
及 \(c^2=7\)，得 \(a^2+b^2=13\). 因此
\[
(a+b)^2=a^2+b^2+2ab=25
\]
由于 \(a\)，\(b>0\)，所以 \(a+b=5\). 三角形周长为 \(a+b+c=5+\sqrt7\).
\end{enumerate}
\end{solution}


\begin{problem}
如图，在以 \(A\)，\(B\)，\(C\)，\(D\)，\(E\)，\(F\) 为顶点的五面体中，面 \(ABEF\) 为正方形，\(AF=2FD\)，\(\angle AFD=90^\circ\)，且二面角 \(D-AF-E\) 与二面角 \(C-BE-F\) 都是 \(60^\circ\).

\begin{enumerate}
\item 证明平面 \(ABEF\perp\) 平面 \(EFDC\)；
\item 求二面角 \(E-BC-A\) 的余弦值.
\end{enumerate}

\bitmapfigure[max width=0.48\linewidth]{img/2016/national_paper_1_science/q18_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 因 \(ABEF\) 为正方形，\(AF\perp FE\). 又已知 \(AF\perp FD\)，且 \(FE\)、\(FD\) 是平面 \(EFDC\) 内相交的两条直线，所以 \(AF\perp\) 平面 \(EFDC\). 而 \(AF\subset\) 平面 \(ABEF\)，故平面 \(ABEF\perp\) 平面 \(EFDC\).
\item 设正方形边长为 \(s\)，建立空间直角坐标系
\(F=(0,0,0)\)，\(A=(s,0,0)\)，\(E=(0,s,0)\)，\(B=(s,s,0)\).
由 \(AF=2FD\) 及 \(\angle AFD=90^\circ\)，得 \(FD=s/2\). 二面角 \(D-AF-E=60^\circ\) 表明 \(\angle DFE=60^\circ\)，故可取
\[
D=\left(0,\frac{s}{4},\frac{\sqrt3s}{4}\right)
\]
由于 \(AB\parallel EF\)，且两平面 \(ABCD\)、\(EFDC\) 的交线为 \(CD\)，有 \(CD\parallel EF\). 因此 \(C\) 与 \(D\) 的 \(x\)、\(z\) 坐标分别相同，设
\[
C=\left(0,y,\frac{\sqrt3s}{4}\right)
\]
二面角 \(C-BE-F=60^\circ\) 给出 \(\angle CEF=60^\circ\). 由于 \(\overrightarrow{EF}=(0,-s,0)\)，\(\overrightarrow{EC}=(0,y-s,\sqrt3s/4)\)，故
\[
\frac{s-y}{\sqrt{(s-y)^2+3s^2/16}}=\frac12
\]
由二面角为锐角知 \(s-y>0\)，解得 \(s-y=s/4\)，即 \(y=3s/4\). 取 \(s=4\)，则
\(A=(4,0,0)\)，\(B=(4,4,0)\)，\(C=(0,3,\sqrt3)\)，\(E=(0,4,0)\).
取平面 \(EBC\)、\(ABC\) 的法向量
\(\bs n_1=\overrightarrow{EC}\times\overrightarrow{EB}=(0,4\sqrt3,4)\)，\(\bs n_2=\overrightarrow{BA}\times\overrightarrow{BC}=(-4\sqrt3,0,-16)\).
按二面角 \(E-BC-A\) 的两侧取上述法向量，二面角余弦为
\[
\frac{\bs n_1\cdot\bs n_2}{|\bs n_1||\bs n_2|}
=\frac{-64}{8\cdot4\sqrt{19}}
=-\frac{2\sqrt{19}}{19}
\]
\end{enumerate}
\end{solution}


\begin{problem}
某公司计划购买 \(2\) 台机器，该种机器使用三年后即被淘汰. 机器有一易损零件，在购进机器时，可以额外购买这种零件作为备件，每个 \(200\) 元. 在机器使用期间，如果备件不足再购买，则每个 \(500\) 元. 现需决策在购买机器时应同时购买几个易损零件，为此搜集并整理了 \(100\) 台这种机器在三年使用期内更换的易损零件数，得下面柱状图：

以这 \(100\) 台机器更换的易损零件数的频率代替 \(1\) 台机器更换的易损零件数发生的概率，记 \(X\) 表示 \(2\) 台机器三年内共需更换的易损零件数，\(n\) 表示购买 \(2\) 台机器的同时购买的易损零件数.

\begin{enumerate}
\item 求 \(X\) 的分布列；
\item 若要求 \(P(X\le n)\ge 0.5\)，确定 \(n\) 的最小值；
\item 以购买易损零件所需费用的期望值为决策依据，在 \(n=19\) 与 \(n=20\) 之中选其一，应选用哪个？
\end{enumerate}

\bitmapfigure[float=inline,max width=0.45\linewidth]{img/2016/national_paper_1_science/q19_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由柱状图，1 台机器更换 \(8\)、\(9\)、\(10\)、\(11\) 个零件的概率分别为
\(0.2\)，\(0.4\)，\(0.2\)，\(0.2\).
设一台机器更换零件数为 \(Y\)，两台机器的更换数相互独立，且 \(X=Y_1+Y_2\). 逐项卷积得
具体地，\(P(X=16)=0.2^2=0.04\)，\(P(X=17)=2(0.2)(0.4)=0.16\)，\(P(X=18)=0.2^2+0.4^2+0.2^2=0.24\)，\(P(X=19)=2(0.2)(0.2)+2(0.4)(0.2)=0.24\)，\(P(X=20)=2(0.4)(0.2)+0.2^2=0.20\)，\(P(X=21)=2(0.2)(0.2)=0.08\)，\(P(X=22)=0.2^2=0.04\).
\begin{center}
\begin{tabular}{c|ccccccc}
\(x\) & \(16\) & \(17\) & \(18\) & \(19\) & \(20\) & \(21\) & \(22\)\\ \hline
\(P(X=x)\) & \(0.04\) & \(0.16\) & \(0.24\) & \(0.24\) & \(0.20\) & \(0.08\) & \(0.04\)
\end{tabular}
\end{center}
表中概率和为 \(1\).
\item 因为 \(X\) 只能取 \(16\) 至 \(22\) 的整数值，
\[
P(X\leqslant18)=0.04+0.16+0.24=0.44<0.5
\]
\[
P(X\leqslant19)=0.44+0.24=0.68\geqslant0.5
\]
所以满足条件的最小整数为 \(n=19\).
\item 若预购 \(n\) 个，购买和临时补购的总费用为
\[
Y=200n+500\max(X-n,0)
\]
当 \(n=19\) 时，只有 \(X>19\) 需要补购，因此
\[
E(Y)=200\cdot19+500\bigl[1\cdot0.20+2\cdot0.08+3\cdot0.04\bigr]=4040
\]
当 \(n=20\) 时，只有 \(X=21\)，\(22\) 需要补购，因此
\[
E(Y)=200\cdot20+500\bigl[1\cdot0.08+2\cdot0.04\bigr]=4080
\]
因为 \(4040<4080\)，所以应选购 \(19\) 个易损零件.
\end{enumerate}
\end{solution}


\begin{problem}
设圆 \(x^2+y^2+2x-15=0\) 的圆心为 \(A\)，直线 \(l\) 过点 \(B(1,0)\) 且与 \(x\) 轴不重合，\(l\) 交圆 \(A\) 于 \(C\)，\(D\) 两点，过 \(B\) 作 \(AC\) 的平行线交 \(AD\) 于点 \(E\).

\begin{enumerate}
\item 证明 \(|EA|+|EB|\) 为定值，并写出点 \(E\) 的轨迹方程；
\item 设点 \(E\) 的轨迹为曲线 \(C_1\)，直线 \(l\) 交 \(C_1\) 于 \(M\)，\(N\) 两点，过 \(B\) 且与 \(l\) 垂直的直线与圆 \(A\) 交于 \(P\)，\(Q\) 两点，求四边形 \(MPNQ\) 面积的取值范围.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 将圆方程化为
\[
(x+1)^2+y^2=16
\]
所以 \(A=(-1,0)\)，圆半径为 \(4\)，而 \(B=(1,0)\) 在圆内. 由于 \(C\)，\(D\) 在圆上，\(AC=AD\)，故 \(\angle ACD=\angle CDA\). 又因 \(BE\parallel AC\)、\(B\)，\(D\)，\(C\) 共线、\(A\)，\(D\)，\(E\) 共线，有 \(\angle EBD=\angle ACD\)，\(\angle EDB=\angle CDA\)，从而 \(EB=ED\). 又因 \(E\) 在线段 \(AD\) 上，
\[
EA+EB=EA+ED=AD=4
\]
因此 \(E\) 的轨迹是以 \(A\)，\(B\) 为焦点、焦距为 \(2\)、长轴长为 \(4\) 的椭圆. 椭圆中心为原点，半长轴为 \(2\)，半焦距为 \(1\)，故半短轴平方为 \(2^2-1^2=3\)，轨迹方程为
\[
\frac{x^2}{4}+\frac{y^2}{3}=1
\]
\item 先设 \(l:y=k(x-1)\)，其中 \(k\ne0\). 将其代入椭圆方程，得
\[
(3+4k^2)x^2-8k^2x+4k^2-12=0
\]
其两根之差的绝对值为 \(12\sqrt{1+k^2}/(3+4k^2)\)，再乘以直线的距离因子 \(\sqrt{1+k^2}\)，得到
\[
|MN|=\frac{12(1+k^2)}{4k^2+3}
\]
过 \(B\) 且垂直于 \(l\) 的直线为 \(x+ky-1=0\). 圆心 \(A\) 到此直线的距离为 \(2/\sqrt{1+k^2}\)，故圆的弦 \(PQ\) 长为
\[
|PQ|=4\sqrt{\frac{4k^2+3}{1+k^2}}
\]
两条弦 \(MN\)、\(PQ\) 所在直线互相垂直，并交于圆内点 \(B\)，所以四边形 \(MPNQ\) 的两条对角线为 \(MN\)、\(PQ\)，且
\[
S_{MPNQ}=\frac12|MN|\cdot|PQ|
=24\sqrt{\frac{1+k^2}{4k^2+3}}
\]
令 \(t=k^2>0\)，则
\[
\frac{1+t}{4t+3}
\]
严格递减，且当 \(t\to0^+\) 时趋于 \(1/3\)，当 \(t\to+\infty\) 时趋于 \(1/4\). 因此斜率有限且非零时
\[
12<S_{MPNQ}<8\sqrt3
\]
当 \(l\) 为竖直线 \(x=1\) 时，\(MN\) 为椭圆的竖直弦，\(|MN|=2\sqrt3\)，\(PQ\) 为圆的水平弦，\(|PQ|=4\sqrt3\)，面积为 \(12\). 故取值范围为
\[
[12,8\sqrt3)
\]
\end{enumerate}
\end{solution}


\begin{problem}
已知函数 \(f(x)=(x-2)\e^x+a(x-1)^2\) 有两个零点.

\begin{enumerate}
\item 求 \(a\) 的取值范围；
\item 设 \(x_1\)，\(x_2\) 是 \(f(x)\) 的两个零点，证明：\(x_1+x_2<2\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 先求导得
\[
f'(x)=(x-1)(e^x+2a)
\]
当 \(a>0\) 时，\(e^x+2a>0\)，故 \(f\) 在 \((-\infty,1)\) 上递减、在 \((1,+\infty)\) 上递增，且
\(\lim_{x\to-\infty}f(x)=+\infty\)，\(f(1)=-e<0\)，\(\lim_{x\to+\infty}f(x)=+\infty\).
所以此时恰有两个零点. 若 \(a=0\)，则 \(f(x)=(x-2)e^x\)，只有一个零点 \(x=2\).
若 \(a<0\)，令 \(x_0=\ln(-2a)\). 当 \(x_0<1\) 时，唯一可能的左侧极值点为 \(x_0\)，且
\[
f(x_0)=-\frac{e^{x_0}}2\bigl[(x_0-2)^2+1\bigr]<0
\]
左侧极大值小于零，左侧没有零点；当 \(x_0\geqslant1\) 时，\(f\) 在 \((-\infty,1]\) 上递增，在 \([1,x_0]\) 上递减，在 \([x_0,+\infty)\) 上递增，而 \(f(1)=-e<0\)、\(f(x_0)<0\)，也只有右侧一个零点. 因此
\[
a\in(0,+\infty)
\]
\item 由第一问 \(a>0\)，两个零点满足 \(x_1<1<x_2\). 因 \(f(2)=a>0\)，且 \(f\) 在 \((1,+\infty)\) 上递增，有 \(x_2<2\). 令 \(t=x_2-1\)，则 \(0<t<1\)，且由 \(f(x_2)=0\) 得
\[
a=\frac{(1-t)e^{1+t}}{t^2}
\]
\[
f(2-x_2)=f(1-t)=-(1+t)e^{1-t}+at^2
=e^{1-t}\bigl[(1-t)e^{2t}-(1+t)\bigr]
\]
令
\[
\phi(t)=\ln\frac{1+t}{1-t}-2t
\]
则 \(\phi(0)=0\)，且
\(\phi'(t)=\frac1{1+t}+\frac1{1-t}-2 =\frac{2t^2}{1-t^2}>0\)，\((0<t<1)\).
所以 \(\phi(t)>0\)，即 \((1-t)e^{2t}<1+t\)，从而 \(f(2-x_2)<0\). 由于 \(f\) 在 \((-\infty,1)\) 上严格递减，且 \(f(x_1)=0\)，故 \(x_1<2-x_2\)，即 \(x_1+x_2<2\).
\end{enumerate}
\end{solution}

\begin{problem}
如图，\(\triangle OAB\) 是等腰三角形，\(\angle AOB=120^\circ\). 以 \(O\) 为圆心，\(\dfrac12 OA\) 为半径作圆.

\begin{enumerate}
\item 证明：直线 \(AB\) 与 \(\odot O\) 相切；
\item 点 \(C\)，\(D\) 在 \(\odot O\) 上，且 \(A\)，\(B\)，\(C\)，\(D\) 四点共圆，证明：\(AB\parallel CD\).
\end{enumerate}

\bitmapfigure[max width=0.38\linewidth]{img/2016/national_paper_1_science/q22_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设 \(OA=OB=R\)，取 \(P\) 为 \(AB\) 的中点. 等腰三角形的中线 \(OP\) 同时是高，所以 \(OP\perp AB\)，且
\[
OP=OA\cos\frac{\angle AOB}{2}=R\cos60^\circ=\frac R2
\]
而 \(\odot O\) 的半径为 \(R/2\)，所以圆心 \(O\) 到直线 \(AB\) 的距离等于圆半径，故 \(AB\) 与 \(\odot O\) 相切.
\item 设过 \(A\)，\(B\)，\(C\)，\(D\) 的圆心为 \(O'\)，取 \(Q\) 为 \(CD\) 的中点. 因为 \(P\)、\(Q\) 分别是圆 \(O'\) 的两条弦 \(AB\)、\(CD\) 的中点，所以
\(O'P\perp AB\)，\(O'Q\perp CD\).
又因为 \(A\)，\(B\)，\(C\)，\(D\) 在 \(\odot O\) 上，且 \(A\)，\(B\) 不在半径为 \(R/2\) 的 \(\odot O\) 上，所以 \(O'\ne O\)，并且
\(OP\perp AB\)，\(OQ\perp CD\).
两条过 \(P\) 且垂直于 \(AB\) 的直线重合，故 \(O\)，\(O'\)，\(P\) 共线；同理 \(O\)，\(O'\)，\(Q\) 共线. 因此 \(P\)，\(O\)，\(Q\) 共线，而 \(AB\)、\(CD\) 都垂直于 \(POQ\)，所以 \(AB\parallel CD\).
\end{enumerate}
\end{solution}


\begin{problem}

在直角坐标系 \(xOy\) 中，曲线 \(C_1\) 的参数方程为
\(\begin{cases}
x=a\cos t,\\
y=1+a\sin t
\end{cases}\)
（\(t\) 为参数，\(a>0\)）.
在以坐标原点为极点，\(x\) 轴正半轴为极轴的极坐标系中，曲线 \(C_2:\rho=4\cos\theta\).

\begin{enumerate}
\item 说明 \(C_1\) 是哪种曲线，并将 \(C_1\) 的方程化为极坐标方程；
\item 直线 \(C_3\) 的极坐标方程为 \(\theta=\alpha_0\)，其中 \(\alpha_0\) 满足 \(\tan\alpha_0=2\)，若曲线 \(C_1\) 与 \(C_2\) 的公共点都在 \(C_3\) 上，求 \(a\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由参数方程
\[
x^2+(y-1)^2=a^2
\]
所以 \(C_1\) 是圆心为 \((0,1)\)、半径为 \(a\) 的圆. 代入 \(x=\rho\cos\theta\)、\(y=\rho\sin\theta\)，得其极坐标方程
\[
\rho^2-2\rho\sin\theta+1-a^2=0
\]
\item \(C_2:\rho=4\cos\theta\) 化为 \(x^2+y^2=4x\)，而 \(\tan\alpha_0=2\) 的直线 \(C_3\) 可写为 \(y=2x\). 联立 \(C_2\) 与 \(C_3\)：
\[
5x^2=4x
\]
所以 \(C_2\) 与 \(C_3\) 的交点只能是 \(O=(0,0)\) 和 \(R=(4/5,8/5)\). 题设公共点存在且都在 \(C_3\) 上，故任取公共点都必须是这两个点之一；而这两个点到 \(C_1\) 圆心 \((0,1)\) 的距离均为
\(|O-(0,1)|=1\)，\(|R-(0,1)|=\sqrt{\left(\frac45\right)^2+\left(\frac35\right)^2}=1\).
因此公共点在 \(C_1\) 上要求
\[
a^2=1
\]
因 \(a>0\)，得 \(a=1\).
\end{enumerate}
\end{solution}


\begin{problem}

已知函数 \(f(x)=|x+1|-|2x-3|\).

\begin{enumerate}
\item 在图中画出 \(y=f(x)\) 的图象；
\item 求不等式 \(|f(x)|>1\) 的解集.
\end{enumerate}

\bitmapfigure[float=inline,max width=0.45\linewidth]{img/2016/national_paper_1_science/q24_fig1.png}
\FloatBarrier
\end{problem}

\begin{solution}
\begin{enumerate}
\item 按 \(x=-1\) 与 \(x=\dfrac32\) 分段，有
\[
f(x)=\begin{cases}
x-4,&x\leqslant-1,\\
3x-2,&-1<x<\dfrac{3}{2},\\
4-x,&x\geqslant\dfrac{3}{2}.
\end{cases}
\]
左段经过连接点 \((-1,-5)\)，中段经过 \((-1,-5)\)、\((\frac32,\frac52)\)，右段从 \((\frac32,\frac52)\) 出发；中段与 \(x\) 轴交于 \((\frac23,0)\). 据此在题图坐标系中作出三段直线.
\item 分段解 \(|f(x)|>1\). 当 \(x\leqslant-1\) 时，\(f(x)=x-4\leqslant-5\)，故整段都满足不等式；当 \(-1<x<\dfrac32\) 时，
\[
|3x-2|>1
\Longleftrightarrow x<\frac13\ \text{或}\ x>1;
\]
\[
x\geqslant\frac32\Rightarrow |4-x|>1
\Rightarrow x<3\ \text{或}\ x>5
\]
合并三段并注意不等式严格，解集为
\[
\left(-\infty,\frac13\right)\cup(1,3)\cup(5,+\infty)
\]
\end{enumerate}
\end{solution}
