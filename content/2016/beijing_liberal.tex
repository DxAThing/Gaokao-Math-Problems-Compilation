\section{单选题}

\begin{problem}
已知集合 \( A=\{x\mid 2<x<4\} \)，\( B=\{x\mid x<3 \text{ 或 } x>5\} \)，则 \( A\cap B=\)

\choices
    {\(\{x\mid 2<x<5\}\)}
    {\(\{x\mid x<4 \text{ 或 } x>5\}\)}
    {\(\{x\mid 2<x<3\}\)}
    {\(\{x\mid x<2 \text{ 或 } x>5\}\)}

\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
因为集合 \( A=\{x\mid 2<x<4\} \)，\( B=\{x\mid x<3 \text{ 或 } x>5\} \)，所以 \( A\cap B=\{x\mid 2<x<3\} \).
故选 C.
\end{solution}

\begin{problem}
复数 \( \dfrac{1+2\i}{2-\i}=\)

\choices
    {\(\i\)}
    {\(1+\i\)}
    {\(-\i\)}
    {\(1-\i\)}

\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
\( \dfrac{1+2\i}{2-\i}=\dfrac{(1+2\i)(2+\i)}{(2-\i)(2+\i)}=\dfrac{5\i}{5}=\i \).
故选 A.
\end{solution}

\begin{problem}
下列函数中，在区间（\( -1 \)，\( 1 \)）上为减函数的是

\choices
    {\( y=\dfrac{1}{1-x} \)}
    {\( y=\cos x \)}
    {\( y=\ln\left( x+1 \right) \)}
    {\( y=2^{-x} \)}

\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
对于选项 \(A\)，当 \( x \) 增大时，\( -x \) 减小，\( 1-x \) 减小，所以 \( \dfrac{1}{1-x} \) 增大，因此函数 \( y=\dfrac{1}{1-x} \) 在区间（\( -1 \)，\( 1 \)）上为增函数，选项 \(A\) 错误.

对于选项 \(B\)，函数 \( y=\cos x \) 在区间（\( -1 \)，\( 1 \)）上不单调，选项 \(B\) 错误.

对于选项 \(C\)，当 \( x \) 增大时，\( x+1 \) 增大，\( \ln\left( x+1 \right) \) 增大，因此函数 \( y=\ln\left( x+1 \right) \) 在区间（\( -1 \)，\( 1 \)）上为增函数，选项 \(C\) 错误.
对于选项 \(D\)，当 \( x \) 增大时，\( -x \) 减小. 因为函数 \( y=2^x \) 为增函数，所以函数 \( y=2^{-x} \) 在区间（\( -1 \)，\( 1 \)）上为减函数，选项 \(D\) 正确.

故选 D.
\end{solution}

\begin{problem}
圆 \( \left( x+1 \right)^2+y^2=2 \) 的圆心到直线 \( y=x+3 \) 的距离为

\choices
    {\(1\)}
    {\(2\)}
    {\(\sqrt2\)}
    {\(2\sqrt2\)}

\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
圆 \( \left( x+1 \right)^2+y^2=2 \) 的圆心为 \( (-1,0) \).
直线 \( y=x+3 \) 化为一般式为 \( x-y+3=0 \)，所以圆心到直线的距离为 \( d=\dfrac{|{-1}+3|}{\sqrt{1^2+(-1)^2}}=\dfrac{2}{\sqrt2}=\sqrt2 \).
故选 C.
\end{solution}

    {\(\dfrac25\)}

\end{answer}

故选 B.
\end{solution}

\begin{problem}
已知向量 \( \bs a=(1,\sqrt3) \)，\( \bs b=(\sqrt3,1) \)，则 \( \bs a \) 与 \( \bs b \) 夹角的大小为\fillinblank{}.

\end{problem}

\begin{answer}
\(\dfrac{\pi}{6}\)
\end{answer}

\begin{solution}
因为 \( \bs a=(1,\sqrt3) \)，\( \bs b=(\sqrt3,1) \)，\( \cos\theta=\dfrac{\bs a\cdot\bs b}{|\bs a||\bs b|}=\dfrac{2\sqrt3}{2\times 2}=\dfrac{\sqrt3}{2} \).
又 \( \theta\in[0,\pi] \)，所以 \( \theta=\dfrac{\pi}{6} \).
故答案为：\( \dfrac{\pi}{6} \).
\end{solution}

因此当 \( x=2 \) 时，\( f(x) \) 取最大值 \( 2 \).
故答案为：\( 2 \).
\end{solution}

\begin{problem}
已知双曲线 \( \dfrac{x^2}{a^2}-\dfrac{y^2}{b^2}=1 \)（\( a>0 \)，\( b>0 \)）的一条渐近线为 \( 2x+y=0 \)，一个焦点为 \( (\sqrt5,0) \)，则 \( a=\fillinblank{} \)，\( b=\fillinblank{} \).

\end{problem}

\begin{answer}
\(1\)，\(2\)
\end{answer}

\begin{solution}
因为双曲线 \( \dfrac{x^2}{a^2}-\dfrac{y^2}{b^2}=1 \) 的一条渐近线为 \( 2x+y=0 \)，所以 \( \dfrac{b}{a}=2 \).
又一个焦点为 \( (\sqrt5,0) \)，所以 \( \sqrt{a^2+b^2}=\sqrt5 \).
联立解得 \(a=1\)，\(b=2\).

\begin{problem}
在 \( \triangle ABC \) 中，\( \angle A=\dfrac{2\pi}{3} \)，\( a=\sqrt3\,c \)，则 \( \dfrac{b}{c}=\fillinblank{} \).

\end{problem}

\begin{answer}
\(1\)
\end{answer}

\begin{solution}
在 \( \triangle ABC \) 中，\( \angle A=\dfrac{2\pi}{3} \)，\( a=\sqrt3\,c \).
由正弦定理得 \( \dfrac{a}{\sin A}=\dfrac{c}{\sin C} \)，所以 \( \dfrac{\sqrt3\,c}{\sin\frac{2\pi}{3}}=\dfrac{c}{\sin C} \)，从而 \( \sin C=\dfrac12 \).
因为 \( \angle A=\dfrac{2\pi}{3} \)，所以 \( C=\dfrac{\pi}{6} \)，进而 \( B=\pi-\dfrac{2\pi}{3}-\dfrac{\pi}{6}=\dfrac{\pi}{6} \).
所以三角形是等腰三角形，\( b=c \)，即 \( \dfrac{b}{c}=1 \).
故答案为：\( 1 \).
\end{solution}

\begin{problem}
某网店统计了连续三天售出商品的种类情况：第一天售出 \( 19 \) 种商品，第二天售出 \( 13 \) 种商品，第三天售出 \( 18 \) 种商品；前两天都售出的商品有 \( 3 \) 种，后两天都售出的商品有 \( 4 \) 种，则该网店

\begin{enumerate}

\item 这三天售出的商品最少有\fillinblank{} 种.
\end{enumerate}

\end{problem}

\begin{answer}
（1）16；（2）29
\end{answer}

\begin{solution}

\begin{problem}
已知 \( \{a_n\} \) 是等差数列，\( \{b_n\} \) 是等比数列，且 \( b_2=3 \)，\( b_3=9 \)，\( a_1=b_1 \)，\( a_{14}=b_4 \).

\begin{enumerate}
\item 求 \( \{a_n\} \) 的通项公式；
\item 设 \( c_n=a_n+b_n \)，求数列 \( \{c_n\} \) 的前 \( n \) 项和.
\end{enumerate}

\end{problem}

\begin{solution}
（1）设 \( \{a_n\} \) 是公差为 \( d \) 的等差数列，\( \{b_n\} \) 是公比为 \( q \) 的等比数列.
由 \( b_2=3 \)，\( b_3=9 \)，可得 \( q=\dfrac{b_3}{b_2}=3 \).
所以 \( b_n=b_2q^{n-2}=3\cdot 3^{n-2}=3^{n-1} \).
于是 \(a_1=b_1=1\)，\(a_{14}=b_4=27\).
所以 \( d=\dfrac{a_{14}-a_1}{13}=\dfrac{27-1}{13}=2 \).
从而 \( a_n=a_1+(n-1)d=1+2(n-1)=2n-1 \).

（2）因为 \( c_n=a_n+b_n=2n-1+3^{n-1} \)，所以数列 \( \{c_n\} \) 的前 \( n \) 项和为 \( S_n=(1+3+\cdots+(2n-1))+(1+3+9+\cdots+3^{n-1})=n^2+\dfrac{3^n-1}{2} \).
\end{solution}

\begin{problem}
已知函数 \( f(x)=2\sin\omega x\cos\omega x+\cos2\omega x \)（\( \omega>0 \)）的最小正周期为 \( \pi \).

\begin{enumerate}
\item 求 \( \omega \) 的值；
\item 求 \( f(x) \) 的单调递增区间.
\end{enumerate}

\end{problem}

\begin{solution}
（1）因为 \( f(x)=2\sin\omega x\cos\omega x+\cos2\omega x=\sin2\omega x+\cos2\omega x=\sqrt2\sin\left( 2\omega x+\frac{\pi}{4} \right) \)，所以它的最小正周期为 \( T=\frac{2\pi}{2\omega}=\frac{\pi}{\omega} \).
由 \( T=\pi \) 得 \( \omega=1 \).

（2）由（1）得 \( f(x)=\sqrt2\sin\left( 2x+\frac{\pi}{4} \right) \).
由 \( -\frac{\pi}{2}+2k\pi\le 2x+\frac{\pi}{4}\le \frac{\pi}{2}+2k\pi \) 得 \(-\frac{3\pi}{8}+k\pi\le x\le \frac{\pi}{8}+k\pi\)，\(k\in\Z\).
所以 \( f(x) \) 的单调递增区间为 \(\left[ -\frac{3\pi}{8}+k\pi,\frac{\pi}{8}+k\pi \right]\)，\(k\in\Z\).
\end{solution}

\begin{problem}
如图，在四棱锥 \( P-ABCD \) 中，\( PC\perp \) 平面 \(ABCD\)，\( AB\parallel DC \)，\( DC\perp AC \).

\begin{enumerate}
\item 求证：\( DC\perp \) 平面 \(PAC\)；
\item 求证：平面 \( PAB\perp \) 平面 \(PAC\)；
\item 设点 \(E\) 为 \(AB\) 的中点，在棱 \(PB\) 上是否存在点 \(F\)，使得 \( PA\parallel \) 平面 \(CEF\)？说明理由.
\end{enumerate}

\bitmapfigure[max width=0.52\linewidth,max totalheight=4.8cm]{img/2016/beijing_liberal/q18_fig1.png}

\end{problem}

\begin{solution}
（1）证明：因为 \( PC\perp \) 平面 \(ABCD\)，\( DC\subset \) 平面 \(ABCD\)，所以 \( PC\perp DC \).
又 \( DC\perp AC \)，且 \( PC\cap AC=C \)，所以 \( DC\perp \text{平面 } PAC \).

（2）证明：因为 \( AB\parallel DC \)，\( DC\perp AC \)，所以 \( AB\perp AC \).
又 \( PC\perp \) 平面 \(ABCD\)，\( AB\subset \) 平面 \(ABCD\)，所以 \( PC\perp AB \).
因为 \( PC\cap AC=C \)，所以 \( AB\perp \text{平面 } PAC \).
又 \( AB\subset \) 平面 \(PAB\)，所以平面 \( PAB\perp \) 平面 \(PAC\).

（3）在棱 \(PB\) 上存在中点 \(F\)，使得 \( PA\parallel \) 平面 \(CEF\).
因为点 \(E\) 为 \(AB\) 的中点，所以 \( EF\parallel PA \).
又 \( PA\nsubseteq \) 平面 \(CEF\)，\( EF\subset \) 平面 \(CEF\)，所以 \( PA\parallel \text{平面 } CEF \).
\end{solution}

\begin{problem}
已知椭圆 \( C:\dfrac{x^2}{a^2}+\dfrac{y^2}{b^2}=1 \) 过点 \( A(2,0) \)，\( B(0,1) \) 两点.

\begin{enumerate}
\item 求椭圆 \(C\) 的方程及离心率；
\item 设 \(P\) 为第三象限内一点且在椭圆 \(C\) 上，直线 \(PA\) 与 \( y \) 轴交于点 \(M\)，直线 \(PB\) 与 \( x \) 轴交于点 \(N\)，求证：四边形 \(ABNM\) 的面积为定值.
\end{enumerate}

\bitmapfigure[max width=0.56\linewidth,max totalheight=4.8cm]{img/2016/beijing_liberal/q19_fig1.png}

\end{problem}

\begin{solution}
（1）因为椭圆 \(C\) 过点 \( A(2,0) \)，\( B(0,1) \) 两点，所以 \(a=2\)，\(b=1\).
从而 \( c=\sqrt{a^2-b^2}=\sqrt3 \)，所以椭圆 \(C\) 的方程为 \( \frac{x^2}{4}+y^2=1 \)，离心率为 \( e=\frac{\sqrt3}{2} \).

（2）证明：设 \( P(x_0,y_0) \)，则 \(P\) 在椭圆上，满足 \( \frac{x_0^2}{4}+y_0^2=1 \).
直线 \(PA\) 的斜率为 \( \dfrac{y_0}{x_0-2} \)，所以其方程为 \( y=\frac{y_0}{x_0-2}(x-2) \).
取 \( x=0 \)，得 \( y_M=-\frac{2y_0}{x_0-2} \).
直线 \(PB\) 的斜率为 \( \dfrac{y_0-1}{x_0} \)，所以其方程为 \( y=\frac{y_0-1}{x_0}x+1 \).
取 \( y=0 \)，得 \( x_N=\frac{x_0}{1-y_0} \).
于是 \( |AN|=2-\frac{x_0}{1-y_0}=\frac{2-2y_0-x_0}{1-y_0} \)，\( |BM|=1+\frac{2y_0}{x_0-2}=\frac{x_0+2y_0-2}{x_0-2} \).
所以 \(S_{ABNM}=\frac12|AN|\cdot |BM|=\frac12\cdot \frac{2-2y_0-x_0}{1-y_0}\cdot \frac{x_0+2y_0-2}{x_0-2}=-\frac12\cdot \frac{(x_0+2y_0-2)^2}{(1-y_0)(x_0-2)}\).
化简得 \( S_{ABNM}=2 \).
所以四边形 \(ABNM\) 的面积为定值 \( 2 \).
\end{solution}

\begin{problem}
设函数 \( f(x)=x^3+ax^2+bx+c \).

\begin{enumerate}
\item 求曲线 \( y=f(x) \) 在点 \( (0,f(0)) \) 处的切线方程；
\item 设 \( a=b=4 \)，若函数 \( f(x) \) 有三个不同零点，求 \( c \) 的取值范围；
\item 求证：\( a^2-3b>0 \) 是 \( f(x) \) 有三个不同零点的必要而不充分条件.
\end{enumerate}

\end{problem}

\begin{solution}
（1）函数 \( f(x)=x^3+ax^2+bx+c \) 的导数为 \( f'(x)=3x^2+2ax+b \).
所以曲线 \( y=f(x) \) 在点 \( (0,f(0)) \) 处的切线斜率为 \( k=f'(0)=b \)，切点为 \( (0,c) \).
因此所求切线方程为 \( y=bx+c \).

（2）设 \( a=b=4 \)，则 \( f(x)=x^3+4x^2+4x+c \).
由 \( f(x)=0 \) 可得 \( -c=x^3+4x^2+4x \).
设 \( g(x)=x^3+4x^2+4x \)，则 \( g'(x)=3x^2+8x+4=(x+2)(3x+2) \).

当 \( x>- \dfrac23 \) 或 \( x<-2 \) 时，\( g'(x)>0 \)，\( g(x) \) 递增；

当 \( -2<x<-\dfrac23 \) 时，\( g'(x)<0 \)，\( g(x) \) 递减.

所以 \( g(x) \) 在 \( x=-2 \) 处取得极大值 \( 0 \)，在 \( x=-\dfrac23 \) 处取得极小值 \( -\frac{32}{27} \).
由函数 \( f(x) \) 有三个不同零点，得 \( -\frac{32}{27}<-c<0 \).
所以 \( 0<c<\frac{32}{27} \).

（3）证明：若 \( f(x) \) 有三个不同零点，则方程 \( f(x)=0 \) 有三个不同实根，所以函数 \( f(x) \) 有 \( 3 \) 个单调区间，即导数 \( f'(x)=3x^2+2ax+b \) 的图象与 \( x \) 轴有两个不同交点.
因此其判别式满足 \( \Delta>0 \)，即 \( 4a^2-12b>0 \)，所以 \( a^2-3b>0 \).

下面说明它不是充分条件.

当 \( c=0 \)，\( a=b=4 \) 时，\( a^2-3b=16-12=4>0 \)，但 \( f(x)=x^3+4x^2+4x=x(x+2)^2 \)，零点只有 \( 0 \) 和 \( -2 \) 两个.
所以 \( a^2-3b>0 \) 是 \( f(x) \) 有三个不同零点的必要而不充分条件.
\end{solution}
