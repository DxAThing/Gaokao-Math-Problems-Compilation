\section{单选题}

\begin{problem}
已知集合 \( P=\{x\in\R\mid 1\le x\le 3\} \)，\( Q=\{x\in\R\mid x^2\ge 4\} \)，则 \( P\cup \left( \complement_{\R} Q \right)= \)（\quad）

\choices
    {\([2,3]\)}
    {\(( -2,3]\)}
    {\([1,2)\)}
    {\(( -\infty,-2]\cup [1,+\infty)\)}

\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
因为 \(Q=\{x\in\R\mid x\ge 2 \text{ 或 } x\le -2\}\)，所以 \(\complement_{\R}Q=\{x\in\R\mid -2<x<2\}\). 因此 \(P\cup \left( \complement_{\R}Q \right)=(-2,3]\). 故选 B.
\end{solution}

\begin{problem}
已知互相垂直的平面 \(\alpha\)，\(\beta\) 交于直线 \( l \)，若直线 \(m\)，\(n\) 满足 \( m\parallel \alpha \)，\( n\perp \beta \)，则（\quad）

\choices
    {\(m\parallel l\)}
    {\(m\parallel n\)}
    {\(n\perp l\)}
    {\(m\perp n\)}

\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
因为 \( l\subset \beta \)，且 \( n\perp \beta \)，所以 \( n\perp l \). 故选 C.
\end{solution}

\begin{problem}
在平面上，过点 \(P\) 作直线 \( l \) 的垂线所得的垂足称为点 \(P\) 在直线 \( l \) 上的投影，由区域 \( \left\{\begin{array}{l} x-2\le 0, \\ x+y\ge 0, \\ x-3y+4\ge 0 \end{array}\right. \) 中的点在直线 \( x+y-2=0 \) 上的投影构成的线段记为 \(AB\)，则 \( |AB|= \)（\quad）

    {\(4\)}
    {\(3\sqrt2\)}
    {\(6\)}

\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
区域内的点在直线 \( x+y-2=0 \) 上的投影构成线段 \(AB\).

\begin{problem}
命题“\( \forall x\in\R \)，\( \exists n\in\N^* \)，使得 \( n\ge x^2 \)”的否定形式是（\quad）

\choices
    {\( \forall x\in\R \)，\( \exists n\in\N^* \)，使得 \( n<x^2 \)}
    {\( \forall x\in\R \)，\( \forall n\in\N^* \)，使得 \( n<x^2 \)}
    {\( \exists x\in\R \)，\( \exists n\in\N^* \)，使得 \( n<x^2 \)}
    {\( \exists x\in\R \)，\( \forall n\in\N^* \)，使得 \( n<x^2 \)}

\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
全称命题的否定是特称命题，所以该命题的否定形式是 \(\exists x\in\R\)，\(\forall n\in\N^*\)，\(n<x^2\). 故选 D.
\end{solution}

\begin{problem}
设函数 \( f(x)=\sin^2x+b\sin x+c \)，则 \( f(x) \) 的最小正周期（\quad）

\choices
    {\text{与 \( b \) 有关，且与 \( c \) 有关}}
    {\text{与 \( b \) 有关，但与 \( c \) 无关}}
    {\text{与 \( b \) 无关，且与 \( c \) 无关}}
    {\text{与 \( b \) 无关，但与 \( c \) 有关}}

\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
常数 \( c \) 只会使函数图象上下平移，所以周期与 \( c \) 无关.

当 \( b=0 \) 时，\( f(x)=\sin^2x+c=\frac{1-\cos 2x}{2}+c \)，其最小正周期为 \( \pi \).

当 \( b\ne 0 \) 时，\( \sin x \) 的最小正周期为 \( 2\pi \)，所以 \( f(x) \) 的最小正周期为 \( 2\pi \).

因此 \( f(x) \) 的最小正周期与 \( b \) 有关，但与 \( c \) 无关.

故选 B.
\end{solution}

\begin{problem}
如图，点列 \( \{A_n\}\)，\( \{B_n\}\) 分别在某锐角的两边上，且 \( |A_nA_{n+1}|=|A_{n+1}A_{n+2}| \)，\( A_n\ne A_{n+1} \)，\( n\in\N^* \)，\( |B_nB_{n+1}|=|B_{n+1}B_{n+2}| \)，\( B_n\ne B_{n+1} \)，\( n\in\N^* \)（\( P\ne Q \) 表示点 \(P\) 与点 \(Q\) 不重合）. 若 \( d_n=|A_nB_n| \)，\( S_n \) 为 \( \triangle A_nB_nB_{n+1} \) 的面积，则（\quad）

\bitmapfigure[max width=0.62\textwidth]{img/2016/zhejiang_science/q6_fig1.png}

\choices
    {\(\{S_n\}\) 是等差数列}
    {\(\{S_n^2\}\) 是等差数列}
    {\(\{d_n\}\) 是等差数列}
    {\(\{d_n^2\}\) 是等差数列}

\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
设锐角的顶点为 \(O\)，再设 \( |OA_1|=a \)，\( |OB_1|=b \)，\( |A_nA_{n+1}|=|A_{n+1}A_{n+2}|=c \)，\( |B_nB_{n+1}|=|B_{n+1}B_{n+2}|=d \).

由于 \(a\)，\(b\) 不确定，所以 \( \{d_n\} \) 不一定是等差数列，\( \{d_n^2\} \) 也不一定是等差数列.

设 \( \triangle A_nB_nB_{n+1} \) 的底边 \( B_nB_{n+1} \) 上的高为 \( h_n \). 由三角形相似可得 \( \frac{h_n}{h_{n+1}}=\frac{a+(n-1)c}{a+nc} \)，\( \frac{h_{n+2}}{h_{n+1}}=\frac{a+(n+1)c}{a+nc} \).

两式相加得 \(\frac{h_n+h_{n+2}}{h_{n+1}}=2\)，即 \(h_n+h_{n+2}=2h_{n+1}\).

又 \(S_n=\frac12dh_n\)，所以 \( S_n+S_{n+2}=2S_{n+1} \)，即 \( S_{n+2}-S_{n+1}=S_{n+1}-S_n \).

因此数列 \( \{S_n\} \) 为等差数列.

故选 A.
\end{solution}

\begin{problem}
已知椭圆 \( C_1:\dfrac{x^2}{m^2}+y^2=1 \)（\( m>1 \)）与双曲线 \( C_2:\dfrac{x^2}{n^2}-y^2=1 \)（\( n>0 \)）的焦点重合，\(e_1\)，\(e_2\) 分别为 \(C_1\)，\(C_2\) 的离心率，则（\quad）

\choices
    {\(m>n\) 且 \( e_1e_2>1 \)}
    {\(m>n\) 且 \( e_1e_2<1 \)}
    {\(m<n\) 且 \( e_1e_2>1 \)}
    {\(m<n\) 且 \( e_1e_2<1 \)}

\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
因为椭圆与双曲线的焦点重合，所以 \(m^2-1=n^2+1\). 于是 \(m^2-n^2=2>0\)，从而 \( m>n \). 又 \(e_1=\frac{c}{m}\)，\( e_2=\frac{c}{n} \)，所以 \(e_1e_2=\frac{c^2}{mn}>1\). 故选 A.
\end{solution}

\begin{problem}
已知实数 \(a\)，\(b\)，\(c\)，则（\quad）

\choices
    {\(\bigl|a^2+b+c\bigr|+\bigl|a+b^2+c\bigr|\le 1\) 时，\( a^2+b^2+c^2<100 \)}
    {\(\bigl|a^2+b+c\bigr|+\bigl|a^2+b-c\bigr|\le 1\) 时，\( a^2+b^2+c^2<100 \)}
    {\(\bigl|a+b+c^2\bigr|+\bigl|a+b-c^2\bigr|\le 1\) 时，\( a^2+b^2+c^2<100 \)}
    {\(\bigl|a^2+b+c\bigr|+\bigl|a+b^2-c\bigr|\le 1\) 时，\( a^2+b^2+c^2<100 \)}

\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
A 中取 \( a=b=10 \)，\( c=-110 \)，则 \( |a^2+b+c|+|a+b^2+c|=0\le 1 \)，但 \( a^2+b^2+c^2>100 \).

B 中取 \( a=10 \)，\( b=-100 \)，\( c=0 \)，则 \( |a^2+b+c|+|a^2+b-c|=0\le 1 \)，但 \( a^2+b^2+c^2>100 \).

C 中取 \( a=100 \)，\( b=-100 \)，\( c=0 \)，则 \( |a+b+c^2|+|a+b-c^2|=0\le 1 \)，但 \( a^2+b^2+c^2>100 \).

故选 D.
\end{solution}

\begin{problem}
若抛物线 \( y^2=4x \) 上的点 \(M\) 到焦点的距离为 \( 10 \)，则 \(M\) 到 \( y \) 轴的距离是\fillinblank{}.

\end{problem}

\begin{answer}
\(9\)
\end{answer}

\begin{solution}
抛物线的准线为 \( x=-1 \). 因为点 \(M\) 到焦点的距离为 \( 10 \)，所以点 \(M\) 到准线的距离也为 \( 10 \). 因此点 \(M\) 到 \( y \) 轴的距离为 \( 9 \). 故答案为：\( 9 \).
\end{solution}

\begin{problem}
已知 \( 2\cos^2x+\sin 2x=A\sin(\omega x+\varphi)+b \)（\( A>0 \)），则 \( A=\fillinblank{} \)，\( b=\fillinblank{} \).

\end{problem}

\begin{answer}
\(\sqrt2\)，\(1\)
\end{answer}

\begin{solution}
因为
\(2\cos^2x+\sin 2x=1+\cos 2x+\sin 2x=1+\sqrt2\left( \frac{\sqrt2}{2}\cos 2x+\frac{\sqrt2}{2}\sin 2x \right)=1+\sqrt2\sin\left( 2x+\frac{\pi}{4} \right)\)，
所以 \(A=\sqrt2\)，\(b=1\). 故答案为：\( \sqrt2 \)；\( 1 \).
\end{solution}

\begin{problem}
已知 \( a>b>1 \)，若 \( \log_a b+\log_b a=\dfrac52 \)，\( a^b=b^a \)，则 \( a=\fillinblank{} \)，\( b=\fillinblank{} \).

\end{problem}

\begin{answer}
\(4\)，\(2\)
\end{answer}

\begin{solution}
设 \( t=\log_b a \)，由 \( a>b>1 \) 知 \( t>1 \). 代入条件得 \( t+\frac{1}{t}=\frac52 \)，即 \(2t^2-5t+2=0\). 解得 \( t=2 \) 或 \( t=\dfrac12 \)（舍去）. 所以 \( \log_b a=2 \)，即 \( a=b^2 \). 又因为 \( a^b=b^a \)，所以 \( b^{2b}=b^a \)，故 \( a=2b=b^2 \)，解得 \(b=2\)，\(a=4\). 故答案为：\( 4 \)；\( 2 \).
\end{solution}

\begin{problem}
设数列 \( \{a_n\} \) 的前 \( n \) 项和为 \( S_n \)，若 \( S_2=4 \)，\( a_{n+1}=2S_n+1 \)，\( n\in\N^* \)，则 \( a_1=\fillinblank{} \)，\( S_5=\fillinblank{} \).

\end{problem}

\begin{answer}
\(1\)，\(121\)
\end{answer}

\begin{solution}
当 \( n=1 \) 时，\( a_1=S_1 \)，所以 \( a_2=2S_1+1=2a_1+1 \). 又 \( S_2=4 \)，即 \( a_1+a_2=4 \)，所以 \( 3a_1+1=4 \)，解得 \( a_1=1 \). 又 \( a_{n+1}=S_{n+1}-S_n \)，所以 \( S_{n+1}=3S_n+1 \). 由 \( S_2=4 \) 得 \( S_3=3\times 4+1=13 \)，\( S_4=3\times 13+1=40 \)，\( S_5=3\times 40+1=121 \). 故答案为：\( 1 \)，\( 121 \).
\end{solution}

\begin{problem}
如图，在 \( \triangle ABC \) 中，\( AB=BC=2 \)，\( \angle ABC=120^\circ \). 若平面 \(ABC\) 外的点 \(P\) 和线段 \(AC\) 上的点 \(D\)，满足 \( PD=DA \)，\( PB=BA \)，则四面体 \(PBCD\) 的体积的最大值是\fillinblank{}.

\bitmapfigure[max width=0.42\textwidth]{img/2016/zhejiang_science/q14_fig1.png}

\end{problem}

\begin{answer}
\(\dfrac12\)
\end{answer}

\begin{solution}
如图，\(M\) 是 \(AC\) 的中点.
\circled{1} 当 \( AD=t<AM=\sqrt3 \) 时，此时高为 \(P\) 到 \(BD\) 的距离，也就是 \(A\) 到 \(BD\) 的距离，即图中 \(AE\).
由于 \(DM=\sqrt3-t\)，由 \( \triangle ADE\sim \triangle BDM \) 可得 \( h=\frac{t}{\sqrt{(\sqrt3-t)^2+1}} \). 于是 \( V=\frac13\cdot \frac12(2\sqrt3-t)\cdot 1\cdot \frac{t}{\sqrt{(\sqrt3-t)^2+1}}=\frac16\cdot \frac{3-(\sqrt3-t)^2}{\sqrt{(\sqrt3-t)^2+1}} \)，\( t\in(0,\sqrt3) \).

\circled{2} 当 \( AD=t>AM=\sqrt3 \) 时，此时高仍为 \(P\) 到 \(BD\) 的距离，也就是 \(A\) 到 \(BD\) 的距离，即图中 \(AH\).
由于 \(DM=t-\sqrt3\)，且由等面积有 \( \frac12AD\cdot BM=\frac12BD\cdot AH \)，故 \( \frac12\cdot t\cdot 1=\frac12\sqrt{(t-\sqrt3)^2+1}\cdot h \)，从而 \(h=\frac{t}{\sqrt{(\sqrt3-t)^2+1}} \). 于是 \( V=\frac16\cdot \frac{3-(\sqrt3-t)^2}{\sqrt{(\sqrt3-t)^2+1}} \)，\( t\in(\sqrt3,2\sqrt3) \).

综上，\( V=\frac16\cdot \frac{3-(\sqrt3-t)^2}{\sqrt{(\sqrt3-t)^2+1}} \)，\( t\in(0,2\sqrt3) \). 令 \( m=\sqrt{(\sqrt3-t)^2+1}\in[1,2) \)，则 \( V=\frac16\cdot \frac{4-m^2}{m} \). 当 \( m=1 \) 时，\(V\) 取最大值 \( V_{\max}=\frac12 \). 故答案为：\( \dfrac12 \).
\end{solution}

\begin{problem}
已知平面向量 \(\bs a\)，\(\bs b\)，\( |\bs a|=1 \)，\( |\bs b|=2 \)，若对任意单位向量 \( \bs e \)，均有 \( |\bs a\cdot \bs e|+|\bs b\cdot \bs e|\le \sqrt6 \)，则 \( \bs a\cdot \bs b \) 的最大值是\fillinblank{}.

\end{problem}

\begin{answer}
\(\dfrac12\)
\end{answer}

\begin{solution}
因为 \( |(\bs a+\bs b)\cdot \bs e|=|\bs a\cdot \bs e+\bs b\cdot \bs e|\le |\bs a\cdot \bs e|+|\bs b\cdot \bs e|\le \sqrt6 \)，所以 \( |(\bs a+\bs b)\cdot \bs e|\le |\bs a+\bs b|\le \sqrt6 \). 两边平方得 \( |\bs a|^2+|\bs b|^2+2\bs a\cdot \bs b\le 6 \). 即 \( 1^2+2^2+2\bs a\cdot \bs b\le 6 \)，所以 \( \bs a\cdot \bs b\le \frac12 \). 故 \( \bs a\cdot \bs b \) 的最大值是 \( \dfrac12 \).
\end{solution}

\begin{problem}
在 \( \triangle ABC \) 中，内角 \(A\)，\(B\)，\(C\) 所对的边分别为 \(a\)，\(b\)，\(c\)，已知 \( b+c=2a\cos B \).

\begin{enumerate}
\item 证明：\( A=2B \)；
\item 若 \( \triangle ABC \) 的面积 \( S=\dfrac{a^2}{4} \)，求角 \(A\) 的大小.
\end{enumerate}

\end{problem}

\begin{solution}
（1）证明：因为 \( b+c=2a\cos B \)，由正弦定理得 \( \sin B+\sin C=2\sin A\cos B \). 又 \( C=\pi-A-B \)，所以 \( \sin B+\sin(A+B)=2\sin A\cos B \). 即 \( \sin B+\sin A\cos B+\cos A\sin B=2\sin A\cos B \). 所以 \( \sin B=\sin A\cos B-\cos A\sin B=\sin(A-B) \). 因为 \(A\)，\(B\) 是三角形内角，所以 \( B=A-B \)，故 \( A=2B \).

（2）因为 \( S=\frac{a^2}{4} \)，所以 \( \frac12bc\sin A=\frac{a^2}{4} \)，即 \( 2bc\sin A=a^2 \). 由正弦定理可得 \(2\sin B\sin C=\sin A=\sin2B\)，所以 \( \sin C=\cos B \). 于是 \( B+C=90^\circ \) 或 \( C=B+90^\circ \). 又 \( A=2B \)，所以 \( A=90^\circ \) 或 \( A=45^\circ \).
\end{solution}

\begin{problem}
如图，在三棱台 \( ABC-DEF \) 中，已知平面 \( BCFE\perp \) 平面 \(ABC\)，\( \angle ACB=90^\circ \)，\( BE=EF=FC=1 \)，\( BC=2 \)，\( AC=3 \).

\end{enumerate}

（2）【法一】过点 \(F\) 作 \( FQ\perp AK \)，连接 \(BQ\). 因为 \( BF\perp \) 平面 \(ACFD\)，所以 \( BF\perp AK \). 又 \( FQ\perp AK \)，所以 \( AK\perp \) 平面 \(BQF\)，从而 \( BQ\perp AK \). 因此 \( \angle BQF \) 是二面角 \( B-AD-F \) 的平面角. 在直角三角形 \(ACK\) 中，\( AC=3 \)，\( CK=2 \)，可得 \( FQ=\frac{3\sqrt{13}}{13} \). 在直角三角形 \(BQF\) 中，\( BF=\sqrt3 \)，\( FQ=\dfrac{3\sqrt{13}}{13} \)，所以 \( \cos\angle BQF=\frac{\sqrt3}{4} \). 故二面角 \( B-AD-F \) 的余弦值为 \( \frac{\sqrt3}{4} \).

【法二】延长 \(AD\)，\(BE\)，\(CF\) 相交于点 \(K\)，则 \( \triangle BCK \) 为等边三角形. 取 \(BC\) 的中点 \(O\)，则 \( KO\perp BC \). 又平面 \( BCFE\perp \) 平面 \(ABC\)，所以 \( KO\perp \) 平面 \(BAC\). 以点 \(O\) 为原点，分别以 \(OB\)，\(OK\) 的方向为 \( x \)，\( z \) 轴正方向，建立空间直角坐标系 \( O-xyz \). 可得 \(B(1,0,0)\)，\(C(-1,0,0)\)，\(K(0,0,\sqrt3)\)，\(A(-1,-3,0)\)，\(E\left( \frac12,0,\frac{\sqrt3}{2} \right)\)，\(F\left( -\frac12,0,\frac{\sqrt3}{2} \right)\). 于是 \(\overrightarrow{AC}=(0,3,0)\)，\(\overrightarrow{AK}=(1,3,\sqrt3)\)，\(\overrightarrow{AB}=(2,3,0)\). 设平面 \(ACK\) 的法向量为 \( \bs m=(x_1,y_1,z_1) \)，平面 \(ABK\) 的法向量为 \( \bs n=(x_2,y_2,z_2) \). 由 \( \overrightarrow{AC}\cdot \bs m=0 \)，\( \overrightarrow{AK}\cdot \bs m=0 \)，可取 \( \bs m=(\sqrt3,0,-1) \). 由 \( \overrightarrow{AB}\cdot \bs n=0 \)，\( \overrightarrow{AK}\cdot \bs n=0 \)，可取 \( \bs n=(3,-2,\sqrt3) \). 所以 \(\cos\langle \bs m,\bs n\rangle=\frac{\bs m\cdot \bs n}{|\bs m||\bs n|}=\frac{\sqrt3}{4}\). 故二面角 \( B-AD-F \) 的余弦值为 \( \frac{\sqrt3}{4} \).
\end{solution}

\begin{problem}
如图，设椭圆 \( C:\dfrac{x^2}{a^2}+y^2=1 \)（\( a>1 \)）.

\begin{enumerate}
\item 求直线 \( y=kx+1 \) 被椭圆截得到的弦长（用 \(a\)，\(k\) 表示）；
\item 若任意以点 \( A(0,1) \) 为圆心的圆与椭圆至多有三个公共点，求椭圆的离心率的取值范围.
\end{enumerate}

\bitmapfigure[max width=0.42\textwidth]{img/2016/zhejiang_science/q19_fig1.png}

\end{problem}

\begin{solution}
（1）由题意 \( \left\{\begin{array}{l} y=kx+1,\\[2pt] \dfrac{x^2}{a^2}+y^2=1 \end{array}\right. \) 消去 \( y \) 得 \( (1+a^2k^2)x^2+2ka^2x=0 \). 所以 \( x_1=0 \)，\( x_2=-\frac{2ka^2}{1+a^2k^2} \). 故直线 \( y=kx+1 \) 被椭圆截得到的弦长为 \( \sqrt{1+k^2}\,|x_1-x_2|=\frac{2a^2|k|}{1+a^2k^2}\sqrt{1+k^2} \).

（2）假设圆 \(A\) 与椭圆有 \( 4 \) 个公共点. 由对称性可设 \( y \) 轴左侧的椭圆上有两个不同的点 \(P\)，\(Q\)，满足 \( |AP|=|AQ| \). 记直线 \(AP\)，\(AQ\) 的斜率分别为 \(k_1\)，\(k_2\)，且 \(k_1\)，\(k_2>0\)，\( k_1\ne k_2 \). 由（1）知 \( |AP|=\frac{2a^2|k_1|\sqrt{1+k_1^2}}{1+a^2k_1^2} \)，\( |AQ|=\frac{2a^2|k_2|\sqrt{1+k_2^2}}{1+a^2k_2^2} \). 由 \( |AP|=|AQ| \) 得 \((k_1^2-k_2^2)\bigl[1+k_1^2+k_2^2+a^2(2-a^2)k_1^2k_2^2\bigr]=0\). 因为 \( k_1\ne k_2 \)，所以 \(1+k_1^2+k_2^2+a^2(2-a^2)k_1^2k_2^2=0\). 该式关于 \(k_1\)，\(k_2\) 有解的必要条件是 \( a>\sqrt2 \). 因此，任意以点 \( A(0,1) \) 为圆心的圆与椭圆至多有三个公共点的充要条件为 \(1<a\le \sqrt2\). 又椭圆的离心率为 \( e=\frac{\sqrt{a^2-1}}{a} \)，所以所求离心率的取值范围是 \( 0<e\le \frac{\sqrt2}{2} \).
\end{solution}

\begin{problem}
设数列满足 \( \left|a_n-\dfrac{a_{n+1}}{2}\right|\le 1 \)，\( n\in\N^* \).

\begin{enumerate}
\item 求证：\( |a_n|\ge 2^{n-1}\left( |a_1|-2 \right) \)（\( n\in\N^* \)）；
\item 若 \( |a_n|\le \left( \dfrac32 \right)^n \)，\( n\in\N^* \)，证明：\( |a_n|\le 2 \)，\( n\in\N^* \).
\end{enumerate}

\end{problem}

\begin{solution}
（1）证明：因为
\( \left|a_n-\frac{a_{n+1}}{2}\right|\le 1 \)，
所以
\( |a_n|-\frac12|a_{n+1}|\le 1 \)，
即
\(\frac{|a_n|}{2^n}-\frac{|a_{n+1}|}{2^{n+1}}\le \frac{1}{2^n}\)，\(n\in\N^*\).

于是
\[
\begin{aligned}
\frac{|a_1|}{2}-\frac{|a_n|}{2^n}
&=\left( \frac{|a_1|}{2}-\frac{|a_2|}{2^2} \right)+\left( \frac{|a_2|}{2^2}-\frac{|a_3|}{2^3} \right)+\cdots \\
&\quad+\left( \frac{|a_{n-1}|}{2^{n-1}}-\frac{|a_n|}{2^n} \right)
\end{aligned}
\]
所以 \( \frac{|a_1|}{2}-\frac{|a_n|}{2^n}\le \frac12+\frac{1}{2^2}+\cdots+\frac{1}{2^n}=1-\frac{1}{2^n}<1 \). 所以 \(|a_n|\ge 2^{n-1}\left( |a_1|-2 \right)\)，\(n\in\N^*\).

（2）证明：任取 \( n\in\N^* \)，由（1）知，对于任意 \( m>n \)，
\[
\begin{aligned}
\frac{|a_n|}{2^n}-\frac{|a_m|}{2^m}
&=\left( \frac{|a_n|}{2^n}-\frac{|a_{n+1}|}{2^{n+1}} \right)+\left( \frac{|a_{n+1}|}{2^{n+1}}-\frac{|a_{n+2}|}{2^{n+2}} \right)+\cdots \\
&\quad+\left( \frac{|a_{m-1}|}{2^{m-1}}-\frac{|a_m|}{2^m} \right) \\
&\le \frac{1}{2^n}+\frac{1}{2^{n+1}}+\cdots+\frac{1}{2^{m-1}} \\
&<\frac{1}{2^{n-1}}
\end{aligned}
\]

所以
\[
\begin{aligned}
|a_n|
&<\left( \frac{1}{2^{n-1}}+\frac{|a_m|}{2^m} \right)\cdot 2^n \\
&\le \left[ \frac{1}{2^{n-1}}+\frac12\left( \frac34 \right)^m \right]\cdot 2^n
=2+\left( \frac34 \right)^m\cdot 2^n. \circled{1}
\end{aligned}
\]

下面证明 \( |a_n|\le 2 \).

否则，存在 \( n_0\in\N^* \)，使得 \( |a_{n_0}|>2 \).

取正整数 \(m_0\)，使 \(m_0>\log_{\frac34}\dfrac{|a_{n_0}|-2}{2^{n_0}}\)
且 \(m_0>n_0\)，则
\( 2^{n_0}\left( \frac34 \right)^{m_0}<|a_{n_0}|-2 \)，
这与 \circled{1} 式矛盾.

综上，对于任意 \( n\in\N^* \)，都有
\( |a_n|\le 2 \).
\end{solution}
