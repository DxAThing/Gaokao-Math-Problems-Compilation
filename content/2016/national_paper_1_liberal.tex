\section{单选题}

\begin{problem}
设集合 \(A=\{1,3,5,7\}\)，\(B=\{x\mid 2\le x\le 5\}\)，则 \(A\cap B=\)（\quad）

\choices
    {\(\{1,3\}\)}
    {\(\{3,5\}\)}
    {\(\{5,7\}\)}
    {\(\{1,7\}\)}

\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
由 \(A\cap B=\{3,5\}\)，故选 B.
\end{solution}

\begin{problem}
设 \((1+2\i)(a+\i)\) 的实部与虚部相等，其中 \(a\) 为实数，则 \(a=\)（\quad）

\choices
    {\(-3\)}
    {\(-2\)}
    {\(2\)}
    {\(3\)}

\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
计算得 \((1+2\i)(a+\i)=a-2+(2a+1)\i\)，由实部与虚部相等得 \(a-2=2a+1\)，解得 \(a=-3\).
\end{solution}

\begin{problem}
为美化环境，从红，黄，白，紫 4 种颜色的花中任选 2 种花种在一个花坛中，余下的 2 种花种在另一个花坛中，则红色和紫色的花不在同一花坛的概率是（\quad）

\choices
    {\(\dfrac13\)}
    {\(\dfrac12\)}
    {\(\dfrac23\)}
    {\(\dfrac34\)}

\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
共有 \(\mathrm{C}_4^2=6\) 种分法，其中红紫同坛有 2 种，因此所求概率为 \(\dfrac{4}{6}=\dfrac23\).
\end{solution}

\begin{problem}
\(\triangle ABC\) 的内角 \(A\)，\(B\)，\(C\) 的对边分别为 \(a\)，\(b\)，\(c\). 已知 \(a=\dfrac{4}{\sqrt3}\)，\(c=2\)，\(\cos A=\dfrac{\sqrt3}{2}\)，则 \(b=\)（\quad）

\choices
    {\(1\)}
    {\(2\)}
    {\(\sqrt2\)}
    {\(3\)}

\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
由余弦定理 \( \cos A=\dfrac{b^2+c^2-a^2}{2bc}=\dfrac{b^2+4-\frac{16}{3}}{4b} \)，整理得 \(3b^2-8b-3=0\)，取正根得 \(b=3\).
\end{solution}

\begin{problem}
直线 \(l\) 经过椭圆的一个顶点和一个焦点，若椭圆中心到 \(l\) 的距离为其短轴长的 \(\dfrac14\)，则该椭圆的离心率为（\quad）

\choices
    {\(\dfrac13\)}
    {\(\dfrac12\)}
    {\(\dfrac23\)}
    {\(\dfrac{\sqrt3}{2}\)}

\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
设椭圆为 \(\dfrac{x^2}{a^2}+\dfrac{y^2}{b^2}=1\)（\(a>b>0\)），取焦点 \(F(c,0)\) 与短轴端点 \(V(0,b)\)，直线 \(FV\) 到原点距离为 \(\dfrac{bc}{\sqrt{b^2+c^2}}=\dfrac b2\)，故 \(b^2=3c^2\)，又 \(a^2=b^2+c^2=4c^2\)，所以 \(e=\dfrac ca=\dfrac12\).
\end{solution}

\section{填空题}

\begin{problem}
设向量 \(\bs a=(x,x+1)\)，\(\bs b=(1,2)\)，且 \(\bs a\perp\bs b\)，则 \(x=\fillinblank{}\).

\end{problem}

\begin{answer}
\(-\dfrac23\)
\end{answer}

\begin{solution}
由 \(\bs a\cdot\bs b=x+2x+2=0\)，解得 \(x=-\dfrac23\).
\end{solution}

\begin{problem}
已知 \(\theta\) 是第四象限角，且 \(\sin\!\left(\theta+\dfrac{\pi}{4}\right)=\dfrac35\)，则 \(\tan\!\left(\theta-\dfrac{\pi}{4}\right)=\fillinblank{}\).

\end{problem}

\begin{answer}
\(-\dfrac43\)
\end{answer}

\begin{solution}
由 \(\sin(\theta+\dfrac{\pi}{4})=\dfrac35\) 且 \(\theta\) 在第四象限，可得 \(\sin\theta=-\dfrac{1}{\sqrt{10}}\)，\(\cos\theta=\dfrac{3}{\sqrt{10}}\)，于是 \(\tan\!\left(\theta-\frac{\pi}{4}\right)=\frac{\tan\theta-1}{1+\tan\theta}=-\frac43\).
\end{solution}

\begin{problem}
设直线 \(y=x+2a\) 与圆 \(C:x^2+y^2-2ay-2=0\) 相交于 \(A\)，\(B\) 两点，若 \(|AB|=2\)，则圆 \(C\) 的面积为\fillinblank{}.

\end{problem}

\begin{answer}
\(4\pi\)
\end{answer}

\begin{solution}
圆心为 \((0,a)\)，半径 \(r=\sqrt{a^2+2}\). 由弦长公式可得 \(d=\dfrac{\sqrt2|a|}{2}\)，联立 \(|AB|=2\) 得 \(a^2=2\)，故 \(r=2\)，面积为 \(4\pi\).
\end{solution}

\begin{problem}
某高科技企业生产产品 \(A\) 和产品 \(B\) 需要甲，乙两种新型材料. 生产一件产品 \(A\) 需要甲材料 \(1.5\text{ kg}\)，乙材料 \(1\text{ kg}\)，用 \(5\) 个工时；生产一件产品 \(B\) 需要甲材料 \(0.5\text{ kg}\)，乙材料 \(0.3\text{ kg}\)，用 \(3\) 个工时，生产一件产品 \(A\) 的利润为 \(2100\) 元，生产一件产品 \(B\) 的利润为 \(900\) 元. 该企业现有甲材料 \(150\text{ kg}\)，乙材料 \(90\text{ kg}\)，则在不超过 \(600\) 个工时的条件下，生产产品 \(A\)，产品 \(B\) 的利润之和的最大值为\fillinblank{} 元.

\end{problem}

\begin{answer}
\(216000\)
\end{answer}

\begin{solution}
设生产 \(A\) 产品 \(x\) 件，\(B\) 产品 \(y\) 件，则约束为
\[
\begin{cases}
1.5x+0.5y\le 150,\\
x+0.3y\le 90,\\
5x+3y\le 600,\\
x,y\ge 0.
\end{cases}
\]
目标函数 \(z=2100x+900y\) 在可行域顶点 \((60,100)\) 处取得最大值，故最大利润为 \(216000\) 元.
\end{solution}
\section{解答题}

\begin{problem}
已知 \(\{a_n\}\) 是公差为 \(3\) 的等差数列，数列 \(\{b_n\}\) 满足 \(b_1=1\)，\(b_2=\dfrac13\)，\(a_nb_{n+1}+b_{n+1}=nb_n\).

\begin{enumerate}
\item 求 \(\{a_n\}\) 的通项公式；
\item 求 \(\{b_n\}\) 的前 \(n\) 项和.
\end{enumerate}

\end{problem}

\begin{solution}
（1）由 \(a_nb_{n+1}+b_{n+1}=nb_n\)，当 \(n=1\) 时，\(a_1b_2+b_2=b_1\).
因为 \(b_1=1\)，\(b_2=\dfrac13\)，所以 \(\frac{a_1+1}{3}=1\)，从而 \(a_1=2\).
又 \(\{a_n\}\) 是公差为 \(3\) 的等差数列，所以 \(a_n=a_1+(n-1)\cdot 3=3n-1\).

（2）由（1）可得 \(a_n=3n-1\)，代入已知递推式，得 \((3n-1)b_{n+1}+b_{n+1}=nb_n\)，即 \(3nb_{n+1}=nb_n\).
因此 \(b_{n+1}=\frac13 b_n\)，所以数列 \(\{b_n\}\) 是以 \(1\) 为首项，\(\dfrac13\) 为公比的等比数列，从而 \(b_n=\left(\frac13\right)^{n-1}\).
故其前 \(n\) 项和为 \(S_n=\frac{1-\left(\frac13\right)^n}{1-\frac13} =\frac32\left(1-3^{-n}\right)\).
\end{solution}

\begin{problem}
如图，已知正三棱锥 \(P-ABC\) 的侧面是直角三角形，\(PA=6\)，顶点 \(P\) 在平面 \(ABC\) 内的正投影为点 \(D\)，\(D\) 在平面 \(PAB\) 内的正投影为点 \(E\)，连接 \(PE\) 并延长交 \(AB\) 于点 \(G\).

\begin{enumerate}
\item 证明：\(G\) 是 \(AB\) 的中点；
\item 在图中作出点 \(E\) 在平面 \(PAC\) 内的正投影 \(F\)（说明作法及理由），并求四面体 \(PDEF\) 的体积.
\end{enumerate}

\bitmapfigure[max width=0.48\linewidth,max totalheight=4.8cm]{img/2016/national_paper_1_liberal/q19_fig1.png}

\end{problem}

\begin{solution}
（1）证明：因为 \(P-ABC\) 为正三棱锥，且 \(D\) 为顶点 \(P\) 在平面 \(ABC\) 内的正投影，\(PD\perp \text{平面 }ABC\)，所以 \(PD\perp AB\).
又 \(E\) 为 \(D\) 在平面 \(PAB\) 内的正投影，所以 \(DE\perp \text{平面 }PAB\)，从而 \(DE\perp AB\).
因为 \(PD\cap DE=D\)，所以 \(AB\perp \text{平面 }PDE\).
连接 \(PE\) 并延长交 \(AB\) 于点 \(G\)，则 \(PE\subset \text{平面 }PDE\)，故 \(AB\perp PG\).
又因为 \(PA=PB\)，所以 \(G\) 是 \(AB\) 的中点.

（2）在平面 \(PAB\) 内，过点 \(E\) 作 \(EF\parallel PB\) 交 \(PA\) 于点 \(F\)，则 \(F\) 即为点 \(E\) 在平面 \(PAC\) 内的正投影.
因为正三棱锥 \(P-ABC\) 的侧面是直角三角形，所以 \(PB\perp PA\)，\(PB\perp PC\).
又 \(EF\parallel PB\)，所以 \(EF\perp PA\)，\(EF\perp PC\).
因此 \(EF\perp \text{平面 }PAC\)，即点 \(F\) 为点 \(E\) 在平面 \(PAC\) 内的正投影.
连接 \(CG\). 因为 \(P\) 在平面 \(ABC\) 内的正投影为 \(D\)，所以 \(D\) 是正三角形 \(ABC\) 的中心. 由（1）知 \(G\) 是 \(AB\) 的中点，所以 \(D\) 在 \(CG\) 上，故 \(CD=\frac23 CG\).
又由题设可得 \(PC\perp \text{平面 }PAB\)，\(DE\perp \text{平面 }PAB\)，所以 \(DE\parallel PC\).
因此 \(PE=\frac23 PG\)，\(DE=\frac13 PC\).
由已知，正三棱锥的侧面是直角三角形且 \(PA=6\)，可得 \(DE=2\)，\(PG=3\sqrt2\)，\(PE=2\sqrt2\).
在等腰直角三角形 \(EFP\) 中，可得 \(EF=PF=2\).
所以 \(S_{\triangle PEF}=\frac12\cdot EF\cdot PF=\frac12\cdot 2\cdot 2=2\)，从而 \(V_{PDEF}=\frac13\cdot DE\cdot S_{\triangle PEF} =\frac13\cdot 2\cdot 2 =\frac43\).
\end{solution}

\begin{problem}
某公司计划购买 \(1\) 台机器，该种机器使用三年后即被淘汰. 机器有一易损零件，在购进机器时，可以额外购买这种零件作为备件，每个 \(200\) 元. 在机器使用期间，如果备件不足再购买，则每个 \(500\) 元. 现需决策在购买机器时应同时购买几个易损零件，为此搜集并整理了 \(100\) 台这种机器在三年使用期内更换的易损零件数，得如图柱状图：

记 \(x\) 表示 \(1\) 台机器在三年使用期内需更换的易损零件数，\(y\) 表示 \(1\) 台机器在购买易损零件上所需的费用（单位：元），\(n\) 表示购机的同时购买的易损零件数.

\begin{enumerate}
\item 若 \(n=19\)，求 \(y\) 与 \(x\) 的函数解析式；
\item 若要求“需更换的易损零件数不大于 \(n\)”的频率不小于 \(0.5\)，求 \(n\) 的最小值；
\item 假设这 \(100\) 台机器在购机的同时每台都购买 \(19\) 个易损零件，或每台都购买 \(20\) 个易损零件，分别计算这 \(100\) 台机器在购买易损零件上所需费用的平均数，以此作为决策依据，购买 \(1\) 台机器的同时应购买 \(19\) 个还是 \(20\) 个易损零件？
\end{enumerate}

\texfigure{img_repaint/2016/national_paper_1_liberal/q19_fig2.tex}

\end{problem}

\begin{solution}
（1）当 \(n=19\) 时，
\[
y=
\begin{cases}
3800, & x\le 19,\\
3800+500(x-19), & x>19.
\end{cases}
\]

（2）由柱状图知，更换的易损零件数为 \(16\) 个，\(17\) 个，\(18\) 个，\(19\) 个，\(20\) 个，\(21\) 个的频率分别为
\(0.06\)，\(\ 0.16\)，\(\ 0.24\)，\(\ 0.24\)，\(\ 0.20\)，\(\ 0.10\).
于是“更换的易损零件数不大于 \(18\)”的频率为
\(0.06+0.16+0.24=0.46<0.5\)，
“更换的易损零件数不大于 \(19\)”的频率为
\(0.06+0.16+0.24+0.24=0.70\ge 0.5\).
所以 \(n\) 的最小值为 \(19\).

（3）若这 \(100\) 台机器在购机的同时每台都购买 \(19\) 个易损零件，则所需费用的平均数为
\(\frac{70\times 19\times 200+20\times 4300+10\times 4800}{100}=4000\text{（元）}\).

若这 \(100\) 台机器在购机的同时每台都购买 \(20\) 个易损零件，则所需费用的平均数为
\(\frac{90\times 4000+10\times 4500}{100}=4050\text{（元）}\).
因为
\(4000<4050\)，
所以购买 \(1\) 台机器的同时应购买 \(19\) 个易损零件.
\end{solution}

\begin{problem}
在直角坐标系 \(xOy\) 中，直线 \(l:y=t\)（\(t\ne 0\)） 交 \(y\) 轴于点 \(M\)，交抛物线 \(C:y^2=2px\)（\(p>0\)） 于点 \(P\)，\(M\) 关于点 \(P\) 的对称点为 \(N\)，连结 \(ON\) 并延长交 \(C\) 于点 \(H\).

\begin{enumerate}
\item 求 \(\dfrac{|OH|}{|ON|}\)；
\item 除 \(H\) 以外，直线 \(MH\) 与 \(C\) 是否有其它公共点？说明理由.
\end{enumerate}

\end{problem}

\begin{solution}
（1）将直线 \(l:y=t\) 与抛物线 \(C:y^2=2px\) 联立，得
\(P\left(\frac{t^2}{2p},t\right)\).
因为点 \(N\) 关于点 \(P\) 与点 \(M(0,t)\) 对称，所以
\(N\left(\frac{t^2}{p},t\right)\).
于是直线 \(ON\) 的方程为
\(y=\frac{p}{t}x\).
将它与抛物线方程联立，得另一交点
\(H\left(\frac{2t^2}{p},2t\right)\).
因此
\(\frac{|OH|}{|ON|}=2\).

（2）由（1）知
\(k_{MH}=\frac{2t-t}{\frac{2t^2}{p}-0}=\frac{p}{2t}\)，
故直线 \(MH\) 的方程为
\(y=\frac{p}{2t}x+t\).
将它与抛物线方程
\(y^2=2px\)
联立，消去 \(x\)，得
\(y^2-4ty+4t^2=0\).
其判别式为
\(\Delta=16t^2-16t^2=0\).
因此直线 \(MH\) 与抛物线 \(C\) 除点 \(H\) 外没有其它公共点.
\end{solution}

\begin{problem}
已知函数 \(f(x)=(x-2)\e^x+a(x-1)^2\).

\begin{enumerate}
\item 讨论 \(f(x)\) 的单调性；
\item 若 \(f(x)\) 有两个零点，求 \(a\) 的取值范围.
\end{enumerate}

\end{problem}

\begin{solution}
（1）由
\(f(x)=(x-2)\e^x+a(x-1)^2\)
得
\(f'(x)=(x-1)\e^x+2a(x-1)=(x-1)(\e^x+2a)\).

当 \(a\ge 0\) 时，由 \(f'(x)>0\) 可得 \(x>1\)，由 \(f'(x)<0\) 可得 \(x<1\)，
所以 \(f(x)\) 在 \((-\infty,1)\) 递减，在 \((1,+\infty)\) 递增.

当 \(a<0\) 时：

若 \(a=-\dfrac{\e}{2}\)，则 \(f'(x)\ge 0\) 恒成立，所以 \(f(x)\) 在 \(\mathbb{R}\) 上递增；

若 \(a<-\dfrac{\e}{2}\)，则由 \(f'(x)>0\) 可得 \(x<1\) 或 \(x>\ln(-2a)\)，由 \(f'(x)<0\) 可得 \(1<x<\ln(-2a)\)，
所以 \(f(x)\) 在 \((-\infty,1)\) 和 \((\ln(-2a),+\infty)\) 递增，在 \((1,\ln(-2a))\) 递减；

若 \(-\dfrac{\e}{2}<a<0\)，则由 \(f'(x)>0\) 可得 \(x<\ln(-2a)\) 或 \(x>1\)，由 \(f'(x)<0\) 可得 \(\ln(-2a)<x<1\)，
所以 \(f(x)\) 在 \((-\infty,\ln(-2a))\) 和 \((1,+\infty)\) 递增，在 \((\ln(-2a),1)\) 递减.

（2）当 \(a>0\) 时，由（1）知 \(f(x)\) 在 \((-\infty,1)\) 递减，在 \((1,+\infty)\) 递增，
且
\(f(1)=-\e<0\)，
又当 \(x\to-\infty\) 时，\(f(x)\to+\infty\)；当 \(x\to+\infty\) 时，\(f(x)\to+\infty\)，
所以 \(f(x)\) 有两个零点.

当 \(a=0\) 时，
\(f(x)=(x-2)\e^x\)，
只有一个零点 \(x=2\).

当 \(a<0\) 时，若 \(x\le 1\)，则
\(x-2<0\)，\((x-1)^2\ge 0\)，
从而
\(f(x)=(x-2)\e^x+a(x-1)^2<0\).

再结合（1）中的单调性讨论可得，\(f(x)\) 至多只有一个零点，因此不可能有两个零点.

综上，\(f(x)\) 有两个零点时，
\(a\in(0,+\infty)\).
\end{solution}

\begin{problem}
如图，\(\triangle OAB\) 是等腰三角形，\(\angle AOB=120^\circ\). 以 \(O\) 为圆心，\(\dfrac12 OA\) 为半径作圆.

\begin{enumerate}
\item 证明：直线 \(AB\) 与 \(\odot O\) 相切；
\item 点 \(C\)，\(D\) 在 \(\odot O\) 上，且 \(A\)，\(B\)，\(C\)，\(D\) 四点共圆，证明：\(AB\parallel CD\).
\end{enumerate}

\bitmapfigure[max width=0.38\linewidth,max totalheight=4.8cm]{img/2016/national_paper_1_liberal/q22_fig1.png}

\end{problem}

\begin{solution}
（1）设 \(K\) 为 \(AB\) 中点，连接 \(OK\).
因为 \(\triangle OAB\) 是等腰三角形，且 \(OA=OB\)，\(\angle AOB=120^\circ\)，所以
\(OK\perp AB\)，\(\angle A=30^\circ\).
于是
\(OK=OA\sin30^\circ=\frac12 OA\).
故直线 \(AB\) 与 \(\odot O\) 相切.

（2）因为 \(OA=2OD\)，所以 \(O\) 不是 \(A\)，\(B\)，\(C\)，\(D\) 四点所在圆的圆心. 设 \(T\) 是 \(A\)，\(B\)，\(C\)，\(D\) 四点所在圆的圆心.
因为
\(OA=OB\)，\(TA=TB\)，
所以 \(OT\) 为 \(AB\) 的中垂线.
又因为
\(OC=OD\)，\(TC=TD\)，
所以 \(OT\) 为 \(CD\) 的中垂线.
因此
\(AB\parallel CD\).
\end{solution}

\begin{problem}

在直角坐标系 \(xOy\) 中，曲线 \(C_1\) 的参数方程为 \(x=a\cos t\)，\(y=1+a\sin t\)（\(t\) 为参数，\(a>0\)）.
在以坐标原点为极点，\(x\) 轴正半轴为极轴的极坐标系中，曲线 \(C_2:\rho=4\cos\theta\).

\begin{enumerate}
\item 说明 \(C_1\) 是哪种曲线，并将 \(C_1\) 的方程化为极坐标方程；
\item 直线 \(C_3\) 的极坐标方程为 \(\theta=\alpha_0\)，其中 \(\alpha_0\) 满足 \(\tan\alpha_0=2\)，若曲线 \(C_1\) 与 \(C_2\) 的公共点都在 \(C_3\) 上，求 \(a\).
\end{enumerate}

\end{problem}

\begin{solution}
（1）由参数方程
\(x=a\cos t\)，\(y=1+a\sin t\)
可得
\(x^2+(y-1)^2=a^2\).
所以 \(C_1\) 是以 \((0,1)\) 为圆心，\(a\) 为半径的圆.
将其化为一般式，得
\(x^2+y^2-2y+1-a^2=0\).
由
\(x^2+y^2=\rho^2\)，\(y=\rho\sin\theta\)，
可得 \(C_1\) 的极坐标方程为
\(\rho^2-2\rho\sin\theta+1-a^2=0\).

（2）由
\(C_2:\rho=4\cos\theta\)
两边同时乘 \(\rho\)，得
\(\rho^2=4\rho\cos\theta\)，
即
\(x^2+y^2=4x\).
又由 \(\tan\alpha_0=2\)，得 \(C_3\) 的直角坐标方程为
\(y=2x\).
因为曲线 \(C_1\) 与 \(C_2\) 的公共点都在 \(C_3\) 上，所以 \(y=2x\) 为圆 \(C_1\) 与圆 \(C_2\) 的公共弦所在直线.
将 \(C_1\) 与 \(C_2\) 的直角坐标方程相减，得公共弦所在直线为
\(4x-2y+1-a^2=0\).
它与 \(y=2x\) 表示同一直线，所以
\(1-a^2=0\).
又因为 \(a>0\)，所以
\(a=1\).
\end{solution}

\begin{problem}

已知函数 \(f(x)=|x+1|-|2x-3|\).

\begin{enumerate}
\item 在图中画出 \(y=f(x)\) 的图象；
\item 求不等式 \(|f(x)|>1\) 的解集.
\end{enumerate}

\bitmapfigure[max width=0.45\linewidth,max totalheight=4.8cm]{img/2016/national_paper_1_liberal/q24_fig1.png}

\end{problem}

\begin{solution}
（1）由绝对值的定义可得
\[
f(x)=
\begin{cases}
x-4, & x\le -1,\\
3x-2, & -1<x<\dfrac32,\\
4-x, & x\ge \dfrac32.
\end{cases}
\]
由分段函数的图象画法，可得 \(f(x)\) 的图象为三段直线组成的折线.

（2）由 \(|f(x)|>1\)，分类讨论：

当 \(x\le -1\) 时，
\(|x-4|>1\)，
解得 \(x>5\) 或 \(x<3\)，结合 \(x\le -1\)，得 \(x\le -1\)；

当 \(-1<x<\dfrac32\) 时，
\(|3x-2|>1\)，
解得 \(x>1\) 或 \(x<\dfrac13\)，结合 \(-1<x<\dfrac32\)，得
\(-1<x<\frac13\)或\(1<x<\frac32;\)

当 \(x\ge \dfrac32\) 时，
\(|4-x|>1\)，
解得 \(x>5\) 或 \(x<3\)，结合 \(x\ge \dfrac32\)，得
\(\frac32\le x<3\)或\(x>5\).

综上，
\(x\in\left(-\infty,\frac13\right)\cup(1,3)\cup(5,+\infty)\).
\end{solution}
