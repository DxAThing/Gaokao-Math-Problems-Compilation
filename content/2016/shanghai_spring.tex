\section{填空题}

\begin{problem}
复数 \(3+4\i\)（\(\i\) 为虚数单位）的实部是\fillinblank{}.

\end{problem}

\begin{answer}
\(3\)
\end{answer}

\begin{solution}
复数 \(3+4\i\) 的实部是 \(3\). 故答案为 \(3\).
\end{solution}
\begin{problem}
若 \(\log_2(x+1)=3\)，则 \(x=\)\fillinblank{}.

\end{problem}

\begin{answer}
\(7\)
\end{answer}

\begin{solution}
由 \(\log_2(x+1)=3\) 得 \(x+1=2^3=8\)，所以 \(x=7\). 故答案为 \(7\).
\end{solution}
\begin{problem}
直线 \(y=x-1\) 与直线 \(y=2\) 的夹角为\fillinblank{}.

\end{problem}

\begin{answer}
\(\dfrac{\pi}{4}\)
\end{answer}

\begin{solution}
直线 \(y=x-1\) 的斜率为 \(1\)，直线 \(y=2\) 的斜率为 \(0\). 故夹角满足 \(\tan\theta=1\)，所以 \(\theta=\dfrac{\pi}{4}\).
\end{solution}
\begin{problem}
函数 \(f(x)=\sqrt{x-2}\) 的定义域为\fillinblank{}.

\end{problem}

\begin{answer}
\([2,+\infty)\)
\end{answer}

\begin{solution}
要使 \(\sqrt{x-2}\) 有意义，需 \(x-2\ge0\)，即 \(x\ge2\). 故定义域为 \([2,+\infty)\).
\end{solution}
\begin{problem}
三阶行列式
\[
\begin{vmatrix}
1 & -3 & 5\\
4 & 0 & 0\\
-1 & 2 & 1
\end{vmatrix}
\]
中，元素 \(5\) 的代数余子式的值为\fillinblank{}.

\end{problem}

\begin{answer}
\(8\)
\end{answer}

\begin{solution}
元素 \(5\) 位于第 \(1\) 行第 \(3\) 列，其代数余子式为
\[
(-1)^{1+3}\begin{vmatrix}
4 & 0\\
-1 & 2
\end{vmatrix}
=4\cdot2-0\cdot(-1)=8.
\]
故答案为 \(8\).
\end{solution}
\begin{problem}
函数 \(f(x)=\dfrac1x+a\) 的反函数的图像经过点 \((2,1)\)，则实数 \(a=\)\fillinblank{}.

\end{problem}

\begin{answer}
\(1\)
\end{answer}

\begin{solution}
反函数的图像经过点 \((2,1)\)，说明原函数图像经过点 \((1,2)\). 代入 \(f(x)=\dfrac1x+a\) 得
\[
f(1)=1+a=2,
\]
所以 \(a=1\).
\end{solution}
\begin{problem}
在 \(\triangle ABC\) 中，若 \(A=30^\circ\)，\(B=45^\circ\)，\(BC=\sqrt6\)，则 \(AC=\)\fillinblank{}.

\end{problem}

\begin{answer}
\(2\sqrt3\)
\end{answer}

\begin{solution}
由正弦定理，
\[
\frac{AC}{\sin45^\circ}=\frac{BC}{\sin30^\circ}.
\]
所以
\[
AC=\sqrt6\cdot\frac{\sin45^\circ}{\sin30^\circ}
=\sqrt6\cdot\frac{\sqrt2/2}{1/2}
=2\sqrt3.
\]
故答案为 \(2\sqrt3\).
\end{solution}
\begin{problem}
\(4\) 个人排成一排照相，不同排列方式的种数为\fillinblank{}（结果用数值表示）.

\end{problem}

\begin{answer}
\(24\)
\end{answer}

\begin{solution}
\(4\) 个人全排列共有 \(4!=24\) 种. 故答案为 \(24\).
\end{solution}
\begin{problem}
无穷等比数列 \(\{a_n\}\) 的首项为 \(2\)，公比为 \(\dfrac13\)，则 \(\{a_n\}\) 的各项的和为\fillinblank{}.

\end{problem}

\begin{answer}
\(3\)
\end{answer}

\begin{solution}
无穷等比数列前无穷项和为
\[
\frac{a_1}{1-q}=\frac2{1-\frac13}=3.
\]
故答案为 \(3\).
\end{solution}
\begin{problem}
若 \(2+\i\)（\(\i\) 为虚数单位）是关于 \(x\) 的实系数一元二次方程 \(x^2+ax+5=0\) 的一个虚根，则 \(a=\)\fillinblank{}.

\end{problem}

\begin{answer}
\(-4\)
\end{answer}

\begin{solution}
实系数方程的虚根共轭成对出现，所以 \(2-\i\) 也是根. 两根之和为
\[
(2+\i)+(2-\i)=4=-a,
\]
故 \(a=-4\).
\end{solution}
\begin{problem}
函数 \(y=x^2-2x+1\) 在区间 \([0,m]\) 上的最小值为 \(0\)，最大值为 \(1\)，则实数 \(m\) 的取值范围是\fillinblank{}.

\end{problem}

\begin{answer}
\([1,2]\)
\end{answer}

\begin{solution}
因为
\[
y=x^2-2x+1=(x-1)^2,
\]
最小值为 \(0\) 要求区间 \([0,m]\) 包含 \(x=1\)，所以 \(m\ge1\). 又最大值为 \(1\) 要求端点处函数值不超过 \(1\)，即 \((m-1)^2\le1\)，所以 \(m\le2\). 故 \(m\in[1,2]\).
\end{solution}
\begin{problem}
在平面直角坐标系 \(xOy\) 中，点 \(A,B\) 是圆 \(x^2+y^2-6x+5=0\) 上的两个动点，且满足 \(\lvert AB\rvert=2\sqrt3\)，则 \(\lvert\overrightarrow{OA}+\overrightarrow{OB}\rvert\) 的最小值为\fillinblank{}.

\end{problem}

\begin{answer}
\(4\)
\end{answer}

\begin{solution}
圆的标准方程为 \((x-3)^2+y^2=4\)，所以圆心为 \(C(3,0)\)，半径为 \(2\). 设弦 \(AB\) 的中点为 \(M\)，则
\[
\overrightarrow{OA}+\overrightarrow{OB}=2\overrightarrow{OM}.
\]
又 \(|AB|=2\sqrt3\)，所以
\[
CM=\sqrt{2^2-\left(\sqrt3\right)^2}=1.
\]
于是点 \(M\) 在以 \(C(3,0)\) 为圆心、半径为 \(1\) 的圆上，故 \(OM\) 的最小值为 \(3-1=2\). 因此
\[
\min\lvert\overrightarrow{OA}+\overrightarrow{OB}\rvert=2\cdot2=4.
\]
\end{solution}

\begin{problem}
如图，已知正三棱柱 \(ABC-A_1B_1C_1\) 的体积为 \(9\sqrt3\)，底面边长为 \(3\)，求异面直线 \(BC_1\) 与 \(AC\) 所成的角的大小. 

\bitmapfigure[max width=0.26\linewidth,max totalheight=4.8cm]{img/2016/shanghai_spring/q25_fig1.png}

\end{problem}

\begin{solution}
底面正三角形 \(ABC\) 的面积为
\[
S_{\triangle ABC}=\frac{3^2\sqrt3}{4}=\frac{9\sqrt3}{4},
\]
所以棱柱高为
\[
h=\frac{9\sqrt3}{9\sqrt3/4}=4.
\]
取坐标 \(A(0,0,0)\)，\(B(3,0,0)\)，\(C\left(\frac32,\frac{3\sqrt3}{2},0\right)\)，\(C_1\left(\frac32,\frac{3\sqrt3}{2},4\right)\). 于是
\[
\overrightarrow{BC_1}=\left(-\frac32,\frac{3\sqrt3}{2},4\right),\qquad
\overrightarrow{AC}=\left(\frac32,\frac{3\sqrt3}{2},0\right).
\]
故
\[
\cos\theta=\frac{\overrightarrow{BC_1}\cdot\overrightarrow{AC}}{\lvert\overrightarrow{BC_1}\rvert\lvert\overrightarrow{AC}\rvert}=\frac{3}{10},
\]
所以所成角为
\[
\theta=\arccos\frac{3}{10}.
\]
\end{solution}

如图，汽车前灯反射镜与轴截面的交线是抛物线的一部分，灯口所在的圆面与反射镜的轴垂直，灯泡位于抛物线的焦点 \(F\) 处，已知灯口直径是 \(24\text{ cm}\)，灯深 \(10\text{ cm}\)，求灯泡与反射镜的顶点 \(O\) 的距离.

\begin{problem}
已知数列 \(\{a_n\}\) 是公差为 \(2\) 的等差数列. 

\begin{enumerate}
\item 若 \(a_1,a_3,a_4\) 成等比数列，求 \(a_1\) 的值；
\item 设 \(a_1=-19\)，数列 \(\{a_n\}\) 的前 \(n\) 项和为 \(S_n\). 数列 \(\{b_n\}\) 满足 \(b_1=1\)，\(b_{n+1}-b_n=\left(\dfrac12\right)^n\). 记 \(c_n=S_n+2^{n-1}b_n\)（\(n\in\N^*\)），求数列 \(\{c_n\}\) 的最小项 \(c_{n_0}\)（即 \(c_{n_0}\le c_n\) 对任意 \(n\in\N^*\) 成立）. 
\end{enumerate}

\end{problem}

\begin{solution}
(1) 设公差为 \(2\)，则 \(a_3=a_1+4\)，\(a_4=a_1+6\). 由等比中项性质，
\[
(a_1+4)^2=a_1(a_1+6),
\]
整理得
\[
2a_1+16=0,
\]
故 \(a_1=-8\). 

(2) 因为 \(a_1=-19\)，公差为 \(2\)，所以
\[
a_n=-19+2(n-1)=2n-21,
\]
从而
\[
S_n=\frac n2\bigl(2a_1+(n-1)2\bigr)=n(n-20).
\]
又由 \(b_{n+1}-b_n=2^{-n}\)，\(b_1=1\) 得
\[
b_n=1+\sum_{k=1}^{n-1}2^{-k}=2-2^{1-n}.
\]
所以
\[
2^{n-1}b_n=2^n-1,
\]
从而
\[
c_n=n(n-20)+2^n-1=n^2-20n+2^n-1.
\]
于是
\[
c_{n+1}-c_n=(n+1)^2-20(n+1)+2^{n+1}-1-\bigl(n^2-20n+2^n-1\bigr)=2n-19+2^n.
\]
当 \(n=1,2,3\) 时，上式为负；当 \(n\ge4\) 时，上式为正，故 \(\{c_n\}\) 先减后增，最小项在 \(n=4\) 取得. 
计算得
\[
c_4=4^2-20\cdot4+2^4-1=-49.
\]
故最小项为 \(c_4=-49\).
\end{solution}
\begin{problem}
对于函数 \(f(x)\) 与 \(g(x)\)，记集合 \(D_{f>g}=\{x\mid f(x)>g(x)\}\). 

\begin{enumerate}
\item 设 \(f(x)=2\lvert x\rvert\)，\(g(x)=x+3\)，求 \(D_{f>g}\)；
\item 设 \(f_1(x)=x-1\)，\(f_2(x)=\left(\dfrac13\right)^x+a\cdot3^x+1\)，\(h(x)=0\). 如果 \(D_{f_1>h}\cup D_{f_2>h}=\R\)，求实数 \(a\) 的取值范围. 
\end{enumerate}

\end{problem}

\begin{solution}
(1) 当 \(x<0\) 时，\(f(x)=2|x|=-2x\)，不等式变为 \(-2x>x+3\)，即 \(x<-1\)；当 \(x\ge0\) 时，不等式变为 \(2x>x+3\)，即 \(x>3\). 所以
\[
D_{f>g}=(-\infty,-1)\cup(3,+\infty).
\]

(2) 由 \(f_1(x)=x-1\) 可知
\[
D_{f_1>h}=(1,+\infty).
\]
若 \(D_{f_1>h}\cup D_{f_2>h}=\R\)，则必须有 \(D_{f_2>h}\supset(-\infty,1]\). 对 \(x\le1\)，令
\[
t=\left(\frac13\right)^x,
\]
则 \(t\in[\frac13,+\infty)\)，且
\[
f_2(x)>0\iff t+a/t+1>0\iff t^2+t+a>0.
\]
函数 \(t^2+t\) 在 \([\frac13,+\infty)\) 上单调递增，最小值取于 \(t=\frac13\)，故只需
\[
\frac19+\frac13+a>0,
\]
即
\[
a>-\frac49.
\]
所以所求范围为 \(\left(-\frac49,+\infty\right)\).
\end{solution}

\begin{problem}
小明用数列 \(\{a_k\}\) 记录某地区 2015 年 12 月份 \(31\) 天中每天是否下过雨，方法为：当第 \(k\) 天下过雨时，记 \(a_k=1\)；当第 \(k\) 天没下过雨时，记 \(a_k=-1\)（\(1\le k\le31\)）. 他用数列 \(\{b_k\}\) 记录该地区该月每天气象台预报是否有雨，方法为：当预报第 \(k\) 天有雨时，记 \(b_k=1\)；当预报第 \(k\) 天没有雨时，记 \(b_k=-1\)（\(1\le k\le31\)）. 记录完毕后，小明计算出 \(a_1b_1+a_2b_2+a_3b_3+\cdots+a_{31}b_{31}=25\)，那么该月气象台预报准确的总天数为\fillinblank{}.

\end{problem}

\begin{answer}
\(28\)
\end{answer}

若某天预报准确，则 \(a_kb_k=1\)；若预报不准确，则 \(a_kb_k=-1\). 设预报准确的天数为 \(x\)，则不准确的天数为 \(31-x\)，所以

\]

\begin{problem}
对于数列 \(\{a_n\}\) 与 \(\{b_n\}\)，若对数列 \(\{c_n\}\) 的每一项 \(c_k\)，均有 \(c_k=a_k\) 或 \(c_k=b_k\)，则称数列 \(\{c_n\}\) 是 \(\{a_n\}\) 与 \(\{b_n\}\) 的一个“并数列”. 

\begin{enumerate}
\item 设数列 \(\{a_n\}\) 与 \(\{b_n\}\) 的前三项分别为 \(a_1=1,a_2=3,a_3=5,b_1=1,b_2=2,b_3=3\)，若 \(\{c_n\}\) 是 \(\{a_n\}\) 与 \(\{b_n\}\) 的一个“并数列”，求所有可能的有序数组 \((c_1,c_2,c_3)\)；
\item 已知数列 \(\{a_n\},\{c_n\}\) 均为等差数列，\(\{a_n\}\) 的公差为 \(1\)，首项为正整数 \(t\)；\(\{c_n\}\) 的前 \(10\) 项和为 \(-30\)，前 \(20\) 项和为 \(-260\)，若存在唯一的数列 \(\{b_n\}\)，使得 \(\{c_n\}\) 是 \(\{a_n\}\) 与 \(\{b_n\}\) 的一个“并数列”，求 \(t\) 的值所构成的集合. 
\end{enumerate}

\end{problem}

\begin{solution}
(1) 由定义可知，\(c_1\) 只能取 \(1\)；\(c_2\) 可取 \(3\) 或 \(2\)；\(c_3\) 可取 \(5\) 或 \(3\). 故所有可能的有序数组为
\[
(1,3,5),\ (1,3,3),\ (1,2,5),\ (1,2,3).
\]

(2) 设 \(\{c_n\}\) 的公差为 \(d\)，首项为 \(c_1\). 由前 \(10\) 项和、前 \(20\) 项和可得
\[
5(2c_1+9d)=-30,\qquad 10(2c_1+19d)=-260.
\]
解得 \(d=-2\)，\(c_1=6\)，所以
\[
c_n=6-2(n-1)=8-2n.
\]
又 \(\{a_n\}\) 的公差为 \(1\)，首项为 \(t\)，故
\[
a_n=t+n-1.
\]
若 \(\{b_n\}\) 存在且唯一，则每一项都必须有 \(c_n\neq a_n\)，即
\[
t+n-1\neq 8-2n\qquad(n\in\N^*).
\]
化简得
\[
t\neq 9-3n.
\]
由于 \(t\) 为正整数，故只需排除 \(t=6,3\). 因此所求集合为
\[
\{t\in\N^*\mid t\neq3,\ t\neq6\}.
\]
\end{solution}
