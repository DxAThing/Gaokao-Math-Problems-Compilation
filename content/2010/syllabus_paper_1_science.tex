\section{单选题}

\begin{problem}
复数 \(\frac{3 + 2\i}{2 - 3\i} =\)（\quad）

\choices
    {\(\i\)}
    {\(-\i\)}
    {\(12 - 13\i\)}
    {\(12 + 13\i\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
将分式的分母实数化，得
\[
\frac{3+2\i}{2-3\i}
=\frac{(3+2\i)(2+3\i)}{(2-3\i)(2+3\i)}
=\frac{6+9\i+4\i+6\i^{2}}{13}
=\i
\]
所以应选 A.
\end{solution}

\begin{problem}
记 \(\cos (-80^{\circ}) = k\)，那么 \(\tan 100^{\circ} =\)（\quad）

\choices
    {\(\frac{\sqrt{1 - k^2}}{k}\)}
    {\(-\frac{\sqrt{1 - k^2}}{k}\)}
    {\(\frac{k}{\sqrt{1 - k^2}}\)}
    {\(-\frac{k}{\sqrt{1 - k^2}}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
因为 \(k=\cos(-80^{\circ})=\cos80^{\circ}>0\)，且 \(\sin80^{\circ}>0\)，所以
\(\sin80^{\circ}=\sqrt{1-k^{2}}\). 又由诱导公式，\(\tan100^{\circ}=-\tan80^{\circ}\)，从而
\[
\tan100^{\circ}
=-\frac{\sin80^{\circ}}{\cos80^{\circ}}
=-\frac{\sqrt{1-k^{2}}}{k}
\]
所以应选 B.
\end{solution}

\begin{problem}
若变量 \(x\)，\(y\) 满足约束条件 \(\begin{cases}
y \leqslant 1,\\
x + y \geqslant 0,\\
x - y - 2 \leqslant 0,
\end{cases}\) 则 \(z = x - 2y\) 的最大值为（\quad）

\choices
    {\(4\)}
    {\(3\)}
    {\(2\)}
    {\(1\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
由 \(x+y\geqslant0\) 和 \(x-y-2\leqslant0\)，分别得 \(x\geqslant-y\) 和 \(x\leqslant y+2\). 因而必须有 \(-y\leqslant y+2\)，即 \(y\geqslant-1\). 对任意满足条件的 \(x\)，有
\[
z=x-2y\leqslant(y+2)-2y=2-y\leqslant3
\]
当 \(y=-1\)，\(x=1\) 时，三个约束条件均满足，且 \(z=1-2(-1)=3\). 所以最大值为 \(3\)，应选 B.
\end{solution}

\begin{problem}
已知各项均为正数的等比数列 \(\{a_{n}\}\)，\(a_{1}a_{2}a_{3} = 5\)，\(a_{7}a_{8}a_{9} = 10\)，则 \(a_{4}a_{5}a_{6} =\)（\quad）

\choices
    {\(5\sqrt{2}\)}
    {\(7\)}
    {\(6\)}
    {\(4\sqrt{2}\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
设等比数列的公比为 \(q\). 因为各项均为正数，所以 \(q>0\). 由等比数列的性质，
\(a_{1}a_{2}a_{3}=a_{2}^{3}=5\)，\(a_{7}a_{8}a_{9}=a_{8}^{3}=10\).
又 \(a_{8}=a_{2}q^{6}\)，所以 \(q^{18}=2\)，从而 \(q^{9}=\sqrt2\). 因此
\[
a_{4}a_{5}a_{6}=a_{5}^{3}=a_{2}^{3}q^{9}=5\sqrt2
\]
所以应选 A.
\end{solution}

\begin{problem}
\((1 + 2\sqrt{x})^3 (1 - \sqrt[3]{x})^5\) 的展开式中 \(x\) 的系数是（\quad）

\choices
    {\(-4\)}
    {\(-2\)}
    {\(2\)}
    {\(4\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
在 \((1+2\sqrt{x})^{3}\) 中取含 \(i\) 个 \(2\sqrt{x}\) 的项，在
\((1-\sqrt[3]{x})^{5}\) 中取含 \(j\) 个 \(-\sqrt[3]{x}\) 的项时，所得幂为
\(x^{i/2+j/3}\). 要得到 \(x\) 项，必须满足
\(\frac{i}{2}+\frac{j}{3}=1\)，\(0\leqslant i\leqslant3\)，\(0\leqslant j\leqslant5\).
即 \(3i+2j=6\)，只有 \((i,j)=(0,3)\) 和 \((2,0)\). 所以 \(x\) 的系数为
\[
\mathrm C_{3}^{0}\mathrm C_{5}^{3}(-1)^{3}+\mathrm C_{3}^{2}2^{2}=-10+12=2
\]
所以应选 C.
\end{solution}

\begin{problem}
某校开设 \(A\) 类选修课\(3\text{ 门}\)，\(B\) 类选修课\(4\text{ 门}\). 一位同学从中共选\(3\text{ 门}\). 若要求两类课程中各至少选一门，则不同的选法共有（\quad）

\choices
    {\(30\text{ 种}\)}
    {\(35\text{ 种}\)}
    {\(42\text{ 种}\)}
    {\(48\text{ 种}\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
从 \(7\) 门课程中任选 \(3\) 门共有 \(\mathrm C_{7}^{3}\) 种. 其中只选 \(A\) 类课程的选法有
\(\mathrm C_{3}^{3}\) 种，只选 \(B\) 类课程的选法有 \(\mathrm C_{4}^{3}\) 种. 因而两类课程中各至少选一门的选法数为
\[
\mathrm C_{7}^{3}-\mathrm C_{3}^{3}-\mathrm C_{4}^{3}=35-1-4=30
\]
所以应选 A.
\end{solution}

\begin{problem}
正方体 \(ABCD - A_{1}B_{1}C_{1}D_{1}\) 中，\(BB_{1}\) 与平面 \(ACD_{1}\) 所成角的余弦值为（\quad）

\choices
    {\(\frac{\sqrt{2}}{3}\)}
    {\(\frac{\sqrt{3}}{3}\)}
    {\(\frac{2}{3}\)}
    {\(\frac{\sqrt{6}}{3}\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
设正方体棱长为 \(1\)，建立空间直角坐标系，取
\(A(0,0,0)\)，\(B(1,0,0)\)，\(C(1,1,0)\)，\(D_{1}(0,1,1)\). 平面 \(ACD_{1}\) 的两个方向向量可取
\(\overrightarrow{AC}=(1,1,0)\) 和 \(\overrightarrow{AD_{1}}=(0,1,1)\)，故其一个法向量为
\(\bs n=(1,-1,1)\). 直线 \(BB_{1}\) 的方向向量为 \(\bs v=(0,0,1)\). 若直线与平面所成角为 \(\theta\)，则
\(\sin\theta=\frac{|\bs v\cdot\bs n|}{|\bs v||\bs n|}=\frac{1}{\sqrt3}\)，\(\cos\theta=\sqrt{1-\frac13}=\frac{\sqrt6}{3}\).
所以应选 D.
\end{solution}

\begin{problem}
设 \(a = \log_32\)，\(b = \ln 2\)，\(c = 5^{-1/2}\)，则（\quad）

\choices
    {\(a < b < c\)}
    {\(b < c < a\)}
    {\(c < a < b\)}
    {\(c < b < a\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
有 \(c=1/\sqrt5\). 又因为 \(\sqrt5>2\)，所以 \(2^{\sqrt5}>2^{2}=4>3\)，从而
\(\log_{3}2>1/\sqrt5=c\). 另一方面，\(\ln3>1\)，所以
\[
a=\frac{\ln2}{\ln3}<\ln2=b
\]
因此 \(c<a<b\)，应选 C.
\end{solution}

\begin{problem}
已知 \(F_{1}\)，\(F_{2}\) 为双曲线 \(C: x^{2} - y^{2} = 1\) 的左，右焦点，点 \(P\) 在 \(C\) 上，\(\angle F_{1}PF_{2} = 60^{\circ}\)，则 \(P\) 到 \(x\) 轴的距离为（\quad）

\choices
    {\(\frac{\sqrt{3}}{2}\)}
    {\(\frac{\sqrt{6}}{2}\)}
    {\(\sqrt{3}\)}
    {\(\sqrt{6}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
双曲线的焦点为 \(F_{1}(-\sqrt2,0)\)，\(F_{2}(\sqrt2,0)\)，所以
\(F_{1}F_{2}=2\sqrt2\). 设 \(u=PF_{1}\)，\(v=PF_{2}\). 由双曲线定义，\(|u-v|=2\)；关于 \(y\) 轴对称时 \(u\)，\(v\) 互换，故不妨设 \(u-v=2\). 在三角形 \(F_{1}PF_{2}\) 中应用余弦定理，得
\[
8=u^{2}+v^{2}-2uv\cos60^{\circ}=u^{2}+v^{2}-uv
\]
代入 \(u=v+2\)，得 \(v^{2}+2v-4=0\)，因 \(v>0\)，故 \(v=\sqrt5-1\)，\(u=\sqrt5+1\). 于是
\(u+v=2\sqrt5\)，\(u^{2}-v^{2}=4\sqrt5\).
又由坐标计算，\(u^{2}-v^{2}=4\sqrt2\,|x_{P}|\)，所以 \(|x_{P}|=\sqrt{5/2}\). 代入
\(x_{P}^{2}-y_{P}^{2}=1\)，得 \(y_{P}^{2}=3/2\)，故 \(P\) 到 \(x\) 轴的距离为
\(\sqrt6/2\)，应选 B.
\end{solution}

\begin{problem}
已知函数 \( f(x) = |\lg x| \). 若 \( 0 < a < b \)，且 \( f(a) = f(b) \)，则 \( a + 2b \) 的取值范围是（\quad）

\choices
    {\(\left(2\sqrt{2},+\infty\right)\)}
    {\(\left[2\sqrt{2},+\infty\right)\)}
    {\(\left(3,+\infty\right)\)}
    {\(\left[3,+\infty\right)\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
由 \(0<a<b\) 且 \(|\lg a|=|\lg b|\)，可知 \(\lg a\) 与 \(\lg b\) 符号相反. 因为 \(a\ne b\)，所以
\(\lg a=-\lg b\)，即 \(ab=1\). 从而 \(0<a<1\)，且 \(b=1/a\). 令
\(g(a)=a+2/a\)，则 \(g'(a)=1-2/a^{2}<0\)（\(0<a<1\)）. 因此 \(g\) 在 \((0,1)\) 上递减，且
\(\lim_{a\to1^{-}}g(a)=3\)，\(\lim_{a\to0^{+}}g(a)=+\infty\).
所以 \(a+2b\in(3,+\infty)\)，应选 C.
\end{solution}

\begin{problem}
已知圆 \(O\) 的半径为 \(1\)，\(PA\)，\(PB\) 为该圆的两条切线，\(A\)，\(B\) 为两切点，那么 \(\overrightarrow{PA} \cdot \overrightarrow{PB}\) 的最小值为（\quad）

\choices
    {\(-4 + \sqrt{2}\)}
    {\(-3 + \sqrt{2}\)}
    {\(-4 + 2\sqrt{2}\)}
    {\(-3 + 2\sqrt{2}\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
设 \(OP=r>1\)，切线长 \(PA=PB=t\)，则由直角三角形 \(OAP\) 得
\(t^{2}=r^{2}-1\). 设 \(\angle APB=\theta\). 由于 \(OP\) 平分两条切线的夹角，在直角三角形 \(OAP\) 中有
\(\sin(\theta/2)=1/r\)，故
\[
\cos\theta=1-2\sin^{2}\frac{\theta}{2}=1-\frac{2}{r^{2}}
\]
因此
\[
\overrightarrow{PA}\cdot\overrightarrow{PB}
=(r^{2}-1)\left(1-\frac{2}{r^{2}}\right)
=r^{2}-3+\frac{2}{r^{2}}
\]
令 \(u=r^{2}>1\)，上式关于 \(u\) 的导数为 \(1-2/u^{2}\)，故在
\(u=\sqrt2\) 时取得最小值
\[
\sqrt2-3+\frac{2}{\sqrt2}=2\sqrt2-3
\]
所以应选 D.
\end{solution}

\begin{problem}
已知在半径为 \(2\) 的球面上有 \(A\)，\(B\)，\(C\)，\(D\) 四点，若 \(AB = CD = 2\)，则四面体 \(ABCD\) 的体积的最大值为（\quad）

\choices
    {\(\frac{2\sqrt{3}}{3}\)}
    {\(\frac{4\sqrt{3}}{3}\)}
    {\(2\sqrt{3}\)}
    {\(\frac{8\sqrt{3}}{3}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
设球心为 \(O\)，\(M\)，\(N\) 分别为弦 \(AB\)，\(CD\) 的中点. 因为球半径为 \(2\)，且
\(AB=CD=2\)，所以
\[
OM=ON=\sqrt{2^{2}-1^{2}}=\sqrt3
\]
设 \(\overrightarrow{MA}=\bs u\)，\(\overrightarrow{NC}=\bs v\)，则
\(|\bs u|=|\bs v|=1\)，并且 \(\overrightarrow{OM}\perp\bs u\)，\(\overrightarrow{ON}\perp\bs v\). 令
\(\bs w=\overrightarrow{MN}\). 于是
\(A=M+\bs u\)，\(B=M-\bs u\)，\(C=N+\bs v\)，\(D=N-\bs v\). 由四面体体积的混合积公式，
\[
V_{ABCD}=\frac16\left|(B-A)\cdot\bigl((C-A)\times(D-A)\bigr)\right|
=\frac23\left|\bs u\cdot(\bs w\times\bs v)\right|
\leqslant\frac{2}{3}|\bs u|\,|\bs w|\,|\bs v|
\leqslant\frac{2}{3}\cdot1\cdot(2\sqrt3)\cdot1
=\frac{4\sqrt3}{3}
\]
其中 \(|\bs w|=|MN|\leqslant OM+ON=2\sqrt3\). 当 \(M\)，\(N\) 位于球心两侧，且 \(\bs u\)，\(\bs v\)，\(\bs w\) 两两垂直时可取到等号；此时四个顶点均在球面上，故最大值为
\(\frac{4\sqrt3}{3}\)，应选 B.
\end{solution}

\section{填空题}

\begin{problem}
不等式 \( \sqrt{2x^{2}+1}-x\leqslant1 \) 的解集是\fillinblank{}.
\end{problem}

\begin{answer}
\([0,2]\)
\end{answer}

\begin{solution}
原不等式等价于 \(\sqrt{2x^{2}+1}\leqslant x+1\). 因为左边非负，所以必须有 \(x+1\geqslant0\). 在此条件下两边平方，得
\[
2x^{2}+1\leqslant(x+1)^{2}
\Longleftrightarrow x^{2}-2x\leqslant0
\Longleftrightarrow 0\leqslant x\leqslant2
\]
这个范围自动满足 \(x+1\geqslant0\)，所以解集为 \([0,2]\).
\end{solution}

\begin{problem}
已知 \(\alpha\) 为第三象限的角，\(\cos 2\alpha = -\frac{3}{5}\)，则 \(\tan \left(\frac{\pi}{4} + 2\alpha\right) =\)\fillinblank{}.
\end{problem}

\begin{answer}
\(-\frac{1}{7}\)
\end{answer}

\begin{solution}
因为 \(\alpha\) 是第三象限的角，所以 \(2\alpha\) 的终边在第一或第二象限；又
\(\cos2\alpha=-3/5<0\)，故 \(2\alpha\) 的终边在第二象限，从而
\(\sin2\alpha=4/5\)，\(\tan2\alpha=-4/3\). 因此
\[
\tan\left(\frac{\pi}{4}+2\alpha\right)
=\frac{1+\tan2\alpha}{1-\tan2\alpha}
=\frac{1-\frac43}{1+\frac43}
=-\frac17
\]
\end{solution}

\begin{problem}
直线 \( y = 1 \) 与曲线 \( y = x^2 - |x| + a \) 有四个交点，则 \( a \) 的取值范围是\fillinblank{}.
\end{problem}

\begin{answer}
\(\left(1,\frac{5}{4}\right)\)
\end{answer}

\begin{solution}
令 \(t=|x|\geqslant0\). 联立两条曲线，得到
\[
t^{2}-t+a-1=0
\]
要有四个不同的交点，关于 \(t\) 的方程必须有两个不相等的正根. 其判别式满足
\(1-4(a-1)>0\)，且两根之积 \(a-1>0\)，两根之和为 \(1>0\). 因而
\[
1<a<\frac54
\]
此时两个根均为正数，各自对应 \(x=\pm t\)，确有四个不同交点，所以
\(a\in\left(1,\frac54\right)\).
\end{solution}

\begin{problem}
已知 \(F\) 是椭圆 \(C\) 的一个焦点，\(B\) 是短轴的一个端点，线段 \(BF\) 的延长线交 \(C\) 于点 \(D\)，且 \(\overrightarrow{BF} = 2\overrightarrow{FD}\)，则 \(C\) 的离心率为\fillinblank{}.
\end{problem}

\begin{answer}
\(\frac{\sqrt{3}}{3}\)
\end{answer}

\begin{solution}
设椭圆的长半轴、短半轴和焦距参数分别为 \(a\)，\(b\)，\(c\)，则
\(b^{2}=a^{2}-c^{2}\)，离心率为 \(e=c/a\). 建立坐标系，取焦点
\(F(c,0)\)，短轴端点 \(B(0,b)\). 由
\(\overrightarrow{BF}=2\overrightarrow{FD}\)，得
\[
D=F+\frac12(F-B)=\left(\frac{3c}{2},-\frac b2\right)
\]
因为 \(D\) 在椭圆上，所以
\[
\frac{(3c/2)^{2}}{a^{2}}+\frac{(-b/2)^{2}}{b^{2}}=1
\Longleftrightarrow \frac{9c^{2}}{4a^{2}}+\frac14=1
\Longleftrightarrow \frac{c^{2}}{a^{2}}=\frac13
\]
故 \(e=\frac ca=\frac{\sqrt3}{3}\).
\end{solution}

\begin{problem}
已知 \(\triangle ABC\) 的内角 \(A\)，\(B\) 及其对边 \(a\)，\(b\) 满足 \(a + b = a \cot A + b \cot B\)，求内角 \(C\).
\end{problem}

\begin{solution}
由正弦定理，存在正数 \(k\)，使得 \(a=k\sin A\)，\(b=k\sin B\). 因此
\(a\cot A=k\cos A\)，\(b\cot B=k\cos B\). 代入已知等式并约去 \(k>0\)，得
\[
\sin A+\sin B=\cos A+\cos B
\]
利用和差化积公式，
\[
2\sin\frac{A+B}{2}\cos\frac{A-B}{2}
=2\cos\frac{A+B}{2}\cos\frac{A-B}{2}
\]
三角形内角满足 \(|A-B|<\pi\)，故 \(\cos\frac{A-B}{2}>0\)，可以约去；又
\(0<A+B<\pi\)，所以 \(0<(A+B)/2<\pi/2\). 从而
\(\tan\frac{A+B}{2}=1\)，即 \(A+B=\pi/2\). 因为
\(A+B+C=\pi\)，所以 \(C=\pi/2\).
\end{solution}

\begin{problem}
投到某杂志的稿件，先由两位初审专家进行评审. 若能通过两位初审专家的评审，则予以录用；若两位初审专家都未予通过，则不予录用；若恰能通过一位初审专家的评审，则再由第三位专家进行复审，若能通过复审专家的评审，则予以录用，否则不予录用. 设稿件能通过各初审专家评审的概率均为 \(0.5\)，复审的稿件能通过评审的概率为 \(0.3\). 各专家独立评审.
\begin{enumerate}
\item 求投到该杂志的 \(1\text{ 篇}\)稿件被录用的概率；
\item 记 \(X\) 表示投到该杂志的 \(4\text{ 篇}\)稿件中被录用的篇数，求 \(X\) 的分布列及期望.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 一篇稿件被录用有两种互斥情形：两位初审专家都通过，或恰有一位初审专家通过且复审专家通过. 因而录用概率为
\[
p=\left(\frac12\right)^{2}
+2\cdot\frac12\cdot\frac12\cdot0.3
=0.25+0.15=0.4=\frac25
\]
\item 各篇稿件的评审过程相互独立，所以 \(X\) 服从参数为 \(4\)，\(\frac25\) 的二项分布. 对
\(k=0\)，\(1\)，\(2\)，\(3\)，\(4\)，有
\[
P(X=k)=\mathrm C_{4}^{k}\left(\frac25\right)^{k}\left(\frac35\right)^{4-k}
\]
其分布列为
\begin{center}
\begin{tabular}{c|ccccc}
\(X\) & \(0\) & \(1\) & \(2\) & \(3\) & \(4\) \\
\hline
\(P\) & \(\dfrac{81}{625}\) & \(\dfrac{216}{625}\) & \(\dfrac{216}{625}\) & \(\dfrac{96}{625}\) & \(\dfrac{16}{625}\)
\end{tabular}
\end{center}
所以
\[
E(X)=4\cdot\frac25=\frac85
\]
\end{enumerate}
\end{solution}

\begin{problem}
如图，四棱锥 \(S - ABCD\) 中，\(SD \perp\) 底面 \(ABCD\)，\(AB \parallel DC\)，\(AD \perp DC\)，\(AB = AD = 1\)，\(DC = SD = 2\)，\(E\) 为棱 \(SB\) 上的一点，平面 \(EDC \perp\) 平面 \(SBC\).

\item 建立空间直角坐标系，取

\item 由 \(\lambda=2/3\)，得 \(E=(2/3,2/3,2/3)\). 平面 \(ADE\) 和
 \(CDE\) 相交于 \(DE\). 取 \(F\) 为 \(DE\) 的中点，则

\overrightarrow{FG}=\overrightarrow{FC}-\frac12\overrightarrow{DE}
\]
所以 \(G\) 在平面 \(CDE\) 内. 又
\(\overrightarrow{FG}\cdot\overrightarrow{DE}=0\)，故 \(\overrightarrow{FG}\perp DE\). 因而
\(\angle AFG\) 是二面角 \(A-DE-C\) 的平面角. 计算得
\[
\cos\angle AFG
=\frac{\overrightarrow{FA}\cdot\overrightarrow{FG}}
{|\overrightarrow{FA}|\,|\overrightarrow{FG}|}
=\frac{-2/3}{(\sqrt6/3)(2\sqrt6/3)}=-\frac12
\]
因此二面角 \(A-DE-C\) 的大小为 \(2\pi/3\).
\end{enumerate}
\end{solution}

\begin{problem}
已知函数 \( f(x) = (x + 1)\ln x - x + 1 \).
\begin{enumerate}
\item 若 \(x f'(x) \leqslant x^2 + ax + 1\)，求 \(a\) 的取值范围；
\item 证明：\((x - 1)f(x) \geqslant 0\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 因为 \(x>0\)，有
\(f'(x)=\ln x+\frac{x+1}{x}-1=\ln x+\frac1x\)，\(xf'(x)=x\ln x+1\).
所以题设不等式等价于
\[
x\ln x+1\leqslant x^{2}+ax+1
\Longleftrightarrow \ln x-x\leqslant a
\]
令 \(g(x)=\ln x-x\)，则 \(g'(x)=1/x-1\). 因此 \(g\) 在 \((0,1)\) 上递增，在
\((1,+\infty)\) 上递减，最大值为 \(g(1)=-1\). 所以不等式对一切 \(x>0\) 成立的充要条件为
\(a\geqslant-1\)，即 \(a\in[-1,+\infty)\).
\item 由
\[
f'(x)=\ln x+\frac1x
\]
以及
\[
\left(\ln x+\frac1x\right)'=\frac{x-1}{x^{2}}
\]
可知 \(\ln x+1/x\) 在 \(x=1\) 处取得最小值 \(1\)，故 \(f'(x)>0\) 对一切 \(x>0\) 成立. 又
\(f(1)=0\)，所以当 \(0<x<1\) 时 \(f(x)<0\)，当 \(x>1\) 时 \(f(x)>0\). 因而
\((x-1)f(x)\geqslant0\).
\end{enumerate}
\end{solution}

\begin{problem}
已知抛物线 \(C: y^{2} = 4x\) 的焦点为 \(F\)，过点 \(K(-1,0)\) 的直线 \(l\) 与 \(C\) 相交于 \(A\)，\(B\) 两点，点 \(A\) 关于 \(x\) 轴的对称点为 \(D\).
\begin{enumerate}
\item 证明：点 \(F\) 在直线 \(BD\) 上；
\item 设 \(\overrightarrow{FA} \cdot \overrightarrow{FB} = \frac{8}{9}\)，求 \(\triangle BDK\) 的内切圆 \(M\) 的方程.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 抛物线 \(y^{2}=4x\) 的参数方程为 \((t^{2},2t)\). 设
\(A=(t^{2},2t)\)，\(B=(s^{2},2s)\). 因为 \(A\)，\(B\)，\(K(-1,0)\) 共线，所以
\[
\frac{2t}{t^{2}+1}=\frac{2s}{s^{2}+1}
\]
因 \(A\ne B\)，化简得 \(ts=1\). 于是可设
\(A=(t^{2},2t)\)，\(B=\left(\frac1{t^{2}},\frac2t\right)\)，\(D=(t^{2},-2t)\).
抛物线焦点为 \(F(1,0)\). 计算
\[
\begin{aligned}
\det\left(\overrightarrow{DF},\overrightarrow{DB}\right)
&=(1-t^{2})\left(2t+\frac2t\right)-2t\left(\frac1{t^{2}}-t^{2}\right)\\
&=2\frac{1-t^{4}}{t}-2\frac{1-t^{4}}{t}=0.
\end{aligned}
\]
因此 \(F\)，\(B\)，\(D\) 三点共线.
\item 由
\[
\overrightarrow{FA}\cdot\overrightarrow{FB}
=(t^{2}-1)\left(\frac1{t^{2}}-1\right)+4
=6-t^{2}-\frac1{t^{2}}=\frac89
\]
得 \(t^{2}+t^{-2}=46/9\)，从而
\[
\left(t+\frac1t\right)^{2}=\frac{64}{9}
\]
关于 \(x\) 轴对称时，三角形和内切圆同时对称，故不妨设 \(t>0\)，于是
\(t+1/t=8/3\)，且 \(t^{2}+t^{-2}=46/9\). 由顶点坐标，
\[
DK^{2}=(t^{2}+1)^{2}+4t^{2}
=t^{2}\left(t^{2}+t^{-2}+6\right)=\frac{100t^{2}}{9}
\]
\[
BK^{2}=\left(1+\frac1{t^{2}}\right)^{2}+\frac4{t^{2}}
=\frac1{t^{2}}\left(t^{2}+t^{-2}+6\right)=\frac{100}{9t^{2}}
\]
所以 \(DK=10t/3\)，\(BK=10/(3t)\). 又
\[
BD^{2}=\left(t^{2}-\frac1{t^{2}}\right)^{2}
+4\left(t+\frac1t\right)^{2}
=\left(t+\frac1t\right)^{4}=\frac{4096}{81}
\]
故 \(BD=64/9\). 三角形面积为
\[
S_{\triangle BDK}
=\frac12\left|\det\left(\overrightarrow{KB},\overrightarrow{KD}\right)\right|
=\frac12\cdot4\left(t+\frac1t\right)=\frac{16}{3}
\]
因此三角形三边长分别为
\(DK=\frac{10t}{3}\)，\(BK=\frac{10}{3t}\)，\(BD=\frac{64}{9}\).
所以三角形 \(BDK\) 的周长为
\[
\frac{10t}{3}+\frac{10}{3t}+\frac{64}{9}=16
\]
半周长为 \(8\)，故内切圆半径为 \(r=S_{\triangle BDK}/8=2/3\). 设内心为 \(M\)，由内心坐标公式，
\[
M=\frac{\frac{10t}{3}\left(\frac1{t^{2}},\frac2t\right)
+\frac{10}{3t}(t^{2},-2t)+\frac{64}{9}(-1,0)}{16}
=\left(\frac19,0\right)
\]
因此内切圆 \(M\) 的方程为
\[
\left(x-\frac19\right)^{2}+y^{2}=\frac49
\]
\end{enumerate}
\end{solution}

\section{选考题：解答题}

\begin{problem}
已知数列 \(\{a_{n}\}\) 中，\(a_1 = 1\)，\(a_{n+1} = c - \frac{1}{a_n}\).
\begin{enumerate}
\item 设 \(c = \frac{5}{2}\)，\(b_n = \frac{1}{a_n - 2}\)，求数列 \(\{b_n\}\) 的通项公式；
\item 求使不等式 \(a_{n} < a_{n+1} < 3\) 成立的 \(c\) 的取值范围.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 当 \(c=5/2\) 时，
由归纳法可知 \(1\leqslant a_n<2\)：\(a_1=1\)，若 \(1\leqslant a_n<2\)，则
\(3/2\leqslant a_{n+1}=5/2-1/a_n<2\). 因而 \(a_n-2\ne0\)，数列 \(\{b_n\}\) 有定义.
\[
a_{n+1}-2=\frac12-\frac1{a_n}=\frac{a_n-2}{2a_n}
\]
因此
\[
b_{n+1}=\frac1{a_{n+1}-2}
=\frac{2a_n}{a_n-2}
=2+\frac4{a_n-2}=2+4b_n
\]
又 \(b_1=1/(1-2)=-1\). 令 \(d_n=b_n+2/3\)，则
\(d_{n+1}=4d_n\)，且 \(d_1=-1/3\). 所以
\(d_n=-\frac13\,4^{n-1}\)，\(b_n=-\frac{4^{n-1}+2}{3}\).
\item 由 \(a_1=1\) 及 \(a_1<a_2<3\)，必要地有
\[
1<c-1<3
\]
即 \(2<c<4\). 对 \(c>2\)，方程
\(x=c-1/x\) 的两个根为
\(r_{-}=\frac{c-\sqrt{c^{2}-4}}2\)，\(r_{+}=\frac{c+\sqrt{c^{2}-4}}2\)
满足 \(0<r_{-}<1<r_{+}\)，且 \(r_{-}r_{+}=1\). 记
\(\varphi(x)=c-1/x\). 当 \(r_{-}<x<r_{+}\) 时，
\[
\varphi(x)-x
=-\frac{x^{2}-cx+1}{x}
=\frac{(r_{+}-x)(x-r_{-})}{x}>0
\]
且 \(\varphi(x)<r_{+}\). 由于 \(r_{-}<a_1=1<r_{+}\)，归纳可得
\[
1=a_1<a_2<\cdots<a_n<r_{+}
\]
并且数列严格递增. 若 \(r_{+}\leqslant3\)，则对所有 \(n\) 都有
\(a_n<a_{n+1}<3\). 因为 \(c=r_{+}+1/r_{+}\)，且 \(r_{+}>1\)，条件
\(r_{+}\leqslant3\) 等价于
\[
2<c\leqslant3+\frac13=\frac{10}{3}
\]
反之，若 \(c>10/3\)，则 \(r_{+}>3\). 上述递增数列有上界 \(r_{+}\)，其极限只能是方程
\(x=c-1/x\) 的大根 \(r_{+}>3\)，所以数列的某项最终会超过 \(3\)，不满足题设不等式. 故
\[
c\in\left(2,\frac{10}{3}\right]
\]
\end{enumerate}
\end{solution}
