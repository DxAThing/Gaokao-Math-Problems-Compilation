\section{解答题}

\begin{problem}
已知 \( \{a_{n}\} \) 是公差不为零的等差数列，\( a_{1}=1 \) 且 \(a_{1}\)，\(a_{3}\)，\(a_{9}\) 成等比数列.

\begin{enumerate}
\item 求数列 \( \{a_{n}\} \) 的通项；
\item 求数列 \( \{2^{a_{n}}\} \) 的前 \(n\) 项和 \( S_{n} \).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设等差数列 \(\{a_n\}\) 的公差为 \(d\)，则 \(d\neq 0\)，且 \(a_3=1+2d\)，\(a_9=1+8d\). 由 \(a_1\)，\(a_3\)，\(a_9\) 成等比数列，得
\[
(1+2d)^2=a_1a_9=1+8d.
\]
整理得 \(4d(d-1)=0\). 因为 \(d\neq 0\)，所以 \(d=1\)，从而数列的通项为 \(a_n=a_1+(n-1)d=n\).
\item 由 \(a_n=n\)，得 \(2^{a_n}=2^n\). 因此前 \(n\) 项和为
\[
S_n=\sum_{k=1}^{n}2^k=\frac{2(2^n-1)}{2-1}=2^{n+1}-2.
\]
\end{enumerate}
\end{solution}

\begin{problem}
如图，在 \(\triangle ABC\) 中，已知 \(\angle B = 45^{\circ}\)，\(D\) 是 \(BC\) 边上的一点，\(AD = 10\)，\(AC = 14\)，\(DC = 6\). 求 \(AB\) 的长.

\bitmapfigure[max width=0.46\linewidth]{img/2010/shaanxi_liberal/q17_fig1.png}
\end{problem}

\begin{solution}
以 \(B\) 为原点，\(BC\) 所在直线为 \(x\) 轴建立直角坐标系. 设 \(BD=r\)，并设 \(A=(t,t)\)，其中 \(t>0\). 由 \(DC=6\)，可取 \(D=(r,0)\)，\(C=(r+6,0)\). 由已知长度得
\[
AD^2=(t-r)^2+t^2=100
\qquad
AC^2=(t-r-6)^2+t^2=196
\]
两式相减，得 \((t-r-6)^2-(t-r)^2=96\)，即 \(-12(t-r)+36=96\)，所以 \(t-r=-5\). 代入第一式得 \(25+t^2=100\)，又因 \(t>0\)，所以 \(t=5\sqrt{3}\). 因而
\[
AB=\sqrt{t^2+t^2}=t\sqrt{2}=5\sqrt{6}.
\]
\end{solution}

\begin{problem}
如图，在四棱锥 \(P - ABCD\) 中，底面 \(ABCD\) 是矩形，\(PA \perp\) 平面 \(ABCD\)，\(AP = AB = 2\)，\(BC = 2\sqrt{2}\)，\(E\)，\(F\) 分别是 \(AD\)，\(PC\) 的中点.

\begin{enumerate}
\item 证明：\(PC \perp\) 平面 \(BEF\)；
\item 求平面 \(BEF\) 与平面 \(BAP\) 夹角的大小.
\end{enumerate}

\bitmapfigure[max width=0.50\linewidth]{img/2010/shaanxi_liberal/q18_fig1.png}
\end{problem}

\begin{solution}
以 \(A\) 为原点，分别以 \(AB\)，\(AD\)，\(AP\) 所在直线为 \(x\) 轴、\(y\) 轴、\(z\) 轴，建立空间直角坐标系. 由题意可得
\(A(0,0,0)\)，\(B(2,0,0)\)，\(D(0,2\sqrt{2},0)\)，\(C(2,2\sqrt{2},0)\)，\(P(0,0,2)\).
因为 \(E\)，\(F\) 分别是 \(AD\)，\(PC\) 的中点，所以
\(E(0,\sqrt{2},0)\)，\(F(1,\sqrt{2},1)\).
\begin{enumerate}
\item 由坐标得 \(\overrightarrow{PC}=(2,2\sqrt{2},-2)\)，\(\overrightarrow{BE}=(-2,\sqrt{2},0)\)，\(\overrightarrow{EF}=(1,0,1)\). 于是 \(\overrightarrow{PC}\cdot\overrightarrow{BE}=-4+4=0\)，\(\overrightarrow{PC}\cdot\overrightarrow{EF}=2-2=0\).
直线 \(BE\)，\(EF\) 在平面 \(BEF\) 内相交，且 \(PC\) 同时垂直于这两条相交直线，所以 \(PC\perp\) 平面 \(BEF\).
\item 取平面 \(BAP\) 的一个法向量为 \(\bs{n}_1=(0,1,0)\). 平面 \(BEF\) 的一个法向量可取
\[
\bs{n}_2=\overrightarrow{BE}\times\overrightarrow{EF}=(\sqrt{2},2,-\sqrt{2}).
\]
设两平面的夹角为 \(\theta\)，则
\[
\cos\theta=\frac{|\bs{n}_1\cdot\bs{n}_2|}{|\bs{n}_1||\bs{n}_2|}=\frac{2}{\sqrt{8}}=\frac{\sqrt{2}}{2}.
\]
由于平面夹角取锐角或直角，故 \(\theta=45^{\circ}\).
\end{enumerate}
\end{solution}

\begin{problem}
为了解学生身高情况，某校以 \(10\%\) 的比例对全校 \(700\) 名学生按性别进行分层抽样调查，测得身高情况的统计图如下：

\begin{enumerate}
\item 估计该校男生的人数；
\item 估计该校学生身高在 \(170 \sim 185 \mathrm{~cm}\) 之间的概率；
\item 从样本中身高在 \(165 \sim 180 \mathrm{~cm}\) 之间的女生中任选 \(2\) 人，求至少有 \(1\) 人身高在 \(170 \sim 180 \mathrm{~cm}\) 之间的概率.
\end{enumerate}

\bitmapfigure[max width=0.72\linewidth]{img/2010/shaanxi_liberal/q19_fig1.png}
\end{problem}

\begin{solution}
由统计图可知，男生各组频数依次为 \(2\)，\(5\)，\(14\)，\(13\)，\(4\)，\(2\)，女生各组频数依次为 \(1\)，\(7\)，\(12\)，\(6\)，\(3\)，\(1\). 因此样本中男生人数为 \(2+5+14+13+4+2=40\)，女生人数为 \(1+7+12+6+3+1=30\)，样本容量为 \(70\).
\begin{enumerate}
\item 分层抽样比例为 \(10\%\)，所以全校男生人数的估计值为 \(40\div 10\%=400\) 人.
\item 样本中身高在 \(170\sim185\mathrm{~cm}\) 之间的男生有 \(14+13+4=31\) 人，女生有 \(12+6+3=21\) 人，共有 \(52\) 人. 因而该校学生身高在这一范围内的概率估计值为
\[
\frac{52}{70}=\frac{26}{35}.
\]
\item 样本中身高在 \(165\sim180\mathrm{~cm}\) 之间的女生有 \(7+12+6=25\) 人，其中身高在 \(170\sim180\mathrm{~cm}\) 之间的有 \(12+6=18\) 人. 任选 \(2\) 人时，所求事件的对立事件是两人身高都在 \(165\sim170\mathrm{~cm}\) 之间，因此所求概率为
\[
1-\frac{\mathrm C_{7}^{2}}{\mathrm C_{25}^{2}}=1-\frac{21}{300}=\frac{93}{100}.
\]
\end{enumerate}
\end{solution}

\begin{problem}
如图，椭圆 \(C: \frac{x^2}{a^2} + \frac{y^2}{b^2} = 1\) 的顶点为 \(A_1\)，\(A_2\)，\(B_1\)，\(B_2\)，焦点为 \(F_1\)，\(F_2\)，\(|A_1B_1| = \sqrt{7}\)，\(S_{\square A_1B_1A_2B_2} = 2S_{\square B_1F_1B_2F_2}\).
\begin{enumerate}
\item 求椭圆 \(C\) 的方程.
\item 设 \(n\) 是过原点的直线，\(l\) 是与 \(n\) 垂直相交于 \(P\) 点，与椭圆相交于 \(A\)，\(B\) 两点的直线，\(\overline{OP}=1\). 是否存在上述直线 \(l\) 使 \(\overrightarrow{AP}\cdot\overrightarrow{PB}=1\) 成立？若存在，求出直线 \(l\) 的方程；若不存在，请说明理由.
\end{enumerate}

\bitmapfigure[max width=0.52\linewidth]{img/2010/shaanxi_liberal/q20_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设椭圆焦点到原点的距离为 \(c\)，则 \(c^2=a^2-b^2\). 由顶点坐标可知 \(A_1=(-a,0)\)，\(B_1=(0,b)\)，所以
\[
|A_1B_1|^2=a^2+b^2=7.
\]
四边形 \(A_1B_1A_2B_2\) 的两条对角线长分别为 \(2a\)，\(2b\)，四边形 \(B_1F_1B_2F_2\) 的两条对角线长分别为 \(2b\)，\(2c\). 故 \(S_{\square A_1B_1A_2B_2}=2ab\)，\(S_{\square B_1F_1B_2F_2}=2bc\).
由题设得 \(2ab=2\times2bc\)，所以 \(a=2c\). 于是 \(a^2=4c^2=4(a^2-b^2)\)，即 \(b^2=\frac34a^2\). 联立 \(a^2+b^2=7\)，得 \(a^2=4\)，\(b^2=3\). 因而椭圆 \(C\) 的方程为
\[
\frac{x^2}{4}+\frac{y^2}{3}=1.
\]
\item 设单位法向量 \(\bs{e}=(\cos\theta,\sin\theta)\)，并取与之垂直的单位方向向量 \(\bs{v}=(-\sin\theta,\cos\theta)\). 因为直线 \(l\) 到原点的距离为 \(OP=1\)，且 \(P\) 是垂足.
所以 \(P=(\cos\theta,\sin\theta)\)，直线 \(l\) 的方程为 \(x\cos\theta+y\sin\theta=1\).
直线 \(l\) 上任一点可表示为 \(P+t\bs{v}\). 代入椭圆方程，设交点 \(A\)，\(B\) 对应的参数分别为 \(t_1\)，\(t_2\)，则 \(t_1<t_2\)，且方程为
\[
\left(\frac{\sin^2\theta}{4}+\frac{\cos^2\theta}{3}\right)t^2
+\frac{\sin\theta\cos\theta}{6}t
+\frac{\cos^2\theta}{4}+\frac{\sin^2\theta}{3}-1=0.
\]
又因为 \(P\) 在椭圆内部，确有 \(t_1<0<t_2\). 由韦达定理，
\[
-t_1t_2=\frac{1-\frac{\cos^2\theta}{4}-\frac{\sin^2\theta}{3}}{\frac{\sin^2\theta}{4}+\frac{\cos^2\theta}{3}}=\frac{8+\cos^2\theta}{3+\cos^2\theta}>1.
\]
由于 \(\bs{v}\) 是单位向量，
\[
\overrightarrow{AP}\cdot\overrightarrow{PB}=(-t_1\bs{v})\cdot(t_2\bs{v})=-t_1t_2>1.
\]
所以不存在满足 \(\overrightarrow{AP}\cdot\overrightarrow{PB}=1\) 的直线 \(l\).
\end{enumerate}
\end{solution}

\begin{problem}
已知函数 \( f(x) = \sqrt{x} \)，\( g(x) = a \ln x \)，\( a \in \R \).

\begin{enumerate}
\item 若曲线 \( y = f(x) \) 与曲线 \( y = g(x) \) 相交，且在交点处有共同的切线，求 \( a \) 的值和该切线方程；
\item 设函数 \( h(x) = f(x) - g(x) \)，当 \( h(x) \) 存在最小值时，求其最小值 \( \varphi(a) \) 的解析式；
\item 对上一问中的 \(\varphi(a)\) 和任意的 \(a > 0\)，\(b > 0\)，证明：\(\varphi'\left(\frac{a + b}{2}\right) \leqslant \frac{\varphi'(a) + \varphi'(b)}{2} \leqslant \varphi'\left(\frac{2ab}{a + b}\right)\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设两曲线的交点为 \((x_0,y_0)\)，则 \(x_0>0\). 令 \(t=\sqrt{x_0}>0\)，则交点条件和共同切线条件分别给出
\[
t=a\ln x_0=a\ln(t^2)
\]
以及
\[
\frac{1}{2t}=\frac{1}{2\sqrt{x_0}}=\frac{a}{x_0}=\frac{a}{t^2}.
\]
由后一式得 \(a=\frac{t}{2}\). 代入前一式，得 \(t=t\ln t\). 因为 \(t>0\)，所以 \(\ln t=1\)，从而 \(t=\e\). 因此 \(x_0=\e^2\)，\(y_0=\e\)，且 \(a=\frac{\e}{2}\). 共同切线的斜率为 \(\frac{1}{2\e}\)，故切线方程为
\[
y-\e=\frac{1}{2\e}(x-\e^2),
\]
即 \(y=\frac{x}{2\e}+\frac{\e}{2}\).
\item 函数 \(h\) 的定义域为 \(x>0\)，且
\[
h'(x)=\frac{1}{2\sqrt{x}}-\frac{a}{x}=\frac{\sqrt{x}-2a}{2x}.
\]
当 \(a>0\) 时，\(h'(x)<0\) 对 \(0<x<4a^2\) 成立，\(h'(x)>0\) 对 \(x>4a^2\) 成立. 又有 \(\displaystyle\lim_{x\to0^+}h(x)=+\infty\)，以及 \(\displaystyle\lim_{x\to+\infty}h(x)=+\infty\)，所以 \(h\) 在 \(x=4a^2\) 处取得最小值. 此时
\[
h(4a^2)=2a-a\ln(4a^2)=2a-2a\ln(2a).
\]
当 \(a=0\) 时，\(h(x)=\sqrt{x}\) 只有下确界 \(0\) 而无最小值；当 \(a<0\) 时，\(h(x)=\sqrt{x}+|a|\ln x\)，且 \(x\to0^+\) 时 \(h(x)\to-\infty\)，也不存在最小值. 因此，\(h\) 存在最小值的条件是 \(a>0\)，并且
\[
\varphi(a)=2a-2a\ln(2a)\quad(a>0).
\]
\item 由上式得 \(\varphi'(x)=2-2\bigl(\ln(2x)+1\bigr)=-2\ln(2x)\)，且 \(\varphi''(x)=-\frac{2}{x}<0\).
所以 \(\varphi'\) 在 \((0,+\infty)\) 上单调递减. 另一方面，
\[
\frac{\varphi'(a)+\varphi'(b)}{2}=-\ln(4ab)=\varphi'(\sqrt{ab}).
\]
由算术平均数不小于几何平均数不小于调和平均数，得
\[
\frac{2ab}{a+b}\leqslant\sqrt{ab}\leqslant\frac{a+b}{2}.
\]
由于 \(\varphi'\) 单调递减，故
\[
\varphi'\left(\frac{a+b}{2}\right)\leqslant\varphi'(\sqrt{ab})\leqslant\varphi'\left(\frac{2ab}{a+b}\right),
\]
结合前面的等式即得所要证明的不等式.
\end{enumerate}
\end{solution}
