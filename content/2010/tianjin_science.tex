\begin{problem}
已知函数 \( f(x) = 2\sqrt{3}\sin x\cos x + 2\cos^2 x - 1 (x \in \R) \).
\begin{enumerate}[label={（\arabic*）},itemsep=0.2em]
\item 求函数 \( f(x) \) 的最小正周期及在区间 \( \left[0, \frac{\pi}{2}\right] \) 上的最大值和最小值；
\item 若 \( f(x_{0})=\frac{6}{5}, x_{0}\in\left[\frac{\pi}{4},\frac{\pi}{2}\right] \)，求 \( \cos 2x_{0} \) 的值.
\end{enumerate}
\end{problem}

\begin{problem}
某射手每次射击击中目标的概率是 \(\frac{2}{3}\)，且各次射击的结果互不影响.
\begin{enumerate}[label={（\arabic*）},itemsep=0.2em]
\item 假设这名射手射击 5 次，求恰有 2 次击中目标的概率；
\item 假设这名射手射击 5 次，求有 3 次连续击中目标，另外 2 次未击中目标的概率；
\item 假设这名射手射击 3 次，每次射击，击中目标得 1 分，未击中目标得 0 分，在 3 次射击中，若有 2 次连续击中，而另外 1 次未击中，则额外加 1 分；若 3 次全击中，则额外加 3 分. 记 \(\xi\) 为射手射击 3 次后的总的分数，求 \(\xi\) 的分布列.
\end{enumerate}
\end{problem}

\begin{problem}
已知椭圆 \(\frac{x^2}{a^2} + \frac{y^2}{b^2} = 1 (a > b > 0)\) 的离心率 \(e = \frac{\sqrt{3}}{2}\)，连接椭圆的四个顶点得到的菱形面积为 4.
\begin{enumerate}[label={（\arabic*）},itemsep=0.2em]
\item 求椭圆的方程；
\item 设直线 \(l\) 与椭圆相交于不同的两点 \(A, B\). 已知点 \(A\) 的坐标为 \((-a, 0)\)，点 \(Q(0, y_{0})\) 在线段 \(AB\) 的垂直平分线上，且 \(\overrightarrow{QA} \cdot \overrightarrow{QB} = 4\). 求 \(y_{0}\) 的值.
\end{enumerate}

\begin{problem}
已知函数 \( f(x) = x\e^{-x} (x \in \R) \).
\begin{enumerate}[label={（\arabic*）},itemsep=0.2em]
\item 求函数 \( f(x) \) 的单调区间和极值；
\item 已知函数 \( y = g(x) \) 的图象与函数 \( y = f(x) \) 的图象关于直线 \( x = 1 \) 对称. 证明：当 \( x > 1 \) 时，\( f(x) > g(x) \)；
\item 如果 \(x_{1} \neq x_{2}\)，且 \(f(x_{1}) = f(x_{2})\)，证明：\(x_{1} + x_{2} > 2\).
\end{enumerate}
\end{problem}

\begin{problem}
在数列 \(\{a_{n}\}\) 中，\(a_{1} = 0\)，且对任意 \(k \in \mathbf{N}^{*}\)，\(a_{2k-1}, a_{2k}, a_{2k+1}\) 成等差数列，其公差为 \(d_{k}\).
\begin{enumerate}[label={（\arabic*）},itemsep=0.2em]
\item 若 \(d_{k} = 2k\)，证明 \(a_{2k}, a_{2k+1}, a_{2k+2}\) 成等比数列 \((k \in \mathbf{N}^{*})\)；
\item 若对任意 \(k \in \mathbf{N}^{*}\)，\(a_{2k}, a_{2k+1}, a_{2k+2}\) 成等比数列，其公比为 \(q_k\).
\begin{enumerate}[label={\circled{\arabic*}},itemsep=0.2em]
\item 设 \(q_{1} \neq 1\). 证明 \(\left\{\frac{1}{q_{k-1}-1}\right\}\) 是等差数列；
\item 若 \(a_2 = 2\)，证明 \(\frac{3}{2} < 2n - \sum_{k=2}^{n} \frac{k^2}{a_k} \leqslant 2 (n \geqslant 2)\).
\end{enumerate}
\end{enumerate}
\end{problem}
