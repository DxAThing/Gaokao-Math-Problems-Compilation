\section{解答题}

\begin{problem}
在 \(\triangle ABC\) 中，角 \(A\)，\(B\)，\(C\) 所对的边分别为 \(a\)，\(b\)，\(c\). 设 \(S\) 为 \(\triangle ABC\) 的面积，满足 \(S = \frac{\sqrt{3}}{4}(a^2 + b^2 - c^2)\).

\begin{enumerate}
\item 求角 \(C\) 的大小；
\item 求 \(\sin A + \sin B\) 的最大值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由三角形面积公式和余弦定理，得
\(S = \frac{1}{2}ab\sin C\)，\(a^2+b^2-c^2=2ab\cos C\). 将它们代入已知条件，得到
\(\frac{1}{2}ab\sin C=\frac{\sqrt{3}}{2}ab\cos C\). 由于 \(a>0\)，\(b>0\)，且 \(0<C<\pi\)，所以 \(\sin C>0\)，从而 \(\cos C>0\). 约去 \(\frac{1}{2}ab\)，可得 \(\tan C=\sqrt{3}\)，故 \(C=\frac{\pi}{3}\).
\item 因为 \(A+B=\pi-C=\frac{2\pi}{3}\)，所以
\(\sin A+\sin B=2\sin\frac{A+B}{2}\cos\frac{A-B}{2}=\sqrt{3}\cos\frac{A-B}{2}\). 又因为 \(A>0\)，\(B>0\)，所以 \(\left|\frac{A-B}{2}\right|<\frac{\pi}{3}\)，从而 \(\cos\frac{A-B}{2}\leqslant1\). 因此 \(\sin A+\sin B\leqslant\sqrt{3}\). 当 \(A=B=\frac{\pi}{3}\) 时取等号，所以最大值为 \(\sqrt{3}\).
\end{enumerate}
\end{solution}

\begin{problem}
设 \(a_1\)，\(d\) 为实数，首项为 \(a_1\)，公差为 \(d\) 的等差数列 \(\{a_n\}\) 的前 \(n\) 项和为 \(S_n\)，满足 \(S_5S_6 + 15 = 0\).

\begin{enumerate}
\item 若 \(S_5 = 5\)，求 \(S_6\) 及 \(a_1\)；
\item 求 \(d\) 的取值范围.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由 \(S_5S_6+15=0\) 及 \(S_5=5\)，得 \(S_6=-3\). 因此 \(a_6=S_6-S_5=-8\). 另一方面，等差数列的前 \(n\) 项和公式给出
\(S_5=5a_1+10d=5\)，即 \(a_1+2d=1\)；又 \(a_6=a_1+5d=-8\). 两式相减得 \(3d=-9\)，所以 \(d=-3\)，进而 \(a_1=7\).
\item 由 \(S_5=5a_1+10d\)，可得 \(a_1=\frac{S_5}{5}-2d\). 代入 \(S_6=6a_1+15d\)，得 \(S_6=\frac{6}{5}S_5+3d\). 于是已知条件等价于
\(S_5\left(\frac{6}{5}S_5+3d\right)+15=0\)，即
\(2S_5^2+5dS_5+25=0\). 要使实数 \(S_5\) 存在，该关于 \(S_5\) 的二次方程判别式必须满足
\(\Delta=25d^2-200\geqslant0\)，所以 \(d^2\geqslant8\)，即 \(d\leqslant-2\sqrt{2}\) 或 \(d\geqslant2\sqrt{2}\). 反之，当上述条件成立时，二次方程有实根，取该实根为 \(S_5\)，再令 \(a_1=\frac{S_5}{5}-2d\)，即可得到满足条件的等差数列，因此取值范围为 \((-\infty,-2\sqrt{2}]\cup[2\sqrt{2},+\infty)\).
\end{enumerate}
\end{solution}

\begin{problem}
如图，在平行四边形 \(ABCD\) 中，\(AB = 2BC\)，\(\angle ABC = 120^{\circ}\). \(E\) 为线段 \(AB\) 的中点，将 \(\triangle ADE\) 沿直线 \(DE\) 翻折成 \(\triangle A'DE\)，使平面 \(A'DE \perp\) 平面 \(BCD\)，\(F\) 为线段 \(A'C\) 的中点.

\begin{enumerate}
\item 求证：\(BF \parallel\) 平面 \(A'DE\)；
\item 设 \(M\) 为线段 \(DE\) 的中点，求直线 \(FM\) 与平面 \(A'DE\) 所成角的余弦值.
\end{enumerate}

\bitmapfigure[max width=0.42\linewidth]{img/2010/zhejiang_liberal/q20_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 取 \(BC=1\)，建立直角坐标系，使平面 \(BCD\) 为 \(z=0\)，并令 \(B=(0,0,0)\)，\(C=(1,0,0)\). 由 \(AB=2BC\) 及 \(\angle ABC=120^{\circ}\)，可取
\(A=(-1,\sqrt{3},0)\)，从而 \(D=(0,\sqrt{3},0)\). 于是
\(E=\left(-\frac{1}{2},\frac{\sqrt{3}}{2},0\right)\)，\(M=\left(-\frac{1}{4},\frac{3\sqrt{3}}{4},0\right)\). 由此可见 \(\triangle ADE\) 为等边三角形，且折叠后 \(A'M\perp DE\)，\(A'M=\frac{\sqrt{3}}{2}\). 因为平面 \(A'DE\) 垂直于平面 \(BCD\)，可取
\(A'=\left(-\frac{1}{4},\frac{3\sqrt{3}}{4},\frac{\sqrt{3}}{2}\right)\).

平面 \(A'DE\) 的一个法向量为 \(\bs{n}=(\sqrt{3},-1,0)\)，其方程为 \(\sqrt{3}x-y+\sqrt{3}=0\). 又 \(F\) 为 \(A'C\) 的中点，所以
\(F=\left(\frac{3}{8},\frac{3\sqrt{3}}{8},\frac{\sqrt{3}}{4}\right)\). 于是 \(\overrightarrow{BF}=\left(\frac{3}{8},\frac{3\sqrt{3}}{8},\frac{\sqrt{3}}{4}\right)\)，满足 \(\overrightarrow{BF}\cdot\bs{n}=0\). 因此直线 \(BF\) 的方向向量平行于平面 \(A'DE\)，且 \(B\) 不在该平面上，故 \(BF\parallel\) 平面 \(A'DE\).
\item 设直线 \(FM\) 与平面 \(A'DE\) 所成的角为 \(\theta\). 取直线 \(FM\) 的方向向量
\(\bs{u}=\overrightarrow{MF}=\left(\frac{5}{8},-\frac{3\sqrt{3}}{8},\frac{\sqrt{3}}{4}\right)\). 则 \(|\bs{u}|=1\)，\(|\bs{n}|=2\)，且 \(\bs{u}\cdot\bs{n}=\sqrt{3}\). 所以
\(\sin\theta=\frac{|\bs{u}\cdot\bs{n}|}{|\bs{u}|\,|\bs{n}|}=\frac{\sqrt{3}}{2}\). 由于 \(0\leqslant\theta\leqslant\frac{\pi}{2}\)，故 \(\cos\theta=\frac{1}{2}\).
\end{enumerate}
\end{solution}

\begin{problem}
已知函数 \(f(x) = (x - a)^2(x - b)\) （\(a, b \in \R, a < b\)）.

\begin{enumerate}
\item 当 \(a = 1\)，\(b = 2\) 时，求曲线 \(y = f(x)\) 在点 \((2, f(2))\) 处的切线方程；
\item 设 \(x_1\)，\(x_2\) 是 \(f(x)\) 的两个极值点，\(x_3\) 是 \(f(x)\) 的一个零点，且 \(x_3 \neq x_1\)，\(x_3 \neq x_2\). 证明：存在实数 \(x_4\)，使得 \(x_1\)，\(x_2\)，\(x_3\)，\(x_4\) 按某种顺序排列后成等差数列，并求 \(x_4\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 当 \(a=1\)，\(b=2\) 时，\(f(x)=(x-1)^2(x-2)\)，所以
\(f'(x)=2(x-1)(x-2)+(x-1)^2=(x-1)(3x-5)\). 又 \(f(2)=0\)，\(f'(2)=1\)，故曲线在点 \((2,0)\) 处的切线方程为 \(y-0=1\cdot(x-2)\)，即 \(y=x-2\).
\item 对一般的 \(a<b\)，有
\(f'(x)=2(x-a)(x-b)+(x-a)^2=(x-a)(3x-a-2b)\). 因此 \(f(x)\) 的两个极值点的横坐标为 \(a\) 和 \(\frac{a+2b}{3}\). 而 \(f(x)=0\) 的零点为 \(a\) 和 \(b\)，其中 \(a\) 是二重零点. 由题设 \(x_3\neq x_1\) 且 \(x_3\neq x_2\)，可知 \(x_3=b\).

由于 \(a<b\)，有
\(a<\frac{2a+b}{3}<\frac{a+2b}{3}<b\)，并且相邻两数之差均为 \(\frac{b-a}{3}\). 所以按 \(a\)，\(\frac{2a+b}{3}\)，\(\frac{a+2b}{3}\)，\(b\) 的顺序排列时，四个数成等差数列，故 \(x_4=\frac{2a+b}{3}\).
\end{enumerate}
\end{solution}

\begin{problem}
已知 \(m\) 是非零实数，抛物线 \(C: y^{2} = 2px\) （\(p > 0\)） 的焦点 \(F\) 在直线 \(l: x - my - \frac{m^{2}}{2} = 0\) 上.

\begin{enumerate}
\item 若 \(m = 2\)，求抛物线 \(C\) 的方程；
\item 设直线 \(l\) 与抛物线 \(C\) 交于 \(A\)，\(B\) 两点，过 \(A\)，\(B\) 分别作抛物线 \(C\) 的准线的垂线，垂足为 \(A_{1}\)，\(B_{1}\)，\(\triangle AA_{1}F\)，\(\triangle BB_{1}F\) 的重心分别为 \(G\)，\(H\). 求证：对任意非零实数 \(m\)，抛物线 \(C\) 的准线与 \(x\) 轴的交点在以线段 \(GH\) 为直径的圆外.
\end{enumerate}

\bitmapfigure[max width=0.38\linewidth]{img/2010/zhejiang_liberal/q22_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 抛物线 \(y^2=2px\) 的焦点为 \(F\left(\frac{p}{2},0\right)\). 由于焦点在直线 \(l\) 上，代入直线方程得
\(\frac{p}{2}-\frac{m^2}{2}=0\)，所以 \(p=m^2\). 当 \(m=2\) 时，\(p=4\)，故抛物线 \(C\) 的方程为 \(y^2=8x\).
\item 由 \(p=m^2\)，可知 \(F=\left(\frac{m^2}{2},0\right)\)，抛物线的准线为 \(x=-\frac{m^2}{2}\). 设 \(A=(x_A,y_A)\)，\(B=(x_B,y_B)\). 由直线 \(l\) 得 \(x=my+\frac{m^2}{2}\)，代入抛物线方程，得到
\(y^2=2m^2\left(my+\frac{m^2}{2}\right)=2m^3y+m^4\). 因此 \(y_A,y_B\) 是方程 \(y^2-2m^3y-m^4=0\) 的两个根，从而
\(y_A+y_B=2m^3\)，\(y_Ay_B=-m^4\).

过 \(A\)，\(B\) 作准线的垂线都是水平线，所以 \(A_1=\left(-\frac{m^2}{2},y_A\right)\)，\(B_1=\left(-\frac{m^2}{2},y_B\right)\). 根据三角形重心坐标公式，得
\(G=\left(\frac{x_A}{3},\frac{2y_A}{3}\right)\)，\(H=\left(\frac{x_B}{3},\frac{2y_B}{3}\right)\). 记准线与 \(x\) 轴的交点为 \(T=\left(-\frac{m^2}{2},0\right)\). 由于 \(x_A=my_A+\frac{m^2}{2}\)，有
\(\overrightarrow{TG}=\left(\frac{m(2m+y_A)}{3},\frac{2y_A}{3}\right)\)，\(\overrightarrow{TH}=\left(\frac{m(2m+y_B)}{3},\frac{2y_B}{3}\right)\). 于是
\(\overrightarrow{TG}\cdot\overrightarrow{TH}=\frac{m^2(2m+y_A)(2m+y_B)+4y_Ay_B}{9}\). 再由根与系数的关系，得
\((2m+y_A)(2m+y_B)=4m^2+2m(y_A+y_B)+y_Ay_B=4m^2+3m^4\)，所以
\(\overrightarrow{TG}\cdot\overrightarrow{TH}=\frac{m^2(4m^2+3m^4)-4m^4}{9}=\frac{m^6}{3}>0\).

设以 \(GH\) 为直径的圆的圆心为 \(O_1\). 对任意点 \(X\)，有 \(\overrightarrow{XG}\cdot\overrightarrow{XH}=|XO_1|^2-\left(\frac{|GH|}{2}\right)^2\). 因此当该数量大于零时，点 \(X\) 在圆外. 取 \(X=T\)，由上式可知 \(T\) 在以 \(GH\) 为直径的圆外，证毕.
\end{enumerate}
\end{solution}
