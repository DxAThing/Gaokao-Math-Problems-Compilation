\begin{problem}

\begin{problem}
已知函数 \( f(x) = x^2 + bx + c (b, c \in \R) \)，对任意的 \( x \in \R \)，恒有 \( f'(x) \leqslant f(x) \).

\begin{enumerate}
\item 证明：当 \(x \geqslant 0\) 时，\(f(x) \leqslant (x + c)^2\)；
\item 若对满足题设条件的任意 \(b, c\)，不等式 \(f(c) - f(b) \leqslant M (c^2 - b^2)\) 恒成立，求 \(M\) 的最小值.
\end{enumerate}
\end{problem}
