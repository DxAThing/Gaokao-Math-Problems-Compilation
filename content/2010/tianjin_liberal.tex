\begin{problem}
在数列 \(\{a_n\}\) 中，\(a_1 = 0\)，且对任意 \(k \in \mathbf{N}^{*}\)，\(a_{2k-1}\)，\(a_{2k}\)，\(a_{2k+1}\) 成等差数列，其公差为 \(2k\).

\begin{enumerate}
\item 证明 \(a_4\)，\(a_5\)，\(a_6\) 成等比数列；
\item 求数列 \(\{a_n\}\) 的通项公式；
\item 记 \(T_n = \frac{2^2}{a_2} + \frac{3^2}{a_3} + \cdots + \frac{n^2}{a_n}\)，证明 \(\frac{3}{2} < 2n - T_n \leqslant 2\)（\(n \geqslant 2\)）.
\end{enumerate}
\end{problem}

\begin{solution}
短答案：

\begin{enumerate}
\item \(a_4\)，\(a_5\)，\(a_6\) 成等比数列，公比为 \(\frac{3}{2}\)；
\item \(a_n = \begin{cases}
\frac{n^2 - 1}{2}, & n\text{ 为奇数},\\
\frac{n^2}{2}, & n\text{ 为偶数};
\end{cases}\)
\item 当 \(n\) 为偶数时，\(2n - T_n = \frac{3}{2} + \frac{1}{n}\)；当 \(n\) 为奇数时，\(2n - T_n = \frac{3}{2} + \frac{1}{n+1}\)，故 \(\frac{3}{2} < 2n - T_n \leqslant 2\).
\end{enumerate}
\end{solution}
