\section{单选题}

\begin{problem}
\(\i\) 是虚数单位，复数 \(\frac{3 + \i}{1 - \i} =\)（\quad）

\choices
    {\(1 + 2\i\)}
    {\(2 + 4\i\)}
    {\(-1 - 2\i\)}
    {\(2 - \i\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
将分子、分母同乘以 \(1+\i\)，得
\[
\frac{3+\i}{1-\i}=\frac{(3+\i)(1+\i)}{(1-\i)(1+\i)}
=\frac{2+4\i}{2}=1+2\i
\]
故选 A.
\end{solution}

\begin{problem}
设变量 \(x\)，\(y\) 满足约束条件 \(\begin{cases}
x + y \leqslant 3,\\
x - y \geqslant -1,\\
y \geqslant 1.
\end{cases}\)
则目标函数 \(z = 4x + 2y\) 的最大值为（\quad）

\choices
    {\(12\)}
    {\(10\)}
    {\(8\)}
    {\(2\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
由约束条件可得
\(y-1\leqslant x\leqslant 3-y\)，\(1\leqslant y\leqslant 2\).
可行域的顶点为 \((0,1)\)、\((2,1)\)、\((1,2)\). 目标函数在这些顶点处的值分别为
\(2\)，\(10\)，\(8\).
所以目标函数的最大值为 \(10\)，故选 B.
\end{solution}

\begin{problem}
阅读下面的程序框图，若输出 \(s\) 的值为 \(-7\)，则判断框内可填写（\quad）

\bitmapfigure[max width=0.26\linewidth]{img/2010/tianjin_liberal/q3_fig1.png}

\choices
    {\(-1\)}
    {\(0\)}
    {\(1\)}
    {\(3\)}
\end{problem}

\begin{answer}
题面与选项矛盾，无正确选项
\end{answer}

\begin{solution}
按照框图逐次计算. 初始 \(s=1\)，\(\ i=1\)，每次循环先更新 \(s\)，再令 \(i\) 增加 1. 第一次循环后
\(s=1(3-1)+1=3\)，\(i=2\).
第二次循环后
\(s=3(3-2)+1=4\)，\(i=3\).
第三次循环后
\(s=4(3-3)+1=1\)，\(i=4\).
第四次循环后
\(s=1(3-4)+1=0\)，\(i=5\).
若将判断框内依次填入四个选项中的数值 \(-1\)，\(0\)，\(1\)，\(3\)，前三种情形均在第一次循环后因 \(i=2\) 满足条件而停止，输出 \(s=3\)；填入 \(3\) 时，前三次循环后的 \((s,i)\) 依次为 \((3,2)\)、\((4,3)\)、\((1,4)\)，第三次循环后停止，输出 \(s=1\)，均不可能输出 \(-7\).

原 PDF 中判断框清晰印为 \(i>4\). 按此条件，第四次循环后 \(s=0\)、\(i=5\)，程序停止并输出 \(0\)，仍与题干的 \(-7\) 不符. 因此按原 PDF 的题面，四个选项均不能成立，答案为“题面与选项矛盾，无正确选项”；公开文科版本将题干改为“运行相应的程序，则输出 \(s\) 的值为”时才对应选项 B.
\end{solution}

\begin{problem}
函数 \(f(x) = \e^{x} + x - 2\) 的零点所在的一个区间是（\quad）

\choices
    {\((-2,-1)\)}
    {\((-1,0)\)}
    {\((0,1)\)}
    {\((1,2)\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
\(f(x)=\e^x+x-2\) 在 \(\R\) 上连续，且
\(f(0)=1+0-2=-1<0\)，\(f(1)=\e+1-2=\e-1>0\).
由零点存在定理，\(f\) 在 \((0,1)\) 内至少有一个零点. 因此零点所在的一个区间是 \((0,1)\)，故选 C.
\end{solution}

\begin{problem}
下列命题中，真命题是（\quad）

\choices
    {\(\exists m \in \R\)，使函数 \(f(x) = x^{2} + mx\)（\(x \in \R\)）是偶函数}
    {\(\exists m \in \R\)，使函数 \(f(x) = x^{2} + mx\)（\(x \in \R\)）是奇函数}
    {\(\forall m \in \R\)，使函数 \(f(x) = x^{2} + mx\)（\(x \in \R\)）是偶函数}
    {\(\forall m \in \R\)，使函数 \(f(x) = x^{2} + mx\)（\(x \in \R\)）是奇函数}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
对任意 \(x\in\R\)，有
\[
f(-x)=x^2-mx
\]
若 \(f\) 为偶函数，则应有 \(f(-x)=f(x)\)，从而
\[
x^2-mx=x^2+mx\quad\text{对任意 }x\text{成立}
\]
即 \(2mx=0\) 对任意 \(x\) 成立，所以 \(m=0\). 此时 \(f(x)=x^2\) 确为偶函数，故选项 A 为真.

若 \(f\) 为奇函数，则应有
\[
x^2-mx=-x^2-mx
\]
即 \(2x^2=0\) 对任意 \(x\) 成立，这是不可能的. 因此 B、C、D 均不成立，故选 A.
\end{solution}

\begin{problem}
设 \(a = \log_{5} 4\)，\(b = (\log_{5} 3)^{2}\)，\(c = \log_{4} 5\)，则（\quad）

\choices
    {\(a < c < b\)}
    {\(b < c < a\)}
    {\(a < b < c\)}
    {\(b < a < c\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
因为 \(3<4<5\)，所以
\[
0<\log_5 3<\log_5 4<1
\]

\[
b=(\log_5 3)^2<\log_5 3<\log_5 4=a

又 \(0<a<1\)，所以 \(c=\log_4 5=1/\log_5 4=1/a>a\). 综上
\[

\]
故选 D.
\end{solution}

\begin{problem}
设集合 \(A = \{x \mid |x - a| < 1,\ x \in \R\}\)，\(B = \{x \mid 1 < x < 5,\ x \in \R\}\). 若 \(A \cap B = \varnothing\)，则实数 \(a\) 的取值范围是（\quad）

\choices
    {\(\{a \mid 0 \leqslant a \leqslant 6\}\)}
    {\(\{a \mid a \leqslant 2\text{ 或 }a \geqslant 4\}\)}
    {\(\{a \mid a \leqslant 0\text{ 或 }a \geqslant 6\}\)}
    {\(\{a \mid 2 \leqslant a \leqslant 4\}\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
\(A=(a-1,a+1)\)，\(B=(1,5)\).
要使两个开区间没有公共点，必须满足左侧区间在 \(B\) 的左边，或在 \(B\) 的右边，即
\[
a+1\leqslant 1\quad\text{或}\quad a-1\geqslant 5
\]
这分别给出 \(a\leqslant0\) 或 \(a\geqslant6\). 端点不属于相应开区间，所以等号允许. 故选 C.
\end{solution}

\begin{problem}
如图，在 \(\triangle ABC\) 中，\(AD \perp AB\)，\(\overrightarrow{BC} = \sqrt{3}\overrightarrow{BD}\)，\(|\overrightarrow{AD}| = 1\)，则 \(\overrightarrow{AC} \cdot \overrightarrow{AD} =\)（\quad）

\bitmapfigure[max width=0.46\linewidth]{img/2010/tianjin_liberal/q9_fig1.png}

\choices
    {\(2\sqrt{3}\)}
    {\(\frac{\sqrt{3}}{2}\)}
    {\(\frac{\sqrt{3}}{3}\)}
    {\(\sqrt{3}\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
取 \(B=(0,0)\)，设 \(D=(d,0)\)，则由
\(\overrightarrow{BC}=\sqrt3\overrightarrow{BD}\) 得
\[
C=(\sqrt3d,0)
\]
设 \(A=(u,v)\). 由 \(AD\perp AB\) 得
\[
(u-d,v)\mathbin{\cdot}(u,v)=0,\qquad\text{即 }u^2+v^2=ud
\]
又 \(|AD|=1\)，所以
\[
1=(u-d)^2+v^2=u^2+v^2-2ud+d^2=d^2-ud
\]
于是
\[
d(d-u)=1
\]
因此
\[
\overrightarrow{AC}\mathbin{\cdot}\overrightarrow{AD}
=(C-A)\mathbin{\cdot}(D-A)
=(\sqrt3d-u,-v)\mathbin{\cdot}(d-u,-v)
=\sqrt3d(d-u)-ud+u^2+v^2
=\sqrt3d(d-u)=\sqrt3
\]
故选 D.
\end{solution}

\begin{problem}
设函数 \(g(x) = x^{2} - 2\) （\(x \in \R\)），\(f(x) = \begin{cases}
g(x) + x + 4, & x < g(x),\\
g(x) - x, & x \geqslant g(x),
\end{cases}\)
则 \(f(x)\) 的值域是（\quad）

\choices
    {\(\left[-\frac{9}{4}, 0\right] \cup (1, +\infty)\)}
    {\([0, +\infty)\)}
    {\(\left[-\frac{9}{4}, +\infty\right)\)}
    {\(\left[-\frac{9}{4}, 0\right] \cup (2, +\infty)\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
由 \(x<g(x)\) 得
\[
x<x^2-2\iff x^2-x-2>0\iff x<-1\text{ 或 }x>2
\]
在这部分，
\[
f(x)=x^2-2+x+4=x^2+x+2
\]
当 \(x<-1\) 时，该函数在此区间上递减，且趋近于 \(-1\) 时趋近于 \(2\)，所以值域为 \((2,+\infty)\)；当 \(x>2\) 时，函数递增，值域为 \((8,+\infty)\)，这是 \((2,+\infty)\) 的子集. 因而第一部分总值域为 \((2,+\infty)\).

由 \(x\geqslant g(x)\) 得
\[
x\geqslant x^2-2\iff x^2-x-2\leqslant0\iff -1\leqslant x\leqslant2
\]
在此区间，
\[
f(x)=x^2-x-2=\left(x-\frac12\right)^2-\frac94
\]
故值域为 \(\left[-\frac94,0\right]\). 合并两部分得到
\[
\left[-\frac94,0\right]\cup(2,+\infty)
\]
故选 D.
\end{solution}

\begin{problem}
如图，四边形 \(ABCD\) 是圆 \(O\) 的内接四边形，延长 \(AB\) 和 \(DC\) 相交于点 \(P\). 若 \(PB = 1\)，\(PD = 3\)，则 \(\frac{BC}{AD}\) 的值为\fillinblank{}.

\bitmapfigure[max width=0.42\linewidth]{img/2010/tianjin_liberal/q11_fig1.png}
\end{problem}

\begin{answer}
\(\frac{1}{3}\)
\end{answer}

\begin{solution}
因为四边形 \(ABCD\) 内接于同一圆，所以对角互补：
\[
\angle ABC+\angle CDA=\pi
\]
由图中点的顺序 \(P\)，\(B\)，\(A\) 共线、\(P\)，\(C\)，\(D\) 共线，射线 \(BP\) 是射线 \(BA\)
的反向延长线，从而
\[
\angle PBC=\pi-\angle ABC=\angle CDA
\]
又因为 \(P\)，\(B\)，\(A\) 共线且 \(P\)，\(C\)，\(D\) 共线，
\[
\angle BPC=\angle DPA
\]
所以由两角分别相等，
\[
\triangle PBC\sim\triangle PDA
\]
对应边满足
\[
\frac{BC}{AD}=\frac{PB}{PD}=\frac{1}{3}
\]
\end{solution}

\begin{problem}
一个几何体的三视图如图所示，则这个几何体的体积为\fillinblank{}.

\bitmapfigure[max width=0.62\linewidth]{img/2010/tianjin_liberal/q12_fig1.png}
\end{problem}

\begin{answer}
3
\end{answer}

\begin{problem}
已知双曲线 \(\frac{x^2}{a^2} - \frac{y^2}{b^2} = 1\) （\(a > 0\)，\(\ b > 0\)） 的一条渐近线方程是 \(y = \sqrt{3}x\)，它的一个焦点与抛物线 \(y^2 = 16x\) 的焦点相同. 则双曲线的方程为\fillinblank{}.
\end{problem}

\begin{answer}
\(\frac{x^2}{4} - \frac{y^2}{12} = 1\)
\end{answer}

\begin{solution}
双曲线的渐近线为 \(y=\pm\frac{b}{a}x\)，故
\(\frac{b}{a}=\sqrt3\)，\(b^2=3a^2\).
抛物线 \(y^2=16x\) 的焦点为 \((4,0)\). 双曲线的焦点为 \((\pm c,0)\)，且
\[
c^2=a^2+b^2=4a^2
\]
由题意 \(c=4\)，所以 \(2a=4\)，即 \(a=2\)，从而 \(b^2=12\). 因此双曲线方程为
\[
\frac{x^2}{4}-\frac{y^2}{12}=1
\]
\end{solution}

\begin{problem}
已知圆 \(C\) 的圆心是直线 \(x - y + 1 = 0\) 与 \(x\) 轴的交点，且圆 \(C\) 与直线 \(x + y + 3 = 0\) 相切，则圆 \(C\) 的方程为\fillinblank{}.
\end{problem}

\begin{answer}
\((x+1)^2+y^2=2\)
\end{answer}

\begin{solution}
圆心是直线 \(x-y+1=0\) 与 \(x\) 轴的交点. 令 \(y=0\)，得 \(x=-1\)，所以圆心为 \((-1,0)\).

圆心到切线 \(x+y+3=0\) 的距离为
\[
r=\frac{|-1+0+3|}{\sqrt{1^2+1^2}}=\sqrt2
\]
故圆的方程为
\[
(x+1)^2+y^2=2
\]
\end{solution}

\begin{problem}
设 \(\{a_n\}\) 是等比数列，公比 \(q = \sqrt{2}\). \(S_n\) 为 \(\{a_n\}\) 的前 \(n\) 项和. 记 \(T_n = \frac{17S_n - S_{2n}}{a_{n+1}}\)，\(\ n \in \mathbf{N}^{*}\). 设 \(T_{n_0}\) 为数列 \(\{T_n\}\) 的最大项，则 \(n_0 =\fillinblank{}\).
\end{problem}

\begin{answer}
4
\end{answer}

\begin{solution}
设首项为 \(a_1\). 由于题设中的 \(T_n\) 有定义，故 \(a_1\ne0\). 公比 \(q=\sqrt2\ne1\)，
\(S_n=a_1\frac{q^n-1}{q-1}\)，\(S_{2n}=a_1\frac{q^{2n}-1}{q-1}\).
代入 \(T_n\)，并利用 \(a_{n+1}=a_1q^n\)，得
\[
T_n=\frac{17-q^n-16q^{-n}}{q-1}
\]
于是
\[
T_{n+1}-T_n=-q^n+16q^{-(n+1)}
\]
因为 \(q>0\)，差值为正当且仅当
\[
q^{2n+1}<16
\]
当 \(n=1\)，\(2\)，\(3\) 时该不等式成立，而当 \(n\geqslant4\) 时不成立. 因此 \(T_n\) 先增后减，最大项为 \(T_4\)，即 \(n_0=4\).
\end{solution}

\begin{problem}
设函数 \(f(x) = x - \frac{1}{x}\)，对任意 \(x \in [1,+\infty)\)，\(f(mx) + mf(x) < 0\) 恒成立，则实数 \(m\) 的取值范围是\fillinblank{}.
\end{problem}

\begin{answer}
\(m < -1\)
\end{answer}

\begin{solution}
题设对任意 \(x\geqslant1\) 都要求 \(f(mx)\) 有意义，因此 \(m\ne0\). 直接计算得
\[
f(mx)+mf(x)=mx-\frac{1}{mx}+m\left(x-\frac1x\right)
=2mx-\frac{m+1/m}{x}
=\frac{2mx^2-m-1/m}{x}
\]
若 \(m>0\)，当 \(x\) 足够大时上式为正，不可能恒小于 0. 所以必须有 \(m<0\).

当 \(m<0\) 时，分母 \(x>0\)，且分子 \(2mx^2-m-1/m\) 关于 \(x^2\) 递减. 因此对所有 \(x\geqslant1\) 为负，等价于在 \(x=1\) 时为负：
\[
2m-m-\frac1m=m-\frac1m<0
\]
令 \(m=-t\)，其中 \(t>0\)，则上式等价于
\[
-t+\frac1t<0\iff t>1\iff m<-1
\]
故 \(m\) 的取值范围为 \(m<-1\).
\end{solution}

\begin{problem}
在 \(\triangle ABC\) 中，\(\frac{AC}{AB} = \frac{\cos B}{\cos C}\).

\begin{enumerate}
\item 证明：\(B = C\)；
\item 若 \(\cos A = -\frac{1}{3}\)，求 \(\sin \left(4B + \frac{\pi}{3}\right)\) 的值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由正弦定理，
\[
\frac{AC}{AB}=\frac{\sin B}{\sin C}
\]
结合题设得
\[
\frac{\sin B}{\sin C}=\frac{\cos B}{\cos C}
\]
所以 \(\sin B\cos C=\sin C\cos B\)，即 \(\sin(B-C)=0\). 由于 \(B\)，\(C\in(0,\pi)\)，且 \(B-C\in(-\pi,\pi)\)，只能有 \(B=C\).

\item 由 \(B=C\) 及三角形内角和，\(A+2B=\pi\). 题设给出
\[
\cos A=\cos(\pi-2B)=-\cos2B=-\frac13
\]
故 \(\cos2B=\frac13\). 因为 \(0<B<\frac{\pi}{2}\)，所以
\[
\sin2B=\sqrt{1-\cos^22B}=\frac{2\sqrt2}{3}
\]
于是
\(\sin4B=2\sin2B\cos2B=\frac{4\sqrt2}{9}\)，\(\cos4B=2\cos^22B-1=-\frac79\).
最后
\[
\sin\left(4B+\frac{\pi}{3}\right)
=\sin4B\cos\frac{\pi}{3}+\cos4B\sin\frac{\pi}{3}
=\frac{4\sqrt2-7\sqrt3}{18}
\]
\end{enumerate}
\end{solution}

\begin{problem}
有编号为 \(A_1\)，\(A_2\)，\(\cdots\)，\(A_{10}\) 的 10 个零件，测量其直径（单位：\(\mathrm{cm}\)），得到下面数据：

\begin{center}

直径 & 1.51 & 1.49 & 1.49 & 1.51 & 1.49 & 1.51 & 1.47 & 1.46 & 1.53 & 1.47 \\
\hline
\end{tabular}

其中直径在区间 \([1.48,1.52]\) 内的零件为一等品.

\item 从一等品零件中，随机抽取 2 个.

\end{enumerate}

\begin{enumerate}

\begin{problem}
如图，在五面体 \(ABCDEF\) 中，四边形 \(ADEF\) 是正方形，\(FA \perp\) 平面 \(ABCD\)，\(BC \parallel AD\)，\(CD = 1\)，\(AD = 2\sqrt{2}\)，\(\angle BAD = \angle CDA = 45^\circ\).

\begin{enumerate}
\item 求异面直线 \(CE\) 与 \(AF\) 所成角的余弦值；
\item 证明 \(CD \perp\) 平面 \(ABF\)；
\item 求二面角 \(B - EF - A\) 的正切值.
\end{enumerate}

\bitmapfigure[max width=0.36\linewidth]{img/2010/tianjin_liberal/q19_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 建立直角坐标系，使平面 \(ABCD\) 为 \(z=0\)，取
\(A=(0,0,0)\)，\(D=(2\sqrt2,0,0)\)，\(F=(0,0,2\sqrt2)\)，\(E=(2\sqrt2,0,2\sqrt2)\).
由 \(CD=1\)、\(\angle CDA=45^\circ\) 及 \(BC\parallel AD\)，可取
\(B=\left(\frac{\sqrt2}{2},\frac{\sqrt2}{2},0\right)\)，\(C=\left(\frac{3\sqrt2}{2},\frac{\sqrt2}{2},0\right)\).
于是
\(\overrightarrow{CE}=\left(\frac{\sqrt2}{2},-\frac{\sqrt2}{2},2\sqrt2\right)\)，\(\overrightarrow{AF}=(0,0,2\sqrt2)\).
两向量的夹角可作为异面直线 \(CE\) 与 \(AF\) 所成的角，故
\[
\cos\theta
=\frac{\overrightarrow{CE}\mathbin{\cdot}\overrightarrow{AF}}
{|\overrightarrow{CE}|\,|\overrightarrow{AF}|}
=\frac{8}{3\cdot2\sqrt2}
=\frac{2\sqrt2}{3}
\]

\item 由坐标可得
\(\overrightarrow{CD}=\left(\frac{\sqrt2}{2},-\frac{\sqrt2}{2},0\right)\)，\(\overrightarrow{AB}=\left(\frac{\sqrt2}{2},\frac{\sqrt2}{2},0\right)\)，\(\overrightarrow{AF}=(0,0,2\sqrt2)\).
平面 \(ABF\) 的一个法向量为
\[
\overrightarrow{AB}\times\overrightarrow{AF}=(2,-2,0)
\]
而 \(\overrightarrow{CD}\parallel(2,-2,0)\). 又点
\[
P=(\sqrt2,\sqrt2,0)=A+2\overrightarrow{AB}=C-\overrightarrow{CD}
\]
同时在直线 \(CD\) 和平面 \(ABF\) 内，所以 \(CD\) 与平面 \(ABF\) 相交且平行于该平面的法向量. 因此
\[
CD\perp\text{平面 }ABF
\]

\item 平面 \(AEF\) 的一个法向量可取 \(\bs{n}_1=(0,1,0)\). 由
\(\overrightarrow{EF}=(-2\sqrt2,0,0)\)，\(\overrightarrow{EB}=\left(-\frac{3\sqrt2}{2},\frac{\sqrt2}{2},-2\sqrt2\right)\)
可得平面 \(BEF\) 的一个法向量
\[
\bs{n}_2=\overrightarrow{EF}\times\overrightarrow{EB}
=(0,-8,-2)
\]
因此二面角 \(B-EF-A\) 的正切值等于两法向量夹角正切值的绝对值：
\[
\tan\theta
=\frac{|\bs{n}_1\times\bs{n}_2|}
{|\bs{n}_1\mathbin{\cdot}\bs{n}_2|}
=\frac{2}{8}
=\frac14
\]
\end{enumerate}
\end{solution}

\begin{problem}
已知函数 \(f(x) = ax^3 - \frac{3}{2}x^2 + 1\) （\(x \in \R\)），其中 \(a > 0\).

\begin{enumerate}
\item 若 \(a = 1\)，求曲线 \(y = f(x)\) 在点 \((2,f(2))\) 处的切线方程；
\item 若在区间 \(\left[-\frac{1}{2}, \frac{1}{2}\right]\) 上，\(f(x) > 0\) 恒成立，求 \(a\) 的取值范围.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 当 \(a=1\) 时，
\(f(x)=x^3-\frac32x^2+1\)，\(f(2)=8-6+1=3\).
又
\(f'(x)=3x^2-3x\)，\(f'(2)=6\).
所以曲线在 \((2,3)\) 处的切线为
\[
y-3=6(x-2)
\]
即 \(y=6x-9\).

\item 必要性方面，若 \(f(x)>0\) 在区间上恒成立，则特别有
\[
f\left(-\frac12\right)
=-\frac a8-\frac38+1
=\frac{5-a}{8}>0
\]
所以 \(a<5\). 结合题设 \(a>0\)，必要条件为 \(0<a<5\).

下面证明该条件充分. 对 \(x\in[-\frac12,0]\)，有
\[
 f'(x)=3x(ax-1)\geqslant0
\]
（因为 \(x\leqslant0\)、\(a>0\)，且 \(ax-1<0\)），故 \(f\) 在该区间上递增，最小值为
\[
f\left(-\frac12\right)=\frac{5-a}{8}>0
\]
当 \(0\leqslant x\leqslant\frac12\) 时分两种情况讨论：

若 \(0<a\leqslant2\)，则 \(ax\leqslant1\)，从而 \(f'(x)\leqslant0\)，最小值在 \(x=\frac12\) 处，且
\[
f\left(\frac12\right)=\frac a8-\frac38+1=\frac{a+5}{8}>0
\]

若 \(2<a<5\)，则 \(f'(x)\) 在 \(x=1/a\) 处由负变正，故最小值为
\[
f\left(\frac1a\right)
=1+\frac1{a^2}-\frac{3}{2a^2}
=1-\frac1{2a^2}>0
\]
所以当且仅当 \(0<a<5\) 时，区间 \([-\frac12,\frac12]\) 上恒有 \(f(x)>0\).
\end{enumerate}
\end{solution}

\begin{problem}
已知椭圆 \(\frac{x^2}{a^2} + \frac{y^2}{b^2} = 1\) （\(a > b > 0\)） 的离心率 \(e = \frac{\sqrt{3}}{2}\)，连接椭圆的四个顶点得到的菱形的面积为 4.

\begin{enumerate}
\item 求椭圆的方程；
\item 设直线 \(l\) 与椭圆相交于不同的两点 \(A\)，\(B\)，已知点 \(A\) 的坐标为 \((-a,0)\).

\begin{enumerate}
\item 若 \(|AB| = \frac{4\sqrt{2}}{5}\)，求直线 \(l\) 的倾斜角；
\item 若点 \(Q(0,y_0)\) 在线段 \(AB\) 的垂直平分线上，且 \(\overrightarrow{QA} \cdot \overrightarrow{QB} = 4\). 求 \(y_0\) 的值.
\end{enumerate}
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设椭圆的半长轴、半短轴分别为 \(a\)、\(b\)，焦距的一半为 \(c\). 由离心率
\[
\frac ca=\frac{\sqrt3}{2}
\]
得
\(c^2=\frac34a^2\)，\(b^2=a^2-c^2=\frac14a^2\).
四个顶点为 \((\pm a,0)\)、\((0,\pm b)\)，连接它们所得菱形的两条对角线长分别为 \(2a\)、\(2b\)，所以面积为
\[
\frac{(2a)(2b)}2=2ab=a^2=4
\]
因 \(a>0\)，得 \(a=2\)、\(b=1\). 椭圆方程为
\[
\frac{x^2}{4}+y^2=1
\]

\item
\begin{enumerate}
\item 因 \(A=(-2,0)\)，非竖直直线 \(l\) 可设为
\[
y=k(x+2)
\]
竖直直线 \(x=-2\) 与椭圆只有一个交点，不能满足交于不同两点，所以不漏掉情形. 将直线代入椭圆：
\[
\frac{x^2}{4}+k^2(x+2)^2=1
\]
整理得
\[
(1+4k^2)x^2+16k^2x+16k^2-4=0
\]
其中一个根为 \(x_A=-2\)，设另一个交点的横坐标为 \(x_B\). 由根与系数的关系，
\(x_B=\frac{2-8k^2}{1+4k^2}\)，\(x_B+2=\frac4{1+4k^2}\).
因此
\[
|AB|=\frac{4\sqrt{1+k^2}}{1+4k^2}
\]
由题设 \(|AB|=\frac{4\sqrt2}{5}\)，平方并化简得
\[
25(1+k^2)=2(1+4k^2)^2
\]
即
\[
32k^4-9k^2-23=0
\]
令 \(u=k^2\)，则
\[
32u^2-9u-23=0
\]
其根为
\[
u=\frac{9\pm55}{64}
\]
因 \(u\geqslant0\)，只能取 \(u=1\)，故 \(k=\pm1\). 直线的倾斜角为
\[
\frac{\pi}{4}\quad\text{或}\quad\frac{3\pi}{4}
\]

\item 由于 \(A\)，\(B\) 为不同交点，\(l\) 不可能是竖直直线 \(x=-2\)，故可设直线 \(l\) 的斜率为 \(k\)，则
\[
y=k(x+2)
\]
将其代入椭圆方程，得
\[
(1+4k^2)x^2+16k^2x+16k^2-4=0
\]
一个根为 \(-2\)，故另一个交点 \(B=(x_1,y_1)\) 满足
\(x_1=\frac{2-8k^2}{1+4k^2}\)，\(y_1=\frac{4k}{1+4k^2}\).

先考虑 \(k=0\). 此时 \(B=(2,0)\)，线段 \(AB\) 的垂直平分线为 \(x=0\)，而
\(\overrightarrow{QA}=(-2,-y_0)\)，\(\overrightarrow{QB}=(2,-y_0)\).
由数量积条件，
\[
-4+y_0^2=4
\]
所以
\[
y_0=\pm2\sqrt2
\]

再考虑 \(k\ne0\). 线段 \(AB\) 的中点为
\[
M=\left(-\frac{8k^2}{1+4k^2},\frac{2k}{1+4k^2}\right)
\]
由于 \(AB\) 的斜率为 \(k\)，其垂直平分线斜率为 \(-1/k\). 将 \(x=0\) 代入这条过 \(M\) 的直线，得
\[
y_0=-\frac{6k}{1+4k^2}
\]
又
\(\overrightarrow{QA}=(-2,-y_0)\)，\(\overrightarrow{QB}=(x_1,y_1-y_0)\).
代入数量积条件并化简：
\[
\frac{4(16k^4+15k^2-1)}{(1+4k^2)^2}=4
\]
所以
\[
7k^2=2
\]
于是
\[
y_0^2=\frac{36k^2}{(1+4k^2)^2}
=\frac{36\cdot\frac27}{\left(1+\frac87\right)^2}
=\frac{56}{25}
\]
从而
\[
y_0=\pm\frac{2\sqrt{14}}5
\]
综上，
\[
y_0=\pm2\sqrt2\quad\text{或}\quad y_0=\pm\frac{2\sqrt{14}}5
\]
\end{enumerate}
\end{enumerate}
\end{solution}

\begin{problem}
在数列 \(\{a_n\}\) 中，\(a_1 = 0\)，且对任意 \(k \in \mathbf{N}^{*}\)，\(a_{2k-1}\)，\(a_{2k}\)，\(a_{2k+1}\) 成等差数列，其公差为 \(2k\).

\begin{enumerate}
\item 证明 \(a_4\)，\(a_5\)，\(a_6\) 成等比数列；
\item 求数列 \(\{a_n\}\) 的通项公式；
\item 记 \(T_n = \frac{2^2}{a_2} + \frac{3^2}{a_3} + \cdots + \frac{n^2}{a_n}\)，证明 \(\frac{3}{2} < 2n - T_n \leqslant 2\)（\(n \geqslant 2\)）.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 因为 \(a_{2k-1}\)、\(a_{2k}\)、\(a_{2k+1}\) 成等差数列且公差为 \(2k\)，所以
\(a_{2k}=a_{2k-1}+2k\)，\(a_{2k+1}=a_{2k}+2k=a_{2k-1}+4k\).
取 \(k=1\)，得 \(a_2=2\)、\(a_3=4\)；取 \(k=2\)，得
\(a_4=8\)，\(a_5=12\).
再取 \(k=3\)，由 \(a_5\)，\(a_6\)，\(a_7\) 成等差数列且公差为 \(6\)，得 \(a_6=18\).
由于
\[
a_5^2=12^2=144=8\cdot18=a_4a_6
\]
所以 \(a_4\)、\(a_5\)、\(a_6\) 成等比数列，公比为 \(12/8=3/2\).

\item 由
\(a_{2k+1}=a_{2k-1}+4k\)，\(a_1=0\)
逐项累加得
\[
a_{2k+1}=4\sum_{j=1}^{k}j=2k(k+1)
\]
令 \(n=2k+1\)，得到
\(a_n=\frac{n^2-1}{2}\)，\((n\text{为奇数})\).
再由 \(a_{2k}=a_{2k-1}+2k\)，得
\[
a_{2k}=2k(k-1)+2k=2k^2=\frac{(2k)^2}{2}
\]
因此
\[
a_n=\begin{cases}
\dfrac{n^2-1}{2},&n\text{为奇数},\\[4pt]
\dfrac{n^2}{2},&n\text{为偶数}.
\end{cases}
\]

\item 当 \(k\) 为偶数时，由通项公式
\[
\frac{k^2}{a_k}=2
\]
当 \(k=2j+1\) 为奇数时
\[
\frac{k^2}{a_k}
=\frac{2k^2}{k^2-1}
=2+\frac{2}{k^2-1}
=2+\frac{1}{2j(j+1)}
\]
设 \(m\) 为不超过 \(n\) 的奇数 \(3\)，\(5\)，\(\ldots\) 的个数，则
\[
\sum_{j=1}^{m}\frac{1}{2j(j+1)}
=\frac12\sum_{j=1}^{m}\left(\frac1j-\frac1{j+1}\right)
=\frac12-\frac{1}{2(m+1)}
\]
若 \(n=2r\)，则 \(m=r-1\)，从而
\(T_n=2(n-1)+\frac12-\frac1n\)，\(2n-T_n=\frac32+\frac1n\).
若 \(n=2r+1\)，则 \(m=r\)，从而
\(T_n=2(n-1)+\frac12-\frac1{n+1}\)，\(2n-T_n=\frac32+\frac1{n+1}\).
当 \(n\geqslant2\) 时，上述两种情形均有 \(2n-T_n>\frac32\). 又当 \(n\) 为偶数时 \(1/n\leqslant1/2\)，当 \(n\) 为奇数时 \(1/(n+1)\leqslant1/2\)，所以 \(2n-T_n\leqslant2\). 因而
\[
\frac32<2n-T_n\leqslant2
\]
\end{enumerate}
\end{solution}
