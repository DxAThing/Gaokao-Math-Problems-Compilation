\section{单选题}

\begin{problem}
已知 \(A\)，\(B\) 均为集合 \(U = \{1, 3, 5, 7, 9\}\) 的子集，且 \(A \cap B = \{3\}\)，\(\complement_U B \cap A = \{9\}\)，则 \(A =\)（\quad）

\choices
    {\(\{1, 3\}\)}
    {\(\{3,7,9\}\)}
    {\(\{3,5,9\}\)}
    {\(\{3,9\}\)}
\end{problem}

\begin{problem}
设 \(a\)，\(b\) 为实数，若复数 \(\frac{1 + 2\i}{a + ib} = 1 + \i\)，则（\quad）

\choices

    {\(a = 3\)，\(b = 1\)}
    {\(a = \frac{1}{2}\)，\(b = \frac{3}{2}\)}
    {\(a = 1\)，\(b = 3\)}
\end{problem}
