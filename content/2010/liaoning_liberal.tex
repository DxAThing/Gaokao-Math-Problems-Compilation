\begin{problem}
已知集合 \(U=\{1,3,5,7,9\}\)，\(A=\{1,5,7\}\)，则 \(\complement_U A=\)（\quad）

\choices
    {\(\{1,3\}\)}
    {\(\{3,7,9\}\)}
    {\(\{3,5,9\}\)}
    {\(\{3,9\}\)}
\end{problem}

\begin{problem}
设 \(a\)，\(b\) 为实数，若复数 \(\frac{1+2\i}{a+b\i}=1+\i\)，则（\quad）

\choices
    {\(a=\frac{3}{2},b=\frac{1}{2}\)}
    {\(a=3\)，\(b=1\)}
    {\(a=\frac{1}{2}\)，\(b=\frac{3}{2}\)}
    {\(a=1\)，\(b=3\)}
\end{problem}

    {\(\exists x\in\R\)，\(f(x)\geqslant f(x_0)\)}

\begin{problem}

\end{problem}
