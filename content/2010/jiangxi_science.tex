\section{单选题}

\begin{problem}
已知 \((x + i) (1 - i) = y\)，则实数 \(x, y\) 分别为（\quad）

\choices
    {\(x = -1, y = 1\)}
    {\(x = -1, y = 2\)}
    {\(x = 1, y = 1\)}
    {\(x = 1, y = 2\)}
\end{problem}

\begin{problem}

\choices
    {\(\{x \mid -1 \leqslant x \leqslant 1\}\)}
    {\(\{x \mid x \geqslant 0\}\)}
    {\(\{x \mid 0 \leqslant x \leqslant 1\}\)}
    {\(\varnothing\)}
\end{problem}

\begin{problem}
已知函数 \( f(x) = (1 + \cot x)\sin^{2} x + m\sin\left(x + \frac{\pi}{4}\right)\sin\left(x - \frac{\pi}{4}\right) \).

\begin{enumerate}[label={（\arabic*）},itemsep=0.2em]
\item 当 \(m = 0\) 时，求 \(f(x)\) 在区间 \(\left[\frac{\pi}{8}, \frac{3\pi}{4}\right]\) 上的取值范围；
\item 当 \(\tan \alpha = 2\) 时，\(f(\alpha) = \frac{3}{5}\)，求 \(m\) 的值.
\end{enumerate}
\end{problem}

\begin{problem}
某迷宫有三个通道，进入迷宫的每个人都要经过一个智能门.首次到达此门，系统会随机（即等可能）为你打开一个通道.若是 1 号通道，则需要 1 小时走出迷宫；若是 2 号，3 号通道，则分别需要 2 小时，3 小时返回智能门.再次到达智能门时，系统会随机打开一个你未到过的通道，直至走出迷宫为止.令 \(\xi\) 表示走出迷宫所需的时间.

\begin{enumerate}[label={（\arabic*）},itemsep=0.2em]
\item 求 \(\xi\) 的分布列；
\item 求 \(\xi\) 的数学期望.
\end{enumerate}
\end{problem}

\begin{problem}
设函数 \( f(x) = \ln x + \ln(2 - x) + ax \,(a > 0) \).

\begin{enumerate}[label={（\arabic*）},itemsep=0.2em]
\item 当 \(a = 1\) 时，求 \(f(x)\) 的单调区间；
\item 若 \(f(x)\) 在 \((0,1]\) 上的最大值为 \(\frac{1}{2}\)，求 \(a\) 的值.
\end{enumerate}
\end{problem}

\begin{problem}

\begin{enumerate}[label={（\arabic*）},itemsep=0.2em]

\item 存在无穷多个互不相似的三角形 \(\Delta_{n}\)，其边长 \(a_{n}, b_{n}, c_{n}\) 为正整数且 \(a_{n}^{2}, b_{n}^{2}, c_{n}^{2}\) 成等差数列.
\end{enumerate}
\end{problem}
