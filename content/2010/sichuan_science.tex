\begin{problem}
已知数列 \(\{a_{n}\}\) 满足 \(a_1 = 0\)，\(a_2 = 2\)，且对任意 \(m\)，\(n \in \mathbf{N}^*\) 都有 \(a_{2m-1} + a_{2n-1} = 2a_{m+n-1} + 2(m - n)^2\).
\begin{enumerate}
\item 求 \(a_{3}\)，\(a_{5}\)；
\item 设 \( b_{n} = a_{2n + 1} - a_{2n - 1}\)（\(n \in \mathbf{N}^{*}\)），证明：数列 \(\{b_{n}\}\) 是等差数列；
\item 设 \(c_{n} = (a_{n + 1} - a_{n})q^{n - 1}\)（\(q\ne 0,\ n\in \mathbf{N}^{*}\)），求数列 \(\{c_n\}\) 的前 \(n\) 项和 \(S_{n}\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 短答案：\(a_{3}=6\)，\(a_{5}=20\).
\item 短答案：\(\{b_n\}\) 是首项为 \(6\)、公差为 \(8\) 的等差数列.
\item 短答案：\(c_n=2nq^{n-1}\)，且
\[
S_n=
\begin{cases}
n(n+1), & q=1,\\
\dfrac{2\left(nq^{n+1}-(n+1)q^n+1\right)}{\left(q-1\right)^2}, & q\ne 1.
\end{cases}
\]
\end{enumerate}
\end{solution}

\begin{problem}

\item 当 \(a = \e\)（\(\e\) 为自然对数的底数）时，证明：\(\sum_{k=2}^{n} g(k) > \frac{2 - n - n^2}{\sqrt{2n(n+1)}}\)；

\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 短答案：\(t\in[5,32]\).
\item 短答案：题设不等式成立.
\item 短答案：\(\left|\sum_{k=1}^{n}f(k)-n\right|<4\).
\end{enumerate}
\end{solution}
