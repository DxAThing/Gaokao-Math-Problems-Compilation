\section{单选题}

\begin{problem}
在等比数列 \(\{a_{n}\}\) 中，\(a_{2010} = 8a_{2007}\)，则公比 \(q\) 的值为（\quad）

\choices
    {\(2\)}
    {\(3\)}
    {\(4\)}
    {\(8\)}

\begin{answer}
A
\end{answer}

\begin{solution}
由等比数列的递推关系，\(a_{2010}=a_{2007}q^3\). 因为题设给出 \(a_{2010}=8a_{2007}\)，所以 \(q^3=8\)，解得 \(q=2\).
\end{solution}
\end{problem}

\begin{problem}
已知向量 \(\bs{a}\)，\(\bs{b}\) 满足 \(\bs{a} \cdot \bs{b} = 0\)，\(|\bs{a}| = 1\)，\(|\bs{b}| = 2\)，则 \(|2\bs{a} - \bs{b}| =\)（\quad）

\choices
    {\(0\)}
    {\( 2\sqrt{2} \)}
    {\(4\)}
    {\(8\)}

\begin{answer}
B
\end{answer}

\begin{solution}
因为 \(\bs{a}\) 与 \(\bs{b}\) 垂直，所以 \(\bs{a}\cdot\bs{b}=0\). 于是 \(\left|2\bs{a}-\bs{b}\right|^2=4|\bs{a}|^2-4\bs{a}\cdot\bs{b}+|\bs{b}|^2=4+0+4=8\)，从而 \(\left|2\bs{a}-\bs{b}\right|=2\sqrt{2}\)，故选 B.
\end{solution}
\end{problem}

\begin{problem}
\(\displaystyle\lim_{x\to 2}\left(\frac{4}{x^2 - 4} -\frac{1}{x - 2}\right) =\)（\quad）

\choices
    {\(-1\)}
    {\( -\frac{1}{4} \)}
    {\( \frac{1}{4} \)}
    {\(1\)}

\begin{answer}
B
\end{answer}

\begin{solution}
将分式通分，得
\[
\frac{4}{x^2-4}-\frac{1}{x-2}=\frac{4-(x+2)}{(x-2)(x+2)}=-\frac{1}{x+2}.
\]
因此
\[
\displaystyle\lim_{x\to2}\left(\frac{4}{x^2-4}-\frac{1}{x-2}\right)=\lim_{x\to2}\left(-\frac{1}{x+2}\right)=-\frac14
\]
故选 B.
\end{solution}
\end{problem}

\begin{problem}
设变量 \(x\)，\(y\) 满足约束条件 \(\begin{cases}
y \geqslant 0,\\
x - y + 1 \geqslant 0,\\
x + y - 3 \leqslant 0,
\end{cases}\) 则 \(z = 2x + y\) 的最大值为（\quad）

\choices
    {\(-2\)}
    {\(4\)}
    {\(6\)}
    {\(8\)}

\begin{answer}
C
\end{answer}

\begin{solution}
由约束条件得 \(x\geqslant y-1\)，\(x\leqslant 3-y\)，并且 \(y\geqslant0\). 固定 \(y\) 时，\(z=2x+y\) 随 \(x\) 增大而增大，所以取 \(x=3-y\). 此时 \(z=2(3-y)+y=6-y\leqslant6\)，当 \(y=0\)，\(x=3\) 时等号成立，故最大值为 \(6\)，选 C.
\end{solution}
\end{problem}

\begin{problem}
函数 \( f(x) = \frac{4^x + 1}{2^x} \) 的图象（\quad）

\choices
    {关于原点对称}
    {关于直线 \( y = x \) 对称}
    {关于 \(x\) 轴对称}
    {关于 \(y\) 轴对称}

\begin{answer}
D
\end{answer}

\begin{solution}
将函数式化为 \(f(x)=2^x+2^{-x}\). 因此 \(f(-x)=2^{-x}+2^x=f(x)\)，所以函数为偶函数，图象关于 \(y\) 轴对称，选 D.
\end{solution}
\end{problem}

\begin{problem}
已知函数 \(y = \sin\left(\omega x + \varphi\right)\)（\(\omega > 0, |\varphi| < \frac{\pi}{2}\)）的部分图象如图所示. 则（\quad）

\bitmapfigure[max width=0.42\linewidth]{img/2010/chongqing_science/q6_fig1.png}

\choices
    {\(\omega = 1\)，\(\varphi = \frac{\pi}{6}\)}
    {\(\omega = 1\)，\(\varphi = -\frac{\pi}{6}\)}
    {\(\omega = 2\)，\(\varphi = \frac{\pi}{6}\)}
    {\(\omega = 2\)，\(\varphi = -\frac{\pi}{6}\)}

\begin{answer}
D

\begin{problem}
已知 \(x > 0\)，\(y > 0\)，\(x + 2y + 2xy = 8\)，则 \(x + 2y\) 的最小值是（\quad）

\choices
    {\(3\)}
    {\(4\)}
    {\( \frac{9}{2} \)}
    {\(\frac{11}{2}\)}

\begin{answer}
B
\end{answer}

\begin{solution}
由题设，\((x+1)(2y+1)=x+2y+2xy+1=9\). 因为 \(x>0\)，\(y>0\)，所以 \(x+1>0\)，\(2y+1>0\). 由基本不等式，\(x+1+2y+1\geqslant2\sqrt{(x+1)(2y+1)}=6\)，故 \(x+2y\geqslant4\). 当 \(x+1=2y+1=3\)，即 \(x=2\)，\(y=1\) 时等号成立，所以最小值为 \(4\)，选 B.
\end{solution}
\end{problem}

\begin{problem}
直线 \( y = \frac{\sqrt{3}}{3} x + \sqrt{2} \) 与圆心为 \(D\) 的圆 \( \begin{cases}
x = \sqrt{3} + \sqrt{3}\cos \theta,\\
y = 1 + \sqrt{3}\sin \theta,
\end{cases}\) （\(\theta \in [0,2\pi)\)） 交于 \(A\)，\(B\) 两点，则直线 \(AD\) 与 \(BD\) 的倾斜角之和为（\quad）

\choices
    {\(\frac{7}{6}\pi\)}
    {\(\frac{5}{4}\pi\)}
    {\(\frac{4}{3}\pi\)}
    {\(\frac{5}{3}\pi\)}

\begin{answer}
C
\end{answer}

\begin{solution}
圆心为 \(D(\sqrt{3},1)\)，半径为 \(\sqrt{3}\). 设直线与圆的垂足为 \(H\)，则直线的方向向量可取 \((\sqrt{3},1)\)，而从 \(D\) 指向 \(H\) 的方向向量可取 \((-1,\sqrt{3})\). 将直线方程写成 \(\frac{\sqrt{3}}{3}x-y+\sqrt{2}=0\)，可得圆心到直线的距离为 \(\frac{\sqrt{6}}{2}\)，故弦的一半长为 \(\sqrt{3-\frac{3}{2}}=\frac{\sqrt{6}}{2}\). 由垂足方向和弦方向相加、相减可得两条半径的方向向量可取 \((3\sqrt{2}-\sqrt{6},3\sqrt{2}+\sqrt{6})\) 和 \((-(3\sqrt{2}+\sqrt{6}),3\sqrt{2}-\sqrt{6})\)，它们的倾斜角分别为 \(\frac{5\pi}{12}\) 和 \(\frac{11\pi}{12}\)，所以倾斜角之和为 \(\frac{4\pi}{3}\)，选 C.
\end{solution}
\end{problem}

\begin{problem}
某单位安排 7 位员工在 10 月 1 日至 7 日值班，每天 1 人，每人值班 1 天，若 7 位员工中的甲、乙排在相邻两天，丙不排在 10 月 1 日，丁不排在 10 月 7 日，则不同的安排方案共有（\quad）

\choices
    {\(504\) 种}
    {\(960\) 种}
    {\(1008\) 种}
    {\(1108\) 种}

\begin{answer}
C
\end{answer}

\begin{solution}
先把甲、乙看作一个整体. 甲、乙相邻时，这个整体在 7 个位置中有 6 种放法，甲、乙在整体内还有 2 种次序，因此不考虑其他限制时共有 \(6!\cdot2\) 种安排.

其中丙排在 10 月 1 日时，甲、乙整体只能放在第 2 至第 7 天中相邻的位置，共有 5 种放法，故有 \(5\cdot2\cdot4!\) 种安排. 类似地，丁排在 10 月 7 日时也有 \(5\cdot2\cdot4!\) 种安排.

丙排在 10 月 1 日且丁排在 10 月 7 日时，甲、乙整体在第 2 至第 6 天中有 4 种相邻放法，故有 \(4\cdot2\cdot3!\) 种安排. 由容斥原理，不同的安排方案数为
\[
6!\cdot2-2\cdot5\cdot2\cdot4!+4\cdot2\cdot3!=1440-480+48=1008.
\]
故选 C.
\end{solution}
\end{problem}

\begin{problem}
到两互相垂直的异面直线的距离相等的点，在过其中一条直线且平行于另一条直线的平面内的轨迹是（\quad）

\choices
    {直线}
    {椭圆}
    {抛物线}
    {双曲线}

\begin{answer}
D
\end{answer}

\begin{solution}
设过第一条直线且平行于第二条直线的平面为 \(\pi\)，把第二条直线在 \(\pi\) 上的正投影记为 \(l_2'\). 由于两条异面直线互相垂直，\(l_2'\) 与第一条直线在平面 \(\pi\) 内相交且互相垂直. 设两条直线之间的距离为 \(h\)，则 \(h>0\).

在平面 \(\pi\) 内建立直角坐标系，使第一条直线为 \(x\) 轴，\(l_2'\) 为 \(y\) 轴. 对平面 \(\pi\) 内任一点 \(P(x,y)\)，有
\[
d(P,l_1)=|y|,
\qquad d(P,l_2)^2=d(P,l_2')^2+h^2=x^2+h^2.
\]
题设条件给出
\[
|y|=\sqrt{x^2+h^2},
\quad\text{即}\quad y^2-x^2=h^2.
\]
这是以坐标轴方向为渐近线的双曲线，故选 D.
\end{solution}
\end{problem}

\section{填空题}

\begin{problem}
已知复数 \(z = 1 + \i\)，则 \(\frac{2}{z} - z =\fillinblank{}\).

\begin{answer}
\(-2\i\)
\end{answer}

\begin{solution}
因为 \(\frac{2}{z}=\frac{2}{1+\i}=\frac{2(1-\i)}{(1+\i)(1-\i)}=1-\i\)，所以 \(\frac{2}{z}-z=(1-\i)-(1+\i)=-2\i\).
\end{solution}
\end{problem}

\begin{problem}
设 \(U = \{0, 1, 2, 3\}\)，\(A = \{x \in U \mid x^2 + mx = 0\}\)，若 \(\complement_U A = \{1, 2\}\)，则实数 \(m =\fillinblank{}\).

\begin{answer}
\(-3\)
\end{answer}

\begin{solution}
由 \(\complement_U A=\{1,2\}\)，知 \(A=\{0,3\}\). 对任意实数 \(m\)，\(0^2+m\cdot0=0\)，所以 \(0\in A\) 自动成立. 又因为 \(3\in A\)，应满足
\[
3^2+3m=0,
\]
解得 \(m=-3\). 此时 \(1^2-3=-2\ne0\)，\(2^2-6=-2\ne0\)，确有 \(A=\{0,3\}\)，故 \(m=-3\).
\end{solution}
\end{problem}

\begin{problem}
某篮球队员在比赛中每次罚球的命中率相同，且在两次罚球中至多命中一次的概率为 \(\frac{16}{25}\)，则该队员每次罚球的命中率为\fillinblank{}.

\begin{answer}
\(\frac{3}{5}\)
\end{answer}

\begin{solution}
设每次罚球命中的概率为 \(p\)，则两次罚球都命中的概率为 \(p^2\). 由“至多命中一次”的对立事件为“两次都命中”，得 \(1-p^2=\frac{16}{25}\)，所以 \(p^2=\frac{9}{25}\). 由于 \(0\leqslant p\leqslant1\)，故 \(p=\frac{3}{5}\).
\end{solution}
\end{problem}

\begin{problem}
已知以 \(F\) 为焦点的抛物线 \(y^{2} = 4x\) 上的两点 \(A\)，\(B\) 满足 \(\overrightarrow{AF} = 3\overrightarrow{FB}\)，则弦 \(AB\) 的中点到准线的距离为\fillinblank{}.

\begin{answer}
\(\frac{8}{3}\)
\end{answer}

\begin{solution}
抛物线 \(y^2=4x\) 可写成 \(y^2=4px\)，故 \(p=1\)，焦点为 \(F(1,0)\)，准线为 \(x=-1\).

设 \(A\left(\frac{u^2}{4},u\right)\)，\(B\left(\frac{v^2}{4},v\right)\). 由 \(\overrightarrow{AF}=3\overrightarrow{FB}\)，比较纵坐标得
\[
0-u=3(v-0),
\quad\text{即}\quad u=-3v.
\]
比较横坐标得
\[
1-\frac{u^2}{4}=3\left(\frac{v^2}{4}-1\right).
\]
代入 \(u=-3v\)，得
\[
1-\frac{9v^2}{4}=\frac{3v^2}{4}-3,
\quad\text{即}\quad 3v^2=4.
\]
所以 \(x_A=\frac{u^2}{4}=3\)，\(x_B=\frac{v^2}{4}=\frac13\). 弦 \(AB\) 的中点横坐标为
\[
x_M=\frac{x_A+x_B}{2}=\frac{3+\frac13}{2}=\frac53.
\]
因此中点到准线 \(x=-1\) 的距离为
\[
x_M-(-1)=\frac53+1=\frac83.
\]
\end{solution}
\end{problem}

\begin{problem}
已知函数 \( f(x) \) 满足： \(f(1) = \frac{1}{4}\)，\(4f(x)f(y) = f(x + y) + f(x - y)\) （\(x, y \in \R\)），则 \( f(2010) = \)\fillinblank{}.

\begin{answer}
\(\frac{1}{2}\)
\end{answer}

\begin{solution}
令 \(y=0\)，原函数方程变为
\[
4f(x)f(0)=f(x)+f(x)=2f(x).
\]
取 \(x=1\)，由 \(f(1)=\frac14\ne0\) 得 \(f(0)=\frac12\). 再令 \(x=0\)，可得
\[
4f(0)f(y)=f(y)+f(-y),
\]
从而 \(f(-y)=f(y)\).

令 \(y=1\)，利用 \(4f(1)=1\)，得到
\[
f(x+1)+f(x-1)=f(x).
\]
对整数依次取值，结合 \(f(0)=\frac12\)、\(f(1)=\frac14\)，有
\[
f(2)=f(1)-f(0)=-\frac14,
\]
\[
f(3)=f(2)-f(1)=-\frac12,
\qquad
f(4)=f(3)-f(2)=-\frac14,
\]
\[
f(5)=f(4)-f(3)=\frac14,
\qquad
f(6)=f(5)-f(4)=\frac12.
\]
该递推关系给出 \(f(n+6)=f(n)\)（例如由 \(f(7)=f(6)-f(5)=\frac14=f(1)\)、\(f(8)=f(7)-f(6)=-\frac14=f(2)\) 可归纳得到），而 \(2010=6\cdot335\)，所以
\[
f(2010)=f(0)=\frac12.
\]
\end{solution}
\end{problem}

\begin{problem}
设函数 \(f(x) = \cos \left(x + \frac{2\pi}{3}\right) + 2\cos^2 \frac{x}{2}\)，\(x \in \R\).

\begin{enumerate}
\item 求 \( f(x) \) 的值域；
\item 记 \(\triangle ABC\) 的内角 \(A\)，\(B\)，\(C\) 的对边长分别为 \(a\)，\(b\)，\(c\). 若 \(f(B) = 1\)，\(b = 1\)，\(c = \sqrt{3}\)，求 \(a\) 的值.
\end{enumerate}

\begin{solution}
\begin{enumerate}
\item 由 \(2\cos^2\frac{x}{2}=1+\cos x\)，得 \(f(x)=1+\cos x+\cos\left(x+\frac{2\pi}{3}\right)=1+\frac{1}{2}\cos x-\frac{\sqrt{3}}{2}\sin x=1+\cos\left(x+\frac{\pi}{3}\right)\). 因此 \(0\leqslant f(x)\leqslant2\)，值域为 \([0,2]\).
\item 由 \(f(B)=1\) 得 \(\cos\left(B+\frac{\pi}{3}\right)=0\)，所以 \(B=\frac{\pi}{6}+k\pi\). 因为 \(0<B<\pi\)，故 \(B=\frac{\pi}{6}\). 在 \(\triangle ABC\) 中，由余弦定理，\(b^2=a^2+c^2-2ac\cos B\)，即 \(1=a^2+3-2a\sqrt{3}\cdot\frac{\sqrt{3}}{2}=a^2-3a+3\)，从而 \(a^2-3a+2=0\)，解得 \(a=1\) 或 \(a=2\). 两者均满足 \(|\sqrt{3}-1|<a<\sqrt{3}+1\)，故 \(a=1\) 或 \(a=2\).
\end{enumerate}
\end{solution}
\end{problem}

\begin{problem}
在甲、乙等 6 个单位参加的一次“唱读讲传”演出活动中，每个单位的节目集中安排在一起，若采用抽签的方式随机确定各单位的演出顺序（序号为 \(1\)，\(2\)，\(\cdots\)，\(6\)），求：

\begin{enumerate}
\item 甲、乙两单位的演出序号至少有一个为奇数的概率；
\item 甲，乙两单位之间的演出单位个数 \(\xi\) 的分布列与期望.
\end{enumerate}

\begin{solution}
\begin{enumerate}
\item 甲、乙两单位的演出序号是从 \(1,2,\ldots,6\) 中随机取出的两个不同序号. 两序号都为偶数的概率为 \(\frac{\mathrm C_3^2}{\mathrm C_6^2}=\frac{1}{5}\)，所以至少有一个为奇数的概率为 \(1-\frac{1}{5}=\frac{4}{5}\).
\item 两单位的演出序号之差为 \(d\) 时，满足条件的无序位置对有 \(6-d\) 个，全部无序位置对有 \(\mathrm C_6^2=15\) 个. 由于 \(\xi=d-1\)，所以 \(\xi\) 的分布列为
\begin{center}
\begin{tabular}{c|ccccc}
\(\xi\) & \(0\) & \(1\) & \(2\) & \(3\) & \(4\) \\
\hline
\(P\) & \(\frac{5}{15}\) & \(\frac{4}{15}\) & \(\frac{3}{15}\) & \(\frac{2}{15}\) & \(\frac{1}{15}\)
\end{tabular}
\end{center}
从而 \(E(\xi)=0\cdot\frac{5}{15}+1\cdot\frac{4}{15}+2\cdot\frac{3}{15}+3\cdot\frac{2}{15}+4\cdot\frac{1}{15}=\frac{4}{3}\).
\end{enumerate}
\end{solution}
\end{problem}

\begin{problem}
已知函数 \( f(x) = \frac{x - 1}{x + a} + \ln\left(x + 1\right) \)，其中实数 \( a \neq 1 \).

\begin{enumerate}
\item 若 \(a = 2\)，求曲线 \(y = f(x)\) 在点 \((0, f(0))\) 处的切线方程；
\item 若 \( f(x) \) 在 \( x = 1 \) 处取得极值，试讨论 \( f(x) \) 的单调性.
\end{enumerate}

\begin{solution}
\begin{enumerate}
\item 当 \(a=2\) 时，\(f(0)=-\frac{1}{2}\)，且 \(f'(x)=\frac{3}{(x+2)^2}+\frac{1}{x+1}\)，所以 \(f'(0)=\frac{7}{4}\). 因此曲线在 \((0,f(0))=\left(0,-\frac{1}{2}\right)\) 处的切线方程为 \(y+\frac{1}{2}=\frac{7}{4}x\)，即 \(y=\frac{7}{4}x-\frac{1}{2}\).
\item 因为函数在 \(x=1\) 处取得极值，且 \(x=1\) 必须属于定义域，所以 \(a\neq-1\). 又 \(f'(1)=\frac{1}{a+1}+\frac{1}{2}=0\)，解得 \(a=-3\). 此时定义域为 \((-1,3)\cup(3,+\infty)\)，并且
\[
f'(x)=\frac{-2}{(x-3)^2}+\frac{1}{x+1}=\frac{(x-1)(x-7)}{(x-3)^2(x+1)}.
\]
由于定义域内分母为正，故 \(f'(x)>0\) 在 \((-1,1)\) 和 \((7,+\infty)\) 上成立，\(f'(x)<0\) 在 \((1,3)\) 和 \((3,7)\) 上成立. 因此 \(f(x)\) 在 \((-1,1)\)、\((7,+\infty)\) 上单调递增，在 \((1,3)\)、\((3,7)\) 上单调递减.
\end{enumerate}
\end{solution}
\end{problem}

\begin{problem}
如图，四棱锥 \(P - ABCD\) 中，底面 \(ABCD\) 为矩形，\(PA \perp\) 底面 \(ABCD\)，\(PA = AB = \sqrt{6}\)，点 \(E\) 是棱 \(PB\) 的中点.

\begin{enumerate}
\item 求直线 AD 与平面 PBC 的距离；
\item 若 \(AD = \sqrt{3}\)，求二面角 \(A - EC - D\) 的平面角的余弦值.
\end{enumerate}

\bitmapfigure[max width=0.36\linewidth]{img/2010/chongqing_science/q19_fig1.png}

\begin{solution}
\begin{enumerate}
\item 建立如图所示的直角坐标系，设 \(s=\sqrt{6}\)，\(AD=d\)，取 \(A(0,0,0)\)，\(B(s,0,0)\)，\(D(0,d,0)\)，\(C(s,d,0)\)，\(P(0,0,s)\). 由 \(E\) 为 \(PB\) 的中点，得 \(E\left(\frac{s}{2},0,\frac{s}{2}\right)\). 平面 \(PBC\) 的方程为 \(x+z=s\)，直线 \(AD\) 的方向为 \(y\) 轴方向，且与平面 \(PBC\) 平行. 因此直线 \(AD\) 到平面 \(PBC\) 的距离为 \(\frac{s}{\sqrt{1^2+1^2}}=\frac{\sqrt{6}}{\sqrt{2}}=\sqrt{3}\).
\item 当 \(d=\sqrt{3}\) 时，取平面 \(AEC\) 的两个方向向量 \(\overrightarrow{AC}=(s,d,0)\)、\(\overrightarrow{AE}=\left(\frac{s}{2},0,\frac{s}{2}\right)\)，取平面 \(DEC\) 的两个方向向量 \(\overrightarrow{DC}=(s,0,0)\)、\(\overrightarrow{DE}=\left(\frac{s}{2},-d,\frac{s}{2}\right)\). 相应法向量可取 \(\bs{n}_1=(ds,-s^2,-ds)\)、\(\bs{n}_2=(0,-s^2,-2ds)\). 代入 \(s=\sqrt{6}\)、\(d=\sqrt{3}\)，得 \(\bs{n}_1=(3\sqrt{2},-6,-3\sqrt{2})\)、\(\bs{n}_2=(0,-6,-6\sqrt{2})\). 所以二面角的平面角 \(\alpha\) 满足 \(\cos\alpha=\frac{|\bs{n}_1\cdot\bs{n}_2|}{|\bs{n}_1||\bs{n}_2|}=\frac{72}{\sqrt{72}\sqrt{108}}=\frac{\sqrt{6}}{3}\).
\end{enumerate}
\end{solution}

\begin{problem}
已知以原点 \(O\) 为中心，\(F(\sqrt{5}, 0)\) 为右焦点的双曲线 \(C\) 的离心率 \(e = \frac{\sqrt{5}}{2}\).

\begin{enumerate}
\item 求双曲线 \(C\) 的标准方程及其渐近线方程；
\item 如图，已知过点 \(M(x_{1}, y_{1})\) 的直线 \(l_{1}: x_{1}x + 4y_{1}y = 4\) 与过点 \(N(x_{2}, y_{2})\)（其中 \(x_{2} \neq x_{1}\)）的直线 \(l_{2}: x_{2}x + 4y_{2}y = 4\) 的交点 \(E\) 在双曲线 \(C\) 上，直线 \(MN\) 与两条渐近线分别交于 \(G\)，\(H\) 两点，求 \(\triangle OGH\) 的面积.
\end{enumerate}

\bitmapfigure[max width=0.38\linewidth]{img/2010/chongqing_science/q20_fig1.png}

\begin{solution}
\begin{enumerate}
\item 设双曲线的标准方程为 \(\frac{x^2}{a^2}-\frac{y^2}{b^2}=1\)，其中 \(a>0\)，\(b>0\)，焦半距为 \(c\). 由题意，\(c=\sqrt{5}\)，\(e=\frac{c}{a}=\frac{\sqrt{5}}{2}\)，所以 \(a=2\). 又 \(b^2=c^2-a^2=5-4=1\)，故双曲线方程为 \(\frac{x^2}{4}-y^2=1\)，渐近线方程为 \(y=\pm\frac{x}{2}\).
\item 设 \(E(x_E,y_E)\) 为两直线的交点. 因为 \(E\in l_1\cap l_2\)，所以 \(x_1x_E+4y_1y_E=4\)，\(x_2x_E+4y_2y_E=4\). 这说明 \(M(x_1,y_1)\)、\(N(x_2,y_2)\) 均在直线 \(x_E x+4y_E y=4\) 上，故直线 \(MN\) 的方程为 \(x_E x+4y_E y=4\). 分别联立渐近线 \(y=\frac{x}{2}\) 和 \(y=-\frac{x}{2}\)，得
\[
G\left(\frac{4}{x_E+2y_E},\frac{2}{x_E+2y_E}\right),\qquad H\left(\frac{4}{x_E-2y_E},-\frac{2}{x_E-2y_E}\right).
\]
于是 \(S_{\triangle OGH}=\frac{1}{2}\left|x_Gy_H-y_Gx_H\right|=\frac{8}{|x_E^2-4y_E^2|}\). 又因为 \(E\) 在双曲线上，满足 \(\frac{x_E^2}{4}-y_E^2=1\)，即 \(x_E^2-4y_E^2=4\)，所以 \(S_{\triangle OGH}=2\).
\end{enumerate}
\end{solution}

\begin{problem}
在数列 \(\{a_{n}\}\) 中，\(a_{1} = 1\)，\(a_{n + 1} = c a_{n} + c^{n + 1}(2n + 1)\)（\(n \in \mathbf{N}^{*}\)），其中实数 \(c \neq 0\).

\begin{enumerate}
\item 求 \( \{a_{n}\} \) 的通项公式；
\item 若对一切 \(k \in \mathbf{N}^*\) 有 \(a_{2k} > a_{2k-1}\)，求 \(c\) 的取值范围.
\end{enumerate}

\begin{solution}
\begin{enumerate}
\item 令 \(b_n=\frac{a_n}{c^n}\). 由递推式得 \(b_{n+1}=\frac{a_{n+1}}{c^{n+1}}=\frac{a_n}{c^n}+2n+1=b_n+2n+1\)，且 \(b_1=\frac{1}{c}\). 因此 \(b_n=\frac{1}{c}+\sum_{j=1}^{n-1}(2j+1)=\frac{1}{c}+n^2-1\)，从而 \(a_n=c^n b_n=c^{n-1}\left(c(n^2-1)+1\right)\).
\item 由通项公式，
\[
a_{2k}-a_{2k-1}=c^{2k-2}\left((4k^2-1)c^2+(-4k^2+4k+1)c-1\right).
\]
由于 \(c^{2k-2}>0\)，条件等价于 \(D_k(c)>0\)，其中 \(D_k(c)=(4k^2-1)c^2+(-4k^2+4k+1)c-1\). 特别地，\(D_1(c)=3c^2+c-1>0\)，所以 \(c<\frac{-1-\sqrt{13}}{6}\) 或 \(c>\frac{-1+\sqrt{13}}{6}\). 另一方面，直接计算得 \(D_k(c)=D_1(c)+4(k-1)c((k+1)c-k)\). 当 \(c<\frac{-1-\sqrt{13}}{6}<0\) 时，后加项为正，故所有 \(D_k(c)>0\). 当 \(c\geqslant1\) 时，\(D_1(c)>0\)，且后加项非负，故所有 \(D_k(c)>0\). 若 \(0<c<1\)，则 \(D_k(c)/(4k^2)\to c^2-c<0\)，故充分大的 \(k\) 使 \(D_k(c)<0\)；若 \(\frac{-1-\sqrt{13}}{6}\leqslant c\leqslant0\)，则 \(D_1(c)\leqslant0\)，也不满足严格不等式. 综上，\(c\) 的取值范围为 \(\left(-\infty,\frac{-1-\sqrt{13}}{6}\right)\cup[1,+\infty)\).
\end{enumerate}
\end{solution}
\end{problem}
