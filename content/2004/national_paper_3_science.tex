\section{单选题}
\begin{problem}
设集合 \(M = \{(x, y)\mid x^2 + y^2 = 1, x \in \R, y \in \R\}\)，\(N = \{(x, y)\mid x^2 - y = 0, x \in \R, y \in \R\}\)，则集合 \(M \cap N\) 中元素的个数为（\quad）

\choices
    {\(1\)}
    {\(2\)}
    {\(3\)}
    {\(4\)}
\end{problem}

\begin{answer}B\end{answer}

\begin{solution}
由 \(N\) 的条件得 \(y=x^2\)，代入 \(M\) 的条件可得 \(x^2+x^4=1\).令 \(t=x^2\geqslant0\)，则 \(t^2+t-1=0\)，所以 \(t=\frac{\sqrt{5}-1}{2}\)，另一个根为负数，不合题意.于是 \(x=\pm\sqrt{\frac{\sqrt{5}-1}{2}}\)，且对应的 \(y=x^2\) 相同，故共有 \(2\) 个交集元素，选择 B.
\end{solution}

\begin{problem}
函数 \(y = \left|\sin \frac{x}{2}\right|\) 的最小正周期是（\quad）

\choices
    {\(\frac{\pi}{2}\)}
    {\(\pi\)}
    {\(2\pi\)}
    {\(4\pi\)}
\end{problem}

\begin{answer}C\end{answer}

\begin{solution}
因为 \(\sin t\) 的周期为 \(2\pi\)，而取绝对值后 \(\lvert\sin t\rvert\) 的最小正周期为 \(\pi\)，所以令 \(t=\frac{x}{2}\) 时，函数的周期 \(T\) 满足 \(\frac{T}{2}=\pi\)，即 \(T=2\pi\).故选择 C.
\end{solution}

\begin{problem}
设数列 \(\{a_{n}\}\) 是等差数列，且 \(a_{2} = -6\)，\(a_{8} = 6\)，\(S_{n}\) 是数列 \(\{a_{n}\}\) 的前 \(n\) 项和，则（\quad）

\choices
    {\(S_{4} < S_{5}\)}
    {\(S_{4} = S_{5}\)}
    {\(S_{6} < S_{5}\)}
    {\(S_{6} = S_{5}\)}
\end{problem}

\begin{answer}B\end{answer}

\begin{solution}
设等差数列的公差为 \(d\)，则 \(a_8-a_2=6d=12\)，从而 \(d=2\)，\(a_1=-8\).因此 \(a_4=-2\)，\(a_5=0\)，\(a_6=2\)，有 \(S_4=2(a_1+a_4)=-20\)，\(S_5=\frac{5}{2}(a_1+a_5)=-20\)，而 \(S_6=3(a_1+a_6)=-18\).所以 \(S_4=S_5\)，选择 B.
\end{solution}

\begin{problem}
圆 \(x^{2} + y^{2} - 4x = 0\) 在点 \(P(1, \sqrt{3})\) 处的切线方程为（\quad）

\choices
    {\(x + \sqrt{3} y - 2 = 0\)}
    {\(x + \sqrt{3} y - 4 = 0\)}
    {\(x - \sqrt{3} y + 4 = 0\)}
    {\(x - \sqrt{3} y + 2 = 0\)}
\end{problem}

\begin{answer}D\end{answer}

\begin{solution}
圆的圆心为 \(O(2,0)\).半径 \(OP\) 的斜率为 \(\frac{\sqrt{3}}{-1}=-\sqrt{3}\)，所以切线斜率为 \(\frac{1}{\sqrt{3}}\).切线过 \(P(1,\sqrt{3})\)，故 \(y-\sqrt{3}=\frac{1}{\sqrt{3}}(x-1)\)，整理得 \(x-\sqrt{3}y+2=0\).故选择 D.
\end{solution}

\begin{problem}
函数 \(y = \sqrt{\log_{\frac{1}{2}}(x^2 - 1)}\) 的定义域为（\quad）

\choices
    {\(\left[-\sqrt{2}, -1\right) \cup (1, \sqrt{2}]\)}
    {\((-\sqrt{2}, -1) \cup (1, \sqrt{2})\)}
    {\(\left[-2, -1\right) \cup (1, 2]\)}
    {\((-2, -1) \cup (1, 2)\)}
\end{problem}

\begin{answer}A\end{answer}

\begin{solution}
根式有意义要求 \(\log_{\frac12}(x^2-1)\geqslant0\).由于底数 \(\frac12\) 小于 \(1\)，这等价于 \(0<x^2-1\leqslant1\)，即 \(1<x^2\leqslant2\).因此定义域为 \(\left[-\sqrt{2},-1\right)\cup\left(1,\sqrt{2}\right]\)，选择 A.
\end{solution}

\begin{problem}
设复数 \(z\) 的辐角的主值为 \(\frac{2\pi}{3}\)，虚部为 \(\sqrt{3}\)，则 \(z^2 =\)（\quad）

\choices
    {\(-2 - 2\sqrt{3}\i\)}
    {\(-2\sqrt{3} - 2\i\)}
    {\(2 + 2\sqrt{3}\i\)}
    {\(2\sqrt{3} + 2\i\)}
\end{problem}

\begin{answer}A\end{answer}

\begin{solution}
设 \(z=r\left(\cos\frac{2\pi}{3}+\i\sin\frac{2\pi}{3}\right)\).由虚部为 \(\sqrt{3}\) 得 \(r\cdot\frac{\sqrt{3}}{2}=\sqrt{3}\)，所以 \(r=2\)，从而 \(z=-1+\sqrt{3}\i\).于是 \(z^2=(-1+\sqrt{3}\i)^2=1-2\sqrt{3}\i-3=-2-2\sqrt{3}\i\)，选择 A.
\end{solution}

\begin{problem}
设双曲线的焦点在 \(x\) 轴上，两条渐近线为 \(y = \pm \frac{1}{2} x\)，则该双曲线的离心率 \(e =\)（\quad）

\choices
    {\(5\)}
    {\(\sqrt{5}\)}
    {\(\frac{\sqrt{5}}{2}\)}
    {\(\frac{5}{4}\)}
\end{problem}

\begin{answer}C\end{answer}

\begin{solution}
焦点在 \(x\) 轴上的双曲线可写为 \(\frac{x^2}{a^2}-\frac{y^2}{b^2}=1\)，其渐近线为 \(y=\pm\frac{b}{a}x\).由题意 \(\frac{b}{a}=\frac12\)，所以 \(c^2=a^2+b^2=\frac54a^2\).因此离心率 \(e=\frac{c}{a}=\frac{\sqrt{5}}{2}\)，选择 C.
\end{solution}

\begin{problem}
不等式 \(1 < |x + 1| < 3\) 的解集为（\quad）

\choices
    {\((0,2)\)}
    {\((-2,0) \cup (2,4)\)}
    {\((-4,0)\)}
    {\((-4,-2) \cup (0,2)\)}
\end{problem}

\begin{answer}D\end{answer}

\begin{solution}
由 \(1<\lvert x+1\rvert<3\)，分两种情况：当 \(x+1>0\) 时，\(1<x+1<3\)，得 \(0<x<2\)；当 \(x+1<0\) 时，\(1<-x-1<3\)，得 \(-4<x<-2\).所以解集为 \((-4,-2)\cup(0,2)\)，选择 D.
\end{solution}

\begin{problem}
正三棱锥的底面边长为 \(2\)，侧面均为直角三角形，则此三棱锥的体积为（\quad）

\choices
    {\(\frac{2\sqrt{2}}{3}\)}
    {\(\sqrt{2}\)}
    {\(\frac{\sqrt{2}}{3}\)}
    {\(\frac{4\sqrt{2}}{3}\)}
\end{problem}

\begin{answer}C\end{answer}

\begin{solution}
正三棱锥的底面为边长 \(2\) 的正三角形，底面面积为 \(\sqrt{3}\).每个侧面是等腰直角三角形，底边 \(2\) 是斜边，所以两条侧棱均为 \(\sqrt{2}\).设顶点到底面中心的高为 \(h\)，正三角形外接圆半径为 \(\frac{2\sqrt{3}}{3}\)，于是 \(h^2=\left(\sqrt{2}\right)^2-\left(\frac{2\sqrt{3}}{3}\right)^2=2-\frac43=\frac23\)，即 \(h=\frac{\sqrt{6}}{3}\).故体积 \(V=\frac13\cdot\sqrt{3}\cdot\frac{\sqrt{6}}{3}=\frac{\sqrt{2}}{3}\)，选择 C.
\end{solution}

\begin{problem}
在 \(\triangle ABC\) 中，\(AB = 3\)，\(BC = \sqrt{13}\)，\(AC = 4\)，则边 \(AC\) 上的高为（\quad）

\choices
    {\(\frac{3\sqrt{2}}{2}\)}
    {\(\frac{3\sqrt{3}}{2}\)}
    {\(\frac{3}{2}\)}
    {\(3\sqrt{3}\)}
\end{problem}

\begin{answer}B\end{answer}

\begin{solution}
设 \(D\) 为 \(AC\) 上高的垂足，令 \(AD=t\)，高为 \(h\)，则 \(DC=4-t\).由勾股定理 \(t^2+h^2=AB^2=9\)，\((4-t)^2+h^2=BC^2=13\).两式相减得 \(16-8t=4\)，所以 \(t=\frac32\)，进而 \(h^2=9-\frac94=\frac{27}{4}\)，故 \(h=\frac{3\sqrt{3}}2\)，选择 B.
\end{solution}

\begin{problem}
设函数 \(f(x) = \begin{cases}
(x + 1)^2, & x < 1,\\
4 - \sqrt{x - 1}, & x \geqslant 1,
\end{cases}\) 则使得 \(f(x) \geqslant 1\) 的自变量 \(x\) 的取值范围为（\quad）

\choices
    {\((-\infty, -2] \cup [0, 10]\)}
    {\((-\infty, -2] \cup [0, 1]\)}
    {\((-\infty, -2] \cup [1, 10]\)}
    {\([-2,0] \cup [1,10]\)}
\end{problem}

\begin{answer}A\end{answer}

\begin{solution}
当 \(x<1\) 时，\((x+1)^2\geqslant1\) 等价于 \(x\leqslant-2\) 或 \(x\geqslant0\)，结合 \(x<1\) 得 \((-\infty,-2]\cup[0,1)\).当 \(x\geqslant1\) 时，\(4-\sqrt{x-1}\geqslant1\) 等价于 \(\sqrt{x-1}\leqslant3\)，所以 \(1\leqslant x\leqslant10\).合并得 \((-\infty,-2]\cup[0,10]\)，选择 A.
\end{solution}

\begin{problem}
将 \(4\text{ 名}\) 教师分配到 \(3\text{ 所}\) 中学任教，每所中学至少 \(1\text{ 名}\)，则不同的分配方案共有（\quad）

\choices
    {\(12\text{ 种}\)}
    {\(24\text{ 种}\)}
    {\(36\text{ 种}\)}
    {\(48\text{ 种}\)}
\end{problem}

\begin{answer}C\end{answer}

\begin{solution}
先按人数分配.由于每所学校至少有一名教师，人数分配只能为 \(2\)，\(1\)，\(1\).从 \(4\) 名教师中选出分到同一所学校的 \(2\) 人，有 \(\mathrm C_4^2\) 种；选定这所学校有 \(3\) 种；其余两名教师分到另外两所学校有 \(2\) 种.因此方案数为 \(\mathrm C_4^2\cdot3\cdot2=36\)，选择 C.
\end{solution}

\section{填空题}
\begin{problem}
用平面 \(\alpha\) 截半径为 \(R\) 的球，如果球心到平面 \(\alpha\) 的距离为 \(\frac{R}{2}\)，那么截得小圆的面积与球的表面积的比值为\fillinblank{}.
\end{problem}

\begin{answer}\(\frac{3}{16}\)\end{answer}

\begin{solution}
设截面小圆半径为 \(r\).球心到截面的距离为 \(\frac R2\)，所以在球心、截面圆心和小圆上一点组成的直角三角形中，\(r^2=R^2-\left(\frac R2\right)^2=\frac{3R^2}{4}\).小圆面积为 \(\pi r^2=\frac{3\pi R^2}{4}\)，球的表面积为 \(4\pi R^2\)，故比值为 \(\frac{3}{16}\).
\end{solution}

\begin{problem}
函数 \(y = \sin x + \sqrt{3}\cos x\) 在区间 \(\left[0, \frac{\pi}{2}\right]\) 上的最小值为\fillinblank{}.
\end{problem}

\begin{answer}\(1\)\end{answer}

\begin{solution}
利用辅助角公式，\(\sin x+\sqrt{3}\cos x=2\sin\left(x+\frac\pi3\right)\).当 \(x\in\left[0,\frac\pi2\right]\) 时，\(x+\frac\pi3\in\left[\frac\pi3,\frac{5\pi}{6}\right]\).正弦函数在该区间先增后减，端点值分别为 \(\frac{\sqrt3}{2}\) 和 \(\frac12\)，故最小值为 \(2\cdot\frac12=1\).
\end{solution}

\begin{problem}
已知函数 \(y = f(x)\) 是奇函数，当 \(x \geqslant 0\) 时，\(f(x) = 3^x - 1\)，设 \(f(x)\) 的反函数是 \(y = g(x)\)，则 \(g(-8) =\)\fillinblank{}.
\end{problem}

\begin{answer}\(-2\)\end{answer}

\begin{solution}
由 \(f(x)\) 为奇函数且 \(f(x)=3^x-1\)（\(x\geqslant0\)），有 \(f(-2)=-f(2)=-(3^2-1)=-8\).反函数满足 \(g(f(x))=x\)，因此 \(g(-8)=g(f(-2))=-2\).
\end{solution}

\begin{problem}
设 \(P\) 是由曲线 \(y^{2} = 4(x - 1)\) 上的一个动点，则点 \(P\) 到点 \((0,1)\) 的距离与点 \(P\) 到 \(y\) 轴的距离之和的最小值为\fillinblank{}.
\end{problem}

\begin{answer}\(\sqrt{5}\)\end{answer}

\begin{solution}
抛物线 \(y^2=4(x-1)\) 的焦点为 \(F(2,0)\)，准线为 \(x=0\)，即 \(y\) 轴.对抛物线上任意点 \(P(x,y)\)，由抛物线定义有 \(PF\) 等于 \(P\) 到准线的距离.又因抛物线上 \(x\geqslant1\)，该距离为 \(x\)，所以所求和为 \(P(0,1)P+PF\).由三角不等式，\(P(0,1)P+PF\geqslant\sqrt{(2-0)^2+(0-1)^2}=\sqrt5\).连接 \((0,1)\) 与 \(F(2,0)\) 的线段与抛物线相交：代入 \(y=1-\frac{x}{2}\) 得 \(x^2-20x+20=0\)，其中 \(x=10-4\sqrt5\in[1,2]\) 给出交点.因此等号可以取得，最小值为 \(\sqrt5\).
\end{solution}

\section{解答题}
\begin{problem}
已知 \(\alpha\) 为锐角，且 \(\tan \alpha = \frac{1}{2}\)，求 \(\frac{\sin 2\alpha \cos \alpha - \sin \alpha}{\sin 2\alpha \cos 2\alpha}\) 的值.
\end{problem}

\begin{solution}
因为 \(\alpha\) 为锐角且 \(\tan\alpha=\frac12\)，所以 \(\sin\alpha=\frac1{\sqrt5}\)，\(\cos\alpha=\frac2{\sqrt5}\).于是 \(\sin2\alpha=2\sin\alpha\cos\alpha=\frac45\)，\(\cos2\alpha=\cos^2\alpha-\sin^2\alpha=\frac35\).因此原式
\[
\frac{\frac45\cdot\frac2{\sqrt5}-\frac1{\sqrt5}}{\frac45\cdot\frac35}
=\frac{\frac3{5\sqrt5}}{\frac{12}{25}}
=\frac{\sqrt5}{4}
\]
\end{solution}

\begin{problem}
解方程：\(4^{x} + |1 - 2^{x}| = 11\).
\end{problem}

\begin{solution}
令 \(t=2^x>0\)，则 \(4^x=t^2\)，方程化为 \(t^2+\lvert1-t\rvert=11\).当 \(t\geqslant1\) 时，\(t^2+t-1=11\)，即 \((t+4)(t-3)=0\)，结合 \(t\geqslant1\) 得 \(t=3\).当 \(0<t<1\) 时，\(t^2+1-t=11\)，即 \(t^2-t-10=0\)，其根为 \(\frac{1+\sqrt{41}}2>1\) 和 \(\frac{1-\sqrt{41}}2<0\)，均不合题意.因此 \(2^x=3\)，解得 \(x=\log_2 3\).
\end{solution}

\begin{problem}
某村计划建造一个室内面积为 \(800\mathrm{~m}^{2}\) 的矩形蔬菜温室. 在温室内，沿左、右两侧与后侧内墙各保留 \(1\mathrm{~m}\) 宽的通道，沿前侧内墙保留 \(3\mathrm{~m}\) 宽的空地. 当矩形温室的边长各为多少时？蔬菜的种植面积最大，最大种植面积是多少？
\end{problem}

\begin{solution}
设温室沿左、右两侧方向的边长为 \(x\mathrm{~m}\)，另一边长为 \(y\mathrm{~m}\)，则 \(xy=800\)，且 \(x>2\)，\(y>4\).种植区域的两边长分别为 \(x-2\) 和 \(y-4\)，种植面积为
\[
S=(x-2)(y-4)=(x-2)\left(\frac{800}{x}-4\right)=808-4x-\frac{1600}{x}
\]
由基本不等式 \(4x+\frac{1600}{x}\geqslant2\sqrt{4x\cdot\frac{1600}{x}}=160\)，等号当且仅当 \(4x=\frac{1600}{x}\)，即 \(x=20\).此时 \(y=40\)，且 \(S_{\max}=808-160=648\).所以温室边长为 \(20\mathrm{~m}\) 和 \(40\mathrm{~m}\) 时，种植面积最大，最大种植面积为 \(648\mathrm{~m}^{2}\).
\end{solution}

\begin{problem}
三棱锥 \(P - ABC\) 中，侧面 \(PAC\) 与底面 \(ABC\) 垂直，\(PA = PB = PC = 3\).

\begin{enumerate}
\item 求证：\(AB \perp BC\)；
\item 设 \(AB = BC = 2\sqrt{3}\)，求 \(AC\) 与平面 \(PBC\) 所成角的大小.
\end{enumerate}

\bitmapfigure[max width=0.42\linewidth]{img/2004/national_paper_3_science/q20_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 在平面 \(PAC\) 内取 \(AC\) 的中点 \(O\).因为 \(PA=PC\)，所以 \(PO\perp AC\).又因为平面 \(PAC\perp\) 平面 \(ABC\)，且两平面的交线为 \(AC\)，所以 \(PO\perp\) 平面 \(ABC\)，从而 \(PO\perp OA\)、\(PO\perp OB\).由 \(PA=PB=PC\) 及 \(PA=PC\)，得 \(OA=OC\)，且在两个直角三角形 \(POA\)、\(POB\) 中斜边 \(PA=PB\)，直角边 \(PO\) 相同，故 \(OA=OB\).于是 \(OA=OB=OC\)，点 \(O\) 是 \(\triangle ABC\) 的外心.由于 \(O\) 在 \(AC\) 的中点，\(AC\) 是外接圆直径，所以 \(\angle ABC=90^\circ\)，即 \(AB\perp BC\).
\item 由 \(AB=BC=2\sqrt3\) 及上问知 \(\triangle ABC\) 是等腰直角三角形，所以 \(AC=2\sqrt6\).取直角坐标系，使 \(A=(-\sqrt6,0,0)\)，\(C=(\sqrt6,0,0)\)，\(B=(0,\sqrt6,0)\).由 \(PA=PC=3\)，且 \(P\) 在过 \(AC\) 的垂直平面内，可取 \(P=(0,0,\sqrt3)\)，因为 \(PA^2=6+3=9\).平面 \(PBC\) 的一个法向量为
\[
(\overrightarrow{BC})\times(\overrightarrow{BP})
=(\sqrt6,-\sqrt6,0)\times(0,-\sqrt6,\sqrt3)
=(-3\sqrt2,-3\sqrt2,-6)
\]
直线 \(AC\) 的方向向量可取 \(\bs d=(1,0,0)\)，所以直线与平面所成角 \(\theta\) 满足
\[
\sin\theta=\frac{|\bs d\cdot(-3\sqrt2,-3\sqrt2,-6)|}{|\bs d|\sqrt{(-3\sqrt2)^2+(-3\sqrt2)^2+(-6)^2}}=\frac{3\sqrt2}{6\sqrt2}=\frac12
\]
因为 \(0\leqslant\theta\leqslant\frac\pi2\)，故 \(\theta=30^\circ\).
\end{enumerate}
\end{solution}

\begin{problem}
设椭圆 \(\frac{x^2}{m+1} + y^2 = 1\) 的两个焦点是 \(F_{1}(-c,0)\) 与 \(F_{2}(c,0)\)（\(c > 0\)），且椭圆上存在一点 \(P\)，使得直线 \(PF_{1}\) 与 \(PF_{2}\) 垂直.

\begin{enumerate}
\item 求实数 \(m\) 的取值范围；
\item 设 \(L\) 是相应于焦点 \(F_{2}\) 的准线，直线 \(PF_{2}\) 与 \(L\) 相交于点 \(Q\). 若 \(\left|\frac{QF_{2}}{PF_{2}}\right| = 2 - \sqrt{3}\)，求直线 \(PF_{2}\) 的方程.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 令 \(a^2=m+1\)，\(b^2=1\).由于焦点在 \(x\) 轴上且 \(c>0\)，有 \(m>0\) 和 \(c^2=a^2-b^2=m\).设 \(P(x,y)\)，由 \(PF_1\perp PF_2\) 得
\[
(\overrightarrow{F_1P})\cdot(\overrightarrow{F_2P})=(x+c,y)\cdot(x-c,y)=x^2+y^2-c^2=0
\]
因而 \(x^2+y^2=m\).与椭圆方程联立，消去 \(y^2\) 得
\(\frac{x^2}{m+1}+m-x^2=1\)，\(x^2=\frac{m^2-1}{m}\)，\(y^2=\frac1m\).
存在这样的 \(P\) 当且仅当 \(x^2\geqslant0\)，结合 \(m>0\) 得 \(m\geqslant1\).反之当 \(m\geqslant1\) 时，上述 \(x^2\)，\(y^2\) 均非负，取相应的 \(x\)，\(y\) 即可得到满足条件的点，所以范围为 \(m\geqslant1\).
\item 椭圆相应于 \(F_2\) 的准线为 \(x=\frac{a^2}{c}=\frac{m+1}{\sqrt m}\).记 \(s=2+\sqrt3\)，由
\[
\left|\frac{QF_2}{PF_2}\right|=\frac{\left|\frac{m+1}{\sqrt m}-\sqrt m\right|}{|x-\sqrt m|}=2-\sqrt3=\frac1s
\]
得 \(|x-\sqrt m|=\frac{s}{\sqrt m}\).又 \(x^2=m-\frac1m\).若 \(x=\sqrt m+\frac{s}{\sqrt m}\)，代入上式会得到 \(2s m+s^2=-1\)，不可能；故 \(x=\sqrt m-\frac{s}{\sqrt m}\).代入 \(x^2=m-\frac1m\)，得
\((m-s)^2=m^2-1\)，\(m=\frac{s^2+1}{2s}=2\)
其中用了 \(s+\frac1s=4\).于是 \(c=\sqrt2\)，且 \(y^2=\frac12\).直线 \(PF_2\) 的斜率的绝对值为
\[
\left|\frac{y}{x-\sqrt2}\right|=\frac{1/\sqrt2}{s/\sqrt2}=\frac1s=2-\sqrt3
\]
因此直线方程为 \(y=\pm(2-\sqrt3)(x-\sqrt2)=\pm(\sqrt3-2)(x-\sqrt2)\).
\end{enumerate}
\end{solution}

\begin{problem}
已知数列 \(\{a_{n}\}\) 的前 \(n\) 项和 \(S_{n}\) 满足 \(S_{n} = 2a_{n} + (-1)^{n}\)，\(n \geqslant 1\).

\begin{enumerate}
\item 写出数列 \(\{a_{n}\}\) 的前三项 \(a_{1}\)，\(a_{2}\)，\(a_{3}\)；
\item 求数列 \(\{a_{n}\}\) 的通项公式；
\item 证明：对任意的整数 \(m > 4\)，有 \(\frac{1}{a_4} + \frac{1}{a_5} + \cdots + \frac{1}{a_m} < \frac{7}{8}\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由 \(S_1=a_1=2a_1-1\)，得 \(a_1=1\).当 \(n\geqslant2\) 时，\(S_n-S_{n-1}=a_n\)，所以
\[
a_n=\left(2a_n+(-1)^n\right)-\left(2a_{n-1}+(-1)^{n-1}\right)
\]
即 \(a_n=2a_{n-1}-2(-1)^n\).于是 \(a_2=2a_1-2=0\)，\(a_3=2a_2+2=2\).
\item 令 \(b_n=a_n+\frac23(-1)^n\)，则由递推式
\[
b_n=2a_{n-1}-2(-1)^n+\frac23(-1)^n=2\left(a_{n-1}+\frac23(-1)^{n-1}\right)=2b_{n-1}
\]
又 \(b_1=1-\frac23=\frac13\)，所以 \(b_n=\frac{2^{n-1}}3\).因此
\(a_n=b_n-\frac23(-1)^n=\frac23\left(2^{n-2}+(-1)^{n-1}\right)\)，\(n\geqslant1\).
\item 由通项公式知，\(a_4=2\)，且当 \(n\geqslant5\) 时 \(a_n>0\).设 \(n\geqslant5\) 为奇数，令 \(A=2^{n-2}>1\)，则
\[
\frac1{a_n}+\frac1{a_{n+1}}=\frac32\left(\frac1{A+1}+\frac1{2A-1}\right)=\frac{9A}{2(2A^2+A-1)}<\frac{9}{4A}=\frac32\left(\frac1{2^{n-2}}+\frac1{2^{n-1}}\right)
\]
最后一个不等号等价于 \(4A^2<2(2A^2+A-1)\)，即 \(A>1\).当 \(m>4\) 为偶数时，将从 \(a_5\) 到 \(a_m\) 的项按相邻两项分组，得
\[
\sum_{n=4}^{m}\frac1{a_n}
<\frac12+\frac32\sum_{k=3}^{m-2}\frac1{2^k}=\frac12+\frac38\left(1-\frac1{2^{m-4}}\right)<\frac78
\]
当 \(m>4\) 为奇数时，因 \(a_{m+1}>0\)，有
\[
\sum_{n=4}^{m}\frac1{a_n}<\sum_{n=4}^{m+1}\frac1{a_n}<\frac78
\]
故题设不等式成立.
\end{enumerate}
\end{solution}
