\section{填空题}
\begin{problem}
若 \(\tan \alpha = \frac{1}{2}\)，则 \(\tan \left(\alpha + \frac{\pi}{4}\right) =\fillinblank{}\).
\end{problem}

\begin{answer}
\(3\)
\end{answer}

\begin{solution}
由正切和角公式，得
\(\tan\left(\alpha+\frac{\pi}{4}\right)=\frac{\tan\alpha+\tan\frac{\pi}{4}}{1-\tan\alpha\tan\frac{\pi}{4}}=\frac{\frac{1}{2}+1}{1-\frac{1}{2}\cdot1}=3\).
\end{solution}

\begin{problem}
设抛物线的顶点坐标为 \(\left(2,0\right)\)，准线方程为 \(x = -1\)，则它的焦点坐标为\fillinblank{}.
\end{problem}

\begin{answer}
\(\left(5,0\right)\)
\end{answer}

\begin{solution}
抛物线的顶点为 \(\left(2,0\right)\)，准线为 \(x=-1\)，所以顶点到准线的距离为 \(3\). 抛物线的焦点在顶点与准线相反的一侧，且与顶点的距离为 \(3\)，故焦点坐标为 \(\left(5,0\right)\).
\end{solution}

\begin{problem}
设集合 \(A = \{5, \log_2(a + 3)\}\)，集合 \(B = \{a, b\}\). 若 \(A \cap B = \{2\}\)，则 \(A \cup B =\fillinblank{}\).
\end{problem}

\begin{answer}
\(\{1,2,5\}\)
\end{answer}

\begin{solution}
因为 \(A\cap B=\{2\}\)，所以 \(2\in A\). 又 \(5\in A\) 且 \(5\ne2\)，故 \(2=\log_{2}(a+3)\)，从而 \(a+3=4\)，得 \(a=1\). 此时 \(A=\{2,5\}\). 为使 \(2\in B=\{1,b\}\)，必须有 \(b=2\)，因此 \(A\cup B=\{1,2,5\}\).
\end{solution}

\begin{problem}
设等比数列 \(\{a_n\}\)（\(n \in \N\)）的公比 \(q = -\frac{1}{2}\)，且 \(\displaystyle\lim_{n\to\infty}\left(a_1 + a_3 + a_5 + \cdots + a_{2n-1}\right) = \frac{8}{3}\)，则 \(a_1 =\fillinblank{}\).
\end{problem}

\begin{answer}
\(2\)
\end{answer}

\begin{solution}
奇数项构成首项为 \(a_{1}\)、公比为 \(q^{2}=\frac{1}{4}\) 的等比数列，因此
\[
\lim_{n\to\infty}\left(a_{1}+a_{3}+\cdots+a_{2n-1}\right)=\frac{a_{1}}{1-\frac{1}{4}}=\frac{4a_{1}}{3}.
\]
由题设 \(\frac{4a_{1}}{3}=\frac{8}{3}\)，解得 \(a_{1}=2\).
\end{solution}

\begin{problem}
设奇函数 \(f(x)\) 的定义域为 \([-5, 5]\). 若当 \(x \in [0, 5]\) 时，\(f(x)\) 的图象如右图，则不等式 \(f(x) < 0\) 的解是\fillinblank{}.

\bitmapfigure[max width=0.30\linewidth]{img/2004/shanghai_liberal/q5_fig1.png}
\end{problem}

\begin{answer}
\(\left(-2,0\right)\cup\left(2,5\right]\)
\end{answer}

\begin{solution}
由图可知，在 \([0,5]\) 上，函数在 \((0,2)\) 内为正，在 \(x=0\) 和 \(x=2\) 时为零，在 \((2,5]\) 内为负. 因为 \(f\) 是奇函数，当 \(x\in[-5,0]\) 时，\(f(x)<0\) 等价于 \(f(-x)>0\)，这给出 \(-2<x<0\). 合并正半轴上的解，得 \(x\in\left(-2,0\right)\cup\left(2,5\right]\).
\end{solution}

\begin{problem}
已知点 \(A\left(-1,5\right)\) 和向量 \(\bs{a} = \left(2,3\right)\)，若 \(\overrightarrow{AB} = 3\bs{a}\)，则点 \(B\) 的坐标为\fillinblank{}.
\end{problem}

\begin{answer}
\(\left(5,14\right)\)
\end{answer}

\begin{solution}
由 \(\overrightarrow{AB}=3\bs{a}\)，得 \(\overrightarrow{AB}=3\left(2,3\right)=\left(6,9\right)\). 又 \(\overrightarrow{AB}=\left(x_{B}+1,y_{B}-5\right)\)，所以 \(x_{B}+1=6\)，\(y_{B}-5=9\).
解得 \(x_{B}=5\)，\(y_{B}=14\)，故 \(B\) 的坐标为 \(\left(5,14\right)\).
\end{solution}

\begin{problem}
当 \(x\)，\(y\) 满足不等式 \(\begin{cases}
2 \leqslant x \leqslant 4,\\
y \geqslant 3,\\
x + y \leqslant 8
\end{cases}\) 时，目标函数 \(k = 3x - 2y\) 的最大值为\fillinblank{}.
\end{problem}

\begin{answer}
\(6\)
\end{answer}

\begin{solution}
由于目标函数为 \(k=3x-2y\)，在可行域内应先使 \(y\) 尽量小. 由 \(y\geqslant3\)，取 \(y=3\)；此时约束 \(x+y\leqslant8\) 给出 \(x\leqslant5\)，再结合 \(2\leqslant x\leqslant4\)，可取最大的 \(x=4\). 点 \((4,3)\) 满足全部约束，且 \(k_{\max}=3\cdot4-2\cdot3=6\).
\end{solution}

\begin{problem}
圆心在直线 \(x = 2\) 上的圆 \(C\) 与 \(y\) 轴交于两点 \(A\left(0,-4\right)\)，\(B\left(0,-2\right)\)，则圆 \(C\) 的方程为\fillinblank{}.
\end{problem}

\begin{answer}
\(\left(x-2\right)^2+\left(y+3\right)^2=5\)
\end{answer}

\begin{solution}
设圆心为 \(C\left(2,c\right)\). 因为 \(A\left(0,-4\right)\)、\(B\left(0,-2\right)\) 是圆与 \(y\) 轴的交点，圆心到两点的距离相等，故 \((c+4)^2=(c+2)^2\). 解得 \(c=-3\)，所以圆心为 \(\left(2,-3\right)\). 半径的平方为 \(CA^2=\left(0-2\right)^2+\left(-4+3\right)^2=5\).
因此圆 \(C\) 的方程为 \(\left(x-2\right)^2+\left(y+3\right)^2=5\).
\end{solution}

\begin{problem}
若在二项式 \(\left(x + 1\right)^{10}\) 的展开式中任取一项，则该项的系数为奇数的概率是\fillinblank{}.（结果用分数表示）
\end{problem}

\begin{answer}
\(\frac{4}{11}\)
\end{answer}

\begin{solution}
二项式展开式共有 \(11\) 项，任取一项时每一项被取到的概率相同. 由二项式定理，展开式为
\[
(x+1)^{10}=x^{10}+10x^{9}+45x^{8}+120x^{7}+210x^{6}+252x^{5}+210x^{4}+120x^{3}+45x^{2}+10x+1.
\]
其中系数为奇数的项是系数为 \(1\)、\(45\)、\(45\)、\(1\) 的四项，所以所求概率为 \(\frac{4}{11}\).
\end{solution}

\begin{problem}
若函数 \(f(x) = a\left|x - b\right| + 2\) 在 \([0, +\infty)\) 上为增函数，则实数 \(a\)，\(b\) 的取值范围是\fillinblank{}.
\end{problem}

\begin{answer}
\(a>0\)，\(b\leqslant 0\)
\end{answer}

\begin{solution}
若 \(b\leqslant0\)，则对所有 \(x\geqslant0\) 都有 \(x-b\geqslant0\)，于是 \(f(x)=a(x-b)+2=ax-ab+2\).
要使它在 \([0,+\infty)\) 上为增函数，必须且只需 \(a>0\).

若 \(b>0\)，则在 \([0,b]\) 上有 \(f(x)=a(b-x)+2\)，在 \([b,+\infty)\) 上有 \(f(x)=a(x-b)+2\). 前一段递增要求 \(a<0\)，后一段递增要求 \(a>0\)，不可能同时成立. 因此实数的取值范围为 \(a>0\)，\(b\leqslant0\).
\end{solution}

\begin{problem}
教材中“坐标平面上的直线”与“圆锥曲线”两章内容体现出解析几何的本质是\fillinblank{}.
\end{problem}

\begin{answer}
用代数的方法研究图形的几何性质
\end{answer}

\begin{solution}
解析几何把点、直线和曲线等几何对象用坐标或方程表示，再通过代数运算研究它们的位置关系和性质. 因此这两章内容体现出的解析几何本质是用代数的方法研究图形的几何性质.
\end{solution}

\begin{problem}
若干个能唯一确定一个数列的量称为该数列的“基本量”. 设 \(\{a_n\}\) 是公比为 \(q\) 的无穷等比数列，下列 \(\{a_n\}\) 的四组量中，一定能成为该数列“基本量”的是第\fillinblank{}组.（写出所有符合要求的组号）

\begin{circlelist}
\item \(S_{1}\) 与 \(S_{2}\)；
\item \(a_{2}\) 与 \(S_{3}\)；
\item \(a_{1}\) 与 \(a_{n}\)；
\item \(q\) 与 \(a_{n}\).
\end{circlelist}

其中 \(n\) 为大于 \(1\) 的整数，\(S_{n}\) 为 \(\{a_{n}\}\) 的前 \(n\) 项和.
\end{problem}

\begin{answer}
\(\circled{1}\)，\(\circled{4}\)
\end{answer}

\begin{solution}
对第 \(\circled{1}\) 组，由 \(S_{1}=a_{1}\)、\(S_{2}=a_{1}+a_{2}\)，可得 \(a_{1}=S_{1}\)、\(a_{2}=S_{2}-S_{1}\). 当 \(a_{1}\ne0\) 时，\(q=a_{2}/a_{1}\)，从而整个数列唯一确定；当 \(a_{1}=0\) 时，等比数列的各项均为零，数列同样唯一确定.

第 \(\circled{2}\) 组不能唯一确定数列. 例如取 \(a_{2}=1\)、\(S_{3}=4\)，则 \(a_{1}q=1\)，且 \(a_{1}+q+1=4\)，即 \(a_{1}+q=3\). 因此 \(a_{1}\)、\(q\) 可以分别为 \(\frac{3+\sqrt{5}}{2}\)、\(\frac{3-\sqrt{5}}{2}\)，也可以交换二者，得到两个不同的等比数列.

第 \(\circled{3}\) 组也不能保证唯一. 例如当 \(n=3\)、\(a_{1}=1\)、\(a_{3}=1\) 时，公比可以为 \(q=1\) 或 \(q=-1\)，所得数列分别为 \(1,1,1,\ldots\) 和 \(1,-1,1,\ldots\).

对第 \(\circled{4}\) 组，已知 \(q\) 和 \(a_{n}\)，由 \(a_{n}=a_{1}q^{n-1}\) 得
\(a_{1}=\frac{a_{n}}{q^{n-1}}\).
于是首项和公比都确定，数列唯一确定. 所以符合要求的组号为 \(\circled{1}\)、\(\circled{4}\).
\end{solution}

\section{单选题}
\begin{problem}
在下列关于直线 \(l\)，\(m\) 与平面 \(\alpha\)，\(\beta\) 的命题中，真命题是（\(\quad\)）

\choices
    {若 \(l \subset \beta\) 且 \(\alpha \perp \beta\)，则 \(l \perp \alpha\)}
    {若 \(l \perp \beta\) 且 \(\alpha \parallel \beta\)，则 \(l \perp \alpha\)}
    {若 \(l \perp \beta\) 且 \(\alpha \perp \beta\)，则 \(l \parallel \alpha\)}
    {若 \(\alpha \cap \beta = m\) 且 \(l \parallel m\)，则 \(l \parallel \alpha\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
直线 \(l\perp\beta\) 时，直线 \(l\) 的方向垂直于平面 \(\beta\) 内任意与其相交的直线. 又因为 \(\alpha\parallel\beta\)，两平面的法线方向相同，所以 \(l\perp\alpha\)，选项 B 正确.

其他选项不一定成立. 例如取 \(\beta:z=0\)、\(\alpha:y=0\)，则两平面垂直. 取直线 \(l:y=1,z=0\)，它在 \(\beta\) 内但不垂直于 \(\alpha\)，故 A 错；取 \(l:x=0,y=0\)，它垂直于 \(\beta\) 但包含在 \(\alpha\) 内，故 C 错；取交线 \(m:y=z=0\) 及直线 \(l:y=0,z=1\)，虽有 \(l\parallel m\)，但 \(l\) 包含在 \(\alpha\) 内而不是平行于 \(\alpha\)，故 D 错.
\end{solution}

\begin{problem}
三角方程 \(2\sin \left(\frac{\pi}{2} - x\right) = 1\) 的解集为（\(\quad\)）

\choices
    {\(\left\{x\mid x = 2k\pi +\frac{\pi}{3},k\in \Z\right\}\)}
    {\(\left\{x\mid x = 2k\pi +\frac{5\pi}{3},k\in \Z\right\}\)}
    {\(\left\{x\mid x = 2k\pi \pm \frac{\pi}{3},k\in \Z\right\}\)}
    {\(\left\{x\mid x = k\pi +(-1)^k,k\in \Z\right\}\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
由 \(\sin\left(\frac{\pi}{2}-x\right)=\cos x\)，原方程化为 \(2\cos x=1\)，即 \(\cos x=\frac{1}{2}\). 因此 \(x=2k\pi\pm\frac{\pi}{3}\)，\(k\in\Z\)，故解集对应选项 C.
\end{solution}

\begin{problem}
若函数 \(y = f(x)\) 的图象与函数 \(y = \lg\left(x + 1\right)\) 的图象关于直线 \(x - y = 0\) 对称，则 \(f(x) =\)（\(\quad\)）

\choices
    {\(10^{x} - 1\)}
    {\(1 - 10^{x}\)}
    {\(1 - 10^{-x}\)}
    {\(10^{-x} - 1\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
关于直线 \(x-y=0\)，即 \(y=x\) 对称等价于取反函数. 对 \(y=\lg(x+1)\)，有 \(x=\lg(y+1)\Longleftrightarrow 10^{x}=y+1\Longleftrightarrow y=10^{x}-1\).
所以 \(f(x)=10^{x}-1\)，选项 A 正确.
\end{solution}

\begin{problem}
某地 2004 年第一季度应聘和招聘人数排行榜前 5 个行业的情况列表如下：

\begin{center}
\begin{tabular}{|c|c|c|c|c|c|}
\hline
行业名称 & 计算机 & 机械 & 营销 & 物流 & 贸易 \\
\hline
应聘人数 & 215830 & 200250 & 154676 & 74570 & 65280 \\
\hline
\end{tabular}

\medskip

\begin{tabular}{|c|c|c|c|c|c|}
\hline
行业名称 & 计算机 & 营销 & 机械 & 建筑 & 化工 \\
\hline
招聘人数 & 124620 & 102935 & 89115 & 76516 & 70436 \\
\hline
\end{tabular}
\end{center}

若用同一行业中应聘人数与招聘人数比值的大小来衡量该行业的就业情况，则根据表中数据，就业形势一定是（\(\quad\)）

\choices
    {计算机行业好于化工行业}
    {建筑行业好于物流行业}
    {机械行业最紧张}
    {营销行业比贸易行业紧张}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
用应聘人数与招聘人数的比值衡量就业形势时，比值越小，表示竞争相对越小，就业形势越好. 由于表格分别列出前五名，未列入应聘人数前五名的行业，其应聘人数不超过 \(65280\)；未列入招聘人数前五名的行业，其招聘人数不超过 \(70436\).

对建筑行业和物流行业，建筑行业的比值满足 \(\frac{\text{建筑应聘人数}}{76516}\leqslant\frac{65280}{76516}\)，而物流行业的比值满足 \(\frac{74570}{\text{物流招聘人数}}\geqslant\frac{74570}{70436}\).
又因为 \(\frac{65280}{76516}<\frac{74570}{70436}\)，所以建筑行业的比值一定小于物流行业，就业形势一定好于物流行业. 其他选项均涉及未列出的行业数据，或由表中比值不能推出“最紧张”，故选 B.
\end{solution}

\section{解答题}
\begin{problem}
已知复数 \(z_{1}\) 满足 \((1 + \i)z_{1} = -1 + 5\i\)，\(z_{2} = a - 2 - \i\)，其中 \(\i\) 为虚数单位，\(a \in \R\)，若 \(\left|z_{1} - \overline{z_{2}}\right| < |z_{1}|\)，求 \(a\) 的取值范围.
\end{problem}

\begin{solution}
由 \((1+\i)z_{1}=-1+5\i\)，得
\[
z_{1}=\frac{-1+5\i}{1+\i}=\frac{\left(-1+5\i\right)\left(1-\i\right)}{\left(1+\i\right)\left(1-\i\right)}=2+3\i.
\]
又因为 \(z_{2}=a-2-\i\)，所以 \(\overline{z_{2}}=a-2+\i\)，从而
\[
z_{1}-\overline{z_{2}}=\left(2+3\i\right)-\left(a-2+\i\right)=4-a+2\i.
\]
因此 \(\left|z_{1}-\overline{z_{2}}\right|^{2}=(4-a)^{2}+4\)，而 \(|z_{1}|^{2}=2^{2}+3^{2}=13\). 原不等式等价于 \((4-a)^{2}+4<13\)，
即 \((a-4)^{2}<9\). 所以 \(1<a<7\).
\end{solution}

\begin{problem}
某单位用木料制作如图所示的框架，框架的下部是边长分别为 \(x\)，\(y\)（单位为 \(\mathrm{m}\)）的矩形. 上部是等腰直角三角形. 要求框架围成的总面积 \(8 \mathrm{~cm}^{2}\). 问 \(x\)，\(y\) 分别为多少（精确到 \(0.001 \mathrm{~m}\)）时用料最省？

\FigureLayoutDeclare{img/2004/shanghai_liberal/q18_fig1.png}{3.3cm}{8bp 5bp 36bp 28bp}
\bitmapfigure[max width=0.26\linewidth]{img/2004/shanghai_liberal/q18_fig1.png}
\end{problem}

\begin{solution}
原件在本题中同时把 \(x\)、\(y\) 的单位写成 \(\mathrm{m}\)，把总面积写成 \(8 \mathrm{~cm}^{2}\)，两者量纲不一致. 结合题目要求精确到 \(0.001 \mathrm{~m}\) 以及原题数值解的量级，以下按题设数值条件理解为总面积为 \(8 \mathrm{~m}^{2}\)，并保留题干原件的单位表述.

设上部等腰直角三角形的斜边为 \(x\)，则两条直角边均为 \(\dfrac{x}{\sqrt{2}}\)，所以框架围成的面积满足
\[
xy+\frac{1}{2}\left(\frac{x}{\sqrt{2}}\right)^{2}=xy+\frac{x^{2}}{4}=8.
\]
由于 \(x>0\)，得 \(y=\dfrac{8}{x}-\dfrac{x}{4}\). 框架用料包括矩形的四条边和三角形的两条斜边，故用料长度
\[
L=2x+2y+2\cdot\frac{x}{\sqrt{2}}=\left(2+\sqrt{2}\right)x+2y=\left(\frac{3}{2}+\sqrt{2}\right)x+\frac{16}{x}.
\]
由基本不等式，
\[
L\geqslant2\sqrt{\left(\frac{3}{2}+\sqrt{2}\right)x\cdot\frac{16}{x}},
\]
当且仅当 \(\left(\frac{3}{2}+\sqrt{2}\right)x=\dfrac{16}{x}\) 时取等号. 因此
\(x=\frac{4}{\sqrt{\frac{3}{2}+\sqrt{2}}}=8-4\sqrt{2}\)，\(y=\frac{8}{x}-\frac{x}{4}=2\sqrt{2}\).
所以当 \(x\approx2.343 \mathrm{~m}\)，\(y\approx2.828 \mathrm{~m}\) 时，用料最省.
\end{solution}

\begin{problem}
记函数 \(f(x) = \sqrt{2 - \frac{x + 3}{x + 1}}\) 的定义域为 \(A\)，\(g(x) = \lg\left[\left(x - a - 1\right)\left(2a - x\right)\right]\)（\(a < 1\)）的定义域为 \(D\).

\begin{enumerate}[itemsep=0pt,topsep=0pt]
\item 求 \(A\)；
\item 若 \(B \subseteq A\)，求实数 \(a\) 的取值范围.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由根式有意义且分母不为零，得 \(2-\frac{x+3}{x+1}=\frac{x-1}{x+1}\geqslant0\)，且 \(x\ne-1\).
由符号变化可知 \(x<-1\) 或 \(x\geqslant1\)，所以
\(A=\left(-\infty,-1\right)\cup\left[1,+\infty\right)\).

\item 题目第二问中的 \(B\) 指 \(g(x)\) 的定义域，题干前一行将该定义域记为 \(D\). 由对数真数为正，得 \((x-a-1)(2a-x)>0\).
因为 \(a<1\)，所以 \(2a<a+1\)，从而该不等式的解集为
\(D=\left(2a,a+1\right)\).
要使 \(D\subseteq A\)，开区间 \(D\) 必须完全落在 \(\left(-\infty,-1\right)\) 或 \(\left[1,+\infty\right)\) 中. 前一种情形要求 \(a+1\leqslant-1\)，即 \(a\leqslant-2\)；后一种情形要求 \(2a\geqslant1\)，即 \(a\geqslant\dfrac{1}{2}\). 结合 \(a<1\)，得实数 \(a\) 的取值范围为
\(\left(-\infty,-2\right]\cup\left[\frac{1}{2},1\right)\).
\end{enumerate}
\end{solution}

\begin{problem}
如图，直线 \(y = \frac{1}{2}x\) 与抛物线 \(y = \frac{1}{8}x^2 - 4\) 交于 \(A\)，\(B\) 两点，线段 \(AB\) 的垂直平分线与直线 \(y = -5\) 交于 \(Q\) 点.

\begin{enumerate}[itemsep=0pt,topsep=0pt]
\item 求点 \(Q\) 的坐标；
\item 当 \(P\) 为抛物线上位于线段 \(AB\) 下方（含 \(A\)，\(B\)）的动点时，求 \(\triangle OPQ\) 面积的最大值.
\end{enumerate}

\bitmapfigure[max width=0.36\linewidth]{img/2004/shanghai_liberal/q20_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 联立直线与抛物线方程，得
\[
\frac{x}{2}=\frac{x^{2}}{8}-4\Longleftrightarrow x^{2}-4x-32=0\Longleftrightarrow\left(x-8\right)\left(x+4\right)=0.
\]
所以 \(A\left(-4,-2\right)\)，\(B\left(8,4\right)\)，线段 \(AB\) 的中点为 \(M\left(2,1\right)\). 直线 \(AB\) 的斜率为 \(\dfrac{1}{2}\)，故其垂直平分线为 \(y-1=-2(x-2)\)，
即 \(y=-2x+5\). 令 \(y=-5\)，得 \(x=5\)，所以 \(Q\left(5,-5\right)\).

\item 设 \(P\left(x,\dfrac{x^{2}}{8}-4\right)\). 因为 \(P\) 位于线段 \(AB\) 下方（含端点），所以 \(-4\leqslant x\leqslant8\). 直线 \(OQ\) 的方程为 \(x+y=0\)，于是
\[
S_{\triangle OPQ}=\frac{1}{2}\left|-5x-5\left(\frac{x^{2}}{8}-4\right)\right|=\frac{5}{2}\left|\frac{(x+4)^{2}}{8}-6\right|.
\]
当 \(-4\leqslant x\leqslant8\) 时，\(\dfrac{(x+4)^{2}}{8}-6\) 的取值范围为 \([-6,12]\)，故其绝对值的最大值为 \(12\). 因此 \(\triangle OPQ\) 面积的最大值为 \(\dfrac{5}{2}\times12=30\)，此时 \(x=8\)，即 \(P=B\).
\end{enumerate}
\end{solution}

\begin{problem}
如图，\(P-ABC\) 是底面边长为 \(1\) 的正三棱锥，\(D\)，\(E\)，\(F\) 分别为棱长 \(PA\)，\(PB\)，\(PC\) 上的点，截面 \(DEF \parallel\) 底面 \(ABC\)，且棱台 \(DEF-ABC\) 与棱锥 \(P-ABC\) 的棱长和相等.（棱长和是指多面体中所有棱的长度之和）

\begin{enumerate}[itemsep=0pt,topsep=0pt]
\item 证明： \(P-ABC\) 为正四面体；
\item 若 \(PD = \frac{1}{2}PA\)，求二面角 \(D-BC-A\) 的大小.（结果用反三角函数值表示）
\item 设棱台 \(DEF-ABC\) 的体积为 \(V\)，是否存在体积为 \(V\) 且各棱长均相等的直平行六面体，使得它与棱台 \(DEF-ABC\) 有相同的棱长和？ 若存在，请具体构造出这样的一个直平行六面体，并给出证明；若不存在，请说明理由.
\end{enumerate}

\bitmapfigure[max width=0.28\linewidth]{img/2004/shanghai_liberal/q21_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设正三棱锥的侧棱长为 \(s\)，并令
\[
t=\frac{PD}{PA}=\frac{PE}{PB}=\frac{PF}{PC}.
\]
由截面平行于底面，三角形 \(DEF\) 与三角形 \(ABC\) 相似，所以 \(DE=EF=FD=t\)，且 \(AD=BE=CF=(1-t)s\). 因为截面位于棱锥内部，\(0<t<1\). 棱台的棱长和为 \(3\cdot1+3t+3(1-t)s\)，棱锥的棱长和为 \(3+3s\). 由题设二者相等，得 \(3+3t+3(1-t)s=3+3s\Longrightarrow3t(1-s)=0\).
结合 \(0<t<1\)，可得 \(s=1\). 于是正三棱锥的六条棱长均为 \(1\)，所以 \(P-ABC\) 为正四面体.

\item 取 \(BC\) 的中点 \(M\). 由于三角形 \(ABC\) 和三角形 \(PBC\) 都是边长为 \(1\) 的等边三角形，有 \(MA=MP=\frac{\sqrt{3}}{2}\). 又因为 \(PD=\dfrac{1}{2}PA\)，所以 \(D\) 是 \(PA\) 的中点. 由 \(PB=PC\)、\(AB=AC\) 以及中线定理，得 \(DB=DC\)，所以 \(DM\perp BC\). 在等腰三角形 \(AMP\) 中，\(MD\) 是底边 \(AP\) 上的中线，也是顶角 \(\angle AMP\) 的角平分线. 由余弦定理，\(AP^{2}=AM^{2}+PM^{2}-2AM\cdot PM\cos\angle AMP\)，即 \(1=\frac{3}{4}+\frac{3}{4}-\frac{3}{2}\cos\angle AMP\)，所以 \(\cos\angle AMP=\frac{1}{3}\). 因为 \(AM\perp BC\)，且 \(DM\perp BC\)，所以 \(\angle DMA\) 是二面角 \(D-BC-A\) 的平面角. 因此 \(\angle DMA=\frac{1}{2}\arccos\frac{1}{3}=\arcsin\frac{\sqrt{3}}{3}\).

\item 由相似体积比，设正四面体 \(P-ABC\) 的体积为 \(V_{0}\)，则 \(V=V_{0}(1-t^{3})\)，其中 \(V_{0}=\frac{1}{3}\cdot\frac{\sqrt{3}}{4}\cdot\sqrt{\frac{2}{3}}=\frac{\sqrt{2}}{12}\). 所以 \(0<V<\dfrac{\sqrt{2}}{12}\)，从而 \(0<8V<\dfrac{2\sqrt{2}}{3}<1\). 令 \(\theta=\arcsin(8V)\). 在空间中取三个向量 \(\bs{u}=\left(\frac{1}{2},0,0\right)\)、\(\bs{v}=\left(\frac{1}{2}\cos\theta,\frac{1}{2}\sin\theta,0\right)\)、\(\bs{w}=\left(0,0,\frac{1}{2}\right)\)，
以它们为从同一顶点出发的三条棱构造直平行六面体. 三个向量的长度均为 \(\dfrac{1}{2}\)，且 \(\bs{w}\) 垂直于底面，所以这是各棱长均为 \(\dfrac{1}{2}\) 的直平行六面体. 它的棱长和为 \(12\times\dfrac{1}{2}=6\)，而体积为
\[
\left|\bs{u}\times\bs{v}\cdot\bs{w}\right|=\frac{1}{8}\sin\theta=\frac{1}{8}\sin\left(\arcsin(8V)\right)=V.
\]
因此存在满足条件的直平行六面体，上述构造即为所求.
\end{enumerate}
\end{solution}

\begin{problem}
设 \(P_{1}\left(x_{1},y_{1}\right)\)，\(P_{2}\left(x_{2},y_{2}\right)\)，\(\cdots\)，\(P_{n}\left(x_{n},y_{n}\right)\)（\(n \geqslant 3\)，\(n \in \N\)）是二次曲线 \(C\) 上的点，且 \(a_{1} = |OP_{1}|^{2}\)，\(a_{2} = |OP_{2}|^{2}\)，\(\cdots\)，\(a_{n} = |OP_{n}|^{2}\) 构成了一个公差为 \(d\)（\(d \neq 0\)）的等差数列，其中 \(O\) 是坐标原点. 记 \(S_{n} = a_{1} + a_{2} + \cdots + a_{n}\).

\begin{enumerate}[itemsep=0pt,topsep=0pt]
\item 若 \(C\) 的方程为 \(\frac{x^2}{9} - y^2 = 1\)，\(n = 3\). 点 \(P_{1}\left(3,0\right)\) 及 \(S_{3} = 162\)，求点 \(P_{3}\) 的坐标；（只需写出一个）
\item 若 \(C\) 的方程为 \(y^{2} = 2px\)（\(p \neq 0\)）. 点 \(P_{1}\left(0,0\right)\)，对于给定的自然数 \(n\)，证明： \((x_{1} + p)^{2}\)，\((x_{2} + p)^{2}\)，\(\cdots\)，\((x_{n} + p)^{2}\) 成等差数列；
\item 若 \(C\) 的方程为 \(\frac{x^2}{a^2} + \frac{y^2}{b^2} = 1\)（\(a > b > 0\)）. 点 \(P_{1}\left(a,0\right)\)，对于给定的自然数 \(n\)，当公差 \(d\) 变化时，求 \(S_{n}\) 的最小值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 因为 \(P_{1}\left(3,0\right)\)，所以 \(a_{1}=|OP_{1}|^{2}=3^{2}=9\). 等差数列的前三项和为 \(S_{3}=\frac{3}{2}\left(a_{1}+a_{3}\right)\). 故 \(162=\dfrac{3}{2}(9+a_{3})\)，解得 \(a_{3}=99\). 又因为 \(P_{3}\) 在双曲线 \(\dfrac{x^{2}}{9}-y^{2}=1\) 上，所以 \(x_{3}^{2}-9y_{3}^{2}=9\)，\(x_{3}^{2}+y_{3}^{2}=99\). 两式相减，得 \(10y_{3}^{2}=90\)，即 \(y_{3}^{2}=9\)，从而 \(x_{3}^{2}=90\). 因此点 \(P_{3}\) 可以取为 \(\left(3\sqrt{10},3\right)\).

\item 对任意 \(1\leqslant k\leqslant n\)，由 \(P_{k}\) 在抛物线 \(y^{2}=2px\) 上，得
\[
a_{k}=|OP_{k}|^{2}=x_{k}^{2}+y_{k}^{2}=x_{k}^{2}+2px_{k}=(x_{k}+p)^{2}-p^{2}.
\]
又因为 \(P_{1}\left(0,0\right)\)，所以 \(a_{1}=0\). 由 \(a_{1},a_{2},\cdots,a_{n}\) 成等差数列，得 \(a_{k}=(k-1)d\). 于是 \((x_{k}+p)^{2}=p^{2}+a_{k}=p^{2}+(k-1)d\). 因此 \((x_{1}+p)^{2}\)，\((x_{2}+p)^{2}\)，\(\cdots\)，\((x_{n}+p)^{2}\) 是首项为 \(p^{2}\)、公差为 \(d\) 的等差数列.

\item 由椭圆方程，点 \(P_{i}\) 的坐标满足 \(\frac{x_{i}^{2}}{a^{2}}+\frac{y_{i}^{2}}{b^{2}}=1\). 因为 \(a>b>0\)，可得
\[
|OP_{i}|^{2}=x_{i}^{2}+y_{i}^{2}=a^{2}-\frac{a^{2}-b^{2}}{b^{2}}y_{i}^{2}\leqslant a^{2},
\]
同理可得 \(|OP_{i}|^{2}\geqslant b^{2}\). 所以每一项都满足 \(b^{2}\leqslant a_{i}\leqslant a^{2}\). 又因 \(P_{1}\left(a,0\right)\)，有 \(a_{1}=a^{2}\)，故等差数列的公差必须满足 \(d<0\) 且 \(a^{2}+(n-1)d=a_{n}\geqslant b^{2}\)，即 \(d\geqslant\dfrac{b^{2}-a^{2}}{n-1}\). 于是
\[
S_{n}=na^{2}+\frac{n(n-1)}{2}d\geqslant na^{2}+\frac{n(n-1)}{2}\cdot\frac{b^{2}-a^{2}}{n-1}=\frac{n}{2}\left(a^{2}+b^{2}\right).
\]
当 \(d=\dfrac{b^{2}-a^{2}}{n-1}\) 时，数列各项从 \(a^{2}\) 等差递减到 \(b^{2}\)，每一项均属于 \([b^{2},a^{2}]\). 对任意 \(r\in[b^{2},a^{2}]\)，取椭圆参数方程 \(x=a\cos t\)、\(y=b\sin t\)，则
\[
|OP|^{2}=x^{2}+y^{2}=a^{2}\cos^{2}t+b^{2}\sin^{2}t=b^{2}+(a^{2}-b^{2})\cos^{2}t,
\]
随着 \(\cos^{2}t\) 在 \([0,1]\) 内取值，该平方距离可以取得 \([b^{2},a^{2}]\) 内的每一个数. 因而上述等差数列的每一项都能由椭圆上的点实现，下界能够取得. 故 \(S_{n}\) 的最小值为
\[
\boxed{\frac{n}{2}\left(a^{2}+b^{2}\right)}.
\]
\end{enumerate}
\end{solution}
