\section{单选题}
\begin{problem}
设集合 \(U = \{0, 1, 2, 3, 4, 5\}\)，集合 \(M = \{0, 3, 5\}\)，\(N = \{1, 4, 5\}\)，则 \(M \cap (\complement_U N) =\)（\quad）

\choices
    {\(\{5\}\)}
    {\(\{0,3\}\)}
    {\(\{0, 2, 3, 5\}\)}
    {\(\{0, 1, 3, 4, 5\}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
在全集 \(U\) 中，\(N\) 的补集为 \(\complement_U N=\{0,2,3\}\)，所以 \(M\cap(\complement_U N)=\{0,3\}\)，故选 B.
\end{solution}

\begin{problem}
函数 \(y = \e^{2x}\)（\(x \in \R\)）的反函数为（\quad）

\choices
    {\(y = 2\ln x\)（\(x > 0\)）}
    {\(y = \ln(2x)\)（\(x > 0\)）}
    {\(y = \frac{1}{2}\ln x\)（\(x > 0\)）}
    {\(y = \frac{1}{2}\ln 2x\)（\(x > 0\)）}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
令 \(y=\e^{2x}\)，交换 \(x\)，\(y\) 的位置，得 \(x=\e^{2y}\). 两边取自然对数，得 \(y=\frac{1}{2}\ln x\)，且 \(x>0\)，故选 C.
\end{solution}

\begin{problem}
正三棱柱侧面的一条对角线长为 \(2\)，且与底面成 \(45^{\circ}\) 角，则此三棱柱的体积（\quad）

\choices
    {\(\frac{\sqrt{6}}{2}\)}
    {\(\sqrt{6}\)}
    {\(\frac{\sqrt{6}}{6}\)}
    {\(\frac{\sqrt{6}}{3}\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
设正三棱柱的底面边长为 \(a\)，高为 \(h\). 侧面对角线长为 \(2\)，其在底面和高方向上的分量分别为 \(a\) 和 \(h\)，又因为该对角线与底面成 \(45^{\circ}\) 角，所以 \(a=2\cos45^{\circ}=\sqrt{2}\)，\(h=2\sin45^{\circ}=\sqrt{2}\). 因此底面积为 \(\frac{\sqrt{3}}{4}a^2=\frac{\sqrt{3}}{2}\)，三棱柱的体积为 \(\frac{\sqrt{3}}{2}\cdot\sqrt{2}=\frac{\sqrt{6}}{2}\)，故选 A.
\end{solution}

\begin{problem}
函数 \( y = (x + 1)^2 (x - 1) \) 在 \( x = 1 \) 处的导数等于（\quad）

\choices
    {\(1\)}
    {\(2\)}
    {\(3\)}
    {\(4\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
令 \(f(x)=(x+1)^2(x-1)\)，则 \(f'(x)=2(x+1)(x-1)+(x+1)^2\). 代入 \(x=1\)，得 \(f'(1)=2\cdot2\cdot0+2^2=4\)，故选 D.
\end{solution}

\begin{problem}
为了得到函数 \( y = 3 \times \left(\frac{1}{3}\right)^x \) 的图象，可以把函数 \( y = \left(\frac{1}{3}\right)^x \) 的图象（\quad）

\choices
    {向左平移 \(3\) 个单位长度}
    {向右平移 \(3\) 个单位长度}
    {向左平移 \(1\) 个单位长度}
    {向右平移 \(1\) 个单位长度}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
设 \(f(x)=\left(\frac{1}{3}\right)^x\)，则 \(3f(x)=\left(\frac{1}{3}\right)^{x-1}=f(x-1)\). 函数 \(y=f(x-1)\) 的图象由 \(y=f(x)\) 的图象向右平移 \(1\) 个单位长度得到，故选 D.
\end{solution}

\begin{problem}
等差数列 \(\{a_{n}\}\) 中，\(a_{1} + a_{2} + a_{3} = -24\)，\(a_{18} + a_{19} + a_{20} = 78\)，则此数列前 \(20\) 项和等于（\quad）

\choices
    {\(160\)}
    {\(180\)}
    {\(200\)}
    {\(220\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
设等差数列的公差为 \(d\). 由 \(a_1+a_2+a_3=3a_2=-24\)，得 \(a_2=-8\). 又由 \(a_{18}+a_{19}+a_{20}=3a_{19}=78\)，得 \(a_{19}=26\). 于是 \(17d=a_{19}-a_2=34\)，所以 \(d=2\)，\(a_1=-10\)，\(a_{20}=28\). 因此前 \(20\) 项和为 \(S_{20}=\frac{20}{2}(a_1+a_{20})=10\cdot18=180\)，故选 B.
\end{solution}

\begin{problem}
已知函数 \( y = \log_{\frac{1}{4}} x \) 与 \( y = kx \) 的图象有公共点 \(A\)，且点 \(A\) 的横坐标为 \(2\)，则 \( k = \)（\quad）

\choices
    {\(-\frac{1}{4}\)}
    {\(\frac{1}{4}\)}
    {\(-\frac{1}{2}\)}
    {\(\frac{1}{2}\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
点 \(A\) 的横坐标为 \(2\)，代入 \(y=\log_{\frac{1}{4}}x\)，得 \(y=\log_{\frac{1}{4}}2=-\frac{1}{2}\). 又点 \(A\) 在 \(y=kx\) 上，所以 \(2k=-\frac{1}{2}\)，得 \(k=-\frac{1}{4}\)，故选 A.
\end{solution}

\begin{problem}
已知圆 \(C\) 的半径为 \(2\)，圆心在 \(x\) 轴的正半轴上，直线 \(3x + 4y + 4 = 0\) 与圆 \(C\) 相切，则圆 \(C\) 的方程为（\quad）

\choices
    {\(x^{2} + y^{2} - 2x - 3 = 0\)}
    {\(x^{2} + y^{2} + 4x = 0\)}
    {\(x^{2} + y^{2} + 2x - 3 = 0\)}
    {\(x^{2} + y^{2} - 4x = 0\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
设圆心为 \((t,0)\)，其中 \(t>0\). 由直线 \(3x+4y+4=0\) 与圆相切，得圆心到该直线的距离等于半径 \(2\)，即 \(\frac{3t+4}{5}=2\). 解得 \(t=2\)，所以圆 \(C\) 的方程为 \((x-2)^2+y^2=4\)，即 \(x^2+y^2-4x=0\)，故选 D.
\end{solution}

\begin{problem}
从 \(5\) 位男教师和 \(4\) 位女教师中选出 \(3\) 位教师，派到 \(3\) 个班担任班主任（每班 \(1\) 位班主任）. 要求这 \(3\) 位班主任中男、女教师都要有，则不同的选派方案共有（\quad）

\choices
    {\(210\) 种}
    {\(420\) 种}
    {\(630\) 种}
    {\(840\) 种}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
按男教师人数分类. 当选 \(2\) 位男教师、\(1\) 位女教师时，先选人再分配到三个班，共有 \(\mathrm C_5^2\mathrm C_4^1\cdot3!=240\) 种；当选 \(1\) 位男教师、\(2\) 位女教师时，共有 \(\mathrm C_5^1\mathrm C_4^2\cdot3!=180\) 种. 因此不同的选派方案共有 \(240+180=420\) 种，故选 B.
\end{solution}

\begin{problem}
函数 \(y = 2\sin \left(\frac{\pi}{3} - x\right) - \cos \left(\frac{\pi}{6} + x\right)\)（\(x \in \R\)）的最小值等于（\quad）

\choices
    {\(-3\)}
    {\(-2\)}
    {\(-1\)}
    {\(-\sqrt{5}\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
利用和角公式，得 \(2\sin\left(\frac{\pi}{3}-x\right)=\sqrt{3}\cos x-\sin x\)，\(\cos\left(\frac{\pi}{6}+x\right)=\frac{\sqrt{3}}{2}\cos x-\frac{1}{2}\sin x\). 所以原函数为 \(\frac{\sqrt{3}}{2}\cos x-\frac{1}{2}\sin x\)，其振幅为 \(\sqrt{\left(\frac{\sqrt{3}}{2}\right)^2+\left(-\frac{1}{2}\right)^2}=1\)，故最小值为 \(-1\)，选 C.
\end{solution}

\begin{problem}
已知球的表面积为 \(20\pi\)，球面上有 \(A\)，\(B\)，\(C\) 三点. 如果 \(AB = AC = 2\)，\(BC = 2\sqrt{3}\)，则球心到平面 \(ABC\) 的距离为（\quad）

\choices
    {\(1\)}
    {\(\sqrt{2}\)}
    {\(\sqrt{3}\)}
    {\(2\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
由球的表面积 \(4\pi R^2=20\pi\)，得球半径 \(R=\sqrt{5}\). 在三角形 \(ABC\) 中，从 \(A\) 向 \(BC\) 作高，得高为 \(\sqrt{2^2-(\sqrt{3})^2}=1\)，所以三角形面积为 \(\sqrt{3}\). 其外接圆半径为 \(\frac{AB\cdot AC\cdot BC}{4[\triangle ABC]}=\frac{2\cdot2\cdot2\sqrt{3}}{4\sqrt{3}}=2\). 设球心到平面 \(ABC\) 的距离为 \(d\)，则平面截球所得圆的半径为 \(2\)，从而 \(d^2+2^2=R^2=5\)，得 \(d=1\)，故选 A.
\end{solution}

\begin{problem}
\(\triangle ABC\) 中，\(a\)，\(b\)，\(c\) 分别为 \(\angle A\)，\(\angle B\)，\(\angle C\) 的对边. 如果 \(a\)，\(b\)，\(c\) 成等差数列，\(\angle B = 30^{\circ}\)，\(\triangle ABC\) 的面积为 \(\frac{3}{2}\)，那么 \(b =\)（\quad）

\choices
    {\(\frac{1 + \sqrt{3}}{2}\)}
    {\(1 + \sqrt{3}\)}
    {\(\frac{2 + \sqrt{3}}{2}\)}
    {\(2 + \sqrt{3}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
由 \(a\)，\(b\)，\(c\) 成等差数列，得 \(a+c=2b\). 又三角形面积为 \(\frac{1}{2}ac\sin B=\frac{3}{2}\)，所以 \(ac=6\). 由余弦定理，\(b^2=a^2+c^2-2ac\cos30^{\circ}=a^2+c^2-\sqrt{3}ac\). 代入 \(a+c=2b\) 和 \(ac=6\)，得 \(b^2=4b^2-12-6\sqrt{3}\)，即 \(b^2=4+2\sqrt{3}=(1+\sqrt{3})^2\). 因为 \(b>0\)，所以 \(b=1+\sqrt{3}\)，故选 B.
\end{solution}

\begin{problem}
\(\left(x - \frac{1}{\sqrt{x}}\right)^8\) 展开式中 \(x^5\) 的系数为\fillinblank{}.
\end{problem}

\begin{answer}
28
\end{answer}

\begin{solution}
展开式的通项为 \(\mathrm C_8^k x^{8-k}\left(-\frac{1}{\sqrt{x}}\right)^k=(-1)^k\mathrm C_8^k x^{8-\frac{3k}{2}}\). 令 \(8-\frac{3k}{2}=5\)，得 \(k=2\). 因此 \(x^5\) 的系数为 \((-1)^2\mathrm C_8^2=28\).
\end{solution}

\begin{problem}
已知函数 \(y = \frac{1}{2} \sin \frac{x + \pi}{A}\)（\(A > 0\)）的最小正周期为 \(3\pi\)，则 \(A =\)\fillinblank{}.
\end{problem}

\begin{answer}
\(\frac{3}{2}\)
\end{answer}

\begin{solution}
函数 \(y=\frac{1}{2}\sin\frac{x+\pi}{A}\) 的自变量系数为 \(\frac{1}{A}\)，所以最小正周期为 \(T=\frac{2\pi}{1/A}=2\pi A\). 由 \(2\pi A=3\pi\) 且 \(A>0\)，得 \(A=\frac{3}{2}\).
\end{solution}

\begin{problem}
向量 \(\bs{a}\)，\(\bs{b}\) 满足 \(\left(\bs{a} - \bs{b}\right) \cdot \left(2\bs{a} + \bs{b}\right) = -4\)，且 \(|\bs{a}| = 2\)，\(\left|\bs{b}\right| = 4\)，则 \(\bs{a}\) 与 \(\bs{b}\) 夹角的余弦值等于\fillinblank{}.
\end{problem}

\begin{answer}
\(-\frac{1}{2}\)
\end{answer}

\begin{solution}
展开已知等式，得 \((\bs{a}-\bs{b})\cdot(2\bs{a}+\bs{b})=2|\bs{a}|^2-\bs{a}\cdot\bs{b}-|\bs{b}|^2\). 代入 \(|\bs{a}|=2\)，\(|\bs{b}|=4\)，得 \(8-\bs{a}\cdot\bs{b}-16=-4\)，所以 \(\bs{a}\cdot\bs{b}=-4\). 设 \(\bs{a}\) 与 \(\bs{b}\) 的夹角为 \(\theta\)，则 \(\cos\theta=\frac{\bs{a}\cdot\bs{b}}{|\bs{a}||\bs{b}|}=\frac{-4}{2\cdot4}=-\frac{1}{2}\).
\end{solution}

\begin{problem}
设 \(x\)，\(y\) 满足约束条件：\(\begin{cases}
x + y \leqslant 1,\\
y \leqslant x,\\
y \geqslant 0,
\end{cases}\) 则 \(z = 2x + y\) 的最大值是\fillinblank{}.
\end{problem}

\begin{answer}
2
\end{answer}

\begin{solution}
由 \(y\geqslant0\) 且 \(y\leqslant x\)，得 \(x\geqslant0\). 可行域由 \(y=0\)，\(y=x\) 和 \(x+y=1\) 围成，三个顶点为 \((0,0)\)，\((1,0)\)，\((\frac{1}{2},\frac{1}{2})\). 在这些顶点处，\(z=2x+y\) 的值分别为 \(0\)，\(2\)，\(\frac{3}{2}\)，所以 \(z\) 的最大值为 \(2\).
\end{solution}

\begin{problem}
已知 \(\alpha\) 为第二象限角，且 \(\sin \alpha = \frac{\sqrt{15}}{4}\)，求 \(\frac{\sin (\alpha + \frac{\pi}{4})}{\sin 2\alpha + \cos 2\alpha + 1}\) 的值.
\end{problem}

\begin{solution}
因为 \(\alpha\) 是第二象限角，所以 \(\cos\alpha<0\)，从而 \(\cos\alpha=-\sqrt{1-\sin^2\alpha}=-\frac{1}{4}\). 于是 \(\sin2\alpha=2\sin\alpha\cos\alpha=-\frac{\sqrt{15}}{8}\)，\(\cos2\alpha=1-2\sin^2\alpha=-\frac{7}{8}\)，所以 \(\sin2\alpha+\cos2\alpha+1=-\frac{\sqrt{15}}{8}-\frac{7}{8}+1=\frac{1-\sqrt{15}}{8}\). 另一方面，\(\sin\left(\alpha+\frac{\pi}{4}\right)=\frac{\sin\alpha+\cos\alpha}{\sqrt{2}}=\frac{\sqrt{15}-1}{4\sqrt{2}}\). 所以原式为 \(\frac{\sqrt{15}-1}{4\sqrt{2}}\div\frac{1-\sqrt{15}}{8}=-\sqrt{2}\).
\end{solution}

\begin{problem}
已知数列 \(\{a_{n}\}\) 为等比数列，\(a_{2} = 6\)，\(a_{5} = 162\).

\begin{enumerate}
\item 求数列 \(\{a_{n}\}\) 的通项公式；
\item 设 \(S_{n}\) 是数列 \(\{a_{n}\}\) 的前 \(n\) 项，证明 \(\frac{S_{n} \cdot S_{n+2}}{S_{n+1}^{2}} \leqslant 1\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设等比数列的公比为 \(q\). 由 \(a_5=a_2q^3\)，得 \(162=6q^3\)，所以 \(q=3\). 于是 \(a_1=\frac{a_2}{q}=2\)，数列的通项公式为 \(a_n=2\cdot3^{n-1}\).
\item 由等比数列求和公式，得 \(S_n=2\frac{3^n-1}{3-1}=3^n-1\). 因此
\[
S_{n+1}^2-S_nS_{n+2}=(3^{n+1}-1)^2-(3^n-1)(3^{n+2}-1)=4\cdot3^n>0
\]
所以 \(S_nS_{n+2}<S_{n+1}^2\)，从而 \(\frac{S_n\cdot S_{n+2}}{S_{n+1}^2}\leqslant1\).
\end{enumerate}
\end{solution}

\begin{problem}
已知直线 \(l_{1}\) 为曲线 \(y = x^{2} + x - 2\) 在点 \((1,0)\) 处的切线，\(l_{2}\) 为该曲线的另一条切线，且 \(l_{1} \perp l_{2}\).

\begin{enumerate}
\item 求直线 \(l_{2}\) 的方程；
\item 求由直线 \(l_{1}\)，\(l_{2}\) 和 \(x\) 轴所围成的三角形的面积.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设 \(l_2\) 与曲线相切于点 \((t,t^2+t-2)\). 曲线的导数为 \(y'=2x+1\)，所以 \(l_1\) 的斜率为 \(3\)，方程为 \(l_1:y=3x-3\). 直线 \(l_2\) 的斜率为 \(2t+1\)，由 \(l_1\perp l_2\)，得 \(3(2t+1)=-1\)，解得 \(t=-\frac{2}{3}\). 曲线在 \((t,t^2+t-2)\) 处的切线方程为 \(y=(2t+1)x-t^2-2\)，代入 \(t=-\frac{2}{3}\)，得 \(l_2:y=-\frac{1}{3}x-\frac{22}{9}\)，即 \(3x+9y+22=0\).
\item \(l_1\) 与 \(x\) 轴的交点为 \((1,0)\)，\(l_2\) 与 \(x\) 轴的交点为 \((-\frac{22}{3},0)\). 联立 \(y=3x-3\) 和 \(y=-\frac{1}{3}x-\frac{22}{9}\)，得两直线交点为 \((\frac{1}{6},-\frac{5}{2})\). 因此所围三角形的底边长为 \(1+\frac{22}{3}=\frac{25}{3}\)，高为 \(\frac{5}{2}\)，面积为 \(\frac{1}{2}\cdot\frac{25}{3}\cdot\frac{5}{2}=\frac{125}{12}\).
\end{enumerate}
\end{solution}

\begin{problem}
某同学参加科普知识竞赛，需回答 \(3\) 个问题. 竞赛规则规定：答对第一、二、三问题分别得 \(100\) 分，\(100\) 分，\(200\) 分，答错得零分. 假设这名同学答对第一、二、三个问题的概率分别为 \(0.8\)，\(0.7\)，\(0.6\)，且各题答对与否相互之间没有影响.

\begin{enumerate}
\item 求这名同学得 \(300\) 分的概率；
\item 求这名同学至少得 \(300\) 分的概率.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 得 \(300\) 分有两种互斥情形：第一、三题答对而第二题答错，或第一题答错而第二、三题答对. 因此所求概率为 \(0.8\times0.3\times0.6+0.2\times0.7\times0.6=0.144+0.084=0.228\).

\end{enumerate}
\end{solution}

\begin{problem}
双曲线 \(\frac{x^2}{a^2} - \frac{y^2}{b^2} = 1\)（\(a > 1\)，\(b > 0\)）的焦点距为 \(2c\)，直线 \(l\) 过点 \((a,0)\) 和 \((0,b)\)，且点 \((1,0)\) 到直线 \(l\) 的距离与点 \((-1,0)\) 到直线 \(l\) 的距离之和 \(s \geqslant \frac{4}{5} c\). 求双曲线的离心率 \(e\) 的取值范围.
\end{problem}

\begin{solution}
直线 \(l\) 的方程为 \(\frac{x}{a}+\frac{y}{b}=1\)，即 \(bx+ay-ab=0\). 双曲线满足 \(c^2=a^2+b^2\). 因为 \(a>1\)，点 \((1,0)\) 和 \((-1,0)\) 到直线 \(l\) 的距离之和为
\[
\frac{|b-ab|}{\sqrt{a^2+b^2}}+\frac{|-b-ab|}{\sqrt{a^2+b^2}}=\frac{b(a-1)+b(a+1)}{c}=\frac{2ab}{c}
\]
由条件 \(\frac{2ab}{c}\geqslant\frac{4}{5}c\)，得 \(5ab\geqslant2c^2=2(a^2+b^2)\).
令 \(t=\frac{b}{a}>0\)，则 \(5t\geqslant2(1+t^2)\)，即 \(2t^2-5t+2\leqslant0\)，所以 \(\frac{1}{2}\leqslant t\leqslant2\).
双曲线的离心率 \(e=\frac{c}{a}=\sqrt{1+\frac{b^2}{a^2}}=\sqrt{1+t^2}\)，故 \(e\) 的取值范围为 \(\left[\frac{\sqrt{5}}{2},\sqrt{5}\right]\).
\end{solution}
