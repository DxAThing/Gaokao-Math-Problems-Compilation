\section{解答题}
\begin{problem}
等差数列 \(\{a_{n}\}\) 的前 \(n\) 项和记为 \(S_{n}\)，已知 \(a_{10} = 30\)，\(a_{20} = 50\).

\begin{enumerate}
\item 求通项 \(a_{n}\)；
\item 若 \(S_{n} = 242\)，求 \(n\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item \(a_{n} = 2n + 10\).
\item \(n = 11\).
\end{enumerate}
\end{solution}

\begin{problem}
设双曲线 \(C: \frac{x^2}{a^2} - y^2 = 1\)（\(a > 0\)）与直线 \(l: x + y = 1\) 相交于两个不同的点 \(A\)，\(B\).

\end{problem}
