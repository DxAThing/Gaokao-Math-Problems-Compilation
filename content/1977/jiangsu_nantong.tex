\begin{problem}
回答下列问题：

\begin{enumerate}[label=\arabic*.]
\item 当 \(a\) 为何值时，\(2a>a\)；当 \(a\) 为何值时，\(2a=a\)；当 \(a\) 为何值时，\(2a<a\).
\item 当 \(x\) 为何值时，\(|x|=x\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}[label=\arabic*.]
\item 当 \(a>0\) 时，\(2a>a\).

当 \(a=0\) 时，\(2a=a\).

当 \(a<0\) 时，\(2a<a\).
\item 当 \(x\ge0\) 时，\(|x|=x\).
\end{enumerate}
\end{solution}

\begin{problem}
写出下列函数的自变量 \(x\) 的取值范围（在实数范围内）：

\begin{enumerate}[label=\arabic*.]
\item \( y=\frac{1}{2-x} \).
\item \( y=\sqrt{2x-1} \).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}[label=\arabic*.]
\item 当 \(2-x\neq0\)，即 \(x\neq2\) 时分母有意义，所以函数 \(y=\frac{1}{2-x}\) 的定义域为 \(x\neq2\).
\item 当 \(2x-1\ge0\)，即 \(x\ge\frac12\) 时，根式 \(\sqrt{2x-1}\) 在实数范围内有意义，所以函数 \(y=\sqrt{2x-1}\) 的定义域为 \(x\ge\frac12\).
\end{enumerate}
\end{solution}

\begin{problem}
计算：\( \left( \frac{2}{3} \right)^2+\left( -5.8 \right)^0-\left( \frac{8}{27} \right)^{-\frac{2}{3}}+\left( \frac{1}{2} \right)^{-2} \).
\end{problem}

\begin{solution}
\[
\left( \frac{2}{3} \right)^2+\left( -5.8 \right)^0-\left( \frac{8}{27} \right)^{-\frac{2}{3}}+\left( \frac{1}{2} \right)^{-2}
=\frac49+1-\frac94+4
=3\frac{7}{36}.
\]
\end{solution}

\begin{problem}
\(k\) 为何值时，方程 \(x^2-2kx+4=0\) 有两个相等的实数根？并求出这方程的根.
\end{problem}

\begin{solution}
\[
\Delta=(-2k)^2-4\times4=4k^2-16.
\]

当 \(4k^2-16=0\)，即 \(k=\pm2\) 时，方程有两个相等的实数根.

将 \(k=2\) 代入原方程得
\[
x^2-4x+4=0,
\]
\[
(x-2)(x-2)=0,
\]
\[
x_1=2,\qquad x_2=2.
\]

将 \(k=-2\) 代入原方程得
\[
x^2+4x+4=0,
\]
\[
(x+2)(x+2)=0,
\]
\[
x_1=-2,\qquad x_2=-2.
\]
\end{solution}

\begin{problem}
已知 \( \sin\alpha=\frac{3}{5}\,(90^\circ<\alpha<180^\circ) \)，求 \( \cos\alpha \)、\( \tan\alpha \)、\( \cot\alpha \) 的值.
\end{problem}

\begin{solution}
\[
90^\circ<\alpha<180^\circ.
\]

\[
\cos\alpha=-\sqrt{1-\sin^2\alpha}
=-\sqrt{1-\frac{9}{25}}
=-\sqrt{\frac{16}{25}}
=-\frac45.
\]

\[
\tan\alpha=\frac{\sin\alpha}{\cos\alpha}
=\frac{3/5}{-4/5}
=-\frac34.
\]

\[
\cot\alpha=\frac{1}{\tan\alpha}
=-\frac43.
\]
\end{solution}

\begin{problem}
试写出抛物线 \( y=1+4x-2x^2 \) 的顶点坐标、对称轴方程和函数 \(y\) 的极值.
\end{problem}

\begin{solution}
\[
y=1+4x-2x^2
=-2\left(x^2-2x-\frac12\right)
=-2\left[(x-1)^2-\frac32\right]
=-2(x-1)^2+3.
\]

顶点坐标为 \((1,3)\).

对称轴方程为 \(x=1\).

因为 \(-2<0\)，所以函数 \(y\) 有极大值.

当 \(x=-\frac{b}{2a}=1\) 时，\(y_{\max}= \frac{4ac-b^2}{4a}=3\).
\end{solution}

\begin{problem}
解方程：\( \dfrac{\lg\left(2x\right)}{\lg\left(x-4\right)}=2 \).
\end{problem}

\begin{solution}
\[
\lg(2x)=2\lg(x-4)
\]
\[
\lg(2x)=\lg(x-4)^2
\]
\[
2x=(x-4)^2
\]
\[
x^2-10x+16=0
\]
\[
(x-8)(x-2)=0
\]

由 \(x-8=0\)，得 \(x=8\).

由 \(x-2=0\)，得 \(x=2\).

经检验知 \(x=8\) 是原方程的根，\(x=2\) 是增根，应舍去.
\end{solution}

\begin{problem}
求 \(200\) 以内的能被 \(6\) 整除的所有自然数的和.
\end{problem}

\begin{solution}
\[
200=33\times6+2.
\]

\[
S_{33}=6\times1+6\times2+6\times3+\cdots+6\times33
\]
\[
=\frac{33}{2}\times(6+6\times33)
\]
\[
=\frac{33}{2}\times204
=3366.
\]
\end{solution}

\begin{problem}
已知正四棱锥底面边长为 \(12\mathrm{cm}\)，高为 \(2\sqrt7\mathrm{cm}\)，求它的侧棱.
\end{problem}

\begin{solution}
如图，\(P-ABCD\)是正四棱锥，\(PO\)是高，并且 \(PO=2\sqrt7\mathrm{cm}\).

\bitmapfigure[max width=0.42\linewidth,max totalheight=4.8cm]{img/1977/jiangsu_nantong/q9_solution_fig1.png}

\[
AB=BC=12\mathrm{cm},
\qquad
AC=\sqrt{AB^2+BC^2}=\sqrt{12^2+12^2}=12\sqrt2,
\]
\[
AO=\frac{AC}{2}=6\sqrt2,
\]
\[
PA=\sqrt{PO^2+AO^2}
=\sqrt{(2\sqrt7)^2+(6\sqrt2)^2}
=\sqrt{28+72}
=\sqrt{100}
=10\mathrm{cm}.
\]

答：它的侧棱为 \(10\mathrm{cm}\).
\end{solution}

\begin{problem}
两根钢绳悬吊货物时，两条钢绳与铅垂线的夹角分别为 \( \alpha \) 和 \( \beta \)，货物的重量为 \( P \)，试用表达式分别写出钢绳所承受的拉力 \( F_1 \) 和 \( F_2 \).
\end{problem}

\begin{solution}
如图，

\bitmapfigure[max width=0.34\linewidth,max totalheight=4.8cm]{img/1977/jiangsu_nantong/q10_solution_fig1.png}

\[
\angle 1=180^\circ-\alpha,\qquad \angle 2=180^\circ-\beta,\qquad \angle 3=180^\circ-\beta,
\]
\[
\angle 4=180^\circ-\bigl[(180^\circ-\beta)+(180^\circ-\alpha)\bigr]
=180^\circ-2\times180^\circ+\alpha+\beta
=\alpha+\beta-180^\circ.
\]

据正弦定理得
\[
\frac{F_1}{\sin(180^\circ-\beta)}=\frac{-P}{\sin(\alpha+\beta-180^\circ)}.
\]
\[
F_1=\frac{-P\sin(180^\circ-\beta)}{-\sin[180^\circ-(\alpha+\beta)]}
=\frac{P\sin(180^\circ-\beta)}{\sin[180^\circ-(\alpha+\beta)]}
=\frac{P\sin\beta}{\sin(\alpha+\beta)}.
\]

同理可得
\[
F_2=\frac{P\sin\alpha}{\sin(\alpha+\beta)}.
\]
\end{solution}
