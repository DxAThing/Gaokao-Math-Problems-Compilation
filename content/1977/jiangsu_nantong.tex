\chapter{1977年江苏卷}

\begin{problem}
\begin{enumerate}[label=（\arabic*）]
    \item 计算：\( \left(2\frac{1}{4}\right)^{\frac{1}{2}}+\left(\frac{1}{10}\right)^{-2}-(3.14)^0+\left(-\frac{27}{8}\right)^{-\frac{1}{2}} \).

    \item 求函数 \( y=\sqrt{x-2}+\frac{1}{x-3}+\lg\left( 5-x \right) \) 的定义域.

    \item 解方程：\( 5^{x^2+2x}=125 \).

    \item 计算：\( -\log_3\left( \log_3\sqrt[3]{\sqrt[3]{\sqrt[3]{3}}} \right) \).

    \item 把直角坐标方程 \( \left( x-3 \right)^2+y^2=9 \) 化为极坐标方程.

    \item 计算：\( \displaystyle\lim_{n\to\infty}\frac{1+2+3+\cdots+n}{n^2} \).

    \item 分解因式：\( x^4-2x^2y-3y^2+8y-4 \).
\end{enumerate}

\end{problem}

\begin{solution}
\begin{enumerate}[label=（\arabic*）]
    \item 原式\(=99\).

    \item 由题意得
    \[
    x-2\ge 0,\quad x\neq 3,\quad 5-x>0.
    \]
    故函数的定义域为 \( [2,3)\cup(3,5) \).

    \item 由 \( 125=5^3 \)，得
    \[
    x^2+2x=3.
    \]
    \[
    x^2+2x-3=0.
    \]
    \[
    \left( x-1 \right)\left( x+3 \right)=0.
    \]
    所以 \( x=1 \) 或 \( x=-3 \).

    \item
    \[
    \log_3\sqrt[3]{\sqrt[3]{\sqrt[3]{3}}}
    =\log_3\left( 3^{\frac{1}{27}} \right)
    =\frac{1}{27}.
    \]
    所以
    \[
    -\log_3\left( \frac{1}{27} \right)=3.
    \]

    \item
    \[
    \left( x-3 \right)^2+y^2=9
    \iff x^2+y^2-6x=0.
    \]
    设 \( x=r\cos\theta \)，\( y=r\sin\theta \)，则
    \[
    r^2-6r\cos\theta=0.
    \]
    所以极坐标方程为
    \[
    r=6\cos\theta.
    \]

    \item
    \[
    \lim_{n\to\infty}\frac{1+2+3+\cdots+n}{n^2}
    =\lim_{n\to\infty}\frac{n(n+1)}{2n^2}
    =\lim_{n\to\infty}\frac{n+1}{2n}
    =\frac{1}{2}.
    \]

    \item
    \[
    x^4-2x^2y-3y^2+8y-4
    =\left( x^2+y-2 \right)\left( x^2-3y+2 \right).
    \]
\end{enumerate}
\end{solution}

\begin{problem}
过抛物线 \( y^2=4x \) 的焦点作倾斜角为 \( \frac{3}{4}\pi \) 的直线，它与抛物线相交于 \( A \)、\( B \) 两点. 求 \( A \)、\( B \) 两点间的距离.

\end{problem}

\begin{solution}
抛物线 \( y^2=4x \) 的焦点为 \( (1,0) \)，所求直线的斜率为 \( \tan\frac{3\pi}{4}=-1 \)，故直线方程为
\[
y=-x+1.
\]
代入抛物线方程，得
\[
\left( -x+1 \right)^2=4x,
\]
\[
x^2-6x+1=0.
\]
设两交点的横坐标分别为 \( x_1,x_2 \)，则
\[
x_1=3-2\sqrt2,\qquad x_2=3+2\sqrt2.
\]
于是
\[
A\left( 3-2\sqrt2,\,-2+2\sqrt2 \right),\quad B\left( 3+2\sqrt2,\,-2-2\sqrt2 \right).
\]
所以
\[
AB=\sqrt{\left( 4\sqrt2 \right)^2+\left( 4\sqrt2 \right)^2}=8.
\]
\end{solution}

\begin{problem}
在直角三角形 \( ABC \) 中，\( \angle ACB=90^\circ \)，\( CD \)、\( CE \) 分别为斜边 \( AB \) 上的高和中线，且 \( \angle BCD \) 与 \( \angle ACD \) 之比为 \( 3:1 \)，求证：\( CD=DE \).

\bitmapfigure[max width=6.2cm,max totalheight=3.2cm]{img/1977/jiangsu/q3_fig1.png}

\end{problem}

\begin{solution}
设 \( \angle ACD=\alpha \)，则 \( \angle BCD=3\alpha \).
因为 \( \angle ACB=90^\circ \)，所以
\[
\alpha+3\alpha=90^\circ,
\]
故 \( \alpha=22.5^\circ \).

设 \( CD=x \). 则
\[
AD=x\tan22.5^\circ=x(\sqrt2-1),
\]
\[
DB=x\tan67.5^\circ=x(\sqrt2+1).
\]
又 \( CE \) 是斜边 \( AB \) 上的中线，所以 \( E \) 是 \( AB \) 的中点，故
\[
AE=\frac{AB}{2}=\frac{AD+DB}{2}=\sqrt2\,x.
\]
于是
\[
DE=AE-AD=\sqrt2\,x-x(\sqrt2-1)=x=CD.
\]
所以 \( CD=DE \).
\end{solution}

\begin{problem}
在周长为 \( 300\,\mathrm{cm} \) 的圆周上，有甲、乙两球以大小不等的速度作匀速圆周运动. 甲球从 \( A \) 点出发按逆时针方向运动，乙球从 \( B \) 点出发按顺时针方向运动，两球相遇于 \( C \) 点. 相遇后，两球各自反方向作匀速圆周运动，但这时甲球速度的大小是原来的 \( 2 \) 倍，乙球速度的大小是原来的一半，以后他们第二次相遇于 \( D \) 点. 已知 \( \widehat{AmC}=40 \) 厘米，\( \widehat{BnD}=20 \) 厘米，求 \( \widehat{ACB} \) 的长度.

\end{problem}

\begin{solution}
设 \( \widehat{BC}=x \) 厘米. 由第一次相遇可得
\[
\frac{40}{v_{\text{甲}}}=\frac{x}{v_{\text{乙}}}.
\]
故
\[
v_{\text{甲}}:v_{\text{乙}}=40:x.
\]

第二次相遇时，甲球走的弧长为 \( 260-x \)，乙球走的弧长为 \( x+20 \)，于是
\[
\frac{260-x}{2v_{\text{甲}}}=\frac{x+20}{\frac12 v_{\text{乙}}}.
\]
代入 \( v_{\text{甲}}:v_{\text{乙}}=40:x \)，得
\[
x(260-x)=160(x+20).
\]
整理得
\[
\left( x-40 \right)\left( x-80 \right)=0.
\]
因 \( v_{\text{甲}}\neq v_{\text{乙}} \)，故 \( x\neq40 \)，所以 \( x=80 \).

因此
\[
\widehat{ACB}=40+80=120\text{ 厘米}.
\]
\end{solution}

\begin{problem}
\begin{enumerate}[label=（\arabic*）]
    \item 若三角形三内角成等差数列，求证：必有一内角为 \( 60^\circ \).

    \item 若三角形三内角成等差数列，而且三边又成等比数列，求证：三角形三内角都是 \( 60^\circ \).
\end{enumerate}

\end{problem}

\begin{solution}
\begin{enumerate}[label=（\arabic*）]
    \item 设三内角为 \( 60^\circ-\delta \)、\( 60^\circ \)、\( 60^\circ+\delta \)，则必有一内角为 \( 60^\circ \).

    \item 设三内角为 \( 60^\circ-\delta \)、\( 60^\circ \)、\( 60^\circ+\delta \)，其对边分别为 \( a,b,c \).
    因为三边成等比数列，所以
    \[
    b^2=ac.
    \]
    由正弦定理，得
    \[
    \sin^2 60^\circ=\sin\left( 60^\circ-\delta \right)\sin\left( 60^\circ+\delta \right).
    \]
    而
    \[
    \sin\left( 60^\circ-\delta \right)\sin\left( 60^\circ+\delta \right)
    =\sin^2 60^\circ-\sin^2\delta.
    \]
    故 \( \sin\delta=0 \)，从而 \( \delta=0 \)，即三内角都是 \( 60^\circ \).
\end{enumerate}
\end{solution}

\begin{problem}
在两条平行直线 \( AB \) 和 \( CD \) 上分别取定一点 \( M \) 和 \( N \)，在直线 \( AB \) 上取一定线段 \( ME=a \)；在线段 \( MN \) 上取一点 \( K \)，连结 \( EK \) 并延长交 \( CD \) 于 \( F \). 试问 \( K \) 取在哪里时，\( \triangle EMK \) 与 \( \triangle FNK \) 的面积之和最小？最小值是多少？

\end{problem}

\begin{solution}
设两条平行线间的距离为 \( h \). 过 \( K \) 作两条平行线的公垂线，交 \( AB \)、\( CD \) 于 \( P \)、\( Q \)，令 \( PK=x \)，则 \( KQ=h-x \).

由 \( \triangle EMK\sim \triangle FNK \)，并结合 \( \triangle MKP\sim \triangle NKQ \)，得
\[
\frac{EM}{FN}=\frac{PK}{KQ}=\frac{x}{h-x},
\]
所以
\[
FN=a\frac{h-x}{x}.
\]

于是面积和
\[
S=\frac12\cdot a\cdot x+\frac12\cdot FN\cdot (h-x)
\]
\[
=\frac12 a\left( x+\frac{(h-x)^2}{x} \right)
=\frac12 a\left( \frac{h^2}{x}+2x-2h \right).
\]
由均值不等式，
\[
\frac{h^2}{x}+2x\ge 2\sqrt2\,h,
\]
当且仅当 \( x=\frac{h}{\sqrt2} \) 时等号成立.

故当
\[
PK=\frac{1}{\sqrt2}h
\]
时，面积和最小；等价地，
\[
MK:KN=\sqrt2+1:1.
\]
最小值为
\[
ah(\sqrt2-1).
\]
\end{solution}

\begin{problem}
求极限：\( \displaystyle\lim_{x\to\infty}\sqrt{x}\left( \sqrt{x+1}-\sqrt{x} \right) \).

\end{problem}

\begin{solution}
\[
\sqrt{x}\left( \sqrt{x+1}-\sqrt{x} \right)
=\frac{\sqrt{x}}{\sqrt{x+1}+\sqrt{x}}
=\frac{1}{\sqrt{1+\frac1x}+1}.
\]
当 \( x\to\infty \) 时，上式趋于
\[
\frac12.
\]
因此所求极限为 \( \frac12 \).
\end{solution}

\begin{problem}
求不定积分：\( \displaystyle\int \frac{\mathrm{d}x}{\left( 1+\e^x \right)^2} \).
\end{problem}

\begin{solution}
令 \( u=\e^x \)，则 \( \mathrm{d}u=\e^x\mathrm{d}x=u\,\mathrm{d}x \)，于是
\[
\int \frac{\mathrm{d}x}{\left( 1+\e^x \right)^2}
=\int \frac{\mathrm{d}u}{u(1+u)^2}.
\]
分解得
\[
\frac{1}{u(1+u)^2}=\frac{1}{u}-\frac{1}{1+u}-\frac{1}{(1+u)^2}.
\]
故
\[
\int \frac{\mathrm{d}x}{\left( 1+\e^x \right)^2}
=\ln u-\ln(1+u)+\frac{1}{1+u}+C.
\]
代回 \( u=\e^x \)，得
\[
x-\ln\left( 1+\e^x \right)+\frac{1}{1+\e^x}+C.
\]
\end{solution}
