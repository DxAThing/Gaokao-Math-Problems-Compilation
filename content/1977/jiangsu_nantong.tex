\chapter{1977年江苏南通地区（试点）}

\section{解答题}

\begin{problem}
\begin{enumerate}
\item
当 \(a\) 为何值时，\(2a>a\)；当 \(a\) 为何值时，\(2a=a\)；当 \(a\) 为何值时，\(2a<a\).

\item
当 \(x\) 为何值时，\(|x|=x\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 当 \(a>0\) 时，\(2a>a\)；当 \(a=0\) 时，\(2a=a\)；当 \(a<0\) 时，\(2a<a\).

\item 当 \(x\geqslant0\) 时，\(|x|=x\).
\end{enumerate}
\end{solution}

\begin{problem}
写出下列函数的自变量 \(x\) 的取值范围（在实数范围内）：

\begin{enumerate}
\item \(y=\dfrac{1}{2-x}\).
\item \(y=\sqrt{2x-1}\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 当 \(2-x\neq0\)，即 \(x\neq2\) 时分母有意义，所以函数 \(y=\dfrac{1}{2-x}\) 的定义域为 \(x\neq2\).

\item 当 \(2x-1\geqslant0\)，即 \(x\geqslant\dfrac12\) 时，根式 \(\sqrt{2x-1}\) 在实数范围内有意义，所以函数 \(y=\sqrt{2x-1}\) 的定义域为 \(x\geqslant\dfrac12\).
\end{enumerate}
\end{solution}

\begin{problem}
计算：\(\left(\dfrac{2}{3}\right)^2+\left(-5.8\right)^0-\left(\dfrac{8}{27}\right)^{-\frac23}+\left(\dfrac12\right)^{-2}\).
\end{problem}

\begin{solution}
\(\left(\dfrac{2}{3}\right)^2+\left(-5.8\right)^0-\left(\dfrac{8}{27}\right)^{-\frac23}+\left(\dfrac12\right)^{-2}=\dfrac49+1-\dfrac94+4=3\dfrac7{36}\).
\end{solution}

\begin{problem}
\(k\) 为何值时，方程 \(x^2-2kx+4=0\) 有两个相等的实数根？并求出这方程的根.
\end{problem}

\begin{solution}
由 \(\Delta=(-2k)^2-4\times4=4k^2-16\)，当 \(4k^2-16=0\)，即 \(k=\pm2\) 时，方程有两个相等的实数根.

将 \(k=2\) 代入原方程得 \(x^2-4x+4=0\)，即 \((x-2)^2=0\)，所以 \(x_1=x_2=2\).

将 \(k=-2\) 代入原方程得 \(x^2+4x+4=0\)，即 \((x+2)^2=0\)，所以 \(x_1=x_2=-2\).
\end{solution}

\begin{problem}
已知 \(\sin\alpha=\dfrac35\)（\(90^\circ<\alpha<180^\circ\)），求 \(\cos\alpha\)、\(\tan\alpha\)、\(\cot\alpha\) 的值.
\end{problem}

\begin{solution}
因为 \(90^\circ<\alpha<180^\circ\)，所以 \(\cos\alpha=-\sqrt{1-\sin^2\alpha}=-\sqrt{1-\dfrac9{25}}=-\sqrt{\dfrac{16}{25}}=-\dfrac45\). 所以 \(\tan\alpha=\dfrac{\sin\alpha}{\cos\alpha}=\dfrac{3/5}{-4/5}=-\dfrac34\)，\(\cot\alpha=\dfrac1{\tan\alpha}=-\dfrac43\).
\end{solution}

\begin{problem}
试写出抛物线 \(y=1+4x-2x^2\) 的顶点坐标、对称轴方程和函数 \(y\) 的极值.
\end{problem}

\begin{solution}
\(y=1+4x-2x^2=-2\left(x^2-2x-\dfrac12\right)=-2\left[(x-1)^2-\dfrac32\right]=-2(x-1)^2+3\)，因此顶点坐标为 \((1,3)\)，对称轴方程为 \(x=1\). 因为 \(-2<0\)，所以函数 \(y\) 有极大值；当 \(x=-\dfrac{b}{2a}=1\) 时，\(y_{\max}=\dfrac{4ac-b^2}{4a}=3\).
\end{solution}

\begin{problem}
解方程：\(\dfrac{\lg(2x)}{\lg(x-4)}=2\).
\end{problem}

\begin{solution}
由 \(\lg(2x)=2\lg(x-4)\) 得 \(\lg(2x)=\lg(x-4)^2\)，所以 \(2x=(x-4)^2\)，即 \(x^2-10x+16=0\)，亦即 \((x-8)(x-2)=0\). 由 \(x-8=0\)，得 \(x=8\)；由 \(x-2=0\)，得 \(x=2\). 经检验知 \(x=8\) 是原方程的根，\(x=2\) 是增根，应舍去.
\end{solution}

\begin{problem}
求 \(200\) 以内的能被 \(6\) 整除的所有自然数的和.
\end{problem}

\begin{solution}
由 \(200=33\times6+2\)，得 \(S_{33}=6\times1+6\times2+6\times3+\cdots+6\times33=\dfrac{33}{2}\times(6+6\times33)=\dfrac{33}{2}\times204=3366\).
\end{solution}

\begin{problem}
已知正四棱锥底面边长为 \(12\mathrm{~cm}\)，高为 \(2\sqrt7\mathrm{~cm}\)，求它的侧棱.
\end{problem}

\begin{solution}
如图，\(P-ABCD\) 是正四棱锥，\(PO\) 是高，并且 \(PO=2\sqrt7\mathrm{~cm}\).

\solutionfigure{\bitmapfigure[max width=0.42\linewidth]{img/1977/jiangsu_nantong/q9_solution_fig1.png}}

所以 \(AB=BC=12\mathrm{~cm}\)，\(AC=\sqrt{AB^2+BC^2}=12\sqrt2\)，\(AO=\dfrac{AC}{2}=6\sqrt2\)，\(PA=\sqrt{PO^2+AO^2}=\sqrt{(2\sqrt7)^2+(6\sqrt2)^2}=\sqrt{28+72}=\sqrt{100}=10\mathrm{~cm}\).

答：它的侧棱为 \(10\mathrm{~cm}\).
\end{solution}

\begin{problem}
两根钢绳悬吊货物时，两条钢绳与铅垂线的夹角分别为 \(\alpha\) 和 \(\beta\)，货物的重量为 \(P\)，试用表达式分别写出钢绳所承受的拉力 \(F_1\) 和 \(F_2\).
\end{problem}

\begin{solution}
如图：

\solutionfigure{\bitmapfigure[max width=0.34\linewidth]{img/1977/jiangsu_nantong/q10_solution_fig1.png}}

由几何关系得 \(\angle1=180^\circ-\alpha\)，\(\angle2=180^\circ-\beta\)，\(\angle3=180^\circ-\beta\)，\(\angle4=\alpha+\beta-180^\circ\). 据正弦定理得 \(\dfrac{F_1}{\sin(180^\circ-\beta)}=\dfrac{-P}{\sin(\alpha+\beta-180^\circ)}\)，所以 \(F_1=\dfrac{P\sin\beta}{\sin(\alpha+\beta)}\). 同理可得 \(F_2=\dfrac{P\sin\alpha}{\sin(\alpha+\beta)}\).
\end{solution}
