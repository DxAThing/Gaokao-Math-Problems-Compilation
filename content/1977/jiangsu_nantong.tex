\chapter{1977年江苏南通卷}

\begin{problem}
一、回答下列问题：

1. 当 \(a\) 为何值时，\(2a>a\)；当 \(a\) 为何值时，\(2a=a\)；当 \(a\) 为何值时，\(2a<a\).

2. 当 \(x\) 为何值时，\(\left| x \right|=x\).

\end{problem}

\begin{solution}
1. 当 \(a>0\) 时，\(2a>a\)；当 \(a=0\) 时，\(2a=a\)；当 \(a<0\) 时，\(2a<a\).

2. 当 \(x\ge 0\) 时，\(\left| x \right|=x\).
\end{solution}

\begin{problem}
二、写出下列函数的自变量 \(x\) 的取值范围（在实数范围内）：

1. \(y=\frac{1}{2-x}\)

2. \(y=\sqrt{2x-1}\)

\end{problem}

\begin{solution}
1. 当 \(2-x\neq 0\)，即 \(x\neq 2\) 时，分母有意义，函数 \(y=\frac{1}{2-x}\) 的定义域为 \(x\neq 2\).

2. 当 \(2x-1\ge 0\)，即 \(x\ge \frac{1}{2}\) 时，根式 \(\sqrt{2x-1}\) 在实数范围内有意义，函数 \(y=\sqrt{2x-1}\) 的定义域为 \(x\ge \frac{1}{2}\).
\end{solution}

\begin{problem}
四、\(k\) 为何值时，方程 \(x^2-2kx+4=0\) 有两个相等的实数根？并求出这方程的根.

\end{problem}

\begin{solution}
\[
\Delta=\left( -2k \right)^2-4\times 4=4k^2-16.
\]

当 \(4k^2-16=0\)，即 \(k=\pm 2\) 时，方程有两个相等的实数根.

将 \(k=2\) 代入原方程，得
\[
x^2-4x+4=0,
\]
\[
\left( x-2 \right)\left( x-2 \right)=0.
\]
所以 \(x_1=2\)，\(x_2=2\).

将 \(k=-2\) 代入原方程，得
\[
x^2+4x+4=0,
\]
\[
\left( x+2 \right)\left( x+2 \right)=0.
\]
所以 \(x_1=-2\)，\(x_2=-2\).

当 \(k=2\) 时，这方程的相等实数根为 \(x_1=2\)，\(x_2=2\). 当 \(k=-2\) 时，这方程的相等实数根为 \(x_1=-2\)，\(x_2=-2\).
\end{solution}

\begin{problem}
五、已知 \(\sin \alpha=\frac{3}{5}\)（\(90^\circ<\alpha<180^\circ\)），求 \(\cos \alpha\)、\(\operatorname{tg} \alpha\)、\(\operatorname{ctg} \alpha\).

\end{problem}

\begin{solution}
因为 \(90^\circ<\alpha<180^\circ\)，所以
\[
\cos \alpha=-\sqrt{1-\sin^2 \alpha}=-\sqrt{1-\frac{9}{25}}=-\sqrt{\frac{16}{25}}=-\frac{4}{5}.
\]

\[
\operatorname{tg} \alpha=\frac{\sin \alpha}{\cos \alpha}=\frac{\frac{3}{5}}{-\frac{4}{5}}=-\frac{3}{4}.
\]

\[
\operatorname{ctg} \alpha=\frac{1}{\operatorname{tg} \alpha}=-\frac{4}{3}=-1\frac{1}{3}.
\]
\end{solution}

\begin{problem}
六、试写出抛物线 \(y=1+4x-2x^2\) 的顶点坐标，对称轴方程和函数 \(y\) 的极值.

\end{problem}

\begin{solution}
\begin{align*}
y&=1+4x-2x^2\\
&=-2\left( x^2-2x-\frac{1}{2} \right)\\
&=-2\left[ \left( x-1 \right)^2-\frac{3}{2} \right]\\
&=-2\left( x-1 \right)^2+3.
\end{align*}

顶点坐标为 \(\left( 1,3 \right)\). 

对称轴方程为 \(x=1\).

因为 \(a=-2<0\)，所以函数 \(y\) 有极大值. 当 \(x=-\frac{b}{2a}=1\) 时，\(y\) 的最大值为
\[
\frac{4ac-b^2}{4a}=3.
\]
\end{solution}

\begin{problem}
七、解方程 \(\frac{\lg 2x}{\lg \left( x-4 \right)}=2\).

\end{problem}

\begin{solution}
\begin{align*}
\lg 2x&=2\lg \left( x-4 \right)\\
&=\lg \left( x-4 \right)^2.
\end{align*}

所以
\[
2x=\left( x-4 \right)^2.
\]

\[
x^2-10x+16=0.
\]

\[
\left( x-8 \right)\left( x-2 \right)=0.
\]

由 \(x-8=0\)，得 \(x=8\). 由 \(x-2=0\)，得 \(x=2\). 经验算知 \(x=8\) 是原方程的根，\(x=2\) 是增根，应舍去.
\end{solution}

\begin{problem}
八、求 \(200\) 以内的能被 \(6\) 整除的所有自然数的和.

\end{problem}

\begin{solution}
因为 \(200=33\times 6+2\)，所以
\begin{align*}
S_{33}&=6\times 1+6\times 2+6\times 3+\dots+6\times 33\\
&=\frac{33}{2}\times \left( 6+6\times 33 \right)\\
&=\frac{33}{2}\times 204\\
&=3366.
\end{align*}
\end{solution}

\begin{problem}
九、已知正四棱锥底面边长为 \(12\text{cm}\)，高为 \(2\sqrt{7}\text{cm}\)，求它的侧棱.

\end{problem}

\begin{solution}
如图，\(P-ABCD\) 是正四棱锥，\(PO\) 是高，并且 \(PO=2\sqrt{7}\text{cm}\).

\bitmapfigure[max width=4.8cm,max totalheight=4.6cm]{img/1977/jiangsu_nantong/q9_solution_fig1.png}

因为 \(AB=BC=12\text{cm}\)，所以
\[
AC=\sqrt{AB^2+BC^2}=\sqrt{12^2+12^2}=12\sqrt{2}.
\]

\[
AO=\frac{AC}{2}=6\sqrt{2}.
\]

所以
\begin{align*}
PA&=\sqrt{PO^2+AO^2}\\
&=\sqrt{\left( 2\sqrt{7} \right)^2+\left( 6\sqrt{2} \right)^2}\\
&=\sqrt{28+72}\\
&=\sqrt{100}=10\text{cm}.
\end{align*}

答：它的侧棱为 \(10\text{cm}\).
\end{solution}

\begin{problem}
十、两根钢绳悬吊货物时，两条钢绳与铅垂线的夹角分别为 \(\alpha\) 和 \(\beta\)，货物的重量为 \(P\)，试用表达式分别写出钢绳所承受的拉力 \(F_1\) 和 \(F_2\).

\end{problem}

\begin{solution}
如图，

\bitmapfigure[max width=3.9cm,max totalheight=4.4cm]{img/1977/jiangsu_nantong/q10_solution_fig1.png}

由图可知，\(\angle 1=180^\circ-\alpha\)，\(\angle 2=180^\circ-\beta\)，\(\angle 3=180^\circ-\beta\).

\begin{align*}
\angle 4&=180^\circ-\left[ \left( 180^\circ-\beta \right)+\left( 180^\circ-\alpha \right) \right]\\
&=180^\circ-2\times 180^\circ+\alpha+\beta\\
&=\alpha+\beta-180^\circ.
\end{align*}

据正弦定理，得
\[
\frac{F_1}{\sin \left( 180^\circ-\beta \right)}=\frac{-P}{\sin \left( \alpha+\beta-180^\circ \right)}.
\]

\begin{align*}
F_1&=\frac{-P\sin \left( 180^\circ-\beta \right)}{-\sin \left[ 180^\circ-\left( \alpha+\beta \right) \right]}\\
&=\frac{P\sin \left( 180^\circ-\beta \right)}{\sin \left[ 180^\circ-\left( \alpha+\beta \right) \right]}\\
&=\frac{P\sin \beta}{\sin \left( \alpha+\beta \right)}.
\end{align*}

同理可得
\[
F_2=\frac{P\sin \alpha}{\sin \left( \alpha+\beta \right)}.
\]
\end{solution}
