\begin{problem}
计算：\( \left( 4.8-3.7 \right)^2-\sqrt[3]{-27}\times\left( 5-1\div 5 \right)\times\left( -\frac{5}{8} \right) \).

\end{problem}

\begin{solution}
\begin{align*}
\text{原式}
&=1.1^2-\left( -3 \right)\times\left( 5-\frac{1}{5} \right)\times\left( -\frac{5}{8} \right) \\
&=1.21-3\times\frac{24}{5}\times\frac{5}{8} \\
&=1.21-9 \\
&=-7.79.
\end{align*}
\end{solution}

\begin{problem}
求分式方程 \( \frac{1-x}{x}=x \) 在 \( 0 \) 到 \( 1 \) 范围内精确到小数三位的根.

\end{problem}

\begin{solution}
去分母，移项后得 \(x^2+x-1=0\). 所以 \(x=\frac{-1\pm\sqrt{5}}{2}\). 经验根知 \( \frac{-1\pm\sqrt{5}}{2} \) 都是原方程的根，\( 0 \) 到 \( 1 \) 范围内的根是
\[
x=\frac{-1+\sqrt{5}}{2}\approx\frac{-1+2.236}{2}=0.618.
\]
\end{solution}

\begin{problem}
已知 \( \log_{10} 2=0.3010 \)，\( \log_{10} 5=0.6990 \)，试计算 \( \left( 2\log_{10} 4 \right)\left( \log_2 5 \right)+\log_{10} \frac{5}{2} \) 的值（精确到小数三位）.

\end{problem}

\begin{solution}
\begin{align*}
\left( 2\log_{10} 4 \right)\left( \log_2 5 \right)+\log_{10} \frac{5}{2}
&=\left( 4\log_{10} 2 \right)\left( \frac{\log_{10} 5}{\log_{10} 2} \right)+\log_{10} 5-\log_{10} 2 \\
&=4\log_{10} 5+\log_{10} 5-\log_{10} 2 \\
&=5\log_{10} 5-\log_{10} 2 \\
&=5\times 0.6990-0.3010 \\
&\approx 3.194.
\end{align*}
\end{solution}

\begin{problem}
利用特殊角三角比，求 \( \sin 75^\circ \) 的值.

\end{problem}

\begin{solution}
\begin{align*}
\sin 75^\circ
&=\sin \left( 45^\circ+30^\circ \right) \\
&=\sin 45^\circ\cos 30^\circ+\cos 45^\circ\sin 30^\circ \\
&=\frac{\sqrt{2}}{2}\cdot\frac{\sqrt{3}}{2}+\frac{\sqrt{2}}{2}\cdot\frac{1}{2} \\
&=\frac{\sqrt{6}+\sqrt{2}}{4}.
\end{align*}
\end{solution}

\begin{problem}
证明恒等式：\( \left( \cos \alpha-\cos \beta \right)^2+\left( \sin \alpha-\sin \beta \right)^2=2-2\cos \left( \alpha-\beta \right) \).

\end{problem}

\begin{solution}
\begin{align*}
\left( \cos \alpha-\cos \beta \right)^2+\left( \sin \alpha-\sin \beta \right)^2
&=\cos^2 \alpha-2\cos \alpha\cos \beta+\cos^2 \beta+\sin^2 \alpha-2\sin \alpha\sin \beta+\sin^2 \beta \\
&=2-2\left( \cos \alpha\cos \beta+\sin \alpha\sin \beta \right) \\
&=2-2\cos \left( \alpha-\beta \right).
\end{align*}

所以 \( \left( \cos \alpha-\cos \beta \right)^2+\left( \sin \alpha-\sin \beta \right)^2=2-2\cos \left( \alpha-\beta \right) \).
\end{solution}

\begin{problem}
红旗工厂要围成一个周长为 \( 120 \) 米的矩形场地可以存放原料. 问矩形的长和宽应各取多少米，才能使存放场地的面积最大？并说明这最大面积的场地是什么形状？

\end{problem}

\begin{solution}
设围成的矩形场地的长为 \( x \) 米，则宽为 \( \frac{120-2x}{2} \) 米. 依题意，矩形场地的面积
\[
S=x\cdot\frac{120-2x}{2}=x\left( 60-x \right)=-\left( x-30 \right)^2+900.
\]

当 \( x=30 \) 米时，\( S \) 得极大值 \( 900 \) 平方米，这时 \( \frac{120-2x}{2}=\frac{120-2\times 30}{2}=30\).

答：矩形场地的长和宽相等，都为 \( 30 \) 米时，围成的面积最大. 最大面积是 \( 900 \) 平方米.
\end{solution}

\begin{problem}
有一正四棱台，上底边长是 \( 40 \) 厘米，下底边长是 \( 52 \) 厘米，高是 \( 8 \) 厘米. 求：

\begin{enumerate}[label=（\arabic*）]
    \item 它的下底面的面积是多少？
    \item 它的侧面积是多少？
    \item 它的容积是多少？
\end{enumerate}

\end{problem}

\begin{solution}
（1）正四棱台的下底面是边长为 \( 52 \) 厘米的正方形，因此下底面的面积为 \( S_{\text{下底}}=52^2=2704\).

（2）正四棱台的侧面是四个全等的等腰梯形，梯形的上底为 \( 40 \) 厘米，下底为 \( 52 \) 厘米. 又因正四棱台的高为 \( 8 \) 厘米，因此等腰梯形的高是 \( \sqrt{\left( \frac{52-40}{2} \right)^2+8^2}=10 \) 厘米，故侧面积为 \( S_{\text{侧}}=4\times\frac{52+40}{2}\times 10=1840\).

（3）正四棱台的上底面积为 \( 40^2 \) 平方厘米，下底面积为 \( 52^2 \) 平方厘米，高是 \( 8 \) 厘米，因此容积为 \( V=\frac{1}{3}\times 8\left( 52^2+40^2+\sqrt{52^2\times 40^2} \right)=17024\).

答：正四棱台的下底面积是 \( 2704 \) 平方厘米，侧面积是 \( 1840 \) 平方厘米，容积是 \( 17024 \) 立方厘米.
\end{solution}

\begin{problem}
已知直线 \( L_2 \) 经过点 \( \left( 7,1 \right) \) 并且与直线 \( L_1:y=\frac{1}{3}x+2 \) 垂直，求直线 \( L_2 \) 的方程.

\end{problem}

\begin{solution}
因为 \( L_2 \) 与 \( L_1 \) 垂直，所以 \( L_2 \) 的斜率为 \( -3 \)，故 \( L_2 \) 的方程可写成 \( y=-3x+b\).

其中 \( b \) 为 \( L_2 \) 的截距. 又 \( L_2 \) 过 \( \left( 7,1 \right) \) 点，因此 \( 1=-3\times 7+b\),
所以 \( b=22 \).

因此 \( L_2 \) 的方程是 \( y=-3x+22\).
\end{solution}


\begin{problem}
求证：\( 1^3+2^3+3^3+\cdots+n^3=\left[ \frac{n\left( n+1 \right)}{2} \right]^2 \).

\end{problem}

\begin{solution}
用数学归纳法证.

当 \( n=1 \) 时，左边 \( =1^3=1 \)，右边 \( =\left[ \frac{1\cdot\left( 1+1 \right)}{2} \right]^2=1 \)，等式成立.

假定 \( n=k \) 时等式成立，即 \( 1^3+2^3+3^3+\cdots+k^3=\left[ \frac{k\left( k+1 \right)}{2} \right]^2\).

那么
\begin{align*}
1^3+2^3+3^3+\cdots+k^3+\left( k+1 \right)^3
&=\left[ \frac{k\left( k+1 \right)}{2} \right]^2+\left( k+1 \right)^3 \\
&=\frac{\left( k+1 \right)^2\left( k^2+4k+4 \right)}{4} \\
&=\frac{\left( k+1 \right)^2\left( k+2 \right)^2}{4} \\
&=\left[ \frac{\left( k+1 \right)\left( k+2 \right)}{2} \right]^2.
\end{align*}

即证得 \( n=k+1 \) 时等式也成立.

故对任意自然数 \( n \)，都有 \( 1^3+2^3+3^3+\cdots+n^3=\left[ \frac{n\left( n+1 \right)}{2} \right]^2\).
\end{solution}

\begin{problem}
计算定积分的值：\( \int_0^{\frac{\pi}{4}} \cos 2x \,\mathrm{d}x \).

\end{problem}

\begin{solution}
\begin{align*}
\int_0^{\frac{\pi}{4}} \cos 2x \,\mathrm{d}x
&=\frac{1}{2}\int_0^{\frac{\pi}{4}} \cos 2x \,\mathrm{d}\left( 2x \right) \\
&=\frac{1}{2}\sin 2x \bigg|_0^{\frac{\pi}{4}} \\
&=\frac{1}{2}.
\end{align*}
\end{solution}
