\chapter{1977年江西卷}

\section{填空题}

\begin{problem}
若 \( f(x)=3x^3-5x^2+7x-5 \)，则 \( f(0)=\fillinblank{} \).
\end{problem}

\begin{answer}
\(-5\)
\end{answer}

\begin{solution}
把 \(x=0\) 代入函数式，得 \(f(0)=3\cdot0^3-5\cdot0^2+7\cdot0-5=-5\). 因此填 \(-5\).
\end{solution}

\begin{problem}
\(\arctan 1=\fillinblank{}\).
\end{problem}

\begin{answer}
\(\frac{\pi}{4}\)
\end{answer}

\begin{solution}
反正切函数的值域为 \(\left(-\frac{\pi}{2},\frac{\pi}{2}\right)\)，且 \(\tan\frac{\pi}{4}=1\). 所以 \(\arctan1=\frac{\pi}{4}\).
\end{solution}

\begin{problem}
\(\log_a a^2=\fillinblank{}\).
\end{problem}

\begin{answer}
\(2\)
\end{answer}

\begin{solution}
对数式有意义时，底数满足 \(a>0\) 且 \(a\ne1\)，真数 \(a^2>0\). 由于真数正好是底数 \(a\) 的二次幂，根据 \(\log_a\left(a^k\right)=k\)，得 \(\log_a a^2=2\).
\end{solution}

\begin{problem}
球面积是大圆面积的\fillinblank{} 倍.
\end{problem}

\begin{answer}
\(4\)
\end{answer}

\begin{solution}
设球的半径为 \(R\). 球面积为 \(4\pi R^2\)，大圆是半径为 \(R\) 的圆，其面积为 \(\pi R^2\). 因而两者之比为 \(\dfrac{4\pi R^2}{\pi R^2}=4\)，所以填 \(4\).
\end{solution}

\begin{problem}
若 \(a<0\)，则 \( \sqrt{a^2}\div a=\fillinblank{} \).
\end{problem}

\begin{answer}
\(-1\)
\end{answer}

\begin{solution}
因为平方根取非负值，\(\sqrt{a^2}=|a|\). 当 \(a<0\) 时，\(|a|=-a\)，故 \(\sqrt{a^2}\div a=\dfrac{-a}{a}=-1\).
\end{solution}

\begin{problem}
\( \frac{1}{2}+\frac{\sqrt{3}}{2}\i \) 的模数\fillinblank{}，幅角\fillinblank{}，三角函数式\fillinblank{}.
\end{problem}

\begin{answer}
\(1\)；\(\frac{\pi}{3}\)；\(\left( \cos \frac{\pi}{3}+\i\sin \frac{\pi}{3} \right)\)
\end{answer}

\begin{solution}
设复数 \(z=\frac{1}{2}+\frac{\sqrt{3}}{2}\i\). 其模数为
\( |z|=\sqrt{\left(\frac{1}{2}\right)^2+\left(\frac{\sqrt{3}}{2}\right)^2}=1\).

实部和虚部都为正，故主幅角 \(\theta\) 在第一象限，并且
\(\cos\theta=\frac{\frac{1}{2}}{|z|}=\frac{1}{2}\)，\(\sin\theta=\frac{\frac{\sqrt{3}}{2}}{|z|}=\frac{\sqrt{3}}{2}\). 因此 \(\theta=\frac{\pi}{3}\)，且三角函数式为
\(z=|z|\left(\cos\theta+\i\sin\theta\right)=\cos\frac{\pi}{3}+\i\sin\frac{\pi}{3}\).
\end{solution}

\begin{problem}
若 \(\alpha\) 为锐角，\(\sin 2\alpha=1\)，则 \(\sin \alpha=\fillinblank{}\).
\end{problem}

\begin{answer}
\(\frac{\sqrt{2}}{2}\)
\end{answer}

\begin{solution}
由 \(\alpha\) 为锐角，知 \(0<2\alpha<\pi\). 在区间 \((0,\pi)\) 内，正弦值等于 \(1\) 只有 \(2\alpha=\frac{\pi}{2}\)，所以 \(\alpha=\frac{\pi}{4}\). 因而 \(\sin\alpha=\sin\frac{\pi}{4}=\frac{\sqrt{2}}{2}\).
\end{solution}

\begin{problem}
若 \(a>b\)，\(c<0\)，则 \(ac\)\fillinblank{} \(bc\).
\end{problem}

\begin{answer}
\(<\)
\end{answer}

\begin{solution}
由 \(a>b\) 得 \(a-b>0\). 由于 \(c<0\)，相乘后有 \(c(a-b)<0\)，即 \(ac-bc<0\). 因此 \(ac<bc\)，填 \(<\).
\end{solution}

\begin{problem}
三角形的重心是三条\fillinblank{} 线的交点.
\end{problem}

\begin{answer}
中
\end{answer}

\begin{solution}
三角形的中线是顶点与其所对边中点的连线. 三角形的三条中线共点，该公共点定义为三角形的重心，所以填“中”.
\end{solution}

\begin{problem}
若 \(\mathrm C_n^3=\mathrm C_n^5\)，则 \(n=\fillinblank{}\).
\end{problem}

\begin{answer}
\(8\)
\end{answer}

\begin{solution}
组合数有意义需 \(n\geqslant5\). 用组合数公式并约去非零因子 \(\dfrac{n(n-1)(n-2)}{6}\)，得
\[
\frac{n(n-1)(n-2)}{6}=\frac{n(n-1)(n-2)(n-3)(n-4)}{120}
\Longleftrightarrow (n-3)(n-4)=20
\]
于是 \(n^2-7n-8=0\)，即 \((n-8)(n+1)=0\). 结合 \(n\geqslant5\)，得 \(n=8\).
\end{solution}

\begin{problem}
直线方程 \(y=-3x+2\)，它的斜率是\fillinblank{}.
\end{problem}

\begin{answer}
\(-3\)
\end{answer}

\begin{solution}
直线的斜截式为 \(y=kx+b\)，其中 \(k\) 是斜率. 与 \(y=-3x+2\) 对比可得 \(k=-3\)，所以填 \(-3\).
\end{solution}

\begin{problem}
直圆锥的底半径为 \(3\)，母线为 \(5\)，则高为\fillinblank{}.
\end{problem}

\begin{answer}
\(4\)
\end{answer}

\begin{solution}
直圆锥的高、底面半径和母线构成直角三角形. 设高为 \(h\)，则
\(h^2+3^2=5^2\)，所以 \(h^2=16\). 因为高度为正，得 \(h=4\).
\end{solution}

\begin{problem}
\(y=2\sin \left( 3x+\frac{\pi}{6} \right)\) 的振幅是\fillinblank{}，周期是\fillinblank{}.
\end{problem}

\begin{answer}
\(2\)；\(\frac{2\pi}{3}\)
\end{answer}

\begin{solution}
正弦函数 \(y=A\sin\left(\omega x+\varphi\right)\) 的振幅为 \(|A|\)，周期为 \(\dfrac{2\pi}{|\omega|}\). 这里 \(A=2\)，\(\omega=3\)，故振幅为 \(2\)，周期为 \(\dfrac{2\pi}{3}\).
\end{solution}

\begin{problem}
\(\frac{\sqrt{3}+\sqrt{2}}{\sqrt{3}-\sqrt{2}}\) 与 \(\frac{\sqrt{3}-\sqrt{2}}{\sqrt{3}+\sqrt{2}}\) 的等差中项是\fillinblank{}，等比中项是\fillinblank{}.
\end{problem}

\begin{answer}
\(5\)；\(\pm 1\)
\end{answer}

\begin{solution}
记
\(u=\dfrac{\sqrt{3}+\sqrt{2}}{\sqrt{3}-\sqrt{2}}\)，\(v=\dfrac{\sqrt{3}-\sqrt{2}}{\sqrt{3}+\sqrt{2}}\). 分别乘以分母的共轭式，得
\(u=\frac{\left(\sqrt{3}+\sqrt{2}\right)^2}{3-2}=5+2\sqrt{6}\)，\(v=\frac{\left(\sqrt{3}-\sqrt{2}\right)^2}{3-2}=5-2\sqrt{6}\).

所以等差中项为 \(\dfrac{u+v}{2}=5\). 又 \(uv=1\)，若等比中项为 \(w\)，则 \(w^2=uv=1\)，故 \(w=\pm1\).
\end{solution}

\begin{problem}
圆半径为 \(R\)，则内接正方形的边长为\fillinblank{}，内接正三角形的边长为\fillinblank{}.
\end{problem}

\begin{answer}
\(\sqrt{2}R\)；\(\sqrt{3}R\)
\end{answer}

\begin{solution}
内接正方形的对角线是圆的直径 \(2R\). 若其边长为 \(s\)，则 \(s\sqrt{2}=2R\)，所以 \(s=\sqrt{2}R\).

内接正三角形的中心到顶点的半径为 \(R\)，其中心把正三角形分成三个顶角为 \(120^\circ\) 的等腰三角形. 若边长为 \(t\)，则
\(t=2R\sin60^\circ=\sqrt{3}R\).
\end{solution}

\section{解答题}

\begin{problem}
若方程 \(5x^2-10x\cos \alpha+7\cos \alpha+6=0\) 的两个根相等，试求两邻边和为 \(6\)，且夹角为 \(\alpha\) 的平行四边形的最大面积.
\end{problem}

\begin{solution}
设平行四边形的一边长为 \(u\)，则邻边长为 \(6-u\)，且 \(0<u<6\). 由两边及其夹角的面积公式，面积为 \(S=u(6-u)\sin\alpha\).

方程有两个相等的根，所以判别式为零：
\[
\left(-10\cos\alpha\right)^2-4\cdot5\left(7\cos\alpha+6\right)=0
\]
整理得 \(5\cos^2\alpha-7\cos\alpha-6=0\)，即
\(\left(5\cos\alpha+3\right)\left(\cos\alpha-2\right)=0\).
由于 \(\cos\alpha\in[-1,1]\)，舍去 \(\cos\alpha=2\)，得 \(\cos\alpha=-\frac{3}{5}\).

平行四边形的夹角满足 \(0<\alpha<\pi\)，所以 \(\sin\alpha>0\)，从而
\(\sin\alpha=\sqrt{1-\cos^2\alpha}=\sqrt{1-\left(-\frac{3}{5}\right)^2}=\frac{4}{5}\).

因此
\[
S=\frac{4}{5}u(6-u)=-\frac{4}{5}u^2+\frac{24}{5}u
=-\frac{4}{5}\left(u-3\right)^2+\frac{36}{5}
\]
当 \(u=3\) 时等号成立，故最大面积为 \(\frac{36}{5}\).
\end{solution}

\begin{problem}
用 \(0\)，\(2\)，\(5\)，\(7\)，\(9\) 诸数字组成没有重复数字的四位数中：

\begin{enumerate}
\item 大于 \(5000\) 的有多少个？
\item 是 \(4\) 的倍数的有多少个？
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 四位数的首位不能为 \(0\). 要大于 \(5000\)，首位只能取 \(5\)，\(7\)，\(9\)，共有 \(3\) 种取法. 首位确定后，从其余四个数字中取三个排列，得到 \(\mathrm A_4^3\) 个数，因此总数为
\(3\mathrm A_4^3=3\times4\times3\times2=72\).

\item 四位数是否为 \(4\) 的倍数只由末两位决定. 从给定数字中选取不重复的末两位，先逐一列出全部候选并检验其是否能被 \(4\) 整除.

不重复条件下末两位的全部候选为 \(02\)，\(05\)，\(07\)，\(09\)，\(20\)，\(25\)，\(27\)，\(29\)，\(50\)，\(52\)，\(57\)，\(59\)，\(70\)，\(72\)，\(75\)，\(79\)，\(90\)，\(92\)，\(95\)，\(97\). 逐一检验末两位能否被 \(4\) 整除，保留下来的正好是 \(20\)，\(52\)，\(72\)，\(92\).

末两位固定为 \(92\)，\(72\) 或 \(52\) 时，前两位从剩余三个数字中排列，但首位不能为 \(0\)，每种均有 \(\mathrm A_3^2-\mathrm A_2^1=6-2=4\) 个. 末两位固定为 \(20\) 时，剩余的 \(5\)，\(7\)，\(9\) 均非零，前两位有 \(\mathrm A_3^2=6\) 个.

所以四位数总数为
\(3\left(\mathrm A_3^2-\mathrm A_2^1\right)+\mathrm A_3^2=3(6-2)+6=18\).
\end{enumerate}
\end{solution}

\begin{problem}
若 \(5\left| \frac{a+2b-3}{b+1} \right|+3 \left( a+3b \right)^2=0\)，求出无穷数列 \(ab\)，\(b\)，\(\frac{b}{a}\)，\(\cdots\) 的和.
\end{problem}

\begin{solution}
先注意分式有意义，故 \(b\ne-1\). 两项 \(5\left|\dfrac{a+2b-3}{b+1}\right|\) 和 \(3(a+3b)^2\) 都是非负数，而它们的和为零，所以两项都必须为零. 因此
\[
\begin{cases}
\dfrac{a+2b-3}{b+1}=0,\\
a+3b=0.
\end{cases}
\]
结合 \(b\ne-1\)，第一式等价于 \(a+2b-3=0\)，于是
\[
\begin{cases}
a+2b-3=0,\\
a+3b=0.
\end{cases}
\]
两式相减得 \(b=-3\)，再代入得 \(a=9\).

故数列的前三项为 \(ab=-27\)，\(b=-3\)，\(\dfrac{b}{a}=-\frac{1}{3}\)，相邻两项的比为 \(q=\frac{1}{9}\). 因为 \(|q|<1\)，无穷等比数列的和为
\[
S=\frac{-27}{1-\frac{1}{9}}=-\frac{243}{8}=-30\frac{3}{8}
\]
\end{solution}

\begin{problem}
三角形的三边为 \(a\)，\(b\)，\(c\)，它们的对角分别为 \(\angle A\)，\(\angle B\)，\(\angle C\)，若 \(a^2=b \left( b+c \right)\)，求证：\(\angle A=2\angle B\).
\end{problem}

\begin{solution}
在射线 \(CA\) 上取点 \(D\)，使 \(A\) 位于 \(C\)，\(D\) 之间且 \(AD=AB=c\)，并连接 \(DB\). 因为 \(C\)，\(A\)，\(D\) 共线，所以 \(CD=CA+AD=b+c\).

由 \(AD=AB\)，三角形 \(ADB\) 为等腰三角形，故其底角相等. 记 \(\angle ADB=\angle1\)，\(\angle DBA=\angle2\)，则 \(\angle1=\angle2\).

由已知条件 \(a^2=b(b+c)\) 得 \(\dfrac{b}{a}=\dfrac{a}{b+c}\). 又因为 \(AC=b\)，\(BC=a\)，\(CD=b+c\)，所以
\[
\frac{AC}{BC}=\frac{b}{a}=\frac{a}{b+c}=\frac{BC}{CD}
\]
又 \(\angle ACB=\angle BCD\)，所以由两边及其夹角对应成比例，\(\triangle ABC\sim\triangle BDC\)，对应关系为 \(A\leftrightarrow B\)，\(B\leftrightarrow D\)，\(C\leftrightarrow C\). 因此 \(\angle B=\angle BDC=\angle1\)，且
\(\angle A=\angle DBC=\angle DBA+\angle ABC=\angle2+\angle B\).

于是 \(\angle A=\angle B+\angle2=\angle B+\angle1=2\angle B\)，命题得证.
\end{solution}

\begin{problem}
已知 \(AB=AC\)，\(\angle ABC=75^\circ\)，\(BC\) 在平面 \(P\) 内，\(A\) 到平面 \(P\) 的距离 \(AD=4\text{ 厘米}\)，\(\triangle ABC\) 所在平面与平面 \(P\) 的夹角为 \(60^\circ\)，求棱锥 \(A-BCD\) 的体积.
\end{problem}

\begin{solution}
取 \(BC\) 的中点 \(E\)，连接 \(AE\)，\(DE\).

\solutionfigure{\bitmapfigure[max width=6.6cm]{img/1977/jiangxi/q6_solution_fig1.png}}

因为 \(AB=AC\)，所以等腰三角形 \(ABC\) 的中线 \(AE\) 同时是高，即 \(AE\perp BC\). 又因为 \(AD\perp P\)，且 \(BC\subset P\)，所以 \(AD\perp BC\). 两条相交直线 \(AE\)，\(AD\) 都垂直于 \(BC\)，故 \(BC\perp\) 平面 \(AED\)，从而 \(BC\perp DE\).

平面 \(ABC\) 与平面 \(P\) 的交线是 \(BC\)，而 \(AE\) 与 \(DE\) 分别位于两平面内且都垂直于 \(BC\)，所以二面角的平面角为 \(\angle AED=60^\circ\). 在直角三角形 \(AED\) 中，\(AD=4\)，于是
\(DE=AD\cot60^\circ=4\cdot\frac{\sqrt{3}}{3}=\frac{4\sqrt{3}}{3}\)，\(AE=\frac{AD}{\sin60^\circ}=\frac{4}{\sqrt{3}/2}=\frac{8\sqrt{3}}{3}\).

又因为 \(E\) 在 \(BC\) 上，三角形 \(ABE\) 在 \(E\) 处直角，且 \(\angle ABE=\angle ABC=75^\circ\). 因而
\[
BE=AE\cot75^\circ
\]
由 \(\tan75^\circ=\tan(45^\circ+30^\circ)=2+\sqrt{3}\)，得 \(\cot75^\circ=2-\sqrt{3}\). 所以
\(BE=\frac{8\sqrt{3}}{3}(2-\sqrt{3})=\frac{8(2\sqrt{3}-3)}{3}\)，\(BC=2BE=\frac{16(2\sqrt{3}-3)}{3}\).

三角形 \(BCD\) 在平面 \(P\) 内，且 \(DE\perp BC\)，故其面积为 \(\frac{1}{2}DE\cdot BC\). 棱锥的高为 \(AD=4\)，所以
\[
V_{A-BCD}=\frac{1}{3}\cdot\frac{1}{2}DE\cdot BC\cdot AD
=\frac{1}{3}\cdot\frac{1}{2}\cdot\frac{4\sqrt{3}}{3}\cdot\frac{16(2\sqrt{3}-3)}{3}\cdot4
\]
\[
V_{A-BCD}=\frac{128(2-\sqrt{3})}{9}\text{ 立方厘米}
\]
故棱锥 \(A-BCD\) 的体积为 \(\dfrac{128(2-\sqrt{3})}{9}\text{ 立方厘米}\).
\end{solution}

\section{附加题}

\begin{problem}
已知抛物线 \(y^2=x\) 上两点 \(A(1,1)\)，\(B(4,-2)\)，求直线 \(AB\) 与抛物线 \(y^2=x\) 所围成的面积.

\bitmapfigure[max width=7.2cm]{img/1977/jiangxi/q7_fig1.png}
\end{problem}

\begin{solution}
直线 \(AB\) 的斜率为 \(\dfrac{-2-1}{4-1}=-1\)，所以其方程满足 \(y-1=-(x-1)\)，即 \(x=2-y\).

抛物线方程写成 \(x=y^2\). 联立两式得 \(y^2=2-y\)，即 \((y-1)(y+2)=0\)，故交点对应的纵坐标为 \(y=-2\) 和 \(y=1\). 当 \(-2\leqslant y\leqslant1\) 时，
\(2-y-y^2=-(y-1)(y+2)\geqslant0\)，所以直线在右侧，所围面积为
\[
S=\int_{-2}^{1}\left(2-y-y^2\right)\mathrm{d}y
=\left(2y-\frac{y^2}{2}-\frac{y^3}{3}\right)\Bigg|_{-2}^{1}
\]
计算得
\[
S=\left(2-\frac{1}{2}-\frac{1}{3}\right)-\left(-4-2+\frac{8}{3}\right)=\frac{9}{2}
\]
故直线 \(AB\) 与抛物线所围成的面积为 \(\frac{9}{2}\).
\end{solution}
