\chapter{1977年江苏镇江地区初试卷}

\section{解答题}

\begin{problem}
\begin{enumerate}
\item 计算：\( \left(-1\right)^4 - 8\times\left(-1\frac{1}{2}\right)^3 + 20.1\div\left(-0.1^2\right) \).
\item \( \dfrac{1}{2\sqrt{2}}-\dfrac{2\sqrt{3}}{3-\sqrt{3}}-\dfrac{1}{2}\left(\sqrt{\dfrac{1}{2}}-\sqrt{12}\right) \).
\item \( \lg 1-\lg 0.001+\log_{2}\left(-2\right)^4+\log_{3}\sqrt{3} \).
\item 当 \( a=\sqrt{10.1} \)，\( b=8 \) 时，求式子 \( \dfrac{\left(a^{\frac{1}{2}}b^{-1}\right)^3\sqrt{a b^{\frac{2}{3}}}}{a^2b^{-4}} \) 的值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 因为 \(\left(-1\right)^4=1\)，\(-1\frac{1}{2}=-\frac{3}{2}\)，\(\left(-1\frac{1}{2}\right)^3=-\frac{27}{8}\)，且 \(-0.1^2=-0.01\)，所以原式
\[
=1-8\times\left(-\frac{27}{8}\right)+20.1\div\left(-0.01\right)=1+27-2010=-1982
\]

\item 分别有理化分母并化简根式：
\[
\frac{1}{2\sqrt{2}}=\frac{\sqrt{2}}{2\times2}=\frac{\sqrt{2}}{4}
\]
\[
\frac{2\sqrt{3}}{3-\sqrt{3}}=\frac{2\sqrt{3}\left(3+\sqrt{3}\right)}{\left(3-\sqrt{3}\right)\left(3+\sqrt{3}\right)}=\frac{6\sqrt{3}+6}{6}=\sqrt{3}+1
\]
\[
\frac{1}{2}\left(\sqrt{\frac{1}{2}}-\sqrt{12}\right)=\frac{1}{2}\left(\frac{\sqrt{2}}{2}-2\sqrt{3}\right)=\frac{\sqrt{2}}{4}-\sqrt{3}
\]
因此原式
\[
=\frac{\sqrt{2}}{4}-\left(\sqrt{3}+1\right)-\left(\frac{\sqrt{2}}{4}-\sqrt{3}\right)=-1
\]

\item 因为 \(\lg1=0\)，\(0.001=10^{-3}\)，\((-2)^4=16\)，且 \(\sqrt{3}=3^{\frac{1}{2}}\)，所以原式
\[
=0-(-3)+4+\frac{1}{2}=\frac{15}{2}
\]

\item 由 \(a>0\)，\(b>0\)，有
\[
\left(a^{\frac{1}{2}}b^{-1}\right)^3\sqrt{a b^{\frac{2}{3}}}=a^{\frac{3}{2}}b^{-3}\cdot a^{\frac{1}{2}}b^{\frac{1}{3}}=a^2b^{-\frac{8}{3}}
\]
所以原式
\[
=\frac{a^2b^{-\frac{8}{3}}}{a^2b^{-4}}=b^{\frac{4}{3}}=8^{\frac{4}{3}}=\left(2^3\right)^{\frac{4}{3}}=16
\]
\end{enumerate}
\end{solution}

\begin{problem}
\begin{enumerate}
\item 求 \( \sin75^\circ+\frac{1}{2}\cos\left(-585^\circ\right)\cdot\tan240^\circ \) 的值.
\item 证明 \( \dfrac{2}{\tan x+\cot x}=(\sin x+\cos x)^2-1 \).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由诱导公式，\(-585^\circ+720^\circ=135^\circ\)，所以 \(\cos\left(-585^\circ\right)=\cos135^\circ=-\frac{\sqrt{2}}{2}\)；又 \(240^\circ-180^\circ=60^\circ\)，所以 \(\tan240^\circ=\tan60^\circ=\sqrt{3}\). 由和角公式，\(\sin75^\circ=\sin\left(45^\circ+30^\circ\right)=\frac{\sqrt{2}}{2}\cdot\frac{\sqrt{3}}{2}+\frac{\sqrt{2}}{2}\cdot\frac{1}{2}=\frac{\sqrt{6}+\sqrt{2}}{4}\). 因此
\[
\sin75^\circ+\frac{1}{2}\cos\left(-585^\circ\right)\tan240^\circ
=\frac{\sqrt{6}+\sqrt{2}}{4}+\frac{1}{2}\left(-\frac{\sqrt{2}}{2}\right)\sqrt{3}=\frac{\sqrt{2}}{4}
\]

\item 在等式两边有意义的条件 \(\sin x\ne0\)，\(\cos x\ne0\) 下，左边
\[
\frac{2}{\tan x+\cot x}=\frac{2}{\dfrac{\sin x}{\cos x}+\dfrac{\cos x}{\sin x}}=\frac{2\sin x\cos x}{\sin^2x+\cos^2x}=2\sin x\cos x
\]
右边
\[
(\sin x+\cos x)^2-1=\sin^2x+2\sin x\cos x+\cos^2x-1=2\sin x\cos x
\]
所以 \(\dfrac{2}{\tan x+\cot x}=(\sin x+\cos x)^2-1\).
\end{enumerate}
\end{solution}

\begin{problem}
解方程 \( \dfrac{x-1}{x^2+4x+3}-\dfrac{2}{x^2-1}=0 \).
\end{problem}

\begin{solution}
由分式有意义可知 \(x^2+4x+3\ne0\)，\(x^2-1\ne0\)，即 \(x\ne-3\)，\(-1\)，\(1\). 在此条件下，原方程化为
\[
\frac{x-1}{(x+1)(x+3)}-\frac{2}{(x-1)(x+1)}=0
\]
\[
\frac{(x-1)^2-2(x+3)}{(x-1)(x+1)(x+3)}=0
\]
因此 \((x-1)^2-2(x+3)=0\)，即 \(x^2-4x-5=0\)，所以 \((x-5)(x+1)=0\)，得 \(x=5\) 或 \(x=-1\). 其中 \(x=-1\) 不满足原分式有意义的条件；代入 \(x=5\) 时，原式为
\[
\frac{5-1}{5^2+4\times5+3}-\frac{2}{5^2-1}=\frac{4}{48}-\frac{2}{24}=0
\]
所以原方程的解为 \(x=5\).
\end{solution}

\begin{problem}
求下列各函数的定义域（自变量 \(x\) 的取值范围）：

\begin{enumerate}
\item \( y=-x+3 \).
\item \( y=\dfrac{1}{x^2+2x-3} \).
\item \( y=\sqrt{-2x+10} \).
\item \( y=\lg\left(-x^2+5x-4\right) \).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 一次函数对任意实数 \(x\) 都有意义，所以定义域为 \(\R\).
\item 分母不能为零，故 \(x^2+2x-3=(x-1)(x+3)\ne0\)，所以定义域为 \(\{x\in\R\mid x\ne-3,\ x\ne1\}\).
\item 被开方数须满足 \(-2x+10\geqslant0\)，解得 \(x\leqslant5\)，所以定义域为 \((-\infty,5]\).
\item 对数的真数须满足 \(-x^2+5x-4>0\)，即 \(x^2-5x+4<0\)，所以 \((x-1)(x-4)<0\). 该二次式开口向上，乘积在两个根 \(1\) 和 \(4\) 之间为负，故 \(1<x<4\)，定义域为 \((1,4)\).
\end{enumerate}
\end{solution}

\begin{problem}
从圆外一点引圆的两条割线，试证：一条割线和它的圆外部分的乘积等于另一条割线和它圆外部分的乘积.
\end{problem}

\begin{solution}
设圆外一点为 \(P\)，两条割线分别交圆于 \(A\)，\(B\) 和 \(C\)，\(D\)，其中点的次序为 \(P\)，\(A\)，\(B\) 和 \(P\)，\(C\)，\(D\). 连接 \(AD\) 和 \(BC\). 因为 \(PA\)、\(PB\) 在同一直线上，\(PC\)、\(PD\) 在同一直线上，所以 \(\angle APD=\angle CPB\).

又因为 \(\angle BAD\) 和 \(\angle BCD\) 都是圆周角，且所对的弦都是 \(BD\)，所以 \(\angle BAD=\angle BCD\). 射线 \(AP\) 与 \(AB\) 反向，射线 \(CP\) 与 \(CD\) 反向，因此
\(\angle PAD+\angle BAD=180^\circ\)，\(\angle PCB+\angle BCD=180^\circ\)，从而 \(\angle PAD=\angle PCB\). 因此 \(\triangle PAD\sim\triangle PCB\)，从而
\[
\frac{PA}{PC}=\frac{PD}{PB}
\]
即 \(PA\cdot PB=PC\cdot PD\). 这就证明了一条割线与其圆外部分的乘积等于另一条割线与其圆外部分的乘积.
\end{solution}

\begin{problem}
为了测量底部不能直接到达的高塔 \(AB\) 的高，在地面上引一条基线 \(CD=60 \mathrm{~m}\)（\(B\)，\(C\)，\(D\) 不在一直线上），今测得 \(\angle ACB=30^\circ\)，\(\angle BCD=75^\circ\)，\(\angle BDC=45^\circ\)，求 \(AB\).

\FigureLayoutDeclare{img/1977/jiangsu_zhenjiang_region/q6_fig1.png}{8.0cm}{10bp 0bp 22bp 12bp}
\bitmapfigure[max width=6.2cm]{img/1977/jiangsu_zhenjiang_region/q6_fig1.png}
\end{problem}

\begin{solution}
在 \(\triangle BCD\) 中，\(\angle CBD=180^\circ-75^\circ-45^\circ=60^\circ\). 由正弦定理，
\[
\frac{BC}{\sin45^\circ}=\frac{CD}{\sin60^\circ}
\]
所以
\[
BC=60\times\frac{\sin45^\circ}{\sin60^\circ}=60\times\frac{\frac{\sqrt{2}}{2}}{\frac{\sqrt{3}}{2}}=20\sqrt{6}\ \mathrm{~m}
\]
高塔 \(AB\) 垂直于地面，故 \(AB\perp BC\)，\(\triangle ABC\) 是直角三角形. 由 \(\angle ACB=30^\circ\)，得
\(\tan30^\circ=\frac{AB}{BC}\)，\(AB=BC\tan30^\circ=20\sqrt{6}\times\frac{\sqrt{3}}{3}=20\sqrt{2}\ \mathrm{~m}\).
因此高塔 \(AB\) 的高为 \(20\sqrt{2}\ \mathrm{~m}\).
\end{solution}

\begin{problem}
\begin{enumerate}
\item 画出抛物线 \(y=x^2+2\) 的图象，并写出它的顶点坐标，对称轴的方程和极值.
\item 如果上述抛物线的顶点作为新的坐标原点 \(O_1\)，把它的对称轴作为新的纵轴 \(O_1Y_1\)，求这抛物线在新的直角坐标系 \(X_1O_1Y_1\) 中的方程，并写出它的准线方程和焦点坐标.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 将函数配方，得 \(y=x^2+2=(x-0)^2+2\). 所以抛物线的顶点坐标为 \((0,2)\)，对称轴方程为 \(x=0\)，开口向上，极小值为 \(2\)，没有最大值. 作图时可取 \(x=-2\)，\(-1\)，\(0\)，\(1\)，\(2\)，分别得到 \(y=6\)，\(3\)，\(2\)，\(3\)，\(6\)，描出这些点并关于直线 \(x=0\) 对称地用平滑曲线连接，即得所求图象.
\item 设新坐标系中的坐标为 \((x_1,y_1)\). 新原点 \(O_1\) 是原点 \(O\) 沿 \(y\) 轴正方向平移 \(2\) 个单位得到的，且新坐标轴与原坐标轴方向相同，因此任一点满足 \(x=x_1\)，\(y=y_1+2\). 代入原方程得
\(y_1+2=x_1^2+2\)，\(y_1=x_1^2\).
即新坐标系中的抛物线方程为 \(x_1^2=y_1\). 对于标准形式 \(x_1^2=4py_1\)，参数 \(p\) 表示顶点到焦点的距离；比较系数得 \(4p=1\)，所以 \(p=\frac{1}{4}\). 抛物线开口向上，故焦点在 \(O_1Y_1\) 正半轴上，焦点坐标为 \(\left(0,\frac{1}{4}\right)\)，准线为焦点关于顶点的对称直线，方程为 \(y_1=-\frac{1}{4}\).
\end{enumerate}
\end{solution}

\begin{problem}
某工厂四年来的产量，第一年到第三年的产量是按等差数列增长的，三年的总产量共 \(1500\text{ 吨}\)，第二年到第四年的产量是按等比数列增长的，这三年的产量共 \(1820\text{ 吨}\). 求每年的产量各是多少吨？
\end{problem}

\begin{solution}
设四年的产量依次为 \(a_1\)，\(a_2\)，\(a_3\)，\(a_4\)，第一年至第三年构成等差数列，公差为 \(d\). 由等差数列的性质，\(a_1+a_2+a_3=3a_2\)，所以
\(a_2=\frac{1500}{3}=500\)，\(a_1=500-d\)，\(a_3=500+d\).
设第二年至第四年的等比数列公比为 \(q\). 由于产量为正且按增长的等比数列变化，\(q>0\)，于是 \(a_3=500q\)，\(a_4=500q^2\). 由三年的总产量得
\[
500+500q+500q^2=1820
\]
即
\[
25q^2+25q-66=0
\]
解得
\[
q=\frac{-25\pm\sqrt{25^2+4\times25\times66}}{50}=\frac{-25\pm85}{50}
\]
因为 \(q>0\)，所以 \(q=\frac{6}{5}\). 因此 \(a_3=600\)，\(a_4=720\)，再由 \(a_3=500+d\) 得 \(d=100\)，从而 \(a_1=400\). 经检验，\(400+500+600=1500\)，\(500+600+720=1820\)，故四年的产量依次为 \(400\text{ 吨}\)，\(500\text{ 吨}\)，\(600\text{ 吨}\)，\(720\text{ 吨}\).
\end{solution}

\begin{problem}
三角形 \(ABC\) 中，\(AE\) 和 \(BD\) 相交于 \(G\)（如图），已知 \(AD:DC=3:1\)，\(BG=GD\)，求 \(\dfrac{AG}{GE}\).

\FigureLayoutDeclare{img/1977/jiangsu_zhenjiang_region/q9_fig1.png}{7.0cm}{0bp 16bp 26bp 24bp}
\bitmapfigure[float=inline,max width=6.2cm]{img/1977/jiangsu_zhenjiang_region/q9_fig1.png}
\end{problem}

\begin{solution}
设 \(\overrightarrow{AB}=\bs{b}\)，\(\overrightarrow{AC}=\bs{c}\). 由 \(AD:DC=3:1\)，得
\[
\overrightarrow{AD}=\frac{3}{4}\overrightarrow{AC}=\frac{3}{4}\bs{c}
\]
又因为 \(BG=GD\)，所以 \(G\) 是 \(BD\) 的中点，从而
\[
\overrightarrow{AG}=\frac{1}{2}\left(\overrightarrow{AB}+\overrightarrow{AD}\right)=\frac{1}{2}\bs{b}+\frac{3}{8}\bs{c}
\]
设 \(\dfrac{AG}{GE}=r\). 由于 \(A\)，\(G\)，\(E\) 共线且 \(G\) 在 \(A\)，\(E\) 之间，有
\[
\overrightarrow{AE}=\left(1+\frac{1}{r}\right)\overrightarrow{AG}=\left(1+\frac{1}{r}\right)\left(\frac{1}{2}\bs{b}+\frac{3}{8}\bs{c}\right)
\]
另一方面，点 \(E\) 在 \(BC\) 上，可设 \(0<\lambda<1\)，使得
\[
\overrightarrow{AE}=(1-\lambda)\bs{b}+\lambda\bs{c}
\]
比较 \(\bs{b}\) 和 \(\bs{c}\) 的系数并相加，得
\[
1=\left(1+\frac{1}{r}\right)\left(\frac{1}{2}+\frac{3}{8}\right)=\frac{7}{8}\left(1+\frac{1}{r}\right)
\]
所以 \(1+\frac{1}{r}=\frac{8}{7}\)，即 \(r=7\). 因此 \(\dfrac{AG}{GE}=7\).
\end{solution}
