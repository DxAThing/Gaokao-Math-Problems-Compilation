\chapter{1977年黑龙江卷}

\section{解答题}

\begin{problem}
解答下列各题.

\begin{enumerate}
    \item 解方程：\(\sqrt{3x+4}=4\).

    \item 解不等式：\(\left|x\right|<5\).

    \item 已知正三角形的外接圆半径为 \(6\sqrt{3} \mathrm{~cm}\)，求它的边长.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
    \item 因为根式有意义，所以 \(3x+4\geqslant 0\). 原方程两边均为非负数，可以平方得 \(3x+4=16\)，所以 \(x=4\). 代回原方程，得 \(\sqrt{3\times4+4}=4\)，故原方程的解为 \(x=4\).
    \item 由绝对值的意义，\(\left|x\right|<5\) 等价于 \(-5<x<5\)，故不等式的解集为 \(\left(-5,5\right)\).
    \item 设正三角形的边长为 \(a\). 正三角形的高为 \(h=\sqrt{a^2-\left(\frac{a}{2}\right)^2}=\frac{\sqrt{3}}{2}a\)，其外心在中线上，且外心到顶点的距离是中线长的 \(\frac{2}{3}\)，所以外接圆半径为 \(R=\frac{2}{3}h=\frac{a}{\sqrt{3}}\). 因而

    \[
    a=R\sqrt{3}=6\sqrt{3}\times\sqrt{3}=18.
    \]

    因而正三角形的边长为 \(18 \mathrm{~cm}\).
\end{enumerate}
\end{solution}

\begin{problem}
计算下列各题.

\begin{enumerate}
    \item \(\sqrt{m^2-2ma+a^2}\).

    \item \(\cos 78^\circ\cdot\cos 3^\circ+\cos 12^\circ\cdot\sin 3^\circ\).

    \item \(\arcsin\left(\cos\frac{\pi}{6}\right)\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
    \item \(m^2-2ma+a^2=(m-a)^2\)，而算术平方根的值非负，所以 \(\sqrt{m^2-2ma+a^2}=\sqrt{(m-a)^2}=\left|m-a\right|\).
    \item 因为 \(\cos78^\circ=\sin12^\circ\)，所以原式

    \[
    =\sin12^\circ\cos3^\circ+\cos12^\circ\sin3^\circ
    =\sin15^\circ=\sin\left(45^\circ-30^\circ\right)
    =\frac{\sqrt{2}}{2}\cdot\frac{\sqrt{3}}{2}
    -\frac{\sqrt{2}}{2}\cdot\frac{1}{2}
    =\frac{\sqrt{6}-\sqrt{2}}{4}.
    \]

    \item 因为 \(\cos\frac{\pi}{6}=\frac{\sqrt{3}}{2}=\sin\frac{\pi}{3}\)，且 \(\frac{\pi}{3}\in\left[-\frac{\pi}{2},\frac{\pi}{2}\right]\)，符合反正弦函数的值域，所以

    \[
    \arcsin\left(\cos\frac{\pi}{6}\right)=\arcsin\left(\sin\frac{\pi}{3}\right)=\frac{\pi}{3}.
    \]
\end{enumerate}
\end{solution}

\begin{problem}
解下列各题.

\begin{enumerate}
    \item 解方程：\(3^{x+1}-9^{\frac{x}{2}}=18\).

    \item 求数列 \(2\)，\(4\)，\(8\)，\(16\)，\(\cdots\) 前十项的和.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
    \item 因为 \(9^{\frac{x}{2}}=\left(3^2\right)^{\frac{x}{2}}=3^x\)，所以原方程等价于 \(3\cdot3^x-3^x=18\)，即 \(2\cdot3^x=18\). 因而 \(3^x=9=3^2\)，所以 \(x=2\). 原方程的解为 \(x=2\).
    \item 这是首项为 \(2\)、公比为 \(2\) 的等比数列，前十项和为

    \[
    S_{10}=2\frac{2^{10}-1}{2-1}=2\left(1024-1\right)=2046.
    \]

    所以前十项的和为 \(2046\).
\end{enumerate}
\end{solution}

\begin{problem}
解下列各题.

\begin{enumerate}
    \item 圆锥的高为 \(6 \mathrm{~cm}\)，母线和底面半径成 \(30^\circ\) 角，求它的侧面积.

    \item 求过点 \(\left(1,4\right)\) 且与直线 \(2x-5y+3=0\) 垂直的直线方程.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
    \item 设圆锥的母线长为 \(l\)，底面半径为 \(r\). 在圆锥的轴截面中，母线、底面半径和高组成直角三角形，且母线与底面半径的夹角为 \(30^\circ\). 因而

    \[
    \sin30^\circ=\frac{6}{l},\qquad l=12.
    \]

    又因为

    \[
    r=l\cos30^\circ=12\times\frac{\sqrt{3}}{2}=6\sqrt{3},
    \]

    所以圆锥的侧面积为

    \[
    S_{\text{侧}}=\pi r l=\pi\times6\sqrt{3}\times12=72\sqrt{3}\pi\mathrm{~cm}^2.
    \]

    \item 直线 \(2x-5y+3=0\) 化为 \(y=\frac{2}{5}x+\frac{3}{5}\)，所以它的斜率为 \(\frac{2}{5}\). 所求直线与它垂直，斜率为 \(-\frac{5}{2}\)，又经过点 \(\left(1,4\right)\)，由点斜式得

    \[
    y-4=-\frac{5}{2}\left(x-1\right).
    \]

    化简得所求直线方程为 \(5x+2y-13=0\).
\end{enumerate}
\end{solution}

\begin{problem}
如果 \( \triangle ABC \) 的 \( \angle A \) 的平分线交 \(BC\) 于 \(D\)，交它的外接圆于 \(E\)，那么 \( AB\cdot AC=AD\cdot AE \).

\bitmapfigure[width=4.8cm]{img/1977/heilongjiang/q5_fig1.png}
\end{problem}

\begin{solution}
因为 \(A,D,E\) 三点共线，且 \(AD\) 是 \(\angle A\) 的平分线，所以 \(\angle BAD=\angle EAC\). 又因为 \(A,B,C,E\) 共圆，\(\angle ABD=\angle ABC\) 与 \(\angle AEC\) 都是所对弦 \(AC\) 的圆周角，所以 \(\angle ABD=\angle AEC\). 因此 \(\triangle ABD\sim\triangle AEC\)，从而

\[
\frac{AD}{AC}=\frac{AB}{AE}.
\]

交叉相乘，得 \(AB\cdot AC=AD\cdot AE\)，结论成立.
\end{solution}

\begin{problem}
前进大队响应毛主席关于“绿化祖国”的伟大号召，\(1975\) 年造林 \(200\text{ 亩}\)，又知 \(1975\) 年至 \(1977\) 年这三年内共造林 \(728\text{ 亩}\)，求后两年造林面积的年平均增长率是多少？
\end{problem}

\begin{solution}
设后两年造林面积的年平均增长率为 \(r\)，则 \(1+r\) 为每年的增长倍数. 于是 \(1976\) 年和 \(1977\) 年的造林面积分别为 \(200\left(1+r\right)\text{ 亩}\) 和 \(200\left(1+r\right)^2\text{ 亩}\). 由三年总面积得

\[
200+200\left(1+r\right)+200\left(1+r\right)^2=728.
\]

令 \(q=1+r\)，则

\[
q^2+q+1=\frac{728}{200}=\frac{91}{25},
\]

即 \(q^2+q-\frac{66}{25}=0\). 解得

\[
q=\frac{-1\pm\sqrt{1+\frac{264}{25}}}{2}=\frac{-1\pm\frac{17}{5}}{2},
\]

所以 \(q=\frac{6}{5}\) 或 \(q=-\frac{11}{5}\). 因为造林面积为正，增长倍数 \(q=1+r\) 必须为正，故 \(q=\frac{6}{5}\)，从而

\[
r=\frac{6}{5}-1=\frac{1}{5}=20\%.
\]

所以后两年造林面积的年平均增长率为 \(20\%\).
\end{solution}

\begin{problem}
解方程：\(\lg\left(2^x+2x-16\right)=x\left(1-\lg 5\right)\).
\end{problem}

\begin{solution}
因为常用对数的真数必须为正，所以先要求 \(2^x+2x-16>0\). 又因为 \(\lg 10=1\)，所以

\[
x\left(1-\lg5\right)=x\left(\lg10-\lg5\right)=x\lg2=\lg2^x.
\]

原方程因此化为 \(\lg\left(2^x+2x-16\right)=\lg2^x\). 两边真数均为正，利用常用对数函数的单调性，得

\[
2^x+2x-16=2^x,
\]

所以 \(2x-16=0\)，即 \(x=8\). 此时真数为 \(2^8+16-16=256>0\)，满足定义域，故原方程的解为 \(x=8\).
\end{solution}

\begin{problem}
已知三角形的三边成等差数列，周长为 \(36 \mathrm{~cm}\)，面积为 \(54 \mathrm{~cm}^2\)，求三边的长.
\end{problem}

\begin{solution}
设三边按从小到大的顺序为 \(12-d\)，\(12\)，\(12+d\)，其中 \(d\geqslant0\). 半周长为 \(s=18\). 由三角形不等式，需有 \(12-d+12>12+d\)，即 \(d<6\). 根据海伦公式，

\[
\begin{aligned}
54^2&=18\left[18-\left(12-d\right)\right]\left(18-12\right)\left[18-\left(12+d\right)\right]\\
&=18\left(6+d\right)\cdot6\left(6-d\right)=108\left(36-d^2\right).
\end{aligned}
\]

因此 \(2916=108\left(36-d^2\right)\)，即 \(d^2=9\). 因为 \(d\geqslant0\)，所以 \(d=3\)，三边长分别为 \(9 \mathrm{~cm}\)、\(12 \mathrm{~cm}\)、\(15 \mathrm{~cm}\). 检验得 \(9+12>15\)，且由海伦公式面积确为 \(54 \mathrm{~cm}^2\)，所以答案成立.
\end{solution}

\section{附加题}

\begin{problem}
如图，\(AP\) 表示发动机的连杆，\(OA\) 表示它的曲柄. 当 \(A\) 在圆上作圆周运动时，\(P\) 在 \(x\) 轴上作直线运动，求 \(P\) 点的横坐标. 为什么当 \(\alpha\) 是直角时，\(\angle P\) 是最大？

\bitmapfigure[width=6.4cm]{img/1977/heilongjiang/q9_fig1.png}
\end{problem}

\begin{solution}
取 \(O\) 为原点，题图中的 \(x\) 轴为横轴. 设 \(P\) 点的横坐标为 \(x\)，则 \(P=\left(x,0\right)\). 由 \(OA=R\) 及 \(\angle AOP=\alpha\)，得

\[
A=\left(R\cos\alpha,R\sin\alpha\right).
\]

又因为连杆长度 \(AP=l\)，所以

\[
l^2=\left(x-R\cos\alpha\right)^2+R^2\sin^2\alpha.
\]

整理得

\[
\left(x-R\cos\alpha\right)^2=l^2-R^2\sin^2\alpha.
\]

在图示机构的取支中，\(P\) 位于 \(A\) 的右侧，故 \(x-R\cos\alpha>0\). 因此 \(P\) 点的横坐标为

\[
x=R\cos\alpha+\sqrt{l^2-R^2\sin^2\alpha},
\]

其中必须满足 \(l^2\geqslant R^2\sin^2\alpha\).

设 \(\theta=\angle APO\). 在三角形 \(AOP\) 中，\(\angle AOP=\alpha\)，边 \(OA=R\) 和边 \(AP=l\) 分别为 \(\angle APO\) 和 \(\angle AOP\) 的对边. 由正弦定理，

\[
\frac{\sin\theta}{R}=\frac{\sin\alpha}{l},
\qquad \sin\theta=\frac{R}{l}\sin\alpha.
\]

由上面的取支条件，\(\cos\theta=\frac{x-R\cos\alpha}{l}\geqslant0\)，所以图示位置的 \(\theta\) 为锐角. 当 \(\alpha\) 可以取到直角时，\(\sin\alpha\leqslant1\)，且等号当且仅当 \(\alpha=\frac{\pi}{2}\). 因而 \(\sin\theta\) 以及 \(\theta=\angle P\) 在 \(\alpha=\frac{\pi}{2}\) 时取得最大值，这就说明了题中的结论.
\end{solution}

\section{参考题}

\begin{problem}
求曲线 \(y=\sin x\) 在 \(\left[0,\pi\right]\) 上的曲边梯形绕 \(x\) 轴旋转一周所形成的旋转体的体积.
\end{problem}

\begin{solution}
在区间 \(\left[0,\pi\right]\) 上，\(\sin x\geqslant0\)，曲边梯形绕 \(x\) 轴旋转所得截面为半径 \(\sin x\) 的圆. 利用恒等式 \(\sin^2x=\frac{1-\cos2x}{2}\)，旋转体体积为

\[
\begin{gathered}
V=\pi\int_0^\pi\sin^2x\,\mathrm{d}x=\pi\int_0^\pi\frac{1-\cos2x}{2}\,\mathrm{d}x\\
=\pi\left[\frac{x}{2}-\frac{\sin2x}{4}\right]_{0}^{\pi}=\pi\cdot\frac{\pi}{2}=\frac{\pi^2}{2}.
\end{gathered}
\]

所以旋转体的体积为 \(\frac{\pi^2}{2}\).
\end{solution}
