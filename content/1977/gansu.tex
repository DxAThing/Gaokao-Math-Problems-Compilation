\chapter{1977年甘肃卷}

\begin{problem}
比较：\( 0.25+3\frac{1}{8} \) 与 \( \left| -3.375 \right| \)；

\end{problem}

\begin{solution}
因为 \( 0.25+3\frac{1}{8}=3\frac{3}{8}=3.375 \)，\( \left| -3.375 \right|=3.375 \)，所以 \( 0.25+3\frac{1}{8}=\left| -3.375 \right| \).
\end{solution}

\begin{problem}
比较：\( \left( \frac{\sqrt{3}}{3}-\frac{1}{\sqrt{3}} \right)^3 \) 与 \( \left( -3 \right)^0 \)；

\end{problem}

\begin{solution}
因为 \( \left( \frac{\sqrt{3}}{3}-\frac{1}{\sqrt{3}} \right)^3=\left( \frac{\sqrt{3}-\sqrt{3}}{3} \right)^3=0 \)，且 \( \left( -3 \right)^0=1 \)，所以 \( \left( \frac{\sqrt{3}}{3}-\frac{1}{\sqrt{3}} \right)^3<\left( -3 \right)^0 \).
\end{solution}

\begin{problem}
比较：\( -\lg 0.01 \) 与 \( \log_4 \frac{1}{16} \)；

\end{problem}

\begin{solution}
因为 \( -\lg 0.01 \) 是个正值，而 \( \log_4 \frac{1}{16} \) 是个负值，所以 \( -\lg 0.01>\log_4 \frac{1}{16} \).
\end{solution}

\begin{problem}
比较：\( \tan 44^\circ \) 与 \( \cot 44^\circ \).

\end{problem}

\begin{solution}
因为 \( \cot 44^\circ=\tan 46^\circ \)，正切函数在第一象限是增函数，有 \( \tan 44^\circ<\tan 46^\circ \)，所以 \( \tan 44^\circ<\cot 44^\circ \).
\end{solution}


\begin{problem}
计算：\( \frac{2a}{a-3b}-\frac{6ab-24b^2}{a^2-7ab+12b^2} \).

\end{problem}

\begin{solution}
\[
\text{原式}=\frac{2a}{a-3b}-\frac{6b}{a-3b}=\frac{2\left(a-3b\right)}{a-3b}=2
\]
\end{solution}

\begin{problem}
化简：\( \frac{\left( a^{\frac{8}{5}}b^{-\frac{6}{5}} \right)^{-\frac{1}{2}}\sqrt[5]{a^4}}{\sqrt[5]{b^3}} \).

\end{problem}

\begin{solution}
原式 \( =\frac{a^{-\frac{4}{5}}b^{\frac{3}{5}}a^{\frac{4}{5}}}{b^{\frac{3}{5}}}=1 \).
\end{solution}

\begin{problem}
证明：\( \frac{2\cot x}{1+\cot^2 x}=\sin 2x \).

\end{problem}

\begin{solution}
\[
\text{左边}=\frac{2\cot x}{\sec^2 x}=2\cot x\cos^2 x=2\sin x\cos x=\sin 2x=\text{右边}
\]
所以 \( \frac{2\cot x}{1+\cot^2 x}=\sin 2x \).
\end{solution}

\begin{problem}
求函数 \( y=\frac{1}{2}x^2+3x+9 \) 的极值.

\end{problem}

\begin{solution}
\[
y=\frac{1}{2}x^2+3x+9=\frac{1}{2}\left( x^2+6x \right)+9=\frac{1}{2}\left( x+3 \right)^2+\frac{9}{2}
\]
因此当 \( x=-3 \) 时，有 \( y \) 最小 \( =\frac{9}{2} \).
\end{solution}

\begin{problem}
\(AD\) 是 \( \triangle ABC \) 外接圆的直径，\( AD=6 \)，\( \angle DAC=\angle ABC \)，求 \(AC\)（精确到 \( 0.01 \)）.

\bitmapfigure[max width=0.36\linewidth,max totalheight=4.8cm]{img/1977/gansu/q3_fig1.png}

\end{problem}

\begin{solution}
连接 \(CD\).
因为 \(AD\) 是直径，所以 \( \angle ACD=90^\circ \).
又因为 \( \angle ADC=\angle ABC \)（同弧上的圆周角相等），而 \( \angle ABC=\angle DAC \)（已知），所以 \( \angle ADC=\angle DAC \).
因此 \( \triangle ADC \) 是等腰直角三角形，
\[
AC=\frac{\sqrt{2}}{2}AD=\frac{\sqrt{2}}{2}\times 6=3\sqrt{2}\approx 4.24
\]
答：\(AC\) 约为 \( 4.24 \).
\end{solution}

\begin{problem}
在 \( \triangle ABC \) 中，\( \angle A=30^\circ \)，\( \angle B-\angle C=60^\circ \)，\( BC=2 \)，求 \(AC\)（用根式表示）.

\bitmapfigure[max width=0.52\linewidth,max totalheight=4.2cm]{img/1977/gansu/q4_fig1.png}

\end{problem}

\begin{solution}
因为 \( \angle B+\angle C=180^\circ-\angle A=150^\circ \)，又已知 \( \angle B-\angle C=60^\circ \)，所以 \( \angle B=105^\circ \)，\( \angle C=45^\circ \). 根据正弦定理，有
\[
AC=\frac{BC\sin B}{\sin A}=\frac{2\sin 105^\circ}{\sin 30^\circ}=4\sin 105^\circ
\]
又
\[
\sin 105^\circ=\sin \left( 60^\circ+45^\circ \right)=\sin 60^\circ\cos 45^\circ+\cos 60^\circ\sin 45^\circ=\frac{\sqrt{6}+\sqrt{2}}{4}
\]
所以 \( AC=4\cdot \frac{\sqrt{6}+\sqrt{2}}{4}=\sqrt{6}+\sqrt{2} \).
\end{solution}

\begin{problem}
某生产队现已平整土地 \( 100 \) 亩，计划两年后平整土地的总面积要在 \( 225 \) 亩以上，求每年平均增长率.

\end{problem}

\begin{solution}
（注）此题题意不清.

（1）“现已平整土地 \( 100 \) 亩”没有交代是否在这一年内平整的，因此不能把 \( 100 \) 作为基数，题无法解.

（2）“两年后平整土地的总面积”这句话意思不清，是那一年里平整的总面积呢，还是所有平整的土地的总面积？若是后一种，求得百分数是下降，而不是增加，与实际不符.

如题改为“今年已平整土地 \( 100 \) 亩，计划在后年要达到一年内平整 \( 225 \) 亩以上，求每年平均增长率”，那么可列出方程 \( 100\left( 1+x \right)^2>225 \).

求得 \( x>50\% \). 因此每年平均增长率应大于 \( 50\% \).
\end{solution}

\begin{problem}
右图的二视图表示什么形体？用直径为 \( 40 \) 毫米的圆柱形钢材锻压这样一个工件，需截取多长一段钢材？

\bitmapfigure[max width=0.44\linewidth,max totalheight=6.8cm]{img/1977/gansu/q6_fig1.png}

\end{problem}

\begin{solution}
这个视图表示一个圆台，这个圆台的上下底面的半径分别是 \( r_1=10 \) 毫米，\( r_2=30 \) 毫米，高 \( h=20\tan 45^\circ=20 \) 毫米.
这个圆台的体积
\[
\begin{aligned}
V&=\frac{1}{3}\left( \pi r_1^2+\pi r_2^2+\pi r_1r_2 \right)h\\
&=\frac{\pi}{3}\left( 10^2+30^2+10\times 20 \right)\times 20
=\frac{26000\pi}{3}\text{（立方毫米）}.
\end{aligned}
\]
设锻压这样一个工件需截取 \( x \) 毫米长的直径为 \( 40 \) 毫米的圆柱形钢材，则依题意有 \( \pi \left( \frac{40}{2} \right)^2x=\frac{26000\pi}{3} \).
所以
\[
x=\frac{26000}{3\cdot 20^2}=\frac{65}{3}=21\frac{2}{3}\approx 21.7\text{（毫米）}
\]
答：锻压这样一个工件，需截取 \( 21.7 \) 毫米长，直径为 \( 40 \) 毫米的圆柱形钢材.
\end{solution}

\begin{problem}
一条直线通过 \( \left( 3,0 \right) \)，并且和直线 \( y=-2x+9 \) 垂直，求这条直线与直线 \( 2y+5x=3 \) 的交点的坐标.

\end{problem}

\begin{solution}
直线 \( y=-2x+9 \) 的斜率是 \( -2 \)，所以和它垂直的直线的斜率是 \( \frac{1}{2} \). 因此，所求的直线方程为 \( y-0=\frac{1}{2}\left( x-3 \right) \)，即 \( x-2y=3 \).
这条直线与直线 \( 2y+5x=3 \) 的交点坐标 \( \left( x,y \right) \) 应满足方程组 \( x-2y=3 \)，\( 2y+5x=3 \). 解得 \( x=1 \)，\( y=-1 \). 因此，所求交点坐标是 \( \left( 1,-1 \right) \).
\end{solution}

\begin{problem}
已知 \( \frac{dy}{dx}=x\e^x+2\cos x+3x^2 \)，求 \( y \).

\end{problem}

\begin{solution}
在已知等式两边积分，得到 \( y=\int \left( x\e^x+2\cos x+3x^2 \right)\,dx \)，积分得 \( y=\e^x\left( x-1 \right)+2\sin x+x^3+C \)，其中 \(C\) 是任意常数.
\end{solution}

\begin{problem}
过点 \( P\left( 4,0 \right) \) 向曲线 \( x^2+y^2=1 \) 作切线，求切线方程.

\end{problem}

\begin{solution}
设切点为 \( \left( x_0,y_0 \right) \)，则切线方程是
\[
x_0x+y_0y=1. \tag{1}
\]
因为 \( \left( x_0,y_0 \right) \) 在曲线上，所以
\[
x_0^2+y_0^2=1. \tag{2}
\]
又因 \( P\left( 4,0 \right) \) 点在所求的切线上，所以 \( \left( 4,0 \right) \) 适合方程 \circled{1}，即
\[
4x_0=1. \tag{3}
\]
由 \circled{3} 求得 \( x_0=\frac{1}{4} \)，代入 \circled{2}，求得 \( y_0=\pm \frac{\sqrt{15}}{4} \).
因此切点坐标为 \( \left( \frac{1}{4},\frac{\sqrt{15}}{4} \right) \) 和 \( \left( \frac{1}{4},-\frac{\sqrt{15}}{4} \right) \).
把切点坐标代入 \circled{1}，得所求的切线方程为 \( x+\sqrt{15}y=4 \) 和 \( x-\sqrt{15}y=4 \).
\end{solution}
