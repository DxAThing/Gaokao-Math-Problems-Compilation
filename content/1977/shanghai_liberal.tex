\begin{problem}
计算：\(\left[\left(\frac{1}{2}-\frac{1}{3}\right)\left(-\frac{3}{2}\right)-\left(-1+\frac{1}{3}\right)\times\left(-\frac{3}{4}\right)\right]\div\frac{3}{2}\).
\end{problem}

\begin{problem}
某生产队去年养猪 \(96\text{ 头}\)，今年养猪 \(120\text{ 头}\)，问今年比去年增加百分之几？计划明年比今年多养 \(40\%\)，明年养猪几头？
\end{problem}

\begin{problem}
在 \(\triangle ABC\) 中，\(\angle C\) 的平分线交 \(AB\) 于 \(D\)，过 \(D\) 作 \(BC\) 的平分线交 \(AC\) 于 \(E\)，已知 \(BC=a\)，\(AC=b\)，求 \(DE\) 的长.

\end{problem}

\begin{problem}
化简：\(\left(\frac{a}{a+b}-\frac{a^2}{a^2+2ab+b^2}\right)\div\left(\frac{a}{a+b}-\frac{a^2}{a^2-b^2}\right)\).
\end{problem}

\begin{problem}
解不等式：\(\frac{2x-1}{3}>\frac{3x-1}{2}-4\).
\end{problem}

\begin{problem}
解方程：\(\frac{4}{x+3}-\frac{1}{x-3}=1-\frac{2x}{x^2-9}\).

\end{problem}

\begin{problem}
计算：\(\frac{\sin 225^\circ+\tan 330^\circ}{\cos(-120^\circ)}\).
\end{problem}

\begin{problem}
求证：\(\tan x+\cot x=\frac{2}{\sin 2x}\).
\end{problem}

\begin{problem}
\(\triangle ABC\) 中，\(\angle A=45^\circ\)，\(\angle B=75^\circ\)，\(AB=12\)，求 \(BC\) 的长.

\end{problem}

\begin{problem}
求直线 \(x+\sqrt{3}y+3\sqrt{3}=0\) 的斜率和倾角，并画出它的图形.

\end{problem}

\begin{problem}
当 \(x\) 为何值时，函数 \(y=x^2-8x+5\) 的值最小，并求出这个最小值.

\end{problem}

\begin{problem}
将浓度为 \(96\%\) 和 \(36\%\) 的甲、乙两种流酸配制成浓度为 \(70\%\) 的流酸 \(600\text{ 升}\)，问应从甲、乙两种流酸中各取多少升.
\end{problem}
