\begin{problem}
\begin{enumerate}
\item 化简：\(\left(\frac{a}{a+b}-\frac{a^2}{a^2+2ab+b^2}\right)\div\left(\frac{a}{a+b}-\frac{a^2}{a^2-b^2}\right)\).

\item 解不等式：\(\frac{2x-1}{3}>\frac{3x-1}{2}-4\).

\item 解方程：\(\frac{4}{x+3}-\frac{1}{x-3}=1-\frac{2x}{x^2-9}\).
\end{enumerate}

\end{problem}
