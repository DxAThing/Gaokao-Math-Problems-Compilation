\chapter{1977年江苏南通地区初试卷}

\section{解答题}

\begin{problem}
\begin{enumerate}
\item 当 \(a\) 为何值时，\(2a>a\)；当 \(a\) 为何值时，\(2a=a\)；当 \(a\) 为何值时，\(2a<a\).
\item 当 \(x\) 为何值时，\(|x|=x\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 分别为 \(a>0\)、\(a=0\)、\(a<0\).

\item \(x\geqslant0\).
\end{enumerate}
\end{solution}

\begin{problem}
写出下列函数的自变量 \(x\) 的取值范围（在实数范围内）：

\begin{enumerate}
\item \(y=\dfrac{1}{2-x}\).
\item \(y=\sqrt{2x-1}\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item \(x\neq2\).

\item \(x\geqslant\dfrac12\).
\end{enumerate}
\end{solution}

\begin{problem}
计算：\(\left(\dfrac{2}{3}\right)^2+\left(-5.8\right)^0-\left(\dfrac{8}{27}\right)^{-\frac23}+\left(\dfrac12\right)^{-2}\).
\end{problem}

\begin{solution}
\(3\dfrac7{36}\)
\end{solution}

\begin{problem}
\(k\) 为何值时，方程 \(x^2-2kx+4=0\) 有两个相等的实数根？并求出这方程的根.
\end{problem}

\begin{solution}
\(k=2\) 时，\(x_1=x_2=2\)；\(k=-2\) 时，\(x_1=x_2=-2\).
\end{solution}

\begin{problem}
已知 \(\sin\alpha=\dfrac35\)（\(90^\circ<\alpha<180^\circ\)），求 \(\cos\alpha\)，\(\tan\alpha\)，\(\cot\alpha\) 的值.
\end{problem}

\begin{solution}
\(\cos\alpha=-\dfrac45\)，\(\tan\alpha=-\dfrac34\)，\(\cot\alpha=-\dfrac43\).
\end{solution}

\begin{problem}
试写出抛物线 \(y=1+4x-2x^2\) 的顶点坐标，对称轴方程和函数 \(y\) 的极值.
\end{problem}

\begin{solution}
顶点坐标为 \((1,3)\)，对称轴方程为 \(x=1\)，极大值为 \(3\).
\end{solution}

\begin{problem}
解方程：\(\dfrac{\lg(2x)}{\lg(x-4)}=2\).
\end{problem}

\begin{solution}
\(x=8\).
\end{solution}

\begin{problem}
求 \(200\) 以内的能被 \(6\) 整除的所有自然数的和.
\end{problem}

\begin{solution}
\(3366\).
\end{solution}

\begin{problem}
已知正四棱锥底面边长为 \(12\mathrm{~cm}\)，高为 \(2\sqrt7\mathrm{~cm}\)，求它的侧棱.
\end{problem}

\begin{solution}
\(10\mathrm{~cm}\).
\end{solution}

\begin{problem}
两根钢绳悬吊货物时，两条钢绳与铅垂线的夹角分别为 \(\alpha\) 和 \(\beta\)，货物的重量为 \(P\)，试用表达式分别写出钢绳所承受的拉力 \(F_1\) 和 \(F_2\).
\end{problem}

\begin{solution}
\(F_1=\dfrac{P\sin\beta}{\sin(\alpha+\beta)}\)，\(F_2=\dfrac{P\sin\alpha}{\sin(\alpha+\beta)}\).
\end{solution}
