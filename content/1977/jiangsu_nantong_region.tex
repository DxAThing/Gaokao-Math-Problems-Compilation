\chapter{1977年江苏南通地区初试卷}

\begin{problem}
当 \(a\) 为何值时，\(2a>a\).
\end{problem}

\begin{problem}
当 \(a\) 为何值时，\(2a=a\).
\end{problem}

\begin{problem}
当 \(a\) 为何值时，\(2a<a\).
\end{problem}

\begin{problem}
当 \(x\) 为何值时，\(|x|=x\).
\end{problem}

\begin{problem}
写出函数 \(y=\frac{1}{2-x}\) 的自变量 \(x\) 的取值范围（在实数范围内）.
\end{problem}

\begin{problem}
写出函数 \(y=\sqrt{2x-1}\) 的自变量 \(x\) 的取值范围（在实数范围内）.
\end{problem}

\begin{problem}
计算：\( \left( \frac{2}{3} \right)^2+\left( -5.8 \right)^0-\left( \frac{8}{27} \right)^{-\frac{2}{3}}+\left( \frac{1}{2} \right)^{-2} \).
\end{problem}

\begin{problem}
\(k\) 为何值时，方程 \(x^2-2kx+4=0\) 有两个相等的实数根？并求出这方程的根.
\end{problem}

\begin{problem}
已知 \(\sin\alpha=\frac{3}{5}\)（\(90^\circ<\alpha<180^\circ\)），求 \(\cos\alpha\)、\(\tan\alpha\)、\(\cot\alpha\) 的值.
\end{problem}

\begin{problem}
试写出抛物线 \(y=1+4x-2x^2\) 的顶点坐标、对称轴方程和函数 \(y\) 的极值.
\end{problem}

\begin{problem}
解方程：\( \dfrac{\lg\left(2x\right)}{\lg\left(x-4\right)}=2 \).
\end{problem}

\begin{problem}
求 \(200\) 以内的能被 \(6\) 整除的所有自然数的和.
\end{problem}

\begin{problem}
已知正四棱锥底面边长为 \(12\mathrm{~cm}\)，高为 \(2\sqrt{7}\mathrm{~cm}\)，求它的侧棱.
\end{problem}

\begin{problem}
两根钢绳悬吊货物时，两条钢绳与铅垂线的夹角分别为 \(\alpha\) 和 \(\beta\)，货物的重量为 \(P\)，试用表达式分别写出钢绳所承受的拉力 \(F_1\) 和 \(F_2\).
\end{problem}
