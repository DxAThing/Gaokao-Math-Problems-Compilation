\section{解答题}

\begin{problem}
计算 \(\left(2\frac{7}{9}\right)^{\frac{1}{2}}+3^{-2}-\left(\frac{1}{3}\right)^{0}\).
\end{problem}

\begin{solution}
因为 \(2\frac{7}{9}=\frac{25}{9}\)，且平方根取算术平方根，所以 \(\left(2\frac{7}{9}\right)^{\frac{1}{2}}=\sqrt{\frac{25}{9}}=\frac{5}{3}\). 又因为 \(3\neq0\)，所以 \(3^{-2}=\frac{1}{3^{2}}=\frac{1}{9}\)，并且 \(\left(\frac{1}{3}\right)^{0}=1\). 因此原式
\[
=\frac{5}{3}+\frac{1}{9}-1=\frac{15+1-9}{9}=\frac{7}{9}
\]
\end{solution}

\begin{problem}
化简：\(\sqrt{12}-\sqrt{48}-3\sqrt{3}\).
\end{problem}

\begin{solution}
将被开方数分解为完全平方因数与 \(3\) 的乘积，得 \(\sqrt{12}=\sqrt{4\times3}=2\sqrt{3}\)，\(\sqrt{48}=\sqrt{16\times3}=4\sqrt{3}\). 所以原式
\[
=2\sqrt{3}-4\sqrt{3}-3\sqrt{3}=(2-4-3)\sqrt{3}=-5\sqrt{3}
\]
\end{solution}

\begin{problem}
分解因式：\(mx^{2}+2mxy+my^{2}\).
\end{problem}

\begin{solution}
各项都有公因式 \(m\)，先提取公因式，再利用完全平方公式，得 \(mx^{2}+2mxy+my^{2}=m\left(x^{2}+2xy+y^{2}\right)=m\left(x+y\right)^{2}\).
\end{solution}

\begin{problem}
解方程：\(2x^{2}-3x+1=0\).
\end{problem}

\begin{solution}
将左端分解因式，得 \(2x^{2}-3x+1=\left(2x-1\right)\left(x-1\right)\)，所以 \(\left(2x-1\right)\left(x-1\right)=0\). 由零因式定理，得 \(2x-1=0\) 或 \(x-1=0\)，从而 \(x=\frac{1}{2}\) 或 \(x=1\). 代回原方程，两值均使左端为零，故解集为 \(\left\{\frac{1}{2},1\right\}\).
\end{solution}

\begin{problem}
已知 \(\sin\alpha=\frac{3}{5}\)（\(\alpha\) 是锐角），求

\begin{enumerate}
\item \(\cos\alpha\)；
\item \(\tan\alpha\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由同角三角函数的基本关系式，
\[
\cos^{2}\alpha=1-\sin^{2}\alpha=1-\left(\frac{3}{5}\right)^{2}=\frac{16}{25}
\]
由于 \(\alpha\) 是锐角，\(\cos\alpha>0\)，所以 \(\cos\alpha=\frac{4}{5}\).

\item 由正切与正弦、余弦的关系式，
\[
\tan\alpha=\frac{\sin\alpha}{\cos\alpha}=\frac{\frac{3}{5}}{\frac{4}{5}}=\frac{3}{4}
\]
\end{enumerate}
\end{solution}

\begin{problem}
不查表求：\(\lg 8+\lg 125-\lg 1\).
\end{problem}

\begin{solution}
因为 \(8=2^{3}\)，\(125=5^{3}\)，且 \(\lg1=0\)，所以
\[
\lg8+\lg125-\lg1=3\lg2+3\lg5=3\left(\lg2+\lg5\right)=3\lg10=3
\]
\end{solution}

\begin{problem}
求经过点 \((4,-4)\) 和原点的直线方程.
\end{problem}

\begin{solution}
设所求直线的斜率为 \(k\). 由于直线经过原点，其方程可写为 \(y=kx\). 将点 \((4,-4)\) 代入，得 \(-4=4k\)，所以 \(k=-1\). 因此所求直线方程为 \(y=-x\).
\end{solution}

\begin{problem}
解不等式：\(\left(x+1\right)\left(x-2\right)<0\).
\end{problem}

\begin{solution}
乘积为负，两个因式必须异号，因此原不等式等价于下面两个不等式组的并集：
\[
\begin{cases}
x+1>0,\\
x-2<0,
\end{cases}
\qquad\text{或}\qquad
\begin{cases}
x+1<0,\\
x-2>0.
\end{cases}
\]
第一个不等式组给出 \(-1<x<2\). 第二个不等式组要求同时满足 \(x<-1\) 和 \(x>2\)，无解. 因而原不等式的解集为 \((-1,2)\).
\end{solution}

\begin{problem}
求 \(\frac{1}{2}+\frac{1}{4}+\frac{1}{8}+\frac{1}{16}+\cdots+\frac{1}{2^{10}}\).
\end{problem}

\begin{solution}
这是首项为 \(\frac{1}{2}\)、公比为 \(\frac{1}{2}\)、项数为 \(10\) 的等比数列的前 \(10\) 项和. 由等比数列求和公式，
\[
S=\frac{1}{2}\frac{1-\left(\frac{1}{2}\right)^{10}}{1-\frac{1}{2}}
 =1-\frac{1}{2^{10}}
 =\frac{2^{10}-1}{2^{10}}
 =\frac{1023}{1024}
\]
\end{solution}

\begin{problem}
已知：在 \(\triangle ABC\) 中，\(\angle B=60^{\circ}\)，\(AD\) 垂直于 \(BC\)，\(AD=3\)，\(AC=5\)，求：\(BC\)，\(AB\).

\bitmapfigure[max width=0.38\linewidth]{img/1977/ningxia/q10_fig1.png}
\end{problem}

\begin{solution}
因为 \(AD\perp BC\)，所以 \(\triangle ABD\) 和 \(\triangle ACD\) 都是直角三角形. 在 \(\triangle ABD\) 中，
由 \(\tan\angle B=\frac{AD}{BD}\)，故 \(BD=\frac{AD}{\tan60^{\circ}}=\frac{3}{\sqrt{3}}=\sqrt{3}\).
由勾股定理，
\[
AB=\sqrt{AD^{2}+BD^{2}}=\sqrt{3^{2}+\left(\sqrt{3}\right)^{2}}=\sqrt{12}=2\sqrt{3}
\]

在 \(\triangle ACD\) 中，由勾股定理，
\(DC=\sqrt{AC^{2}-AD^{2}}=\sqrt{5^{2}-3^{2}}=4\). 图中 \(D\) 在 \(BC\) 上，所以 \(BC=BD+DC=\sqrt{3}+4\).
\end{solution}

\section{选作题}

\begin{problem}
在 \(\triangle ABC\) 中，已知：\(a=\sqrt{2}\)，\(A=45^{\circ}\)，\(B=60^{\circ}\)，求 \(C\)，\(b\)，\(c\) 及面积 \(\triangle ABC\).

\bitmapfigure[max width=0.34\linewidth]{img/1977/ningxia/q11_fig1.png}
\end{problem}

\begin{solution}
三角形内角和为 \(180^{\circ}\)，所以
\[
C=180^{\circ}-A-B=180^{\circ}-45^{\circ}-60^{\circ}=75^{\circ}
\]
由正弦定理，
\[
\frac{b}{\sin B}=\frac{a}{\sin A}
\]
从而
\[
b=\frac{a\sin B}{\sin A}=\frac{\sqrt{2}\sin60^{\circ}}{\sin45^{\circ}}=\frac{\sqrt{2}\cdot\frac{\sqrt{3}}{2}}{\frac{\sqrt{2}}{2}}=\sqrt{3}
\]
同理，\(c=\frac{a\sin C}{\sin A}=\frac{\sqrt{2}\sin75^{\circ}}{\sin45^{\circ}}=2\sin75^{\circ}\).
利用和角公式，
\[
\sin75^{\circ}=\sin\left(30^{\circ}+45^{\circ}\right)
 =\sin30^{\circ}\cos45^{\circ}+\cos30^{\circ}\sin45^{\circ}
 =\frac{1}{2}\cdot\frac{\sqrt{2}}{2}+\frac{\sqrt{3}}{2}\cdot\frac{\sqrt{2}}{2}
 =\frac{\sqrt{2}+\sqrt{6}}{4}
\]
因此 \(c=\frac{\sqrt{2}+\sqrt{6}}{2}\).
三角形面积为
\[
S=\frac{1}{2}ab\sin C
 =\frac{1}{2}\cdot\sqrt{2}\cdot\sqrt{3}\cdot\frac{\sqrt{2}+\sqrt{6}}{4}
 =\frac{\sqrt{12}+6}{8}
 =\frac{\sqrt{3}+3}{4}
\]
所以 \(C=75^{\circ}\)，\(b=\sqrt{3}\)，\(c=\frac{\sqrt{2}+\sqrt{6}}{2}\)，面积为 \(\frac{\sqrt{3}+3}{4}\).
\end{solution}

\begin{problem}
求与直线 \(x+y=1\) 相切于点 \((2,-1)\)，并且圆心在直线 \(y=-2x\) 上的圆的方程.
\end{problem}

\begin{solution}
设圆心为 \(O(a,b)\)，半径为 \(R\). 切点为 \(P(2,-1)\). 圆在 \(P\) 点的半径 \(OP\) 垂直于切线 \(x+y=1\). 切线的一个方向向量为 \((1,-1)\)，而半径方向向量可取切线的法向量 \((1,1)\)，故
 \(\overrightarrow{OP}=(2-a,-1-b)\) 的两个分量相等，即 \(2-a=-1-b\)，从而 \(a-b=3\).
又因为圆心在 \(y=-2x\) 上，所以 \(b=-2a\). 联立得
 \(
\begin{cases}
a-b=3,\\
b=-2a,
\end{cases}
\) 解得 \(a=1\)，\(b=-2\).
于是圆心为 \(O(1,-2)\). 半径的平方为
\[
R^{2}=OP^{2}=\left(2-1\right)^{2}+\left(-1+2\right)^{2}=2
\]
因此所求圆的方程为
\[
\left(x-1\right)^{2}+\left(y+2\right)^{2}=2
\]
\end{solution}

\begin{problem}
已知圆内接三角形 \(ABC\)，\(AB\)，\(BC\) 上的高 \(CE\)，\(AD\) 相交于 \(H\)，\(AD\) 延长线与 \(\widehat{BC}\) 交于 \(F\). 求证：\(HD=FD\).

\bitmapfigure[max width=0.38\linewidth]{img/1977/ningxia/q13_fig1.png}
\end{problem}

\begin{solution}
连接 \(FC\). 按图中角标，\(\angle1\) 是 \(\angle ABC\)，\(\angle2\) 是直线 \(BC\) 与高 \(CE\) 所成的锐角，\(\angle3\) 是直线 \(AD\) 与高 \(CE\) 在 \(H\) 处所成的锐角，\(\angle4\) 是 \(FC\) 与 \(FD\) 在 \(F\) 处所成的角.

因为 \(CE\perp AB\)，所以 \(\angle1+\angle2=90^{\circ}\). 又因为 \(AD\perp BC\)，所以 \(\angle3+\angle2=90^{\circ}\)，从而 \(\angle1=\angle3\). 四边形 \(ABFC\) 内接于同一圆，\(\angle ABC\) 与 \(\angle AFC\) 同对弧 \(AC\)，所以 \(\angle1=\angle AFC\). 又因 \(A\)，\(D\)，\(F\) 三点共线，\(\angle AFC=\angle4\)，从而 \(\angle1=\angle4\)，因此 \(\angle3=\angle4\).

由于 \(HD\) 和 \(FD\) 都在直线 \(AD\) 上，且 \(AD\perp BC\)，所以 \(\triangle HDC\) 与 \(\triangle FDC\) 都是以 \(D\) 为直角顶点的直角三角形. 在这两个三角形中，\(\angle H=\angle3=\angle4=\angle F\)，且 \(\angle D=90^{\circ}\)，故两三角形相似；又它们的对应边 \(DC\) 是同一条线段，故相似比为 \(1\)，两三角形全等. 因此对应边相等，即 \(HD=FD\).
\end{solution}

\begin{problem}
用 \(6\text{ 米}\) 长的条形木料做如图的矩形窗框（包括中间的两个横档），问窗框的长，宽尺寸如何选择，窗框的面积最大？

\bitmapfigure[max width=0.22\linewidth]{img/1977/ningxia/q14_fig1.png}
\end{problem}

\begin{solution}
设窗框的长为 \(x\text{ 米}\)，宽为 \(y\text{ 米}\). 窗框有两根竖档和四根横档，所以木料总长满足
\[
2x+4y=6
\]
窗框的长、宽均为正数，因此 \(0<x<3\)，且
\[
y=\frac{6-2x}{4}=\frac{3-x}{2}
\]
窗框面积为
\[
S=xy=x\frac{3-x}{2}
 =-\frac{1}{2}x^{2}+\frac{3}{2}x
 =-\frac{1}{2}\left(x-\frac{3}{2}\right)^{2}+\frac{9}{8}
\]
因为平方项满足 \(\left(x-\frac{3}{2}\right)^{2}\geqslant0\)，所以当 \(x=\frac{3}{2}\text{ 米}\) 时，面积取得最大值 \(\frac{9}{8}\text{ 平方米}\). 此时
\[
y=\frac{3-\frac{3}{2}}{2}=\frac{3}{4}\text{ 米}
\]
所以窗框的长为 \(\frac{3}{2}\text{ 米}\)、宽为 \(\frac{3}{4}\text{ 米}\) 时，面积最大.
\end{solution}

\begin{problem}
一大正六边形内接一个小正六边形，小正六边形各顶点分大正六边形各边成 \(m\)，\(n\) 两段. （\(S_{\text{大}}\) 表示大正六边形的面积）
\begin{enumerate}
\item 求小正六边形的面积 \(S_{\text{小}}\)；
\item 证明 \(4S_{\text{小}}\geqslant 3S_{\text{大}}\).
\end{enumerate}

\bitmapfigure[max width=0.38\linewidth]{img/1977/ningxia/q15_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设大正六边形相邻顶点为 \(A\)，\(B\)，\(C\)，小正六边形在边 \(AB\)、\(BC\) 上的相邻顶点为 \(A'\)、\(B'\). 由图，\(A'B=n\)，\(BB'=m\)，且 \(m>0\)，\(n>0\). 大正六边形的内角为 \(120^{\circ}\)，所以在 \(\triangle A'BB'\) 中，由余弦定理，
\[
A'B'^{2}=A'B^{2}+BB'^{2}-2A'B\cdot BB'\cos120^{\circ}
 =n^{2}+m^{2}-2mn\left(-\frac{1}{2}\right)
 =n^{2}+m^{2}+mn
\]
\(A'B'\) 是小正六边形的边长. 正六边形可分成六个边长等于其边长的正三角形，因此边长为 \(l\) 的正六边形面积为 \(6\cdot\frac{\sqrt{3}}{4}l^{2}=\frac{3\sqrt{3}}{2}l^{2}\). 于是
\[
S_{\text{小}}=\frac{3\sqrt{3}}{2}A'B'^{2}=\frac{3\sqrt{3}}{2}\left(m^{2}+n^{2}+mn\right)
\]

\item 大正六边形的边长为 \(AB=m+n\)，所以
\[
S_{\text{大}}=\frac{3\sqrt{3}}{2}\left(m+n\right)^{2}
\]
由 \(\left(m-n\right)^{2}\geqslant0\)，得
\[
mn\leqslant\left(\frac{m+n}{2}\right)^{2}
\]
因此
\[
\begin{gathered}
4S_{\text{小}}=6\sqrt{3}\left(m^{2}+n^{2}+mn\right)
 =6\sqrt{3}\left(\left(m+n\right)^{2}-mn\right)\\
\geqslant6\sqrt{3}\left(\left(m+n\right)^{2}-\left(\frac{m+n}{2}\right)^{2}\right)
 =\frac{9\sqrt{3}}{2}\left(m+n\right)^{2}=3S_{\text{大}}.
\end{gathered}
\]
所以 \(4S_{\text{小}}\geqslant3S_{\text{大}}\). 当且仅当 \(m=n\) 时取等号.
\end{enumerate}
\end{solution}

\section{附加题}

\begin{problem}
已知 \(\cos\alpha+\sin\alpha=\frac{1}{\sqrt{2}}\)，试求：

\begin{enumerate}
\item \(\sin^{4}\alpha+\cos^{4}\alpha\)；
\item \(\sin^{5}\alpha+\cos^{5}\alpha\).
\end{enumerate}
\end{problem}

\begin{solution}
令 \(s=\sin\alpha\)，\(c=\cos\alpha\). 已知 \(s+c=\frac{1}{\sqrt{2}}\). 两边平方，得
\[
1+2sc=\frac{1}{2}
\]
所以
\[
sc=-\frac{1}{4}
\]

\begin{enumerate}
\item 由 \(s^{2}+c^{2}=1\)，
\[
s^{4}+c^{4}=\left(s^{2}+c^{2}\right)^{2}-2s^{2}c^{2}
 =1-2\left(sc\right)^{2}
 =1-2\left(-\frac{1}{4}\right)^{2}
 =\frac{7}{8}
\]
因此 \(\sin^{4}\alpha+\cos^{4}\alpha=\frac{7}{8}\).

\item 先将五次幂和按 \(s+c\) 分解：
\[
s^{5}+c^{5}=(s^{4}+c^{4})(s+c)-sc(s^{3}+c^{3})
\]
又
\[
s^{3}+c^{3}=(s+c)^{3}-3sc(s+c)
\]
代入 \(s+c=\frac{1}{\sqrt{2}}\)、\(sc=-\frac{1}{4}\) 以及上面的结果，得
\[
\begin{aligned}
s^{5}+c^{5}&=(s+c)\left[(s^{4}+c^{4})-sc(s+c)^{2}+3s^{2}c^{2}\right]\\
&=\frac{1}{\sqrt{2}}\left[\frac{7}{8}-\left(-\frac{1}{4}\right)\left(\frac{1}{\sqrt{2}}\right)^{2}+3\left(-\frac{1}{4}\right)^{2}\right]\\
&=\frac{1}{\sqrt{2}}\left(\frac{7}{8}+\frac{1}{8}+\frac{3}{16}\right)=\frac{19}{16\sqrt{2}}=\frac{19\sqrt{2}}{32}.
\end{aligned}
\]
因此 \(\sin^{5}\alpha+\cos^{5}\alpha=\frac{19\sqrt{2}}{32}\).
\end{enumerate}
\end{solution}

\begin{problem}
\begin{enumerate}
\item 若 \(a_i>0\)（\(i=1\)，\(2\)，\(\dots\)，\(n\)），证明：\(\displaystyle\sum_{i=1}^{n}a_i\sum_{i=1}^{n}\frac{1}{a_i}\geqslant n^{2}\)；
\item 若 \(f(x)\) 在 \([0,1]\) 上连续，且 \(f(x)>0\)，证明：\(\displaystyle\int_{0}^{1}f(x)\,\mathrm{d}x\int_{0}^{1}\frac{\mathrm{d}x}{f(x)}\geqslant1\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由于所有 \(a_i\) 都是正数，算术平均数不小于几何平均数，故
\[
\frac{1}{n}\sum_{i=1}^{n}a_i\geqslant\sqrt[n]{\prod_{i=1}^{n}a_i}
\]
同理，
\[
\frac{1}{n}\sum_{i=1}^{n}\frac{1}{a_i}\geqslant\sqrt[n]{\prod_{i=1}^{n}\frac{1}{a_i}}
\]
两式左、右两边分别都是正数，可以相乘，得
\[
\frac{1}{n^{2}}\left(\sum_{i=1}^{n}a_i\right)\left(\sum_{i=1}^{n}\frac{1}{a_i}\right)
\geqslant
\sqrt[n]{\prod_{i=1}^{n}a_i\cdot\prod_{i=1}^{n}\frac{1}{a_i}}=1
\]
两边乘以 \(n^{2}\)，即得
\[
\left(\sum_{i=1}^{n}a_i\right)\left(\sum_{i=1}^{n}\frac{1}{a_i}\right)\geqslant n^{2}
\]
当且仅当 \(a_1=a_2=\cdots=a_n\) 时取等号.

\item 因为 \(f\) 在 \([0,1]\) 上连续且处处为正，所以 \(\frac{1}{f(x)}\) 也在 \([0,1]\) 上连续. 对任意正整数 \(n\)，取等分点 \(x_i=\frac{i}{n}\)（\(i=0\)，\(1\)，\(\dots\)，\(n\)），则每个小区间长度为 \(\frac{1}{n}\). 由定积分的定义，
\[
\int_{0}^{1}f(x)\,\mathrm{d}x=\displaystyle\lim_{n\to\infty}\frac{1}{n}\sum_{i=1}^{n}f(x_i)
\]
同理，
\[
\int_{0}^{1}\frac{\mathrm{d}x}{f(x)}=\displaystyle\lim_{n\to\infty}\frac{1}{n}\sum_{i=1}^{n}\frac{1}{f(x_i)}
\]
对每一个固定的 \(n\)，数 \(f(x_i)\)（\(i=1\)，\(2\)，\(\dots\)，\(n\)）均为正数. 应用第（1）问的不等式，得
\[
\left(\frac{1}{n}\sum_{i=1}^{n}f(x_i)\right)\left(\frac{1}{n}\sum_{i=1}^{n}\frac{1}{f(x_i)}\right)\geqslant1
\]
令
\[
A_n=\frac{1}{n}\sum_{i=1}^{n}f(x_i)
\]
并令
\[
B_n=\frac{1}{n}\sum_{i=1}^{n}\frac{1}{f(x_i)}
\]
由上式，\(A_nB_n\geqslant1\). 由定积分定义，\(A_n\to I\)，\(B_n\to J\)，其中
\[
I=\int_{0}^{1}f(x)\,\mathrm{d}x
\]
并且
\[
J=\int_{0}^{1}\frac{\mathrm{d}x}{f(x)}
\]
下面说明乘积的极限. 令 \(M=1+\lvert I\rvert+\lvert J\rvert\). 对任意 \(\varepsilon>0\)，由 \(A_n\to I\)、\(B_n\to J\)，存在正整数 \(N\)，使得当 \(n\geqslant N\) 时
\[
\lvert A_n-I\rvert<\min\left\{1,\frac{\varepsilon}{2M}\right\}
\]
且
\[
\lvert B_n-J\rvert<\frac{\varepsilon}{2M}
\]
此时 \(\lvert A_n\rvert\leqslant\lvert I\rvert+1\leqslant M\)，从而
\[
\lvert A_nB_n-IJ\rvert
\leqslant\lvert A_n\rvert\lvert B_n-J\rvert+\lvert J\rvert\lvert A_n-I\rvert
<\varepsilon
\]
因此 \(A_nB_n\to IJ\). 若 \(IJ<1\)，取 \(\varepsilon=\frac{1-IJ}{2}>0\)，则当 \(n\geqslant N\) 时
\[
A_nB_n<IJ+\varepsilon=\frac{IJ+1}{2}<1
\]
这与对所有 \(n\) 都有 \(A_nB_n\geqslant1\) 矛盾. 所以
\[
\int_{0}^{1}f(x)\,\mathrm{d}x\int_{0}^{1}\frac{\mathrm{d}x}{f(x)}=IJ\geqslant1
\]
这就证明了所要的不等式.
\end{enumerate}
\end{solution}
