\begin{problem}
\(\sqrt{\left( -a \right)^2}=?\)

\end{problem}

\begin{solution}
\[
\sqrt{\left( -a \right)^2}=
\begin{cases}
a & \text{（\(a\geqslant 0\)）},\\
-a & \text{（\(a<0\)）}.
\end{cases}
\]
\end{solution}

\begin{problem}
平行四边形 \(ABCD\) 的两邻边之比为 \(1:2\)，\(M\) 为大边 \(AB\) 的中点，连接 \(MD\)，\(MC\)，求证：\(\angle DMC=\frac{\pi}{2}\).

\end{problem}

\begin{solution}
\bitmapfigure[max width=6.0cm,max totalheight=3.8cm]{img/1977/shandong_b/q7_solution_fig1.png}

因为 \(AB:AD=2:1\)，\(AM=\frac{1}{2}AB\)，所以 \(AM=AD\). 所以 \(\angle ADM=\angle AMD\). 又因为 \(AB\parallel DC\)，且 \(A\)，\(M\)，\(B\) 三点共线，所以 \(\angle AMD=\angle MDC\). 因此 \(\angle MDC=\frac{1}{2}\angle ADC\).

又因为 \(M\) 为 \(AB\) 的中点，所以 \(MB=AM\). 而平行四边形 \(ABCD\) 中 \(AD=BC\)，又已知 \(AM=AD\)，所以 \(MB=BC\). 因此 \(\triangle MBC\) 为等腰三角形，\(\angle BMC=\angle MCB\). 又因为 \(MB\parallel DC\)，所以 \(\angle BMC=\angle MCD\). 因此 \(\angle BCD=\angle MCB+\angle MCD=2\angle MCD\)，即 \(\angle MCD=\frac{1}{2}\angle BCD\).

\[
\angle MDC+\angle MCD=\frac{1}{2}\left( \angle ADC+\angle BCD \right)=\frac{1}{2}\pi=\frac{\pi}{2}
\]

所以 \(\angle DMC=\frac{\pi}{2}\).
\end{solution}

\begin{problem}
\(\frac{\sin\alpha}{2}x^2+x+\frac{1}{2\cos\alpha}=0\)（\(0<\alpha<\frac{\pi}{2}\)）为关于 \(x\) 的二次方程. 当 \(\alpha\) 为何值时，方程有两个不相等的实数根？它的根是什么？

\end{problem}

\begin{solution}
因为方程有两个不相等的实数根，所以 \(b^2-4ac>0\)，即 \(1-\frac{\sin\alpha}{\cos\alpha}=1-\tan\alpha>0\)，亦即 \(\tan\alpha<1\). 因此 \(0<\alpha<\frac{\pi}{4}\) 时，方程有两个不相等的实数根.

\(x=\frac{-1\pm\sqrt{1-\tan\alpha}}{\sin\alpha}\)
\end{solution}

\begin{problem}
求极限：\(\lim\limits_{x\to 0}\frac{\ln(1+x)}{x}\).

\end{problem}

\begin{solution}
\[
\lim\limits_{x\to 0}\frac{\ln(1+x)}{x}
=\lim\limits_{x\to 0}\frac{[\ln(1+x)]'}{x'}
=\lim\limits_{x\to 0}\frac{1}{1+x}=1
\]
\end{solution}
\endgroup

\begin{problem}
用定积分求抛物线 \(y^2=4x\) 与 \(x^2=4y\) 所围成的曲边形面积.

\end{problem}

\begin{solution}
\begin{center}
\includegraphics[page=6,trim=8.3cm 7.8cm 3.9cm 15.2cm,clip,width=5.0cm]{source/普通高考/1977/1977山东B.pdf}
\end{center}

设曲边形面积为 \(S\). 解方程组
\[
\begin{cases}
y^2=4x,\\
x^2=4y.
\end{cases}
\]
得两交点的横坐标 \(x=0\) 与 \(x=4\).

\[
\begin{aligned}
S&=\int_0^4 2\sqrt{x}\,\mathrm{d}x-\int_0^4 \frac{1}{4}x^2\,\mathrm{d}x
=\left. \frac{4}{3}x^{\frac{3}{2}} \right|_0^4-\left. \frac{1}{12}x^3 \right|_0^4 \\
&=\frac{32}{3}-\frac{16}{3}=\frac{16}{3}.
\end{aligned}
\]
\end{solution}
\endgroup

\begin{problem}
试比较等底等高的正四棱锥和正圆锥的侧面积，看哪个的侧面积较小.

\end{problem}

\begin{solution}
设正四棱锥和圆锥的底面积为 \(S\)，高为 \(h\). 则正四棱锥底面边长为 \(\sqrt{S}\)，斜高为 \(\sqrt{\frac{S}{4}+h^2}\). 圆锥的底面半径 \(r=\sqrt{\frac{S}{\pi}}\)，母线 \(l=\sqrt{\frac{S}{\pi}+h^2}\).

\[
S_{\text{正四棱锥侧}}=4\cdot\frac{1}{2}\cdot\sqrt{S}\cdot\sqrt{\frac{S}{4}+h^2}=\sqrt{S^2+4Sh^2}
\]

\[
S_{\text{圆锥侧}}=\pi\cdot\sqrt{\frac{S}{\pi}}\cdot\sqrt{\frac{S}{\pi}+h^2}=\sqrt{S^2+\pi Sh^2}
\]

因为 \(\pi<4\)，所以 \(S_{\text{圆锥侧}}<S_{\text{正四棱锥侧}}\).
\end{solution}

\begin{problem}
一只船以 \(v\) 海里/小时的速度向正北航行，在 \(A\) 处望见灯塔在船的北 \(\alpha\) 度东，继续航行 \(t\) 小时后，在 \(B\) 处望见 \(S\) 在船的北 \(\beta\) 度东，求第二次望见灯塔 \(S\) 时船和灯塔的距离 \(BS\).

\end{problem}

\begin{solution}
\bitmapfigure[max width=4.2cm,max totalheight=4.6cm]{img/1977/shandong_b/q10_solution_fig1.png}

在 \(\triangle ABS\) 中，\(AB=vt\)，\(\angle A=\alpha\)，\(\angle S=\beta-\alpha\). 由正弦定理得 \(\frac{BS}{\sin\alpha}=\frac{vt}{\sin\left( \beta-\alpha \right)}\)，所以 \(BS=\frac{vt\sin\alpha}{\sin\left( \beta-\alpha \right)}\).
\end{solution}

\begin{problem}
某隧道的横断面如图所示，已知隧道横断面的周长为 \(23\) 米，试根据图所示尺寸，求当 \(x\) 和 \(y\) 为何值时，隧道的横断面的面积最大？最大面积是多少？（取 \(\pi\approx 3\)）（图尺寸单位：米）

\bitmapfigure[max width=4.8cm,max totalheight=6.0cm]{img/1977/shandong_b/q11_fig1.png}

\end{problem}

\begin{solution}
设半圆的半径为 \(x\) 米，隧道横断面面积为 \(S\) 平方米. 则
\[
y\approx\left( 23-3x-0.5\times 2-2x \right)\times\frac{1}{2}=\left( 11-2.5x \right)\text{米}
\]

\[
S\approx\frac{3x^2}{2}+2x\left( 11-2.5x \right)-\left( 2x-1 \right)\times 0.5=-3.5x^2+21x+0.5
\]

因为 \(-3.5<0\)，所以当 \(x=-\frac{21}{2\times\left( -3.5 \right)}=3\) 时，\(S\) 有最大值.

\[
S_{\max}=\frac{4\times\left( -3.5 \right)\times 0.5-21^2}{4\times\left( -3.5 \right)}=32\text{（米}^2\text{）}
\]

当 \(x=3\) 时，\(y=11-2.5\times 3=3.5\)（米）.

答：当半圆的半径 \(x\) 为 \(3\) 米，\(y\) 为 \(3.5\) 米时隧道的横断面面积最大，最大面积为 \(32\) 平方米.
\end{solution}
