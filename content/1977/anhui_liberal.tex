    \item \(\left(3\sqrt[3]{a^2}\cdot\sqrt b\right)\left(-6\sqrt a\cdot\sqrt[3]b\right)\div\left(-2\sqrt[6]a\cdot\sqrt[6]{b^5}\right)\)；
    \item \(\sqrt{(\lg x)^2-2\lg x+1}\).

\begin{solution}
\[
\begin{aligned}

\end{aligned}
\]

（2）原式 \(=\sqrt{(\lg x-1)^2}=|\lg x-1|\).
\end{solution}

\begin{problem}
求下列函数的定义域，并在数轴上把它们表示出来：

\begin{enumerate}
    \item \(y=\frac{\sqrt{3+2x}}{3x-1}\)；
    \item \(y=\arcsin\frac{x}{3}\).
\end{enumerate}

\end{problem}

\begin{solution}
（1）\(x\) 必须满足两条件 \circled{1} \(3+2x\geqslant0\)，\circled{2} \(3x-1\neq0\). 由 \circled{1} 知 \(x\geqslant-\frac{3}{2}\)，由 \circled{2} 知 \(x\neq\frac{1}{3}\)，故定义域为 \(\left[-\frac{3}{2},\frac{1}{3}\right)\) 和 \(\left(\frac{1}{3},+\infty\right)\).

（2）由 \(\sin y=\frac{x}{3}\) 知，\(\left|\frac{x}{3}\right|\leqslant1\)，即 \(-1\leqslant\frac{x}{3}\leqslant1\)，故定义域为 \([-3,3]\).
\end{solution}

\begin{problem}
求证：\(\frac{\sin\alpha+\sin\beta}{\sin(\alpha+\beta)}=\frac{\cos\frac{\alpha-\beta}{2}}{\cos\frac{\alpha+\beta}{2}}\).

\end{problem}

\begin{solution}
\[
\begin{aligned}
\text{左端}&=\frac{2\sin\frac{\alpha+\beta}{2}\cos\frac{\alpha-\beta}{2}}{2\sin\frac{\alpha+\beta}{2}\cos\frac{\alpha+\beta}{2}}\\
&=\frac{\cos\frac{\alpha-\beta}{2}}{\cos\frac{\alpha+\beta}{2}}=\text{右端}
\end{aligned}
\]
所以恒等式成立.
\end{solution}

\begin{problem}
解方程 \(\sin x+\cos x-1=0\).

\end{problem}

\begin{solution}
由 \(\sin x=1-\cos x\) 得 \(2\sin\frac{x}{2}\cos\frac{x}{2}=2\sin^2\frac{x}{2}\)，整理得 \(\sin\frac{x}{2}\left( \cos\frac{x}{2}-\sin\frac{x}{2} \right)=0\).

当 \circled{1} \(\sin\frac{x}{2}=0\) 时，则 \(\frac{x}{2}=k\pi\)，因此 \(x=2k\pi\).

当 \circled{2} \(\cos\frac{x}{2}-\sin\frac{x}{2}=0\) 时，\(\tan\frac{x}{2}=1\)，所以 \(\frac{x}{2}=k\pi+\frac{\pi}{4}\)，\(x=2k\pi+\frac{\pi}{2}\)（\(k=0\)，\(\pm1\)，\(\pm2\)，\(\cdots\)）.

故方程解为：\(x=2k\pi\) 和 \(x=2k\pi+\frac{\pi}{2}\)（\(k=0\)，\(\pm1\)，\(\pm2\)，\(\cdots\)）.
\end{solution}

\begin{problem}
求下列等式里的实数 \(x\) 和 \(y\) 的值：\((3x\i+2x)+(y\i-4)=(2y+5\i)-(3+x\i)\).

\end{problem}

\begin{solution}
原式为 \((2x-4)+(3x+y)\i=(2y-3)+(5-x)\i\)，所以 \(2x-4=2y-3\)，\(3x+y=5-x\).

解方程组：
\[
\begin{cases}
2x-2y=1 \qquad \text{\circled{1}},\\
4x+y=5 \qquad \text{\circled{2}}.
\end{cases}
\]
\(\text{\circled{1}}+2\times\text{\circled{2}}\)，得 \(10x=11\). \(x=\frac{11}{10}\). 代入 \circled{1}，得 \(y=\frac{3}{5}\).
\end{solution}

\begin{problem}
从山顶 \(D\) 测得地面上同一方向的两点 \(A\) 和 \(B\) 的俯角分别是 \(30^\circ\) 和 \(45^\circ\)，已知 \(AB=40\text{ 米}\)，求山高. （精确到 \(0.1\)）

\end{problem}

\begin{solution}
\bitmapfigure[max width=6cm]{img/1977/anhui_liberal/q8_fig1.png}

由题设知 \(\angle DAB=30^\circ\)，\(\angle DBC=45^\circ\)，所以 \(BC=DC\).

在 \(\triangle DAC\) 中，
\[
\begin{aligned}
\tan30^\circ&=\frac{DC}{AC}=\frac{DC}{40+DC}\\
\frac{\sqrt3}{3}&=\frac{DC}{40+DC},\ DC=\frac{\sqrt3}{3}(40+DC)\\
DC&=\frac{\sqrt3\cdot40}{3-\sqrt3}=\frac{40}{\sqrt3-1}=20\left( \sqrt3+1 \right)\approx54.6\text{ 米}
\end{aligned}
\]
\end{solution}

\begin{problem}
红旗大队粮食产量逐年增加，1973年产量为\(90\text{ 万斤}\)，连续三年平均每年比前一年增产 \(10\%\)，这个大队从1973年到1976年总共生产粮食多少万斤？（精确到 \(0.1\)）

\end{problem}

\begin{solution}
设总共产粮为 \(S\text{ 万斤} \).
\[
\begin{aligned}
S&=90\left[1+\left( 1+10\% \right)+\left( 1+10\% \right)^2+\left( 1+10\% \right)^3\right]\\
&\approx90\left[1+1.1+1.21+1.331\right]\\
&\approx90\times4.641=417.69\text{ 万斤}
\end{aligned}
\]
\end{solution}

\begin{problem}
求椭圆 \(25x^2+9y^2=100\) 的长轴和短轴的长，焦点坐标，并且画出它的图象.

\end{problem}

\begin{solution}
原方程为 \(\frac{x^2}{4}+\frac{y^2}{\frac{100}{9}}=1\).

短半轴 \(a=2\)，故短轴 \(2a=4\).

长半轴 \(b=\frac{10}{3}\)，故长轴 \(2b=\frac{20}{3}\).

半焦距 \(C=\sqrt{b^2-a^2}=\frac{8}{3}\).

故焦点坐标为 \(\left(0,\frac{8}{3}\right)\) 和 \(\left(0,-\frac{8}{3}\right)\).

\end{solution}
