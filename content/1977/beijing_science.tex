\chapter{1977年北京卷（理）}

\begin{problem}
解方程：\(\sqrt{x-1}=3-x\).

\end{problem}

\begin{problem}
计算：\(2^{-\frac{1}{2}}+\frac{2^0}{\sqrt{2}}+\frac{1}{\sqrt{2}-1}\).

\end{problem}

\begin{problem}
已知 \(\lg 2=0.3010\)，\(\lg 3=0.4771\)，求 \(\lg\sqrt{45}\).

\end{problem}

\begin{problem}
证明：\(\left(1+\tan\alpha\right)^2=\frac{1+\sin 2\alpha}{\cos^2\alpha}\).

\end{problem}

\begin{problem}
求过两直线 \(x+y-7=0\) 和 \(3x-y-1=0\) 的交点且过 \(\left(1,1\right)\) 点的直线方程.

\end{problem}

\begin{problem}
某工厂今年七月份的产值为 \(100\text{ 万元}\)，以后每月产值比上月增加 \(20\%\)，问今年七月份到十月份总产值是多少?

\end{problem}

\begin{problem}
已知二次函数 \(y=x^2-6x+5\).

\begin{enumerate}
    \item 求出它的图象的顶点坐标和对称轴方程；
    \item 画出它的图象；
    \item 分别求出它的图象和 \(x\) 轴、\(y\) 轴的交点坐标.
\end{enumerate}

\end{problem}

\begin{problem}
一只船以 \(20\text{ 海里/小时}\) 的速度向正东航行，起初船在 \(A\) 处看见一灯塔 \(B\) 在船的北 \(45^\circ\) 东方向，一小时后船在 \(C\) 处看见这个灯塔在船的北 \(15^\circ\) 东方向，求这时船和灯塔的距离 \(CB\).

\end{problem}

\begin{problem}
一个圆内接三角形 \(ABC\)，\(\angle A\) 的平分线交 \(BC\) 于 \(D\)，交外接圆于 \(E\)，求证：\(AD\cdot AE=AC\cdot AB\).

\end{problem}

\begin{problem}
当 \(m\) 取哪些值时，直线 \(y=x+m\) 与椭圆 \(\frac{x^2}{16}+\frac{y^2}{9}=1\) 有一个交点? 有两个交点? 没有交点? 当它们有一个交点时，画出它的图象.
\end{problem}

\begin{problem}
求函数 \(f(x)=\begin{cases}x^2\sin\frac{\pi}{x}, & x\ne0,\\0, & x=0\end{cases}\) 的导数.

\end{problem}

\begin{problem}
求椭圆 \(\frac{x^2}{a^2}+\frac{y^2}{b^2}=1\) 绕 \(x\) 轴旋转而成的旋转体体积.

\end{problem}

\begin{problem}
\begin{enumerate}
    \item 试用 \(\varepsilon-\delta\) 语言叙述“函数 \(f(x)\) 在点 \(x=x_0\) 处连续”的定义.
    \item 试证明：若 \(f(x)\) 在点 \(x=x_0\) 处连续，且 \(f(x_0)>0\)，则存在一个 \(x_0\) 的 \(\left(x_0-\delta,x_0+\delta\right)\)，在这个邻域内，处处有 \(f(x)>0\).
\end{enumerate}
\end{problem}
