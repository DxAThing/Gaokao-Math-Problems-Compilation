\chapter{1977年江苏南通市初试卷}

\section{解答题}

\begin{problem}
化简，计算：

\begin{enumerate}
\item \(\dfrac{-2mn}{m-3n}\sqrt{\dfrac{\left(m+3n\right)^2}{2mn}}-6\)（\(0<m<3n\)）；
\item \(\left(\dfrac14\right)^{-1.5}\cdot\left[\left(\dfrac7{11}\right)^0\right]^{-3}-\left(\dfrac35\right)^{-2}\div\left(2\dfrac79\right)^{\frac12}\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由 \(0<m<3n\) 且 \(m>0\)，知 \(n>0\)，从而 \(2mn>0\)，并且 \(m+3n>0\). 因此
\(\sqrt{\dfrac{\left(m+3n\right)^2}{2mn}}=\dfrac{m+3n}{\sqrt{2mn}}\).
又因为 \(m-3n<0\)，所以原式
\[
=\dfrac{-2mn}{m-3n}\cdot\dfrac{m+3n}{\sqrt{2mn}}-6
=\dfrac{\left(m+3n\right)\sqrt{2mn}}{3n-m}-6
\]
\item \(1.5=\dfrac32\)，且 \(2\dfrac79=\dfrac{25}{9}\)，所以
\[
\left(\dfrac14\right)^{-1.5}\left[\left(\dfrac7{11}\right)^0\right]^{-3}-\left(\dfrac35\right)^{-2}\div\left(2\dfrac79\right)^{\frac12}
=4^{\frac32}\cdot1^{-3}-\left(\dfrac53\right)^2\div\dfrac53
=8-\dfrac{25}{9}\cdot\dfrac35
=\dfrac{19}{3}
\]
\end{enumerate}
\end{solution}

\begin{problem}
在实数范围内分解因式：

\begin{enumerate}
\item \(a^4-10a^2+16\)；
\item \(x^2-4xy+4y^2-3x+6y+2\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 令 \(u=a^2\)，则
\[
a^4-10a^2+16=u^2-10u+16=\left(u-2\right)\left(u-8\right)
\]
代回 \(u=a^2\)，并在实数范围内继续分解，得
\[
a^4-10a^2+16=\left(a^2-2\right)\left(a^2-8\right)=\left(a-\sqrt2\right)\left(a+\sqrt2\right)\left(a-2\sqrt2\right)\left(a+2\sqrt2\right)
\]
\item 注意到前两项可合并为完全平方，且后三项也只含 \(x-2y\)，所以
\[
x^2-4xy+4y^2-3x+6y+2=\left(x-2y\right)^2-3\left(x-2y\right)+2=\left(x-2y-1\right)\left(x-2y-2\right)
\]
\end{enumerate}
\end{solution}

\begin{problem}
求 \(x\)：

\begin{enumerate}
\item \(\dfrac{2x+1}{x-3}-\dfrac{5x-11}{x^2-4x+3}=\dfrac3{x-1}\).
\item \(\sqrt{2x-4}-\sqrt{x+5}=1\).
\item \(\log_{x-1}\left(2x^2-3x-1\right)=2\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 原方程有意义的条件是 \(x\ne1\) 且 \(x\ne3\)，因为 \(x^2-4x+3=\left(x-1\right)\left(x-3\right)\). 在此条件下，方程两边同乘 \(\left(x-1\right)\left(x-3\right)\)，得
\(\left(2x+1\right)\left(x-1\right)-\left(5x-11\right)=3\left(x-3\right)\).
化简得 \(2x^2-9x+19=0\)，其判别式为 \(\Delta=\left(-9\right)^2-4\times2\times19=-71<0\). 因此原方程无实数解.
\item 由根式有意义，须 \(x\geqslant2\). 移项得 \(\sqrt{2x-4}=1+\sqrt{x+5}\).
两边平方，得 \(2x-4=x+6+2\sqrt{x+5}\)，即 \(x-10=2\sqrt{x+5}\)，所以还须 \(x\geqslant10\). 再次平方，得 \(\left(x-10\right)^2=4\left(x+5\right)\).
即 \(x^2-24x+80=0\)，所以 \(x=4\) 或 \(x=20\). 由 \(x\geqslant10\) 取 \(x=20\)，代回原方程检验：\(\sqrt{2\times20-4}-\sqrt{20+5}=6-5=1\)，故原方程的解为 \(x=20\).
\item 对数的底数须满足 \(x-1>0\) 且 \(x-1\ne1\)，即 \(x>1\) 且 \(x\ne2\). 真数还须满足 \(2x^2-3x-1>0\). 由于 \(2x^2-3x-1=0\) 的两根为 \(\frac{3-\sqrt {17}}4\) 和 \(\frac{3+\sqrt {17}}4\)，所以在 \(x>1\) 时，真数为正等价于 \(x>\frac{3+\sqrt {17}}4\). 在这些条件下，原方程等价于
\(2x^2-3x-1=\left(x-1\right)^2\).
整理得 \(x^2-x-2=0\)，所以 \(x=2\) 或 \(x=-1\). 其中 \(x=2\) 使底数为 \(1\)，\(x=-1\) 使底数为负数，均不满足定义域条件，因此原方程无实数解.
\end{enumerate}
\end{solution}

\begin{problem}
求下列函数的定义域：

\begin{enumerate}
\item \(y=\dfrac1{1-\sqrt{x-1}}\)；
\item \(y=\lg\left(\tan x\right)\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 根式有意义要求 \(x-1\geqslant0\)，分母不为零要求 \(1-\sqrt{x-1}\ne0\). 因此定义域为 \(x\geqslant1\) 且 \(x\ne2\).
\item 对数真数须为正，所以要求 \(\tan x>0\)；同时这已保证 \(\tan x\) 有定义. 正切函数在每个周期内的正值区间为 \(\left(k\pi,k\pi+\frac\pi2\right)\)，其中 \(k\in\Z\)，故定义域为
\[
\bigcup_{k\in\Z}\left(k\pi,k\pi+\dfrac\pi2\right)
\]
\end{enumerate}
\end{solution}

\begin{problem}
不查表：

\begin{enumerate}
\item 求 \(4^{1-\log_2\dfrac25+\log_2\sqrt2}\).
\item 解三角形：已知 \(\triangle ABC\) 中，\(\angle B=120^\circ\)，\(\angle C=15^\circ\)，\(b=100\)，求 \(c\)（可用根式表示结果）.
\item 将直角坐标方程 \(x^2-y^2=a^2\) 化成极坐标方程.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 因为 \(\log_2\dfrac25=1-\log_2 5\)，
并且 \(\log_2\sqrt2=\dfrac12\)，
所以指数为 \(1-\left(1-\log_2 5\right)+\frac12=\frac12+\log_2 5\). 因而
\[
4^{1-\log_2\frac25+\log_2\sqrt2}=\left(2^2\right)^{\frac12+\log_2 5}=2^{1+2\log_2 5}=2\cdot5^2=50
\]
\item 由三角形内角和可得 \(\angle A=180^\circ-120^\circ-15^\circ=45^\circ\). 由正弦定理可得 \(\dfrac{c}{\sin15^\circ}=\dfrac{b}{\sin120^\circ}\)，所以
\[
c=100\cdot\dfrac{\sin15^\circ}{\sin120^\circ}=100\cdot\dfrac{\frac{\sqrt6-\sqrt2}{4}}{\frac{\sqrt3}{2}}=\dfrac{50\left(\sqrt6-\sqrt2\right)}{\sqrt3}=\dfrac{50}{3}\left(3\sqrt2-\sqrt6\right)
\]
\item 令极坐标为 \(\left(r,\theta\right)\)，则 \(x=r\cos\theta\)，\(y=r\sin\theta\). 代入直角坐标方程，得 \(r^2\cos^2\theta-r^2\sin^2\theta=a^2\)，即极坐标方程为 \(r^2\cos2\theta=a^2\).
\end{enumerate}
\end{solution}

\begin{problem}
\(\triangle ABC\) 的 \(\angle A\) 的平分线交 \(BC\) 于 \(D\)，延长 \(AD\) 交三角形的外接圆于 \(E\)，求证：\(AB\cdot AC=AD\cdot AE\).
\end{problem}

\begin{solution}
由角平分线定理可得 \(\dfrac{BD}{DC}=\dfrac{AB}{AC}\).
设 \(AB=m\)，\(AC=n\)，\(BD=p\)，\(DC=q\). 由角平分线定理，\(np=mq\). 斯特瓦特定理给出
\[
n^2p+m^2q=\left(p+q\right)\left(AD^2+pq\right)
\]
由 \(np=mq\)，左端可化为 \(mn\left(p+q\right)\)，所以 \(mn\left(p+q\right)=\left(p+q\right)\left(AD^2+pq\right)\). 因 \(p+q>0\)，约去 \(p+q\)，得到 \(AD^2=mn-pq=AB\cdot AC-BD\cdot DC\).
又因为 \(A\)、\(B\)、\(C\)、\(E\) 四点共圆，且 \(D\) 在弦 \(BC\) 与弦 \(AE\) 的交点上，由相交弦定理可得 \(AD\cdot DE=BD\cdot DC\).
由于 \(D\) 在线段 \(AE\) 上，\(AE=AD+DE\). 因此
\[
AB\cdot AC=AD^2+BD\cdot DC=AD^2+AD\cdot DE=AD\left(AD+DE\right)=AD\cdot AE
\]
所以 \(AB\cdot AC=AD\cdot AE\).
\end{solution}

\begin{problem}
已知正三棱锥 \(P-ABC\) 的底面边长是 \(6\mathrm{~cm}\)，侧面与底面所成的角是 \(60^\circ\)，求它的体积（结果可用根式表示）.
\end{problem}

\begin{solution}
设正三角形底面 \(ABC\) 的中心为 \(O\)，\(PO\) 为棱锥的高，设 \(M\) 为 \(BC\) 的中点. 因为棱锥为正三棱锥，所以 \(PO\perp\) 平面 \(ABC\)，且 \(PM\perp BC\)，\(OM\perp BC\). 因此侧面与底面所成的二面角就是 \(\angle PMO=60^\circ\).

底面为边长 \(6\mathrm{~cm}\) 的正三角形，所以 \(OM=\dfrac{6\sqrt3}{6}=\sqrt3\mathrm{~cm}\).
底面积为
\(S_{\triangle ABC}=\dfrac{6^2\sqrt3}{4}=9\sqrt3\mathrm{~cm}^2\).
在直角三角形 \(POM\) 中，
\(PO=OM\tan60^\circ=\sqrt3\cdot\sqrt3=3\mathrm{~cm}\).
所以棱锥的体积为
\[
V=\dfrac13S_{\triangle ABC}\cdot PO=\dfrac13\cdot9\sqrt3\cdot3=9\sqrt3\mathrm{~cm}^3
\]
\end{solution}

\begin{problem}
已知 \(a\)，\(b\)，\(c\) 为三角形的三条边，求证：\(b^2x^2+\left(b^2+c^2-a^2\right)x+c^2=0\) 无实数根.
\end{problem}

\begin{solution}
因为 \(a\)、\(b\)、\(c\) 是三角形的三条边，所以 \(\left|b-c\right|<a<b+c\). 三边均为正数，平方后得 \(\left(b-c\right)^2<a^2<\left(b+c\right)^2\). 移项可得 \(-2bc<b^2+c^2-a^2<2bc\)，因此 \(\left(b^2+c^2-a^2\right)^2<4b^2c^2\). 原方程的判别式为 \(\Delta=\left(b^2+c^2-a^2\right)^2-4b^2c^2<0\).
又因二次项系数 \(b^2>0\)，故原方程无实数根.
\end{solution}

\begin{problem}
方程 \(x^2-\left(\tan\theta+\cot\theta\right)x+1=0\) 有一根为 \(2+\sqrt3\)，求 \(\sin 2\theta\).
\end{problem}

\begin{solution}
设方程的另一根为 \(x_2\). 由韦达定理可得 \(\left(2+\sqrt3\right)x_2=1\)，所以 \(x_2=2-\sqrt3\). 再由两根之和，\(\tan\theta+\cot\theta=\left(2+\sqrt3\right)+\left(2-\sqrt3\right)=4\).
另一方面，由题设中的 \(\tan\theta\) 与 \(\cot\theta\) 有意义，知 \(\sin\theta\cos\theta\ne0\)，因此
\[
\tan\theta+\cot\theta=\dfrac{\sin^2\theta+\cos^2\theta}{\sin\theta\cos\theta}=\dfrac{1}{\sin\theta\cos\theta}=\dfrac{2}{\sin2\theta}
\]
因此 \(\dfrac{2}{\sin2\theta}=4\)，从而 \(\sin2\theta=\dfrac12\).
\end{solution}

\begin{problem}
汽车与自行车分别从相距 \(75\mathrm{~km}\) 的 \(A\)，\(B\) 两地同时出发（\(A\) 地在 \(B\) 地的正东方向），汽车从 \(A\) 地向正西方向行驶，自行车从 \(B\) 地向正南方向行驶，已知汽车和自行车的速度分别为 \(30\mathrm{~km/h}\) 和 \(15\mathrm{~km/h}\)，问出发后多少小时两车相距最短？
\end{problem}

\begin{solution}
取 \(B\) 为坐标原点，正东方向为 \(x\) 轴正向，正北方向为 \(y\) 轴正向，则 \(A=\left(75,0\right)\). 设出发 \(t\mathrm{~h}\) 后汽车和自行车的位置分别为 \(C\)、\(D\)，其中 \(t\geqslant0\). 于是汽车位置为 \(C=\left(75-30t,0\right)\)，自行车位置为 \(D=\left(0,-15t\right)\).
两车距离的平方为
\[
CD^2=\left(75-30t\right)^2+\left(15t\right)^2=1125t^2-4500t+5625=1125\left(t-2\right)^2+1125
\]
当 \(t=2\) 时，上式取得最小值，所以出发后 \(2\mathrm{~h}\) 两车相距最短.
\end{solution}

\begin{problem}
已知等差数列有 \(10\) 项，偶数项之和为 \(55\)，奇数项之和为 \(45\)，求此数列的第五项.
\end{problem}

\begin{solution}
设等差数列的首项为 \(a_1\)，公差为 \(d\). 奇数项之和为
\[
a_1+a_3+a_5+a_7+a_9=5a_1+\left(0+2+4+6+8\right)d=5a_1+20d=45
\]
偶数项之和为
\[
a_2+a_4+a_6+a_8+a_{10}=5a_1+\left(1+3+5+7+9\right)d=5a_1+25d=55
\]
两式相减得 \(5d=10\)，即 \(d=2\). 代入奇数项之和得 \(5a_1+40=45\)，所以 \(a_1=1\). 因此第五项为 \(a_5=a_1+4d=1+4\times2=9\).
\end{solution}

\begin{problem}
一动点到直线 \(x=8\) 的距离是它到点 \(A\left(2,0\right)\) 的距离的 \(2\) 倍.

\begin{enumerate}
\item 求这动点的轨迹方程；
\item 判断这轨迹是何曲线；
\item 求经过这曲线上一点 \(B\left(2,3\right)\) 的切线方程.
\end{enumerate}
\end{problem}

\begin{solution}
设动点为 \(P\left(x,y\right)\).
\begin{enumerate}
\item 由题意，点 \(P\) 到直线 \(x=8\) 的距离为 \(\left|x-8\right|\)，到点 \(A\left(2,0\right)\) 的距离为 \(\sqrt{\left(x-2\right)^2+y^2}\)，所以 \(\left|x-8\right|=2\sqrt{\left(x-2\right)^2+y^2}\).
两边均非负，平方得 \(\left(x-8\right)^2=4\left(\left(x-2\right)^2+y^2\right)\).
化简为 \(3x^2+4y^2=48\)，即轨迹方程为
\[
\dfrac{x^2}{16}+\dfrac{y^2}{12}=1
\]
\item 上式是椭圆的标准方程，且半长轴为 \(4\)，半短轴为 \(2\sqrt3\)，所以这条轨迹是以原点为中心、长轴在 \(x\) 轴上的椭圆.
\item 先检验 \(B\left(2,3\right)\) 在椭圆上：\(\frac{2^2}{16}+\frac{3^2}{12}=1\). 对方程 \(3x^2+4y^2=48\) 隐式求导，得 \(6x+8yy'=0\). 在 \(B\left(2,3\right)\) 处，切线斜率为 \(y'=-\frac12\)，所以切线方程为 \(y-3=-\dfrac12\left(x-2\right)\)，
即 \(x+2y=8\).
\end{enumerate}
\end{solution}
