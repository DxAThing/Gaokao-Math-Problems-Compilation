\begin{problem}
化简下式：\( \left( 1-c^2 \right)^{- \frac{1}{2}}-\left\{ \left[ \left( 1+c \right)\left( 1-c \right) \right]^{\frac{1}{2}}+c^2\left( \left( 1+c \right)\left( 1-c \right) \right)^{- \frac{1}{2}} \right\} \).

\end{problem}

\begin{solution}
\begin{align*}
\text{原式}
&=\left( 1-c^2 \right)^{- \frac{1}{2}}-\left\{ \left( 1-c^2 \right)^{\frac{1}{2}}+c^2\left( 1-c^2 \right)^{- \frac{1}{2}} \right\} \\
&=\left( 1-c^2 \right)^{- \frac{1}{2}}-\left( 1-c^2 \right)^{\frac{1}{2}}-c^2\left( 1-c^2 \right)^{- \frac{1}{2}} \\
&=\left( 1-c^2 \right)^{- \frac{1}{2}}\left[ 1-\left( 1-c^2 \right)-c^2 \right]=0
\end{align*}
\end{solution}

\begin{problem}
求 \( \frac{\lg 3+\lg 2}{\frac{1}{4}\lg 16+\sqrt{\left( \lg 1-\lg 10 \right)^2}+\frac{1}{2}\lg 0.09} \) 的值.

\end{problem}

\begin{solution}
\begin{align*}
\text{原式}
&=\frac{\lg 3+\lg 2}{\lg 2+\sqrt{\left( 0-1 \right)^2}+\lg 0.3}
=\frac{\lg 3+\lg 2}{\lg 2+1+\lg 3-\lg 10} \\
&=\frac{\lg 3+\lg 2}{\lg 2+1+\lg 3-1}=1
\end{align*}
\end{solution}

\begin{problem}
求等式 \( 5^{x^2-2x}=125 \) 中 \( x \) 的值.

\end{problem}

\begin{solution}
由 \(5^{x^2-2x}=125=5^3\) 得 \(x^2-2x=3\)，因此 \(\left( x-3 \right)\left( x+1 \right)=0\)，所以 \(x_1=3\)，\(x_2=-1\).
\end{solution}

\begin{problem}
已知 \( a \)，\( b \)，\( c \) 分别表示 \( \triangle ABC \) 的角 \( A \)，\( B \)，\( C \) 对边的长，求证：\( a\left( \sin B-\sin C \right)+b\left( \sin C-\sin A \right)+c\left( \sin A-\sin B \right)=0 \).

\end{problem}

\begin{solution}
由正弦定理，\(\frac{a}{\sin A}=\frac{b}{\sin B}=\frac{c}{\sin C}\).
可知 \( a\sin B=b\sin A \)，\( a\sin C=c\sin A \)，\( b\sin C=c\sin B \).
因此
\begin{align*}
&a\left( \sin B-\sin C \right)+b\left( \sin C-\sin A \right)+c\left( \sin A-\sin B \right) \\
=\;&a\sin B-a\sin C+b\sin C-b\sin A+c\sin A-c\sin B \\
=\;&b\sin A-c\sin A+c\sin B-b\sin A+c\sin A-c\sin B=0
\end{align*}
\end{solution}

\begin{problem}
如图，已知正方形 \( ABCD \) 的边 \( CD \) 上任意一点 \( E \)，延长 \( BC \) 到 \( F \)，使 \( CF=CE \)，设 \( BE \) 与 \( DF \) 相交于 \( G \)，求证 \( BG \perp DF \).

\bitmapfigure[max width=5.2cm,max totalheight=4.6cm]{img/1977/henan/q5_fig1.png}

\end{problem}

\begin{solution}
在 \( \mathrm{Rt}\triangle BCE \) 和 \( \mathrm{Rt}\triangle DCF \) 中，因为 \( BC=CD \)（正方形的边长相等），\( CE=CF \)（已知），所以 \( \triangle BCE \cong \triangle DCF \).
所以 \( \angle GBC=\angle FDC \)（对应角相等）.
又因为 \( \angle EBC+\angle BEC=90^\circ \)，而 \( \angle BEC=\angle DEG \)（对顶角相等），所以 \( \angle FDC+\angle DEG=90^\circ \).
所以 \( \angle DGB=90^\circ \). 即 \( BG \perp DF \).
\end{solution}

\begin{problem}
求证：\( \tan^2\alpha-\cot^2\alpha=-\frac{2\sin 4\alpha}{\sin^3 2\alpha} \).

\end{problem}

\begin{solution}

\begin{align*}
\text{左边}
=\;&\tan^2\alpha-\cot^2\alpha=\frac{\sin^4\alpha-\cos^4\alpha}{\sin^2\alpha\cos^2\alpha} \\
=\;&\frac{\sin^2\alpha-\cos^2\alpha}{\sin^2\alpha\cos^2\alpha} \\
=\;&-\frac{\cos 2\alpha}{\frac{1}{4}\sin^2 2\alpha} \\
=\;&-\frac{4\cos 2\alpha\cdot\sin 2\alpha}{\sin^2 2\alpha\cdot\sin 2\alpha}=-\frac{2\sin 4\alpha}{\sin^3 2\alpha}=\text{右边}
\end{align*}
所以 \( \tan^2\alpha-\cot^2\alpha=-\frac{2\sin 4\alpha}{\sin^3 2\alpha} \).
\end{solution}

\begin{problem}
已知长方体的对角线长为 \( L \)，它与底面所成的角为 \( \alpha \)，底面两条对角线的夹角为 \( \beta \)，求长方体的体积.

\bitmapfigure[max width=5.8cm,max totalheight=4.8cm]{img/1977/henan/q7_fig1.png}

\end{problem}

\begin{solution}
\( AA_1=L\sin\alpha \)，\( AC=L\cos\alpha \).
因此底面矩形 \( ABCD \) 的面积为 \(S=\frac{1}{2}AC^2\sin\beta=\frac{1}{2}L^2\cos^2\alpha\sin\beta\).
因此长方体的体积
\[
V=AA_1\times S=L\sin\alpha\cdot\frac{1}{2}L^2\cos^2\alpha\sin\beta=\frac{1}{2}L^3\cos^2\alpha\sin\alpha\sin\beta
\]
\end{solution}

\begin{problem}
某工厂科研小组，对一项生产工艺过程总结出产量指标函数和消耗函数为：\( f_1(x)=a_1x^2+\frac{1}{2}x+c \) 和 \( f_2(x)=a_2x^2+bx+\frac{5}{4} \). 且知 \( f_1(-1)=f_2(-1)=f_1(3)=f_2(3)=2 \).

\begin{enumerate}
\item 分别求出产量指标函数 \( f_1(x) \) 和消耗函数 \( f_2(x) \) 的具体表达式；
\item 问因素 \( x \) 取何值时，\( f_1(x) \) 和 \( f_2(x) \) 有最大值或最小值？最大值或最小值各是多少？
\item 画出所求出函数的略图.
\end{enumerate}

\end{problem}

\begin{solution}
（1）对指标函数 \( f_1(x) \) 来说，由 \( f_1(-1)=f_1(3)=2 \)，可得
\[
\begin{cases}
a_1-\frac{1}{2}+c=2, \\
9a_1+\frac{3}{2}+c=2
\end{cases}
\]
解这个方程组，得 \( a_1=-\frac{1}{4} \)，\( c=\frac{11}{4} \). 所以 \(f_1(x)=-\frac{1}{4}x^2+\frac{1}{2}x+\frac{11}{4}\).

对消耗函数 \( f_2(x) \) 来说，由 \( f_2(-1)=f_2(3)=2 \)，可得
\[
\begin{cases}
a_2-b+\frac{5}{4}=2, \\
9a_2+3b+\frac{5}{4}=2
\end{cases}
\]
解这个方程组，得 \( a_2=\frac{1}{4} \)，\( b=-\frac{1}{2} \). 所以 \(f_2(x)=\frac{1}{4}x^2-\frac{1}{2}x+\frac{5}{4}\).

（2）因为
\(f_1(x)=-\frac{1}{4}x^2+\frac{1}{2}x+\frac{11}{4}=-\frac{1}{4}\left( x-1 \right)^2+3\)，
\(f_2(x)=\frac{1}{4}x^2-\frac{1}{2}x+\frac{5}{4}=\frac{1}{4}\left( x-1 \right)^2+1\)，
所以指标函数 \( f_1(x) \) 在 \( x=1 \) 时有极大值 \( 3 \)，消耗函数 \( f_2(x) \) 在 \( x=1 \) 时有极小值 \( 1 \).

（3）略图如下：

\end{solution}

\begin{problem}
已知过点 \( P\left( 0,3\sqrt{2} \right) \) 且斜率为 \( k \) 的直线与圆心在原点，半径为 \( 3 \) 的圆相交于 \( M \)，\( N \) 两点.

\begin{enumerate}
\item 求 \( M \)，\( N \) 的坐标；
\setcounter{enumi}{2}
\item 问当 \( M \)，\( N \) 两点重合时，\( k \) 为何值？过 \( P \) 点的直线与圆的位置关系如何？这样的直线有几条？它们的夹角是多大？
\end{enumerate}

\end{problem}

\begin{solution}
（1）过点 \( P\left( 0,3\sqrt{2} \right) \)，且斜率为 \( k \) 的直线方程是 \(y-3\sqrt{2}=kx\).
圆心在原点，半径为 \( 3 \) 的圆的方程是 \(x^2+y^2=9\).
它们的交点坐标 \( \left( x,y \right) \) 应适合方程组
\[
\begin{cases}
y-3\sqrt{2}=kx, \\
x^2+y^2=9
\end{cases}
\]
解此方程组，得
\[
\begin{cases}
x=\frac{-3\sqrt{2}k\pm3\sqrt{k^2-1}}{1+k^2}, \\
y=\frac{3\sqrt{2}\pm3k\sqrt{k^2-1}}{1+k^2}
\end{cases}
\]
即交点 \( M \)，\( N \) 的坐标是
\[
\left( \frac{-3\sqrt{2}k+3\sqrt{k^2-1}}{1+k^2},\frac{3\sqrt{2}+3k\sqrt{k^2-1}}{1+k^2} \right)
\]
\[
\left( \frac{-3\sqrt{2}k-3\sqrt{k^2-1}}{1+k^2},\frac{3\sqrt{2}-3k\sqrt{k^2-1}}{1+k^2} \right)
\]

（2）当 \( M \)，\( N \) 两点重合时，这时 \( k^2-1=0 \)，\( k=\pm1 \).
这时过 \( P \) 点的直线和圆相切. 这样的直线有两条：\( y=x+3\sqrt{2} \) 和 \( y=-x+3\sqrt{2} \).
这两条与圆相切的直线的斜率互为负倒数，表示这两条直线垂直，因此这两条切线的夹角为 \( 90^\circ \).
\end{solution}

\begin{problem}
某电管所为实现农业现代化，加强电力网的建设，沿着一条通往农村的新公路栽电杆. 已知一辆汽车每次从电管所运 \(3\text{ 根}\)电线杆，相邻两根电线杆相距 \(50\text{ 米}\)，汽车往返的总行程是 \(35.5\text{ 公里}\)，最后一根电线杆与电管所相距 \(2450\text{ 米}\).

\begin{enumerate}
\item 第一根电线杆与电管所相距多少？
\item 共栽了多少根电线杆？
\end{enumerate}

\end{problem}

\begin{solution}
设第一根电线杆离电管站为 \(x\text{ 米}\)，共运电线杆 \(n\text{ 根}\)，汽车共走了 \( \frac{n}{3}\text{ 趟} \). 第一趟走的路程是 \(2\left( x+100 \right)\text{ 米}\)，以后每一趟都比前一趟多走 \(300\text{ 米}\)，最后一趟所走的路程是 \(2\times2450\text{ 米}\).
因此，根据题意，得
\[
\begin{cases}
2450=x+\left( n-1 \right)50, \\
35500=\frac{n}{3}\cdot\frac{2\left( x+100 \right)+2\times2450}{2}
\end{cases}
\]
解这个方程组，得 \( n_1=30 \)，\( x_1=1000 \)；\( n_2=71 \)，\( x_2=-1050 \)（不合题意，舍去）.

（1）第一根电线杆与电管所相距 \(1000\text{ 米}\).

（2）共栽了 \(30\text{ 根}\)电线杆.
\end{solution}


\begin{problem}
已知 \( x+x^{-1}=2\cos\theta \)，求证 \( x^n+x^{-n}=2\cos n\theta \).

\end{problem}

\begin{solution}
由 \( x+x^{-1}=2\cos\theta \) 可得 \(x^2-2\cos\theta x+1=0\).
所以 \( x=\cos\theta\pm\i\sin\theta \).
因此
\begin{align*}
x^n+x^{-n}
=\;&\left( \cos\theta\pm\i\sin\theta \right)^n+\left( \cos\theta\pm\i\sin\theta \right)^{-n} \\
=\;&\cos n\theta\pm\i\sin n\theta+\cos\left( -n\theta \right)\pm\i\sin\left( -n\theta \right) \\
=\;&\cos n\theta\pm\i\sin n\theta+\cos n\theta\mp\i\sin n\theta=2\cos n\theta
\end{align*}
\end{solution}

\begin{problem}
设 \( y=\ln\left( x+\sqrt{x^2-a^2} \right) \)，求 \( y \) 的导数 \( y' \).

\end{problem}

\begin{solution}
由求导法则，\( y'=\frac{1+\frac{2x}{2\sqrt{x^2-a^2}}}{x+\sqrt{x^2-a^2}}=\frac{1}{\sqrt{x^2-a^2}} \)，且 \(\frac{1}{\sqrt{x^2-a^2}}=\frac{\sqrt{x^2-a^2}}{x^2-a^2}\).
\end{solution}

\begin{problem}
求 \( \int_0^{2p}\sqrt{2px}\,\mathrm{d}x \) 的值.

\end{problem}

\begin{solution}
\begin{align*}
\int_0^{2p}\sqrt{2px}\,\mathrm{d}x
&=\frac{1}{2p}\int_0^{2p}\sqrt{2px}\,\mathrm{d}\left( 2px \right) \\
&=\frac{1}{3p}\left( 2px \right)^{\frac{3}{2}}\bigg|_0^{2p}=\frac{8}{3}p^2
\end{align*}
\end{solution}
