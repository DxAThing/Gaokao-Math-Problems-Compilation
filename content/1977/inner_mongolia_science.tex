\begin{problem}
化简：\(\sqrt{4-12a+9a^2}\).
\end{problem}

\begin{solution}

\[
\sqrt{4-12a+9a^2}=\sqrt{\left(2-3a\right)^2}
=
\begin{cases}
2-3a, & a\le\frac{2}{3}\text{时},\\
3a-2, & a>\frac{2}{3}\text{时}.
\end{cases}
\]

\end{solution}

\begin{problem}
计算：\(\frac{(-27)^3\sqrt{\left(\frac{1}{16}\right)^{-\frac{1}{2}}}}{(-9)^0\left(\frac{3}{2}\right)^7\left(\frac{1}{2}\right)^{-5}}\).
\end{problem}

\begin{solution}
\[
\frac{(-27)^3\sqrt{\left(\frac{1}{16}\right)^{-\frac{1}{2}}}}{(-9)^0\left(\frac{3}{2}\right)^7\left(\frac{1}{2}\right)^{-5}}
=\frac{-3^9\cdot2}{1\cdot3^7\cdot2^{-7}\cdot2^5}
=\frac{-3^9\cdot2}{3^7\cdot2^{-2}}
=-3^2\cdot2^3=-72
\]

\end{solution}

\begin{problem}
已知 \(\lg2=0.3010\)，\(f\left(x\right)=\log_{\sqrt{10}}x\)，求 \(f\left(5\right)\).
\end{problem}

\begin{solution}
\[
f\left(5\right)=\log_{\sqrt{10}}5
=\frac{\lg5}{\lg\sqrt{10}}
=\frac{\lg\frac{10}{2}}{\lg\sqrt{10}}
=\frac{\lg10-\lg2}{\frac{1}{2}\lg10}
=\frac{1-0.3010}{\frac{1}{2}\times1}=1.398
\]

\end{solution}

\begin{problem}
求函数 \(y=\frac{\sqrt{1-\frac{2}{3}x}}{\sqrt[3]{3x-1}}\) 的定义域.
\end{problem}

\begin{solution}
要使这个函数有意义，自变量 \(x\) 必须满足下面的不等式组
\[
\begin{cases}
1-\frac{2}{3}x\ge0,\\
3x-1\ne0.
\end{cases}
\]
所以 \(x\le\frac{3}{2}\)，且 \(x\ne\frac{1}{3}\). 因此，函数 \(y=\frac{\sqrt{1-\frac{2}{3}x}}{\sqrt[3]{3x-1}}\) 的定义域为 \(\frac{1}{3}<x\le\frac{3}{2}\) 和 \(x<\frac{1}{3}\).

\end{solution}

\begin{problem}
已知 \(\cos\theta=0.6\)，\(270^\circ<\theta<360^\circ\)，求 \(\left(\sin\frac{\theta}{2}-\cos\frac{\theta}{2}\right)^2\) 的值.
\end{problem}

\begin{solution}
因为 \(270^\circ<\theta<360^\circ\)，所以 \(\sin\theta\) 是负值，所以
\[
\sin\theta=-\sqrt{1-\cos^2\theta}=-\sqrt{1-0.6^2}=-0.8
\]
因此
\[
\left(\sin\frac{\theta}{2}-\cos\frac{\theta}{2}\right)^2
=\sin^2\frac{\theta}{2}+\cos^2\frac{\theta}{2}-2\sin\frac{\theta}{2}\cos\frac{\theta}{2}
=1-\sin\theta=1+0.8=1.8
\]

\end{solution}

\begin{problem}
把点 \(P\) 的极坐标 \(\left(-4,-\frac{4\pi}{3}\right)\) 化成直角坐标.
\end{problem}

\begin{solution}
极坐标和直角坐标的变换公式是 \(x=\rho\cos\theta\)，\(y=\rho\sin\theta\). 因此，\(P\) 点的横坐标为
\[
x=-4\cos\left(-\frac{4\pi}{3}\right)
=-4\cos\left(\pi+\frac{\pi}{3}\right)
=-4\left(-\cos\frac{\pi}{3}\right)
=-4\left(-\frac{1}{2}\right)=2
\]
纵坐标为
\[
y=-4\sin\left(-\frac{4\pi}{3}\right)
=4\sin\left(\pi+\frac{\pi}{3}\right)
=-4\sin\frac{\pi}{3}
=-4\times\frac{\sqrt{3}}{2}
=-2\sqrt{3}
\]
所以，\(P\) 点的直角坐标是 \(\left(2,-2\sqrt{3}\right)\).
\end{solution}

\end{problem}

\begin{solution}
因为 \(AD\parallel MN\parallel BC\).

\begin{problem}
圆台上底直径为 \(12\mathrm{~cm}\)，下底直径为 \(24\mathrm{~cm}\)，高为 \(8\mathrm{~cm}\)，求圆台的全面积. （\(\pi=3.14\)，精确到 \(0.1\mathrm{~cm}^2\)）
\end{problem}

\begin{solution}
由已知，上底半径 \(r_1=\frac{1}{2}\cdot12\mathrm{~cm}=6\mathrm{~cm}\)，下底半径 \(r_2=\frac{1}{2}\times24\mathrm{~cm}=12\mathrm{~cm}\)，圆台的高 \(h=8\mathrm{~cm}\). 由此得圆台的母线 \(L=\sqrt{h^2+\left(r_2-r_1\right)^2}=\sqrt{8^2+\left(12-6\right)^2}=10\mathrm{~cm}\).
因此，圆台的全面积
\[
S=\pi\left[r_1^2+r_2^2+\left(r_1+r_2\right)L\right]
\approx3.14\left[6^2+12^2+\left(6+12\right)\times10\right]
=3.14\times360\approx1130.4\mathrm{~cm}^2
\]

答：圆台的全面积约为 \(1130.4\mathrm{~cm}^2\).
\end{solution}

\begin{problem}

红旗拖拉机厂今年元月份生产出一批甲，乙两种型号的拖拉机，其中生产乙型 \(16\text{ 台}\). 从二月份起，甲型每月增产 \(10\text{ 台}\)，而乙型按每月相同增长率逐步递增. 又知二月份甲，乙两型拖拉机产量之比是 \(3:2\)，三月份甲，乙两型拖拉机产量之和是 \(65\text{ 台}\). 求乙型拖拉机每月的增长率及甲型拖拉机元月份的产量.

\end{problem}

\begin{solution}
设乙型拖拉机每月的增长率为 \(x\)，甲型拖拉机元月份的产量为 \(y\text{ 台}\). 依题意得方程组：
\[
\begin{cases}
\frac{y+10}{16\left(1+x\right)}=\frac{3}{2},\\
y+20+16\left(1+x\right)^2=65.
\end{cases}
\]
由第一式得 \(y=24x+14\). 把 \(y=24x+14\) 代入第二式得 \(16x^2+56x-15=0\)，即 \(\left(4x-1\right)\left(4x+15\right)=0\). 解得 \(x_1=\frac{1}{4}=25\%\)，\(x_2=-\frac{15}{4}\)（不合题意，舍去）. 由 \(x=\frac{1}{4}\) 得 \(y=24x+14=24\times\frac{1}{4}+14=20\text{ 台}\). 经检验，\(x=\frac{1}{4}\)，\(y=20\) 是原方程组的解.

答：红旗拖拉机厂生产的乙型拖拉机每月的增长率为 \(25\%\)，甲型拖拉机元月份的产量为 \(20\text{ 台}\).
\end{solution}

\begin{problem}

已知：直线 \(3x+y=5\) 及 \(2x-3y+4=0\) 交于 \(A\) 点.
求：
\begin{enumerate}
\item 过 \(A\) 点且与 \(y\) 轴垂直的直线方程；
\item 过 \(A\) 点且与圆 \(x^2+y^2=1\) 相切的直线方程.
\end{enumerate}

\end{problem}

\begin{solution}
先求出交点 \(A\) 的坐标 \(\left(x,y\right)\). 因为 \(A\) 既是直线 \(3x+y=5\) 上的点，也是直线 \(2x-3y+4=0\) 上的点，因此 \(A\) 的坐标 \(\left(x,y\right)\) 应适合方程组
\[
\begin{cases}
3x+y=5,\\
2x-3y+4=0.
\end{cases}
\]
解之，得 \(x=1\)，\(y=2\). 所以 \(A\) 的坐标是 \(\left(1,2\right)\).

（1）过 \(A\) 点与 \(y\) 轴垂直的直线方程是 \(y=2\).

（2）设所求直线和圆 \(x^2+y^2=1\) 相切于 \(P\left(x,y\right)\) 点. 因为切线过 \(A\left(1,2\right)\) 点，所以所求切线的斜率是 \(k_1=\frac{y-2}{x-1}\). 而过切点 \(P\) 的圆的直径所在直线的斜率是 \(k_2=\frac{y}{x}\). 根据平面几何知识，这样两条直线是互相垂直的，因此它们的斜率的积应等于 \(-1\)，即 \(k_1\cdot k_2=\left(\frac{y-2}{x-1}\right)\frac{y}{x}=-1\).
而 \(P\left(x,y\right)\) 是圆上的点，又有 \(x^2+y^2=1\). 由上两式解得
\[
\begin{cases}
x_1=1,\\
y_1=0
\end{cases}
\quad\text{和}\quad
\begin{cases}
x_2=-\frac{3}{5},\\
y_2=\frac{4}{5}.
\end{cases}
\]
由两点式，可得所求的直线有两条：\(L_1:\frac{y-2}{x-1}=\frac{2-\frac{4}{5}}{1+\frac{3}{5}}\)，即 \(3x-4y+5=0\)，和 \(L_2:\frac{x-1}{y-2}=\frac{1-1}{2-0}\)，即 \(x=1\).
\end{solution}

\begin{problem}

甲，乙两船，甲船在某岛 \(B\) 的正南方向 \(A\) 处，\(AB=10\text{ 海里}\)，甲船自 \(A\) 处以 \(4\text{ 海里/小时}\) 的速度向正北方向航行；同时，乙船以 \(6\text{ 海里/小时}\) 的速度自岛 \(B\) 出发，向岛的北 \(60^\circ\) 西方向驶去，问几分钟后两船相距最近？（精确到 \(1\text{ 分}\)）

\end{problem}

\begin{solution}
设经过 \(x\) 小时后，甲船航行到 \(C\) 点，乙船航行到 \(D\) 点（如图），因此 \(AC=4x\text{ 海里}\)，\(BC=AB-AC=10-4x\text{ 海里}\)，\(BD=6x\text{ 海里}\).

\begin{center}
\bitmapinclude[max width=3.8cm]{img/1977/inner_mongolia_science/q5_solution_fig1.png}
\end{center}

在 \(\triangle BCD\) 中，\(\angle DBC=180^\circ-60^\circ=120^\circ\). 所以甲，乙船间的距离的平方
\[
\begin{aligned}
DC^2&=BD^2+BC^2-2\cdot BD\cdot BC\cdot\cos120^\circ\\
&=\left(6x\right)^2+\left(10-4x\right)^2-2\cdot6x\cdot\left(10-4x\right)\cdot\left(-\frac{1}{2}\right)\\
&=28x^2-20x+100=28\left(x-\frac{5}{14}\right)^2+\frac{675}{7}
\end{aligned}
\]
因此当 \(x=\frac{5}{14}\text{ 小时}\approx21\text{ 分}\) 时，\(DC^2\) 取最小值，即 \(DC\) 取最小值.

答：大约 \(21\text{ 分钟}\) 后，两船相距最近.
\end{solution}

\begin{problem}

已知：在 \(\triangle ABC\) 中，\(a\)，\(b\)，\(c\) 是三内角 \(A\)，\(B\)，\(C\) 的对边，\(\lg\sin A\)，\(\lg\sin B\)，\(\lg\sin C\) 成等差数列.

\begin{enumerate}
\item 求证：\(\frac{\sin^2 A}{\sin^2 B}=\frac{a}{c}\)；

\item 又若方程 \(cx^2+2cx+a=0\) 有相等两实根，求证：\(\sin A=\sin B=\sin C\).
\end{enumerate}

\end{problem}

\begin{solution}
（1）因为 \(\lg\sin A\)，\(\lg\sin B\)，\(\lg\sin C\) 成等差数列，所以 \(2\lg\sin B=\lg\sin A+\lg\sin C\)，得 \(\sin^2 B=\sin A\cdot\sin C\). 又根据正弦定理，有 \(\sin C=\frac{c\sin A}{a}\). 把上式代入 \(\sin^2 B=\sin A\cdot\sin C\)，得 \(\sin^2 B=\sin A\cdot\frac{c\sin A}{a}\)，即 \(\frac{\sin^2 A}{\sin^2 B}=\frac{a}{c}\).

（2）因为方程 \(cx^2+2cx+a=0\) 有相等两实根，所以 \(\left(2c\right)^2-4ac=0\)，即 \(4c\left(c-a\right)=0\). 故 \(c=0\)（舍去）或 \(c=a\)，从而得 \(\angle C=\angle A\)，\(\sin C=\sin A\). 由（1）知 \(\sin^2 B=\sin A\sin C=\sin^2 A\)，所以 \(\sin B=\sin A\). 因此 \(\sin A=\sin B=\sin C\).
\end{solution}

\begin{problem}

在 \(\triangle ABC\) 中，求证 \(\sin^2 A+\sin^2 B+\cos^2 C+2\sin A\sin B\cos\left(A+B\right)=1\).

\end{problem}

\begin{solution}

\begin{align*}
&\sin^2 A+\sin^2 B+\cos^2 C+2\sin A\sin B\cos\left(A+B\right) \\
=&\sin^2 A+\sin^2 B+\cos^2\left(A+B\right)+2\sin A\sin B\left(\cos A\cos B-\sin A\sin B\right) \\
=&\sin^2 A+\sin^2 B+\left(\cos A\cos B-\sin A\sin B\right)^2+2\sin A\sin B\cos A\cos B-2\sin^2 A\sin^2 B \\
=&\sin^2 A+\sin^2 B+\cos^2 A\cos^2 B-\sin^2 A\sin^2 B=\sin^2 A\left(1-\sin^2 B\right)+\sin^2 B+\cos^2 A\cos^2 B \\
=&\sin^2 A\cos^2 B+\sin^2 B+\cos^2 A\cos^2 B=\cos^2 B\left(\sin^2 A+\cos^2 A\right)+\sin^2 B=\cos^2 B+\sin^2 B=1
\end{align*}
\end{solution}

\begin{problem}

已知：椭圆的中心点在原点，长轴和短轴都在坐标轴上，离心率 \(e=0.8\)，一条准线方程为 \(y=-\frac{25}{4}\)，求内接于椭圆的最大矩形面积.

\end{problem}

\begin{solution}
因为准线方程是 \(y=-\frac{25}{4}\)，它与 \(x\) 轴平行，而中心在原点，所以长轴在 \(y\) 轴上，短轴在 \(x\) 轴上，所以椭圆的方程有如下形式：\(\frac{x^2}{b^2}+\frac{y^2}{a^2}=1\)（其中 \(a\) 是半长轴，\(b\) 是半短轴）. 由已知可得方程组
\[
\begin{cases}
\frac{c}{a}=0.8,\\
-\frac{a^2}{c}=-\frac{25}{4}.
\end{cases}
\]
解之，得 \(a=5\)，\(c=4\). 所以 \(b=\sqrt{a^2-c^2}=\sqrt{5^2-4^2}=3\). 因此，椭圆的方程是 \(\frac{x^2}{3^2}+\frac{y^2}{5^2}=1\). 其内接最大矩形的长和宽，应分别是 \(\sqrt{2}a\) 和 \(\sqrt{2}b\)（参阅陕西参考题1的解），即长和宽是 \(5\sqrt{2}\) 和 \(3\sqrt{2}\)，所以得内接最大矩形的面积为 \(S=3\sqrt{2}\times5\sqrt{2}=30\).
\end{solution}
