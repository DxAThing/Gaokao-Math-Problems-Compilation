\chapter{1977年江苏连云港市初试卷}

\section{填空题}

\begin{problem}
把指数式 \(3^{2}=9\) 改写成对数式\fillinblank{}.
\end{problem}

\begin{answer}
\(\log_{3}9=2\)
\end{answer}

\begin{solution}
由指数式与对数式的互化关系 \(a^{b}=c\Longleftrightarrow\log_{a}c=b\)，将 \(3^{2}=9\) 中的底数、幂和幂的值分别对应到对数的底数、真数和结果，得 \(\log_{3}9=2\).
\end{solution}

\begin{problem}
写出下列数列的通项公式：
\begin{enumerate}
  \item \(2\)，\(4\)，\(6\)，\(8\)，\(10\)，\(\ldots\)，\(a_n=\)\fillinblank{}；
  \item \(-1\)，\(1\)，\(-1\)，\(1\)，\(\ldots\)，\(a_n=\)\fillinblank{}.
\end{enumerate}
\end{problem}

\begin{answer}
\(2n\)，\((-1)^n\)
\end{answer}

\begin{solution}
第一列数列的首项为 \(2\)，公差为 \(2\)，所以其第 \(n\) 项为
\[
a_n=2+(n-1)\cdot2=2n.
\]
第二列数列中相邻两项的比为 \(-1\)，且首项为 \(-1\)，所以它是首项为 \(-1\)、公比为 \(-1\) 的等比数列，从而
\[
a_n=-1\cdot(-1)^{n-1}=(-1)^n.
\]
\end{solution}

\begin{problem}
以点 \((0,0)\) 为圆心，以 \(4\) 为半径的圆的方程是\fillinblank{}.
\end{problem}

\begin{answer}
\(x^2+y^2=16\)
\end{answer}

\begin{solution}
圆心为原点 \(O(0,0)\)，半径为 \(4\). 圆心在原点、半径为 \(r\) 的圆的方程为 \(x^2+y^2=r^2\)，代入 \(r=4\)，得
\[
x^2+y^2=4^2=16.
\]
\end{solution}

\section{解答题}

\begin{problem}
计算：\(1^{0.123}\times\left(\frac{3}{2}\right)^{-3}\div\left(-\frac{9}{10}\right)^{0}\).
\end{problem}

\begin{solution}
因为 \(1^{0.123}=1\)，且 \(-\frac{9}{10}\neq0\)，所以 \(\left(-\frac{9}{10}\right)^0=1\). 又因为
\[
\left(\frac{3}{2}\right)^{-3}=\left(\frac{2}{3}\right)^3=\frac{8}{27},
\]
故原式
\[
=1\times\frac{8}{27}\div1=\frac{8}{27}.
\]
\end{solution}

\begin{problem}
计算：\(\log_{2}781-\lg100\).
\end{problem}

\begin{solution}
因为 \(\lg100=2=\log_{2}4\)，所以由对数的差公式，原式
\[
=\log_{2}781-\log_{2}4=\log_{2}\frac{781}{4}.
\]
\end{solution}

\begin{problem}
计算：\(\frac{\sqrt[3]{x^{\frac{7}{2}}\sqrt{x^{-1}}}}{\sqrt{\sqrt[3]{x^{-7}}\sqrt[3]{x^{13}}}}\).
\end{problem}

\begin{solution}
根式 \(\sqrt{x^{-1}}\) 有意义要求 \(x>0\). 在此条件下，\(\sqrt{x^{-1}}=x^{-\frac12}\)，故分子为
\[
\sqrt[3]{x^{\frac72}\sqrt{x^{-1}}}=\sqrt[3]{x^{\frac72}x^{-\frac12}}=\sqrt[3]{x^3}=x.
\]
分母为
\[
\sqrt{\sqrt[3]{x^{-7}}\sqrt[3]{x^{13}}}=\sqrt{\sqrt[3]{x^{-7+13}}}=\sqrt{x^2}=x.
\]
由于 \(x>0\)，分母不为零，所以原式
\[
=\frac{x}{x}=1.
\]
\end{solution}

\begin{problem}
计算：\(\frac{\left(\sin22^{\circ}30'+\cos22^{\circ}30'\right)^{2}}{2\cos0^{\circ}-2\sin15^{\circ}\cos15^{\circ}+\sin270^{\circ}}\).
\end{problem}

\begin{solution}
记 \(\theta=22^{\circ}30'\)，则
\[
\left(\sin\theta+\cos\theta\right)^2=\sin^2\theta+\cos^2\theta+2\sin\theta\cos\theta=1+\sin2\theta=1+\sin45^{\circ}=1+\frac{\sqrt2}{2}.
\]
另一方面，\(2\cos0^{\circ}=2\)，\(2\sin15^{\circ}\cos15^{\circ}=\sin30^{\circ}=\frac12\)，且 \(\sin270^{\circ}=-1\)，所以分母为
\[
2-\frac12-1=\frac12.
\]
因此原式
\[
=\frac{1+\frac{\sqrt2}{2}}{\frac12}=2+\sqrt2.
\]
\end{solution}

\begin{problem}
对于下列函数，写出自变量 \(x\) 的取值范围：\(y=\frac{6x-1}{2x-1}\).
\end{problem}

\begin{solution}
分式有意义的条件是分母不为零，即 \(2x-1\neq0\)，所以 \(x\neq\frac12\). 因此自变量的取值范围为
\[
\left(-\infty,\frac12\right)\cup\left(\frac12,+\infty\right).
\]
\end{solution}

\begin{problem}
对于下列函数，写出自变量 \(x\) 的取值范围：\(y=\sqrt{x}+\sqrt{-x}\).
\end{problem}

\begin{solution}
由 \(\sqrt{x}\) 有意义，得 \(x\geqslant0\)；由 \(\sqrt{-x}\) 有意义，得 \(-x\geqslant0\)，即 \(x\leqslant0\). 两个条件同时成立时只能有 \(x=0\)，所以自变量的取值范围为 \(\{0\}\).
\end{solution}

\begin{problem}
对于下列函数，写出自变量 \(x\) 的取值范围：\(y=\sqrt{\log_{\frac{1}{2}}(x+1)}\).
\end{problem}

\begin{solution}
首先，必须有 \(x+1>0\)，以保证对数有意义. 其次，外层平方根要求
\[
\log_{\frac12}(x+1)\geqslant0.
\]
因为底数 \(\frac12\) 在 \(0\) 与 \(1\) 之间，对数函数单调递减，所以
\[
\log_{\frac12}(x+1)\geqslant0\Longleftrightarrow 0<x+1\leqslant1.
\]
于是 \(-1<x\leqslant0\)，自变量的取值范围为 \((-1,0]\).
\end{solution}

\begin{problem}
求极限 \(\displaystyle\lim_{n\to\infty}\left[\frac{1+3+5+\cdots+(2n-1)}{n+2}-n\right]\).
\end{problem}

\begin{solution}
前 \(n\) 个奇数的和为 \(n^2\)，所以
\[
\frac{1+3+5+\cdots+(2n-1)}{n+2}-n=\frac{n^2}{n+2}-n=\frac{n^2-n(n+2)}{n+2}=-\frac{2n}{n+2}.
\]
因此
\[
\lim_{n\to\infty}\left[\frac{1+3+5+\cdots+(2n-1)}{n+2}-n\right]=\lim_{n\to\infty}\left(-\frac{2}{1+\frac2n}\right)=-2.
\]
\end{solution}

\begin{problem}
\begin{enumerate}
  \item \(k\) 为何值时，方程 \(x^{2}-2kx+3=0\) 有一个根是 \(-2\)？
  \item 试判别一元二次方程 \(ax^{2}+(a-b)x-(a+b)=0\) 的根的情况.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
  \item 因为 \(-2\) 是方程的一个根，所以代入方程得
  \[
  (-2)^2-2k(-2)+3=0,
  \]
  即 \(4+4k+3=0\)，解得 \(k=-\frac74\).

  \item 题目所说的一元二次方程要求 \(a\neq0\). 它的判别式为
  \[
  \Delta=(a-b)^2-4a\bigl(-(a+b)\bigr)=a^2-2ab+b^2+4a^2+4ab=5a^2+2ab+b^2=(a+b)^2+4a^2.
  \]
  当 \(a\neq0\) 时，\((a+b)^2\geqslant0\) 且 \(4a^2>0\)，所以 \(\Delta>0\). 因此该一元二次方程恒有两个不相等的实数根.

  若不预先限定 \(a\neq0\)，则 \(a=0\) 时原式不再是一元二次方程；此时若 \(b\neq0\)，方程化为 \(-b(x+1)=0\)，只有根 \(x=-1\)，若 \(b=0\)，方程化为恒等式 \(0=0\).
\end{enumerate}
\end{solution}

\begin{problem}
已知椭圆 \(\frac{x^{2}}{5^{2}}+\frac{y^{2}}{3^{2}}=1\)，

\begin{enumerate}
  \item 求此椭圆的长轴的长，离心率，焦点坐标和准线方程.
  \item 如图过椭圆的一个焦点 \(F\)，作 \(FA\) 平行于 \(y\) 轴交椭圆的上半部于 \(A\) 点，求 \(A\) 点的坐标.
  \item 求 \(\triangle AOF\) 的面积.
\end{enumerate}

\FigureLayoutDeclare{img/1977/jiangsu_lianyungang_city/q6_fig1.png}{8.5cm}{6bp 0bp 38bp 38bp}
\bitmapfigure[max width=5.6cm]{img/1977/jiangsu_lianyungang_city/q6_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
  \item 与椭圆标准方程 \(\dfrac{x^2}{a^2}+\dfrac{y^2}{b^2}=1\) 对比，得 \(a=5\)，\(b=3\). 因此
  \[
  c=\sqrt{a^2-b^2}=\sqrt{25-9}=4.
  \]
  所以长轴长为 \(2a=10\)，离心率为 \(e=\dfrac ca=\dfrac45\)，焦点坐标为 \((-4,0)\)、\((4,0)\)，准线方程为
  \[
  x=\pm\frac ae=\pm\frac{25}{4}.
  \]

  \item 由图取右焦点 \(F=(4,0)\). 因为 \(FA\parallel y\) 轴，所以点 \(A\) 的横坐标为 \(4\). 代入椭圆方程并取上半部的正值，得
  \[
  \frac{4^2}{5^2}+\frac{y^2}{3^2}=1,
  \]
  即 \(\dfrac{16}{25}+\dfrac{y^2}{9}=1\)，从而 \(y^2=\dfrac{81}{25}\). 由于 \(A\) 在上半部，\(y=\dfrac95\)，所以 \(A\left(4,\dfrac95\right)\).

  \item \(O=(0,0)\)，\(F=(4,0)\)，所以 \(OF=4\)，而点 \(A\) 到 \(OF\) 所在直线的距离为 \(\dfrac95\). 因此
  \[
  S_{\triangle AOF}=\frac12\times4\times\frac95=\frac{18}{5}.
  \]
\end{enumerate}
\end{solution}

\begin{problem}
如图：把小矩形 \(ABCD\) 扩充为大矩形 \(AEFG\)，并使 \(BE+EF+FG+GD=22\)，已知小矩形的宽 \(AB=2\)，小矩形的周长为 \(12\)，问 \(BE\) 为何值时扩充部分的面积最大？最大面积是多少？（扩充部分为图中阴影部分）

\FigureLayoutDeclare{img/1977/jiangsu_lianyungang_city/q7_fig1.png}{9.0cm}{0bp 0bp 0bp 0bp}
\bitmapfigure[max width=6.0cm]{img/1977/jiangsu_lianyungang_city/q7_fig1.png}
\end{problem}

\begin{solution}
设 \(BE=x\)，\(DG=y\). 小矩形的周长为 \(12\)，且 \(AB=2\)，所以
\[
2(AB+BC)=12\Longrightarrow BC=4.
\]
由图可知大矩形的长为 \(AF=BC+DG=4+y\)，宽为 \(AE=AB+BE=2+x\)，并且 \(EF=AF=4+y\)，\(FG=AE=2+x\). 代入题设条件，得
\[
x+(4+y)+(2+x)+y=22,
\]
即 \(x+y=8\). 因为 \(BE\) 和 \(DG\) 都是长度，所以 \(0\leqslant x\leqslant8\)，且 \(y=8-x\).

扩充部分的面积等于大矩形面积减去小矩形面积，因此
\[
S=(4+y)(2+x)-AB\cdot BC=(12-x)(2+x)-8=16+10x-x^2.
\]
配方得
\[
S=41-(x-5)^2.
\]
因为 \((x-5)^2\geqslant0\)，且 \(x=5\) 在 \([0,8]\) 内，所以当 \(BE=x=5\) 时，扩充部分的面积最大，最大面积为 \(41\).
\end{solution}

\begin{problem}
已知 \(AD\) 是 \(\odot O\) 的弦，且 \(AD=8\)，半径 \(OH\perp AD\)，交 \(AD\) 于 \(B\) 点，且 \(HB=2\)，又 \(AC\) 是圆的直径，试求 \(BC\) 的长.

\bitmapfigure[max width=5.8cm]{img/1977/jiangsu_lianyungang_city/q8_fig1.png}
\end{problem}

\begin{solution}
因为从圆心向弦作的垂线平分弦，所以 \(AB=BD=\dfrac{AD}{2}=4\). 设圆的半径为 \(R=OH=OA\). 由图中点的顺序 \(O\)、\(B\)、\(H\) 共线，且 \(HB=2\)，所以 \(OB=R-2\). 在直角三角形 \(OAB\) 中，由勾股定理得
\[
R^2=(R-2)^2+4^2.
\]
解得 \(4R=20\)，所以 \(R=5\)，从而直径 \(AC=2R=10\). 因为 \(AC\) 是直径，\(\angle ABC=90^{\circ}\)，于是
\[
BC=\sqrt{AC^2-AB^2}=\sqrt{10^2-4^2}=\sqrt{84}=2\sqrt{21}.
\]
\end{solution}

\section{附加题}

\begin{problem}
如图 \(A\) 是双曲线 \(x^{2}-y^{2}=1\) 上的一点，\(A\) 点处的法线交坐标轴于 \(B\)，\(C\) 两点，若 \(\left|BC\right|=42\)，试求 \(\left|OA\right|\). 提示：设 \(A(x_0,y_0)\)，先求出法线斜率，再写出方程，找出 \(B\)，\(C\) 坐标，再用距离公式求出 \(A\) 点坐标.

\FigureLayoutDeclare{img/1977/jiangsu_lianyungang_city/q9_fig1.png}{9.0cm}{23bp 34bp 37bp 0bp}
\bitmapfigure[max width=5.8cm]{img/1977/jiangsu_lianyungang_city/q9_fig1.png}
\end{problem}

\begin{solution}
设 \(A=(x_0,y_0)\). 由图可知 \(A\) 在第一象限，所以 \(x_0>0\)，\(y_0>0\)，后续除以 \(x_0\)、\(y_0\) 合法. 因为 \(A\) 在双曲线 \(x^2-y^2=1\) 上，所以
\[
x_0^2-y_0^2=1.
\]
对方程 \(x^2-y^2=1\) 隐式求导，得 \(2x-2y\dfrac{dy}{dx}=0\)，所以双曲线在 \(A\) 点处的切线斜率为 \(\dfrac{x_0}{y_0}\)，法线斜率为 \(-\dfrac{y_0}{x_0}\). 因而法线方程为
\[
y-y_0=-\frac{y_0}{x_0}(x-x_0),
\]
整理得 \(\dfrac{x}{x_0}+\dfrac{y}{y_0}=2\). 它与 \(y\) 轴的交点为 \(B=(0,2y_0)\)，与 \(x\) 轴的交点为 \(C=(2x_0,0)\).
由 \(\left|BC\right|=42\)，得
\[
42=\sqrt{(2x_0)^2+(2y_0)^2}=2\sqrt{x_0^2+y_0^2},
\]
所以 \(x_0^2+y_0^2=441\). 联立 \(x_0^2-y_0^2=1\) 与 \(x_0^2+y_0^2=441\)，得 \(x_0^2=221\)，\(y_0^2=220\). 因此
\[
\left|OA\right|=\sqrt{x_0^2+y_0^2}=\sqrt{441}=21.
\]
\end{solution}
