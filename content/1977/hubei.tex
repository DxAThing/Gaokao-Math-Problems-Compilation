\section{解答题}

\begin{problem}
\begin{enumerate}
\item
\( \sqrt{\left(-3\right)^2} \)；\( \sqrt[3]{\left(-27\right)^3} \).

\item
\( \log_2 2+\log_5 1-\log_{\frac{1}{3}}\left(\frac{1}{27}\right) \).

\item
\( \left(\sqrt{\frac{2}{3}}\right)^3\cdot\sqrt[3]{\left(\frac{3}{2}\right)^2}\cdot\left(\frac{2}{3}\right)^{-\frac{5}{6}} \).

\item
解方程：\(x-\sqrt{8-x^2}=0\).

\item
当 \(m\leq 1\) 时，计算 \(\sqrt{\left(m-1\right)^2}\).

\item
求函数 \(y=\frac{\sqrt{2-x}}{x+4}\) 的定义域.

\item
分解因式：\(x^4+x^2y^2+y^4\).

\item
证明：\(\tan 22^\circ30'=\sqrt{2}-1\)（不查表）.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 因为平方根表示算术平方根，所以 \(\sqrt{\left(-3\right)^2}=\left|-3\right|=3\)；立方根的定义给出 \(\sqrt[3]{\left(-27\right)^3}=-27\).

\item 由 \(\log_2 2=1\)、\(\log_5 1=0\)，以及 \(\frac{1}{27}=\left(\frac{1}{3}\right)^3\)，得 \(\log_{\frac{1}{3}}\left(\frac{1}{27}\right)=3\). 因此原式 \(=1+0-3=-2\).

\item 三个幂的底数均为正数，可以使用分数指数幂运算，且 \(\sqrt[3]{\left(\frac{3}{2}\right)^2}=\left(\frac{3}{2}\right)^{\frac{2}{3}}=\left(\frac{2}{3}\right)^{-\frac{2}{3}}\). 所以
\[
\left(\sqrt{\frac{2}{3}}\right)^3\cdot\sqrt[3]{\left(\frac{3}{2}\right)^2}\cdot\left(\frac{2}{3}\right)^{-\frac{5}{6}}=\left(\frac{2}{3}\right)^{\frac{3}{2}-\frac{2}{3}-\frac{5}{6}}=\left(\frac{2}{3}\right)^0=1
\]

\item 原方程有意义必须满足 \(8-x^2\geq 0\)，且由 \(x=\sqrt{8-x^2}\) 可知 \(x\geq 0\). 移项并平方，得 \(x^2=8-x^2\)，即 \(x^2=4\)，所以候选值为 \(x=2\) 或 \(x=-2\). 代入原方程检验：\(2-\sqrt{8-2^2}=0\)，而 \(-2-\sqrt{8-(-2)^2}=-4\neq 0\). 因此原方程的解为 \(x=2\).

\item 因为 \(m\leq 1\)，所以 \(m-1\leq 0\)，从而 \(\left|m-1\right|=-(m-1)=1-m\). 因此 \(\sqrt{\left(m-1\right)^2}=\left|m-1\right|=1-m\).

\item 要使根式有意义，必须有 \(2-x\geq 0\)，即 \(x\leq 2\)；同时分母不能为零，即 \(x+4\neq 0\)，也就是 \(x\neq -4\). 因此函数的定义域为 \((-\infty,-4)\cup(-4,2]\).

\item 利用平方差公式，先写成
\[
x^4+x^2y^2+y^4=\left(x^2+y^2\right)^2-x^2y^2=\left(x^2+y^2\right)^2-\left(xy\right)^2
\]
再作因式分解，得 \(x^4+x^2y^2+y^4=\left(x^2-xy+y^2\right)\left(x^2+xy+y^2\right)\).

\item 令 \(\theta=22^\circ30'=\frac{45^\circ}{2}\). 由半角正切公式，且 \(\theta\) 为锐角，得
\[
\tan\theta=\frac{1-\cos45^\circ}{\sin45^\circ}
\]
不查表时可在等腰直角三角形中取两直角边长为 \(1\)，则斜边为 \(\sqrt{2}\)，所以 \(\sin45^\circ=\cos45^\circ=\frac{\sqrt{2}}{2}\). 代入上式并化简，得
\[
\tan22^\circ30'=\frac{1-\frac{\sqrt{2}}{2}}{\frac{\sqrt{2}}{2}}=\frac{2-\sqrt{2}}{\sqrt{2}}=\sqrt{2}-1
\]
\end{enumerate}
\end{solution}

\begin{problem}
围一矩形场地，一边利用房屋的墙，其他三边用 \(L\text{ 米}\) 的篱笆围成，问怎样围才能使面积最大？最大面积是多少？

\bitmapfigure[max width=6.0cm]{img/1977/hubei/q2_fig1.png}
\end{problem}

\begin{solution}
设与墙平行的一边长为 \(x\text{ 米}\)，与墙垂直的两边长均为 \(y\text{ 米}\)，其中 \(L>0\). 由篱笆总长得 \(x+2y=L\)，所以 \(y=\frac{L-x}{2}\)，且矩形存在时 \(0<x<L\). 矩形面积为
\[
S=xy=x\left(\frac{L-x}{2}\right)=-\frac{1}{2}x^2+\frac{L}{2}x=-\frac{1}{2}\left(x-\frac{L}{2}\right)^2+\frac{L^2}{8}
\]
因为 \(\left(x-\frac{L}{2}\right)^2\geq 0\)，所以 \(S\leq\frac{L^2}{8}\)，当且仅当 \(x=\frac{L}{2}\) 时等号成立. 此时 \(y=\frac{L-x}{2}=\frac{L}{4}\).

因此，当与墙平行的一边长为 \(\frac{L}{2}\text{ 米}\)，与墙垂直的两边分别长为 \(\frac{L}{4}\text{ 米}\) 时，面积最大，最大面积为 \(\frac{L^2}{8}\text{ 米}^2\).
\end{solution}

\begin{problem}
已知直线 \(L_1\) 方程为 \(x+y-1=0\) 和点 \(A\left(-3,0\right)\).

\begin{enumerate}
\item 求与直线 \(L_1\) 垂直且通过 \(A\) 的直线方程；

\item 求点 \(A\) 到直线 \(L_1\) 的距离.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 将 \(L_1\) 化为斜截式 \(y=-x+1\)，可知其斜率为 \(k_1=-1\). 设所求直线斜率为 \(k_2\)，两条互相垂直且斜率存在的直线满足 \(k_1k_2=-1\)，所以 \(k_2=1\). 由点斜式，过 \(A\left(-3,0\right)\) 的直线为 \(y-0=1\left(x+3\right)\)，即 \(x-y+3=0\).

\item 对直线 \(x+y-1=0\)，有 \(a=1\)、\(b=1\)、\(c=-1\). 点 \(A\left(-3,0\right)\) 到该直线的距离为
\[
d=\frac{\left|1\cdot(-3)+1\cdot0-1\right|}{\sqrt{1^2+1^2}}=\frac{4}{\sqrt{2}}=2\sqrt{2}
\]
\end{enumerate}
\end{solution}

\begin{problem}
解方程组：\(\begin{cases}
\log_2 x-\log_4 y=0,\\
y^2-13x^2+36=0.
\end{cases}\)
\end{problem}

\begin{solution}
由对数的定义，必须有 \(x>0\)，\(y>0\). 又因为 \(\log_4 y=\frac{\log_2 y}{\log_2 4}=\frac{1}{2}\log_2 y\)，第一式化为 \(2\log_2 x=\log_2 y\). 由于 \(x>0\)，有 \(2\log_2 x=\log_2\left(x^2\right)\)，所以 \(\log_2\left(x^2\right)=\log_2 y\)，从而 \(y=x^2\).

将 \(y=x^2\) 代入第二式，得 \(x^4-13x^2+36=0\)，即 \(\left(x^2-9\right)\left(x^2-4\right)=0\). 因而 \(x^2=9\) 或 \(x^2=4\). 结合 \(x>0\)，得 \(x=3\) 或 \(x=2\)，再由 \(y=x^2\) 得对应的 \(y=9\) 或 \(y=4\). 两组值均满足对数定义域和原方程组，所以原方程组的解为 \(\left(x,y\right)=\left(3,9\right)\) 或 \(\left(x,y\right)=\left(2,4\right)\).
\end{solution}

\begin{problem}
作函数 \(y=\sqrt{3}\cos2x+\sin2x\) 的图象（要求写出作图步骤，并画出草图）.
\end{problem}

\begin{solution}
先将函数化为正弦型：
\[
y=\sqrt{3}\cos2x+\sin2x=2\left(\frac{\sqrt{3}}{2}\cos2x+\frac{1}{2}\sin2x\right)=2\left(\sin\frac{\pi}{3}\cos2x+\cos\frac{\pi}{3}\sin2x\right)=2\sin\left(2x+\frac{\pi}{3}\right)
\]
因此函数的振幅为 \(2\)，角频率为 \(2\)，最小正周期为 \(T=\frac{2\pi}{2}=\pi\). 取一个周期内的相位 \(0\)、\(\frac{\pi}{2}\)、\(\pi\)、\(\frac{3\pi}{2}\)、\(2\pi\)，由 \(2x+\frac{\pi}{3}\) 分别等于这些相位，得到
对应的五个横坐标为 \(-\frac{\pi}{6}\)、\(\frac{\pi}{12}\)、\(\frac{\pi}{3}\)、\(\frac{7\pi}{12}\)、\(\frac{5\pi}{6}\)，相应的函数值依次为 \(0\)、\(2\)、\(0\)、\(-2\)、\(0\). 于是先在坐标系中标出这五个关键点，再按正弦曲线的变化趋势光滑连接，并以周期 \(\pi\) 向左右延拓.

\begin{center}
\begin{tabular}{c|ccccc}
\hline
\(x\) & \(-\dfrac{\pi}{6}\) & \(\dfrac{\pi}{12}\) & \(\dfrac{\pi}{3}\) & \(\dfrac{7\pi}{12}\) & \(\dfrac{5\pi}{6}\) \\
\hline
\(y\) & \(0\) & \(2\) & \(0\) & \(-2\) & \(0\) \\
\hline
\end{tabular}
\end{center}

\end{solution}

\section{参考题}

\begin{problem}
求下列函数的导数：

\begin{enumerate}
\item \(y=\sin x\).
\item \(y=\e^{3x}\).
\item \(y=\ln x\).
\item \(f(t)=A\sin\left(\omega t+\varphi\right)\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由基本导数公式 \(\left(\sin x\right)'=\cos x\)，得 \(y'=\cos x\).

\item 令 \(u=3x\)，则 \(u'=3\). 由复合函数求导法则 \(\left(\e^u\right)'=\e^u u'\)，得 \(y'=\e^{3x}\cdot3=3\e^{3x}\).

\item 在 \(x>0\) 时，基本导数公式为 \(\left(\ln x\right)'=\frac{1}{x}\)，所以 \(y'=\frac{1}{x}\).

\item 令 \(u=\omega t+\varphi\)，其中 \(A\)、\(\omega\)、\(\varphi\) 为常数，则 \(u'=\omega\). 因而由常数因子法则和链式法则，\(f'(t)=A\cos u\cdot u'=\omega A\cos\left(\omega t+\varphi\right)\).
\end{enumerate}
\end{solution}

\begin{problem}
设函数 \(f(x)=x^3-12x+10\).

\begin{enumerate}
\item 求 \(f(x)\) 的极值点；

\item 求 \(f(1)\)，\(f(2)\)，\(f(-3)\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 先求导，得 \(f'(x)=3x^2-12=3\left(x-2\right)\left(x+2\right)\). 当 \(x<-2\) 或 \(x>2\) 时，\(f'(x)>0\)；当 \(-2<x<2\) 时，\(f'(x)<0\). 因此函数在 \(x=-2\) 处由增变减，取得极大值；在 \(x=2\) 处由减变增，取得极小值. 又 \(f(-2)=(-2)^3-12(-2)+10=26\)，\(f(2)=2^3-12\cdot2+10=-6\)，所以极大值点为 \(\left(-2,26\right)\)，极小值点为 \(\left(2,-6\right)\).

\item 直接代入函数式，得 \(f(1)=1^3-12\cdot1+10=-1\)，\(f(2)=2^3-12\cdot2+10=-6\)，\(f(-3)=(-3)^3-12(-3)+10=19\).
\end{enumerate}
\end{solution}

\begin{problem}
解方程：\(y''+3y'-4y=0\)，其中 \(y\big|_{x=0}=0\)，\(y'\big|_{x=0}=-10\).
\end{problem}

\begin{solution}
对应的特征方程为 \(\lambda^2+3\lambda-4=0\)，因式分解得 \(\left(\lambda-1\right)\left(\lambda+4\right)=0\)，所以两个特征根为 \(\lambda_1=1\) 和 \(\lambda_2=-4\). 因此微分方程的通解为 \(y=C_1\e^{-4x}+C_2\e^x\)，其中 \(C_1\)、\(C_2\) 为常数.

对通解求导，得 \(y'=-4C_1\e^{-4x}+C_2\e^x\). 代入初始条件，得到
\[
\begin{cases}
C_1+C_2=0,\\
-4C_1+C_2=-10.
\end{cases}
\]
由第一式得 \(C_2=-C_1\)，代入第二式得 \(-5C_1=-10\)，所以 \(C_1=2\)，\(C_2=-2\). 故初值问题的解为 \(y=2\e^{-4x}-2\e^x\). 代回可得 \(y(0)=2-2=0\)，\(y'(0)=-8-2=-10\)，初始条件确实满足.
\end{solution}
