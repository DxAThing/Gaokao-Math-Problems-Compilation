\begin{problem}
\( y=2\sqrt{x^2} \) 和 \( y=2x \) 是不是同一个函数？为什么？

\end{problem}

\begin{solution}
不是同一个函数.

因为
\[
y=2\sqrt{x^2}=
\begin{cases}
2x, & x\ge 0, \\
-2x, & x<0.
\end{cases}
\]

所以 \( y=2\sqrt{x^2} \) 与 \( y=2x \) 并不恒等，不是同一个函数.
\end{solution}

\begin{problem}
确定下列各式的符号：

\begin{enumerate}[label=（\arabic*），leftmargin=*]
    \item \( 1-\sin 800^\circ \)；
    \item \( \cos 20^\circ-\sin 20^\circ \)；
    \item \( \tan 288^\circ\cdot \tan 510^\circ \).
\end{enumerate}

\end{problem}

\begin{solution}
（1）\( 1-\sin 800^\circ=1-\sin \left( 2\times 360^\circ+80^\circ \right)=1-\sin 80^\circ>0 \).

（2）\( \cos 20^\circ-\sin 20^\circ=\cos \left( 90^\circ-70^\circ \right)-\sin 20^\circ=\sin 70^\circ-\sin 20^\circ>0 \).

（3）
\begin{align*}
\tan 288^\circ\cdot \tan 510^\circ
=\;&\tan \left( 360^\circ-72^\circ \right)\cdot \tan \left( 360^\circ+150^\circ \right) \\
=\;&-\tan 72^\circ\cdot \tan 150^\circ \\
=\;&-\tan 72^\circ\cdot \tan \left( 180^\circ-30^\circ \right) \\
=\;&-\tan 72^\circ\left( -\tan 30^\circ \right)=\tan 72^\circ\cdot \tan 30^\circ>0.
\end{align*}
\end{solution}

\begin{problem}
已知 \( x \)、\( y \) 都是实数，且 \( \left( 2x-1 \right)^2+\left( y-8 \right)^2=0 \)，求 \( \log_2 xy=? \)

\end{problem}

\begin{solution}
由 \( x \)、\( y \) 都是实数，且 \( \left( 2x-1 \right)^2+\left( y-8 \right)^2=0 \) 可知 \( \left( 2x-1 \right)^2=0 \) 和 \( \left( y-8 \right)^2=0 \)，即 \( x=\frac{1}{2} \) 和 \( y=8 \).

所以
\[
\log_2 xy=\log_2 \left( \frac{1}{2}\times 8 \right)=\log_2 4=2\log_2 2=2.
\]
\end{solution}

\begin{problem}
\( \sqrt{\left( \log_3 20 \right)^2-6\log_3 20+9}=? \)

\end{problem}

\begin{solution}
原式 \( =\sqrt{\left( \log_3 20-3 \right)^2}=\left| \log_3 20-3 \right| \).

而 \( 3=\log_3 3^3=\log_3 27>\log_3 20 \).

所以原式 \( =\left| \log_3 20-3 \right|=3-\log_3 20 \).
\end{solution}

\begin{problem}
已知 \( \log_{1.5}7=a \)，\( \log_{1.5}8=b \)，求 \( \log_{56}15=? \)

\end{problem}

\begin{solution}
\[
\log_{56}15=\frac{\log_{1.5}15}{\log_{1.5}56}=\frac{1}{\log_{1.5}\left( 7\times 8 \right)}=\frac{1}{\log_{1.5}7+\log_{1.5}8}=\frac{1}{a+b}.
\]
\end{solution}

\begin{problem}
化简：
\(\left( \frac{x^{-1}-1}{x^{-1}+1}-\frac{x^{-1}+1}{x^{-1}-1} \right)\cdot \left( 2^{-1}-4^{-1}x^{-1}-4^{-1}x \right)\).

\end{problem}

\begin{solution}
\begin{align*}
\text{原式}
=\;&\left( \frac{1-x}{1+x}-\frac{1+x}{1-x} \right)\cdot \left( \frac{1}{2}-\frac{1}{4x}-\frac{x}{4} \right) \\
=\;&\frac{-4x}{1-x^2}\cdot \frac{2x-1-x^2}{4x} \\
=\;&\frac{\left( x-1 \right)^2}{1-x^2}=\frac{1-x}{1+x}.
\end{align*}
\end{solution}

\begin{problem}
\( h \) 为何数时，方程 \( 2hx^2+\left( 8h+1 \right)x=-8h \) 有两个不相等的实根？

\end{problem}

\begin{solution}
要使方程有两个不相等的实根，必须根的判别式大于 \( 0 \)，即
\[
\left( 8h+1 \right)^2-4\times 2h\times 8h>0.
\]

整理为
\[
16h+1>0.
\]

所以 \( h>-\frac{1}{16} \) 时，方程有两个不相等的实根.
\end{solution}

\begin{problem}
证明恒等式：
\(\frac{2\sin \theta\cdot \sin \left( \frac{\pi}{2}-\theta \right)+\sin \left( \frac{3\pi}{2}+\theta \right)}{1+\sin \left( \pi+\theta \right)+\sin^2 \left( \pi-\theta \right)-\sin^2 \left( \frac{\pi}{2}+\theta \right)}=\tan \left( \frac{\pi}{2}-\theta \right)\).

\end{problem}

\begin{solution}
\begin{align*}
\text{左式}
=\;&\frac{2\sin \theta\cdot \cos \theta-\cos \theta}{1-\sin \theta+\sin^2 \theta-\cos^2 \theta} \\
=\;&\frac{\left( 2\sin \theta-1 \right)\cos \theta}{2\sin^2 \theta-\sin \theta} \\
=\;&\frac{\left( 2\sin \theta-1 \right)\cos \theta}{\left( 2\sin \theta-1 \right)\sin \theta} \\
=\;&\frac{\cos \theta}{\sin \theta}=\cot \theta.
\end{align*}

右式 \( =\cot \theta \).

所以左式 \( = \) 右式，从而
\[
\frac{2\sin \theta\cdot \sin \left( \frac{\pi}{2}-\theta \right)+\sin \left( \frac{3\pi}{2}+\theta \right)}{1+\sin \left( \pi+\theta \right)+\sin^2 \left( \pi-\theta \right)-\sin^2 \left( \frac{\pi}{2}+\theta \right)}=\tan \left( \frac{\pi}{2}-\theta \right).
\]
\end{solution}

\begin{enumerate}[label=（\arabic*），leftmargin=*]
    \item 三个半径都是 \( R \) 的圆两两外切，求过三个切点的三个圆弧所围成图形的面积. 
    \item 已知菱形面积为 \( 392\mathrm{cm}^2 \)，有一角为 \( 150^\circ \)，求边长. 
\end{enumerate}

\end{problem}

\begin{solution}
（1）连结 \( A \)、\( B \)、\( C \) 三点.

\bitmapfigure[max width=4.8cm,max totalheight=4.8cm]{img/1977/hunan_zhuzhou_pilot/q9_solution_fig1.png}

由图可知，所求的面积 \( s \) 等于 \( \triangle ABC \) 的面积减去扇形 \( ADF \) 面积的 \( 3 \) 倍.

\[
s=\frac{1}{2}AB\cdot AC\cdot \sin A-3\left( \frac{1}{6}\pi \cdot AD^2 \right)=\frac{1}{2}\cdot 2R\cdot 2R\cdot \sin 60^\circ-\frac{1}{2}\pi R^2
\]
\[
=\sqrt{3}R^2-\frac{\pi}{2}R^2=\left( \sqrt{3}-\frac{\pi}{2} \right)R^2.
\]

（2）如图，因为 \( \angle B=150^\circ \)，由菱形的性质可知 \( \angle A=30^\circ \)，依题意可得

2\left( \frac{1}{2}AB\cdot AD\cdot \sin A \right)=392.

因为 \( AB=AD \)，即 \( AB\cdot AB\cdot \sin 30^\circ=392 \)，\( \frac{1}{2}AB^2=392 \).

\[
AB^2=784,\quad AB=\sqrt{784}=28\mathrm{cm}.

\begin{problem}
在高 \( 150\mathrm{m} \) 的山顶上测得正南两船的俯角分别为 \( 15^\circ \) 和 \( 75^\circ \)，求两船的距离.

\end{problem}

\begin{solution}
\bitmapfigure[max width=5.8cm,max totalheight=4.4cm]{img/1977/hunan_zhuzhou_pilot/q10_solution_fig1.png}

两船的距离
\[
AB=AC-BC=150\tan \left( 90^\circ-15^\circ \right)-150\tan \left( 90^\circ-75^\circ \right)=150\left( \cot 15^\circ-\tan 15^\circ \right).
\]

而
\[
\sin 15^\circ=\sin \left( 45^\circ-30^\circ \right)=\sin 45^\circ \cos 30^\circ-\cos 45^\circ \sin 30^\circ=\frac{\sqrt{6}-\sqrt{2}}{4},
\]
\[
\cos 15^\circ=\cos \left( 45^\circ-30^\circ \right)=\cos 45^\circ \cos 30^\circ+\sin 45^\circ \sin 30^\circ=\frac{\sqrt{6}+\sqrt{2}}{4}.
\]

所以
\[
AB=150\left( \frac{\sqrt{6}+\sqrt{2}}{\sqrt{6}-\sqrt{2}}-\frac{\sqrt{6}-\sqrt{2}}{\sqrt{6}+\sqrt{2}} \right)=150\sqrt{12}=300\sqrt{3}\approx 519\mathrm{m}.
\]

答：两船的距离约为 \( 519\mathrm{m} \).
\end{solution}

\begin{problem}
有某种农药一桶，倒出 \( 8\text{升} \) 后，用水补满；然后又倒出 \( 4\text{升} \)，再用水补满. 于是测得桶中农药与水之比为 \( 18:7 \)，求木桶的容积.

\end{problem}

\begin{solution}
设桶的容积为 \( x\text{升} \)，依题意得
\[
\frac{x-8-\frac{x-8}{x}\times 4}{8-\frac{8}{x}\times 4+4}=\frac{18}{7}.
\]

即
\[
\frac{x^2-12x+32}{12x-32}=\frac{18}{7}.
\]

整理得
\[
7x^2-300x+800=0.
\]
\[
\left( 7x-20 \right)\left( x-40 \right)=0.
\]

所以 \( x_1=\frac{20}{7} \)（不合题意，舍去），\( x_2=40 \).

答：桶的容积为 \( 40\text{升} \).
\end{solution}

\begin{problem}
甲船在 \( A \) 点发现乙船在北偏东 \( 60^\circ \) 的 \( B \) 处，乙船以每小时 \( a\text{里} \) 的速度向北行驶，已知甲船速度是每小时 \( \sqrt{3}a\text{里} \)，问甲船应依什么方向前进，才能最快地与乙船相会？

\end{problem}

\begin{solution}
如图：在 \( \triangle ABC \) 中，

\bitmapfigure[max width=5.0cm,max totalheight=6.0cm]{img/1977/hunan_zhuzhou_pilot/q12_solution_fig1.png}

\[
\angle B=180^\circ-60^\circ=120^\circ.
\]

设 \( t \) 小时后两船相遇于 \( C \). 则 \( AC=\sqrt{3}at \)，\( BC=at \)，根据正弦定理得
\[
\frac{AC}{\sin B}=\frac{BC}{\sin A}.
\]

所以
\[
\sin A=\frac{BC\cdot \sin B}{AC}=\frac{at\sin 120^\circ}{\sqrt{3}at}=\frac{1}{2}.
\]

因为 \( A \) 是锐角，所以 \( A=30^\circ \)，\( \angle NAC=60^\circ-30^\circ=30^\circ \).

答：甲船应以北偏东 \( 30^\circ \) 的方向前进，才能最快地与乙船相会.
\end{solution}

\begin{problem}
双曲线中心在原点，焦点在坐标轴上；焦距为 \( 14 \)，顶点间距离为 \( 12 \)，求双曲线方程.

\end{problem}

\begin{solution}
设所求双曲线方程为
\[
\frac{x^2}{a^2}-\frac{y^2}{b^2}=1.
\]

因为顶点间距 \( 2a=12 \)，所以 \( a=6 \).

又焦距 \( 2c=14 \)，所以 \( c=7 \). 而 \( b^2=c^2-a^2=49-36=13 \).

故所求的方程为
\[
\frac{x^2}{36}-\frac{y^2}{13}=1.
\]

\bitmapfigure[max width=5.0cm,max totalheight=3.2cm]{img/1977/hunan_zhuzhou_pilot/q13_solution_fig1.png}

对于焦点在 \( y \) 轴上的双曲线，同理可求得方程为

\bitmapfigure[max width=3.2cm,max totalheight=4.6cm]{img/1977/hunan_zhuzhou_pilot/q13_solution_fig2.png}

\[
\frac{y^2}{36}-\frac{x^2}{13}=1.
\]
\end{solution}

\begin{problem}
从等腰 \( \triangle ABC \) 的顶点 \( A \) 引直线交 \( BC \) 于 \( D \)，交外接圆于 \( E \)，求证 \( AB \) 是 \( AD \) 和 \( AE \) 的比例中项.

\end{problem}

\begin{solution}
连结 \( BE \).

\bitmapfigure[max width=5.0cm,max totalheight=5.6cm,trim=1.0cm 0 0 0,clip]{img/1977/hunan_zhuzhou_pilot/q14_solution_fig1.png}

因为 \( \angle ABC=\angle ACB \)，而 \( \angle AEB=\angle ACB \)（同弧上的圆周角相等），所以 \( \angle ABC=\angle AEB \).

又 \( \angle BAE \) 是公共角，所以 \( \triangle ABD\sim \triangle AEB \).

故
\[
\frac{AE}{AB}=\frac{AB}{AD}.
\]

即 \( AB^2=AD\cdot AE \).
\end{solution}

\begin{problem}
如果一个直角三角形三边的长成等差数列，试证明三边的比是 \( 3:4:5 \).

\end{problem}

\begin{solution}
依题意，直角三角形三边为 \( a-d \)、\( a \)、\( a+d \)，公差为 \( d \)，如图所示.

\bitmapfigure[max width=4.4cm,max totalheight=4.8cm]{img/1977/hunan_zhuzhou_pilot/q15_solution_fig1.png}

因为
\[
a^2+\left( a-d \right)^2=\left( a+d \right)^2,
\]

所以
\[
a^2=\left( a+d \right)^2-\left( a-d \right)^2=a^2+2ad+d^2-a^2+2ad-d^2=4ad.
\]

因此
\[
d=\frac{a^2}{4a}=\frac{a}{4}.
\]

三边的比是
\[
\left( a-\frac{a}{4} \right):a:\left( a+\frac{a}{4} \right)=\frac{3a}{4}:a:\frac{5a}{4}=3:4:5.
\]
\end{solution}

\begin{problem}
试求 \( f(x)=x^4-42x^3+435x^2+113x-66 \) 和 \( g(x)=x^2-x-462 \) 的公根.

\end{problem}

\begin{solution}
解法一：\( g(x)=x^2-x-462=0 \) 的根是 \( x_1=22 \) 和 \( x_2=-21 \). 代入 \( f(x) \)：\( f(-21)\ne 0 \)，不是公根；\( f(22)=0 \). 因此 \( x=22 \) 是 \( f(x) \) 和 \( g(x) \) 的公根.

解法二：\( g(x)=x^2-x-462 \) 的根是 \( x_1=22 \) 和 \( x_2=-21 \). 用综合除法，\( \frac{f(x)}{x+21} \) 余数不为 \( 0 \)，因此 \( x_2=-21 \) 不是两者的公根. 而 \( \frac{f(x)}{x-22} \) 余数为 \( 0 \)，所以 \( x_1=22 \) 是两者的公根.

解法三：用辗转相除法求 \( f(x) \) 和 \( g(x) \) 的最高公因式，如下图所示.

\bitmapfigure[max width=0.92\linewidth,max totalheight=8.0cm,trim=0 1.4cm 0 1.6cm,clip]{img/1977/hunan_zhuzhou_pilot/q16_solution_fig1.png}

所以 \( -x+22 \) 是 \( f(x) \) 和 \( g(x) \) 的最高公因式.

故 \( -x+22=0 \)，即 \( x=22 \) 是 \( f(x) \) 和 \( g(x) \) 的公根.
\end{solution}

\begin{problem}
求函数 \( y=\left( 2x+1 \right)^{\frac{1}{2}}\left( -\frac{1}{3}x+1 \right)^3 \) 的导数.

\end{problem}

\begin{solution}
\begin{align*}
y'
=\;&\left( -\frac{1}{3}x+1 \right)^3\cdot \frac{1}{2}\left( 2x+1 \right)^{-\frac{1}{2}}\cdot 2 \\
&+\left( 2x+1 \right)^{\frac{1}{2}}\cdot 3\left( -\frac{1}{3}x+1 \right)^2\cdot \left( -\frac{1}{3} \right) \\
=\;&\left( 1-\frac{x}{3} \right)^3\cdot \left( 2x+1 \right)^{-\frac{1}{2}}-\left( 2x+1 \right)^{\frac{1}{2}}\left( 1-\frac{x}{3} \right)^2.
\end{align*}
\end{solution}

\begin{problem}
有一窗户，顶部为一抛物线的拱形，窗宽 \( 120\mathrm{cm} \)，边高 \( 150\mathrm{cm} \)，顶点高 \( 180\mathrm{cm} \)，求它的面积.

\end{problem}

\begin{solution}
取如图所示的直角坐标系，设抛物线方程为 \( x^2=-2py \).

\bitmapfigure[max width=5.8cm,max totalheight=6.4cm,trim=0 0 0 1.2cm,clip]{img/1977/hunan_zhuzhou_pilot/q18_solution_fig1.png}

将 \( A \) 点坐标 \( \left( 60,-30 \right) \) 代入上方程得 \( 60^2=-2p\left( -30 \right) \)，得 \( p=60 \).

所以抛物线方程为
\[
x^2=-2\times 60y,
\]

亦可变成
\[
y=-\frac{1}{120}x^2.
\]

图中所影部分面积为
\[
\int_0^{60}\frac{x^2}{120}\,\mathrm{d}x=\frac{1}{360}x^3\Big|_0^{60}=\frac{1}{360}\times 60^3=600.
\]

窗户面积
\[
S=2\left( 180\times 60-600 \right)=2\times 10200=20400\mathrm{cm}^2=2.04\mathrm{m}^2.
\]

答：所求窗户的面积是 \( 2.04\mathrm{m}^2 \).
\end{solution}
