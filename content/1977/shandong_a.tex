\begin{problem}
\(\sqrt{\left( 3-x \right)^2}=?\)

\end{problem}

\begin{solution}
\[
\sqrt{\left( 3-x \right)^2}=
\begin{cases}
3-x, & x\le 3,\\
x-3, & x>3.
\end{cases}
\]
\end{solution}

\begin{problem}
\(\lg 2+\lg 5+\lg \sqrt{10}+\lg 0.01+\log_8\log_3 3=?\)

\end{problem}

\begin{solution}
原式 \(=\lg 2+\lg 5+\frac{1}{2}\lg 10-2\lg 10+\log_8 1\)
\[
=\lg 10+\frac{1}{2}\lg 10-2\lg 10+0=1+\frac{1}{2}-2=-\frac{1}{2}
\]
\end{solution}

\begin{problem}
写出函数定义域：

\begin{enumerate}
\item \(y=\log_a x\)（\(a>0\)）；
\item \(y=\frac{\sqrt{x-1}}{x+3}\)
\end{enumerate}

\end{problem}

\begin{solution}
（1）\(y=\log_a x\)（\(a>0\)）的定义域为 \(x>0\).

（2）\(y=\frac{\sqrt{x-1}}{x+3}\) 的定义域为 \(x\ge 1\).
\end{solution}

\begin{problem}
写出过点 \((2,3)\) 斜率为 \(-\frac{1}{2}\) 的直线方程.

\end{problem}

\begin{solution}
根据点斜式，所求的直线方程为 \(y-3=-\dfrac{1}{2}\left( x-2 \right)\)，即 \(y=-\dfrac{1}{2}x+4\).
\end{solution}

\begin{problem}
求椭圆 \(\frac{x^2}{25}+\frac{y^2}{16}=1\) 的焦点坐标.

\end{problem}

\begin{solution}
椭圆方程可写为 \(\frac{x^2}{5^2}+\frac{y^2}{4^2}=1\). 其长半轴 \(a=5\)，短半轴 \(b=4\). 因此 \(c=\sqrt{5^2-4^2}=3\)，所以焦点坐标为 \(F_1\left(-3,0\right)\)，\(F_2\left(3,0\right)\).
\end{solution}

\begin{problem}
设 \(AD\) 是 \(\triangle ABC\) 的高，在线段 \(BC\) 上取点 \(E\)，使 \(\angle EAC=\angle BAD\)，延长 \(AE\) 交 \(\triangle ABC\) 外接圆于点 \(F\). 求证：\(AF\) 是外接圆的直径.

\bitmapfigure[max width=4.9cm]{img/1977/shandong_a/q7_fig1.png}

\end{problem}

\begin{solution}
连接 \(FC\).

由于 \(A\)，\(E\)，\(F\) 三点共线，\(\angle FAC=\angle EAC=\angle BAD\). 又因为 \(\angle ABC=\angle AFC\)（同圆中，同弧上圆周角相等），且 \(\angle ADB=90^\circ\)，所以在 \(\triangle ABD\) 和 \(\triangle ACF\) 中，有
\[
\angle ACF=180^\circ-\angle FAC-\angle AFC
=180^\circ-\angle BAD-\angle ABC
=\angle ADB=90^\circ
\]

因此 \(AF\) 是 \(\triangle ABC\) 外接圆的直径.
\end{solution}

\begin{problem}
\(x^2-x\cos \alpha+\frac{1}{4}\sin^2 \alpha=0\)（\(0<\alpha<\frac{\pi}{2}\)）为关于 \(x\) 的二次方程，当 \(\alpha\) 为何值时，方程有两个不相等的实数根？它的根是什么？

\end{problem}

\begin{solution}
当二次方程的根的判别式 \(\Delta>0\) 时，方程 \(x^2-x\cos \alpha+\frac{1}{4}\sin^2 \alpha=0\) 有两个不相等的实数根. 由 \(\Delta=\cos^2 \alpha-\sin^2 \alpha>0\) 得 \(\cos 2\alpha>0\)，所以 \(0<2\alpha<\frac{\pi}{2}\)，即 \(0<\alpha<\frac{\pi}{4}\).

这时方程的两个实数根是 \(x_1=\frac{\cos \alpha+\sqrt{\cos 2\alpha}}{2}\)，\(x_2=\frac{\cos \alpha-\sqrt{\cos 2\alpha}}{2}\).

答：当 \(0<\alpha<\frac{\pi}{4}\) 时，方程 \(x^2-x\cos \alpha+\frac{1}{4}\sin^2 \alpha=0\) 有两个不相等的实数根 \(x=\frac{\cos \alpha\pm\sqrt{\cos 2\alpha}}{2}\).
\end{solution}

\begin{problem}
在半径为 \(R\) 的一块圆形铁板上，剪去一个圆心角为 \(\alpha\) 弧度的扇形，用剩下的铁板做一个圆锥形容器（铁板厚度和加工量不计）.

\begin{enumerate}
\item 用 \(R\) 和 \(\alpha\) 表示圆锥底面半径 \(r\)，圆锥高 \(h\)，圆锥容积 \(V\).
\item 当 \(R=20\mathrm{~cm}\)，\(\alpha=\frac{2\pi}{5}\) 弧度时，计算圆锥容积 \(V\)（取 \(\pi=3.14\)，精确到 \(1\mathrm{~cm^3}\)）.
\end{enumerate}

\end{problem}

\begin{solution}
（1）因为 \(2\pi r=R\left( 2\pi-\alpha \right)\)，所以 \(r=\dfrac{2\pi-\alpha}{2\pi}R\). 而
\[
h^2=R^2-r^2=R^2-\left( \frac{2\pi-\alpha}{2\pi}R \right)^2=R^2-\frac{\left( 2\pi-\alpha \right)^2}{4\pi^2}R^2
=R^2\left[1-\frac{\left( 2\pi-\alpha \right)^2}{4\pi^2}\right]=\frac{4\pi \alpha-\alpha^2}{4\pi^2}R^2
\]
所以 \(h=\dfrac{\sqrt{4\pi \alpha-\alpha^2}}{2\pi}R\)，所以
\[
V=\frac{1}{3}\pi r^2 h=\frac{1}{3}\pi \left( \frac{2\pi-\alpha}{2\pi}R \right)^2\cdot \frac{\sqrt{4\pi \alpha-\alpha^2}}{2\pi}R
=\frac{R^3\left( 2\pi-\alpha \right)^2}{24\pi^2}\sqrt{4\pi \alpha-\alpha^2}
\]

（2）当 \(R=20\mathrm{~cm}\)，\(\alpha=\frac{2\pi}{5}\) 时，
\[
V=\frac{20^3\left( 2\pi-\frac{2\pi}{5} \right)^2}{24\pi^2}\sqrt{\left( 4\pi-\frac{2\pi}{5} \right)\cdot \frac{2\pi}{5}}
=\frac{20^3\cdot \frac{64}{25}\pi^2}{24\pi^2}\cdot \sqrt{\frac{36\pi^2}{25}}
=1024\pi\approx3215\mathrm{~cm^3}
\]
答：当 \(R=20\mathrm{~cm}\)，\(\alpha=\frac{2\pi}{5}\) 时，圆锥的容积约为 \(3215\mathrm{~cm^3}\).
\end{solution}

\begin{problem}
我炮兵阵地位于 \(A\)，两观察所分别设于 \(C\)，\(D\)，已知 \(\triangle ACD\) 为一正三角形，且 \(CD=a\)，当目标出现于 \(B\) 时，测得 \(\angle CDB=45^\circ\)，\(\angle BCD=75^\circ\)，求炮目距离 \(AB\).

\bitmapfigure[max width=6.0cm]{img/1977/shandong_a/q9_fig1.png}

\end{problem}

\begin{solution}
\(\angle B=180^\circ-\left( 45^\circ+75^\circ \right)=60^\circ\).

在 \(\triangle BCD\) 中，由正弦定理知
\[
BC=\frac{a\sin 45^\circ}{\sin 60^\circ}
=\frac{a\cdot \frac{\sqrt{2}}{2}}{\frac{\sqrt{3}}{2}}
=\frac{\sqrt{6}}{3}a
\]

在 \(\triangle ABC\) 中，根据余弦定理
\[
\begin{aligned}
AB&=\sqrt{AC^2+BC^2-2\cdot AC\cdot BC\cdot \cos \angle ACB} \\
  &=\sqrt{a^2+\frac{2}{3}a^2-2a\cdot \frac{\sqrt{6}}{3}a\cos \left( 75^\circ+60^\circ \right)} \\
  &=\sqrt{\frac{5}{3}a^2+\frac{2\sqrt{6}}{3}a^2\cos 45^\circ}
    =\sqrt{\frac{5+2\sqrt{3}}{3}}a
    =\frac{\sqrt{15+6\sqrt{3}}}{3}a.
\end{aligned}
\]

答：炮目距离 \(AB=\frac{\sqrt{15+6\sqrt{3}}}{3}a\).
\end{solution}

\begin{problem}
现有可建猪圈 \(L\) 米的材料，要建彼此相连的 \(m\) 间面积相等的矩形猪圈（如下图）（门在内，墙的厚度不计）. 怎样设计每间猪圈的长和宽，才能使所建的猪圈的总面积最大？按这种设计，\(m\) 间猪圈长的总和与宽的总和有什么关系？

\bitmapfigure[max width=5.8cm]{img/1977/shandong_a/q10_fig1.png}

\end{problem}

\begin{solution}
设此猪圈的长为 \(x\) 米，则宽就为 \(\frac{L-2x}{m+1}\) 米. 猪圈的总面积 \(S=x\cdot\dfrac{L-2x}{m+1}=-\dfrac{2}{m+1}x^2+\dfrac{L}{m+1}x\)，是长度 \(x\) 的二次函数，其中 \(a=-\dfrac{2}{m+1}\)，\(b=\dfrac{L}{m+1}\)，\(c=0\).

因为 \(a<0\)，所以当
\[
x=-\frac{b}{2a}=\frac{\frac{L}{m+1}}{\frac{4}{m+1}}=\frac{L}{4}
\]
时，面积 \(S\) 最大.
\[
S_{\max}=\frac{4ac-b^2}{4a}=\frac{-\left( \frac{L}{m+1} \right)^2}{-\frac{8}{m+1}}=\frac{L^2}{8\left( m+1 \right)}
\]

这时，宽 \(\frac{L-2x}{m+1}=\frac{L}{2\left( m+1 \right)}\)，每间的长 \(\frac{x}{m}=\frac{L}{4m}\). 猪圈长的总和 \(=2x=\frac{L}{2}\)，宽的总和 \(=L-\frac{L}{2}=\frac{L}{2}\).

答：每间猪圈的长为 \(\frac{L}{4m}\)，宽为 \(\frac{L}{2\left( m+1 \right)}\) 时，才能使所建猪圈总面积最大. 按这种设计，\(m\) 间猪圈长的总和与宽的总和相等.
\end{solution}

\begin{problem}
求极限：\(\lim\limits_{x\to\infty}\frac{\e^x}{x^2}\)

\end{problem}

\begin{solution}
根据洛必达法则 \(\lim\limits_{x\to\infty}\frac{\e^x}{x^2}=\lim\limits_{x\to\infty}\frac{\e^x}{2x}=\lim\limits_{x\to\infty}\frac{\e^x}{2}=+\infty\). 因此该函数不存在有限极限.
\end{solution}

\begin{problem}
利用定积分证明：椭圆 \(\frac{x^2}{a^2}+\frac{y^2}{b^2}=1\) 的面积 \(S=\pi ab\).

\end{problem}

\begin{solution}
先求椭圆在第一象限内的面积 \(S'=\displaystyle\int_0^a b\sqrt{1-\frac{x^2}{a^2}}\,\mathrm{d}x=\frac{b}{a}\int_0^a \sqrt{a^2-x^2}\,\mathrm{d}x\).

令 \(x=a\sin t\)，则 \(\sqrt{a^2-x^2}=a\cos t\)，\(\mathrm{d}x=a\cos t\,\mathrm{d}t\)（\(0\le t\le \frac{\pi}{2}\)）.
因此
\[
\begin{aligned}
S'&=\frac{b}{a}\int_0^{\frac{\pi}{2}}a\cos t\cdot a\cos t\,\mathrm{d}t
=ab\int_0^{\frac{\pi}{2}}\cos^2 t\,\mathrm{d}t
=ab\int_0^{\frac{\pi}{2}}\frac{\cos 2t+1}{2}\,\mathrm{d}t \\
&=ab\left( \frac{\sin 2t}{4}+\frac{t}{2} \right)\bigg|_0^{\frac{\pi}{2}}
=\frac{ab\pi}{4}.
\end{aligned}
\]

由对称性，椭圆的面积 \(S=4S'=ab\pi\).
\end{solution}
