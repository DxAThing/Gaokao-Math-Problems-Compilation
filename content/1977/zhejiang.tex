\chapter{1977年浙江卷}

一，计算：

\begin{problem}
\( \sqrt[6]{8^5}\div\sqrt{8}\times\sqrt[3]{8}\div\left( \frac{1}{8} \right)^0 \).

\end{problem}

\begin{solution}
\begin{align*}
\text{原式}
=\;&\sqrt[6]{2^{15}}\div\sqrt{2^3}\times\sqrt[3]{2^3}\div 1 \\
=\;&4\sqrt{2}\div 2\sqrt{2}\times 2\div 1 \\
=\;&4.
\end{align*}
\end{solution}

\begin{problem}
\( \frac{\lg 4+\lg 25-\lg 10}{\lg \sqrt{1000}} \).

\end{problem}

\begin{solution}
\begin{align*}
\text{原式}
=\;&\frac{\lg \frac{4\times 25}{10}}{\lg 10^{\frac{3}{2}}} \\
=\;&\frac{\lg 10}{\frac{3}{2}\lg 10} \\
=\;&\frac{2}{3}.
\end{align*}
\end{solution}

\begin{problem}
\( \frac{1}{100\times 101}+\frac{1}{101\times 102}+\frac{1}{102\times 103}+\cdots+\frac{1}{149\times 150} \).

\end{problem}

\begin{solution}
\begin{align*}
\text{原式}
=\;&\left( \frac{1}{100}-\frac{1}{101} \right)+\left( \frac{1}{101}-\frac{1}{102} \right)+\left( \frac{1}{102}-\frac{1}{103} \right)+\cdots+\left( \frac{1}{149}-\frac{1}{150} \right) \\
=\;&\frac{1}{100}-\frac{1}{150} \\
=\;&\frac{1}{300}.
\end{align*}
\end{solution}

二，

\begin{problem}
已知 \( \tan\alpha=\sqrt{2} \)，求 \( \frac{\cos\alpha+\sin\alpha}{\cos\alpha-\sin\alpha} \) 的值.

\end{problem}

\begin{solution}
\begin{align*}
\frac{\cos\alpha+\sin\alpha}{\cos\alpha-\sin\alpha}
=\;&\frac{\cos\alpha\left( 1+\frac{\sin\alpha}{\cos\alpha} \right)}{\cos\alpha\left( 1-\frac{\sin\alpha}{\cos\alpha} \right)} \\
=\;&\frac{1+\tan\alpha}{1-\tan\alpha} \\
=\;&\frac{1+\sqrt{2}}{1-\sqrt{2}} \\
=\;&-3-2\sqrt{2}.
\end{align*}
\end{solution}

\begin{problem}
求 \( \sin^2 62^\circ+\tan54^\circ\cdot\tan45^\circ\cdot\tan36^\circ+\sin^2 28^\circ \) 的值.

\end{problem}

\begin{solution}
\[
\text{原式}
=\sin^2 62^\circ+\cos^2 62^\circ+\tan54^\circ\cdot\tan45^\circ\cdot\cot54^\circ
=1+1=2.
\]
\end{solution}

\begin{problem}
某生产队要建一个贮粮用的“土圆仓”，如果土圆仓下部圆柱和上部圆锥的底面半径都是 1.5 米，高分别是 3 米和 0.6 米，计算它的容积（\( \pi\approx 3.14 \)，精确到 0.1 立方米）.

\bitmapfigure[max width=6.0cm,max totalheight=5.2cm]{img/1977/zhejiang/q6_fig1.png}

\end{problem}

\begin{solution}
若以 \( h_1 \)，\( h_2 \) 分别表示圆柱的高和上面圆锥的高，则“粮仓”的容积
\begin{align*}
V
=\;&V_{\text{圆柱}}+V_{\text{圆锥}}=\pi r^2 h_1+\frac{1}{3}\pi r^2 h_2 \\
=\;&\pi r^2\left( h_1+\frac{1}{3}h_2 \right) \\
=\;&3.14\times 1.5^2\left( 3+\frac{1}{3}\times 0.6 \right) \\
\approx\;&22.6\text{（立方米）}.
\end{align*}

答：“土圆仓”的容积约为 22.6 立方米.
\end{solution}

三，

\begin{problem}
求平行直线 \( 3x+2y-3=0 \) 和 \( 3x+2y+2=0 \) 之间的距离.

\end{problem}

\begin{solution}
点 \( P(1,0) \) 是直线 \( 3x+2y-3=0 \) 上的点. 点 \( P \) 到直线 \( 3x+2y+2=0 \) 的距离为 \( d=\frac{|3\times 1+2\times 0+2|}{\sqrt{3^2+2^2}}=\frac{5\sqrt{13}}{13} \). 因为这两条直线平行，所求得的 \( d=\frac{5\sqrt{13}}{13} \) 就是这两条平行直线间的距离.
\end{solution}

四，

\begin{problem}
已知 \( \alpha \) 和 \( \beta \) 是三角形的两个内角，并且 \( \alpha \) 是锐角，而 \( \cos\alpha=\frac{3}{5} \)，\( \sin\beta=\frac{5}{13} \)，求 \( \sin(\alpha+\beta) \) 的值.

\end{problem}

\begin{solution}
因为 \( \alpha \) 为锐角，且 \( \cos\alpha=\frac{3}{5} \)，所以 \( \sin\alpha=\frac{4}{5} \).

因为 \( \sin\beta=\frac{5}{13} \)，\( \beta \) 可能是锐角，也可能是钝角，所以 \( \cos\beta=\pm\frac{12}{13} \).

（1）\( \beta \) 为锐角时，\( \cos\beta=\frac{12}{13} \)，
\[
\sin(\alpha+\beta)=\sin\alpha\cos\beta+\cos\alpha\sin\beta=\frac{4}{5}\cdot\frac{12}{13}+\frac{3}{5}\cdot\frac{5}{13}=\frac{63}{65}.
\]

（2）\( \beta \) 为钝角时，\( \cos\beta=-\frac{12}{13} \)，
\[
\sin(\alpha+\beta)=\sin\alpha\cos\beta+\cos\alpha\sin\beta=\frac{4}{5}\cdot\left( -\frac{12}{13} \right)+\frac{3}{5}\cdot\frac{5}{13}=-\frac{33}{65}.
\]

因为 \( \alpha \)，\( \beta \) 是三角形两内角，\( 0^\circ<\alpha+\beta<180^\circ \)，\( \sin(\alpha+\beta) \) 不可能是负值，所以 \( \beta \) 为钝角这种情况应舍去.

故 \( \sin(\alpha+\beta)=\frac{63}{65} \).
\end{solution}

五，

\begin{problem}
解不等式：\( \log_5\left( x^2-2x-15 \right)<\log_5\left( x+13 \right) \).

\end{problem}

\begin{solution}
根据对数的定义和性质，要使原不等式成立，\( x \) 必须同时满足下列不等式：
\[
\begin{cases}
x^2-2x-15<x+13 \\
x^2-2x-15>0 \\
x+13>0
\end{cases}
\]
\[
\begin{cases}
(x+4)(x-7)<0 \\
(x+3)(x-5)>0 \\
x>-13
\end{cases}
\]
\[
\begin{cases}
-4<x<7 \\
x>5\text{ 或者 }x<-3 \\
x>-13
\end{cases}
\]

所以原不等式的解为：\( 5<x<7 \) 和 \( -4<x<-3 \).
\end{solution}

六，

\begin{problem}
点 \( O \) 为 \( \triangle ABC \) 内任意一点，连接 \( AO \) 并延长交 \( BC \) 于点 \( D \)，连接 \( BO \) 并延长交 \( AC \) 于点 \( E \)，连接 \( CO \) 并延长交 \( AB \) 于点 \( F \)，则 \( \frac{OD}{AD}+\frac{OE}{BE}+\frac{OF}{CF}=1 \).

\bitmapfigure[max width=6.2cm,max totalheight=4.8cm]{img/1977/zhejiang/q10_fig1.png}

\end{problem}

\begin{solution}
证明：作 \( \triangle ABC \) 和 \( \triangle OBC \) 的高 \( AK \) 和 \( OH \).

因此 \( \mathrm{Rt}\triangle AKD\sim \mathrm{Rt}\triangle OHD \)，\( \frac{OH}{AK}=\frac{OD}{AD} \).

所以
\[
\frac{\triangle OBC\text{面积}}{\triangle ABC\text{面积}}=\frac{\frac{1}{2}BC\cdot OH}{\frac{1}{2}BC\cdot AK}=\frac{OH}{AK}=\frac{OD}{AD}.
\]

同理，有 \( \frac{\triangle OAC\text{面积}}{\triangle ABC\text{面积}}=\frac{OE}{BE} \)，\( \frac{\triangle OAB\text{面积}}{\triangle ABC\text{面积}}=\frac{OF}{CF} \).

所以
\[
\frac{OD}{AD}+\frac{OE}{BE}+\frac{OF}{CF}
=\frac{\triangle OBC\text{面积}+\triangle OAC\text{面积}+\triangle OAB\text{面积}}{\triangle ABC\text{面积}}
=\frac{\triangle ABC\text{面积}}{\triangle ABC\text{面积}}
=1.
\]

即 \( \frac{OD}{AD}+\frac{OE}{BE}+\frac{OF}{CF}=1 \).
\end{solution}

七，

\begin{problem}
设 \( x \)，\( y \)，\( z \) 为三个互不相等的实数，且 \( x+\frac{1}{y}=y+\frac{1}{z}=z+\frac{1}{x} \)，求证：\( x^2y^2z^2=1 \).

\end{problem}

\begin{solution}
证明：由 \( x+\frac{1}{y}=y+\frac{1}{z} \)，得 \( x-y=\frac{1}{z}-\frac{1}{y}=\frac{y-z}{yz} \)，即 \( yz=\frac{y-z}{x-y} \).

同理可得 \( xy=\frac{x-y}{z-x} \)，\( zx=\frac{z-x}{y-z} \).

所以 \( yz\cdot xy\cdot zx=\frac{y-z}{x-y}\cdot\frac{x-y}{z-x}\cdot\frac{z-x}{y-z}=1 \).

即 \( x^2y^2z^2=1 \).
\end{solution}


\begin{problem}
将抛物线 \( y=x^2 \)，直线 \( x=2 \) 与 \( x \) 轴围成的曲边三角形，绕 \( x \) 轴旋转一周形成一个旋转体，求此旋转体的体积.

\end{problem}

\begin{solution}
旋转体的体积
\[
V=\int_0^2 \pi y^2\,\mathrm{d}x=\int_0^2 \pi x^4\,\mathrm{d}x=\left. \frac{\pi x^5}{5} \right|_0^2=\frac{32}{5}\pi.
\]
\end{solution}

\begin{problem}
求不定积分 \(\int \frac{6x^2+22x+18}{(x+1)(x+2)(x+3)}\,\mathrm{d}x\).

\end{problem}

\begin{solution}
\[
\int \frac{6x^2+22x+18}{(x+1)(x+2)(x+3)}\,\mathrm{d}x
=\int \left( \frac{1}{x+1}+\frac{2}{x+2}+\frac{3}{x+3} \right)\,\mathrm{d}x
\]
\[
=\ln\left| x+1 \right|+2\ln\left| x+2 \right|+3\ln\left| x+3 \right|+C
\]
\[
=\ln\left| (x+1)(x+2)^2(x+3)^2 \right|+C.
\]

其中 \( C \) 是任意常数.
\end{solution}
