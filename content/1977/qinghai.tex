\chapter{1977年青海卷}

\begin{problem}
计算：\( \frac{a}{a-b}-\frac{b}{a+b}+\frac{2ab}{a^2-b^2} \)（\( a\neq\pm b \)）.

\end{problem}

\begin{solution}
\[
\text{原式}=\frac{a\left( a+b \right)-b\left( a-b \right)+2ab}{\left( a-b \right)\left( a+b \right)}=\frac{a^2+b^2+2ab}{\left( a-b \right)\left( a+b \right)}=\frac{\left( a+b \right)^2}{\left( a-b \right)\left( a+b \right)}=\frac{a+b}{a-b}
\]
\end{solution}

\begin{problem}
如图，\(AB\) 是 \(\odot O\) 的直径，\(AC\) 是 \(\odot O\) 的切线，连接 \(BC\) 与 \(\odot O\) 相交于 \(D\) 点，而且 \(\angle ABC=45^\circ\). 求证 \(BD=DC\).

\bitmapfigure[max width=4.8cm]{img/1977/qinghai/q2_fig1.png}

\end{problem}

\begin{solution}
连接 \(AD\).
因为 \(AB\) 是直径，所以 \( \angle ADB=90^\circ \). 又因为 \( \angle ABC=45^\circ \)，所以 \( \angle BAD=45^\circ \). 所以 \(BD=AD\).
因为 \(AC\) 是切线，所以 \( \angle BAC=90^\circ \). 所以 \( \angle DAC=45^\circ \). 而 \( \angle C=90^\circ-45^\circ=45^\circ \). 所以 \(DC=AD\).
从而得 \(BD=DC\).
\end{solution}

\begin{problem}
已知长方体 \(ABCD-A'B'C'D'\) 的对角线 \(AC'\) 与 \(A'C\) 的夹角 \( \angle AOC=120^\circ \)，底面边长 \(AB=4\mathrm{cm}\)，\(BC=3\mathrm{cm}\). 求此长方体的体积.

\bitmapfigure[max width=5.8cm]{img/1977/qinghai/q3_fig1.png}

\end{problem}

\begin{solution}
因为 \( \angle AOC=120^\circ \)，所以 \( \angle COC'=180^\circ-120^\circ=60^\circ \). 又因为 \(CO=C'O\)，所以 \( \angle OCC'=\angle OC'C=60^\circ \).
连接 \(AC\)，在 \(\mathrm{Rt}\triangle ACC'\) 中，\(\angle C'AC=90^\circ-60^\circ=30^\circ\)，所以 \(AC'=2C'C\).
根据勾股定理，
\[
AC=\sqrt{AB^2+BC^2}=\sqrt{4^2+3^2}
=5
\]
于是 \((2C'C)^2-5^2=C'C^2\)，即 \(3C'C^2=25\). 所以 \(C'C=\frac{5\sqrt{3}}{3}\mathrm{cm}\). 所以求得长方体的体积
\[
V=AB\times BC\times C'C=4\times3\times\frac{5\sqrt{3}}{3}
=20\sqrt{3}\mathrm{cm}^3
\]
答：长方体的体积为 \(20\sqrt{3}\mathrm{cm}^3\).
\end{solution}

\begin{problem}
求下列函数的定义域：

\begin{enumerate}
    \item \( y=\frac{7x}{x+1} \).
    \item \( y=\frac{1}{\sqrt{2-x}} \).
\end{enumerate}

\end{problem}

\begin{solution}
（1）函数 \( y=\frac{7x}{x+1} \) 的定义域为：\(x\) 不等于 \(-1\) 的任意实数.

（2）函数 \( y=\frac{1}{\sqrt{2-x}} \) 的定义域为 \(x<2\) 的实数.
\end{solution}

\begin{problem}
化简：\( \left( 27a^{\frac{5}{2}}\sqrt{ab^{-\frac{1}{3}}\sqrt[4]{b^{\frac{4}{3}}}} \right)^{\frac{1}{3}} \)（\( b\neq0 \)）.

\end{problem}

\begin{solution}
\[
\text{原式}=\left[ 27a^{\frac{5}{2}}\left( ab^{-\frac{1}{3}}b^{\frac{1}{3}} \right)^{\frac{1}{2}} \right]^{\frac{1}{3}}=\left[ 27a^{\frac{5}{2}}a^{\frac{1}{2}} \right]^{\frac{1}{3}}=\left( 27a^3 \right)^{\frac{1}{3}}=3a
\]
\end{solution}

\begin{problem}
求 \( \sin435^\circ \) 的值.

\end{problem}

\begin{solution}
\[
\begin{aligned}
\sin435^\circ&=\sin\left( 360^\circ+75^\circ \right)=\sin75^\circ=\sin\left( 45^\circ+30^\circ \right)\\
&=\sin45^\circ\cos30^\circ+\cos45^\circ\sin30^\circ=\frac{\sqrt{2}}{2}\cdot\frac{\sqrt{3}}{2}+\frac{\sqrt{2}}{2}\cdot\frac{1}{2}=\frac{\sqrt{6}+\sqrt{2}}{4}.
\end{aligned}
\]
\end{solution}

\begin{problem}
当锐角 \( \theta \) 为何值时，方程 \( \left( 3\sin \theta \right)x^2-\left( 4\cos \theta \right)x+2=0 \) 有两个相等的实数根？并求出这个方程的实数根.

\end{problem}

\begin{solution}
要使方程有两个相等的实数根，必须使其判别式为 \(0\). 即 \((4\cos\theta)^2-8(3\sin\theta)=0\). 所以 \(16\cos^2\theta-24\sin\theta=0\). 即 \(2\cos^2\theta-3\sin\theta=0\).
又因为 \(\cos^2\theta=1-\sin^2\theta\)，所以 \(2(1-\sin^2\theta)-3\sin\theta=0\). 即 \(2\sin^2\theta+3\sin\theta-2=0\). 因此 \((2\sin\theta-1)(\sin\theta+2)=0\).
所以 \(\sin\theta=\frac12\) 或者 \(\sin\theta=-2\)（不适合，舍去）. 已知 \(\theta\) 是锐角，所以 \(\theta=30^\circ\).
这时，原方程就化为 \(\frac32x^2-2\sqrt3x+2=0\). 或者写成 \(3x^2-4\sqrt3x+4=0\). 即 \((\sqrt3x-2)^2=0\). 所以 \(x=\frac{2}{\sqrt3}=\frac{2\sqrt3}{3}\).
答：当 \(\theta=30^\circ\) 时，原方程有两个相等的实数根. 这两个相等的实数根是 \(\frac{2\sqrt3}{3}\).
\end{solution}

\begin{problem}
求经过抛物线 \( y^2=-2x \) 的焦点，且与直线 \( 2x-y-1=0 \) 垂直的直线方程.

\end{problem}

\begin{solution}
抛物线 \( y^2=-2x \) 的焦点是 \( \left( -\frac{1}{2},0 \right) \)，直线 \( 2x-y-1=0 \) 的斜率为 \(2\)，因此所求的直线过 \( \left( -\frac{1}{2},0 \right) \) 点，而且斜率为 \( -\frac{1}{2} \). 由点斜式直线方程可求得所求的直线是 \( y-0=-\frac{1}{2}\left( x+\frac{1}{2} \right) \)，即 \( 2x+4y+1=0 \).
\end{solution}

\begin{problem}
今年，我省香日德农场春小麦亩产达 \( 1765.5 \) 斤，创全国春小麦亩产新记录，在大治之年作出了贡献. 我省在 \( 1971 \) 年开始出现亩产 \( 1000 \) 斤的高产田. 从 \( 1971 \) 年的高产量 \( 1000 \) 斤到今年的创记录产量 \( 1765.5 \) 斤，平均每年的增长率是多少？（已知 \( \lg1.766=0.2470 \)，已知由 \( \lg A=0.0412 \) 得 \( A=1.100 \)）

\end{problem}

\begin{solution}
设平均每年的增长率为 \(x\)，则 \(1000(1+x)^6=1765.5\). 所以 \((1+x)^6=1.7655\approx1.766\). 于是 \(6\lg(1+x)=\lg1.766=0.2470\). 所以 \(\lg(1+x)=\frac16\times0.2470=0.0412\). 因此 \(1+x=1.100\)，所以 \(x=0.100\approx10.0\%\).

答：平均每年的增长率约为 \(10.0\%\).
\end{solution}


\begin{problem}
若齐次线性方程组 \(\begin{cases}
x+ay+a^2z=0, \\
ax+a^2y+z=0, \\
a^2x+y+az=0
\end{cases}\) 有非零解，那末 \(a\) 应为何值？

\end{problem}

\begin{solution}
齐次线性方程组有非零解，其充要条件是系数行列式等于 \(0\). 即要求
\(\begin{vmatrix}1 & a & a^2 \\ a & a^2 & 1 \\ a^2 & 1 & a\end{vmatrix}=0\).
而
\[
\begin{vmatrix}1 & a & a^2 \\ a & a^2 & 1 \\ a^2 & 1 & a\end{vmatrix}=a^3+a^3+a^3-a^6-1-a^3=2a^3-a^6-1
\]
所以 \(2a^3-a^6-1=0\)，即 \((a^3-1)^2=0\). 所以 \(a=1\).

答：当 \(a=1\) 时，这个齐次线性方程组有非零解.
\end{solution}

\begin{problem}
试求由曲线 \(y=2\cos x\) 之一半被与 \(y=0\) 围成的面积绕 \(ox\) 轴旋转所产生的立体的体积.

\end{problem}

\begin{solution}
所求的体积
\[
V=\int_{-\frac{\pi}{2}}^{\frac{\pi}{2}}\pi y^2 \,\mathrm{d}x
=\int_{-\frac{\pi}{2}}^{\frac{\pi}{2}}4\pi\cos^2x\,\mathrm{d}x
=\int_{-\frac{\pi}{2}}^{\frac{\pi}{2}}2\pi(1+\cos2x)\,\mathrm{d}x
\]
所以 \((2\pi x+\pi\sin2x)\big|_{-\frac{\pi}{2}}^{\frac{\pi}{2}}=2\pi^2.\)
\end{solution}
