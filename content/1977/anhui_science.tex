\begin{problem}
\begin{enumerate}
    \item 计算：\(0.25\times\left( -2 \right)^2-4\div\left( \sqrt{5}-1 \right)^0-\left( \frac{1}{6} \right)^{-\frac{1}{2}}+\frac{\sqrt{3}}{\sqrt{3}-\sqrt{2}}\).
    \item 求函数 \(y=\log_x\left( 4^x-16 \right)\) 的定义域.
    \item 求 \(\sin\left( -915^\circ \right)\) 的值.
    \item 求函数 \(y=\arcsin\left( 1-2x \right)\) 的定义域.
    \item 化复数 \(-1+\i\) 为三角函数式.
    \item 化简：\(\sec A\cdot\sqrt{1-\sin^2 A}\)（\(90^\circ<A\leqslant180^\circ\)）.
    \item 已知圆锥的母线长为 \(3 \mathrm{~cm}\)，底面直径 \(2 \mathrm{~cm}\)，求它的侧面积（\(\pi\) 取 \(3.14\)）.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 原式
\[
=\frac{1}{4}\times4-4\div1-\sqrt{6}+\sqrt{3}\left( \sqrt{3}+\sqrt{2} \right)
=1-4-\sqrt{6}+3+\sqrt{6}=0
\]

\item 底数须满足 \(x>0\) 且 \(x\neq1\)，真数须满足 \(4^x-16>0\). 解不等式组
\[
\left\{
\begin{aligned}
x&>0, \\
x&\neq1, \\
4^x-16&>0
\end{aligned}
\right.{}
\]
得 \(x>2\). 所以函数 \(y=\log_x\left( 4^x-16 \right)\) 的定义域为 \(x>2\).

\item
\[
\begin{aligned}
\sin\left( -915^\circ \right)&=-\sin915^\circ=-\sin\left( 360^\circ\times2+195^\circ \right)=-\sin195^\circ \\
&=-\sin\left( 180^\circ+15^\circ \right)=\sin15^\circ=\sin\left( 45^\circ-30^\circ \right) \\
&=\sin45^\circ\cos30^\circ-\cos45^\circ\sin30^\circ=\frac{\sqrt{2}}{2}\cdot\frac{\sqrt{3}}{2}-\frac{\sqrt{2}}{2}\cdot\frac{1}{2}=\frac{\sqrt{6}-\sqrt{2}}{4}
\end{aligned}
\]

\item 要使函数 \(y=\arcsin\left( 1-2x \right)\) 有意义，必须
\(-1\leqslant1-2x\leqslant1\).
解得 \(0\leqslant x\leqslant1\). 所以定义域为 \(0\leqslant x\leqslant1\).

\item \(-1+\i = \sqrt{2}\left( -\frac{\sqrt{2}}{2}+\frac{\sqrt{2}}{2}\i \right) = \sqrt{2}\left( \cos\frac{3\pi}{4}+\i\sin\frac{3\pi}{4} \right)\).

\item
\[
\sec A\cdot\sqrt{1-\sin^2 A}
=\sec A\cdot\sqrt{\cos^2 A}
=\sec A\cdot\left( -\cos A \right)
=-1
\]

\item
\[
S_{\text{侧}}=\frac{1}{2}CL=\pi RL=3.14\times1\times3=9.42 \mathrm{~cm}^2
\]
式中的 \(C\) 表示底面圆的周长，\(L\) 表示母线长，\(R\) 是底面圆的半径.
\end{enumerate}
\end{solution}

\begin{problem}
用宽为 \(80 \mathrm{~cm}\) 的铁皮制作一个槽，槽的截面 \(ABCD\) 是一个矩形：

\begin{enumerate}
    \item 求槽的截面积 \(y\) 关于槽深 \(x\) 的函数关系式.
    \item 求出这个函数图象的顶点坐标，对称轴方程.
    \item 当槽深为多少厘米时，槽的截面积最大？这个最大面积是多少平方厘米？
\end{enumerate}

\bitmapfigure[width=5.6cm]{img/1977/anhui_science/q2_fig1.png}

\end{problem}

\begin{solution}
\begin{enumerate}
\item 若槽深为 \(x\)，则槽宽为 \(80-2x\)，所以槽的截面积 \(y\) 关于槽深 \(x\) 的函数关系式是 \(y=x\left( 80-2x \right)=-2x^2+80x.\)

\item \(y=-2x^2+80x=-2\left( x-20 \right)^2+800.\)
顶点坐标是 \(\left( 20,800 \right)\)，对称轴方程是 \(x=20\).

\item 当 \(x=20\) 时，\(y\) 有极大值 \(800 \mathrm{~cm}^2\). 答：当槽深为 \(20 \mathrm{~cm}\) 时，槽的截面积最大，这个最大截面积是 \(800 \mathrm{~cm}^2\).
\end{enumerate}
\end{solution}

\begin{problem}
\begin{enumerate}
    \item 写出 \(\left( \sqrt[3]{x}-\frac{2}{3\sqrt[5]{x}} \right)^n\) 的展开式的第 \(\left( r+1 \right)\) 项.
    \item 设 \(AM\) 是直角三角形斜边 \(BC\) 上的中线，求证：\(BC^2+AC^2+AB^2=8AM^2\).
\end{enumerate}

\bitmapfigure[width=3.2cm]{img/1977/anhui_science/q3_fig1.png}

\end{problem}

\begin{solution}
\begin{enumerate}
\item 第 \(\left( r+1 \right)\) 项为
\[
\begin{aligned}
T_{r+1}
&=\mathrm C_n^r\left( \sqrt[3]{x} \right)^{n-r}\left( -\frac{2}{3\sqrt[5]{x}} \right)^r \\
&=\left( -1 \right)^r\mathrm C_n^r\left( \frac{2}{3} \right)^r\left( x^{\frac{1}{3}} \right)^{n-r}\left( x^{-\frac{1}{5}} \right)^r \\
&=\left( -1 \right)^r\mathrm C_n^r\left( \frac{2}{3} \right)^r x^{\frac{5n-8r}{15}}
\end{aligned}
\]

\item 因为 \(AM=\frac{1}{2}BC\)，所以 \(8AM^2=8\cdot\left( \frac{1}{2}BC \right)^2=2BC^2.\)
又根据勾股定理 \(AB^2+AC^2=BC^2\)，所以 \(AB^2+AC^2+BC^2=BC^2+BC^2=2BC^2.\)
因此 \(AB^2+AC^2+BC^2=8AM^2.\)
\end{enumerate}
\end{solution}

\begin{problem}
\begin{enumerate}
    \item 化简：\(\sqrt{5^{2\log_5\left( \lg x \right)}-2\lg x+1}\).
    \item 求下面数列的前 \(n\) 项和：\(\lg\frac{1}{3}+\lg\frac{10}{3^2}+\lg\frac{100}{3^4}+\lg\frac{1000}{3^8}+\dots\)
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item
\[
\sqrt{5^{2\log_5\left( \lg x \right)}-2\lg x+1}
=\sqrt{\left( \lg x \right)^2-2\lg x+1}
=\sqrt{\left( \lg x-1 \right)^2}
\]
\[
\begin{cases}
\lg x-1=\lg\frac{x}{10}, & x\geqslant10,\\
1-\lg x=\lg\frac{10}{x}, & 1<x<10.
\end{cases}
\]
（因为 \(\lg x\) 本身是另一对数的真数，要求 \(\lg x>0\)，\(x>1\)）

\item 设前 \(n\) 项和为
\[
S_n=\lg\frac{1}{3}+\lg\frac{10}{3^2}+\lg\frac{100}{3^4}+\dots+\lg\frac{10^{n-1}}{3^{2^{n-1}}}
\]
\[
\begin{aligned}
S_n
&=\left( \lg1+\lg10+\lg100+\dots+\lg10^{n-1} \right) \\
&\quad-\left( \lg3+\lg3^2+\lg3^4+\dots+\lg3^{2^{n-1}} \right) \\
&=\left( 0+1+2+\dots+\left( n-1 \right) \right)-\left( \lg3 \right)\left( 1+2+4+\dots+2^{n-1} \right) \\
&=\frac{n\left( n-1 \right)}{2}-\left( 2^n-1 \right)\lg3 \\
&=\frac{n\left( n-1 \right)}{2}+\left( 1-2^n \right)\lg3
\end{aligned}
\]
\end{enumerate}
\end{solution}

\begin{problem}
已知 \(\triangle ABC\) 的三个内角 \(A\)，\(B\)，\(C\) 成等差数列，且最大角与最小角的对边之比是 \(\left( \sqrt{3}+1 \right):2\)，试求此三角形三个内角的度数.

\end{problem}

\begin{solution}
假定三内角以 \(A\)，\(B\)，\(C\) 次序由大到小成等差数列，则 \(A+C=2B\). 所以
\[
A+B+C=2B+B=3B=180^\circ
\]
即 \(B=60^\circ\)，且 \(A+C=120^\circ\).

又已知 \(\frac{a}{c}=\frac{\sin A}{\sin C}=\frac{\sqrt{3}+1}{2}.\)
而
\[
\begin{aligned}
\frac{\sin A}{\sin C}
&=\frac{\sin\left( 120^\circ-C \right)}{\sin C}
=\frac{\sin120^\circ\cos C-\cos120^\circ\sin C}{\sin C} \\
&=\sin120^\circ\cot C-\cos120^\circ
=\frac{\sqrt{3}}{2}\cot C+\frac{1}{2}
\end{aligned}
\]

所以 \(\frac{\sqrt{3}}{2}\cot C+\frac{1}{2}=\frac{\sqrt{3}+1}{2}\)，
得 \(\cot C=1\)，故 \(C=45^\circ\). 从而 \(A=120^\circ-45^\circ=75^\circ.\)

答：三角形三内角是 \(75^\circ\)，\(60^\circ\)，\(45^\circ\).
\end{solution}

\begin{enumerate}
    \item 求过 \(P\) 点与直线 \(y-x-1=0\) 平行的直线方程.
    \item 求所求的直线与已知曲线交点的坐标.
    \item 作出它们的图象.
\end{enumerate}

\end{problem}

\begin{solution}
\begin{enumerate}
\item 直线 \(y-x-1=0\) 的斜率 \(k_1=1\)，因此过 \((8,2)\) 且与 \(y-x-1=0\) 平行的直线方程是 \(\frac{y-2}{x-8}=1\)，或者写成 \(y-x+6=0.\)

y-x+6&=0, \\
y^2&=x

\right.{}

解这个方程组，得 \(\left( x_1,y_1 \right)=\left( 9,3 \right)\)，\(\left( x_2,y_2 \right)=\left( 4,-2 \right)\). 因此，它们的交点坐标是 \(\left( 9,3 \right)\)，\(\left( 4,-2 \right)\).

\begin{problem}
判断下列方程的轨迹，并画图象：\(25x^2-14xy+25y^2-288=0\).

\end{problem}

\begin{solution}
作坐标旋转变换，假定坐标系向逆时针方向旋转 \(\varphi\) 角度，在原坐标系中 \((x,y)\) 的点，在新坐标系下为 \((x',y')\). 则有变换公式 \(x=x'\cos\varphi-y'\sin\varphi\)，\(y=x'\sin\varphi+y'\cos\varphi.\)

将它们代入原方程，整理后得
\[
\left( 25-14\sin\varphi\cos\varphi \right)x'^2+\left( 25+14\sin\varphi\cos\varphi \right)y'^2-14\left( \cos^2\varphi-\sin^2\varphi \right)x'y'-288=0
\]

令 \(14\left( \cos^2\varphi-\sin^2\varphi \right)=0\)（目的是使 \(x'y'\) 项的系数为 \(0\)），得 \(\varphi=\frac{\pi}{4}.\)

这时方程化为 \(18x'^2+32y'^2-288=0\)，即 \(9x'^2+16y'^2-144=0\)，也就是 \(\frac{x'^2}{4^2}+\frac{y'^2}{3^2}=1.\)

从而断定，原方程的轨迹是个椭圆，其图象如下：

\end{solution}

\begin{problem}

\begin{enumerate}
    \item 用 \(0\) 到 \(9\) 这 \(10\) 个数字，可以排列成多少个含有两个重复数字的三位数？
    \item 用数学归纳法证明 \(x^n-y^n\) 能够被 \(x-y\) 整除（\(n\) 为自然数）.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item
从 \(0\) 到 \(9\) 排成可以有重复数字的三位数的个数是 \(10^3-10^2\). 从 \(0\) 到 \(9\) 排成没有重复数字的三位数的个数是 \(\mathrm A_{10}^3-\mathrm A_9^2\). 同一个数字组成的三位数（如 \(111\)，\(222\)，\(\dots\)）有 \(9\) 个. 因此，从 \(0\) 到 \(9\) 这 \(10\) 个数字可排成含有两个重复数字的三位数的个数是 \(\left( 10^3-10^2 \right)-\left( \mathrm A_{10}^3-\mathrm A_9^2 \right)-9=243\text{（个）}.\)
\item 当 \(n=1\) 时，\(x^1-y^1\) 能够被 \(x-y\) 整除，命题成立.

假定当 \(n=k\) 时命题成立，即 \(x^k-y^k\) 能被 \(x-y\) 整除. 那么
\[
x^{k+1}-y^{k+1}
=x\cdot x^k-y\cdot y^k+xy^k-xy^k
=x\left( x^k-y^k \right)+y^k\left( x-y \right)
\]

因为 \(y^k\left( x-y \right)\) 能被 \(x-y\) 整除，由假定可知 \(x\left( x^k-y^k \right)\) 也能够被 \(x-y\) 整除，所以 \(x^{k+1}-y^{k+1}\) 也能够被 \(x-y\) 整除. 即 \(n=k+1\) 时，命题也成立.

根据数学归纳法原理，\(n\) 为任何自然数时，\(x^n-y^n\) 都能够被 \(x-y\) 整除.
\end{enumerate}
\end{solution}


\section{附加题}

\begin{problem}
如图，在矩形 \(ABCD\) 中，\(A_1E\parallel AB\)，\(AA_1=A_1A_2=A_2A_3=A_3A_4=A_4D=a\)，\(AB_1=B_1B_2=B_2B=\sqrt{3}a\)，试用三种方法证明：\(\angle B_1A_1E+\angle B_2A_1E+\angle BDC=\frac{\pi}{2}\).
（注：三种方法是指每种方法各以代数，三角，几何知识为主来证明.）

\bitmapfigure[width=4.8cm]{img/1977/anhui_science/q9_fig1.png}

\end{problem}

\begin{solution}
【法一】因为 \(ABCD\) 是矩形，\(A_1E\parallel AB\)，所以
\[
\tan\angle B_1A_1E=\tan\angle A_1B_1A=\frac{AA_1}{AB_1}=\frac{a}{\sqrt{3}a}=\frac{1}{\sqrt{3}}
\]
所以 \(\angle B_1A_1E=\frac{\pi}{6}.\)
又
\[
\tan\angle B_2A_1E=\tan\angle A_1B_2A=\frac{AA_1}{AB_2}=\frac{a}{2\sqrt{3}a}=\frac{\sqrt{3}}{6}
\]
\[
\tan\angle BDC=\tan\angle DBA=\frac{AD}{AB}=\frac{5a}{3\sqrt{3}a}=\frac{5\sqrt{3}}{9}
\]
因此
\[
\tan\left( \angle B_2A_1E+\angle BDC \right)
=\frac{\frac{\sqrt{3}}{6}+\frac{5\sqrt{3}}{9}}{1-\frac{\sqrt{3}}{6}\cdot\frac{5\sqrt{3}}{9}}
=\sqrt{3}
\]
所以 \(\angle B_2A_1E+\angle BDC=\frac{\pi}{3}.\) 从而 \(\angle B_1A_1E+\angle B_2A_1E+\angle BDC=\frac{\pi}{6}+\frac{\pi}{3}=\frac{\pi}{2}.\)

【法二】延长 \(DA\) 到 \(A'\) 使 \(AA'=a\)，过 \(A'\) 作 \(A'B'\parallel AB\) 且交 \(BC\) 的延长线于 \(B'\)，在 \(A'B'\) 上截取 \(A'B_2'=2\sqrt{3}a\)，连 \(DB_2'\)，\(B_2'B\)（如图）：

\solutionfigure{\bitmapfigure[width=4.8cm]{img/1977/anhui_science/q9_solution_fig1.png}}

在直角三角形 \(B_1AA_1\) 及直角三角形 \(DA'B_2'\) 中，\(\frac{B_1A}{DA'}=\frac{\sqrt{3}a}{6a}=\frac{\sqrt{3}}{6}\)，\(\frac{AA_1}{A'B_2'}=\frac{a}{2\sqrt{3}a}=\frac{\sqrt{3}}{6}.\) 所以 \(\frac{B_1A}{DA'}=\frac{AA_1}{A'B_2'}\)，从而 \(\triangle B_1AA_1\sim\triangle DA'B_2'.\) 因此 \(\angle B_1A_1E=\angle A_1B_1A=\angle A'DB_2'.\)

又在 \(\triangle DB_2'B\) 中，\(BD^2=\left( 5a \right)^2+\left( 3\sqrt{3}a \right)^2=52a^2\)，\(DB_2'^2=\left( 6a \right)^2+\left( 2\sqrt{3}a \right)^2=48a^2\)，\(B_2'B^2=\left( \sqrt{3}a \right)^2+a^2=4a^2.\) 所以 \(DB_2'^2+B_2'B^2=BD^2\)，即 \(\angle DB_2'B=\frac{\pi}{2}.\)

在直角三角形 \(DB_2'B\) 及直角三角形 \(B_2AA_1\) 中，\(\frac{DB_2'}{B_2A}=\frac{\sqrt{48a^2}}{2\sqrt{3}a}=2\)，\(\frac{B_2'B}{AA_1}=\frac{2a}{a}=2.\) 所以 \(\frac{DB_2'}{B_2A}=\frac{B_2'B}{AA_1}\)，从而 \(\triangle DB_2'B\sim\triangle B_2AA_1.\) 因此 \(\angle B_2A_1E=\angle AB_2A_1=\angle B_2'DB.\)

于是 \(\angle B_1A_1E+\angle B_2A_1E+\angle BDC =\angle A'DB_2'+\angle B_2'DB+\angle BDC =\frac{\pi}{2}.\)

【法三】在矩形 \(ABCD\) 所在的平面内建立直角坐标系；以 \(A\) 点为坐标系的原点；以 \(AB\) 所在的直线为 \(x\) 轴；以 \(AD\) 所在的直线为 \(y\) 轴；以 \(a\) 为长度单位. 设 \(B_1A_1\)，\(B_2A_1\)，\(BD\) 所在直线的斜率分别为 \(k_1\)，\(k_2\)，\(k_3\). 那么
\[
k_1=\frac{1-0}{0-\sqrt{3}}=-\frac{1}{\sqrt{3}}=-\frac{\sqrt{3}}{3}
\]
\[
k_2=\frac{1-0}{0-2\sqrt{3}}=-\frac{1}{2\sqrt{3}}=-\frac{\sqrt{3}}{6}
\]
\[
k_3=\frac{5-0}{0-3\sqrt{3}}=-\frac{5}{3\sqrt{3}}=-\frac{5\sqrt{3}}{9}
\]

直线 \(B_1A_1\) 与 \(A_1E\) 的夹角是 \(\angle B_1A_1E\)，则
\[
\tan\angle B_1A_1E
=\frac{0-\left( -\frac{\sqrt{3}}{3} \right)}{1+0\cdot\left( -\frac{\sqrt{3}}{3} \right)}
=\frac{\sqrt{3}}{3}
\]
所以 \(\angle B_1A_1E=\frac{\pi}{6}.\)

直线 \(B_2A_1\) 与 \(A_1E\) 的夹角是 \(\angle B_2A_1E\)，则
\[
\tan\angle B_2A_1E
=\frac{0-\left( -\frac{\sqrt{3}}{6} \right)}{1+0\cdot\left( -\frac{\sqrt{3}}{6} \right)}
=\frac{\sqrt{3}}{6}
\]

直线 \(BD\) 与 \(DC\) 的夹角是 \(\angle BDC\)，则
\[
\tan\angle BDC
=\frac{0-\left( -\frac{5\sqrt{3}}{9} \right)}{1+0\cdot\left( -\frac{5\sqrt{3}}{9} \right)}
=\frac{5\sqrt{3}}{9}
\]

因而
\[
\tan\left( \angle B_2A_1E+\angle BDC \right)
=\frac{\frac{\sqrt{3}}{6}+\frac{5\sqrt{3}}{9}}{1-\frac{\sqrt{3}}{6}\cdot\frac{5\sqrt{3}}{9}}
=\sqrt{3}
\]
所以 \(\angle B_2A_1E+\angle BDC=\frac{\pi}{3}.\) 从而 \(\angle B_1A_1E+\angle B_2A_1E+\angle BDC=\frac{\pi}{6}+\frac{\pi}{3}=\frac{\pi}{2}.\)
\end{solution}
