\section{解答题}

\begin{problem}
\begin{enumerate}
\item 什么是因式分解？
\item 把 \(18a^3-48a^2b+32ab^2\) 分解因式.
\end{enumerate}

\end{problem}

\begin{solution}
\begin{enumerate}
\item
把一个多项式分解成几个单项式或多项式连乘的形式，叫做多项式的因式分解.

\item
\(18a^3-48a^2b+32ab^2=2a\left(9a^2-24ab+16b^2\right)=2a\left(3a-4b\right)^2\).
\end{enumerate}
\end{solution}

\begin{problem}
\begin{enumerate}
\item 什么是方程？
\item 什么是方程的解？
\item 解方程：\(\frac{2}{x-2}+x=\frac{x}{x-2}\).
\end{enumerate}

\end{problem}

\begin{solution}
\begin{enumerate}
\item
含有未知数的等式叫做方程.

\item
使方程两端相等的未知数的值叫方程的解.

\item
解方程：\(\frac{2}{x-2}+x=\frac{x}{x-2}\). 去分母，得 \(2+x\left(x-2\right)=x\)，整理得 \(x^2-3x+2=0\)，\(\left(x-2\right)\left(x-1\right)=0\)，所以 \(x_1=2\)，\(x_2=1\). 把 \(x=2\) 代入原方程分母为零，所以 \(x=2\) 是增根，舍去. 故原方程的解是 \(x=1\).
\end{enumerate}
\end{solution}

\begin{problem}
在 \(\triangle ABC\) 中，\(\angle C\) 是直角，\(CD\) 是 \(AB\) 上的高，已知 \(AB=9\)，\(AC=6\). 求 \(CD\).

\end{problem}

\begin{solution}
因为 \(\angle C\) 是直角，所以 \(\triangle ABC\) 是直角三角形. 由勾股定理得 \(BC=\sqrt{AB^2-AC^2}=\sqrt{81-36}=3\sqrt{5}\).

\solutionfigure{\bitmapfigure[max width=4.0cm]{img/1977/jilin/q3_solution_fig1.png}}

又因为 \(CD\) 是 \(AB\) 边上的高，所以 \(\frac{1}{2}AB\cdot CD=\frac{1}{2}AC\cdot BC\)，因此 \(CD=\frac{AC\cdot BC}{AB}=\frac{6\cdot3\sqrt{5}}{9}=2\sqrt{5}\).
\end{solution}

\begin{problem}
\(k\) 是什么值时，方程 \(x^2+4x+k=0\) 有

\begin{enumerate}
\item 两个不相等的实根？
\item 相等的实根？
\item 没有实根？
\end{enumerate}

\end{problem}

\begin{solution}
方程 \(x^2+4x+k=0\) 的判别式是 \(\Delta=4^2-4k=4\left(4-k\right)\).

\begin{enumerate}
\item 当 \(k<4\) 时，\(\Delta>0\)，方程有两个不相等的实根.

\item 当 \(k=4\) 时，\(\Delta=0\)，方程有两个相等的实根.

\item 当 \(k>4\) 时，\(\Delta<0\)，方程没有实根.
\end{enumerate}
\end{solution}

\begin{problem}
\begin{enumerate}
\item 什么是平面上的轨迹？
\item 某港口外海面上，有一个暗礁区，为进港船只的安全，建有 \(A\)，\(B\) 两灯塔，把暗礁区所张的危险的角 \(\angle AMB\) 的大小通知船上. 试用轨迹涵义证明：只要进港船只 \(S\) 的视角 \(\angle ASB<\angle AMB\) 时，船只就可安全进港.
\end{enumerate}

\end{problem}

\begin{solution}
\begin{enumerate}
\item
平面上的动点按照一定规律运动所描绘的图形，就叫做动点在平面上的轨迹.

\item
港口所建的两灯塔 \(A\)，\(B\) 之间距离为一定长，暗礁区所张的角 \(\angle AMB\) 也已知，为一定角. 因此 \(\angle AMB\) 的顶点 \(M\) 的轨迹是以 \(AB\) 为弦，圆周角为定角 \(\angle AMB\) 所划的圆弧，暗礁区就在圆弧内（如图）.

根据圆外任一点到两弦连线的夹角都小于该弦上的圆周角这一道理，所以在 \(\angle ASB<\angle AMB\) 时船总是在暗礁区 \(AMB\) 的外面，所以船就可以安全进港.
\end{enumerate}
\solutionfigure{\bitmapfigure[max width=4.2cm]{img/1977/jilin/q5_solution_fig1.png}}
\end{solution}

\begin{problem}
计算：\(8x^{-\frac{1}{3}}\sqrt{y^{-\frac{1}{3}}x^4\sqrt{y^{\frac{4}{3}}}}\).

\end{problem}

\begin{solution}
原式 \(=8x^{-\frac{1}{3}}\sqrt{y^{-\frac{1}{3}}x^4\sqrt{y^{\frac{4}{3}}}}\).
\[
=8x^{-\frac{1}{3}}\left[y^{-\frac{1}{3}}x^4\left(y^{\frac{4}{3}}\right)^{\frac{1}{2}}\right]^{\frac{1}{2}}=8x^{-\frac{1}{3}}x^2y^{-\frac{1}{6}}y^{\frac{1}{3}}
=8x^{\frac{5}{3}}y^{\frac{1}{6}}=8x\sqrt[6]{x^4y}
\]
\end{solution}

\begin{problem}
证明：\(\log_a b\cdot\log_b c\cdot\log_c a=1\).

\end{problem}

\begin{solution}
根据对数的换底公式，有 \(\log_a b\cdot\log_b c\cdot\log_c a=\frac{\lg b}{\lg a}\cdot\frac{\lg c}{\lg b}\cdot\frac{\lg a}{\lg c}=1\).
\end{solution}

\begin{problem}
证明：\(\frac{\sin\left(\alpha+\beta\right)-2\sin\alpha\cos\beta}{2\sin\alpha\sin\beta+\cos\left(\alpha+\beta\right)}=\tan\left(\beta-\alpha\right)\).

\end{problem}

\begin{solution}
左边
\[
\begin{aligned}
&=\frac{\sin\alpha\cos\beta+\cos\alpha\sin\beta-2\sin\alpha\cos\beta}{2\sin\alpha\sin\beta+\cos\alpha\cos\beta-\sin\alpha\sin\beta}\\
&=\frac{\cos\alpha\sin\beta-\sin\alpha\cos\beta}{\cos\alpha\cos\beta+\sin\alpha\sin\beta}=\frac{\sin\left(\beta-\alpha\right)}{\cos\left(\beta-\alpha\right)}\\
&=\tan\left(\beta-\alpha\right)=\text{右边}.
\end{aligned}
\]
所以原等式成立.
\end{solution}

\begin{problem}
计算：在等比数列里，已知 \(a_1=2\)，\(S_3=26\)，求 \(q\).

\end{problem}

\begin{solution}
因为 \(a_1=2\)，\(S_3=26\)，根据等比数列前 \(n\) 项和的公式，有 \(26=\frac{2\left(1-q^3\right)}{1-q}\)，即 \(1+q+q^2=13\)，\(q^2+q-12=0\)，\(\left(q+4\right)\left(q-3\right)=0\). 所以 \(q_1=-4\)，\(q_2=3\).
\end{solution}

\begin{problem}
修建水库时，欲测两山腰上 \(A\)，\(B\) 两点间的距离，在某处选与 \(A\)，\(B\) 在同一平面内 \(C\)，\(D\) 两点，测得 \(CD=200\text{ 米}\)，\(\angle ACD=105^\circ\)，\(\angle BCD=15^\circ\)，\(\angle BDC=120^\circ\)，\(\angle ADC=30^\circ\)，试求 \(AB\).

\end{problem}

\begin{solution}
\solutionfigure{\bitmapfigure[max width=5.8cm]{img/1977/jilin/q10_solution_fig1.png}}

在 \(\triangle ACD\) 中，\(\angle CAD=180^\circ-\angle ACD-\angle ADC=180^\circ-105^\circ-30^\circ=45^\circ\). 根据正弦定理，\(AC=\frac{200\sin30^\circ}{\sin45^\circ}=100\sqrt{2}\text{ 米}\).
在 \(\triangle BDC\) 中，\(\angle CBD=180^\circ-\angle BDC-\angle BCD=180^\circ-120^\circ-15^\circ=45^\circ\). 由正弦定理，\(BC=\frac{200\sin120^\circ}{\sin45^\circ}=100\sqrt{6}\text{ 米}\).
在 \(\triangle ACB\) 中，\(\angle ACB=\angle ACD-\angle BCD=105^\circ-15^\circ=90^\circ\). 因此，由勾股定理，
\[
AB=\sqrt{AC^2+BC^2}=\sqrt{\left(100\sqrt{2}\right)^2+\left(100\sqrt{6}\right)^2}=200\sqrt{2}\text{ 米}
\]

答：\(AB\) 长为 \(200\sqrt{2}\text{ 米}\).
\end{solution}

\begin{problem}
已知直线 \(4x+3y-12=0\) 和圆心在 \(\left(4,7\right)\) 点的一个圆相切，求这圆的半径及过圆心和切点的直线方程.

\end{problem}

\begin{solution}
根据已知，圆的方程可写为 \(\left(x-4\right)^2+\left(y-7\right)^2=R^2\). 因为它与直线 \(4x+3y-12=0\) 相切，因此圆心 \(\left(4,7\right)\) 到直线 \(4x+3y-12=0\) 的距离等于圆半径 \(R\)，即
\[
R=\frac{\left|4\times4+3\times7-12\right|}{\sqrt{4^2+3^2}}=\frac{25}{5}=5
\]
从而圆的方程是 \(\left(x-4\right)^2+\left(y-7\right)^2=5^2\).

下面求圆和直线的交点坐标 \(\left(x,y\right)\)，解如下方程组
\[
\begin{cases}
\left(x-4\right)^2+\left(y-7\right)^2=5^2,\\
4x+3y-12=0.
\end{cases}
\]
解之，得两组相同解 \(x_1=x_2=0\)，\(y_1=y_2=4\)，表示 \(\left(0,4\right)\) 是切点. 因此过圆心 \(\left(4,7\right)\) 和切点 \(\left(0,4\right)\) 的直线方程为 \(\frac{y-4}{x-0}=\frac{7-4}{4-0}\)，即 \(3x-4y+16=0\).
\end{solution}


\section{参考题}

\begin{problem}
把 \(4x^2+9y^2+8x-36y+4=0\) 化成标准方程，并指出它所代表的曲线的长、短半轴，离心率.

\end{problem}

\begin{solution}
原方程可化为 \(4\left(x^2+2x+1\right)+9\left(y^2-4y+4\right)-36=0\)，即 \(4\left(x+1\right)^2+9\left(y-2\right)^2=36\). 标准式为 \(\frac{\left(x+1\right)^2}{3^2}+\frac{\left(y-2\right)^2}{2^2}=1\). 此方程表示以 \(\left(-1,2\right)\) 为中心的椭圆. 长半轴 \(a=3\)，短半轴 \(b=2\). 其离心率为
\[
e=\frac{c}{a}=\frac{\sqrt{a^2-b^2}}{a}=\frac{\sqrt{3^2-2^2}}{3}=\frac{\sqrt{5}}{3}
\]
\end{solution}

\begin{problem}
试求下列各式的导数：

\begin{enumerate}
\item \(y=3x^2+\sqrt{x}+5\)；
\item \(y=\ln\left(x^{10}\right)+\ln\left(\ln x\right)\)；
\item \(y=x^x\).
\end{enumerate}

\end{problem}

\begin{solution}
\begin{enumerate}
\item \(y'=6x+\frac{1}{2\sqrt{x}}=6x+\frac{\sqrt{x}}{2x}\).

\item \(y'=\frac{10x^9}{x^{10}}+\frac{\frac{1}{x}}{\ln x}=\frac{10}{x}+\frac{1}{x\ln x}\).

\item \(y'=x\cdot x^{x-1}+x^x\ln x=x^x\left(1+\ln x\right)\).
\end{enumerate}
\end{solution}

\begin{problem}
求微分方程的通解：

\begin{enumerate}
\item \(y'=\frac{2+y}{3-x}\)；
\item \(y''-2y'-3y=0\).
\end{enumerate}

\end{problem}

\begin{solution}
\begin{enumerate}
\item 原方程可写成 \(\frac{\mathrm{d}y}{2+y}=\frac{\mathrm{d}x}{3-x}\). 两边积分，得 \(\ln\left|2+y\right|=-\ln\left|3-x\right|+C_1\)（\(C_1\) 为任意常数）.
于是 \(2+y=\frac{C}{3-x}\). 因此通解为 \(y=\frac{C}{3-x}-2\)（其中 \(C=\e^{C_1}\)，是任意常数）.

\item 这是一个二阶常系数齐次线性方程，其特征方程 \(\lambda^2-2\lambda-3=0\) 的根是 \(3\)，\(-1\)，因此原方程的通解为 \(y=C_1\e^{3x}+C_2\e^{-x}\)，其中 \(C_1\)，\(C_2\) 是任意常数.
\end{enumerate}
\end{solution}
