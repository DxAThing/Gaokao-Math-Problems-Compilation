\section{填空题}

\begin{problem}
\begin{enumerate}
\item \( \log_3 \frac{1}{27}=\fillinblank{} \).
\item \( \log_{10}\left( \log_{10}10 \right)=\fillinblank{} \).
\end{enumerate}

\end{problem}

\begin{answer}
\begin{enumerate}
\item \(-3\)
\item \(0\)
\end{enumerate}
\end{answer}

\begin{solution}
\begin{enumerate}
\item
\( \log_3 \frac{1}{27}=\log_3 3^{-3}=-3 \).

\item
\( \log_{10}\left( \log_{10}10 \right)=\log_{10}1=0 \).
\end{enumerate}
\end{solution}

\section{解答题}

\begin{problem}
当 \( a \) 取什么值时，方程 \( x^2-2x+\lg a=0 \) 有实数根.

\end{problem}

\begin{solution}
要使方程 \( x^2-2x+\lg a=0 \) 有实数根，这个方程的判别式必须不小于零，即
\( \left( -2 \right)^2-4\lg a\ge 0 \).
得 \( \lg a\le 1 \)，\( a\le 10 \).
又因为 \( a \) 是真数，所以 \( a>0 \).
综合以上条件，得 \( 0<a\le 10 \).
因此当 \( 0<a\le 10 \) 时，方程 \( x^2-2x+\lg a=0 \) 有实数根.
\end{solution}

\begin{problem}
解方程：\( 7^{\left( 2x^2+x-6 \right)}=1 \).

\end{problem}

\begin{solution}
\( 7^{\left( 2x^2+x-6 \right)}=1 \) 可化为 \( 7^{\left( 2x^2+x-6 \right)}=7^0 \). 所以
\( 2x^2+x-6=0 \).
解此方程得到 \( x_1=\frac{3}{2} \)，\( x_2=-2 \).
\end{solution}

\begin{problem}
求等比数列 \( \sqrt[6]{3} \)，\( \sqrt[3]{3} \)，\( \sqrt{3} \)，\(\ldots\) 的公比 \( q \)，并写出它的第六项 \( a_6 \).

\end{problem}

\begin{solution}
\( q=\frac{\sqrt[3]{3}}{\sqrt[6]{3}}=\sqrt[6]{3} \).
\( a_6=a_1q^5=\sqrt[6]{3}\cdot \left( \sqrt[6]{3} \right)^5=3 \).
\end{solution}

\begin{problem}
在底边为 \(2\text{ 尺}\)，高为 \(1.6\text{ 尺}\)的三角形内截一个最大面积的矩形，并使其一边在三角形的底边上，求矩形的长和宽各是多少？

\end{problem}

\begin{solution}
如图，矩形的一边 \(FG\) 在 \( \triangle ABC \) 的 \(BC\) 边上，因此 \( EH\parallel BC \).

\solutionfigure{\bitmapfigure[max width=5.8cm]{img/1977/xinjiang_science/q5_solution_fig1.png}}

从而有 \( \triangle AEH\sim\triangle ABC \). 因此
\( \frac{EH}{BC}=\frac{AK}{AD} \).
现设矩形的长 \( EH=x\text{ 尺} \)，则 \( \frac{x}{2}=\frac{AK}{1.6} \)，\( AK=0.8x \).
那末矩形的宽 \( HG=KD=1.6-0.8x \).
因此矩形面积
\[
S=x\left( 1.6-0.8x \right)=-0.8x^2+1.6x=-0.8\left( x-1 \right)^2+0.8
\]
所以当 \( x=1 \) 时，函数 \( S=-0.8x^2+1.6x \) 有极大值. 此时
\( HG=KD=1.6-0.8\times 1=0.8 \).

答：当矩形的长为 \(1\text{ 尺}\)，宽为 \(0.8\text{ 尺}\)时，所截矩形面积最大.
\end{solution}

\begin{problem}
求 \( \cot\left( -1200^\circ \right) \) 的值.

\end{problem}

\begin{solution}
\[
\cot\left( -1200^\circ \right)=-\cot\left( 1200^\circ \right)=-\cot\left( 360^\circ\times 3+120^\circ \right)=-\cot 120^\circ.
\]
\[
\cot\left( -1200^\circ \right)=-\cot\left( 180^\circ-60^\circ \right)=\cot 60^\circ=\frac{\sqrt{3}}{3}.
\]
\end{solution}

\begin{problem}
已知 \( \sin\alpha=-\frac{12}{13} \)，并且 \( \pi<\alpha<\frac{3\pi}{2} \)，求 \( \cos \frac{\alpha}{2} \).

\end{problem}

\begin{solution}
因为 \( \pi<\alpha<\frac{3\pi}{2} \)，所以 \( \frac{\pi}{2}<\frac{\alpha}{2}<\frac{3\pi}{4} \)，\( \cos\alpha \) 和 \( \cos \frac{\alpha}{2} \) 都是负值，则
\( \cos \frac{\alpha}{2}=-\sqrt{\frac{1+\cos\alpha}{2}} \).
而
\[
\cos\alpha=-\sqrt{1-\sin^2\alpha}=-\sqrt{1-\left( -\frac{12}{13} \right)^2}=-\frac{5}{13}
\]
因此
\[
\cos \frac{\alpha}{2}=-\sqrt{\frac{1+\cos\alpha}{2}}=-\sqrt{\frac{1-\frac{5}{13}}{2}}=-\frac{2\sqrt{13}}{13}
\]
\end{solution}

\begin{problem}
三角形的三边长分别是 \( 4 \mathrm{~cm} \)，\( 5 \mathrm{~cm} \)，\( \sqrt{61} \mathrm{~cm} \)，求这个三角形内最大角的度数.

\end{problem}

\begin{solution}
在一个三角形内，三内角中最长边所对的角最大. 因此这个三角形内边长为 \( \sqrt{61} \mathrm{~cm} \) 的边所对的角最大.
设 \( \sqrt{61} \mathrm{~cm} \) 的这条边所对的角为 \( \alpha \)，由余弦定理得
\[
\cos\alpha=\frac{-\left( \sqrt{61} \right)^2+4^2+5^2}{2\times 4\times 5}=\frac{-61+16+25}{40}=-\frac{1}{2}
\]
故 \( \alpha=120^\circ \).
答：这个三角形内最大的角是 \( 120^\circ \).
\end{solution}

\begin{problem}
求经过圆 \( 2x^2-20x+2y^2+18=0 \) 上一点 \( P\left( 3,2\sqrt{3} \right) \) 的切线方程.

\end{problem}

\begin{solution}
因为 \( P\left( 3,2\sqrt{3} \right) \) 是圆 \( 2x^2-20x+2y^2+18=0 \) 上一点，所以过 \(P\) 点圆的切线方程为
\( 2\left( 3x \right)-20\left( \frac{3+x}{2} \right)+2\left( 2\sqrt{3}\,y \right)+18=0 \).
化简后得
\( \sqrt{3}\,y-x-3=0 \).
\end{solution}

\begin{problem}
圆弧拱桥的拱跨径 \( AB=16\text{ 米} \)，主拱高为 \( 2.5\text{ 米} \)，计算圆弧 \(\widehat{AB}\) 的半径是多少米？

\end{problem}

\begin{solution}
如图，是圆弧拱桥. 跨径 \(AB\) 是圆弧 \( \widehat{AB} \) 之弦，主拱 \( CD\perp AB \)，并且主拱 \(CD\) 延长通过圆弧 \( \widehat{AB} \) 的圆心.

\solutionfigure{\bitmapfigure[max width=6.2cm]{img/1977/xinjiang_science/q10_solution_fig1.png}}

故 \(CD\) 平分 \(AB\)（垂直于弦的直径平分这条弦）. 即得
\[
AD=DB=\frac{1}{2}AB=8
\]
设圆弧 \( \widehat{AB} \) 的半径为 \(R\)，圆心为 \(O\)，则 \( OB=R \)，\( OD=R-CD=R-2.5 \).

由勾股定理得
\( R^2=\left( R-2.5 \right)^2+8^2 \).
\( R^2=R^2-5R+6.25+64 \).
\( 5R=70.25 \).
所以 \( R=14.05\text{ 米} \).
答：圆弧 \(\widehat{AB}\) 的半径是 \(14.05\text{ 米}\).
\end{solution}

\section{参考题}

\begin{problem}
解方程：\( \log_{16-3x}\left( x-2 \right)=\log_8\left( 2\sqrt{2} \right) \).

\end{problem}

\begin{solution}
\( \log_8\left( 2\sqrt{2} \right)=\log_8 8^{\frac{1}{2}}=\frac{1}{2} \).
因此
\( \log_{16-3x}\left( x-2 \right)=\frac{1}{2} \).
根据对数定义，有
\( \left( 16-3x \right)^{\frac{1}{2}}=x-2 \).
两边平方，
\( 16-3x=x^2-4x+4 \).
即
\(x^2-x-12=0\)，\(\left( x-4 \right)\left( x+3 \right)=0\).
解得 \( x_1=4 \)，\( x_2=-3 \). 其中 \( x=-3 \) 是增根，舍去.
因此原方程的解是 \( x=4 \).
\end{solution}

\begin{problem}
已知 \( 0^\circ<x<45^\circ \)，且有
\( \lg \tan x-\lg \sin x=\lg \cos x-\lg \cot x+2\lg 3-\frac{3}{2}\lg 2 \)，试求 \( \sin x-\cos x \) 的值.

\end{problem}

\begin{solution}
利用对数的性质，已知等式可变换成
\( \lg \frac{\tan x}{\sin x}=\lg \frac{\cos x\cdot 3^2}{\cot x\cdot 2^{\frac{3}{2}}} \).
即
\( \lg \frac{1}{\cos x}=\lg \frac{9\sin x}{2\sqrt{2}} \).
所以
\( \frac{1}{\cos x}=\frac{9\sin x}{2\sqrt{2}} \)，
从而
\( \sin x\cos x=\frac{2\sqrt{2}}{9} \).
因为 \( 0^\circ<x<45^\circ \)，所以 \( \sin x<\cos x \)，也就是 \( \sin x-\cos x<0 \). 因此
\( \sin x-\cos x=-\sqrt{\left( \sin x-\cos x \right)^2} \).
\( \sin x-\cos x=-\sqrt{1-2\sin x\cos x} \).
\( \sin x-\cos x=-\sqrt{1-2\cdot \frac{2\sqrt{2}}{9}} \).
\( \sin x-\cos x=-\frac{\sqrt{9-4\sqrt{2}}}{3} \).
\( \sin x-\cos x=-\frac{2\sqrt{2}-1}{3} \).
\( \sin x-\cos x=\frac{1-2\sqrt{2}}{3} \).
\end{solution}

\begin{problem}
应用导数，求过曲线 \( x^2+y^2-8x-2y+12=0 \) 上一点 \( P\left(5,3\right) \) 的切线方程.

\end{problem}

\begin{solution}
对曲线方程 \( x^2+y^2-8x-2y+12=0 \) 两边求导：
\( 2x+2yy'-8-2y'=0 \).
解出
\( y'=\frac{4-x}{y-1} \).
过曲线上 \( P\left(5,3\right) \) 点的切线的斜率
\( k=\left. \frac{4-x}{y-1} \right|_{x=5,\ y=3}=-\frac{1}{2} \).
因此，所求的切线方程是
\( y-3=-\frac{1}{2}\left( x-5 \right) \).
或者写为
\( y=-\frac{1}{2}x+\frac{11}{2} \).
\end{solution}

\begin{problem}
利用定积分，求椭圆 \( 4x^2+25y^2=100 \) 的面积.

\end{problem}

\begin{solution}
椭圆的方程可以写成
\( \frac{x^2}{5^2}+\frac{y^2}{2^2}=1 \).
因此，椭圆的面积
\( S=4\int_0^5 2\sqrt{1-\frac{x^2}{5^2}}\,\mathrm{d}x=\frac{8}{5}\int_0^5 \sqrt{5^2-x^2}\,\mathrm{d}x \).
\( =\frac{4}{5}\left( x\sqrt{5^2-x^2}+5^2\arcsin \frac{x}{5} \right)\bigg|_0^5=10\pi \).
答：椭圆 \( 4x^2+25y^2=100 \) 的面积是 \( 10\pi \).
\end{solution}
