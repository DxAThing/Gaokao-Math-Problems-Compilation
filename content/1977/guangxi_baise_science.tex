\chapter{1977年广西百色卷（理）}

\begin{problem}
求函数 \( y=\frac{2\sqrt{5-x}}{\sqrt{x-3}} \) 的定义域.

\end{problem}

\begin{solution}
当 \( 5-x\geqslant 0 \) 时，即 \( x\leqslant 5 \) 时，分子才有意义. 当 \( x-3>0 \) 时，即 \( x>3 \) 时，分母才有意义. 故函数 \( y=\frac{2\sqrt{5-x}}{\sqrt{x-3}} \) 的定义域是 \( 3<x\leqslant 5 \).
\end{solution}

\begin{problem}
计算 \( \left( -\frac{1}{2} \right)^{-3}+32^{\frac{2}{5}}+5^0-3^{-2} \).

\end{problem}

\begin{solution}
原式 \( =(-2)^3+\left( 2^5 \right)^{\frac{2}{5}}+1-\frac{1}{3^2} =-8+4+1-\frac{1}{9} =-3\frac{1}{9} \).
\end{solution}

\begin{problem}
求 \( \sin^2 60^\circ+\cot 45^\circ+\frac{3}{4}\tan^2 30^\circ-\cos 60^\circ \) 的值.

\end{problem}

\begin{solution}
原式 \( =\left( \frac{\sqrt{3}}{2} \right)^2+1+\frac{3}{4}\left( \frac{\sqrt{3}}{3} \right)^2-\frac{1}{2} =\frac{3}{4}+1+\frac{1}{4}-\frac{1}{2} =1\frac{1}{2} \).
\end{solution}

\begin{problem}
\( \log_2\sqrt{8}=x \). 求 \( x \).

\end{problem}

\begin{solution}
因为 \( \log_2\sqrt{8}=x \)，所以 \( 2^x=\sqrt{8} \). 又 \( 2^x=2^{\frac{3}{2}} \). 所以 \( x=\frac{3}{2} \).
\end{solution}

\begin{problem}
已知直线过 \( (3,1) \) 和 \( (-6,4) \) 两点.

\begin{enumerate}[label=（\arabic*）]
    \item 求出这直线方程.

    \item 求它的斜率和 \( y \) 轴上的截距.

    \item 这函数是上升或下降.

    \item 求过 \( (0,-5) \) 和这直线平行的直线方程.
\end{enumerate}

\end{problem}

\begin{solution}
（1）设过 \( (3,1) \) 和 \( (-6,4) \) 两点的直线方程为 \( y=kx+b \). 则这两个点的坐标满足方程：\( \begin{cases} 1=3k+b \\ 4=-6k+b \end{cases} \). 解这个方程组得：\( k=-\frac{1}{3} \)，\( b=2 \). 故这条直线方程是 \( y=-\frac{1}{3}x+2 \). 即 \( x+3y-2=0 \).

（2）由上知这条直线的斜率是 \( -\frac{1}{3} \)，\( y \) 轴上截距是 2.

（3）因 \( k<0 \)，故这函数是下降的.

（4）因为过 \( (0,-5) \) 点的直线与这条直线平行，故它们的斜率相等，\( y \) 轴上的截距是 \(-5\)，所以过 \( (0,-5) \) 点的直线方程是：\( y=-\frac{1}{3}x-5 \). 即 \( x+3y+15=0 \).
\end{solution}

\begin{problem}
已知圆 \( O \) 中两弦 \( AB \)，\( CD \) 相交于 \( E \).

\begin{enumerate}[label=（\arabic*）]
    \item 证明 \( AE\cdot BE=CE\cdot DE \).

    \item 当 \( CD \) 为直径（\( 2R \)），\( AB\perp CD \) 时，证明 \( BE^2=CE(2R-CE) \).
\end{enumerate}

\bitmapfigure[max width=0.34\linewidth,max totalheight=4.8cm]{img/1977/guangxi_baise_science/q6_fig1.png}

\end{problem}

\begin{solution}
（1）证明：连结 \( AC \) 和 \( BD \). 则 \( \angle A=\angle D \)，\( \angle B=\angle C \)，所以 \( \triangle AEC\sim\triangle DEB \). 所以 \( AE:DE=CE:BE \). 即 \( AE\cdot BE=CE\cdot DE \).

（2）证明：因为 \( CD \) 为直径（\( 2R \)），\( AB\perp CD \). 所以 \( AE=BE \)，\( DE=2R-CE \). 代入 \( AE\cdot BE=CE\cdot DE \) 得 \( BE^2=CE(2R-CE) \).
\end{solution}

\begin{problem}
一船往返于 48 公里的两地，共航行 10 小时，已知水流速度是 2 公里/小时，求这轮船在静水中航行的速度.

\end{problem}

\begin{solution}
设轮船在静水中航行速度为每小时 \( x \) 公里. 则轮船在顺水时的速度为每小时 \( x+2 \) 公里. 轮船在逆水时的速度为每小时 \( x-2 \) 公里.

依题意得方程：\( \frac{48}{x+2}+\frac{48}{x-2}=10 \). 两边乘以 \( (x+2)(x-2) \) 得：\( 48(x-2)+48(x+2)=10(x+2)(x-2) \).

整理得：\( 5x^2-48x-20=0 \).

解这个方程得：\( x_1=10 \)，\( x_2=-\frac{2}{5} \).

把 \( x_1=10 \)，\( x_2=-\frac{2}{5} \) 代入所乘整式 \( (x+2)(x-2) \) 都不为 0. 因此 \( x_1=10 \)，\( x_2=-\frac{2}{5} \) 都是方程的根，但 \( x_2=-\frac{2}{5} \) 不合题意应舍去，故得 \( x=10 \).

答：轮船在静水中航行的速度是 10 公里/小时.
\end{solution}

\begin{problem}
已知 \( A \)，\( B \)，\( C \) 是 \( \triangle ABC \) 的三个内角，求证 \( a=b\cos C+c\cos B \)，\( b=c\cos A+a\cos C \)，\( c=a\cos B+b\cos A \).

\bitmapfigure[max width=0.34\linewidth,max totalheight=4.6cm]{img/1977/guangxi_baise_science/q8_fig1.png}

\end{problem}

\begin{solution}
证明：作 \( AD\perp BC \)，则 \( \triangle ABD \) 和 \( \triangle ACD \) 都是直角三角形. 所以 \( BD=c\cos B \)，\( DC=b\cos C \). 又 \( a=BC=BD+DC=b\cos C+c\cos B \).

同理可证：\( b=c\cos A+a\cos C \)，\( c=a\cos B+b\cos A \).
\end{solution}

\begin{problem}
当 \( a \)，\( b \)，\( c \) 为实数时，求证方程 \( x^2-(a+b)x+(ab-c^2)=0 \) 有两个实数根，并求出两个根相等的条件.

\end{problem}

\begin{solution}
证明：因为 \( \Delta=\left[ -(a+b) \right]^2-4\times 1\times (ab-c^2) =a^2+2ab+b^2-4ab+4c^2 =(a-b)^2+4c^2 \).

当 \( a \)，\( b \)，\( c \) 为实数时，必有 \( \Delta=(a-b)^2+4c^2\geqslant 0 \). 故当 \( a \)，\( b \)，\( c \) 为实数时，方程 \( x^2-(a+b)x+(ab-c^2)=0 \) 有两个实数根，并且方程有两个等根的条件是 \( a=b \)，\( c=0 \).
\end{solution}


\begin{problem}
已知：\( 2\lg(x-2y)=\lg x+\lg y \). 求：\( x:y \).

\end{problem}

\begin{solution}
因为 \( 2\lg(x-2y)=\lg x+\lg y \)，所以 \( \lg\left( x-2y \right)^2=\lg xy \). 即 \( \left( x-2y \right)^2=xy \). 即 \( x^2-5xy+4y^2=0 \). 所以 \( \left( x-4y \right)\left( x-y \right)=0 \). \( x-4y=0 \) 得 \( x:y=4 \). \( x-y=0 \) 得 \( x:y=1 \).

因为当 \( x>0 \)，\( y>0 \) 时原方程才有意义，并且当 \( x:y=1 \) 时 \( \lg(x-2y) \) 没有意义，所以 \( x:y=4 \) 是原方程的解.
\end{solution}

\begin{problem}
求内接于半径为 10 厘米，圆心角为 \( 120^\circ \) 的弓形的正方形的边长.

\bitmapfigure[max width=0.42\linewidth,max totalheight=5.2cm]{img/1977/guangxi_baise_science/q11_fig1.png}

\end{problem}

\begin{solution}
如图：\( CDEF \) 是内接于半径为 10 厘米，圆心角为 \( 120^\circ \) 的弓形的正方形，作半径 \( OP\perp FE \)，交 \( FE \) 于 \( N \)，\( AB \) 于 \( M \)，则 \( OP \) 平分 \( EF \)，也平分 \( AB \) 及 \( \angle AOB \).

在直角 \( \triangle OMB \) 中，\( \angle BOM=60^\circ \)，\( OB=10\text{cm} \)，所以 \( OM=10\cos 60^\circ=10\times \frac{1}{2}=5\text{cm} \).

设正方形 \( CDEF \) 的边长为 \( x \)，则 \( MN=x \)，\( NE=\frac{1}{2}x \). 连结 \( OE \)，在直角 \( \triangle ONE \) 中，\( OE=10\text{cm} \)，\( NE=\frac{1}{2}x \)，\( ON=5+x \). 所以 \( \left( 5+x \right)^2+\left( \frac{1}{2}x \right)^2=10^2 \). 化简整理得：\( x^2+8x-60=0 \). 解这个方程得：\( x=\frac{-8\pm \sqrt{304}}{2}\approx \frac{-8\pm 17.4}{2} \). 所以 \( x_1=\frac{-8+17.4}{2}=4.7\text{cm} \)，\( x_2=\frac{-8-17.4}{2}=-12.7\text{cm} \)（不合题意，舍去）.

故内接于半径为 \( 10\text{cm} \)，圆心角为 \( 120^\circ \) 的弓形的正方形边长约为 \( 4.7\text{cm} \).
\end{solution}
