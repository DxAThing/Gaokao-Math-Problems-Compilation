\begin{problem}
计算：\( \frac{2}{3}-\frac{1}{2}+0.3\times1.4\div0.6 \).
\end{problem}

\begin{solution}
\[
\frac{2}{3}-\frac{1}{2}+0.3\times1.4\div0.6=\frac{1}{6}+0.42\div0.6=\frac{1}{6}+0.7=\frac{1}{6}+\frac{7}{10}=\frac{13}{15}.
\]
\end{solution}

\begin{problem}
计算：\( -\frac{1}{3}+2\times\left(-\frac{2}{3}\right)^2-3^2\div\left(-1\frac{1}{2}\right) \).
\end{problem}

\begin{solution}
\[
-\frac{1}{3}+2\times\left(-\frac{2}{3}\right)^2-3^2\div\left(-1\frac{1}{2}\right)=-\frac{1}{3}+2\times\frac{4}{9}+9\div\frac{3}{2}=-\frac{3}{9}+\frac{8}{9}+6=6\frac{5}{9}.
\]
\end{solution}

\begin{problem}
已知圆的周长为 \(12\pi\mathrm{~cm}\)，求这个圆的面积（\(\pi=3.14\)）.
\end{problem}

\begin{solution}
由 \(12\pi=2\pi r\) 得 \(r=6\mathrm{~cm}\)，故 \(S=\pi r^2=36\times3.14=113.04\mathrm{~cm}^2\).
\end{solution}

\begin{problem}
解方程：\( \frac{7}{2x+6}-\frac{3}{4}=\frac{5}{x+3} \).
\end{problem}

\begin{solution}
将方程两边同乘以 \(4(x+3)\)，得 \(14-3(x+3)=20\)，
即 \(-3(x+3)=6\)，所以 \(x+3=-2\)，故 \(x=-5\).

将 \(x=-5\) 代入原方程，
\(\text{左边}=\frac{7}{-10+6}-\frac{3}{4}=-\frac{5}{2}\)，
\(\text{右边}=\frac{5}{-5+3}=-\frac{5}{2}\)，
\(\text{左边}=\text{右边}\)，所以 \(x=-5\) 为方程的根.
\end{solution}

\begin{problem}
因式分解：\( 3a^2+6ab+3b^2-12c^2 \).
\end{problem}

\begin{solution}
\[
3a^2+6ab+3b^2-12c^2=3\bigl[a^2+2ab+b^2-4c^2\bigr]=3\bigl[(a+b)^2-4c^2\bigr]=3(a+b-2c)(a+b+2c).
\]
\end{solution}

\begin{problem}
试讨论 \(a\) 与 \(-a\) 的大小.
\end{problem}

\begin{solution}
当 \(a>0\) 时，\(a>-a\)；

当 \(a=0\) 时，\(a=-a\)；

当 \(a<0\) 时，\(a<-a\).
\end{solution}

\begin{problem}
已知 \(AB, AC\) 是 \(\odot O\) 的切线，\(B, C\) 是切点，\(\angle BAC=120^\circ\)，\(AO=10\mathrm{~cm}\)，求 \(\odot O\) 的半径和切线 \(AB\) 的长.

\end{problem}

\begin{solution}
\bitmapfigure[max width=4.8cm,max totalheight=4.2cm]{img/1977/heilongjiang_preliminary/q7_solution_fig1.png}

\(\angle BAO=\frac{1}{2}\angle BAC=60^\circ\)（圆外一点 \(A\) 与圆心的联线平分过这点的两条切线间的夹角）；
连 \(OB\)，则 \(OB\perp AB\)（过切点的半径垂直于切线）；
所以 \(r=OA\sin60^\circ=10\times\frac{\sqrt3}{2}=5\sqrt3\)，\(AB=OA\cos60^\circ=10\times\frac{1}{2}=5\).
\end{solution}

\begin{problem}
在直角 \(\triangle ABC\) 中，\(\angle C=90^\circ\)，\(CD\perp AB\). 求证 \( \frac{CD}{AD}=\frac{BD}{CD} \).

\end{problem}

\begin{solution}
\bitmapfigure[max width=4.8cm,max totalheight=4.2cm]{img/1977/heilongjiang_preliminary/q8_solution_fig1.png}

因为 \(\angle A\) 与 \(\angle B\) 互余，\(\angle A\) 与 \(\angle 2\) 互余，所以 \(\angle B=\angle 2\).
所以 \(\angle BDC=\angle ADC=90^\circ\)（\(CD\perp AB\)），
\(\triangle ADC\sim\triangle BDC\)，
故 \(\dfrac{CD}{BD}=\dfrac{AD}{CD}\)（相似形对应边成比例），即 \(\dfrac{CD}{AD}=\dfrac{BD}{CD}\)（更比定理）.
\end{solution}

\begin{problem}
甲、乙两个车间按计划每月共生产 \(450\text{ 件}\) 产品，粉碎“四人帮”之后，在工业学大寨运动的推动下，甲车间完成了计划的 \(120\%\)，乙车间完成了计划的 \(110\%\)，两车间共生产 \(515\text{ 件}\) 产品，问甲、乙两车间原计划各生产多少件产品？
\end{problem}

\begin{solution}
设甲车间原计划生产 \(x\text{ 件}\) 产品，乙车间原计划生产 \((450-x)\text{ 件}\) 产品，
则 \(x\times120\%+(450-x)\times110\%=515\)，
即 \(12x+4950-11x=5150\)，解得 \(x=200\)，故 \(450-200=250\).

答：甲车间原生产 \(200\text{ 件}\) 产品，乙车间原生产 \(250\text{ 件}\) 产品.
\end{solution}

\begin{problem}
计算（不查表）：
\(\sin(-510^\circ)\).
\end{problem}

\begin{solution}
\[
\sin(-510^\circ)=\sin(-360^\circ-150^\circ)=\sin(-150^\circ)=-\sin150^\circ=-\sin\left(\frac{\pi}{2}+60^\circ\right)=-\cos60^\circ=-\frac{1}{2}.
\]

\end{solution}

\begin{problem}
\(\left(\frac{1}{2}\right)^{-2}-9^{\frac{1}{2}}+5^0-\sqrt2\cdot\sqrt8+\log_{7}7\).
\end{problem}

\begin{solution}
\(\left(\frac{1}{2}\right)^{-2}-9^{\frac{1}{2}}+5^0-\sqrt2\cdot\sqrt8+\log_{7}7 =4-3+1-4+1=-1\).
\end{solution}

\begin{problem}
在 \(\triangle ABC\) 中，已知 \(\angle A=120^\circ\)，\(\angle B=45^\circ\)，\(AC=6\mathrm{~cm}\)，求 \(BC\) 的长.
\end{problem}

\begin{solution}
由正弦定理得 \(\dfrac{BC}{\sin120^\circ}=\dfrac{AC}{\sin45^\circ}\)，
故
\[
BC=\frac{AC}{\sin45^\circ}\cdot\sin120^\circ=\frac{6}{\frac{\sqrt2}{2}}\cdot\cos30^\circ=\frac{12}{\sqrt2}\cdot\frac{\sqrt3}{2}=3\sqrt6\mathrm{~cm}.
\]
\end{solution}

\begin{problem}
求函数 \(y=-x^2-2x+3\) 的最大值.
\end{problem}

\begin{solution}
因为 \(a<0\)，所以 \(y\) 有最大值.
当 \(x=-\dfrac{b}{2a}\) 时，\(y\) 有最大值.
即 \(x=-\dfrac{-2}{-2}=-1\) 时，\(y=4\) 为最大值.
\end{solution}
