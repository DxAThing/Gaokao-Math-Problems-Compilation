\chapter{1977年江苏常州市初试卷}

\section{解答题}

\begin{problem}
计算：\(-1-\left(-5\frac{1}{2}\right)\times\frac{4}{11}+\left(-2\right)^3\div\left[\left(-3\right)^2+7\right]\).
\end{problem}

\begin{solution}
因为 \(-5\frac{1}{2}=-\frac{11}{2}\)，所以
\[
-1-\left(-\frac{11}{2}\right)\times\frac{4}{11}+\left(-2\right)^3\div\left[\left(-3\right)^2+7\right]
=-1-(-2)+\frac{-8}{9+7}=1-\frac{1}{2}=\frac{1}{2}.
\]
\end{solution}

\begin{problem}
计算：\(-\frac{2}{2+\sqrt{2}}+\frac{1}{\sqrt{3}-\sqrt{2}}-\frac{3+\sqrt{3}}{3-\sqrt{3}}\).
\end{problem}

\begin{solution}
分别有 \(-\frac{2}{2+\sqrt{2}}=-\frac{2(2-\sqrt{2})}{4-2}=\sqrt{2}-2\)，
\(\frac{1}{\sqrt{3}-\sqrt{2}}=\frac{\sqrt{3}+\sqrt{2}}{3-2}=\sqrt{3}+\sqrt{2}\)，
\(\frac{3+\sqrt{3}}{3-\sqrt{3}}=\frac{(3+\sqrt{3})^2}{9-3}=2+\sqrt{3}\).
因此原式
\[
=(\sqrt{2}-2)+(\sqrt{3}+\sqrt{2})-(2+\sqrt{3})=2\sqrt{2}-4.
\]
\end{solution}

\begin{problem}
计算：\(\log_2\left(1+\lg 10\right)\).
\end{problem}

\begin{solution}
由常用对数的定义，\(\lg 10=1\)，故
\(\log_2\left(1+\lg 10\right)=\log_2 2=1\).
\end{solution}

\begin{problem}
\(\sin 390^\circ\cos\left(-60^\circ\right)+\sin 120^\circ\cos 750^\circ\).
\end{problem}

\begin{solution}
利用三角函数的周期性和偶性，\(\sin 390^\circ=\sin 30^\circ=\frac{1}{2}\)，
\(\cos\left(-60^\circ\right)=\cos 60^\circ=\frac{1}{2}\)，
\(\sin 120^\circ=\sin 60^\circ=\frac{\sqrt{3}}{2}\)，
\(\cos 750^\circ=\cos 30^\circ=\frac{\sqrt{3}}{2}\). 因此原式
\[
=\frac{1}{2}\cdot\frac{1}{2}+\frac{\sqrt{3}}{2}\cdot\frac{\sqrt{3}}{2}
=\frac{1}{4}+\frac{3}{4}=1.
\]
\end{solution}

\begin{problem}
化简：\(a+\sqrt{a^2-2a+1}\).
\end{problem}

\begin{solution}
因为 \(a^2-2a+1=(a-1)^2\)，所以 \(a+\sqrt{a^2-2a+1}=a+\lvert a-1\rvert\).
当 \(a\geq 1\) 时，\(\lvert a-1\rvert=a-1\)，原式为 \(2a-1\)；
当 \(a<1\) 时，\(\lvert a-1\rvert=1-a\)，原式为 \(1\). 故原式
\[
=\begin{cases}
2a-1,&a\geq 1,\\
1,&a<1.
\end{cases}
\]
\end{solution}

\begin{problem}
已知 \(a=9\)，\(b=\sqrt{3}\)，求 \(a^{\frac{3}{2}}b\left(\frac{b^2}{a}\right)^{\frac{1}{2}}\div\left(\frac{1}{a}\right)^{-\frac{1}{2}}\) 的值.
\end{problem}

\begin{solution}
先分别计算各因式：
\(a^{\frac{3}{2}}=9^{\frac{3}{2}}=27\)，\(\left(\frac{b^2}{a}\right)^{\frac{1}{2}}=\left(\frac{3}{9}\right)^{\frac{1}{2}}=\frac{1}{\sqrt{3}}\)，\(\left(\frac{1}{a}\right)^{-\frac{1}{2}}=\left(\frac{1}{9}\right)^{-\frac{1}{2}}=3\).
所以原式 \(=27\cdot\sqrt{3}\cdot\frac{1}{\sqrt{3}}\div 3=9\).
\end{solution}

\begin{problem}
求函数 \(y=\sqrt{x-3}+\sqrt{5-x}\) 的自变量的取值范围.
\end{problem}

\begin{solution}
要使根式有意义，必须同时满足 \(x-3\geq 0\) 且 \(5-x\geq 0\).
由此得 \(x\geq 3\) 且 \(x\leq 5\)，所以自变量的取值范围是 \([3,5]\).
\end{solution}

\begin{problem}
求证：\(\cot\frac{\pi}{8}-\tan\frac{\pi}{8}=2\).
\end{problem}

\begin{solution}
令 \(\theta=\frac{\pi}{8}\). 由于 \(\sin\theta\ne 0\) 且 \(\cos\theta\ne 0\)，有
\[
\cot\theta-\tan\theta
=\frac{\cos\theta}{\sin\theta}-\frac{\sin\theta}{\cos\theta}
=\frac{\cos^2\theta-\sin^2\theta}{\sin\theta\cos\theta}
=\frac{\cos 2\theta}{\frac{1}{2}\sin 2\theta}=2\cot 2\theta.
\]
代入 \(\theta=\frac{\pi}{8}\)，得 \(\cot\frac{\pi}{8}-\tan\frac{\pi}{8}=2\cot\frac{\pi}{4}=2\).
\end{solution}

\begin{problem}
要建造轮廓为矩形的猪舍四间，按实线部分砌墙，若墙的总长为 \(32\text{ 米}\)，问每间猪舍的长、宽为几米时面积最大？每间猪舍的最大面积是多少？

\FigureLayoutDeclare{img/1977/jiangsu_changzhou_city/q9_fig1.png}{6.2cm}{0bp 0bp 12bp 0bp}
\bitmapfigure[max width=6.2cm]{img/1977/jiangsu_changzhou_city/q9_fig1.png}
\end{problem}

\begin{solution}
设每间猪舍沿四间并列方向的长为 \(x\text{ 米}\)，垂直于并列方向的宽为 \(y\text{ 米}\). 由图可知，实线墙包括一条总长为 \(4x\) 的外侧横墙和五条长为 \(y\) 的竖墙，因此约束条件为 \(4x+5y=32\)，且 \(x>0\)，\(y>0\).
每间猪舍的面积为 \(S=xy\). 由约束条件得 \(y=\frac{32-4x}{5}\)，于是
\[
S=x\cdot\frac{32-4x}{5}=-\frac{4}{5}x^2+\frac{32}{5}x
=-\frac{4}{5}(x-4)^2+\frac{64}{5}.
\]
因为 \(-\frac{4}{5}(x-4)^2\leq 0\)，当且仅当 \(x=4\) 时面积取得最大值. 此时
\(y=\frac{32-4\times 4}{5}=\frac{16}{5}\)，\(S_{\max}=\frac{64}{5}\).
所以每间猪舍的长为 \(4\text{ 米}\)，宽为 \(\frac{16}{5}\text{ 米}\)，每间最大面积为 \(\frac{64}{5}\text{ 平方米}\).
\end{solution}

\begin{problem}
如图：割线 \(PO\) 交 \(\odot O\) 于 \(C\)，\(D\)，割线 \(PA\) 交 \(\odot O\) 于 \(A\)，\(B\)，

\begin{enumerate}
    \item 试证：\(PA\cdot PB=PC\cdot PD\)；
    \item 已知：\(\odot O\) 的半径为 \(1\)，\(PO=2\)，\(AB=1\)，求 \(PA\) 的长.
\end{enumerate}

\FigureLayoutDeclare{img/1977/jiangsu_changzhou_city/q10_fig1.png}{6.4cm}{10bp 32bp 2bp 4bp}
\bitmapfigure[max width=6.2cm]{img/1977/jiangsu_changzhou_city/q10_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 连接 \(AC\) 和 \(BD\). 因为 \(P,B,A\) 共线，\(P,D,C\) 共线，所以 \(\angle APC=\angle DPB\). 又因射线 \(AP\) 与 \(AB\) 同向，射线 \(DP\) 与 \(DC\) 反向，所以 \(\angle PAC=\angle BAC\)，\(\angle PDB=180^\circ-\angle CDB\). 四边形 \(ABDC\) 内接于圆，故圆内接四边形的对角互补，即 \(\angle BAC+\angle CDB=180^\circ\)，从而 \(\angle PAC=\angle PDB\). 所以 \(\triangle PAC\sim\triangle PDB\)，得到
\(\frac{PA}{PD}=\frac{PC}{PB}\)，即 \(PA\cdot PB=PC\cdot PD\).

\item 由半径为 \(1\)、\(PO=2\)，且 \(C,O,D,P\) 依次共线，有 \(PC=PO+OC=3\)，\(PD=PO-OD=1\).
设 \(PB=x\)，则 \(PA=PB+AB=x+1\). 利用（1）的结论，
\(x(x+1)=PA\cdot PB=PC\cdot PD=3\)，即 \(x^2+x-3=0\). 解得 \(x=\frac{-1\pm\sqrt{13}}{2}\). 由于线段长度为正，\(x=PB=\frac{\sqrt{13}-1}{2}\)，故
\(PA=PB+AB=\frac{\sqrt{13}-1}{2}+1=\frac{\sqrt{13}+1}{2}\).
\end{enumerate}
\end{solution}

\begin{problem}
右图是某物体的二视图，按 \(1:1\) 的比例画出这个物体的直观图，并计算它的侧面积.
\FigureLayoutDeclare{img/1977/jiangsu_changzhou_city/q11_fig1.png}{6.0cm}{0bp 16bp 7bp 8bp}
\bitmapfigure[max width=6.0cm]{img/1977/jiangsu_changzhou_city/q11_fig1.png}
\end{problem}

\begin{solution}
二视图表明，该物体是一个底面为边长 \(24\) 的正方形、顶点在底面中心正上方的正四棱锥；正视图中的高为 \(16\). 按 \(1:1\) 比例作直观图时，先作边长为 \(24\) 的正方形底面，再在底面中心正上方取高为 \(16\) 的顶点，并将该顶点与底面四个顶点连接.

取任意一个侧面，设其斜高为 \(l\). 斜高的下端是底面一边的中点，因此由勾股定理
\[
l=\sqrt{16^2+\left(\frac{24}{2}\right)^2}=\sqrt{256+144}=20.
\]
正四棱锥有四个全等的侧面，所以侧面积为
\[
4\times\frac{1}{2}\times 24\times 20=960.
\]
故该物体的侧面积为 \(960\text{ 平方单位}\).
\end{solution}

\begin{problem}
已知：三点 \(A\left(-2,9\right)\)，\(B\left(4,-4\right)\)，\(C\left(10,8\right)\).

试求：

\begin{enumerate}
    \item 过 \(A\) 点且垂直于 \(BC\) 的直线方程；
    \item \(A\) 点到 \(BC\) 的距离.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 直线 \(BC\) 的斜率为 \(k_{BC}=\frac{8-(-4)}{10-4}=2\).
所以过 \(A\) 点且垂直于 \(BC\) 的直线斜率为 \(-\frac{1}{2}\). 由点斜式，所求直线满足
\(y-9=-\frac{1}{2}(x+2)\)，整理得 \(x+2y-16=0\).

\item 由点 \(B\)、\(C\) 可得直线 \(BC\) 的方程为 \(y+4=2(x-4)\)，即 \(2x-y-12=0\). 所以点 \(A\left(-2,9\right)\) 到直线 \(BC\) 的距离为
\[
d=\frac{\left|2\times(-2)-9-12\right|}{\sqrt{2^2+(-1)^2}}=\frac{25}{\sqrt{5}}=5\sqrt{5}.
\]
\end{enumerate}
\end{solution}

\begin{problem}
如图：在直角坐标系中，\(OA_0=1\)，以 \(OA_0\) 为一条直角边画出第一个直角三角形 \(OA_0A_1\)，又以 \(OA_1\) 为一条直角边画出第二个直角三角形 \(OA_1A_2\)，再以 \(OA_2\) 为一条直角边画出第三个直角三角形 \(OA_2A_3\). 依次继续画下去，试求：

\begin{enumerate}
    \item \(A_0\) 点的极坐标；
    \item 第十二个直角三角形的面积（精确到 \(0.01\)）.
\end{enumerate}

已知 \(\lg 2=0.3010\)，\(\lg 3=0.4771\).

\end{center}
\end{problem}

\begin{solution}
由图可知，点 \(A_0\) 在正半轴上，且 \(OA_0=1\)，所以 \(A_0\) 的极坐标为 \(\left(1,0\right)\).

相邻两条射线 \(OA_{n-1}\)、\(OA_n\) 的夹角均为 \(\frac{\pi}{6}\)，且三角形 \(OA_{n-1}A_n\) 在 \(A_{n-1}\) 处为直角. 设 \(r_n=OA_n\)，则在直角三角形 \(OA_{n-1}A_n\) 中，
\[
\cos\frac{\pi}{6}=\frac{OA_{n-1}}{OA_n}=\frac{r_{n-1}}{r_n}
\]
从而
\[
r_n=\frac{r_{n-1}}{\cos\frac{\pi}{6}}=\frac{2}{\sqrt{3}}r_{n-1}.
\]
由 \(r_0=1\)，递推得
\[
r_{11}=\left(\frac{2}{\sqrt{3}}\right)^{11}.
\]
第十二个三角形为 \(OA_{11}A_{12}\). 其一条直角边为 \(r_{11}\)，另一条直角边为
\[
A_{11}A_{12}=r_{11}\tan\frac{\pi}{6}=\frac{r_{11}}{\sqrt{3}}.
\]
因此第十二个三角形的面积为
\[
S_{12}=\frac{1}{2}r_{11}\cdot\frac{r_{11}}{\sqrt{3}}
=\frac{1}{2\sqrt{3}}\left(\frac{2}{\sqrt{3}}\right)^{22}
=\frac{2^{21}}{3^{23/2}}.
\]
利用题给对数，
\[
\lg S_{12}=21\lg 2-\frac{23}{2}\lg 3
=21\times0.3010-\frac{23}{2}\times0.4771=0.83435.
\]
查反对数表并按题意取近似值，得
\[
S_{12}\approx 6.83.
\]
故第十二个直角三角形的面积约为 \(6.83\text{ 平方单位}\).
\end{solution}
