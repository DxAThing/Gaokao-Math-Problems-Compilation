一，将下列各式化成最简形式.

\begin{problem}
将下式化成最简形式：\( \sqrt[4]{a^{-2}b^3}\cdot\sqrt[3]{a^4b^{-2}}\div\sqrt[6]{a^5\cdot b^{-2}} \)

\end{problem}

\begin{solution}
解法一：
\[
\text{原式}
=\frac{\sqrt[12]{a^{-6}b^9a^{16}b^{-8}}}{\sqrt[12]{a^{10}b^{-4}}}
=\sqrt[12]{\frac{a^{10}b}{a^{10}b^{-4}}}
=\sqrt[12]{b^5}.
\]

解法二：
\[
\text{原式}
=a^{-\frac{2}{4}}b^{\frac{3}{4}}\cdot a^{\frac{4}{3}}b^{-\frac{2}{3}}\cdot a^{-\frac{5}{6}}b^{\frac{2}{6}}
=b^{\frac{5}{12}}
=\sqrt[12]{b^5}.
\]
\end{solution}

\begin{problem}
将下式化成最简形式：\( \log_2\cos\frac{\pi}{4}-\log_2\sin\frac{\pi}{6} \)

\end{problem}

\begin{solution}
解法一：
\[
\text{原式}
=\log_2\frac{\sqrt{2}}{2}-\log_2\frac{1}{2}
=\frac{1}{2}\log_2 2-\log_2 2-\log_2 1+\log_2 2
=\frac{1}{2}-1-0+1=\frac{1}{2}.
\]

解法二：
\[
\text{原式}
=\log_2\frac{\sqrt{2}}{2}-\log_2\frac{1}{2}
=\log_2 2^{-\frac{1}{2}}-\log_2 2^{-1}
=-\frac{1}{2}-\left( -1 \right)=\frac{1}{2}.
\]

解法三：
\[
\text{原式}
=\log_2\left( \frac{\sqrt{2}}{2}\div\frac{1}{2} \right)
=\log_2\sqrt{2}
=\frac{1}{2}.
\]
\end{solution}

二，计算下列各题.

\begin{problem}
已知 \( \tan\alpha=\frac{2ab}{a^2-b^2} \)，求 \( \sin\alpha \).

\end{problem}

\begin{solution}
解法一：
\[
\sec^2\alpha
=1+\tan^2\alpha
=1+\frac{4a^2b^2}{\left( a^2-b^2 \right)^2}
=\frac{\left( a^2+b^2 \right)^2}{\left( a^2-b^2 \right)^2}.
\]

所以 \( \sec\alpha=\pm\frac{a^2+b^2}{a^2-b^2} \).

故 \( \cos\alpha=\pm\frac{a^2-b^2}{a^2+b^2} \).

所以
\[
\sin\alpha
=\tan\alpha\cdot\cos\alpha
=\frac{2ab}{a^2-b^2}\cdot\left( \pm\frac{a^2-b^2}{a^2+b^2} \right)
=\pm\frac{2ab}{a^2+b^2}.
\]

解法二：
\[
\left\{
\begin{aligned}
\frac{\sin\alpha}{\cos\alpha}&=\frac{2ab}{a^2-b^2}\\
\sin^2\alpha+\cos^2\alpha&=1
\end{aligned}
\right.
\]

由第一式得 \( \cos\alpha=\frac{a^2-b^2}{2ab}\sin\alpha \).

代入第二式，得 \( \sin^2\alpha=\frac{4a^2b^2}{\left( a^2+b^2 \right)^2} \).

所以 \( \sin\alpha=\pm\frac{2ab}{a^2+b^2} \).
\end{solution}

\begin{problem}
已知 \( m+\frac{1}{m}=2 \)，求 \( m^2+\frac{1}{m^2}=? \) \( m^3+\frac{1}{m^3}=? \)

\end{problem}

\begin{solution}
解法一：
\( m^2+\frac{1}{m^2}=\left( m+\frac{1}{m} \right)^2-2m\cdot\frac{1}{m}=2^2-2=2 \).

\[
m^3+\frac{1}{m^3}
=\left( m+\frac{1}{m} \right)^3-3m\cdot\frac{1}{m}\left( m+\frac{1}{m} \right)
=2^3-3\cdot 2=2.
\]

解法二：
由已知等式 \( m+\frac{1}{m}=2 \)，可得 \( m^2-2m+1=0 \)，即 \( \left( m-1 \right)^2=0 \).

因此 \( m=1 \). 故 \( m^2+\frac{1}{m^2}=1^2+\frac{1}{1^2}=2 \)，\( m^3+\frac{1}{m^3}=1^3+\frac{1}{1^3}=2 \).
\end{solution}

\guizhouproblabel{三，}
\begin{problem}
以直角三角形 \( ABC \) 的一直角边 \( AB \) 为直径作圆，此圆与斜边 \( AC \) 交于 \( D \)，过 \( D \) 引圆的切线交 \( BC \) 于 \( E \). 试证 \( EB=CE \).

已知：\( AB \) 为直径，\( \angle ABC=90^\circ \)，\( AC \) 与圆相交于 \( D \)，\( DE \) 是圆的切线.

求证：\( EB=CE \).

\bitmapfigure[max width=5.8cm,max totalheight=5.4cm]{img/1977/guizhou/q5_fig1.png}

\end{problem}
\endgroup

\begin{solution}
证明：连接 \( BD \) 则 \( \angle ADB=90^\circ \).
因为 \( \angle ABC=90^\circ \)，\( AB \) 又是直径，所以 \( EB \) 是圆的切线.
因此 \( ED=EB \).
所以
\[
\angle 3=\angle 4. \tag*{（1）}
\]
又因为 \( \angle 1+\angle 3=90^\circ \)，\( \angle 2+\angle 4=90^\circ \)，而 \( \angle 3=\angle 4 \)（已证），所以 \( \angle 1=\angle 2 \).
从而
\[
CE=ED. \tag*{（2）}
\]
由（1）和（2）即得 \( EB=CE \).
\end{solution}

\guizhouproblabel{四，}
\begin{problem}
求到点 \( A\left( 1,2 \right) \) 和点 \( B\left( 4,1 \right) \) 等距离的点的轨迹的方程.

\end{problem}
\endgroup

\begin{solution}
解：设 \( P\left( x,y \right) \) 是到 \( A \)，\( B \) 等距离的点，则 \( \left| PA \right|=\left| PB \right| \).

由两点间的距离公式得
\[
\sqrt{\left( x-1 \right)^2+\left( y-2 \right)^2}=\sqrt{\left( x-4 \right)^2+\left( y-1 \right)^2}.
\]

两边平方
\[
\left( x-1 \right)^2+\left( y-2 \right)^2=\left( x-4 \right)^2+\left( y-1 \right)^2.
\]

整理化简后得 \( P\left( x,y \right) \) 的轨迹方程为 \( 3x-y-6=0 \)（或者写成 \( y=3x-6 \)）.
\end{solution}

\guizhouproblabel{五，}
\begin{problem}
用长 8 米的木料，做一矩形的窗框，问长和宽如何选择，才能使窗子透过的光线最多？

\end{problem}
\endgroup

\begin{solution}
解：设窗子的宽为 \( x \) 米，则长为 \( \left( 4-x \right) \) 米. 而窗子的面积 \( S=x\left( 4-x \right)=-\left( x-2 \right)^2+4 \).

所以，当 \( x=2 \) 时，\( S \) 有极大值 4. 这时，宽为 \( 4-x=2 \). 故窗子为边长 2 米的正方形时，透过的光线最多.

答：当窗子的长和宽相等，且都为 2 米时，才能使透过的光线最多.
\end{solution}

\guizhouproblabel{六，}
\begin{problem}
求 \( \displaystyle\lim_{n\to\infty}\sin a\left( 1+\cos a+\cos^2 a+\cdots+\cos^n a \right)=? \)（其中 \( a \) 为任意实常数，\( 0<a<\frac{\pi}{2} \)）

\end{problem}
\endgroup

\begin{solution}
解：数列 \( 1 \)，\( \cos a \)，\( \cos^2 a \)，...，\( \cos^n a \)，... 为无穷递缩等比数列，其公比 \( 0<q=\cos a<1 \). 因此 \( \displaystyle\lim_{n\to\infty}\left( 1+\cos a+\cdots+\cos^n a \right)=\frac{1}{1-\cos a} \).

故
\[
\lim_{n\to\infty}\sin a\left( 1+\cos a+\cdots+\cos^n a \right)
=\sin a\cdot\lim_{n\to\infty}\left( 1+\cos a+\cdots+\cos^n a \right)
=\frac{\sin a}{1-\cos a}
=\cot\frac{a}{2}.
\]
\end{solution}

\guizhouproblabel{七，}
\begin{problem}
已知方程 \( x^2+px+q=0 \) 的两个根为 \( \tan\alpha \) 与 \( \tan\beta \)，求 \( \sin^2\left( \alpha+\beta \right)+p\sin\left( \alpha+\beta \right)\cos\left( \alpha+\beta \right)+q\cos^2\left( \alpha+\beta \right) \) 的值（其中 \( p \)，\( q \) 为任意实常数）.

\end{problem}
\endgroup

\begin{solution}
解：根据一元二次方程根与系数的关系：\( \tan\alpha+\tan\beta=-p \)，\( \tan\alpha\cdot\tan\beta=q \).

当 \( q\neq 1 \) 时，得 \( \tan\left( \alpha+\beta \right)=\dfrac{\tan\alpha+\tan\beta}{1-\tan\alpha\tan\beta}=\dfrac{-p}{1-q} \).

因此
\[
\sin^2\left( \alpha+\beta \right)+p\sin\left( \alpha+\beta \right)\cos\left( \alpha+\beta \right)+q\cos^2\left( \alpha+\beta \right)
\]
\[
=\cos^2\left( \alpha+\beta \right)\left[ \tan^2\left( \alpha+\beta \right)+p\tan\left( \alpha+\beta \right)+q \right]
\]
\[
=\frac{1}{1+\tan^2\left( \alpha+\beta \right)}\left[ \tan^2\left( \alpha+\beta \right)+p\tan\left( \alpha+\beta \right)+q \right]
\]
\[
=\frac{1}{1+\frac{p^2}{\left( 1-q \right)^2}}\left[ \frac{p^2}{\left( 1-q \right)^2}-\frac{p^2}{1-q}+q \right]
=q.
\]

当 \( q=1 \) 时，\( \cos\left( \alpha+\beta \right)=0 \)，\( \sin\left( \alpha+\beta \right)=\pm 1 \)，因此原式 \( =1 \).
\end{solution}

\guizhouproblabel{八，}
\begin{problem}
设有直线（\( L \)）与曲线（\( c \)）：

\( \left( L \right):y=ax-1 \).

\( \left( c \right):r\sin\theta=\sin 2\theta \).

问 \( \left( L \right) \) 与 \( \left( c \right) \) 在什么条件下相交，相切，不相交？

\end{problem}
\endgroup

\begin{solution}
解：将 \( \left( c \right) \) 化成直角坐标方程：
\[
\left( c \right):y=\frac{2xy}{x^2+y^2},
\]
即
\[
\frac{2x}{x^2+y^2}=1,
\]
或者写成
\[
\left( c \right):\left( x-1 \right)^2+y^2=1.
\]

把 \( y=ax-1 \) 代入 \( \left( x-1 \right)^2+y^2=1 \)，整理后得
\[
\left( 1+a^2 \right)x^2-2\left( a+1 \right)x+1=0.
\]

当这个方程的判别式 \( \Delta>0 \) 时，表示 \( \left( L \right) \) 与 \( \left( c \right) \) 有两个交点，即相交；当 \( \Delta=0 \) 时，表示 \( \left( L \right) \) 与 \( \left( c \right) \) 只有一个交点，即相切；当 \( \Delta<0 \) 时，表示 \( \left( L \right) \) 与 \( \left( c \right) \) 没有交点，即不相交. 下面分别求出这三种情况 \( a \) 的值.

当 \( \Delta>0 \) 时，即 \( 4\left( a+1 \right)^2-4\left( 1+a^2 \right)>0 \)，即 \( 8a>0 \).

因此 \( a>0 \) 时 \( \left( L \right) \) 与 \( \left( c \right) \) 相交.

当 \( \Delta=0 \) 时，即 \( 4\left( a+1 \right)^2-4\left( 1+a^2 \right)=0 \)，即 \( 8a=0 \).

因此 \( a=0 \) 时相切.

当 \( \Delta<0 \) 时，即 \( 4\left( a+1 \right)^2-4\left( 1+a^2 \right)<0 \)，即 \( 8a<0 \).

因此 \( a<0 \) 时不相交.

故当 \( a>0 \) 时，\( \left( L \right) \) 与 \( \left( c \right) \) 相交；\( a=0 \) 时，\( \left( L \right) \) 与 \( \left( c \right) \) 相切；\( a<0 \) 时，\( \left( L \right) \) 与 \( \left( c \right) \) 不相交.
\end{solution}

\guizhouproblabel{一，}
\begin{problem}
设某平板上任意一点的电势，在 \( \rho\geqslant 1 \) 时，由 \( E=\ln\rho \) 给出，这里 \( \rho \) 是以原点为起点的矢径. 试求在点 \( \rho=5 \)，\( \theta=\arcsin 0.8 \) 处，沿此点至 \( \rho=10 \)，\( \theta=\frac{\pi}{2} \) 的方向 \( E \) 的变化率.

\end{problem}
\endgroup

\begin{solution}
解：把极坐标化为直角坐标：\( E=\ln\sqrt{x^2+y^2} \).

求在点 \( \left( 3,4 \right) \) 处，沿 \( \left( 3,4 \right) \) 至 \( \left( 0,10 \right) \) 方向上的 \( E \) 的变化率.

\bitmapfigure[max width=4.2cm,max totalheight=6.8cm]{img/1977/guizhou/q11_solution_fig1.png}

从 \( \left( 3,4 \right) \) 至 \( \left( 0,10 \right) \) 的方向记为 \( \bs{L} \)，则 \( \bs{L}=\left( 0-3 \right)\bs{i}+\left( 10-4 \right)\bs{j}=-3\bs{i}+6\bs{j} \).

其中 \( \bs{i} \)，\( \bs{j} \) 分别表示横轴，纵轴上的单位向量. 问题的实质，就是求 \( E \) 在 \( \bs{L} \) 方向上的方向导数 \( \frac{\partial E}{\partial \bs{L}} \). 求方向导数的计算公式是
\[
\frac{\partial E}{\partial \bs{L}}=\frac{\partial E}{\partial x}\cos\alpha+\frac{\partial E}{\partial y}\cos\beta.
\]

其中 \( \cos\alpha \)，\( \cos\beta \) 是 \( \bs{L} \) 的方向余弦.

\[
\frac{\partial E}{\partial x}=\frac{\partial\left( \ln\sqrt{x^2+y^2} \right)}{\partial x}=\frac{x}{x^2+y^2},
\qquad
\frac{\partial E}{\partial y}=\frac{\partial\left( \ln\sqrt{x^2+y^2} \right)}{\partial y}=\frac{y}{x^2+y^2}.
\]
\[
\cos\alpha=\frac{\bs{L}\cdot\bs{i}}{\left| \bs{L} \right|\left| \bs{i} \right|}=\frac{-3}{\sqrt{\left( -3 \right)^2+6^2}}=-\frac{1}{\sqrt{5}}.
\]
\[
\cos\beta=\frac{\bs{L}\cdot\bs{j}}{\left| \bs{L} \right|\left| \bs{j} \right|}=\frac{6}{\sqrt{\left( -3 \right)^2+6^2}}=\frac{2}{\sqrt{5}}.
\]

因此
\[
\frac{\partial E}{\partial \bs{L}}=-\frac{1}{\sqrt{5}}\cdot\frac{x}{x^2+y^2}+\frac{2}{\sqrt{5}}\cdot\frac{y}{x^2+y^2}.
\]

在点 \( \left( 3,4 \right) \) 处，\( \dfrac{\partial E}{\partial \bs{L}}
=-\dfrac{1}{\sqrt{5}}\cdot\dfrac{3}{25}+\dfrac{2}{\sqrt{5}}\cdot\dfrac{4}{25}
=\dfrac{\sqrt{5}}{25} \).

答：\( E \) 在点 \( \rho=5 \)，\( \theta=\arcsin 0.8 \) 处，沿此点至 \( \rho=10 \)，\( \theta=\frac{\pi}{2} \) 方向上的变化率为 \( \frac{\sqrt{5}}{25} \).
\end{solution}

\guizhouproblabel{二，}
\begin{problem}
证明 \( S_1=1 \)，\( S_2=1+\frac{1}{2^2} \)，\( S_3=1+\frac{1}{2^2}+\frac{1}{3^2} \)，...，\( S_n=1+\frac{1}{2^2}+\frac{1}{3^2}+\cdots+\frac{1}{n^2} \)，... 是收敛数列.

\end{problem}
\endgroup

\begin{solution}
证明：因为 \( S_n-S_{n-1}=\dfrac{1}{n^2}>0 \)，所以此数列是单调上升数列.

又因为
\[
S_n=1+\frac{1}{2^2}+\frac{1}{3^2}+\cdots+\frac{1}{n^2}<1+\frac{1}{1\cdot 2}+\frac{1}{2\cdot 3}+\cdots+\frac{1}{\left( n-1 \right)n},
\]
而
\[
1+\frac{1}{1\cdot 2}+\frac{1}{2\cdot 3}+\cdots+\frac{1}{\left( n-1 \right)n}
=1+\left( 1-\frac{1}{2} \right)+\left( \frac{1}{2}-\frac{1}{3} \right)+\cdots+\left( \frac{1}{n-1}-\frac{1}{n} \right)
=2-\frac{1}{n}<2.
\]

此数列又有上界. 单调上升而又有上界的数列，必收敛. 故 \( \{S_n\} \) 收敛.
\end{solution}
