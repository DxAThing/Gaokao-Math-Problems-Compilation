\chapter{1977年江苏苏州地区初试卷}

\section{解答题}

\begin{problem}
化简：\(\left(3y-1\right)^2-\left(3y-1\right)\left(3y+1\right)+3\left(y-1\right)\).
\end{problem}

\begin{solution}
展开各项得 \(\left(3y-1\right)^2=9y^2-6y+1\)，\(\left(3y-1\right)\left(3y+1\right)=9y^2-1\)，所以原式 \(=9y^2-6y+1-9y^2+1+3y-3=-3y-1\).
\end{solution}

\begin{problem}
因式分解：\(4x^4-y^4\).
\end{problem}

\begin{solution}
连续使用平方差公式，得
\[
4x^4-y^4=\left(2x^2\right)^2-\left(y^2\right)^2=\left(2x^2-y^2\right)\left(2x^2+y^2\right)=\left(2x-y\right)\left(2x+y\right)\left(2x^2+y^2\right)
\]
\end{solution}

\begin{problem}
计算：\(\left(a+b-c\right)\left(a-b+c\right)\).
\end{problem}

\begin{solution}
将两因式分别写成 \(a+(b-c)\) 和 \(a-(b-c)\)，利用平方差公式，得 \(\left(a+b-c\right)\left(a-b+c\right)=a^2-(b-c)^2=a^2-b^2+2bc-c^2\).
\end{solution}

\begin{problem}
计算：\(\left(\frac{a-2b}{a+2b}-\frac{a+2b}{a-2b}\right)\div\frac{4ab}{4b^2c-a^2c}\).
\end{problem}

\begin{solution}
原式有意义且除数不为零时，分式中的各分母不为零，并且作为除数的分式不为零，因此
\(a\ne0\)、\(b\ne0\)、\(c\ne0\)、\(a-2b\ne0\)、\(a+2b\ne0\)，即
\(abc(a-2b)(a+2b)\ne0\). 先通分化简括号内的式子：
\[
\frac{a-2b}{a+2b}-\frac{a+2b}{a-2b}=\frac{(a-2b)^2-(a+2b)^2}{a^2-4b^2}=\frac{-8ab}{a^2-4b^2}=\frac{8ab}{4b^2-a^2}
\]
又因为 \(4b^2c-a^2c=c(4b^2-a^2)\)，所以原式
\(=\frac{8ab}{4b^2-a^2}\cdot\frac{c(4b^2-a^2)}{4ab}=2c\).
\end{solution}

\begin{problem}
计算：\(\sqrt[3]{1-\sqrt{2}}\cdot\sqrt[6]{3+2\sqrt{2}}-\frac{1}{1-\sqrt{2}}\).
\end{problem}

\begin{solution}
注意到 \(3+2\sqrt{2}=\left(1+\sqrt{2}\right)^2\)，且 \(1+\sqrt{2}>0\)，所以取算术根时
\(\sqrt[6]{3+2\sqrt{2}}=\sqrt[6]{\left(1+\sqrt{2}\right)^2}=\sqrt[3]{1+\sqrt{2}}\). 因而
\[
\sqrt[3]{1-\sqrt{2}}\cdot\sqrt[6]{3+2\sqrt{2}}=\sqrt[3]{\left(1-\sqrt{2}\right)\left(1+\sqrt{2}\right)}=\sqrt[3]{-1}=-1
\]
另一方面，\(\dfrac{1}{1-\sqrt{2}}=\dfrac{1+\sqrt{2}}{(1-\sqrt{2})(1+\sqrt{2})}=-(1+\sqrt{2})\). 所以原式
\(=-1-\left[-(1+\sqrt{2})\right]=\sqrt{2}\).
\end{solution}

\begin{problem}
解方程：\(3+\frac{5x-1}{2}=x-\frac{1-3x}{5}\).
\end{problem}

\begin{solution}
方程两边同乘 \(10\)，得
\[
30+5(5x-1)=10x-2(1-3x)
\]
化简得 \(25+25x=16x-2\)，所以 \(9x=-27\)，解得 \(x=-3\). 将 \(x=-3\) 代回原方程，左右两边均为 \(-5\)，故原方程的解为 \(x=-3\).
\end{solution}

\begin{problem}
解方程：\(x^2+x-2=0\).
\end{problem}

\begin{solution}
因式分解得 \(x^2+x-2=(x+2)(x-1)\). 因此
\(x+2=0\) 或 \(x-1=0\)，解得 \(x=-2\) 或 \(x=1\). 代回原方程，两值都使左边为零，故解集为 \(\{-2,1\}\).
\end{solution}

\begin{problem}
解方程组：\(\begin{cases}
3x+2y-1=0,\\
2x-3y+8=0.
\end{cases}\)
\end{problem}

\begin{solution}
原方程组等价于
\[
\begin{cases}
3x+2y=1,\\
2x-3y=-8.
\end{cases}
\]
将第一式乘以 \(3\)，第二式乘以 \(2\)，消去 \(y\)，得
\[
\begin{cases}
9x+6y=3,\\
4x-6y=-16.
\end{cases}
\]
两式相加，得 \(13x=-13\)，所以 \(x=-1\). 代入 \(3x+2y=1\)，得 \(-3+2y=1\)，从而 \(y=2\). 将 \(x=-1\)，\(y=2\) 代入原方程组，两式分别成立，因此方程组的解为 \(x=-1\)，\(y=2\).
\end{solution}

\begin{problem}
解不等式：\(3\left(2y-5\right)+4\geqslant8y-7\).
\end{problem}

\begin{solution}
展开并移项，得 \(6y-15+4\geqslant8y-7\)，即 \(6y-11\geqslant8y-7\). 两边减去 \(6y-7\)，得 \(-4\geqslant2y\)，再除以正数 \(2\)，不改变不等号方向，得到 \(y\leqslant-2\).
\end{solution}

\begin{problem}
计算 \(4\sin30^\circ-3\tan45^\circ+2\cos30^\circ\).
\end{problem}

\begin{solution}
由特殊角三角函数值
\(\sin30^\circ=\frac12\)、\(\tan45^\circ=1\)、\(\cos30^\circ=\frac{\sqrt{3}}2\)，得
\[
4\sin30^\circ-3\tan45^\circ+2\cos30^\circ=4\cdot\frac12-3\cdot1+2\cdot\frac{\sqrt{3}}2=\sqrt{3}-1
\]
\end{solution}

\begin{problem}
在 \(\triangle ABC\) 中，\(\angle C=90^\circ\)，\(a=2\)，\(b=1\)，求 \(\sin B\)，\(\tan A\).
\end{problem}

\begin{solution}
按通常的边角记号，\(a=BC=2\)、\(b=CA=1\)，而 \(c=AB\) 是直角三角形的斜边. 由勾股定理，
\(c=\sqrt{a^2+b^2}=\sqrt{2^2+1^2}=\sqrt5\). 对于角 \(B\)，对边为 \(AC=b\)，斜边为 \(AB=c\)，所以
\(\sin B=\dfrac{b}{c}=\dfrac1{\sqrt5}=\dfrac{\sqrt5}{5}\). 对于角 \(A\)，对边为 \(BC=a\)，邻直角边为 \(CA=b\)，所以 \(\tan A=\dfrac{a}{b}=2\).
\end{solution}

\begin{problem}
在 \(\triangle ABC\) 中，\(M\)，\(N\) 分别是 \(AB\)，\(AC\) 的中点，如果 \(\triangle ABC\) 的面积是 \(20\mathrm{~cm}^2\)，求 \(\triangle AMN\) 的面积.
\end{problem}

\begin{solution}
因为 \(M\)，\(N\) 分别是 \(AB\)，\(AC\) 的中点，所以
\(\dfrac{AM}{AB}=\dfrac{AN}{AC}=\dfrac12\)，且夹角 \(\angle MAN=\angle BAC\). 由两边对应成比例且夹角相等，\(\triangle AMN\sim\triangle ABC\)，相似比为 \(\dfrac12\). 相似三角形的面积比等于相似比的平方，因此
\[
\frac{S_{\triangle AMN}}{S_{\triangle ABC}}=\left(\frac12\right)^2=\frac14
\]
所以 \(S_{\triangle AMN}=\dfrac14\times20=5\mathrm{~cm}^2\).
\end{solution}

\begin{problem}
如图，\(AB\) 为半圆的直径，\(CD\perp AB\)，求证：\(CD^2=AD\cdot BD\).

\bitmapfigure[max width=4.0cm]{img/1977/jiangsu_suzhou_region/q4_fig1.png}
\end{problem}

\begin{solution}
因为 \(AB\) 是半圆的直径，所以由直径所对的圆周角为直角，得 \(\angle ACB=90^\circ\). 又因为 \(CD\perp AB\)，所以 \(\angle ADC=\angle CDB=90^\circ\). 由于 \(D\) 在 \(AB\) 上，\(\angle CAD=\angle CAB\)，而在直角三角形 \(ABC\) 中 \(\angle CAB+\angle CBA=90^\circ\). 因此 \(\angle ACD=90^\circ-\angle CAD=\angle CBA=\angle CBD\).

于是 \(\triangle ACD\sim\triangle CDB\)，对应关系为 \(A\leftrightarrow C\)、\(C\leftrightarrow B\)、\(D\leftrightarrow D\). 所以
\[
\frac{AD}{CD}=\frac{CD}{BD}
\]
从而 \(CD^2=AD\cdot BD\).
\end{solution}

\begin{problem}
求 \(P_1\left(0,-5\right)\) 和 \(P_2\left(-1,2\right)\) 两点间的距离.
\end{problem}

\begin{solution}
两点的横坐标之差为 \(-1-0=-1\)，纵坐标之差为 \(2-(-5)=7\). 由两点间距离公式，
\[
P_1P_2=\sqrt{\left[-1-0\right]^2+\left[2-(-5)\right]^2}=\sqrt{1+49}=\sqrt{50}=5\sqrt2
\]
\end{solution}

\begin{problem}
某机器厂第三季度共制造机器 \(331\text{ 台}\)，已知七月份制造 \(100\text{ 台}\)，求第三季度的月平均增长率.
\end{problem}

\begin{solution}
设月平均增长率为 \(r\)，则八月份和九月份的产量分别为 \(100(1+r)\) 台和 \(100(1+r)^2\) 台. 由第三季度总产量得
\[
100+100(1+r)+100(1+r)^2=331
\]
令 \(q=1+r\)，则 \(q>0\)，且
\[
100q^2+100q-231=0
\]
解得
\[
q=\frac{-100\pm\sqrt{100^2+4\times100\times231}}{2\times100}=\frac{-100\pm320}{200}=1.1\ \text{或}\ -2.1
\]
由于 \(q=1+r\) 是相邻月份产量的正比值，必须取正值，故 \(q=1.1\). 因此 \(r=q-1=0.1=10\%\).
\end{solution}

\begin{problem}
在 \(\triangle ABC\) 中，\(\angle A=45^\circ\)，\(AB+AC=10\mathrm{~cm}\).

\begin{enumerate}
\item 用公式法表示 \(\triangle ABC\) 的面积 \(S\) 与一边长 \(AB\)（\(x\)）之间的函数关系；
\item \(x\) 为何值时 \(S\) 为最大；
\item 求 \(S\) 的最大值.
\end{enumerate}
\end{problem}

\begin{solution}
设 \(AB=x\)，则 \(AC=10-x\). 因为三角形两边长均为正，所以定义域为 \(0<x<10\).
\begin{enumerate}
\item 由三角形面积公式，\(S=\dfrac12AB\cdot AC\cdot\sin A\)，所以 \(S=\dfrac12x(10-x)\sin45^\circ=\dfrac{\sqrt2}{4}x(10-x)\).

\item 配方得 \(S=\dfrac{\sqrt2}{4}\left[25-(x-5)^2\right]\). 因为 \((x-5)^2\geqslant0\)，所以 \(S\leqslant\dfrac{25\sqrt2}{4}\). 又 \(x=5\) 属于定义域，故当且仅当 \(x=5\) 时，\(S\) 取得最大值.

\item 当 \(x=5\)，即 \(AB=5\mathrm{~cm}\)、\(AC=5\mathrm{~cm}\) 时，\(S\) 的最大值为 \(\dfrac{25\sqrt2}{4}\mathrm{~cm}^2\).
\end{enumerate}
\end{solution}

\begin{problem}
过点 \(P\left(1,0\right)\) 作圆 \(\left(x+1\right)^2+y^2=1\) 的切线，求切线方程.
\end{problem}

\begin{solution}
圆心为 \(O\left(-1,0\right)\)，半径为 \(1\). 过点 \(P\) 且斜率不存在的直线是 \(x=1\)，其到圆心的距离为 \(2\)，不等于半径，故它不是切线.

设切线斜率为 \(k\)，则切线方程为 \(y=k(x-1)\)，即 \(kx-y-k=0\). 由圆心到切线的距离等于半径，得
\[
\frac{\left|k(-1)-0-k\right|}{\sqrt{k^2+1}}=1
\]
即 \(\dfrac{|-2k|}{\sqrt{k^2+1}}=1\). 两边平方得 \(4k^2=k^2+1\)，所以 \(k=\pm\dfrac1{\sqrt3}\).

因此两条切线方程为
\(y=\dfrac{x-1}{\sqrt3}\) 和 \(y=-\dfrac{x-1}{\sqrt3}\)，分别化为
\(x-\sqrt3y-1=0\) 和 \(x+\sqrt3y-1=0\).
\end{solution}

\section{附加题}

\begin{problem}
求所有被 \(3\) 除余数为 \(1\) 的两位数的和.
\end{problem}

\begin{solution}
这些两位数依次为 \(10\)，\(13\)，\(16\)，\(\ldots\)，\(97\)，构成首项为 \(10\)、末项为 \(97\)、公差为 \(3\) 的等差数列. 项数为
\(n=\dfrac{97-10}{3}+1=30\)，所以它们的和为
\[
\frac{30(10+97)}2=1605
\]
\end{solution}

\begin{problem}
求立方抛物线 \(y=x^3\) 和直线 \(y=1\)，\(x=0\) 所围成的图形的面积.

提示：\(1^3+2^3+3^3+\cdots+n^3=\left[\frac{n\left(n+1\right)}{2}\right]^2\).
\end{problem}

\begin{solution}
曲线 \(y=x^3\) 与直线 \(y=1\) 的交点满足 \(x^3=1\)，所以所围图形在横坐标方向的范围是 \(0\leqslant x\leqslant1\). 将区间 \([0,1]\) 等分成 \(n\) 份，每份宽为 \(\dfrac1n\)，取各小区间的右端点，则曲线下方部分的面积为
\[
S_1=\lim_{n\to\infty}\frac1n\sum_{k=1}^{n}\left(\frac{k}{n}\right)^3=\lim_{n\to\infty}\frac1{n^4}\sum_{k=1}^{n}k^3=\lim_{n\to\infty}\frac1{n^4}\left[\frac{n(n+1)}2\right]^2=\lim_{n\to\infty}\frac{(n+1)^2}{4n^2}=\frac14
\]
直线 \(y=1\) 与坐标轴围成的单位正方形面积为 \(1\)，所求图形是该正方形减去曲线下方部分，因此面积为 \(1-S_1=\dfrac34\).
\end{solution}
