\chapter{1977年江苏淮阴地区初试卷}

\begin{problem}
\begin{enumerate}
    \item 计算：\(4(-1)^5\times 0.25+8(-0.5^3)-0.4\div(-0.2^2)\).

    \item 计算：\(\left[3^{\frac{1}{2}}+\left(\frac{1}{2^{-\frac{2}{3}}}\right)^{\frac{2}{3}}\right]\div\left[3^{0.5}-\left(2\sqrt{2}\right)^{\frac{2}{8}}\right]\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
    \item 先按幂的运算顺序计算各项，得 \(4(-1)^5\times 0.25=-1\)，\(8(-0.5^3)=8\times(-0.125)=-1\)，\(-0.4\div(-0.2^2)=(-0.4)\div(-0.04)=10\). 因此原式的值为 \(-1-1+10=8\).

    \item 因为 \(3^{\frac12}=3^{0.5}=\sqrt{3}\)，且
    \[
    \left(\frac{1}{2^{-\frac23}}\right)^{\frac23}=\left(2^{\frac23}\right)^{\frac23}=2^{\frac49}
    \]
    以及
    \[
    \left(2\sqrt{2}\right)^{\frac28}=\left(2^{\frac32}\right)^{\frac14}=2^{\frac38}.
    \]
    所以原式的值为
    \[
    \frac{\sqrt{3}+2^{\frac49}}{\sqrt{3}-2^{\frac38}}.
    \]
\end{enumerate}
\end{solution}

\begin{problem}
化简：\(\dfrac{\left(a^2+1\right)^2-4a^2}{-2a^4+4a^2-2}\).
\end{problem}

\begin{solution}
原式的分母不能为零，因此 \(a^2\ne1\)，即 \(a\ne\pm1\). 分子、分母分别因式分解，得
\[
\frac{\left(a^2+1\right)^2-4a^2}{-2a^4+4a^2-2}
 =\frac{\left(a^2-1\right)^2}{-2\left(a^2-1\right)^2}=-1.
\]
所以原式的值为 \(-1\)，其中 \(a\ne\pm1\).
\end{solution}

\begin{problem}
求 \(\cos\left(-\dfrac{16\pi}{3}\right)\cdot\sin\dfrac{16\pi}{3}\) 的值.
\end{problem}

\begin{solution}
由三角函数的周期性和奇偶性，\(\cos\left(-\frac{16\pi}{3}\right)=\cos\frac{16\pi}{3}=\cos\frac{4\pi}{3}=-\frac12\)，\(\sin\frac{16\pi}{3}=\sin\frac{4\pi}{3}=-\frac{\sqrt{3}}2\). 因此
\[
\cos\left(-\frac{16\pi}{3}\right)\cdot\sin\frac{16\pi}{3}=\left(-\frac12\right)\left(-\frac{\sqrt{3}}2\right)=\frac{\sqrt{3}}4.
\]
\end{solution}

\begin{problem}
\begin{enumerate}
    \item 把 \(P\) 点的直角坐标 \(\left(\sqrt{3},-1\right)\) 化成极坐标.

    \item 把极坐标方程 \(\rho=2\sin\theta+3\cos\theta\) 化成直角坐标方程.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
    \item 设点 \(P\) 的极坐标为 \(\left(\rho,\theta\right)\)，则
    \[
    \rho=\sqrt{\left(\sqrt{3}\right)^2+(-1)^2}=2.
    \]
    又因为 \(\cos\theta=\frac{\sqrt{3}}2\)，\(\sin\theta=-\frac12\)，所以 \(\theta=-\frac\pi6+2k\pi\)，其中 \(k\in\Z\). 因而点 \(P\) 的极坐标可写为 \(\left(2,-\frac\pi6\right)\)，取辐角在 \([0,2\pi)\) 内时也可写为 \(\left(2,\frac{11\pi}{6}\right)\).

    \item 两边同乘以 \(\rho\)，并利用 \(x=\rho\cos\theta\)、\(y=\rho\sin\theta\)，得
    \[
    \rho^2=2\rho\sin\theta+3\rho\cos\theta
    \quad\Longleftrightarrow\quad
    x^2+y^2=2y+3x.
    \]
    所以对应的直角坐标方程为
    \[
    x^2+y^2-3x-2y=0
    \]
    或写成圆的标准方程
    \[
    \left(x-\frac32\right)^2+\left(y-1\right)^2=\frac{13}{4}.
    \]
\end{enumerate}
\end{solution}

\begin{problem}
以直角三角形 \(ABC\) 的直角边 \(BC\) 为直径作圆，交斜边 \(AB\) 于 \(D\)，\(E\) 是直角边 \(AC\) 的中点，连接 \(DE\)，则 \(DE\) 是此圆的切线（需画图）.
\end{problem}

\begin{solution}
设直角顶点为 \(C\)，取直角坐标系使 \(C=(0,0)\)、\(A=(0,a)\)、\(B=(b,0)\)，其中 \(a>0\)、\(b>0\). 则 \(E=\left(0,\frac a2\right)\)，圆心为 \(O=\left(\frac b2,0\right)\).

直线 \(AB\) 的方程为 \(y=a-\frac abx\)，以 \(BC\) 为直径的圆的方程为
\[
\left(x-\frac b2\right)^2+y^2=\left(\frac b2\right)^2
\]
即 \(x^2+y^2=bx\). 将直线 \(AB\) 的方程代入圆方程，得到
\[
x^2+a^2\left(1-\frac xb\right)^2=bx
\quad\Longleftrightarrow\quad
(x-b)\bigl((a^2+b^2)x-a^2b\bigr)=0.
\]
其中根 \(x=b\) 对应交点 \(B\)，所以另一交点 \(D=(x_D,y_D)\) 满足
\[
x_D=\frac{a^2b}{a^2+b^2}
\]
以及
\[
y_D=a-\frac abx_D=\frac{ab^2}{a^2+b^2}.
\]
于是 \(bx_D=ay_D\). 计算半径 \(OD\) 与直线 \(DE\) 的方向向量的内积，得
\[
\overrightarrow{OD}\cdot\overrightarrow{ED}
=\left(x_D-\frac b2,y_D\right)\cdot\left(x_D,y_D-\frac a2\right)
=x_D^2+y_D^2-\frac12\left(bx_D+ay_D\right).
\]
又因为 \(D\) 在圆上，\(x_D^2+y_D^2=bx_D\)，且 \(bx_D=ay_D\)，所以
\[
\overrightarrow{OD}\cdot\overrightarrow{ED}=bx_D-\frac12\left(bx_D+ay_D\right)=0.
\]
即 \(OD\perp DE\). 半径 \(OD\) 与过圆上点 \(D\) 的直线 \(DE\) 垂直，因此 \(DE\) 是此圆在 \(D\) 点的切线.
\end{solution}

\begin{problem}
用长为 \(8 \mathrm{~m}\) 的木料做一个“目”字形窗框，问窗框的宽和高各多少才能使通过的光线最多？
\end{problem}

\begin{solution}
设窗框的宽为 \(w\mathrm{~m}\)，高为 \(h\mathrm{~m}\)，其中 \(w>0\)、\(h>0\). “目”字形窗框的外框有两条宽边和两条高边，中间横档的长度也是 \(w\)，所以木料总长满足
\[
3w+2h=8.
\]
窗户的透光面积为外框内的矩形面积 \(S=wh\). 由约束条件得 \(h=\frac{8-3w}{2}\)，于是
\[
S=w\cdot\frac{8-3w}{2}=-\frac32\left(w-\frac43\right)^2+\frac83.
\]
因此当且仅当 \(w=\frac43\) 时，透光面积取得最大值. 此时
\[
h=\frac{8-3\times\frac43}{2}=2.
\]
所以窗框的宽为 \(\frac43\mathrm{~m}\)，高为 \(2\mathrm{~m}\)，最大透光面积为 \(\frac83\mathrm{~m}^2\).
\end{solution}

\begin{problem}
\begin{enumerate}
    \item 计算：\(\sqrt[3]{10^{3-\frac{1}{2}\lg 4}}\).

    \item 计算：\(\log_{2-\sqrt{3}}\left(2+\sqrt{3}\right)\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
    \item 因为 \(\lg 4=\lg 2^2=2\lg 2\)，所以
    \[
    10^{3-\frac12\lg4}=10^{3-\lg2}=\frac{10^3}{10^{\lg2}}=\frac{1000}{2}=500.
    \]
    因此
    \[
    \sqrt[3]{10^{3-\frac12\lg4}}=\sqrt[3]{500}=\sqrt[3]{125\times4}=5\sqrt[3]{4}.
    \]

    \item 因为 \(\left(2-\sqrt{3}\right)\left(2+\sqrt{3}\right)=1\)，所以
    \[
    2+\sqrt{3}=\left(2-\sqrt{3}\right)^{-1}.
    \]
    又因为 \(2-\sqrt{3}>0\) 且 \(2-\sqrt{3}\ne1\)，故
    \[
    \log_{2-\sqrt{3}}\left(2+\sqrt{3}\right)=-1.
    \]
\end{enumerate}
\end{solution}

\begin{problem}
有两点 \(A\left(-2,2\right)\) 和 \(B\left(2,-1\right)\)，试在 \(y\) 轴上找出点 \(P\)，使 \(\angle APB=90^\circ\).
\end{problem}

\begin{solution}
设点 \(P\) 在 \(y\) 轴上，记为 \(P=(0,t)\). 则
\[
\overrightarrow{PA}=(-2,2-t)
\]
以及
\[
\overrightarrow{PB}=(2,-1-t).
\]
由 \(\angle APB=90^\circ\)，两向量垂直，故
\[
\overrightarrow{PA}\cdot\overrightarrow{PB}=(-2)\times2+(2-t)(-1-t)=0.
\]
化简得
\[
-4+t^2-t-2=0
\quad\Longleftrightarrow\quad
t^2-t-6=0
\quad\Longleftrightarrow\quad
(t-3)(t+2)=0.
\]
所以 \(t=3\) 或 \(t=-2\)，所求点为 \(P_1=(0,3)\) 或 \(P_2=(0,-2)\).
\end{solution}

\begin{problem}
设在椭圆 \(\dfrac{x^2}{a^2}+\dfrac{y^2}{b^2}=1\) 上任意一点 \(P_1\left(x_1,y_1\right)\) 的切线与椭圆长轴的两端点 \(A'\) 和 \(A\) 的切线相交于 \(Q'\) 和 \(Q\)，求证：\(A'Q'\cdot AQ=b^2\).
\end{problem}

\begin{solution}
设椭圆长轴两端点为 \(A'=(-a,0)\)、\(A=(a,0)\). 题设中的交点 \(Q'\)、\(Q\) 存在时，\(P_1\) 不是长轴端点，故 \(y_1\ne0\).

点 \(P_1\) 处的切线方程为
\[
\frac{xx_1}{a^2}+\frac{yy_1}{b^2}=1.
\]
在 \(A'=(-a,0)\)、\(A=(a,0)\) 处，椭圆的切线分别为 \(x=-a\)、\(x=a\). 设
\[
Q'=\left(-a,y_{Q'}\right),\qquad Q=\left(a,y_Q\right).
\]
将 \(x=-a\) 和 \(x=a\) 分别代入 \(P_1\) 处的切线方程，得到
\[
y_{Q'}=\frac{b^2(a+x_1)}{ay_1}
\]
以及
\[
y_Q=\frac{b^2(a-x_1)}{ay_1}.
\]
因此
\[
A'Q'=|y_{Q'}|=\frac{b^2|a+x_1|}{a|y_1|}
\]
以及
\[
AQ=|y_Q|=\frac{b^2|a-x_1|}{a|y_1|}.
\]
由于 \(P_1\) 不是长轴端点，\(|x_1|<a\)，结合椭圆方程可得
\[
y_1^2=b^2\left(1-\frac{x_1^2}{a^2}\right)=\frac{b^2(a^2-x_1^2)}{a^2}.
\]
于是
\[
A'Q'\cdot AQ
=\frac{b^4(a^2-x_1^2)}{a^2y_1^2}
=\frac{b^4(a^2-x_1^2)}{b^2(a^2-x_1^2)}=b^2.
\]
所以 \(A'Q'\cdot AQ=b^2\).
\end{solution}
