\begin{problem}
计算 \(\sin\left(-990^\circ\right)\).
\end{problem}

\begin{solution}
\(\sin\left(-990^\circ\right)=-\sin990^\circ=-\sin\left(720^\circ+270^\circ\right)=-\sin270^\circ=1\).
\end{solution}

\begin{problem}
化简 \(\sqrt{4-12a+9a^2}\).
\end{problem}

\begin{solution}
\(\sqrt{4-12a+9a^2}=\sqrt{\left(2-3a\right)^2}=\left|2-3a\right|\).
因此
\[
\sqrt{4-12a+9a^2}=
\begin{cases}
2-3a, & a<\frac{2}{3},\\
0, & a=\frac{2}{3},\\
3a-2, & a>\frac{2}{3}.
\end{cases}
\]
\end{solution}

\begin{problem}
计算 \(\frac{(-27)^3\sqrt{\left(\frac{1}{16}\right)^{-\frac{1}{2}}}}{(-9)^0\left(\frac{3}{2}\right)^7\left(\frac{1}{2}\right)^{-5}}\).
\end{problem}

\begin{solution}
\(
\frac{(-27)^3\sqrt{\left(\frac{1}{16}\right)^{-\frac{1}{2}}}}{(-9)^0\left(\frac{3}{2}\right)^7\left(\frac{1}{2}\right)^{-5}}
=\frac{(-3)^9\sqrt{4}}{3^7\cdot\left(\frac{1}{2}\right)^{7-5}}
=-3^2\cdot2^3
=-72
\)
\end{solution}

\begin{problem}
已知 \(\lg2=0.3010\)，\(f\left(x\right)=\log_{\sqrt{10}}x\)，求 \(f\left(5\right)\).
\end{problem}

\begin{solution}
由换底公式知：
\begin{align*}
f\left(5\right)&=\log_{\sqrt{10}}5
=\frac{\lg5}{\lg\sqrt{10}}
=\frac{\lg\frac{10}{2}}{\frac{1}{2}\lg10}\\
&=2\left(\lg10-\lg2\right)
=2\left(1-0.3010\right)
=1.398
\end{align*}
所以 \(f\left(5\right)=1.398\).
\end{solution}

\begin{problem}
求函数 \(y=\frac{\sqrt{1-\frac{2}{3}x}}{\sqrt[3]{3x-1}}\) 的定义域.
\end{problem}

\begin{solution}
当 \(1-\frac{2}{3}x\ge0\) 且 \(3x-1\ne0\) 时，给出的函数才有定义. 因此 \(y=\frac{\sqrt{1-\frac{2}{3}x}}{\sqrt[3]{3x-1}}\) 的定义域为 \(x<\frac{1}{3}\) 及 \(\frac{1}{3}<x\le\frac{3}{2}\).
\end{solution}

\begin{problem}
已知 \(\sin\theta=0.6\)，求 \(\left(\sin\frac{\theta}{2}-\cos\frac{\theta}{2}\right)^2\) 的值.
\end{problem}

\begin{solution}
\[
\left(\sin\frac{\theta}{2}-\cos\frac{\theta}{2}\right)^2=\sin^2\frac{\theta}{2}+\cos^2\frac{\theta}{2}-2\sin\frac{\theta}{2}\cos\frac{\theta}{2}=1-\sin\theta=1-0.6=0.4
\]
\end{solution}

\begin{problem}
已知：梯形 \(ABCD\)，\(AD\parallel BC\)，对角线 \(AC\) 和 \(BD\) 相交于 \(E\)，过 \(E\) 作平行于底的直线，交 \(AB\) 于 \(M\)，交 \(CD\) 于 \(N\)，求证：\(ME=EN\).
\end{problem}

\begin{solution}
证：因为 \(MN\parallel BC\parallel AD\)，因此 \(\frac{AM}{MB}=\frac{DN}{NC}\)，由合比定理有 \(\frac{AM+MB}{MB}=\frac{DN+NC}{NC}\)，即 \(\frac{AB}{MB}=\frac{DC}{NC}\).

\bitmapfigure[max width=4.2cm]{img/1977/inner_mongolia_liberal/q2_solution_fig1.png}

因为 \(\triangle BEM\sim\triangle BDA\)，\(\triangle CNE\sim\triangle CDA\)，所以 \(\frac{AD}{ME}=\frac{AB}{MB}\)，\(\frac{AD}{EN}=\frac{DC}{NC}\). 但 \(\frac{AB}{MB}=\frac{DC}{NC}\)，则 \(\frac{AD}{ME}=\frac{AD}{EN}\)，因此 \(ME=EN\).
\end{solution}

\begin{problem}
圆台上底直径为 \(12\mathrm{~cm}\)，下底直径为 \(24\mathrm{~cm}\)，高为 \(8\mathrm{~cm}\)，求圆台的体积和全面积. （\(\pi=3.14\)，结果精确到 \(0.1\mathrm{~cm}^3\) 或 \(0.1\mathrm{~cm}^2\)）
\end{problem}

\begin{solution}
圆台的上底半径 \(R_1=\frac{12}{2}\mathrm{~cm}=6\mathrm{~cm}\)，下底半径 \(R_2=\frac{24}{2}\mathrm{~cm}=12\mathrm{~cm}\)，高 \(H=8\mathrm{~cm}\)，母线长 \(L=\sqrt{H^2+\left(R_2-R_1\right)^2}=\sqrt{64+36}=10\mathrm{~cm}\). 于是圆台的体积
\[
V=\frac{1}{3}\pi H\left(R_1^2+R_2^2+R_1R_2\right)=\frac{8\pi}{3}\cdot\left(36+144+72\right)\approx 2110\mathrm{~cm}^3
\]
圆台的全面积
\[
S=\pi\left(R_1^2+R_2^2+L\left(R_1+R_2\right)\right)=\pi\left(36+144+180\right)\approx 1130\mathrm{~cm}^2
\]
\end{solution}

\begin{problem}
已知：直线 \(3x+y=5\) 及 \(2x-3y+4=0\) 交于 \(A\) 点.
求：
\begin{enumerate}
\item 过 \(A\) 点且与 \(y\) 轴垂直的直线方程；
\item 过 \(A\) 点且与直线 \(2x+y+1=0\) 平行的直线方程；
\item 过 \(A\) 点且与直线 \(2x+y+1=0\) 垂直的直线方程.
\end{enumerate}

\end{problem}

\begin{solution}
（1）解方程组
\[
\begin{cases}
3x+y=5,\\
2x-3y+4=0
\end{cases}
\]
可得 \(A\) 点的坐标为 \(\left(1,2\right)\)，过 \(A\) 点且与 \(y\) 轴垂直的直线方程为 \(y=2\).

（2）直线 \(2x+y+1=0\) 的斜率为 \(-2\)，因此过 \(A\) 点与上直线平行的直线的斜率也为 \(-2\)，故其方程为 \(y-2=-2\left(x-1\right)\)，即 \(2x+y-4=0\).

（3）过 \(A\) 点与 \(2x+y+1=0\) 垂直的直线的斜率与已知直线的斜率互为负倒数，故所求的直线方程为 \(y-2=\frac{1}{2}\left(x-1\right)\)，即 \(x-2y+3=0\).
\end{solution}

\begin{problem}
红旗拖拉机厂今年元月份生产出一批甲乙两种型号的拖拉机，其中生产乙型 \(16\text{ 台}\). 从二月份起，甲型每月增产 \(10\text{ 台}\)，而乙型按每月相同增长率逐月递增，又知二月份甲，乙两型拖拉机产量之比是 \(3:2\)，三月份甲，乙两型产量之和是 \(65\text{ 台}\)，求乙型拖拉机每月的增长率及甲型拖拉机元月份的产量.

\end{problem}

\begin{solution}
设甲型拖拉机元月份产量为 \(x\)，乙型拖拉机每月增长率为 \(y\). 根据已给条件列方程如下：
\[
\begin{cases}
\frac{x+10}{16\left(1+y\right)}=\frac{3}{2},\\
16\left(1+y\right)^2+\left(x+20\right)=65.
\end{cases}
\]
由第一个方程得 \(x=\frac{3}{2}\cdot16\left(1+y\right)-10=24\left(1+y\right)-10\)，代入第二个方程得
\(16\left(1+y\right)^2+24\left(1+y\right)-10+20=65\)，即 \(16\left(1+y\right)^2+24\left(1+y\right)-55=0\)，分解因式
\(\left[4\left(1+y\right)+11\right]\left[4\left(1+y\right)-5\right]=0\)，
\(4\left(1+y\right)+11=0\)，\(y=-\frac{11}{4}-1\)，不合题意，舍去；\(4\left(1+y\right)-5=0\)，\(y=\frac{5}{4}-1=0.25=25\%\). 又 \(x=24\left(1+25\%\right)-10=20\).

故甲型拖拉机元月份产 \(20\text{ 台}\)，乙型拖拉机每月增长率为 \(25\%\).
\end{solution}
