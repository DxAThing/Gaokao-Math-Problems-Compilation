\chapter{1985年上海卷（文）}

\section{填空题}

\begin{problem}

不等式 \(\left|x+2\right|<1\) 的解集是\fillinblank{}.

\begin{answer}
\(\left(-3,-1\right)\)
\end{answer}

\begin{solution}
由绝对值不等式的意义，\(\left|x+2\right|<1\) 等价于 \(-1<x+2<1\). 不等式两边同时减去 \(2\)，得 \(-3<x<-1\)，所以解集为 \(\left(-3,-1\right)\).
\end{solution}
\end{problem}

\begin{problem}

函数 \(y=\sqrt[8]{x}-1\) 的反函数是\fillinblank{}.

\begin{answer}
\(y=(x+1)^8\)，\(x\geqslant-1\)
\end{answer}

\begin{solution}
原函数的定义域为 \(x\geqslant0\)，且八次方根函数在此定义域上严格递增，所以反函数存在. 由 \(y=\sqrt[8]{x}-1\) 得 \(y+1=\sqrt[8]{x}\)，从而 \(x=(y+1)^8\). 交换 \(x\)，\(y\) 的位置，得到反函数 \(y=(x+1)^8\). 原函数的值域为 \(y\geqslant-1\)，因此反函数的定义域为 \(x\geqslant-1\).
\end{solution}
\end{problem}

\begin{problem}

函数 \(y=\tan 2x\) 的最小正周期是\fillinblank{}.

\begin{answer}
\(\dfrac{\pi}{2}\)
\end{answer}

\begin{solution}
正切函数满足 \(\tan u=\tan v\) 当且仅当 \(u-v=k\pi\)，其中 \(k\in\Z\). 若 \(T\) 是 \(y=\tan 2x\) 的周期，则对定义域内任意 \(x\)，有 \(\tan(2x+2T)=\tan 2x\)，从而 \(2T=k\pi\). 取最小正整数 \(k=1\)，得到 \(T=\dfrac{\pi}{2}\)；反之，\(\tan\left(2x+\pi\right)=\tan 2x\)，所以最小正周期确为 \(\dfrac{\pi}{2}\).
\end{solution}
\end{problem}

\begin{problem}

设等差数列 \(\{a_n\}\) 中，\(a_1=2\)，\(a_2=5\)，则 \(a_{10}=\)\fillinblank{}.

\begin{answer}
\(29\)
\end{answer}

\begin{solution}
等差数列的公差为 \(d=a_2-a_1=5-2=3\). 由通项公式 \(a_n=a_1+(n-1)d\)，得 \(a_{10}=2+9\times3=29\).
\end{solution}
\end{problem}

\begin{problem}

方程 \(9^x=3^{x+1}\) 的解是\fillinblank{}.

\begin{answer}
\(x=1\)
\end{answer}

\begin{solution}
因为 \(9=3^2\)，所以原方程化为 \(3^{2x}=3^{x+1}\). 底数 \(3>0\) 且 \(3\neq1\)，指数函数具有一一对应性，故 \(2x=x+1\)，解得 \(x=1\). 将 \(x=1\) 代回原方程，左右两边均为 \(9\)，所以这是原方程的解.
\end{solution}
\end{problem}

\begin{problem}

用不等号 \(>\) 或 \(<\) 连接：\(\left(\frac{1}{2}\right)^{\frac{1}{7}}\)\fillinblank{}\(\left(\frac{1}{2}\right)^{\frac{1}{6}}\).

\begin{answer}
\(>\)
\end{answer}

\begin{solution}
因为 \(0<\dfrac{1}{2}<1\)，所以指数函数 \(y=\left(\dfrac{1}{2}\right)^t\) 在 \(\R\) 上严格递减. 又因为 \(\dfrac{1}{7}<\dfrac{1}{6}\)，递减性给出 \(\left(\dfrac{1}{2}\right)^{\frac{1}{7}}>\left(\dfrac{1}{2}\right)^{\frac{1}{6}}\)，故应填 \(>\).
\end{solution}
\end{problem}

\begin{problem}

如果一个圆的圆心在点 \((2,4)\)，并且经过点 \((0,3)\)，则这个圆的方程是\fillinblank{}.

\begin{answer}
\(\left(x-2\right)^2+\left(y-4\right)^2=5\)
\end{answer}

\begin{solution}
圆心为 \(C(2,4)\). 由圆经过点 \((0,3)\)，半径的平方为 \(r^2=(0-2)^2+(3-4)^2=4+1=5\). 因此圆的标准方程为 \(\left(x-2\right)^2+\left(y-4\right)^2=5\). 将 \((0,3)\) 代入可得左边为 \(4+1=5\)，与方程一致.
\end{solution}
\end{problem}

\begin{problem}

直线 \(y=-x+3\) 的倾斜角的大小是\fillinblank{}.

\begin{answer}
\(\dfrac{3\pi}{4}\)
\end{answer}

\begin{solution}
直线的斜率为 \(k=-1\). 设倾斜角为 \(\theta\)，按定义有 \(0\leqslant\theta<\pi\) 且 \(\tan\theta=k=-1\). 在这个范围内，\(\tan\theta=-1\) 的角为第二象限角 \(\theta=\dfrac{3\pi}{4}\)，所以倾斜角为 \(\dfrac{3\pi}{4}\).
\end{solution}
\end{problem}

\begin{problem}

设 \(\lg 2=a\)，则 \(\log_2 25=\)\fillinblank{}.

\begin{answer}
\(\dfrac{2(1-a)}{a}\)
\end{answer}

\begin{solution}
由换底公式，\(\log_2 25=\dfrac{\lg25}{\lg2}=\dfrac{2\lg5}{a}\). 又因为 \(\lg5=\lg\dfrac{10}{2}=1-\lg2=1-a\)，所以 \(\log_2 25=\dfrac{2(1-a)}{a}\). 由于 \(a=\lg2>0\)，分母不为零，变形合法.
\end{solution}
\end{problem}

\begin{problem}

已知 \(O\) 为直角坐标系的原点，\(A\) 点的坐标为 \((1,3)\)，\(B\) 点的坐标为 \((0,3)\)，若把 \(\triangle OAB\) 绕 \(y\) 轴旋转一周，则所得旋转体的体积是\fillinblank{}.

\begin{answer}
\(\pi\)
\end{answer}

\begin{solution}
线段 \(OA\) 经过 \((0,0)\) 和 \((1,3)\)，方程为 \(x=\dfrac{y}{3}\). 三角形内部在 \(0\leqslant y\leqslant3\) 时满足 \(0\leqslant x\leqslant\dfrac{y}{3}\). 因而在高度 \(y\) 处的截面绕 \(y\) 轴旋转后是半径为 \(\dfrac{y}{3}\) 的圆盘.

所以旋转体的体积为
\[
V=\int_0^3\pi\left(\frac{y}{3}\right)^2\,\mathrm{d}y=\frac{\pi}{9}\left[\frac{y^3}{3}\right]_0^3=\pi
\]
\end{solution}
\end{problem}

\section{单选题}

\begin{problem}

函数 \(y=-\sin 2x+3\) 的最大值是（\quad）

\choices
    {\(2\)}
    {\(3\)}
    {\(4\)}
    {\(5\)}

\begin{answer}
C
\end{answer}

\begin{solution}
对任意实数 \(x\)，有 \(-1\leqslant\sin 2x\leqslant1\). 两边同乘以 \(-1\) 后不等号反向，得 \(-1\leqslant-\sin 2x\leqslant1\)，再加上 \(3\)，得到 \(2\leqslant-\sin 2x+3\leqslant4\). 当 \(\sin 2x=-1\) 时上界可以取到，例如取 \(2x=\dfrac{3\pi}{2}\)，所以最大值为 \(4\)，选 C.
\end{solution}
\end{problem}

\begin{problem}

设复数 \(z_1=1+\i\)，\(z_2=1-3\i\)，则复数 \(z=z_1\cdot z_2\) 在复平面内所表示的点位于（\quad）

\choices
    {第一象限}
    {第二象限}
    {第三象限}
    {第四象限}

\begin{answer}
D
\end{answer}

\begin{solution}
利用 \(\i^2=-1\)，得
\[
z=(1+\i)(1-3\i)=1-3\i+\i-3\i^2=4-2\i
\]
复数 \(4-2\i\) 在复平面内对应点 \((4,-2)\). 该点的横坐标为正、纵坐标为负，因此位于第四象限，选 D.
\end{solution}
\end{problem}

\begin{problem}

已知函数 \(f(x)=\lg x^2\) 的定义域为 \(F\)，函数 \(g(x)=2\lg x\) 的定义域为 \(G\)，那么（\quad）

\choices
    {\(F \cap G = \varnothing\)}
    {\(F = G\)}
    {\(F \subset G\)}
    {\(G \subset F\)}

\begin{answer}
D
\end{answer}

\begin{solution}
对 \(f(x)=\lg x^2\)，真数必须满足 \(x^2>0\)，所以 \(x\neq0\)，从而 \(F=\{x\mid x\neq0\}\). 对 \(g(x)=2\lg x\)，真数必须满足 \(x>0\)，所以 \(G=(0,+\infty)\). 每个正数都属于 \(F\)，且 \(F\) 还包含负数，因此 \(G\subset F\)，选 D.
\end{solution}
\end{problem}

\begin{problem}

若一个四棱锥的底面是凸四边形，它的顶点到底面各边的距离都相等，则这个棱锥的底面四边形（\quad）

\choices
    {必为正方形}
    {必有内切圆}
    {必有外接圆}
    {必既有内切圆又有外接圆}

\begin{answer}
B
\end{answer}

\begin{solution}
设棱锥顶点为 \(P\)，底面所在平面为 \(\alpha\)，点 \(P\) 在 \(\alpha\) 上的射影为 \(H\). 设四条底边所在直线分别为 \(l_1\)，\(l_2\)，\(l_3\)，\(l_4\)，从 \(H\) 向 \(l_i\) 作垂线，垂足为 \(D_i\). 题设中到各边的距离均由相应的垂足实现，所以各个 \(D_i\) 都在相应的边上. 因为 \(PH\perp\alpha\)，且 \(HD_i\perp l_i\)，所以 \(PD_i\perp l_i\)，并有 \(PD_i^2=PH^2+HD_i^2\). 题设给出 \(PD_1=PD_2=PD_3=PD_4\)，故 \(HD_1=HD_2=HD_3=HD_4=r\). 对于凸四边形，若公共等距点 \(H\) 在四边形外部，则从 \(H\) 向某条边及其相邻边所在直线作垂线时，至少有一个垂足落在对应边的延长线上，这与各个 \(D_i\) 都在边上矛盾. 故 \(H\) 位于四边形内部. 以 \(H\) 为圆心、\(r\) 为半径作圆，它在各边上的切点分别为 \(D_i\)，因而与四条边分别相切. 所以底面四边形必有内切圆，选 B.
\end{solution}
\end{problem}

\begin{problem}

\(a>0\) 且 \(b>0\)，是 \(ab>0\) 的（\quad）

\choices
    {充分但不必要的条件}
    {必要但不充分的条件}
    {充分而且必要的条件}
    {既不充分又不必要的条件}

\begin{answer}
A
\end{answer}

\begin{solution}
若 \(a>0\) 且 \(b>0\)，则正数与正数的乘积为正数，即 \(ab>0\)，所以该条件是充分条件. 但 \(ab>0\) 也可能由 \(a<0\) 且 \(b<0\) 得到，例如 \(a=b=-1\) 时 \(ab=1>0\)，而 \(a>0\) 且 \(b>0\) 不成立. 因此该条件不是必要条件，故选“充分但不必要的条件”，即 A.
\end{solution}
\end{problem}

\begin{problem}

在平面直角坐标系内，画出下列方程的曲线（只要求画出图形）：

\begin{enumerate}
\item \(\frac{x}{3} - \frac{y}{2} = -1\)；
\item \(\frac{x^2}{9} - \frac{y^2}{4} = -1\).
\end{enumerate}

\begin{solution}
\begin{enumerate}
\item 将方程乘以 \(6\)，得 \(2x-3y=-6\)，即 \(y=\dfrac{2}{3}x+2\). 令 \(y=0\)，得横轴截距 \((-3,0)\)；令 \(x=0\)，得纵轴截距 \((0,2)\). 过这两点作直线，即为所求曲线.
\item 移项得 \(\dfrac{y^2}{4}-\dfrac{x^2}{9}=1\). 这是中心在原点、实轴在 \(y\) 轴上的双曲线，令 \(x=0\) 得顶点为 \((0,2)\) 和 \((0,-2)\). 由标准双曲线方程可知其渐近线为 \(y=\dfrac{2}{3}x\) 和 \(y=-\dfrac{2}{3}x\)；也可由 \(y=\pm\dfrac{2}{3}\sqrt{x^2+9}\) 看出这一点. 因而图形有上、下两支，分别位于 \(y\geqslant2\) 和 \(y\leqslant-2\) 的区域，并以这两条直线为渐近线.
\end{enumerate}
\end{solution}
\end{problem}

\begin{problem}

\begin{enumerate}
\item 求 \(\left(2+x^2\right)^8\) 的展开式中 \(x^{10}\) 的系数.

\item 从六个数字 \(1\)，\(2\)，\(3\)，\(5\)，\(7\)，\(9\) 中任取四个不同的数字，有多少种取法？由这六个数字可以组成多少个没有重复数字的四位偶数？

\item 在一列球中，第 \(1\) 个球的半径为 \(1\)，第 \(2\) 个球的直径是第 \(1\) 个球的半径，以后每一个球的直径都是前一个球的半径，这样无限继续下去，求所有这些球的表面积的和.
\end{enumerate}

\begin{solution}
\begin{enumerate}
\item 二项式展开的通项中，从 \(8\) 个因子中选出 \(k\) 个取 \(x^2\)，其余因子取 \(2\)，所得项为 \(\mathrm C_8^k2^{8-k}x^{2k}\). 要得到 \(x^{10}\)，必须满足 \(2k=10\)，所以 \(k=5\). 因此所求系数为 \(\mathrm C_8^5 2^{8-5}=56\times8=448\).
\item 任取四个不同数字时只需从六个数字中选出四个，取法数为 \(\mathrm C_6^4=\dfrac{6\times5\times4\times3}{4\times3\times2\times1}=15\). 组成四位偶数时，个位必须是偶数；给出的六个数字中只有 \(2\) 是偶数，所以个位固定为 \(2\). 千位、百位、十位从其余五个数字中选出三个并按顺序排列，方法数为 \(\mathrm A_5^3=5\times4\times3=60\). 因此两问的答案分别为 \(15\) 种和 \(60\) 个.
\item 记第 \(n\) 个球的半径为 \(r_n\). 由题意 \(r_1=1\)，且对 \(n\geqslant2\)，第 \(n\) 个球的直径为前一个球的半径，所以 \(2r_n=r_{n-1}\)，即 \(r_n=\dfrac{1}{2^{n-1}}\). 第 \(n\) 个球的表面积为 \(S_n=4\pi r_n^2=4\pi\left(\dfrac14\right)^{n-1}\). 因此所有球的表面积构成首项为 \(4\pi\)、公比为 \(\dfrac14\) 的无穷等比数列，且公比的绝对值小于 \(1\)，故其和为 \(S=\dfrac{4\pi}{1-\frac14}=\dfrac{16\pi}{3}\).
\end{enumerate}
\end{solution}
\end{problem}

\begin{problem}

如图，直线 \(a\parallel b\)，点 \(P\) 在 \(a\)，\(b\) 所确定的平面外，\(PA\perp a\)，且交 \(a\) 于 \(A\) 点；\(AB\perp b\)，且交 \(b\) 于 \(B\) 点. 求证：\(PB\perp b\).

\begin{solution}
设 \(a\)，\(b\) 所确定的平面为 \(\alpha\)，过 \(P\)，\(A\)，\(B\) 的平面为 \(\beta\). 因为 \(a\parallel b\)，且 \(AB\perp b\)，所以 \(AB\perp a\). 题设又给出 \(PA\perp a\). 直线 \(PA\) 与 \(AB\) 是平面 \(\beta\) 内相交的两条直线，且直线 \(a\) 同时垂直于它们，因此 \(a\perp\beta\).

由 \(a\parallel b\)，可得 \(b\perp\beta\). 直线 \(PB\) 在平面 \(\beta\) 内，并且与 \(b\) 相交于 \(B\)，所以平面 \(\beta\) 内经过交点 \(B\) 的直线 \(PB\) 垂直于 \(b\). 即 \(PB\perp b\).
\end{solution}
\end{problem}

\begin{problem}

\begin{enumerate}
\item 证明万能公式：\(\sin\alpha=\frac{2\tan\frac{\alpha}{2}}{1+\tan^2\frac{\alpha}{2}}\).

\item 已知：\(\cos\alpha=-\frac{1}{3}\)，\(\pi<\alpha<\frac{3\pi}{2}\)，求：\(\cos\left(\frac{\alpha}{2}+\frac{5\pi}{6}\right)\).
\end{enumerate}

\begin{solution}
\begin{enumerate}
\item 当 \(\tan\dfrac{\alpha}{2}\) 有意义时，即 \(\alpha\neq(2m+1)\pi\)（\(m\in\Z\)），有 \(\cos\dfrac{\alpha}{2}\neq0\). 由正弦二倍角公式并将分子、分母同除以 \(\cos^2\dfrac{\alpha}{2}\)，得
\[
\sin\alpha=\frac{2\sin\frac{\alpha}{2}\cos\frac{\alpha}{2}}{\cos^2\frac{\alpha}{2}}=\frac{2\tan\frac{\alpha}{2}}{\sec^2\frac{\alpha}{2}}=\frac{2\tan\frac{\alpha}{2}}{1+\tan^2\frac{\alpha}{2}}
\]
所以公式得证.
\item 由 \(\pi<\alpha<\dfrac{3\pi}{2}\)，两边同除以 \(2\)，得 \(\dfrac{\pi}{2}<\dfrac{\alpha}{2}<\dfrac{3\pi}{4}\)，所以 \(\dfrac{\alpha}{2}\) 位于第二象限，正弦为正、余弦为负. 由半角公式，
\(\cos\dfrac{\alpha}{2}=-\sqrt{\dfrac{1+\cos\alpha}{2}}=-\sqrt{\dfrac{1-\frac13}{2}}=-\dfrac{1}{\sqrt3}\)，
\(\sin\dfrac{\alpha}{2}=\sqrt{\dfrac{1-\cos\alpha}{2}}=\sqrt{\dfrac{1+\frac13}{2}}=\sqrt{\dfrac23}\).
又因为 \(\cos\dfrac{5\pi}{6}=-\dfrac{\sqrt3}{2}\)，\(\sin\dfrac{5\pi}{6}=\dfrac12\)，由两角和公式得
\[
\cos\left(\frac{\alpha}{2}+\frac{5\pi}{6}\right)=\left(-\frac{1}{\sqrt3}\right)\left(-\frac{\sqrt3}{2}\right)-\sqrt{\frac23}\cdot\frac12=\frac12-\frac{\sqrt6}{6}=\frac{3-\sqrt6}{6}
\]
\end{enumerate}
\end{solution}
\end{problem}

\begin{problem}

设 \(\triangle ABC\) 的两个内角 \(A\)，\(B\) 所对的边分别为 \(a\)，\(b\). 若复数 \(z_1\) 的模为 \(a\)，幅角为 \(B\)；复数 \(z_2\) 的模为 \(b\)，幅角为 \(-A\). 求证：\(z_1+z_2\) 为实数.

\begin{solution}
由复数的三角形式，\(z_1=a\left(\cos B+\i\sin B\right)\)，\(z_2=b\left(\cos A-\i\sin A\right)\). 因此 \(z_1+z_2=\left(a\cos B+b\cos A\right)+\i\left(a\sin B-b\sin A\right)\).

在三角形 \(ABC\) 中，由正弦定理，存在同一个正数 \(k\)，使得 \(a=k\sin A\)，\(b=k\sin B\). 所以虚部中的系数为 \(a\sin B-b\sin A=k\sin A\sin B-k\sin B\sin A=0\). 于是 \(z_1+z_2=a\cos B+b\cos A\) 没有虚部，是实数.
\end{solution}
\end{problem}

\begin{problem}

如图，由抛物线 \(y=x^2+2\) 上的点 \(M(x_0,y_0)\) 向直线 \(y=\frac{1}{2}x\) 作垂线，垂足为 \(N\)，延长 \(MN\) 至 \(P\) 点，使点 \(N\) 分线段 \(MP\) 成定比 \(\lambda=\frac{MN}{NP}=4\).

\begin{enumerate}
\item 用 \(x_0\)，\(y_0\) 表示点 \(N\) 的坐标 \((x_1, y_1)\)；
\item 用 \(x_0\)，\(y_0\) 和 \(x_1\)，\(y_1\) 表示点 \(P\) 的坐标 \((x, y)\)；
\item 求：当点 \(M\) 沿抛物线移动时，点 \(P\) 的轨迹方程，并道明轨迹是什么图形.
\end{enumerate}

\begin{solution}
\begin{enumerate}
\item 直线 \(y=\dfrac12x\) 上的点可设为 \(N=(2t,t)\). 该直线的方向向量为 \((2,1)\)，而 \(MN\) 垂直于该直线，所以 \(\bigl(x_0-2t,y_0-t\bigr)\cdot(2,1)=0\). 展开得 \(2x_0+y_0-5t=0\)，从而 \(t=\dfrac{2x_0+y_0}{5}\). 因此 \(x_1=2t=\dfrac{4x_0+2y_0}{5}\)，\(y_1=t=\dfrac{2x_0+y_0}{5}\).
\item 因为 \(M\)，\(N\)，\(P\) 依次在同一直线上，且 \(MN:NP=4:1\)，所以内分点公式给出 \(N=\dfrac{M+4P}{5}\). 解得 \(P=\dfrac{5N-M}{4}\)，即 \(x=\dfrac{5x_1-x_0}{4}\)，\(y=\dfrac{5y_1-y_0}{4}\).
\item 将第（1）问的结果代入第（2）问，得 \(x=\dfrac{3x_0+2y_0}{4}\)，\(y=\dfrac{x_0}{2}\). 又因为 \(M\) 在抛物线上，所以 \(y_0=x_0^2+2\). 于是 \(x=\dfrac{2x_0^2+3x_0+4}{4}\)，并由 \(x_0=2y\) 消去参数，得到
\[
x=2y^2+\frac32y+1=2\left(y+\frac38\right)^2+\frac{23}{32}
\]
由于 \(x_0\) 可以取任意实数，\(y=\dfrac{x_0}{2}\) 也可以取任意实数，所以轨迹是完整的抛物线 \(x=2\left(y+\dfrac38\right)^2+\dfrac{23}{32}\). 它的顶点为 \(\left(\dfrac{23}{32},-\dfrac38\right)\)，对称轴为 \(y=-\dfrac38\)，开口方向为 \(x\) 轴正向.
\end{enumerate}
\end{solution}
\end{problem}
