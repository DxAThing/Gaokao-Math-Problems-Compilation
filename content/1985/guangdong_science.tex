\chapter{1985年广东卷（理）}

\section{填空题}

\begin{problem}

函数 \(y = \sqrt{\arctan(1 - x)} \) 的定义域是\fillinblank{}.
\end{problem}

\begin{answer}
\(x \leqslant 1\).
\end{answer}

\begin{solution}
反正切函数 \(\arctan t\) 对一切实数 \(t\) 有定义，且与 \(t\) 同号.因此根式有意义的充要条件是 \(\arctan(1-x)\geqslant0\)，等价于 \(1-x\geqslant0\).所以定义域为 \(x\leqslant1\).
\end{solution}

\begin{problem}

如果 \(f\left(\frac{1}{x}\right)=\frac{x}{1-x^{2}} \)，则 \(f(x)= \)\fillinblank{}.
\end{problem}

\begin{answer}
\(\displaystyle \frac{x}{x^{2}-1}\).
\end{answer}

\begin{solution}
原式中的自变量必须满足 \(x\ne0\)，\(\pm1\).令原式中的 \(x=\frac1t\)，则 \(t\ne0\)，\(\pm1\)，并且 \(f(t)=\dfrac{1/t}{1-1/t^{2}}=\dfrac{t}{t^{2}-1}\).因此，在题设关系确定的范围 \(x\ne0\)，\(\pm1\) 内，\(f(x)=\dfrac{x}{x^{2}-1}\).
\end{solution}

\begin{problem}

如果一个圆锥底面的半径增大到原来的三倍，但要保持它的体积不变，则其高必须缩小到原来的\fillinblank{}.
\end{problem}

\begin{answer}
\(\displaystyle \frac{1}{9}\).
\end{answer}

\begin{solution}
设圆锥原来的底面半径和高分别为 \(r\)、\(h\)，则体积为 \(V=\frac13\pi r^{2}h\).设变化后的高为 \(h'\)，因半径变为 \(3r\) 且体积不变，有 \(\frac13\pi(3r)^{2}h'=\frac13\pi r^{2}h\)，从而 \(9h'=h\)，即 \(h'=\frac19h\).所以高缩小到原来的 \(\displaystyle\frac19\).
\end{solution}

\begin{problem}

在直角坐标系中，通过点 \((0,0) \) \((1,0) \) 和 \((0,2) \) 的圆的方程是\fillinblank{}.
\end{problem}

\begin{answer}
\(x^{2}+y^{2}-x-2y=0\).
\end{answer}

\begin{solution}
设圆的方程为 \(x^{2}+y^{2}+Dx+Ey+F=0\).代入 \((0,0)\) 得 \(F=0\)；代入 \((1,0)\) 得 \(1+D=0\)，所以 \(D=-1\)；代入 \((0,2)\) 得 \(4+2E=0\)，所以 \(E=-2\).故圆的方程为 \(x^{2}+y^{2}-x-2y=0\).
\end{solution}

\begin{problem}

\(\tan\frac{\pi}{5}+\tan\frac{2\pi}{5}+\tan\frac{3\pi}{5}+\tan\frac{4\pi}{5} \) 的值等于\fillinblank{}.
\end{problem}

\begin{answer}
\(0\).
\end{answer}

\begin{solution}
利用 \(\tan(\pi-\theta)=-\tan\theta\)，有 \(\tan\frac{3\pi}{5}=-\tan\frac{2\pi}{5}\)，\(\tan\frac{4\pi}{5}=-\tan\frac\pi5\).因此原式为 \(\tan\frac\pi5+\tan\frac{2\pi}5-\tan\frac{2\pi}5-\tan\frac\pi5=0\).
\end{solution}

\begin{problem}

\(\displaystyle\lim_{n\to\infty}\frac{1+4+7+\cdots+(3n-2)}{n(n+1)} \) 的值等于\fillinblank{}.
\end{problem}

\begin{answer}
\(\displaystyle \frac{3}{2}\).
\end{answer}

\begin{solution}
分子是首项为 \(1\)、末项为 \(3n-2\)、项数为 \(n\) 的等差数列之和，因此 \(1+4+7+\cdots+(3n-2)=\frac{n[1+(3n-2)]}{2}=\frac{n(3n-1)}2\).所以原极限为 \(\displaystyle\lim_{n\to\infty}\frac{3n-1}{2(n+1)}=\frac32\).
\end{solution}

\begin{problem}

函数 \(y = \sin^2(2x) \) 的最小正周期是\fillinblank{}.
\end{problem}

\begin{answer}
\(\displaystyle \frac{\pi}{2}\).
\end{answer}

\begin{solution}
函数 \(\sin^{2}u\) 的最小正周期为 \(\pi\).令 \(u=2x\)，则需满足 \(2T=\pi\)，所以函数 \(y=\sin^{2}(2x)\) 的最小正周期为 \(\displaystyle T=\frac\pi2\).
\end{solution}

\begin{problem}

已知函数 \(f(x) = ax^5 + \arctan x \)，其中 \(a \) 为常数，且 \(f(2) = 10 \)，则 \(f(-2) \) 的值是\fillinblank{}.
\end{problem}

\begin{answer}
\(-10\).
\end{answer}

\begin{solution}
因为 \(x^{5}\) 和 \(\arctan x\) 都是奇函数，所以 \(f(x)=ax^{5}+\arctan x\) 是奇函数.由 \(f(2)=10\)，得 \(f(-2)=-f(2)=-10\).
\end{solution}

\begin{problem}

已知圆的方程为 \(x^2 + (y - 2)^2 = 9 \)，用平行于 \(x \) 轴的直径把圆分为上下半圆，则以上半圆弧（包括端点）为图象的函数表达式是\fillinblank{}.
\end{problem}

\begin{answer}
\(y=2+\sqrt{9-x^{2}}\)， \( -3\leqslant x\leqslant 3\).
\end{answer}

\begin{solution}
由圆方程得 \((y-2)^{2}=9-x^{2}\)，所以 \(y=2\pm\sqrt{9-x^{2}}\)，且必须有 \(9-x^{2}\geqslant0\)，即 \(-3\leqslant x\leqslant3\).上半圆弧满足 \(y\geqslant2\)，故取正号，函数为 \(y=2+\sqrt{9-x^{2}}\)，定义域为 \([-3,3]\).
\end{solution}

\begin{problem}

极坐标方程 \(\rho=\frac{10}{5-3\cos\theta} \). 所表示的曲线是（\quad）

\choicesfive
    {椭圆}
    {圆}
    {双曲线}
    {抛物线}
    {双曲线的一支}
\end{problem}

\begin{answer}
A.
\end{answer}

\begin{solution}
极坐标二次曲线的标准形式为 \(\rho=\dfrac{l}{1-e\cos\theta}\).将题式化为 \(\rho=\dfrac{2}{1-\frac35\cos\theta}\)，可得离心率 \(e=\frac35<1\).离心率小于 \(1\) 的圆锥曲线是椭圆，故选 A.
\end{solution}

\begin{problem}

以 \(A(1,3) \)， \(B(-5,1) \) 为端点的线段的垂直平分线的方程是（\quad）

\choicesfive
    {\(3x - y + 8 = 0 \)}
    {\(3x + y + 4 = 0 \)}
    {\(2x - y - 6 = 0\)}
    {\(2x + y + 2 = 0 \)}
    {\(3x + y + 8 = 0 \)}
\end{problem}

\begin{answer}
B.
\end{answer}

\begin{solution}
线段 \(AB\) 的中点为 \(M(-2,2)\)，其斜率为 \(\dfrac{1-3}{-5-1}=\frac13\)，所以垂直平分线的斜率为 \(-3\).由点斜式得 \(y-2=-3(x+2)\)，整理为 \(3x+y+4=0\)，故选 B.
\end{solution}

\begin{problem}

如果双曲线 \(\frac{x^{2}}{a^{2}} - \frac{y^{2}}{b^{2}} = 1 \) 的两条渐近线互相垂直，则双曲线的离心率是（\quad）

\choicesfive
    {\(\sqrt{2} \)}
    {\(2\)}
    {\(\sqrt{3} \)}
    {\(2\sqrt{2} \)}
    {\(\frac{3}{2} \)}
\end{problem}

\begin{answer}
A.
\end{answer}

\begin{solution}
双曲线的渐近线为 \(y=\pm\frac ba x\)，两条直线垂直时斜率乘积为 \(-1\)，即 \(-\frac{b^{2}}{a^{2}}=-1\)，所以 \(a=b\).双曲线的半焦距满足 \(c^{2}=a^{2}+b^{2}=2a^{2}\)，因此离心率 \(e=\frac ca=\sqrt2\)，故选 A.
\end{solution}

\begin{problem}

设集合 \(C = \{ \) 复数 \(\} \)， \(R = \{ \) 实数 \(\} \)， \(M = \{ \) 纯虚数 \(\} \)，其中 \(C\) 为全集，则（\quad）

\choicesfive
    {\(M \cup R = C \)}
    {\(\overline{M} \cup R = C \)}
    {\(R \cup \overline{R} = C \)}
    {\(M \cap R = \{0\} \)}
    {\(C \cap \overline{R} = M \)}
\end{problem}

\begin{answer}
C.
\end{answer}

\begin{solution}
在全集 \(C\) 中，\(\overline{R}\) 是 \(R\) 的补集，所以 \(R\cup\overline{R}=C\)，选项 C 正确.选项 A 不正确，例如 \(1+\i\) 既不是实数也不是纯虚数；选项 B 不正确，例如非零纯虚数 \(\i\) 不属于 \(\overline M\cup R\)；选项 D 不正确，因为纯虚数的实部为零且虚部不为零，故 \(M\cap R=\varnothing\)；选项 E 不正确，因为 \(C\cap\overline R=\overline R\) 包含 \(1+\i\)，不等于 \(M\).故选 C.
\end{solution}

\(y = x^{2}\)（\(x \leqslant 0\)）的反函数是（\quad）

如果 \(y = \log_{(a^2 - 1)} x \) 在 \((0, +\infty) \) 内是减函数，则 \(a \) 的取值范围是（\quad）

\begin{problem}

如果 \(n\) 是正偶数，则 \(\mathrm C_{n}^{0} + \mathrm C_{n}^{2} + \cdots + \mathrm C_{n}^{n-2} + \mathrm C_{n}^{n} = \)（\quad）

\choicesfive

    {\(2^{n-1} \)}
    {\(2^{n-2} \)}
    {\((n-1)2^{n-1} \)}
    {\((n-1)2^{n-2} \)}
\end{problem}

\begin{answer}
B.
\end{answer}

由二项式定理，\((1+1)^{n}=\sum_{k=0}^{n}\mathrm C_n^k=2^n\)，而 \((1-1)^n=\sum_{k=0}^{n}(-1)^k\mathrm C_n^k=0\).正偶数项与奇数项的和相等，所以偶数项之和为 \(2^{n-1}\)，故选 B.

\begin{problem}

已知 \(180^{\circ} < \alpha < 270^{\circ} \)，且 \(\sin(270^{\circ} + \alpha) = \frac{4}{5} \)，则 \(\tan\frac{\alpha}{2} = \)（\quad）

\end{problem}

\begin{answer}
D.
\end{answer}

\begin{problem}

方程 \(\sin 3x - \sin x = 0 \) 的解集是（\quad）

\choicesfive

\begin{answer}
C.

\begin{problem}

\(\arcsin\left(-\frac{1}{3}\right)\)，\(\arctan\left(-\sqrt{2}\right)\)，\(\arccos\left(-\frac{2}{3}\right)\) 的大小关系是（\quad）

\choicesfive
    {\(\arcsin\left(-\frac{1}{3}\right)<\arctan\left(-\sqrt{2}\right)<\arccos\left(-\frac{2}{3}\right) \)}
    {\(\arcsin\left(-\frac{1}{3}\right)<\arccos\left(-\frac{2}{3}\right)<\arctan\left(-\sqrt{2}\right) \)}
    {\(\arctan\left(-\sqrt{2}\right)<\arcsin\left(-\frac{1}{3}\right)<\arccos\left(-\frac{2}{3}\right) \)}
    {\(\arctan\left(-\sqrt{2}\right)<\arccos\left(-\frac{2}{3}\right)<\arcsin\left(-\frac{1}{3}\right) \)}
    {\(\arccos\left(-\frac{2}{3}\right)<\arcsin\left(-\frac{1}{3}\right)<\arctan\left(-\sqrt{2}\right) \)}
\end{problem}

\begin{answer}
C.
\end{answer}

\begin{solution}
令 \(u=\arcsin(-\frac13)\)，则 \(u\in[-\frac\pi2,0]\)，且 \(\tan u=\dfrac{-1/3}{\sqrt{1-1/9}}=-\dfrac{1}{2\sqrt2}\).因为正切函数在 \((-\frac\pi2,\frac\pi2)\) 上递增，且 \(-\sqrt2< -\frac{1}{2\sqrt2}\)，所以 \(\arctan(-\sqrt2)<u\).另一方面，\(u<0\)，而 \(\arccos(-\frac23)\in(\frac\pi2,\pi)\)，故 \(u<\arccos(-\frac23)\).因此顺序为 \(\arctan(-\sqrt2)<\arcsin(-\frac13)<\arccos(-\frac23)\)，故选 C.
\end{solution}

\begin{problem}

直角三角形 \(ABC\) 的斜边 \(BC\) 在平面 \(\alpha \) 内，顶点 \(A\) 在平面 \(\alpha \) 外，则 \(\triangle ABC \) 的两条直角边在平面 \(\alpha \) 上的射影与斜边 \(BC\) 所组成的图形只能是（\quad）

\choicesfive
    {一条线段}
    {一个锐角三角形}
    {一个钝角三角形}
    {一条线段或一个钝角三角形}
    {一条线段或一个锐角三角形}
\end{problem}

\begin{answer}
D.
\end{answer}

\begin{solution}
设点 \(A\) 在平面 \(\alpha\) 上的射影为 \(A'\).因为 \(A\) 不在平面 \(\alpha\) 内，所以 \(AA'\ne0\)，且 \(AA'\perp\alpha\).将 \(\overrightarrow{AB}\)、\(\overrightarrow{AC}\) 分解到平面 \(\alpha\) 内和垂直于平面 \(\alpha\) 的方向，有
\(0=\overrightarrow{AB}\cdot\overrightarrow{AC}=\overrightarrow{A'B}\cdot\overrightarrow{A'C}+AA'^{2}\)，从而 \(\overrightarrow{A'B}\cdot\overrightarrow{A'C}=-AA'^{2}<0\).所以当 \(A'\) 不在直线 \(BC\) 上时，投影三角形在 \(A'\) 处为钝角三角形；当 \(A'\) 在直线 \(BC\) 上时，三条投影共线，组成一条线段.因此只能是一条线段或一个钝角三角形，故选 D.
\end{solution}

\begin{problem}

如果实数 \(x\) 与 \(y\) 满足方程 \(x + y - 4 = 0 \)，则 \(x^{2} + y^{2} \) 的最小值是（\quad）

\choicesfive
    {\(4\)}
    {\(6\)}
    {\(8\)}
    {\(10\)}
    {\(12\)}
\end{problem}

\begin{answer}
C.
\end{answer}

\begin{solution}
由柯西不等式，\((x+y)^2\leqslant2(x^2+y^2)\).又 \(x+y=4\)，所以 \(x^2+y^2\geqslant\frac{16}{2}=8\).当且仅当 \(x=y\)，结合 \(x+y=4\) 得 \(x=y=2\)，等号可以取得.因此最小值为 \(8\)，故选 C.
\end{solution}

\begin{problem}

和 \(y\) 轴相切，且和半圆 \(x^{2} + y^{2} = 4\)（\(0 \leqslant x \leqslant 2\)）相内切的动圆圆心的轨迹方程是（\quad）

\choicesfive
    {\(y^{2} = -4(x - 1)\)（\(0 < x \leqslant 1\)）}
    {\(y^{2} = 4(x - 1)\)（\(0 < x \leqslant 1\)）}
    {\(y^{2}=4(x+1)\)（\(0<x\leqslant1\)）}
    {\(y^{2}=-2(x-1)\)（\(0<x\leqslant1\)）}
    {\(y^{2}=2(x+1)\)（\(0<x\leq1\)）}
\end{problem}

\begin{answer}
A.
\end{answer}

\begin{solution}
设动圆圆心为 \((x,y)\)，半径为 \(r\).与 \(y\) 轴相切且圆心位于右侧时，\(r=x\)，其中 \(x>0\).它与圆心为原点、半径为 \(2\) 的圆内切，所以圆心距满足 \(\sqrt{x^2+y^2}=2-r=2-x\).平方并消去 \(x^2\)，得 \(y^2=4(1-x)=-4(x-1)\).由 \(r>0\) 及内切位置得 \(0<x\leqslant1\)，故选 A.
\end{solution}

\begin{problem}

设 \(a\)，\(b\)，\(c\) 是空间中的三条直线，下面给出四个命题：

\begin{circlelist}
\item 如果 \(a \perp b\)，\(b \perp c\)，则 \(a \parallel c\).
\item 如果 \(a\)，\(b\) 是异面直线，\(b\)，\(c\) 是异面直线，则 \(a\)，\(c\) 也是异面直线.
\item 如果 \(a\) 和 \(b\) 相交，\(b\) 和 \(c\) 相交，则 \(a\) 和 \(c\) 也相交.
\item 如果 \(a\) 和 \(b\) 共面，\(b\) 和 \(c\) 共面，则 \(a\) 和 \(c\) 也共面.
\end{circlelist}

那么，在上述命题中，真命题的个数是（\quad）

\choicesfive
    {\(4\)}
    {\(3\)}
    {\(2\)}
    {\(1\)}
    {\(0\)}
\end{problem}

\begin{answer}
E.
\end{answer}

\begin{solution}
四个命题均不成立.命题（1）取直线 \(b\) 上一点，过该点在垂直于 \(b\) 的平面内作两条不平行直线 \(a\)，\(c\)，则 \(a\perp b\)、\(c\perp b\)，但 \(a\) 不平行于 \(c\).命题（2）取 \(b\) 为 \(z\) 轴，取 \(a=\{(t,1,0)\mid t\in\R\}\)、\(c=\{(t,2,0)\mid t\in\R\}\).直线 \(a\)，\(b\) 异面，直线 \(b\)，\(c\) 也异面，但 \(a\parallel c\)，故 \(a\)，\(c\) 不是异面直线.命题（3）在平面 \(z=0\) 内取两条互相平行而不同的直线 \(a:y=0\)、\(c:y=1\)，再取 \(b:x=0\)，则 \(a\) 和 \(b\) 相交，\(b\) 和 \(c\) 相交，而 \(a\)，\(c\) 不相交.命题（4）取两个相交平面 \(\alpha:y=0\)、\(\beta:x=0\)，交线为 \(b\)（即 \(z\) 轴）；在 \(\alpha\) 内取 \(a=\{(t,0,1)\mid t\in\R\}\)，在 \(\beta\) 内取 \(c=\{(0,t,2)\mid t\in\R\}\)，则 \(a\)，\(c\) 异面.因此真命题个数为 \(0\)，故选 E.
\end{solution}

\begin{problem}

棱柱成为直棱柱的一个必要但不充分的条件是（\quad）

\choicesfive
    {棱柱有一条侧棱与底面垂直}
    {棱柱有一条侧棱与底面的两条边垂直}
    {棱柱有一条侧面与底面的一条边垂直}
    {棱柱有一条侧面是矩形，且它与底面垂直}
    {棱柱有两个相邻的侧面互相垂直}
\end{problem}

\begin{answer}
B.
\end{answer}

\begin{solution}
若棱柱是直棱柱，则每条侧棱都垂直于底面，因而一条侧棱必垂直于底面内的两条边，所以选项 B 是必要条件.反之，取底面为 \(xy\) 平面内的矩形，取其中两条互相平行的边的方向为 \((1,0,0)\)，并令棱柱的侧棱方向为 \((0,1,1)\).此时该侧棱垂直于这两条平行边，但它含有沿底面的分量，因而不垂直于底面，所得棱柱不是直棱柱.因此该条件必要但不充分，故选 B.
\end{solution}

\begin{problem}

用数学归纳法证明： \(|\sin n\theta| \leqslant n|\sin\theta| \).（\(n\) 是自然数）
\end{problem}

\begin{solution}
证：当 \(n=1\) 时，左边 \(|\sin\theta|=\) 右边，所以原不等式成立.

假设当 \(n = k\)（\(k \geq 1\)）时，不等式成立，即
\(\left|\sin k\theta\right|\leqslant k\left|\sin\theta\right|\).
那么，当 \(n=k+1\) 时，
\examdisplaychain{
|\sin(k+1)\theta|
&=\left|\sin k\theta\cos\theta+\cos k\theta\sin\theta\right|\\
&\leqslant\left|\sin k\theta\right|\left|\cos\theta\right|+\left|\cos k\theta\right|\left|\sin\theta\right|\\
&\leqslant|\sin k\theta|+|\sin\theta|
\leqslant k|\sin\theta|+|\sin\theta|
=(k+1)\left|\sin\theta\right|
}
这就是说，原不等式对于 \(n = k + 1\) 成立. 根据上述归纳步骤，原不等式对于任意自然数 \(n\) 都成立.
\end{solution}

\begin{problem}

已知直三棱柱 \(ABC-A_1B_1C_1 \) 的侧棱 \(AA_1=4\mathrm{~cm} \)，它的底面 \(\triangle ABC \) 中有 \(AC=BC=2\mathrm{~cm} \)，\(\angle C=90^\circ \)，\(E\) 是 \(AB\) 的中点.

\begin{enumerate}
\item 求证：\(CE \perp AB_{1}\).
\item 求证：\(CE\) 和 \(AB_{1}\) 所在的异面直线的距离等于 \(\frac{2\sqrt{3}}{3}\mathrm{~cm}\).
\item 求截面 \(ACB_{1}\) 与侧面 \(ABB_{1}A_{1}\) 所成的二面角的大小（只求其中的锐角，可用反三角函数表示）.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 证：因为 \(ABC-A_1B_1C_1\) 是直三棱柱，所以 \(AA_{1}\perp\) 底面 \(\triangle ABC\). 又因为 \(CE \subset\) 平面 \(ABC\)，所以 \(CE \perp AA_{1}\). 由于 \(E\) 是等腰三角形 \(ABC\) 底边 \(AB\) 的中点，所以 \(CE \perp AB\). 故 \(CE \perp\) 平面 \(ABB_1A_1\). 因为 \(AB_1 \subset\) 平面 \(ABB_1A_1\)，因此 \(CE \perp AB_1\).

\item 证：因为 \(CE \perp AB\)，\(CE \perp AB_{1}\)，所以 \(CE \perp\) 平面 \(ABB_{1}A_{1}\). 过 \(E\) 作 \(ED \perp AB_1\)，垂足为 \(D\). 因为 \(ED \subset\) 平面 \(ABB_1A_1\)，所以 \(CE \perp ED\). 故 \(ED\) 是 \(CE\) 和 \(AB_{1}\) 所在的异面直线的距离.
在 \(\mathrm{Rt}\triangle ABB_1\) 和 \(\mathrm{Rt}\triangle ADE\) 中，\(\angle A\) 公共，所以 \(\triangle ABB_1 \sim \triangle ADE\). 因此
\(\frac{ED}{BB_1} = \frac{AE}{AB_1}\)，
即
\(ED = \frac{AE \cdot BB_1}{AB_1}\).
因为 \(AC=BC=2\)，\(\angle ACB=90^{\circ}\)，所以 \(AB=2\sqrt{2}\)，\(AE=\frac{1}{2}AB=\sqrt{2}\).
又在 \(\mathrm{Rt}\triangle ABB_1\) 中，\(BB_1 = 4\)，所以
\[
AB_1 = \sqrt{AB^2 + BB_1^2} = \sqrt{8 + 16} = 2\sqrt{6}
\]
因此
\(ED=\frac{\sqrt{2}\cdot 4}{2\sqrt{6}}=\frac{2\sqrt{3}}{3}\mathrm{~cm}\).

\item 因为 \(ED\perp AB_1\)，\(CE\perp\) 侧面 \(ABB_1A_1\)，连结 \(CD\)，依三垂线定理有 \(CD\perp AB_1\). 故 \(\angle CDE\) 是截面 \(CAB_1\) 和侧面 \(ABB_1A_1\) 所成的二面角的平面角，且为锐角. 又 \(CE\) 是 \(\mathrm{Rt}\triangle ABC\) 的斜边上的中线，所以 \(CE=\frac{1}{2}AB=\sqrt{2}\). 由上可知 \(ED=\frac{2\sqrt{3}}{3}\). 因此在 \(\mathrm{Rt}\triangle CED\) 中，
\(\tan\angle CDE=\frac{CE}{ED}=\frac{\sqrt{6}}{2}\).
故截面 \(CAB_{1}\) 和侧面 \(ABB_{1}A_{1}\) 所成的较小的二面角等于 \(\arctan\frac{\sqrt{6}}{2}\).
\end{enumerate}
\end{solution}

\begin{problem}

设函数 \(f(x)=\log_{3}\left(x^{2}-4mx+4m^{2}+m+\frac{1}{m-1}\right) \)，其中 \(m\) 是实数，又用 \(M\) 表示集合 \(\{m \mid m>1\} \).

\begin{enumerate}
\item 求证：当 \(m \in M \) 时，\(f(x)\) 对所有实数 \(x\) 都有意义；反之，如果 \(f(x)\) 对所有实数 \(x\) 都有意义，则 \(m \in M\).
\item 当 \(m \in M \) 时，求函数 \(f(x)\) 的最小值.
\item 求证：对每一个 \(m \in M \)，函数 \(f(x)\) 的最小值都不小于 \(1\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 证：当 \(m \in M\) 时，有 \(m > 1\)，从而对所有的实数 \(x\) 都有
\(x^{2}-4mx+4m^{2}+m+\frac{1}{m-1}=(x-2m)^{2}+m+\frac{1}{m-1}\geqslant m+\frac{1}{m-1}>0\).
于是，当 \(m \in M\) 时，函数 \(f(x)\) 对所有的实数 \(x\) 都有意义.
反之，如果 \(f(x)\) 对所有的实数 \(x\) 都有意义，则对所有的实数 \(x\)，都有
\(x^{2}-4mx+4m^{2}+m+\frac{1}{m-1}>0\).
而当 \(x=2m\) 时，上式变为
\(m+\frac{1}{m-1}>0\)，
即
\(\frac{m^{2}-m+1}{m-1}>0\)
或
\(\frac{\left(m-\frac{1}{2}\right)^{2}+\frac{3}{4}}{m-1}>0\).
由于上式左端的分子总是正数，所以它的分母 \(m-1>0\)，即 \(m>1\)，从而 \(m\in M\).

\item 因为以 \(3\) 为底的对数函数是增函数，所以
\(f(x)=\log_{3}\left[(x-2m)^{2}+m+\frac{1}{m-1}\right]\geqslant\log_{3}\left(m+\frac{1}{m-1}\right)\).
又因为
\(f(2m)=\log_{3}\left(m+\frac{1}{m-1}\right)\)，
所以，当 \(m \in M\) 时，\(f(x)\) 的最小值是
\(\log_{3}\left(m+\frac{1}{m-1}\right)\).

\item 证：当 \(m \in M\) 时，有 \(m > 1\)，所以
\[
m+\frac{1}{m-1}=(m-1)+\frac{1}{m-1}+1\geqslant2\sqrt{(m-1)\frac{1}{m-1}}+1=3
\]
上式两端取以 \(3\) 为底的对数，得
\(\log_{3}\left(m+\frac{1}{m-1}\right)\geqslant\log_{3}3=1\).
于是，根据（2）的结果可知，对每一个 \(m \in M\)，函数 \(f(x)\) 的最小值都不小于 \(1\).
\end{enumerate}
\end{solution}

\begin{problem}

在直角坐标系中，有椭圆 \(I\) 和抛物线 \(II\)，它们的参数方程分别是 \(I\)：\(\begin{cases}
x = m + 2\cos\phi,\\
y = \sqrt{3}\sin\phi
\end{cases}\)（\(m\) 是常数，\(\phi\) 是参数）；\(II\)：\(\begin{cases}
x = \frac{3}{2}+t^{2},\\
y = \sqrt{6}t
\end{cases}\)（\(t\) 是参数）.

\begin{enumerate}
\item 求证：当 \(m=4\) 时，椭圆 \(I\) 有一个焦点和一条准线分别与抛物线 \(II\) 的焦点和准线重合.
\item 求证：当且仅当 \(m \in [-\frac{1}{2}, \frac{7}{2}]\) 时，椭圆 \(I\) 与抛物线 \(II\) 有交点.
\item 在区间 \([-\frac{1}{2}, \frac{7}{2}]\) 中有哪些 \(m\) 值，使得椭圆 \(I\) 和抛物线 \(II\) 的交点与坐标原点的距离等于这个交点与椭圆 \(I\) 的中心的距离？
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 证：将曲线 \(I\)，\(II\) 的方程化为直角坐标方程：
\(I:\ \frac{(x-m)^{2}}{4}+\frac{y^{2}}{3}=1\)，\(II:\ y^{2}=6\left(x-\frac{3}{2}\right)\).
当 \(m = 4\) 时，椭圆 \(I\) 中心的坐标为 \((4,0)\)，长轴在 \(x\) 轴上，长半轴长 \(a = \sqrt{4} = 2\)，短半轴长 \(b = \sqrt{3}\). 从而半焦距
\(c = \sqrt{a^2 - b^2} = 1\)，
焦点坐标 \((4 \pm c,0)\) 即为 \((5,0)\) 和 \((3,0)\)，准线方程
\(x = 4 \pm \frac{a^2}{c}\)，
即为 \(x = 8\) 和 \(x = 0\).
由抛物线 \(II\) 的方程可知：该抛物线以 \(x\) 轴为对称轴，开口向右，顶点的坐标为 \(\left(\frac{3}{2},0\right)\)，焦点至顶点的距离是 \(\frac{p}{2}=\frac{3}{2}\). 从而焦点的坐标为
\(\left(\frac{3}{2}+\frac{p}{2},0\right)=(3,0)\)，
准线方程是
\(x=\frac{3}{2}-\frac{p}{2}=0\).
所以，椭圆 \(I\) 的左焦点和左准线分别与抛物线 \(II\) 的焦点和准线重合.

\item 证：将曲线 \(I\)，\(II\) 的直角坐标方程联立，得
\examdisplaycases{}{\frac{(x-m)^{2}}{4}+y^{2}=1,\\
y^{2}=6\left(x-\frac{3}{2}\right).}
那么，曲线 \(I\)，\(II\) 有交点，当且仅当方程组有实数解 \(x\)，\(y\). 将第二式代入第一式，整理得
\(x^{2}+(8-2m)x+m^{2}-16=0\).
又由于
\(x - \frac{3}{2} = \frac{y^2}{6} \geq 0\)，
故方程组有实数解，当且仅当上式有不小于 \(\frac{3}{2}\) 的实根.
上式的两个根为
\(x_{1}=-(4-m)+2\sqrt{2(4-m)}\)，\(x_{2}=-(4-m)-2\sqrt{2(4-m)}\).
可见，当且仅当 \(4-m\geqslant0\) 时，\(x_1\)，\(x_2\) 是实根，且这时 \(x_2=-(4-m)-2\sqrt{2(4-m)}\leqslant0\).所以只能由 \(x_1\) 满足 \(x_1\geqslant\frac32\)，即 \(m\) 满足不等式组
\examdisplaycases{}{4-m\geqslant0,\\
-4+m+2\sqrt{2(4-m)}\geqslant\frac{3}{2}.}
令 \(u=\sqrt{2(4-m)}\)，则 \(u\geqslant0\)，且 \(4-m=\frac{u^{2}}2\).第二个不等式化为 \(-\frac{u^{2}}2+2u\geqslant\frac32\)，即 \(u^{2}-4u+3\leqslant0\)，所以 \(1\leqslant u\leqslant3\).于是 \(\frac12\leqslant4-m\leqslant\frac92\)，即 \(-\frac12\leqslant m\leqslant\frac72\).综合起来，当且仅当 \(m\in\left[-\frac12,\frac72\right]\) 时，椭圆 \(I\) 与抛物线 \(II\) 有交点.

\item 因为椭圆 \(I\) 中心的坐标为 \((m,0)\)，所以曲线 \(I\)，\(II\) 的交点 \((x, y)\) 满足题设，等价于 \(x\)，\(y\) 满足
\(x^{2} + y^{2} = (x - m)^{2} + y^{2}\)，
即
\(m(m-2x)=0\).
由（2）的证明可知，曲线 \(I\)，\(II\) 的交点横坐标是
\(x = (m - 4) + 2 \sqrt{2(4 - m)}\)，\(m \in \left[-\frac{1}{2}, \frac{7}{2}\right]\).
将它代入上式，得知所求的 \(m\) 值应满足方程
\(m \left[ 8 - m - 4 \sqrt{2 (4 - m)} \right] = 0\)，\(m \in \left[-\frac{1}{2}, \frac{7}{2}\right]\).
解方程：由 \(m = 0\) 得 \(m_{1} = 0\)；由 \(8 - m - 4\sqrt{2(4 - m)} = 0\)，即
\(m^{2} + 16m - 64 = 0\)，
得 \(m_{2} = 8(\sqrt{2} -1)\in \left[-\frac{1}{2},\frac{7}{2}\right]\)，而 \(m_{3} = -8(\sqrt{2} +1)\) 不在区间 \(\left[-\frac{1}{2},\frac{7}{2}\right]\) 内（应舍去）. 直接检验可知，当 \(m = 0\) 或 \(8(\sqrt{2} - 1)\) 时，曲线 \(I\)，\(II\) 的交点满足题设条件，故所求的 \(m\) 的值是 \(0\) 和 \(8(\sqrt{2} - 1)\).
\end{enumerate}
\end{solution}


\section{附加题}

\begin{problem}

已知函数 \(f(x)=x^{5}-5x+c \)，其中 \(c\) 是实数.

\begin{enumerate}
\item 求 \(f(x)\) 的极大值和极小值.
\item 证明方程 \(f(x)=0\) 的不同实根的个数不大于 \(3\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由 \(f'(x)=5x^{4}-5=5(x^{4}-1)\). 令 \(f'(x)=0\)，即 \(5(x^{4}-1)=0\)，这个方程的实根是 \(x=\pm1\). 而
\(f(-1) = -1 + 5 + c = c + 4\)，\(f(1) = 1 - 5 + c = c - 4\).
函数 \(f(x)\) 在 \((-\infty,-1)\) 与 \((1,+\infty)\) 上递增，在 \((-1,1)\) 上递减. 所以 \(f(x)\) 的极大值是 \(f(-1)=c+4\)，\(f(x)\) 的极小值是 \(f(1)=c-4\).

函数 \(f(x)\) 的变化情况如下表所示：

\examdisplayarray{}{c|ccccc}{
x & (-\infty,-1) & -1 & (-1,1) & 1 & (1,+\infty) \\
\hline
f'(x) & + & 0 & - & 0 & + \\
\hline
f(x) & \nearrow & c+4 & \searrow & c-4 & \nearrow
}{}

\item 证：用反证法证明. 假如方程 \(f(x) = 0\) 的不同实根个数大于 \(3\)，则至少可找到四个不同的实数 \(x_{1} < x_{2} < x_{3} < x_{4}\)，使得
\(f(x_{1}) = f(x_{2}) = f(x_{3}) = f(x_{4}) = 0\).
根据微分中值定理，有 \(\theta_{1}\in(x_{1},x_{2})\)，使得
\(f(x_{2}) - f(x_{1}) = f'(\theta_{1})(x_{2} - x_{1})\).
由上式及 \(x_{2} > x_{1}\)，可知 \(f'(\theta_1) = 0\). 这表明方程 \(f'(x) = 0\) 在区间 \((x_{1},x_{2})\) 中至少有一个实根 \(\theta_{1}\). 同理可证：方程 \(f'(x)=0\) 在区间 \((x_{2}, x_{3})\)，\((x_{3}, x_{4})\) 中分别有实根 \(\theta_{2}\)，\(\theta_{3}\). 显然，有 \(\theta_{1}<\theta_{2}<\theta_{3}\)，即方程 \(f'(x)=0\) 至少有三个不同的实根，这与（1）中所得的结果相矛盾. 从而，方程 \(f(x)=0\) 的不同实根的个数不大于 \(3\).
\end{enumerate}
\end{solution}
