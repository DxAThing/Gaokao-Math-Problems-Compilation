\chapter{1952年全国卷}

\section{填空题}


\begin{problem}
分解因式：\(x^{4} - y^{4} =\)\fillinblank{}.
\end{problem}

\begin{answer}
\(\left( x-y \right)\left( x+y \right)\left( x^{2}+y^{2} \right)\).
\end{answer}

\begin{solution}
利用平方差公式，先得 \(x^{4}-y^{4}=\left( x^{2}-y^{2} \right)\left( x^{2}+y^{2} \right)\)，再将 \(x^{2}-y^{2}\) 分解为 \(\left( x-y \right)\left( x+y \right)\)，所以 \(x^{4}-y^{4}=\left( x-y \right)\left( x+y \right)\left( x^{2}+y^{2} \right)\).
\end{solution}

\begin{problem}
若 \(\log_{10} 2x = 2\log_{10} x\)，问 \(x=\)\fillinblank{}.
\end{problem}

\begin{answer}
\(2\).
\end{answer}

\begin{solution}
由对数式有意义，需 \(x>0\). 由对数的乘方公式，\(2\log_{10}x=\log_{10}x^{2}\)，于是题设等价于 \(\log_{10}(2x)=\log_{10}x^{2}\). 因为底数 \(10>1\)，对数函数在正数上是单调函数，所以 \(2x=x^{2}\). 结合 \(x>0\) 及 \(x^{2}-2x=x(x-2)=0\)，舍去 \(x=0\)，得 \(x=2\).
\end{solution}

\begin{problem}
若方程式 \(x^{3} + bx^{2} + cx + d = 0\) 之三根为 \(1\)，\(-1\)，\(\frac{1}{2}\)，则 \(c=\)\fillinblank{}.
\end{problem}

\begin{answer}
\(-1\).
\end{answer}

\begin{solution}
由韦达定理，三根之两两乘积的和等于一次项系数 \(c\)，故 \(c=1\times(-1)+1\times\frac{1}{2}+(-1)\times\frac{1}{2}=-1+\frac{1}{2}-\frac{1}{2}=-1\).
\end{solution}

\begin{problem}
若 \(\sqrt{x^{2}+7}-4=0\)，则 \(x=\)\fillinblank{}.
\end{problem}

\begin{answer}
\(\pm 3\).
\end{answer}

\begin{solution}
由 \(\sqrt{x^{2}+7}-4=0\)，得 \(\sqrt{x^{2}+7}=4\). 两边平方，得到 \(x^{2}+7=16\)，所以 \(x^{2}=9\). 因而 \(x=3\) 或 \(x=-3\)，即 \(x=\pm3\). 这两个值代回原式均有 \(\sqrt{16}-4=0\)，故都成立.
\end{solution}

\begin{problem}
\(\begin{vmatrix}1 & 2 & 3 \\ 4 & 5 & 0 \\ 3 & 2 & 1\end{vmatrix}=\)\fillinblank{}.
\end{problem}

\begin{answer}
\(-24\).
\end{answer}

\begin{solution}
按第一行展开行列式，得
\[
\begin{vmatrix}1&2&3\\4&5&0\\3&2&1\end{vmatrix}=1\begin{vmatrix}5&0\\2&1\end{vmatrix}-2\begin{vmatrix}4&0\\3&1\end{vmatrix}+3\begin{vmatrix}4&5\\3&2\end{vmatrix}=1(5)-2(4)+3(8-15)=-24.
\]
\end{solution}

\begin{problem}
两个圆的半径都是 \(4\text{ 寸}\)，并且一个圆通过另一圆的圆心，则这两个圆的公共弦之长是\fillinblank{}\(\text{寸}\).
\end{problem}

\begin{answer}
\(4\sqrt{3}\).
\end{answer}

\begin{solution}
设两圆圆心分别为 \(O_{1}\)、\(O_{2}\)，则 \(O_{1}O_{2}=4\)，因为一个圆通过另一个圆的圆心，且两圆半径均为 \(4\text{ 寸}\). 公共弦所在直线到任一圆心的距离相等，且在 \(O_{1}O_{2}\) 的中点处，故该距离为 \(2\text{ 寸}\). 设公共弦长为 \(l\)，则在圆心、公共弦中点和弦端点组成的直角三角形中，半弦长为 \(\frac l2\)，从而
\[
\left(\frac l2\right)^{2}+2^{2}=4^{2}.
\]
所以 \(\frac{l^{2}}{4}=12\)，得 \(l=4\sqrt3\text{ 寸}\).
\end{solution}

\begin{problem}
三角形 \(\triangle ABC\) 的面积是 \(60\text{ 平方寸}\)，\(M\) 是 \(AB\) 的中点，\(N\) 是 \(AC\) 的中点，则 \(\triangle AMN\) 的面积是\fillinblank{}\(\text{平方寸}\).
\end{problem}

\begin{answer}
\(15\).
\end{answer}

\begin{solution}
连接 \(MN\). 因为 \(M\)、\(N\) 分别是 \(AB\)、\(AC\) 的中点，所以中位线定理给出 \(MN\parallel BC\)，且 \(AM=\frac12AB\)、\(AN=\frac12AC\). 因此 \(\triangle AMN\sim\triangle ABC\)，相似比为 \(\frac12\)，面积比为相似比的平方，即 \(\frac{S_{\triangle AMN}}{S_{\triangle ABC}}=\frac14\). 于是 \(S_{\triangle AMN}=\frac14\times60=15\text{ 平方寸}\).
\end{solution}

\begin{problem}
正十边形的一内角是\fillinblank{}\(\text{度}\).
\end{problem}

\begin{answer}
\(144^{\circ}\).
\end{answer}

\begin{solution}
正十边形可分成 \(10\) 个以中心为顶点的等腰三角形，因此每个中心角为 \(\frac{360^{\circ}}{10}=36^{\circ}\). 正十边形的一内角等于两个直角减去一个中心角，所以内角为 \(180^{\circ}-36^{\circ}=144^{\circ}\).
\end{solution}

\begin{problem}
祖冲之的圆周率 \(\pi=\)\fillinblank{}.
\end{problem}

\begin{answer}
\(\frac{22}{7}\)，\(\frac{355}{113}\)，\(3.14159265\).
\end{answer}

\begin{solution}
圆周率 \(\pi\) 是圆周长与直径之比，不能用有限小数精确表示. 祖冲之给出的两个著名近似值分别是约率 \(\frac{22}{7}\) 和密率 \(\frac{355}{113}\)，其中密率更精确；写成小数时可取 \(\pi\approx3.14159265\). 因而按原题答案可填 \(\frac{22}{7}\)、\(\frac{355}{113}\) 或 \(3.14159265\).
\end{solution}

\begin{problem}
球的面积等于大圆面积的\fillinblank{} 倍.
\end{problem}

\begin{answer}
\(4\).
\end{answer}

\begin{solution}
设球的半径为 \(r\). 球的表面积为 \(4\pi r^{2}\)，其大圆半径也是 \(r\)，所以大圆面积为 \(\pi r^{2}\). 因而球的面积与大圆面积之比为 \(\frac{4\pi r^{2}}{\pi r^{2}}=4\)，即球的面积等于大圆面积的 \(4\) 倍.
\end{solution}

\begin{problem}
直圆锥之底之半径为 \(3\text{ 尺}\)，斜高为 \(5\text{ 尺}\)，则其体积为\fillinblank{}\(\text{立方尺}\).
\end{problem}

\begin{answer}
\(12\pi\).
\end{answer}

\begin{solution}
设直圆锥的高为 \(h\). 由轴截面是直角三角形，斜高为斜边，故 \(h^{2}+3^{2}=5^{2}\)，从而 \(h=4\text{ 尺}\). 因此圆锥体积为 \(\frac13\pi\cdot3^{2}\cdot4=12\pi\text{ 立方尺}\).
\end{solution}

\begin{problem}
正多面体有\fillinblank{} 种，其名称为\fillinblank{}.
\end{problem}

\begin{answer}
\(5\)，正四面体，正六面体，正八面体，正十二面体，正二十面体.
\end{answer}

\begin{solution}
设每个面是正 \(p\) 边形，每个顶点有 \(q\) 个面相交. 因为每个顶点周围的面角和必须小于 \(360^{\circ}\)，而正 \(p\) 边形的内角为 \(\frac{(p-2)180^{\circ}}{p}\)，所以 \(q\frac{(p-2)180^{\circ}}{p}<360^{\circ}\)，其中 \(p\ge3\)、\(q\ge3\) 为整数. 当 \(p=3\) 时，\(q=3,4,5\)；当 \(p=4\) 时只能有 \(q=3\)；当 \(p=5\) 时只能有 \(q=3\)；当 \(p\ge6\) 时左边不小于 \(3\times120^{\circ}=360^{\circ}\)，不可能成立. 因而只有五种组合 \((p,q)=(3,3),(3,4),(3,5),(4,3),(5,3)\)，分别得到正四面体、正八面体、正二十面体、正六面体、正十二面体，所以正多面体共有 \(5\) 种.
\end{solution}

\begin{problem}
已知 \(\sin \theta = \frac{1}{3}\)，求 \(\cos 2\theta=\)\fillinblank{}.
\end{problem}

\begin{answer}
\(\frac{7}{9}\).
\end{answer}

\begin{solution}
由二倍角公式，\(\cos2\theta=1-2\sin^{2}\theta\). 代入 \(\sin\theta=\frac13\)，得 \(\cos2\theta=1-2\left(\frac13\right)^{2}=1-\frac29=\frac79\).
\end{solution}

\begin{problem}
方程式 \(\tan 2x = 1\) 的通解为 \(x=\)\fillinblank{}.
\end{problem}

\begin{answer}
\(\frac{1}{2}\left( n\pi + \frac{\pi}{4} \right)\).
\end{answer}

\begin{solution}
由 \(\tan2x=1=\tan\frac{\pi}{4}\)，正切函数的周期为 \(\pi\)，所以 \(2x=\frac{\pi}{4}+n\pi\)，其中 \(n\in\Z\). 两边除以 \(2\)，得 \(x=\frac12\left(n\pi+\frac{\pi}{4}\right)=\frac{\pi}{8}+\frac{n\pi}{2}\)，其中 \(n\in\Z\).
\end{solution}

\begin{problem}
太阳仰角为 \(30^{\circ}\) 时塔影长 \(5\text{ 丈}\)，求塔高 \(=\)\fillinblank{}.
\end{problem}

\begin{answer}
\(\frac{5}{3}\sqrt{3}\).
\end{answer}

\begin{solution}
设塔高为 \(h\)，塔影长为 \(5\text{ 丈}\). 由太阳仰角为 \(30^{\circ}\)，在塔、影子和太阳光线组成的直角三角形中，\(\tan30^{\circ}=\frac{h}{5}\). 因为 \(\tan30^{\circ}=\frac{1}{\sqrt3}\)，所以 \(h=\frac{5}{\sqrt3}=\frac{5\sqrt3}{3}\text{ 丈}\).
\end{solution}

\begin{problem}
三角形 \(\triangle ABC\) 之 \(b\) 边为 \(3\text{ 寸}\)，\(c\) 边为 \(4\text{ 寸}\)，\(A\) 角为 \(30^{\circ}\)，则 \(\triangle ABC\) 的面积为\fillinblank{}\(\text{平方寸}\).
\end{problem}

\begin{answer}
\(3\).
\end{answer}

\begin{solution}
三角形面积由两边及其夹角给出：\(S_{\triangle ABC}=\frac12bc\sin A\). 代入 \(b=3\text{ 寸}\)、\(c=4\text{ 寸}\) 和 \(A=30^{\circ}\)，得 \(S_{\triangle ABC}=\frac12\cdot3\cdot4\cdot\frac12=3\text{ 平方寸}\).
\end{solution}

\begin{problem}
已知一直线经过点 \(\left( 2,-3 \right)\)，其斜率为 \(-1\)，则此直线之方程式为\fillinblank{}.
\end{problem}

\begin{answer}
\(x + y + 1 = 0\).
\end{answer}

\begin{solution}
由点斜式，过 \((2,-3)\) 且斜率为 \(-1\) 的直线满足 \(y-(-3)=-1(x-2)\). 化简得 \(y+3=-x+2\)，即 \(x+y+1=0\).
\end{solution}

\begin{problem}
若原点在一圆上，而此圆的圆心为点 \(\left( 3,4 \right)\)，则此圆的方程式为\fillinblank{}.
\end{problem}

\begin{answer}
\(x^{2} + y^{2} - 6x - 8y = 0\).
\end{answer}

\begin{solution}
圆心为 \((3,4)\)，且圆过原点 \((0,0)\)，所以半径满足 \(r^{2}=(0-3)^{2}+(0-4)^{2}=25\). 圆的标准方程为 \((x-3)^{2}+(y-4)^{2}=25\). 展开并整理，得 \(x^{2}+y^{2}-6x-8y=0\).
\end{solution}

\begin{problem}
原点至 \(3x + 4y + 1 = 0\) 之距离 \(=\)\fillinblank{}.
\end{problem}

\begin{answer}
\(\frac{1}{5}\).
\end{answer}

\begin{solution}
点 \((0,0)\) 到直线 \(3x+4y+1=0\) 的距离为 \(\frac{|3\cdot0+4\cdot0+1|}{\sqrt{3^{2}+4^{2}}}=\frac{1}{5}\).
\end{solution}

\begin{problem}
抛物线 \(y^{2} - 8x + 6y + 17 = 0\) 之顶点之坐标为\fillinblank{}.
\end{problem}

\begin{answer}
\(\left( 1,-3 \right)\).
\end{answer}

\begin{solution}
将方程配方，得 \(y^{2}+6y-8x+17=0\)，即 \((y+3)^{2}=8x-8=8(x-1)\). 与抛物线标准方程 \((y-k)^{2}=4p(x-h)\) 比较，顶点为 \((h,k)=(1,-3)\).
\end{solution}

\section{解答题}


\begin{problem}
解方程式 \(x^{4} + 5x^{3} - 7x^{2} - 8x - 12 = 0\).
\end{problem}

\begin{solution}
设
\[
P(x)=x^{4}+5x^{3}-7x^{2}-8x-12.
\]
将中间项分组，得
\[
P(x)=x^{2}\left(x^{2}+5x-6\right)-\left(x^{2}+8x+12\right).
\]
因为 \(x^{2}+5x-6=(x+6)(x-1)\)，\(x^{2}+8x+12=(x+6)(x+2)\)，所以
\[
P(x)=(x+6)\left[x^{2}(x-1)-(x+2)\right]
=(x+6)\left(x^{3}-x^{2}-x-2\right).
\]
又因为
\[
x^{3}-x^{2}-x-2=(x-2)\left(x^{2}+x+1\right),
\]
所以原方程化为
\[
(x+6)(x-2)(x^{2}+x+1)=0.
\]
由此得 \(x=-6\) 或 \(x=2\)，以及 \(x^{2}+x+1=0\). 后一方程的两个根分别为
\(x=\frac{-1+\sqrt{3}\i}{2}\) 和 \(x=\frac{-1-\sqrt{3}\i}{2}\).
若记 \(\omega=\frac{-1+\sqrt{3}\i}{2}\)，则另一个根为 \(\omega^{2}\). 因而原方程的四个根为
\(2\)，\(-6\)，\(\omega\)，\(\omega^{2}\).
\end{solution}

\begin{problem}
\(\triangle ABC\) 中，\(\angle A\) 的外分角线与此三角形的外接圆相交于 \(D\)，求证：\(BD = CD\).
\end{problem}

\begin{solution}
设三角形 \(ABC\) 的外接圆半径为 \(R\). 由于 \(AD\) 是 \(\angle A\) 的外分角平分线，射线 \(AB\)、\(AC\) 与 \(AD\) 所成的两个角互为补角，因此
\[
\angle BAD+\angle CAD=180^{\circ}.
\]
在三角形 \(ABD\) 和三角形 \(ACD\) 中分别使用正弦定理. 它们的外接圆都是原三角形 \(ABC\) 的外接圆，所以
\[
\begin{cases}
\frac{BD}{\sin\angle BAD}=2R,\\
\frac{CD}{\sin\angle CAD}=2R.
\end{cases}
\]
又因为
\[
\sin\angle BAD=\sin\left(180^{\circ}-\angle CAD\right)=\sin\angle CAD,
\]
所以 \(BD=2R\sin\angle BAD=2R\sin\angle CAD=CD\). 这就证明了 \(BD=CD\).
\end{solution}

\begin{problem}
设三角形的边长为 \(a = 4\)，\(b = 5\)，\(c = 6\)，其对角依次为 \(A\)，\(B\)，\(C\).

\begin{enumerate}
    \item 求 \(\cos C\).
    \item 求 \(\sin C\)，\(\sin B\)，\(\sin A\).
    \item 问 \(A\)，\(B\)，\(C\) 三个角各为锐角或钝角？
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
    \item 由余弦定理，\(c^{2}=a^{2}+b^{2}-2ab\cos C\). 代入 \(a=4\)，\(b=5\)，\(c=6\)，得
    \[
    \cos C=\frac{a^{2}+b^{2}-c^{2}}{2ab}
    =\frac{4^{2}+5^{2}-6^{2}}{2\times4\times5}
    =\frac{5}{40}=\frac{1}{8}.
    \]

    \item 因为 \(C\) 是三角形的内角，所以 \(0<C<180^{\circ}\)，从而 \(\sin C>0\). 由同角平方关系，
    \[
    \sin C=\sqrt{1-\cos^{2}C}
    =\sqrt{1-\frac{1}{64}}
    =\frac{3\sqrt{7}}{8}.
    \]
    由正弦定理，\(\frac{a}{\sin A}=\frac{b}{\sin B}=\frac{c}{\sin C}\). 因此
    \[
    \sin B=\frac{b}{c}\sin C=\frac{5}{6}\cdot\frac{3\sqrt{7}}{8}=\frac{5\sqrt{7}}{16}.
    \]
    同理，
    \[
    \sin A=\frac{a}{c}\sin C=\frac{4}{6}\cdot\frac{3\sqrt{7}}{8}=\frac{\sqrt{7}}{4}.
    \]

    \item 因为 \(c=6>a=4\) 且 \(c=6>b=5\)，所以 \(C\) 是最大角. 又 \(\cos C=\frac18>0\)，故 \(C\) 为锐角. \(A\)、\(B\) 都小于最大角 \(C\)，所以 \(A\)、\(B\) 也都是锐角. 因而 \(A\)、\(B\)、\(C\) 皆为锐角.
\end{enumerate}
\end{solution}

\begin{problem}
一椭圆通过 \(\left( 2,3 \right)\) 及 \(\left( -1,4 \right)\) 两点，中心为原点，长短轴重合于坐标轴，试求其长短轴及焦点.
\end{problem}

\begin{solution}
设椭圆方程为
\[
\frac{x^{2}}{A}+\frac{y^{2}}{B}=1,
\]
其中 \(A>0\)，\(B>0\)，暂不预设长轴所在的坐标轴. 将点 \((2,3)\) 和 \((-1,4)\) 代入，得到
\[
    \begin{cases}
    \frac{4}{A}+\frac{9}{B}=1,\\
    \frac{1}{A}+\frac{16}{B}=1.
    \end{cases}
\]
令 \(u=\frac{1}{A}\)，\(v=\frac{1}{B}\)，则
\[
\begin{cases}
4u+9v=1,\\
u+16v=1.
\end{cases}
\]
由第二式得 \(u=1-16v\)，代入第一式，得 \(4(1-16v)+9v=1\)，即 \(-55v=-3\)，所以 \(v=\frac{3}{55}\)，\(u=\frac{7}{55}\). 因而 \(A=\frac{55}{7}\)，\(B=\frac{55}{3}\)，椭圆方程为
\[
\frac{7x^{2}}{55}+\frac{3y^{2}}{55}=1.
\]
因为 \(\frac{55}{3}>\frac{55}{7}\)，所以长轴在 \(y\) 轴上. 设半长轴为 \(a\)，半短轴为 \(b\). 于是 \(a^{2}=\frac{55}{3}\)，\(b^{2}=\frac{55}{7}\). 长轴长为 \(2a=2\sqrt{\frac{55}{3}}=\frac{2\sqrt{165}}{3}\)，短轴长为 \(2b=2\sqrt{\frac{55}{7}}=\frac{2\sqrt{385}}{7}\).

椭圆的焦距参数满足 \(c^{2}=a^{2}-b^{2}\)，故
\[
c^{2}=\frac{55}{3}-\frac{55}{7}=\frac{220}{21}.
\]
所以 \(c=\sqrt{\frac{220}{21}}=\frac{2\sqrt{1155}}{21}\).
由于焦点在 \(y\) 轴上，两个焦点为 \(\left(0,\frac{2\sqrt{1155}}{21}\right)\) 和 \(\left(0,-\frac{2\sqrt{1155}}{21}\right)\)，即 \(\left(0,\pm\frac{2\sqrt{1155}}{21}\right)\).
\end{solution}
