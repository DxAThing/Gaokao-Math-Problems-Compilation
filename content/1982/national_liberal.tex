\chapter{1982年全国卷（文）}

\section{填空题}

\begin{problem}

填表：

\begin{center}
\renewcommand{\arraystretch}{1.15}
\setlength{\tabcolsep}{4pt}
\begin{tabular}{|c|c|c|}
\hline
\makebox[0.8cm]{} & \makebox[3.0cm]{函数} & \makebox[7.5cm]{使函数有意义的\(x\)的实数范围} \\
\hline
1 & \(y=\sqrt{-x^2}\) &\fillinblank{} \\
\hline
2 & \(y=\sqrt{(-x)^2}\) &\fillinblank{} \\
\hline
3 & \(y=10^{\lg x}\) &\fillinblank{} \\
\hline
4 & \(y=\lg 10^x\) &\fillinblank{} \\
\hline
\end{tabular}
\end{center}
\end{problem}

\begin{answer}
\(\{0\}\)，\(\R\)，\(x>0\)，\(\R\).
\end{answer}

\begin{solution}
逐行考察使函数有意义所需的条件：

\begin{enumerate}
\item 根式要求 \(-x^2\geqslant 0\)，即 \(x^2\leqslant 0\)，所以 \(x=0\)，定义域为 \(\{0\}\).
\item 因为 \((-x)^2\geqslant 0\) 对任意实数 \(x\) 都成立，所以定义域为 \(\R\).
\item 对数 \(\lg x\) 要求 \(x>0\)，而 \(10^{\lg x}\) 对每个这样的 \(x\) 都有意义，所以定义域为 \((0,+\infty)\).
\item 对任意实数 \(x\)，都有 \(10^x>0\)，因此 \(\lg 10^x\) 对任意实数 \(x\) 都有意义，定义域为 \(\R\).
\end{enumerate}
\end{solution}

\section{解答题}

\begin{problem}

求 \((-1+\i)^{20}\) 展开式中第 \(15\) 项的数值.
\end{problem}

\begin{solution}
二项式展开式的第 \(r+1\) 项为：
\[
T_{r+1}=\mathrm C_{20}^{r}(-1)^{20-r}\i^r
\]
第 \(15\) 项对应 \(r=14\)，故：
\[
T_{15}=\mathrm C_{20}^{14}(-1)^6\i^{14}=\mathrm C_{20}^{14}\i^{12}\i^2=-\mathrm C_{20}^{14}=-\mathrm C_{20}^{6}=-38760
\]
所以第 \(15\) 项的数值为 \(-38760\).
\end{solution}

\begin{problem}

在平面直角坐标系内，表中的方程表示什么曲线？并画出它们的图形：

\begin{center}
\renewcommand{\arraystretch}{1.15}
\setlength{\tabcolsep}{4pt}
\begin{tabular}{|c|c|c|c|}
\hline
\makebox[0.8cm]{} & \makebox[3.0cm]{方程} & \makebox[3.0cm]{曲线名称} & \makebox[5.8cm]{图形} \\
\hline
\parbox[c][4.2cm][c]{0.8cm}{\centering 1} &
\parbox[c][4.2cm][c]{3.0cm}{\centering \(4x^2+y^2=4\)} &
\parbox[c][4.2cm][c]{3.0cm}{} &
\parbox[c][4.2cm][c]{5.8cm}{} \\
\hline
\parbox[c][4.2cm][c]{0.8cm}{\centering 2} &
\parbox[c][4.2cm][c]{3.0cm}{\centering \(x-3=0\)} &
\parbox[c][4.2cm][c]{3.0cm}{} &
\parbox[c][4.2cm][c]{5.8cm}{} \\
\hline
\end{tabular}
\end{center}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 将方程化为：
\[
 x^2+\frac{y^2}{4}=1
\]
这是中心在原点、长半轴为 \(2\) 且沿 \(y\) 轴、短半轴为 \(1\) 且沿 \(x\) 轴的椭圆. 它与坐标轴的交点为 \((\pm1,0)\) 和 \((0,\pm2)\)，取这四点并顺次作关于两坐标轴对称的椭圆，即得所求图形.
\item 方程等价于 \(x=3\)，所以它表示一条经过点 \((3,0)\) 且平行于 \(y\) 轴的直线. 作出 \(x=3\) 即得所求图形.
\end{enumerate}
\end{solution}

\begin{problem}

已知 \(x-y=\frac{1}{2}\)，\(x^2+y^2=1\)，求 \(x^2-y^2\) 的值.
\end{problem}

\begin{solution}
由 \(x-y=\frac12\) 得：
\[
 (x-y)^2=x^2-2xy+y^2=\frac14
\]
结合 \(x^2+y^2=1\)，可得：
\[
 2xy=1-\frac14=\frac34
\]
于是：
\[
 (x+y)^2=x^2+2xy+y^2=1+\frac34=\frac74
\]
所以 \(x+y=\pm\frac{\sqrt7}{2}\). 因而：
\[
 x^2-y^2=(x-y)(x+y)=\frac12\left(\pm\frac{\sqrt7}{2}\right)=\pm\frac{\sqrt7}{4}
\]
两种符号分别对应实数解 \(\displaystyle (x,y)=\left(\frac{1\pm\sqrt7}{4},\frac{\pm\sqrt7-1}{4}\right)\)，其中两个 \(\pm\) 取同号，因此两个值都能取到.
\end{solution}

\begin{problem}

以墙为一边，用篱笆围成长方形的场地，并用平行于一边的篱笆隔开（如图）. 已知篱笆的总长为定值 \(L\)，这块场地的长和宽各为多少时场地的面积最大？最大面积是多少？

\FigureLayoutDeclare{img/1982/national_liberal/q5_fig1.png}{4.5cm}{16bp 18bp 15bp 21bp}
\bitmapfigure[max width=0.42\linewidth]{img/1982/national_liberal/q5_fig1.png}
\end{problem}

\begin{solution}
设垂直于墙的宽为 \(x\)，则两条侧边和中间隔栏共需 \(3x\) 的篱笆，沿墙方向的长为 \(L-3x\). 因为长、宽均为正数，所以 \(0<x<\frac{L}{3}\).

场地面积为：
\[
 S=x(L-3x)=-3\left(x-\frac{L}{6}\right)^2+\frac{L^2}{12}
\]
因此当且仅当 \(x=\frac{L}{6}\) 时面积最大. 此时场地的长为 \(L-3x=L-3\cdot\frac{L}{6}=\frac{L}{2}\)，最大面积为 \(\frac{L^2}{12}\).
\end{solution}

\begin{problem}

已知正方体 \(ABCD-A_1B_1C_1D_1\) 的棱长为 \(a\).

\begin{enumerate}
\item 用平面 \(A_1BC_1\) 截去一角后，求剩余部分的体积；
\item 求 \(A_1B\) 和 \(B_1C\) 所成的角.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 取坐标 \(A=(0,0,0)\)、\(B=(a,0,0)\)、\(C=(a,a,0)\)、\(D=(0,a,0)\)，以及 \(A_1=(0,0,a)\)、\(B_1=(a,0,a)\)、\(C_1=(a,a,a)\)、\(D_1=(0,a,a)\).
由 \(\overrightarrow{A_1B}=(a,0,-a)\) 和 \(\overrightarrow{A_1C_1}=(a,a,0)\) 的叉积可取平面法向量 \((1,-1,1)\)，所以平面 \(A_1BC_1\) 的方程为 \(x-y+z=a\). 代入 \(B_1\) 得左端为 \(2a>a\)，而 \(A\)，\(C\)，\(D\)，\(D_1\) 代入得值均小于 \(a\)，因此截去的部分是四面体 \(B_1A_1BC_1\). 从 \(B_1\) 出发的三条棱向量可取为 \(\overrightarrow{B_1A_1}=(-a,0,0)\)、\(\overrightarrow{B_1B}=(0,0,-a)\)、\(\overrightarrow{B_1C_1}=(0,a,0)\).
所以截去部分的体积为：
\[
 V_{\mathrm{cut}}=\frac16\left|\begin{vmatrix}-a&0&0\\0&0&-a\\0&a&0\end{vmatrix}\right|=\frac{a^3}{6}
\]
正方体体积为 \(a^3\)，故剩余部分的体积为：
\[
 V=a^3-\frac{a^3}{6}=\frac56a^3
\]
\item 取直线 \(A_1B\) 和 \(B_1C\) 的方向向量分别为 \(\overrightarrow{A_1B}=(a,0,-a)\) 和 \(\overrightarrow{B_1C}=(0,a,-a)\).
它们的内积为 \(a^2\)，两向量的长度均为 \(a\sqrt2\). 设两直线所成的锐角为 \(\varphi\)，则：
\[
 \cos\varphi=\frac{|a^2|}{a\sqrt2\cdot a\sqrt2}=\frac12
\]
由于 \(0^\circ\leqslant\varphi\leqslant90^\circ\)，得 \(\varphi=60^\circ\).
\end{enumerate}
\end{solution}

\begin{problem}

已知定点 \(A\)，\(B\) 且 \(AB=2a\)，如果动点 \(P\) 到点 \(A\) 的距离和到点 \(B\) 的距离之比为 \(2:1\)，求点 \(P\) 的轨迹方程，并说明它表示什么曲线.
\end{problem}

\begin{solution}
取 \(AB\) 所在直线为横轴，从 \(A\) 到 \(B\) 为正方向，以 \(AB\) 中点为原点，纵轴取为过原点且垂直于 \(AB\) 的直线. 于是
\(A=(-a,0)\)、\(B=(a,0)\).
设动点 \(P=(x,y)\)，由于比值有定义，\(P\neq B\). 由题意：
\[
 \frac{PA}{PB}=2
\]
即：
\[
 \frac{\sqrt{(x+a)^2+y^2}}{\sqrt{(x-a)^2+y^2}}=2
\]
两边均为非负数，平方得：
\[
 (x+a)^2+y^2=4\left((x-a)^2+y^2\right)
\]
展开并移项，得到轨迹方程：
\[
 3x^2-10ax+3y^2+3a^2=0
\]
配方可得：
\[
 \left(x-\frac{5a}{3}\right)^2+y^2=\left(\frac{4a}{3}\right)^2
\]
所以轨迹是圆心为 \(\left(\frac{5a}{3},0\right)\)、半径为 \(\frac{4a}{3}\) 的圆.
\end{solution}

\begin{problem}

求 \(\tan 9^\circ+\cot 117^\circ-\tan 243^\circ-\cot 351^\circ\) 的值.
\end{problem}

\begin{solution}
利用诱导公式，\(\cot117^\circ=-\tan27^\circ\)、\(\tan243^\circ=\tan63^\circ\)、\(\cot351^\circ=-\cot9^\circ\).
又因为 \(\tan63^\circ=\cot27^\circ\)，所以原式为：
\[
 \tan9^\circ+\cot9^\circ-\tan27^\circ-\cot27^\circ
\]
对任意使式子有意义的角 \(\alpha\)，有：
\[
 \tan\alpha+\cot\alpha=\frac{1}{\sin\alpha\cos\alpha}=\frac{2}{\sin2\alpha}
\]
故原式：
\[
 =\frac{2}{\sin18^\circ}-\frac{2}{\sin54^\circ}
 =2\cdot\frac{\sin54^\circ-\sin18^\circ}{\sin18^\circ\sin54^\circ}
\]
又因为：
\[
 \sin54^\circ-\sin18^\circ=2\cos36^\circ\sin18^\circ=2\sin54^\circ\sin18^\circ
\]
所以原式等于 \(2\cdot2=4\).
\end{solution}

\begin{problem}

如图，已知 \(\triangle AOB\) 中，\(OA=b\)，\(OB=a\)，\(\angle AOB=\theta\)（\(a\geqslant b\)，\(\theta\) 是锐角）. 作 \(AB_1\perp OB\)，\(B_1A_1\parallel BA\)；再作 \(A_1B_2\perp OB\)，\(B_2A_2\parallel BA\)；如此无限连续作下去. 设 \(\triangle ABB_1\)，\(\triangle A_1B_1B_2\)，\(\cdots\) 的面积分别为 \(S_1\)，\(S_2\)，\(\cdots\)，求无穷数列 \(S_1\)，\(S_2\)，\(\cdots\) 的和.

\bitmapfigure[max width=0.42\linewidth]{img/1982/national_liberal/q9_fig1.png}
\end{problem}

\begin{solution}
令 \(A_0=A\)，\(B_0=B\)，并令 \(S_{n+1}\) 表示三角形 \(A_nB_nB_{n+1}\) 的面积. 由 \(AB_1\perp OB\)，有 \(AB_1=b\sin\theta\)、\(OB_1=b\cos\theta\)，从而 \(BB_1=a-b\cos\theta\)、\(S_1=\frac12b\sin\theta\left(a-b\cos\theta\right)\).

由每次作出的平行线，三角形 \(OA_nB_n\) 与三角形 \(OAB\) 相似，所以 \(\frac{OA_n}{OB_n}=\frac{b}{a}\). 设 \(r=\frac{b}{a}\cos\theta\).
在三角形 \(OA_nB_n\) 中，\(B_{n+1}\) 是 \(A_n\) 在 \(OB\) 上的垂足，所以 \(OB_{n+1}=OA_n\cos\theta=r\,OB_n\).
因此相邻两个相似三角形的对应边长之比为 \(r\)，即 \(\frac{A_nB_n}{A_{n-1}B_{n-1}}=r\).
同时，三角形 \(A_nB_nB_{n+1}\) 与三角形 \(A_{n-1}B_{n-1}B_n\) 的对应高和对应底边都按比例 \(r\) 缩小，所以 \(\frac{S_{n+1}}{S_n}=r^2=\left(\frac{b}{a}\cos\theta\right)^2\).
因为 \(a\geqslant b>0\)，且 \(\theta\) 为锐角，所以 \(0<r<1\)，并且 \(a-b\cos\theta>0\)，面积数列是无穷等比数列. 于是：
\[
\sum_{n=1}^{\infty}S_n=\frac{S_1}{1-r^2}
=\frac{\frac12b\sin\theta\left(a-b\cos\theta\right)}{1-\frac{b^2}{a^2}\cos^2\theta}
\]
又因为：
\[
\sum_{n=1}^{\infty}S_n
=\frac{\frac12b\sin\theta\left(a-b\cos\theta\right)}{\frac{\left(a-b\cos\theta\right)\left(a+b\cos\theta\right)}{a^2}}
=\frac{a^2b\sin\theta}{2\left(a+b\cos\theta\right)}
\]
\end{solution}
