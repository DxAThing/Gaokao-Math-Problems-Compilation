\begin{problem}

\begin{enumerate}
\item 求该地 1 位车主至少购买甲，乙两种保险中的 1 种的概率；
\item X 表示该地的 100 位车主中，甲，乙两种保险都不购买的车主数. 求 X 的期望.
\end{enumerate}
\end{problem}

\begin{problem}
设数列 \(\{a_{n}\}\) 满足 \(a_1 = 0\) 且 \(\frac{1}{1 - a_{n + 1}} -\frac{1}{1 - a_n} = 1\).
\begin{enumerate}
\item 求 \( \{a_{n}\} \) 的通项公式；
\item 设 \( b_n = \frac{1 - \sqrt{a_{n+1}}}{n} \)，记 \( S_n = \sum_{k=1}^{n} b_k \)，证明：\( S_n < 1 \).
\end{enumerate}
\end{problem}
