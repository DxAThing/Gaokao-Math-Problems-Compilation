\section{单选题}

\begin{problem}
若 \(z = \frac{1 + 2\i}{\i}\)，则复数 \(z =\)（\quad）

\choices
    {\(-2 - \i\)}
    {\(-2 + \i\)}
    {\(2 - \i\)}
    {\(2 + \i\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
因为 \(\dfrac{1}{\i}=-\i\)，所以
\(z=\dfrac{1+2\i}{\i}=(1+2\i)(-\i)=2-\i\)，故选 C.
\end{solution}

\begin{problem}
若集合 \(A = \{x \mid -1 \leqslant 2x + 1 \leqslant 3\}\)，\(B = \left\{x \left| \frac{x - 2}{x} \leqslant 0 \right.\right\}\)，则 \(A \cap B =\)（\quad）

\choices
    {\(\{x\mid -1\leqslant x < 0\}\)}
    {\(\{x\mid 0 < x\leqslant 1\}\)}
    {\(\{x\mid 0\leqslant x\leqslant 2\}\)}
    {\(\{x\mid 0\leqslant x\leqslant 1\}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
由 \(-1\leqslant 2x+1\leqslant 3\)，得 \(-1\leqslant x\leqslant 1\)，即
\(A=[-1,1]\). 因为 \(x\neq 0\)，且
\(\dfrac{x-2}{x}\leqslant 0\) 的解集为 \(B=(0,2]\)，所以
\(A\cap B=(0,1]\)，故选 B.
\end{solution}

\begin{problem}
若 \( f(x) = \frac{1}{\sqrt{\log_{\frac{1}{2}}(2x + 1)}} \)，则 \( f(x) \) 定义域为（\quad）

\choices
    {\(\left(-\frac{1}{2},0\right)\)}
    {\(\left(-\frac{1}{2},0\right]\)}
    {\(\left(-\frac{1}{2}, +\infty\right)\)}
    {\((0, +\infty)\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
要使 \(f(x)\) 有意义，需满足 \(2x+1>0\)，且
\(\log_{\frac{1}{2}}(2x+1)>0\)，因为分母不能为零. 由于底数
\(\frac{1}{2}\) 小于 1，后一个不等式等价于 \(0<2x+1<1\)，所以
\(-\frac{1}{2}<x<0\)，故选 A.
\end{solution}

\begin{problem}
若 \( f(x) = x^2 - 2x - 4\ln x \)，则 \( f'(x) > 0 \) 的解集为（\quad）

\choices
    {\((0, +\infty)\)}
    {\((-1,0)\cup (2, + \infty)\)}
    {\((2, +\infty)\)}
    {\((-1,0)\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
函数的定义域为 \((0,+\infty)\)，且
\[
f'(x)=2x-2-\frac{4}{x}=\frac{2(x-2)(x+1)}{x}.
\]
在 \(x>0\) 时，\(f'(x)>0\) 等价于 \(x>2\)，所以解集为
\((2,+\infty)\)，故选 C.
\end{solution}

\begin{problem}
已知数列 \(\{a_{n}\}\) 的前 \(n\) 项和 \(S_{n}\) 满足： \(S_{n} + S_{m} = S_{n + m}\)，且 \(a_{1} = 1\)，那么 \(a_{10} =\)（\quad）

\choices
    {1}
    {9}
    {10}
    {55}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
取 \(m=1\)，由题意得 \(S_{n+1}=S_n+S_1=S_n+1\)，从而
\(a_{n+1}=S_{n+1}-S_n=1\). 因此 \(a_{10}=1\)，故选 A.
\end{solution}

\begin{problem}
变量 \(X\) 与 \(Y\) 相对应的一组数据为 \((10, 1)\)，\((11.3, 2)\)，\((11.8, 3)\)，\((12.5, 4)\)，\((13, 5)\)；变量 \(U\) 与 \(V\) 相对应的一组数据为 \((10, 5)\)，\((11.3, 4)\)，\((11.8, 3)\)，\((12.5, 2)\)，\((13, 1)\). \(r_{1}\) 表示变量 \(Y\) 与 \(X\) 之间的线性相关系数；\(r_{2}\) 表示变量 \(V\) 与 \(U\) 之间的线性相关系数，则（\quad）

\choices
    {\(r_2 < r_1 < 0\)}
    {\(0 < r_2 < r_1\)}
    {\(r_2 < 0 < r_1\)}
    {\(r_2 = r_1\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
随着 \(X\) 增大，\(Y\) 严格增大，所以 \(r_1>0\). 第二组数据中
\(V=6-Y\)，变量 \(U\) 与 \(X\) 的取值相同且顺序一致，因此
\(V\) 与 \(U\) 的线性相关系数为 \(-r_1<0\). 故 \(r_2<0<r_1\)，选 C.
\end{solution}

\begin{problem}
观察下列各式： \(5^{5} = 3125\)，\(5^{6} = 15625\)，\(5^{7} = 78125\)，\(\cdots\)，则 \(5^{2011}\) 的末四位数字为（\quad）

\choices
    {3125}
    {5625}
    {0625}
    {8125}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
对任意 \(n\geqslant 4\)，\(5^n\) 的末四位数字只由模 \(10000\) 的余数决定. 直接计算得
\(5^4\equiv 0625\)，再依次乘以 5，末四位按
\(0625\)，\(3125\)，\(5625\)，\(8125\) 循环出现，循环长度为 4.
因为 \(2011=4\times 502+3\)，所以 \(5^{2011}\) 的末四位与 \(5^7\) 相同，均为 \(8125\)，故选 D.
\end{solution}

\begin{problem}
已知 \(\alpha_{1}\)，\(\alpha_{2}\)，\(\alpha_{3}\) 是三个相互平行的平面，平面 \(\alpha_{1}\)，\(\alpha_{2}\) 之间的距离为 \(d_{1}\)，平面 \(\alpha_{2}\)，\(\alpha_{3}\) 之间的距离为 \(d_{2}\). 直线 \(l\) 与 \(\alpha_{1}\)，\(\alpha_{2}\)，\(\alpha_{3}\) 分别交于 \(P_{1}\)，\(P_{2}\)，\(P_{3}\). 那么“\(P_{1}P_{2} = P_{2}P_{3}\)”是“\(d_{1} = d_{2}\)”的（\quad）

\choices
    {充分不必要条件}
    {必要不充分条件}
    {充分必要条件}
    {既不充分也不必要条件}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
设直线与相邻两平面所成的夹角为 \(\theta\). 由于三个平面相互平行，沿直线方向的距离投影满足
\[
\frac{P_1P_2}{P_2P_3}=\frac{d_1}{d_2},
\]
其中比例因子由同一个 \(\theta\) 决定. 因而
\(P_1P_2=P_2P_3\) 当且仅当 \(d_1=d_2\)，所以这是充分必要条件，选 C.
\end{solution}

\begin{problem}
若曲线 \(C_{1}: x^{2} + y^{2} - 2x = 0\) 与曲线 \(C_{2}: y (y - mx - m) = 0\) 有四个不同的交点，则实数 \(m\) 的取值范围是（\quad）

\choices
    {\(\left(-\frac{\sqrt{3}}{3}, \frac{\sqrt{3}}{3}\right)\)}
    {\(\left(-\frac{\sqrt{3}}{3}, 0\right) \cup \left(0, \frac{\sqrt{3}}{3}\right)\)}
    {\(\left[-\frac{\sqrt{3}}{3}, \frac{\sqrt{3}}{3}\right]\)}
    {\(\left(-\infty, -\frac{\sqrt{3}}{3}\right) \cup \left(\frac{\sqrt{3}}{3}, +\infty\right)\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
曲线 \(C_2\) 是直线 \(y=0\) 与直线 \(y=m(x+1)\) 的并集. 直线
\(y=0\) 与圆 \(C_1\) 恒有两个交点 \((0,0)\)，\((2,0)\). 要有四个不同交点，另一条直线必须与圆有两个交点，且不能与 \(y=0\) 重合.

直线 \(y=m(x+1)\) 写成 \(mx-y+m=0\). 圆心为 \((1,0)\)，半径为 1，所以直线与圆有两个交点的条件是
\[
\frac{|2m|}{\sqrt{m^2+1}}<1,
\]
即 \(3m^2<1\). 同时 \(m\neq 0\)，故
\(m\in\left(-\frac{\sqrt{3}}{3},0\right)\cup\left(0,\frac{\sqrt{3}}{3}\right)\)，选 B.
\end{solution}

如图，一个直径为 1 的小圆沿着直径为 2 的大圆内壁逆时针方向滚动，\(M\) 和 \(N\) 是小圆的一条固定直径的两个端点. 那么，当小圆这样滚过大圆内壁一周，点 \(M\)，\(N\) 在大圆内所绘出的图形大致是（\quad）

\section{填空题}

\begin{problem}
已知 \(|a| = |b| = 2\)，\((a + 2b) \cdot (a - b) = -2\)，则 \(a\) 与 \(b\) 的夹角为\fillinblank{}.
\end{problem}

\begin{answer}
\(\frac{\pi}{3}\)
\end{answer}

\begin{solution}
展开数量积得
\[
(a+2b)\cdot(a-b)=|a|^2+a\cdot b-2|b|^2=4+a\cdot b-8=a\cdot b-4.
\]
因此 \(a\cdot b=2\). 设 \(a\) 与 \(b\) 的夹角为 \(\theta\)，则
\(a\cdot b=|a||b|\cos\theta=4\cos\theta\)，所以
\(\cos\theta=\frac{1}{2}\). 由于 \(0\leqslant\theta\leqslant\pi\)，得
\(\theta=\frac{\pi}{3}\).
\end{solution}

\begin{problem}
小波通过做游戏的方式来确定周末活动，他随机地往单位圆内投掷一点，若此点到圆心的距离大于 \(\frac{1}{2}\)，则周末去看电影；若此点到圆心的距离小于 \(\frac{1}{4}\)，则去打篮球；否则，在家看书. 则小波周末不在家看书的概率为\fillinblank{}.
\end{problem}

\begin{answer}
\(\frac{13}{16}\)
\end{answer}

\begin{solution}
单位圆内各点等可能，样本空间面积为 \(\pi\). 不在家看书包括到圆心距离大于
\(\frac{1}{2}\) 和小于 \(\frac{1}{4}\) 两个互不相交的区域，故所求概率为
\[
\frac{\pi-\pi\left(\frac{1}{2}\right)^2+\pi\left(\frac{1}{4}\right)^2}{\pi}
=1-\frac{1}{4}+\frac{1}{16}=\frac{13}{16}.
\]
边界圆周的面积为零，不影响概率.
\end{solution}

\begin{problem}
下图是某算法程序框图，则程序运行后输出的结果是\fillinblank{}.

\texfigure[max width=0.96\linewidth]{img_repaint/2011/jiangxi_science/q13_fig1.tex}

\begin{problem}
若椭圆 \(\frac{x^2}{a^2} + \frac{y^2}{b^2} = 1\) 的焦点在 \(x\) 轴上，过点 \(\left(1, \frac{1}{2}\right)\) 作圆 \(x^2 + y^2 = 1\) 的切线. 切点分别为 \(A\)，\(B\)，直线 \(AB\) 恰好经过椭圆的右焦点和上顶点，则椭圆方程是\fillinblank{}.
\end{problem}

\begin{answer}
\(\frac{x^2}{5} + \frac{y^2}{4} = 1\)
\end{answer}

\begin{solution}
从点 \(P\left(1,\frac{1}{2}\right)\) 向单位圆作两条切线，切点弦 \(AB\) 是该圆关于点 \(P\) 的极线，因此其方程为
\[
x+\frac{1}{2}y=1,
\]
即 \(2x+y=2\). 设椭圆的焦距的一半为 \(c\)，则右焦点为 \((c,0)\)，上顶点为 \((0,b)\). 因为这两点都在直线 \(2x+y=2\) 上，所以
\(2c=2\)，\(b=2\)，即 \(c=1\)，\(b^2=4\). 又
\(c^2=a^2-b^2\)，故 \(a^2=c^2+b^2=5\). 因此椭圆方程为
\(\dfrac{x^2}{5}+\dfrac{y^2}{4}=1\).
\end{solution}

\begin{problem}[A]
若曲线的极坐标方程为 \(\rho = 2 \sin \theta + 4 \cos \theta\)，以极点为原点，极轴为 \(x\) 轴正半轴建立直角坐标系，则该曲线的直角坐标方程为\fillinblank{}.
\end{problem}

\begin{answer}
\(x^2 + y^2 - 4x - 2y = 0\)
\end{answer}

\begin{solution}
由极坐标关系 \(x=\rho\cos\theta\)，\(y=\rho\sin\theta\)，将原方程乘以 \(\rho\) 得
\[
\rho^2=2\rho\sin\theta+4\rho\cos\theta=2y+4x.
\]
因为 \(\rho^2=x^2+y^2\)，所以直角坐标方程为
\(x^2+y^2-4x-2y=0\).
\end{solution}

对于实数 \(x\)，\(y\)，若 \(|x - 1| \leqslant 1\)，\(|y - 2| \leqslant 1\)，则 \(|x - 2y + 1|\) 的最大值为\fillinblank{}.

\begin{problem}
某饮料公司招聘一名员工，现对其进行一项测试，以便确定工资级别. 公司准备了两种不同的饮料共 \(8\text{ 杯}\)，其颜色完全相同，并且其中 \(4\text{ 杯}\) 为 A 饮料，另外 \(4\text{ 杯}\) 为 B 饮料，公司要求此员工一一品尝后，从 \(8\text{ 杯}\) 饮料中选出 \(4\text{ 杯}\) A 饮料. 若 \(4\text{ 杯}\) 都选对，则月工资定为 \(3500\text{ 元}\)；若 \(4\text{ 杯}\) 选对 \(3\text{ 杯}\)，则月工资定为 \(2800\text{ 元}\)；否则月工资定为 \(2100\text{ 元}\). 令 \(X\) 表示此人选对 A 饮料的杯数. 假设此人对 A 和 B 两种饮料没有鉴别能力.

\begin{enumerate}
\item 求 \(X\) 的分布列；
\item 求此员工月工资的期望.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 从 \(8\text{ 杯}\) 中选出 \(4\text{ 杯}\) 的结果等可能，共有 \(\mathrm C_8^4=70\) 种. 当选对 \(k\text{ 杯}\) 时，必须从 \(4\text{ 杯}\) A 饮料中选出 \(k\text{ 杯}\)，并从 \(4\text{ 杯}\) B 饮料中误选 \((4-k)\text{ 杯}\)，所以
\[
P(X=k)=\frac{\mathrm C_4^k\mathrm C_4^{4-k}}{\mathrm C_8^4},\qquad k=0,1,2,3,4.
\]
因此 \(X\) 的分布列为
\begin{center}
\begin{tabular}{c|ccccc}
\(X\) & \(0\) & \(1\) & \(2\) & \(3\) & \(4\) \\
\hline
\(P\) & \(\frac{1}{70}\) & \(\frac{8}{35}\) & \(\frac{18}{35}\) & \(\frac{8}{35}\) & \(\frac{1}{70}\)
\end{tabular}
\end{center}

\item 设月工资为随机变量 \(Y\). 当 \(X=4\) 时 \(Y=3500\text{ 元}\)，当 \(X=3\) 时 \(Y=2800\text{ 元}\)，其余情形 \(Y=2100\text{ 元}\). 因而
\[
E(Y)=3500\text{ 元}\cdot\frac{1}{70}+2800\text{ 元}\cdot\frac{8}{35}+2100\text{ 元}\left(1-\frac{1}{70}-\frac{8}{35}\right).
\]
再计算得 \(E(Y)=50\text{ 元}+640\text{ 元}+1590\text{ 元}=2280\text{ 元}\).
所以此员工月工资的期望为 \(2280\text{ 元}\).
\end{enumerate}
\end{solution}

\begin{problem}
在 \(\triangle ABC\) 中，角 \(A\)，\(B\)，\(C\) 的对边分别是 \(a\)，\(b\)，\(c\)，已知 \(\sin C + \cos C = 1 - \sin \frac{C}{2}\).
\begin{enumerate}
\item 求 \(\sin C\) 的值；
\item 若 \(a^2 + b^2 = 4(a + b) - 8\)，求边 \(c\) 的值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 令 \(t=\frac{C}{2}\)，则 \(0<t<\frac{\pi}{2}\)，从而 \(\sin t>0\). 将
\(\sin C=2\sin t\cos t\)，\(\cos C=1-2\sin^2t\) 代入已知等式，得
\[
2\sin t\cos t+1-2\sin^2t=1-\sin t,
\]
即 \(2\sin t-2\cos t=1\). 因此
\(\sin t-\cos t=\frac{1}{2}\). 又
\[
1=\sin^2t+\cos^2t=(\sin t-\cos t)^2+2\sin t\cos t,
\]
所以 \(\sin t\cos t=\frac{3}{8}\)，进而
\(\sin C=2\sin t\cos t=\frac{3}{4}\).

\item 由题设
\[
(a-2)^2+(b-2)^2=a^2+b^2-4(a+b)+8=0,
\]
故 \(a=b=2\). 由上面的关系及
\((\sin t+\cos t)^2=1+2\sin t\cos t=\frac{7}{4}\)，得
\(\cos C=\cos^2t-\sin^2t=(\cos t-\sin t)(\cos t+\sin t)=-\frac{\sqrt7}{4}\).
由余弦定理，
\[
c^2=a^2+b^2-2ab\cos C=8-8\left(-\frac{\sqrt7}{4}\right)=8+2\sqrt7=(1+\sqrt7)^2.
\]
因为 \(c>0\)，所以 \(c=1+\sqrt7\).
\end{enumerate}
\end{solution}

\begin{problem}
已知两个等比数列 \(\{a_{n}\}\)，\(\{b_{n}\}\)，满足 \(a_{1} = a\)（\(a > 0\)），\(b_{1} - a_{1} = 1\)，\(b_{2} - a_{2} = 2\)，\(b_{3} - a_{3} = 3\).
\begin{enumerate}
\item 若 \(a = 1\)，求数列 \(\{a_{n}\}\) 的通项公式；
\item 若数列 \(\{a_{n}\}\) 唯一，求 \(a\) 的值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 当 \(a=1\) 时，\(b_1=2\). 设 \(\{a_n\}\) 的公比为 \(q\)，则
\(a_2=q\)，\(a_3=q^2\)，又由三个差值条件得
\(b_2=q+2\)，\(b_3=q^2+3\). 因为 \(\{b_n\}\) 是等比数列，所以
\[
(q+2)^2=b_2^2=b_1b_3=2(q^2+3).
\]
整理得 \(q^2-4q+2=0\)，从而
\(q=2+\sqrt2\) 或 \(q=2-\sqrt2\). 因此
\[
a_n=(2+\sqrt2)^{n-1}\quad\text{或}\quad a_n=(2-\sqrt2)^{n-1},\qquad n\geqslant1.
\]

\item 设 \(\{a_n\}\) 的公比为 \(q\)，则
\(a_1=a\)，\(a_2=aq\)，\(a_3=aq^2\)，从而
\(b_1=a+1\)，\(b_2=aq+2\)，\(b_3=aq^2+3\). 由 \(\{b_n\}\) 为等比数列，须有
\[
(aq+2)^2=(a+1)(aq^2+3),
\]
即
\[
aq^2-4aq+3a-1=0.
\]
若 \(q\neq0\)，该方程的两个根均给出不同的 \(\{a_n\}\)，不能满足题设的唯一性. 因此必须使两个根中有一个为 \(0\). 代入上式得 \(3a-1=0\)，所以
\[
a=\frac{1}{3}.
\]
此时两个根为 \(0,4\)，按等比数列公比 \(q\neq0\) 的定义，只能取 \(q=4\)，故 \(\{a_n\}\) 确实唯一.
\end{enumerate}
\end{solution}

\begin{problem}
设 \( f(x) = -\frac{1}{3} x^3 + \frac{1}{2} x^2 + 2ax \).
\begin{enumerate}
\item 若 \( f(x) \) 在 \( \left(\frac{2}{3}, +\infty\right) \) 上存在单调递增区间，求 \(a\) 的取值范围；
\item 当 \(0 < a < 2\) 时，\( f(x) \) 在 \([1, 4]\) 上的最小值为 \(-\frac{16}{3}\)，求 \( f(x) \) 在该区间上的最大值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由
\[
f'(x)=-x^2+x+2a
\]
知在 \(\left(\frac{2}{3},+\infty\right)\) 上存在单调递增区间，当且仅当存在
\(x>\frac{2}{3}\) 使 \(-x^2+x+2a>0\)，即
\(2a>x^2-x\). 函数 \(x^2-x\) 在 \(x>\frac{2}{3}\) 上递增，且
\(\inf_{x>2/3}(x^2-x)=-\frac{2}{9}\). 当 \(a=-\frac{1}{9}\) 时，任意
\(x>\frac{2}{3}\) 都有 \(x^2-x>-\frac{2}{9}=2a\)，不能取得正导数；当
\(a>-\frac{1}{9}\) 时，取充分接近 \(\frac{2}{3}\) 的 \(x\)，即可使
\(x^2-x<2a\). 所以
\[
a>-\frac{1}{9}.
\]

\item 因为 \(0<a<2\)，所以
\(f'(1)=2a>0\)，\(f'(4)=-12+2a<0\). 方程 \(f'(x)=0\) 在 \([1,4]\) 内的根为
\(x=\frac{1+\sqrt{1+8a}}{2}\)，该点是极大值点，故最小值只能在端点取得. 又
\[
f(1)=2a+\frac{1}{6}>0,\qquad f(4)=8a-\frac{40}{3}.
\]
已知最小值为 \(-\frac{16}{3}\)，所以必有
\(8a-\frac{40}{3}=-\frac{16}{3}\)，解得 \(a=1\). 此时
\(f'(x)=-(x-2)(x+1)\)，函数在 \([1,2]\) 上递增、在 \([2,4]\) 上递减，最大值为
\[
f(2)=-\frac{8}{3}+2+4=\frac{10}{3}.
\]
所以该区间上的最大值为 \(\frac{10}{3}\).
\end{enumerate}
\end{solution}

\begin{problem}
\(P(x_0, y_0)\)（\(x_0 \neq \pm a\)）是双曲线 \(E: \frac{x^2}{a^2} - \frac{y^2}{b^2} = 1\)（\(a > 0\)，\(b > 0\)）上一点，\(M\)，\(N\) 分别是双曲线 \(E\) 的左，右顶点. 直线 \(PM\)，\(PN\) 的斜率之积为 \(\frac{1}{5}\).
\begin{enumerate}
\item 求双曲线的离心率；
\item 过双曲线 \(E\) 的右焦点且斜率为 1 的直线交双曲线于 \(A\)，\(B\) 两点，\(O\) 为坐标原点，\(C\) 为双曲线上的一点. 满足 \(\overrightarrow{OC} = \lambda \overrightarrow{OA} + \overrightarrow{OB}\)，求 \(\lambda\) 的值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 双曲线的左、右顶点分别为 \(M(-a,0)\)，\(N(a,0)\)，所以
\[
k_{PM}k_{PN}=\frac{y_0}{x_0+a}\cdot\frac{y_0}{x_0-a}
=\frac{y_0^2}{x_0^2-a^2}=\frac{1}{5}.
\]
又因 \(P\) 在双曲线上，
\(y_0^2=\frac{b^2}{a^2}(x_0^2-a^2)\)，故
\(\frac{b^2}{a^2}=\frac{1}{5}\). 设双曲线的半焦距为 \(c\)，则
\(c^2=a^2+b^2=\frac{6}{5}a^2\)，所以离心率为
\[
e=\frac{c}{a}=\sqrt{\frac{6}{5}}=\frac{\sqrt{30}}{5}.
\]

\item 由 \(b^2=\frac{a^2}{5}\)，有 \(c^2=a^2+b^2=\frac{6}{5}a^2\). 过右焦点
\(F(c,0)\) 且斜率为 1 的直线为 \(y=x-c\). 设
\(A(x_1,y_1)\)，\(B(x_2,y_2)\)，则代入双曲线方程并乘以 \(a^2\)，得
\[
x^2-5(x-c)^2=a^2,
\]
即
\(4x^2-10cx+5c^2+a^2=0\). 因此
\[
x_1+x_2=\frac{5c}{2},\qquad x_1x_2=\frac{5c^2+a^2}{4}.
\]
由于 \(y_i=x_i-c\)，所以
\[
y_1y_2=x_1x_2-c(x_1+x_2)+c^2=\frac{a^2-c^2}{4}=-\frac{a^2}{20},
\]
并且
\[
x_1x_2-5y_1y_2=\frac{7a^2}{4}+\frac{a^2}{4}=2a^2.
\]
设 \(C=(\lambda x_1+x_2,\lambda y_1+y_2)\). 因为 \(C\) 在双曲线上，使用方程
\(x^2-5y^2=a^2\) 得
\[
a^2=(\lambda x_1+x_2)^2-5(\lambda y_1+y_2)^2
=\lambda^2a^2+a^2+2\lambda(x_1x_2-5y_1y_2)
=\lambda^2a^2+a^2+4\lambda a^2.
\]
于是 \(\lambda^2+4\lambda=0\)，故
\(\lambda=0\) 或 \(\lambda=-4\).
\end{enumerate}
\end{solution}

\begin{problem}
\begin{enumerate}
    \item 如图，对于任一给定的四面体 \(A_{1}A_{2}A_{3}A_{4}\)，找出依次排列的四个相互平行的平面 \(\alpha_{1}\)，\(\alpha_{2}\)，\(\alpha_{3}\)，\(\alpha_{4}\)，使得 \(A_{i} \in \alpha_{i}\) （\(i = 1\)，\(2\)，\(3\)，\(4\)），且其中每相邻两个平面间的距离都相等；
    \item 给定依次排列的四个相互平行的平面 \(\alpha_{1}\)，\(\alpha_{2}\)，\(\alpha_{3}\)，\(\alpha_{4}\)，其中每相邻两个平面间的距离都为 1，若一个正四面体 \(A_{1}A_{2}A_{3}A_{4}\) 的四个顶点满足： \(A_{i} \in \alpha_{i}\) （\(i = 1\)，\(2\)，\(3\)，\(4\)），求该正四面体 \(A_{1}A_{2}A_{3}A_{4}\) 的体积.
\end{enumerate}

\bitmapfigure[max width=0.34\linewidth]{img/2011/jiangxi_science/q21_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 记
\(\bs d_1=\overrightarrow{A_1A_2}\)，\(\bs d_2=\overrightarrow{A_2A_3}\)，\(\bs d_3=\overrightarrow{A_3A_4}\). 取非零向量 \(\bs n\)，使得
\[
\bs n\cdot(\bs d_1-\bs d_2)=0,\qquad
\bs n\cdot(\bs d_2-\bs d_3)=0.
\]
这样的向量存在：它可以取为两个向量 \(\bs d_1-\bs d_2\)，\(\bs d_2-\bs d_3\) 的公共垂线方向；若这两个向量共线，则在其垂直平面内选取满足 \(\bs n\cdot\bs d_1\neq0\) 的方向. 后一个条件一定可以满足，否则 \(\bs n\) 将同时垂直于 \(\bs d_1\)，\(\bs d_2\)，\(\bs d_3\)，与四面体不共面矛盾.

令 \(\bs n\) 的方向使 \(\bs n\cdot\bs d_1>0\)，并定义
\[
\alpha_i:\ \bs n\cdot\overrightarrow{OX}=\bs n\cdot\overrightarrow{OA_i},\qquad i=1,2,3,4.
\]
由所取 \(\bs n\) 的性质，
\[
\bs n\cdot\bs d_1=\bs n\cdot\bs d_2=\bs n\cdot\bs d_3.
\]
所以四个平面的方程右端依次构成等差数列，且相邻平面的距离均为
\(\dfrac{\bs n\cdot\bs d_1}{|\bs n|}\). 这些平面即为所求.

\item 设正四面体的棱长为 \(s\)，取四面体重心为原点. 在适当的直角坐标系中，四个顶点可表示为
\[
(u,u,u),\quad(u,-u,-u),\quad(-u,u,-u),\quad(-u,-u,u),
\]
其中 \(s=2\sqrt2u\). 设四个顶点在平面法向单位向量
\(\bs n=(p,q,r)\) 上的有向坐标分别为 \(z_1,z_2,z_3,z_4\). 由于相邻平面距离为 1，且四个平面依次对应四个顶点，四个 \(z_i\) 构成公差为 1 的等差数列. 它们的平均值为重心在该方向上的坐标，故减去平均值后得到
\(-\frac{3}{2},-\frac{1}{2},\frac{1}{2},\frac{3}{2}\)，从而
\[
\sum_{i=1}^{4}(z_i-\overline z)^2=5.
\]
另一方面，由上述四个坐标直接计算，且 \(p^2+q^2+r^2=1\)，有
\[
\sum_{i=1}^{4}(z_i-\overline z)^2
=u^2\bigl[(p+q+r)^2+(p-q-r)^2+(-p+q-r)^2+(-p-q+r)^2\bigr]
=4u^2=\frac{s^2}{2}.
\]
所以 \(\frac{s^2}{2}=5\)，即 \(s=\sqrt{10}\). 正四面体体积为
\[
V=\frac{\sqrt2}{12}s^3=\frac{\sqrt2}{12}(\sqrt{10})^3=\frac{5\sqrt5}{3}.
\]
\end{enumerate}
\end{solution}
