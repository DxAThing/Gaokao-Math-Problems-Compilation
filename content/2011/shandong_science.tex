\begin{problem}

\item 求 \(\frac{\sin C}{\sin A}\) 的值；
\item 若 \(\cos B = \frac{1}{4}\)，\(b = 2\)，求 \(\triangle ABC\) 的面积 \(S\).
\end{enumerate}
\end{problem}
