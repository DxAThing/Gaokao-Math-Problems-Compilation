\section{单选题}

\begin{problem}
设集合 \(M = \{x \mid x^2 + x - 6 < 0\}\)，\(N = \{x \mid 1 \leqslant x \leqslant 3\}\). 则 \(M \cap N =\)（\quad）

\choices
    {\([1, 2)\)}
    {\([1, 2]\)}
    {\((2, 3]\)}
    {\([2, 3]\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
由 \(x^{2}+x-6<0\) 得 \((x+3)(x-2)<0\)，所以 \(M=(-3,2)\). 又 \(N=[1,3]\)，故 \(M\cap N=[1,2)\)，选 A.
\end{solution}

\begin{problem}
复数 \(z = \frac{2 - \i}{2 + \i}\)（\(\i\) 为虚数单位）在复平面内对应的点所在的象限为（\quad）

\choices
    {第一象限}
    {第二象限}
    {第三象限}
    {第四象限}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
将分子、分母同乘 \(2-\i\)，得 \(z=\frac{(2-\i)^2}{(2+\i)(2-\i)}=\frac{3-4\i}{5}=\frac35-\frac45\i\). 实部为正、虚部为负，所以对应点在第四象限，选 D.
\end{solution}

\begin{problem}
若点 \((a,9)\) 在函数 \(y = 3^{x}\) 的图象上，则 \(\tan \frac{a\pi}{6}\) 的值为（\quad）

\choices
    {0}
    {\( \frac{\sqrt{3}}{3} \)}
    {1}
    {\(\sqrt{3}\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
因为点 \((a,9)\) 在 \(y=3^x\) 上，所以 \(3^a=9=3^2\)，从而 \(a=2\). 因此 \(\tan\frac{a\pi}{6}=\tan\frac{\pi}{3}=\sqrt3\)，选 D.
\end{solution}

\begin{problem}
不等式 \( |x - 5| + |x + 3| \geqslant 10 \) 的解集是（\quad）

\choices
    {\([-5,7]\)}
    {\([-4,6]\)}
    {\((- \infty, -5] \cup [7, +\infty)\)}
    {\((- \infty, -4] \cup [6, +\infty)\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
在 \(-3\leqslant x\leqslant5\) 时，\(|x-5|+|x+3|=8<10\)，不满足不等式. 在 \(x<-3\) 时，不等式化为 \(2-2x\geqslant10\)，得 \(x\leqslant-4\). 在 \(x>5\) 时，不等式化为 \(2x-2\geqslant10\)，得 \(x\geqslant6\). 补上分界点后，解集为 \((-\infty,-4]\cup[6,+\infty)\)，选 D.
\end{solution}

\begin{problem}
对于函数 \(y = f(x)\)，\(x \in \R\). “\( y = |f(x)| \) 的图象关于 \( y \) 轴对称”是“\( y = f(x) \) 是奇函数”的（\quad）

\choices
    {充分而不必要条件}
    {必要而不充分条件}
    {充要条件}
    {既不充分也不必要条件}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
若 \(y=f(x)\) 是奇函数，则 \(f(-x)=-f(x)\)，于是 \(|f(-x)|=|f(x)|\)，所以 \(y=|f(x)|\) 的图象关于 \(y\) 轴对称. 反之不成立，例如取 \(f(x)=1\)，则 \(y=|f(x)|=1\) 关于 \(y\) 轴对称，但 \(f\) 不是奇函数. 因此前一个条件是后一个条件的必要而不充分条件，选 B.
\end{solution}

\begin{problem}
若函数 \( f(x) = \sin \omega x \)（\(\omega > 0\)）在区间 \(\left[0, \frac{\pi}{3}\right]\) 上单调递增，在区间 \(\left[\frac{\pi}{3}, \frac{\pi}{2}\right]\) 上单调递减，则 \(\omega =\)（\quad）

\choices
    {3}
    {2}
    {\(\frac{3}{2}\)}
    {\(\frac{2}{3}\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
在第一段区间上单调递增要求 \(\cos(\omega x)\geqslant0\)，在第二段区间上单调递减要求 \(\cos(\omega x)\leqslant0\). 由于相位从 \(0\) 开始连续增加，分界点必须恰好对应正弦函数的最大值，即 \(\omega\cdot\frac{\pi}{3}=\frac{\pi}{2}\). 故 \(\omega=\frac32\)，选 C.
\end{solution}

\begin{problem}
某产品的广告费用 \(x\) 与销售额 \(y\) 的统计数据如下表：

\begin{center}
\begin{tabular}{c|cccc}
广告费用 \(x\) （万元） & 4 & 2 & 3 & 5 \\
\hline
销售额 \(y\) （万元） & 49 & 26 & 39 & 54
\end{tabular}
\end{center}

根据上表可得回归方程 \( \hat{y} = \hat{b}x + \hat{a} \) 中的 \(\hat{b}\) 为 \(9.4\)，据此模型预报广告费用为 \(6\text{ 万元}\) 时销售额为（\quad）

\choices
    {\(63.6\text{ 万元}\)}
    {\(65.5\text{ 万元}\)}
    {\(67.7\text{ 万元}\)}
    {\(72.0\text{ 万元}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
由表中数据得 \(\bar{x}=\frac{4+2+3+5}{4}=\frac72\)，\(\bar{y}=\frac{49+26+39+54}{4}=42\). 回归直线 \(\hat y=\hat b x+\hat a\) 经过点 \((\bar x,\bar y)\)，所以 \(\hat a=42-9.4\times\frac72=9.1\). 当广告费用为 \(6\text{ 万元}\) 时，\(\hat y=9.4\times6+9.1=65.5\text{ 万元}\)，选 B.
\end{solution}

\begin{problem}
已知双曲线 \(\frac{x^2}{a^2} - \frac{y^2}{b^2} = 1 \) （\(a > 0, b > 0\)）的两条渐近线均和圆 \(C: x^2 + y^2 - 6x + 5 = 0\) 相切，且双曲线右焦点为圆 \(C\) 的圆心，则该双曲线的方程为（\quad）

\choices
    {\(\frac{x^2}{5} -\frac{y^2}{4} = 1\)}
    {\(\frac{x^2}{4} -\frac{y^2}{5} = 1\)}
    {\(\frac{x^2}{3} -\frac{y^2}{6} = 1\)}
    {\(\frac{x^2}{6} -\frac{y^2}{3} = 1\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
圆 \(C\) 的圆心为 \((3,0)\)，半径为 \(2\). 双曲线右焦点为该圆心，所以焦距满足 \(c=3\)，即 \(a^{2}+b^{2}=9\). 双曲线的渐近线为 \(y=\pm\frac{b}{a}x\)，圆心到任一渐近线的距离为 \(\frac{3b}{\sqrt{a^{2}+b^{2}}}=b\). 渐近线与圆相切，故该距离等于半径 \(2\)，从而 \(b=2\)，\(a^{2}=9-4=5\). 所以双曲线方程为 \(\frac{x^{2}}5-\frac{y^{2}}4=1\)，选 A.
\end{solution}

\begin{problem}
函数 \( y = \frac{x}{2} - 2\sin x \) 的图象大致是（\quad）

\choices
    {\choicebitmap[width=0.15\paperwidth]{img/2011/shandong_science/q9_choice_a.png}}
    {\choicebitmap[width=0.15\paperwidth]{img/2011/shandong_science/q9_choice_b.png}}
    {\choicebitmap[width=0.15\paperwidth]{img/2011/shandong_science/q9_choice_c.png}}
    {\choicebitmap[width=0.15\paperwidth]{img/2011/shandong_science/q9_choice_d.png}}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
记 \(g(x)=\frac{x}{2}-2\sin x\)，则 \(g(0)=0\)，且 \(g'(x)=\frac12-2\cos x\). 在 \(x=0\) 和 \(x=2\pi\) 处均有 \(g'(x)<0\)，所以图象在原点向下穿过 \(x\) 轴，并在 \(2\pi\) 附近仍向下；又 \(g(2\pi)=\pi>0\)，且 \(g(x+2\pi)=g(x)+\pi\)，所以图象的波动周期为 \(2\pi\) 而中线不断上移. 选项 C 同时符合这些性质，选 C.
\end{solution}

\begin{problem}
已知 \( f(x) \) 是 \( \R \) 上最小正周期为 2 的周期函数，且当 \( 0 \leqslant x < 2 \) 时，\( f(x) = x^3 - x \)，则函数 \( y = f(x) \) 的图象在区间 \([0, 6]\) 上与 \( x \) 轴的交点的个数为（\quad）

\choices
    {6}
    {7}
    {8}
    {9}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
在 \(0\leqslant x<2\) 时，\(f(x)=x^{3}-x=x(x-1)(x+1)\)，零点为 \(0\) 和 \(1\). 由于函数的周期为 \(2\)，在每个区间 \([2k,2k+2)\) 中的零点为 \(2k\) 和 \(2k+1\). 在闭区间 \([0,6]\) 内这些零点为 \(0,1,2,3,4,5,6\)，共有 \(7\) 个，选 B.
\end{solution}

\begin{problem}
如图是长和宽分别相等的两个矩形. 给定下列三个命题：

\begin{circlelist}
\item 存在三棱柱，其正（主）视图，俯视图如图；
\item 存在四棱柱，其正（主）视图，俯视图如图；
\item 存在圆柱，其正（主）视图，俯视图如图.
\end{circlelist}

其中真命题的个数是（\quad）
\bitmapfigure[max width=0.44\linewidth]{img/2011/shandong_science/q11_fig1.png}

\choices
    {3}
    {2}
    {1}
    {0}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
设两个矩形的长、宽分别为 \(l\)、\(w\). 取底面为 \(l\times w\) 的长方形、高为 \(w\) 的直四棱柱，其正视图和俯视图均为 \(l\times w\) 的矩形，所以命题 \(\circled{2}\) 为真. 取轴长为 \(l\)、底面为等腰直角三角形且两直角边均为 \(w\) 的直三棱柱，并使三角形的两条直角边分别沿正视和俯视方向，从相应方向观察时，三角形的投影分别为长度 \(w\) 的线段，两个视图均为 \(l\times w\) 的矩形，所以命题 \(\circled{1}\) 为真. 取轴长为 \(l\)、底面半径为 \(\frac w2\) 的圆柱，其正视图和俯视图也均为 \(l\times w\) 的矩形，所以命题 \(\circled{3}\) 为真. 因此真命题有 \(3\) 个，选 A.
\end{solution}

\begin{problem}
设 \(A_{1}\)，\(A_{2}\)，\(A_{3}\)，\(A_{4}\) 是平面直角坐标系中两两不同的四点. 若 \(\overrightarrow{A_{1}A_{3}} = \lambda \overrightarrow{A_{1}A_{2}}\) （\(\lambda \in \R\)），\(\overrightarrow{A_{1}A_{4}} = \mu \overrightarrow{A_{1}A_{2}}\) （\(\mu \in \R\)），且 \(\frac{1}{\lambda} + \frac{1}{\mu} = 2\)，则称 \(A_{3}\)，\(A_{4}\) 调和分割 \(A_{1}\)，\(A_{2}\). 已知平面上的点 \(C\)，\(D\) 调和分割点 \(A\)，\(B\)，则下面说法正确的是（\quad）

\choices
    {\(C\) 可能是线段 \(AB\) 的中点}
    {\(D\) 可能是线段 \(AB\) 的中点}
    {\(C\)，\(D\) 可能同时在线段 \(AB\) 上}
    {\(C\)，\(D\) 不可能同时在线段 \(AB\) 的延长线上}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
以 \(A\)、\(B\) 为基准，设 \(\overrightarrow{AC}=\lambda\overrightarrow{AB}\)，\(\overrightarrow{AD}=\mu\overrightarrow{AB}\). 由题设 \(\frac1\lambda+\frac1\mu=2\)，所以 \(\frac1\mu=2-\frac1\lambda\). 若 \(C\) 是线段 \(AB\) 的中点，则 \(\lambda=\frac12\)，这会使 \(\frac1\mu=0\)，不可能；同理 \(D\) 不可能是中点. 若 \(0<\lambda<1\)，当 \(0<\lambda<\frac12\) 时有 \(\mu<0\)，当 \(\frac12<\lambda<1\) 时有 \(\mu=\frac{\lambda}{2\lambda-1}>1\)，所以 \(C,D\) 不可能同时在线段 \(AB\) 上. 若 \(\lambda<0\)，则 \(0<\mu<\frac12\)；若 \(\lambda>1\)，则 \(\frac12<\mu<1\)，因此 \(C,D\) 也不可能同时在线段 \(AB\) 的延长线上. 故只有 D 正确，选 D.
\end{solution}

\begin{problem}
执行如图所示的程序框图，输入 \(l = 2\)，\(m = 3\)，\(n = 5\)，则输出的 \(y\) 的值是\fillinblank{}.

\bitmapfigure[width=6.5cm]{img/2011/shandong_science/q13_fig1.png}
\end{problem}

\begin{answer}
68
\end{answer}

\begin{solution}
输入 \(l=2\)、\(m=3\)、\(n=5\) 后，\(l^{2}+m^{2}+n^{2}=4+9+25=38\ne0\)，故先令 \(y=70l+21m+15n=70\times2+21\times3+15\times5=278\). 因为 \(278>105\)，执行一次 \(y=y-105\) 得 \(173\)，仍有 \(173>105\)，再执行一次得 \(68\). 此时 \(68\leqslant105\)，输出 \(y=68\).
\end{solution}

\begin{problem}
若 \(\left(x - \frac{\sqrt{a}}{x^{2}}\right)^6\) 展开式的常数项为 60，则常数 \(a\) 的值为\fillinblank{}.
\end{problem}

\begin{answer}
4
\end{answer}

\begin{solution}
展开式的通项为 \(\mathrm C_6^k x^{6-k}\left(-\frac{\sqrt a}{x^{2}}\right)^k=\mathrm C_6^k(-\sqrt a)^k x^{6-3k}\). 常数项要求 \(6-3k=0\)，所以 \(k=2\). 因此常数项为 \(\mathrm C_6^2a=15a\)，由 \(15a=60\) 得 \(a=4\).
\end{solution}

\begin{problem}
设函数 \(f(x) = \frac{x}{x + 2}\)（\(x > 0\)），观察：\(f_{1}(x) = f(x) = \frac{x}{x + 2}\)，\(f_{2}(x) = f\!\left(f_{1}(x)\right) = \frac{x}{3x + 4}\)，\(f_{3}(x) = f\!\left(f_{2}(x)\right) = \frac{x}{7x + 8}\)，\(f_{4}(x) = f\!\left(f_{3}(x)\right) = \frac{x}{15x + 16}\). 根据以上事实，由归纳推理可得：当 \(n \in \N^{*}\) 且 \(n \geqslant 2\) 时，\(f_n(x) = f(f_{n-1}(x)) =\)\fillinblank{}.
\end{problem}

\begin{answer}
\(\displaystyle \frac{x}{(2^{n}-1)x+2^{n}}\)
\end{answer}

\begin{solution}
设归纳结论为 \(f_n(x)=\frac{x}{(2^n-1)x+2^n}\). 当 \(n=2\) 时，题中已给出 \(f_2(x)=\frac{x}{3x+4}\)，结论成立. 若结论对 \(n-1\) 成立，则 \(f_n(x)=f(f_{n-1}(x))=\frac{\frac{x}{(2^{n-1}-1)x+2^{n-1}}}{\frac{x}{(2^{n-1}-1)x+2^{n-1}}+2}\). 分子、分母同乘 \((2^{n-1}-1)x+2^{n-1}\)，得 \(f_n(x)=\frac{x}{x+2(2^{n-1}-1)x+2^n}=\frac{x}{(2^n-1)x+2^n}\). 由数学归纳法，结论对所有 \(n\geqslant2\) 成立.
\end{solution}

\begin{problem}
已知函数 \( f(x) = \log_a x + x - b \)（\(a > 0\)，且 \(a \neq 1\)）. 当 \( 2 < a < 3 < b < 4 \) 时，函数 \( f(x) \) 的零点 \(x_0 \in (n, n+1)\)，\(n \in \N^{*}\)，则 \( n = \)\fillinblank{}.
\end{problem}

\begin{answer}
2
\end{answer}

\begin{solution}
因为 \(a>2\)，所以 \(\log_a2<1\)；又 \(b>3\)，于是 \(f(2)=\log_a2+2-b<1+2-3=0\). 因为 \(a<3\)，所以 \(\log_a3>1\)；又 \(b<4\)，于是 \(f(3)=\log_a3+3-b>1+3-4=0\). 函数在 \((0,+\infty)\) 上连续且 \(f'(x)=\frac{1}{x\ln a}+1>0\)，故其零点唯一，并且 \(x_0\in(2,3)\)，从而 \(n=2\).
\end{solution}

\section{解答题}

\begin{problem}
在 \(\triangle ABC\) 中，内角 \(A\)，\(B\)，\(C\) 的对边分别为 \(a\)，\(b\)，\(c\)，已知 \(\frac{\cos A - 2 \cos C}{\cos B} = \frac{2c - a}{b}\).
\begin{enumerate}
\item 求 \(\frac{\sin C}{\sin A}\) 的值；
\item 若 \(\cos B = \frac{1}{4}\)，\(b = 2\)，求 \(\triangle ABC\) 的面积 \(S\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由正弦定理，存在 \(R>0\)，使得 \(a=2R\sin A\)、\(b=2R\sin B\)、\(c=2R\sin C\)，因此 \(\frac{2c-a}{b}=\frac{2\sin C-\sin A}{\sin B}\). 原条件可改写为 \(\cos A-2\cos C=\cot B(2\sin C-\sin A)\). 移项并分组，得到 \(\cos A+\cot B\sin A=2(\cos C+\cot B\sin C)\). 又 \(A+B+C=\pi\)，所以 \(\cos A+\cot B\sin A=\frac{\sin(A+B)}{\sin B}=\frac{\sin C}{\sin B}\)，以及 \(\cos C+\cot B\sin C=\frac{\sin(B+C)}{\sin B}=\frac{\sin A}{\sin B}\). 故 \(\sin C=2\sin A\)，从而 \(\frac{\sin C}{\sin A}=2\).
\item 由第一问和正弦定理得 \(c=2a\). 由余弦定理，\(b^2=a^2+c^2-2ac\cos B=a^2+4a^2-2a\cdot2a\cdot\frac14=4a^2\). 代入 \(b=2\) 得 \(a=1\)，从而 \(c=2\). 又 \(\sin B=\sqrt{1-\cos^2B}=\frac{\sqrt{15}}4\)，故 \(S=\frac12ac\sin B=\frac12\cdot1\cdot2\cdot\frac{\sqrt{15}}4=\frac{\sqrt{15}}4\).
\end{enumerate}
\end{solution}

\begin{problem}
红队队员甲，乙，丙与蓝队队员 \(A\)，\(B\)，\(C\) 进行围棋比赛，甲对 \(A\)，乙对 \(B\)，丙对 \(C\) 各一盘. 已知甲胜 \(A\)，乙胜 \(B\)，丙胜 \(C\) 的概率分别为 \(0.6\)，\(0.5\)，\(0.5\). 假设各盘比赛结果相互独立.
\begin{enumerate}
\item 求红队至少两名队员获胜的概率；
\item 用 \(\xi\) 表示红队队员获胜的总数，求 \(\xi\) 的分布列和数学期望 \(E(\xi)\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 红队恰有两名队员获胜的概率为 \(0.6\times0.5\times0.5+0.6\times0.5\times0.5+0.4\times0.5\times0.5=0.40\)，三名队员全部获胜的概率为 \(0.6\times0.5\times0.5=0.15\). 所以红队至少两名队员获胜的概率为 \(0.40+0.15=0.55\).
\item \(\xi\) 的分布列为
\begin{center}
\begin{tabular}{c|cccc}
\(\xi\) & 0 & 1 & 2 & 3 \\
\hline
\(P\) & \(0.10\) & \(0.35\) & \(0.40\) & \(0.15\)
\end{tabular}
\end{center}
其中 \(P(\xi=0)=0.4\times0.5\times0.5=0.10\)，\(P(\xi=1)=0.6\times0.5\times0.5+0.4\times0.5\times0.5+0.4\times0.5\times0.5=0.35\)，\(P(\xi=2)=0.40\)，\(P(\xi=3)=0.15\). 因此 \(E(\xi)=0\times0.10+1\times0.35+2\times0.40+3\times0.15=1.6\).
\end{enumerate}
\end{solution}

\begin{problem}
在如图所示的几何体中，四边形 \(ABCD\) 为平行四边形，\(\angle ACB = 90^{\circ}\)，\(EA \perp\) 平面 \(ABCD\)，\(EF \parallel AB\)，\(FG \parallel BC\)，\(EG \parallel AC\)，\(AB = 2EF\).
\begin{enumerate}
\item 若 \(M\) 是线段 \(AD\) 的中点，求证：\(GM \parallel\) 平面 \(ABFE\)；
\item 若 \(AC = BC = 2AE\)，求二面角 \(A - BF - C\) 的大小.
\end{enumerate}

\bitmapfigure[max width=0.44\linewidth]{img/2011/shandong_science/q19_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
    \item 设 \(\overrightarrow{AB}=\bs b\)，\(\overrightarrow{BC}=\bs d\)，\(\overrightarrow{AE}=\bs e\). 由 \(EF\parallel AB\)、\(FG\parallel BC\)、\(EG\parallel AC\) 以及 \(AB=2EF\)，三角形 \(EFG\) 与三角形 \(ABC\) 的对应边比为 \(\frac12\)，故可取 \(F=E+\frac12\bs b\)，\(G=E+\frac12(\bs b+\bs d)\). 平行四边形给出 \(M=A+\frac12\bs d\). 于是 \(\overrightarrow{GM}=\frac12\bs d-\bs e-\frac12(\bs b+\bs d)=-\bs e-\frac12\bs b\)，它是 \(\bs e\) 与 \(\bs b\) 的线性组合，而平面 \(ABFE\) 的方向向量由 \(\bs b\)、\(\bs e\) 张成. 又因平面 \(ABFE\) 与平面 \(ABCD\) 的交线为 \(AB\)，而 \(M\) 不在 \(AB\) 上，所以 \(GM\parallel\) 平面 \(ABFE\).
\item 令 \(AE=h\)，建立空间直角坐标系，取 \(A=(0,0,0)\)、\(C=(2h,0,0)\)、\(B=(2h,2h,0)\)、\(E=(0,0,h)\). 由 \(AB=2EF\) 及 \(EF\parallel AB\)，得 \(F=(h,h,h)\). 于是 \(\overrightarrow{BA}=(-2h,-2h,0)\)、\(\overrightarrow{BF}=(-h,-h,h)\)、\(\overrightarrow{BC}=(0,-2h,0)\). 平面 \(ABF\) 和平面 \(CBF\) 的法向量可分别取 \(\bs n_1=\overrightarrow{BA}\times\overrightarrow{BF}=(-2h^2,2h^2,0)\) 和 \(\bs n_2=\overrightarrow{BC}\times\overrightarrow{BF}=(-2h^2,0,-2h^2)\). 故两平面夹角的余弦为 \(\frac{|\bs n_1\cdot\bs n_2|}{|\bs n_1||\bs n_2|}=\frac12\)，二面角 \(A-BF-C\) 的大小为 \(60^{\circ}\).
\end{enumerate}
\end{solution}

\begin{problem}
等比数列 \(\{a_{n}\}\) 中，\(a_{1}\)，\(a_{2}\)，\(a_{3}\) 分别是下表第一，二，三行中的某一个数，且 \(a_{1}\)，\(a_{2}\)，\(a_{3}\) 中的任何两个数不在下表的同一列.

\begin{center}
\begin{tabular}{c|ccc}
 & 第一列 & 第二列 & 第三列 \\
\hline
第一行 & 3 & 2 & 10 \\
第二行 & 6 & 4 & 14 \\
第三行 & 9 & 8 & 18
\end{tabular}
\end{center}

\begin{enumerate}
\item 求数列 \(\{a_{n}\}\) 的通项公式；
\item 若数列 \(\{b_{n}\}\) 满足： \(b_{n} = a_{n} + (-1)^{n} \ln a_{n}\)，求数列 \(\{b_{n}\}\) 的前 \(n\) 项和 \(S_{n}\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 若 \(a_1=3\)，则 \(a_2=6,4,14\) 时分别要求 \(a_3=12,\frac{16}{3},\frac{196}{3}\)，均不在第三行；若 \(a_1=2\)，则 \(a_2=6\) 时 \(a_3=18\)，且三项所在列依次为第二、第一、第三列，符合条件，而 \(a_2=4\) 时 \(a_3=8\) 与 \(a_1\) 同列，\(a_2=14\) 时 \(a_3=98\) 也不合条件；若 \(a_1=10\)，则 \(a_2=6,4,14\) 时分别要求 \(a_3=\frac{18}{5},\frac85,\frac{98}{5}\)，均不在第三行. 因此唯一可能的三项为 \(a_1=2\)、\(a_2=6\)、\(a_3=18\). 公比 \(q=\frac{a_2}{a_1}=3\)，数列通项为 \(a_n=2\cdot3^{n-1}\).
\item 由 \(a_n=2\cdot3^{n-1}\)，得 \(b_n=2\cdot3^{n-1}+(-1)^n\bigl(\ln2+(n-1)\ln3\bigr)\). 所以
\(S_n=\sum_{k=1}^{n}2\cdot3^{k-1}+\ln2\sum_{k=1}^{n}(-1)^k+\ln3\sum_{k=1}^{n}(-1)^k(k-1)\). 第一项为 \(3^n-1\). 当 \(n\) 为偶数时，后两项分别为 \(0\) 和 \(\frac n2\ln3\)；当 \(n\) 为奇数时，后两项分别为 \(-\ln2\) 和 \(-\frac{n-1}{2}\ln3\). 故
当 \(n\) 为偶数时，\(S_n=3^n-1+\frac n2\ln3\)；当 \(n\) 为奇数时，\(S_n=3^n-1-\ln2-\frac{n-1}{2}\ln3\).
\end{enumerate}
\end{solution}

\begin{problem}
某企业拟建造如图所示的容器（不计厚度，长度单位：米），其中容器的中间为圆柱形，左右两端均为半球形，按照设计要求容器的容积为 \(\frac{80\pi}{3}\text{ 立方米}\)，且 \(l\geqslant2r\). 假设该容器的建造费用仅与其表面积有关. 已知圆柱形部分每平方米建造费用为 \(3\text{ 千元}\)，半球形部分每平方米建造费用为 \(c\text{ 千元}\)（\(c>3\)）. 设该容器的建造费用为 \(y\text{ 千元}\).
\begin{enumerate}
\item 写出 \(y\) 关于 \(r\) 的函数表达式，并求该函数的定义域；
\item 求该容器的建造费用最小时的 \(r\).
\end{enumerate}

\bitmapfigure[max width=0.42\linewidth]{img/2011/shandong_science/q21_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 容器的容积由圆柱和两个半球组成，故 \(\pi r^2l+\frac43\pi r^3=\frac{80\pi}{3}\)，从而 \(l=\frac{80}{3r^2}-\frac43r\). 条件 \(l\geqslant2r\) 等价于 \(r^3\leqslant8\). 又 \(r>0\)，所以 \(0<r\leqslant2\). 圆柱侧面积为 \(2\pi rl\)，两个半球的总表面积为 \(4\pi r^2\)，故费用为 \(y=3\cdot2\pi rl+c\cdot4\pi r^2\). 代入 \(l\) 得 \(y=4\pi(c-2)r^2+\frac{160\pi}{r}\)，定义域为 \(0<r\leqslant2\).
\item 在 \(0<r\leqslant2\) 上，\(y'(r)=8\pi(c-2)r-\frac{160\pi}{r^2}=\frac{8\pi}{r^2}\bigl((c-2)r^3-20\bigr)\). 当 \(3<c\leqslant\frac92\) 时，\((c-2)r^3-20\leqslant8(c-2)-20\leqslant0\)，所以 \(y\) 在定义域上递减，最小值在 \(r=2\) 处取得. 当 \(c>\frac92\) 时，令 \(y'(r)=0\)，得 \(r_0=\sqrt[3]{\frac{20}{c-2}}<2\)，且导数在 \(r_0\) 左侧为负、右侧为正，所以最小值在 \(r=r_0\) 处取得. 综上，\(r=2\)（\(3<c\leqslant\frac92\)），\(r=\sqrt[3]{\frac{20}{c-2}}\)（\(c>\frac92\)）.
\end{enumerate}
\end{solution}

\begin{problem}
已知动直线 \(l\) 与椭圆 \(C: \frac{x^2}{3} + \frac{y^2}{2} = 1\) 交于 \(P(x_1, y_1)\)，\(Q(x_2, y_2)\) 两不同点，且 \(\triangle OPQ\) 的面积 \(S_{\triangle OPQ} = \frac{\sqrt{6}}{2}\). 其中 \(O\) 为坐标原点.
\begin{enumerate}
\item 证明： \(x_1^2 + x_2^2\) 和 \(y_1^2 + y_2^2\) 均为定值；
\item 设线段 \(PQ\) 的中点为 \(M\)，求 \(|OM| \cdot |PQ|\) 的最大值；
\item 椭圆 \(C\) 上是否存在三点 \(D\)，\(E\)，\(G\)，使得 \(S_{\triangle DOE} = S_{\triangle ODG} = S_{\triangle OEG} = \frac{\sqrt{6}}{2}\)；若存在，判断 \(\triangle DEG\) 的形状；若不存在，请说明理由.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 作伸缩变换 \(X=\frac{x}{\sqrt3}\)、\(Y=\frac{y}{\sqrt2}\). 椭圆上的点 \(P,Q\) 变为单位圆上的点 \(P'(X_1,Y_1)\)、\(Q'(X_2,Y_2)\). 由面积条件，\(\left|x_1y_2-x_2y_1\right|=\sqrt6\)，而 \(x_1y_2-x_2y_1=\sqrt6(X_1Y_2-X_2Y_1)\)，所以 \(\left|X_1Y_2-X_2Y_1\right|=1\). 又 \(P'\)、\(Q'\) 都是单位向量，故由行列式与内积的恒等式 \((X_1Y_2-X_2Y_1)^2+(X_1X_2+Y_1Y_2)^2=1\)，得 \(X_1X_2+Y_1Y_2=0\). 因此 \(P'\)、\(Q'\) 是一组互相垂直的单位向量，故 \(X_1^2+X_2^2=1\)、\(Y_1^2+Y_2^2=1\)，从而 \(x_1^2+x_2^2=3\)，\(y_1^2+y_2^2=2\).
\item 设 \(P'=(\cos\theta,\sin\theta)\)，\(Q'=(-\sin\theta,\cos\theta)\). 记 \(t=\sin2\theta\)，则 \(P=(\sqrt3\cos\theta,\sqrt2\sin\theta)\)，\(Q=(-\sqrt3\sin\theta,\sqrt2\cos\theta)\)，中点 \(M\) 满足 \(|OM|^2=\frac{5-t}{4}\)，且 \(|PQ|^2=5+t\). 因此 \((|OM|\cdot|PQ|)^2=\frac{(5-t)(5+t)}4=\frac{25-t^2}{4}\leqslant\frac{25}{4}\). 当 \(t=0\) 时等号成立，故 \(|OM|\cdot|PQ|\) 的最大值为 \(\frac52\).
\item 若存在这样的 \(D,E,G\)，经同一伸缩变换后得到单位圆上的 \(D',E',G'\). 每个条件 \(S_{\triangle ODE}=\frac{\sqrt6}{2}\) 等价于 \(|\det(D',E')|=1\)，从而 \(D'\perp E'\)；另外两个面积条件同理给出 \(D'\perp G'\)、\(E'\perp G'\). 平面内不可能存在三个两两垂直的非零向量，而单位圆上的向量均非零，所以这样的三点不存在.
\end{enumerate}
\end{solution}
