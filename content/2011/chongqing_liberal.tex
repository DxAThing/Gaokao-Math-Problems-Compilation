\section{单选题}

\begin{problem}
在等差数列 \(\{a_{n}\}\) 中，\(a_{2} = 2\)，\(a_{3} = 4\)，则 \(a_{10} =\)（\quad）

\choices
    {12}
    {14}
    {16}
    {18}
\end{problem}

\begin{problem}
设 \(U = \R\)，\(M = \{x \mid x^2 - 2x > 0\}\)，则 \(\mathbb{C}_U M =\)（\quad）

\choices
    {[0,2]}
    {(0,2)}
    {\((- \infty, 0) \cup (2, +\infty)\)}
    {\((- \infty, 0] \cup [2, +\infty)\)}
\end{problem}

\begin{problem}
曲线 \( y = -x^{3} + 3x^{2} \) 在点(1,2)处的切线方程为（\quad）

\choices
    {\(y = 3x - 1\)}
    {\(y = -3x + 5\)}
    {\(y = 3x + 5\)}
    {\(y = 2x\)}
\end{problem}

\begin{problem}
从一堆苹果中任取 10 只，称得它们的质量如下 （单位：克）：125 120 122 105 130 114 116 95 120 134 则样本数据落在 \([114.5, 124.5]\) 内的频率为（\quad）

\choices
    {0.2}
    {0.3}
    {0.4}
    {0.5}
\end{problem}

\begin{problem}
已知向量 \(a = (1, k)\)，\(b = (2, 2)\)，且 \(a + b\) 与 \(a\) 共线，那么 \(a \cdot b\) 的值为（\quad）

\choices
    {1}
    {2}
    {3}
    {4}
\end{problem}

\begin{problem}
设 \(a = \log_{\frac{1}{3}} \frac{1}{2}\)，\(b = \log_{\frac{1}{3}} \frac{2}{3}\)，\(c = \log_3 \frac{4}{3}\)，则 \(a\)，\(b\)，\(c\) 的大小关系是（\quad）

\choices
    {\(a < b < c\)}
    {\(c < b < a\)}
    {\(b < a < c\)}
    {\(b < c < a\)}
\end{problem}

\begin{problem}
若函数 \( f(x) = x + \frac{1}{x - 2} (x > 2) \) 在 \( x = a \) 处取最小值，则 \( a = \)（\quad）

\choices
    {\(1 + \sqrt{2}\)}
    {\(1 + \sqrt{3}\)}
    {3}
    {4}
\end{problem}

\begin{problem}
若 \(\triangle ABC\) 的内角 \(A\)，\(B\)，\(C\) 满足 \(6\sin A = 4\sin B = 3\sin C\)，则 \(\cos B =\)（\quad）

\choices
    {\(\frac{\sqrt{15}}{4}\)}
    {\(\frac{3}{4}\)}
    {\(\frac{3\sqrt{15}}{16}\)}
    {\(\frac{11}{16}\)}
\end{problem}

\begin{problem}
设双曲线的左准线与两条渐近线交于 A，B 两点，左焦点在以 \(AB\) 为直径的圆内，则该双曲线的离心率取值范围为（\quad）

\choices
    {\((0, \sqrt{2})\)}
    {\((1, \sqrt{2})\)}
    {\(\left(\frac{\sqrt{2}}{2}, 1\right)\)}
    {\((\sqrt{2}, +\infty)\)}
\end{problem}

\begin{problem}
高为 \(\sqrt{2}\) 的四棱锥 \(S - ABCD\) 的底面是边长为 1 的正方形，点 \(S\)，\(A\)，\(B\)，\(C\)，\(D\) 均在半径为 1 的同一球面上. 则底面 \(ABCD\) 的中心与顶点 \(S\) 之间的距离为（\quad）

    {\(\frac{\sqrt{2} + \sqrt{3}}{2}\)}

\section{解答题}

\begin{problem}
设 \(\{a_{n}\}\) 是公比为正数的等比数列，\(a_{1} = 2\)，\(a_{3} = a_{2} + 4\).
\begin{enumerate}
\item 求 \( \{a_{n}\} \) 的通项公式；
\item 设 \( \{b_{n}\} \) 是首项为 1，公差为 2 的等差数列. 求数列 \( \{a_{n} + b_{n}\} \) 的前 n 项和 \( S_{n} \).
\end{enumerate}
\end{problem}

\begin{problem}
某市公租房的房源位于 A，B，C 三个片区. 设每位申请人只申请其中一个片区的房源，且申请其中任一个片区的房源是等可能的，求该市的任 4 位申请人中：
\begin{enumerate}
\item 没有人申请 A 片区房源的概率；
\item 每个片区的房源都有人申请的概率.
\end{enumerate}
\end{problem}

\begin{problem}
设函数 \(f(x) = \sin x \cos x - \sqrt{3}\cos(\pi + x)\cos x\) （\(x \in \R\)）.
\begin{enumerate}
\item 求 \(f(x)\) 的最小正周期；
\item 若函数 \(y = f(x)\) 的图象按 \(b = \left(\frac{\pi}{4}, \frac{\sqrt{3}}{2}\right)\) 平移后得到函数 \(y = g(x)\) 的图象，求 \(y = g(x)\) 在 \(\left[0, \frac{\pi}{4}\right]\) 上的最大值.
\end{enumerate}
\end{problem}

\begin{problem}
设 \( f(x) = 2x^{3} + ax^{2} + bx + 1 \) 的导数为 \( f'(x) \). 若函数 \( y = f'(x) \) 的图象关于直线 \( x = -\frac{1}{2} \) 对称，且 \( f'(1) = 0 \).
\begin{enumerate}
\item 求实数 \(a\)，\(b\) 的值；
\item 求函数 \( f(x) \) 的极值.
\end{enumerate}
\end{problem}
