\section{解答题}

\begin{problem}
在 \(\triangle ABC\) 中，\(a, b, c\) 分别为内角 \(A, B, C\) 所对的边长，\(a = \sqrt{3}, b = \sqrt{2}\)，\(1 + 2\cos (B + C) = 0\). 求边 \(BC\) 上的高.
\end{problem}

\begin{problem}
设直线 \(l_{1}: y = k_{1}x + 1, l_{2}: y = k_{2}x - 1\)，其中实数 \(k_{1}, k_{2}\) 满足 \(k_{1}k_{2} + 2 = 0\). (1) 证明 \(l_{1}\) 与 \(l_{2}\) 相交； (2) 证明 \(l_{1}\) 与 \(l_{2}\) 的交点在椭圆 \(2x^{2} + y^{2} = 1\) 上.
\end{problem}

\begin{problem}
设 \( f(x) = \frac{\e^x}{1 + ax^2} \)，其中 \( a \) 为正实数. (1) 当 \(a = \frac{4}{3}\) 时，求 \(f(x)\) 的极值点; (2) 若 \( f(x) \) 为 R 上的单调函数，求 a 的取值范围.
\end{problem}

\end{problem}

\begin{problem}

\end{problem}
