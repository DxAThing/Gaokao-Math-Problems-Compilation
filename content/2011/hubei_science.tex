\section{单选题}

\begin{problem}
\(\i\) 为虚数单位，则 \(\left(\frac{1+\i}{1-\i}\right)^{2011}=\)（\quad）

\choices
    {\(-\i\)}
    {\(-1\)}
    {\(\i\)}
    {\(1\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
因为 \(\dfrac{1+\i}{1-\i}=\dfrac{(1+\i)^2}{1+1}=\i\)，而 \(2011=4\times502+3\)，所以 \(\left(\dfrac{1+\i}{1-\i}\right)^{2011}=\i^{2011}=\i^3=-\i\)，故选 A.
\end{solution}

\begin{problem}
已知 \(U=\{y\mid y=\log_2x,x>1\}\)，\(P=\left\{y\mid y=\dfrac{1}{x},x>2\right\}\)，则 \(\complement_U P=\)（\quad）

\choices
    {\(\left[\dfrac{1}{2},+\infty\right)\)}
    {\(\left(0,\dfrac{1}{2}\right)\)}
    {\((0,+\infty)\)}
    {\(\left(-\infty,0\right)\cup\left[\dfrac{1}{2},+\infty\right)\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
当 \(x>1\) 时，\(\log_2x>0\)，所以 \(U=(0,+\infty)\). 当 \(x>2\) 时，\(0<\dfrac{1}{x}<\dfrac{1}{2}\)，所以 \(P=\left(0,\dfrac{1}{2}\right)\). 因此在全集 \(U\) 中取补集，得 \(\complement_U P=\left[\dfrac{1}{2},+\infty\right)\)，故选 A.
\end{solution}

\begin{problem}
已知函数 \(f(x)=\sqrt{3}\sin x-\cos x\)，\(x\in\R\). 若 \(f(x)\geqslant1\)，则 \(x\) 的取值范围为（\quad）

\choices
    {\(\left\{x\mid k\pi+\dfrac{\pi}{3}\leqslant x\leqslant k\pi+\pi,k\in\Z\right\}\)}
    {\(\left\{x\mid 2k\pi+\dfrac{\pi}{3}\leqslant x\leqslant2k\pi+\pi,k\in\Z\right\}\)}
    {\(\left\{x\mid k\pi+\dfrac{\pi}{6}\leqslant x\leqslant k\pi+\dfrac{5\pi}{6},k\in\Z\right\}\)}
    {\(\left\{x\mid 2k\pi+\dfrac{\pi}{6}\leqslant x\leqslant2k\pi+\dfrac{5\pi}{6},k\in\Z\right\}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
由 \(\sqrt{3}\sin x-\cos x=2\sin\left(x-\dfrac{\pi}{6}\right)\)，原不等式等价于 \(\sin\left(x-\dfrac{\pi}{6}\right)\geqslant\dfrac{1}{2}\). 因此 \(\dfrac{\pi}{6}+2k\pi\leqslant x-\dfrac{\pi}{6}\leqslant\dfrac{5\pi}{6}+2k\pi\)，其中 \(k\in\Z\)，即 \(2k\pi+\dfrac{\pi}{3}\leqslant x\leqslant2k\pi+\pi\). 故选 B.
\end{solution}

\begin{problem}
将两个顶点在抛物线 \(y^2=2px\)（\(p>0\)）上，另一个顶点是此抛物线焦点的正三角形个数记为 \(n\)，则（\quad）

\choices
    {\(n=0\)}
    {\(n=1\)}
    {\(n=2\)}
    {\(n\geqslant3\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
抛物线的焦点为 \(F\left(\dfrac{p}{2},0\right)\). 用参数 \(t\) 表示抛物线上的点 \(P(t)=\left(\dfrac{p}{2}t^2,pt\right)\)，则
\[
\overrightarrow{FP(t)}=\left(\dfrac{p}{2}(t^2-1),pt\right)
\]
\[
|FP(t)|=\dfrac{p}{2}(t^2+1)
\]
若 \(P(t)\)、\(P(s)\) 与 \(F\) 构成正三角形，则 \(|FP(t)|=|FP(s)|\)，从而 \(s^2=t^2\). 因两点不能重合，\(t\ne0\)，故 \(s=-t\). 此时
\(\overrightarrow{FP(t)}\cdot\overrightarrow{FP(-t)}=\frac{p^2}{4}\left(t^4-6t^2+1\right)\)，\(|FP(t)|=|FP(-t)|=\frac{p}{2}(t^2+1)\).
正三角形在顶点 \(F\) 处的角为 \(60^\circ\)，所以
\[
\frac{p^2}{4}\left(t^4-6t^2+1\right)=|FP(t)|\,|FP(-t)|\cos60^\circ=\frac{p^2}{8}(t^2+1)^2
\]
约去正数 \(\dfrac{p^2}{8}\)，得 \(2\left(t^4-6t^2+1\right)=(t^2+1)^2\)，即 \(t^4-14t^2+1=0\). 令 \(u=t^2\)，则 \(u^2-14u+1=0\)，所以 \(u=7\pm4\sqrt{3}=(2\pm\sqrt{3})^2\)，有两个正值. 每个正值确定一对点 \(P(t)\)，\(P(-t)\)，因此正三角形共有 \(2\) 个，故选 C.
\end{solution}

\begin{problem}
已知随机变量 \(\xi\) 服从正态分布 \(N(2,\sigma^2)\)，且 \(P(\xi<4)=0.8\)，则 \(P(0<\xi<2)=\)（\quad）

\choices
    {\(0.6\)}
    {\(0.4\)}
    {\(0.3\)}
    {\(0.2\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
正态分布关于其均值 \(2\) 对称，所以 \(P(\xi<2)=P(\xi>2)=0.5\). 又因为 \(P(\xi<4)=0.8\)，故 \(P(2<\xi<4)=0.8-0.5=0.3\). 由关于 \(2\) 的对称性，\(P(0<\xi<2)=P(2<\xi<4)=0.3\)，故选 C.
\end{solution}

\begin{problem}
已知定义在 \(\R\) 上的奇函数 \(f(x)\) 和偶函数 \(g(x)\) 满足 \(f(x)+g(x)=a^x-a^{-x}+2\)（\(a>0\)，且 \(a\neq1\)）. 若 \(g(2)=a\)，则 \(f(2)=\)（\quad）

\choices
    {\(2\)}
    {\(\dfrac{15}{4}\)}
    {\(\dfrac{17}{4}\)}
    {\(a^2\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
令 \(x=2\)，利用 \(f\) 为奇函数、\(g\) 为偶函数，分别得到
\[
f(2)+g(2)=a^2-a^{-2}+2
\]
\[
-f(2)+g(2)=a^{-2}-a^2+2
\]
两式相加并代入 \(g(2)=a\)，得 \(2a=4\)，所以 \(a=2\). 再代入第一式，得 \(f(2)=4-\dfrac{1}{4}+2-2=\dfrac{15}{4}\)，故选 B.
\end{solution}

\begin{problem}
已知向量 \(\bs{a}=(x+z,3)\)，\(\bs{b}=(2,y-z)\)，且 \(\bs{a}\perp\bs{b}\). 若 \(x\)，\(y\) 满足不等式 \(|x|+|y|\leqslant1\)，则 \(z\) 的取值范围为（\quad）

\choices
    {\([-2,2]\)}
    {\([-2,3]\)}
    {\([-3,2]\)}
    {\([-3,3]\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
由向量垂直得 \(2(x+z)+3(y-z)=0\)，所以 \(z=2x+3y\). 在区域 \(|x|+|y|\leqslant1\) 上，\(|z|\leqslant2|x|+3|y|\leqslant3(|x|+|y|)\leqslant3\). 当 \((x,y)=(0,1)\) 时 \(z=3\)，当 \((x,y)=(0,-1)\) 时 \(z=-3\)；由于区域连通且线性函数取得介于最大值和最小值之间的所有值，故 \(z\in[-3,3]\)，选 D.
\end{solution}

\begin{problem}
若实数 \(a\)，\(b\) 满足 \(a\geqslant0\)，\(b\geqslant0\)，且 \(ab=0\)，则称 \(a\) 与 \(b\) 互补. 记 \(\varphi(a,b)=\sqrt{a^2+b^2}-a-b\)，那么 \(\varphi(a,b)=0\) 是 \(a\) 与 \(b\) 互补的（\quad）

\choices
    {必要而不充分条件}
    {充分而不必要条件}
    {充要条件}
    {既不充分也不必要条件}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
若 \(a\)、\(b\) 互补，则 \(ab=0\)，从而 \(a^2+b^2=(a+b)^2\). 因 \(a+b\geqslant0\)，有 \(\sqrt{a^2+b^2}=a+b\)，即 \(\varphi(a,b)=0\). 反之，若 \(\varphi(a,b)=0\)，则同样有 \(\sqrt{a^2+b^2}=a+b\)，两边平方得 \(a^2+b^2=a^2+2ab+b^2\)，所以 \(ab=0\). 因此该条件是充要条件，选 C.
\end{solution}

\begin{problem}
放射性元素由于不断有原子放射出微粒子而变成其他元素，其含量不断减少，这种现象称为衰变. 假设在放射性同位素铯 137 的衰变过程中，其含量 \(M\)（单位：太贝克）与时间 \(t\)（单位：年）满足函数关系：\(M(t)=M_0 2^{-t/30}\)，其中 \(M_0\) 为 \(t=0\) 时铯 137 的含量. 已知 \(t=30\) 时，铯 137 含量的变化率是 \(-10\ln2\)（太贝克/年），则 \(M(60)=\)（\quad）

\choices
    {\(5\text{ 太贝克}\)}
    {\(75\ln2\text{ 太贝克}\)}
    {\(150\ln2\text{ 太贝克}\)}
    {\(150\text{ 太贝克}\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
由 \(M(t)=M_0 2^{-t/30}\)，得 \(M'(t)=-\dfrac{\ln2}{30}M_0 2^{-t/30}\). 代入 \(t=30\) 及 \(M'(30)=-10\ln2\)，得 \(-\dfrac{\ln2}{60}M_0=-10\ln2\)，所以 \(M_0=600\). 因此 \(M(60)=600\cdot2^{-2}=150\)，故选 D.
\end{solution}

\begin{problem}
\(\left(x-\dfrac{1}{3\sqrt{x}}\right)^{18}\) 的展开式中含 \(x^{15}\) 的项的系数为\fillinblank{}.（结果用数值表示）
\end{problem}

\begin{answer}
\(17\)
\end{answer}

\begin{solution}
展开时取第二项 \(\dfrac{1}{3\sqrt{x}}\) 共 \(k\) 次，所得项的幂为 \(x^{18-k}x^{-k/2}=x^{18-3k/2}\). 令 \(18-\dfrac{3k}{2}=15\)，得 \(k=2\). 故所求系数为 \(\mathrm C_{18}^{2}\left(-\dfrac{1}{3}\right)^2=\dfrac{18\times17}{2\times9}=17\).
\end{solution}

\end{problem}

\(\dfrac{28}{145}\)
\end{answer}

\begin{problem}
《九章算术》“竹九节”问题：现有一根 \(9\) 节的竹子，自上而下各节的容积成等差数列，上面 \(4\) 节的容积共 \(3\text{ 升}\)，下面 \(3\) 节的容积共 \(4\text{ 升}\)，则第 \(5\) 节的容积为\fillinblank{}\(\text{升}\).
\end{problem}

\begin{answer}
\(\dfrac{67}{66}\)
\end{answer}

\begin{solution}
设第 \(1\) 节容积为 \(a\)，公差为 \(d\). 由前 \(4\) 节和后 \(3\) 节的和分别为 \(3\)、\(4\)，得 \(4a+6d=3\) 及 \(3a+21d=4\). 两式相减得 \(a-15d=-1\)，即 \(a=15d-1\). 代入第一式，得 \(4(15d-1)+6d=3\)，所以 \(d=\dfrac{7}{66}\)，进而 \(a=\dfrac{13}{22}\). 因此第 \(5\) 节容积为 \(a+4d=\dfrac{13}{22}+\dfrac{28}{66}=\dfrac{67}{66}\text{ 升}\).
\end{solution}

\begin{problem}
设 \(\triangle ABC\) 的内角 \(A\)，\(B\)，\(C\) 所对的边分别为 \(a\)，\(b\)，\(c\). 已知 \(a=1\)，\(b=2\)，\(\cos C=\dfrac{1}{4}\).
\begin{enumerate}
\item 求 \(\triangle ABC\) 的周长；
\item 求 \(\cos(A-C)\) 的值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由余弦定理，\(\cos C=\dfrac{a^2+b^2-c^2}{2ab}\)，所以 \(\dfrac{1+4-c^2}{4}=\dfrac{1}{4}\)，得 \(c^2=4\). 因边长为正，\(c=2\)，故三角形周长为 \(a+b+c=1+2+2=5\).
\item 由余弦定理，\(\cos A=\dfrac{b^2+c^2-a^2}{2bc}=\dfrac{4+4-1}{8}=\dfrac{7}{8}\). 因 \(A\)、\(C\) 是三角形内角，\(\sin A=\dfrac{\sqrt{15}}{8}\)，\(\sin C=\dfrac{\sqrt{15}}{4}\). 因此 \(\cos(A-C)=\cos A\cos C+\sin A\sin C=\dfrac{7}{32}+\dfrac{15}{32}=\dfrac{11}{16}\).
\end{enumerate}
\end{solution}

\begin{problem}
提高过江大桥的车辆通行能力可改善整个城市的交通状况. 在一般情况下，大桥上的车流速度 \(v\)（单位：千米/小时）是车流密度 \(x\)（单位：辆/千米）的函数. 当桥上的车流密度达到 \(200\text{ 辆/千米}\) 时，造成堵塞，此时车流速度为 \(0\)；当车流密度不超过 \(20\text{ 辆/千米}\) 时，车流速度为 \(60\text{ 千米/小时}\)；研究表明：当 \(20\leqslant x\leqslant200\) 时，车流速度 \(v\) 是车流密度 \(x\) 的一次函数.
\begin{enumerate}
\item 当 \(0\leqslant x\leqslant200\) 时，求函数 \(v(x)\) 的表达式；
\item 当车流密度 \(x\) 为多大时，车流量（单位时间内通过桥上某观测点的车辆数，单位：辆/小时）\(f(x)=x\cdot v(x)\) 可以达到最大，并求出最大值.（精确到 \(1\text{ 辆/小时}\)）
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 当 \(0\leqslant x\leqslant20\) 时，\(v(x)=60\). 当 \(20\leqslant x\leqslant200\) 时，直线经过 \((20,60)\) 和 \((200,0)\)，斜率为 \(\dfrac{0-60}{200-20}=-\dfrac{1}{3}\)，故 \(v(x)=60-\dfrac{1}{3}(x-20)=\dfrac{200-x}{3}\). 因此
\[
v(x)=\begin{cases}
60,&0\leqslant x\leqslant20,\\[2pt]
\dfrac{200-x}{3},&20<x\leqslant200.
\end{cases}
\]
\item 当 \(0\leqslant x\leqslant20\) 时，\(f(x)=60x\)，且 \(f'(x)=60>0\)，所以其最大值为 \(1200\). 当 \(20<x\leqslant200\) 时，\(f(x)=\dfrac{x(200-x)}{3}=\dfrac{10000-(x-100)^2}{3}\)，所以在 \(x=100\) 时取得最大值 \(\dfrac{10000}{3}\). 比较 \(\dfrac{10000}{3}>1200\)，故车流密度为 \(100\text{ 辆/千米}\) 时车流量最大；按题意精确到 \(1\) 辆/小时，车流量最大值为 \(3333\text{ 辆/小时}\).
\end{enumerate}
\end{solution}

\begin{problem}
如图，已知正三棱柱 \(ABC-A_{1}B_{1}C_{1}\) 的各棱长都是 \(4\)，\(E\) 是 \(BC\) 的中点，动点 \(F\) 在侧棱 \(CC_{1}\) 上，且不与点 \(C\) 重合.
\begin{enumerate}
\item 当 \(CF=1\) 时，求证：\(EF\perp A_{1}C\)；
\item 设二面角 \(C-AF-E\) 的大小为 \(\theta\)，求 \(\tan\theta\) 的最小值.
\end{enumerate}

\bitmapfigure[max width=0.32\linewidth]{img/2011/hubei_science/q18_fig1.png}
\end{problem}

\begin{solution}
建立空间直角坐标系，取 \(A=(0,0,0)\)，\(B=(4,0,0)\)，\(C=(2,2\sqrt{3},0)\)，并令竖直方向为 \(z\) 轴，则 \(A_{1}=(0,0,4)\)，\(C_{1}=(2,2\sqrt{3},4)\)，\(E=(3,\sqrt{3},0)\). 设 \(CF=t\)，则 \(0<t\leqslant4\)，且 \(F=(2,2\sqrt{3},t)\).
\begin{enumerate}
\item 当 \(t=1\) 时，\(\overrightarrow{EF}=(-1,\sqrt{3},1)\)，\(\overrightarrow{A_{1}C}=(2,2\sqrt{3},-4)\). 于是 \(\overrightarrow{EF}\cdot\overrightarrow{A_{1}C}=-2+6-4=0\)，所以 \(EF\perp A_{1}C\).
\item 由 \(\overrightarrow{AC}=(2,2\sqrt{3},0)\)，\(\overrightarrow{AF}=(2,2\sqrt{3},t)\)，\(\overrightarrow{AE}=(3,\sqrt{3},0)\)，有
\[
\overrightarrow{AC}\times\overrightarrow{AF}=2t\left(\sqrt{3},-1,0\right)
\]
\[
\overrightarrow{AE}\times\overrightarrow{AF}=\left(\sqrt{3}t,-3t,4\sqrt{3}\right)
\]
因此可取平面 \(ACF\) 和平面 \(AEF\) 的法向量
\(\bs{n}_1=\left(\sqrt{3},-1,0\right)\)，\(\bs{n}_2=\left(\sqrt{3}t,-3t,4\sqrt{3}\right)\).
二面角取其锐角，故
\[
\tan\theta=\frac{|\bs{n}_1\times\bs{n}_2|}{|\bs{n}_1\cdot\bs{n}_2|}=\frac{\sqrt{t^2+16}}{\sqrt{3}t}
\]
因此
\[
\tan^2\theta=\frac{t^2+16}{3t^2}=\frac{1}{3}+\frac{16}{3t^2}
\]
由于 \(0<t\leqslant4\)，上式随 \(t\) 增大而减小，所以在 \(t=4\)，即 \(F=C_{1}\) 时取得最小值，且 \(\tan\theta=\dfrac{\sqrt{32}}{4\sqrt{3}}=\dfrac{\sqrt{6}}{3}\).
\end{enumerate}
\end{solution}

\begin{problem}
已知数列 \(\{a_n\}\) 的前 \(n\) 项和为 \(S_n\)，且满足 \(a_1=a\)，\(a\neq0\)，\(a_{n+1}=rS_n\)，\(n\in\N^*\)，\(r\in\R\)，\(r\neq-1\).
\begin{enumerate}
\item 求数列 \(\{a_n\}\) 的通项公式；
\item 若存在 \(k\in\N^*\)，使得 \(S_{k+1}\)，\(S_k\)，\(S_{k+2}\) 成等差数列，试判断：对于任意的 \(m\in\N^*\)，且 \(m\geqslant2\)，\(a_{m+1}\)，\(a_m\)，\(a_{m+2}\) 是否成等差数列，并证明你的结论.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由 \(a_{n+1}=rS_n\) 得 \(a_{n+2}=rS_{n+1}=r(S_n+a_{n+1})=(1+r)a_{n+1}\). 又 \(a_2=ra\). 当 \(r=0\) 时，\(a_1=a\)，且 \(a_n=0\)（\(n\geqslant2\)）. 当 \(r\neq0\) 时，递推关系给出 \(a_n=ar(1+r)^{n-2}\)（\(n\geqslant2\)）.
\item 由等差数列条件，\(2S_k=S_{k+1}+S_{k+2}\). 利用 \(S_{k+1}=S_k+a_{k+1}\) 和 \(S_{k+2}=S_k+a_{k+1}+a_{k+2}\)，得 \(a_{k+2}=-2a_{k+1}\). 若 \(r=0\)，则对任意 \(m\geqslant2\)，\(a_m=a_{m+1}=a_{m+2}=0\)，结论成立. 若 \(r\neq0\)，则由第一问及 \(a\neq0\)、\(r\neq-1\) 可知 \(a_{k+1}\neq0\)，且 \(a_{k+2}=(1+r)a_{k+1}\)，所以 \(1+r=-2\)，即 \(r=-3\). 于是对任意 \(m\geqslant2\)，有 \(a_{m+1}=-2a_m\)、\(a_{m+2}=-2a_{m+1}=4a_m\)，从而 \(a_{m+1}+a_{m+2}=2a_m\). 因此 \(a_{m+1}\)，\(a_m\)，\(a_{m+2}\) 对任意 \(m\geqslant2\) 均成等差数列.
\end{enumerate}
\end{solution}

\begin{problem}
平面内与两定点 \(A_{1}(-a,0)\)，\(A_{2}(a,0)\)（\(a>0\)）连线的斜率之积等于非零常数 \(m\) 的点的轨迹，加上 \(A_{1}\)，\(A_{2}\) 两点所成的曲线 \(C\) 可以是圆、椭圆或双曲线.
\begin{enumerate}
\item 求曲线 \(C\) 的方程，并讨论 \(C\) 的形状与 \(m\) 值的关系；
\item 当 \(m=-1\) 时，对应的曲线为 \(C_1\)；对给定的 \(m\in(-1,0)\cup(0,+\infty)\)，对应的曲线为 \(C_2\). 设 \(F_1\)，\(F_2\) 是 \(C_2\) 的两个焦点. 试问：在 \(C_1\) 上，是否存在点 \(N\)，使得 \(\triangle F_1NF_2\) 的面积 \(S=|m|a^2\)？若存在，求 \(\tan\angle F_1NF_2\) 的值；若不存在，请说明理由.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设轨迹上的一般点为 \(M(x,y)\)，当 \(x\neq\pm a\) 时，两条连线的斜率乘积为 \(\dfrac{y}{x+a}\cdot\dfrac{y}{x-a}=\dfrac{y^2}{x^2-a^2}=m\)，所以 \(mx^2-y^2=ma^2\). 两定点 \(A_1\)、\(A_2\) 也满足该方程，故曲线 \(C\) 的方程为 \(mx^2-y^2=ma^2\). 当 \(m<-1\) 时，方程化为 \(\dfrac{x^2}{a^2}+\dfrac{y^2}{(-m)a^2}=1\)，是焦点在 \(y\) 轴上的椭圆；当 \(m=-1\) 时，方程为 \(x^2+y^2=a^2\)，是圆；当 \(-1<m<0\) 时，方程化为 \(\dfrac{x^2}{a^2}+\dfrac{y^2}{(-m)a^2}=1\)，是焦点在 \(x\) 轴上的椭圆；当 \(m>0\) 时，方程化为 \(\dfrac{x^2}{a^2}-\dfrac{y^2}{ma^2}=1\)，是焦点在 \(x\) 轴上的双曲线.
\item 由第一问，\(C_1\) 为圆 \(x^2+y^2=a^2\). 无论 \(-1<m<0\) 还是 \(m>0\)，曲线 \(C_2\) 的焦点均为 \(F_1\left(-a\sqrt{1+m},0\right)\)、\(F_2\left(a\sqrt{1+m},0\right)\). 设 \(N=(x,y)\in C_1\)，则
\[
S=\frac{1}{2}|F_1F_2||y|=a\sqrt{1+m}|y|
\]
由 \(S=|m|a^2\)，得 \(|y|=\dfrac{|m|a}{\sqrt{1+m}}\). 在圆 \(C_1\) 上存在这样的点的充要条件是 \(|y|\leqslant a\)，即 \(|m|\leqslant\sqrt{1+m}\). 当 \(-1<m<0\) 时，\(|m|=-m\)，于是 \(m^2\leqslant1+m\)，即 \(m^2-m-1\leqslant0\)，结合该区间得 \(m\in\left[\dfrac{1-\sqrt{5}}{2},0\right)\). 当 \(m>0\) 时，同样由 \(m^2-m-1\leqslant0\)，并结合该区间得 \(m\in\left(0,\dfrac{1+\sqrt{5}}{2}\right]\). 所以存在条件为 \(m\in\left[\dfrac{1-\sqrt{5}}{2},0\right)\cup\left(0,\dfrac{1+\sqrt{5}}{2}\right]\).

当存在时，令 \(\bs{u}=\overrightarrow{NF_1}\)、\(\bs{v}=\overrightarrow{NF_2}\). 由 \(N\in C_1\) 得 \(\bs{u}\cdot\bs{v}=a^2-a^2(1+m)=-ma^2\)，又 \(|\bs{u}\times\bs{v}|=2a\sqrt{1+m}|y|=2|m|a^2\). 因此
\[
\tan\angle F_1NF_2=\frac{|\bs{u}\times\bs{v}|}{\bs{u}\cdot\bs{v}}=-\frac{2|m|}{m}
\]
所以当 \(m\in\left[\dfrac{1-\sqrt{5}}{2},0\right)\) 时，\(\tan\angle F_1NF_2=2\)；当 \(m\in\left(0,\dfrac{1+\sqrt{5}}{2}\right]\) 时，\(\tan\angle F_1NF_2=-2\). 其余给定的 \(m\) 不存在满足条件的点 \(N\).
\end{enumerate}
\end{solution}

\begin{problem}
\begin{enumerate}
\item 已知函数 \(f(x)=\ln x-x+1\)，\(x\in(0,+\infty)\)，求函数 \(f(x)\) 的最大值.

\item 设 \(a_k\)，\(b_k\)（\(k=1\)，\(2\)，\(\dots\)，\(n\)）均为正数，证明：

\begin{circlelist}
\item 若 \(a_1b_1+a_2b_2+\dots+a_nb_n\leqslant b_1+b_2+\dots+b_n\)，则 \(a_1^{b_1}a_2^{b_2}\cdots a_n^{b_n}\leqslant1\)；
\item 若 \(b_1+b_2+\dots+b_n=1\)，则 \(\dfrac{1}{n}\leqslant b_1^{b_1}b_2^{b_2}\cdots b_n^{b_n}\leqslant b_1^2+b_2^2+\dots+b_n^2\).
\end{circlelist}
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 因为 \(f'(x)=\dfrac{1}{x}-1=\dfrac{1-x}{x}\)，所以 \(f'(x)>0\) 在 \((0,1)\) 上成立，\(f'(1)=0\)，且 \(f'(x)<0\) 在 \((1,+\infty)\) 上成立. 因此 \(f(x)\) 在 \((0,1)\) 上递增、在 \((1,+\infty)\) 上递减，最大值在 \(x=1\) 处取得，且 \(f(1)=0\).
\item 由第一问，对任意正数 \(u\)，都有 \(\ln u-u+1\leqslant0\)，即 \(\ln u\leqslant u-1\).
\begin{enumerate}
\item 将 \(u=a_k\) 代入并乘以正数 \(b_k\)，再求和，得 \(\sum_{k=1}^{n}b_k\ln a_k\leqslant\sum_{k=1}^{n}a_kb_k-\sum_{k=1}^{n}b_k\leqslant0\). 两边取指数，得 \(a_1^{b_1}a_2^{b_2}\cdots a_n^{b_n}\leqslant1\).
\item 记 \(B=b_1^{b_1}b_2^{b_2}\cdots b_n^{b_n}\)，\(s=b_1^2+b_2^2+\dots+b_n^2\). 先取 \(a_k=\dfrac{1}{nb_k}\)，则 \(\sum_{k=1}^{n}a_kb_k=1=\sum_{k=1}^{n}b_k\)，由上一结论得 \(\prod_{k=1}^{n}\left(\dfrac{1}{nb_k}\right)^{b_k}\leqslant1\)，即 \(\dfrac{1}{nB}\leqslant1\)，所以 \(B\geqslant\dfrac{1}{n}\). 再取 \(a_k=\dfrac{b_k}{s}\)，则 \(\sum_{k=1}^{n}a_kb_k=1=\sum_{k=1}^{n}b_k\)，由上一结论得 \(\prod_{k=1}^{n}\left(\dfrac{b_k}{s}\right)^{b_k}\leqslant1\). 由于 \(\sum_{k=1}^{n}b_k=1\)，故 \(\dfrac{B}{s}\leqslant1\)，即 \(B\leqslant s=b_1^2+b_2^2+\dots+b_n^2\).
\end{enumerate}
\end{enumerate}
\end{solution}
