\begin{problem}
数列 \(\{a_{n}\}\) 的前 \(n\) 项和记为 \(S_{n}\)，\(a_{1} = 1\)，\(a_{n+1} = 2S_{n} + 1\)（\(n \geqslant 1\)）.

\begin{enumerate}
\item 求 \(\{a_{n}\}\) 的通项公式；
\item 等差数列 \(\{b_{n}\}\) 的各项为正，其前 \(n\) 项和 \(T_{n}\)，且 \(T_{3}=15\)，又 \(a_{1}+b_{1}\)，\(a_{2}+b_{2}\)，\(a_{3}+b_{3}\) 成等比数列，求 \(T_{n}\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由 \(S_n=S_{n-1}+a_n\)，当 \(n\geqslant 2\) 时，
\[
a_{n+1}-a_n=(2S_n+1)-(2S_{n-1}+1)=2a_n.
\]
又 \(a_2=2S_1+1=3=3a_1\)，所以对任意 \(n\geqslant 1\)，都有 \(a_{n+1}=3a_n\). 因此 \(\{a_n\}\) 是首项为 \(1\)，公比为 \(3\) 的等比数列，从而 \(a_n=3^{n-1}\).

\item 设等差数列 \(\{b_n\}\) 的首项为 \(b_1=u\)，公差为 \(d\). 由 \(T_3=15\)，得
\(3u+3d=15\)，即 \(u+d=5\).
由第一问，\(a_1=1\)，\(a_2=3\)，\(a_3=9\). 三个数成等比数列，所以
\[
\bigl(3+u+d\bigr)^2=(1+u)\bigl(9+u+2d\bigr).
\]
利用 \(d=5-u\)，上式化为 \(8^2=(1+u)(19-u)\)，即 \(u^2-18u+45=0\)，解得 \(u=3\) 或 \(u=15\). 当 \(u=15\) 时，\(d=-10\)，有 \(b_3=-5<0\)，不合题意；故 \(u=3\)，\(d=2\). 因此
\[
T_n=\frac{n}{2}\bigl[2u+(n-1)d\bigr]=\frac{n}{2}\bigl[6+2(n-1)\bigr]=n^2+2n.
\]
\end{enumerate}
\end{solution}

\begin{problem}
已知 \(A\)，\(B\)，\(C\) 是 \(\triangle ABC\) 三内角，向量 \(\bs{m} = (-1, \sqrt{3})\)，\(\bs{n} = (\cos A, \sin A)\)，且 \(\bs{m} \cdot \bs{n} = 1\).

\begin{enumerate}
\item 求角 \(A\)；
\item 若 \(\frac{1 + \sin 2B}{\cos^2 B - \sin^2 B} = -3\)，求 \(\tan C\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由向量数量积条件，得
\[
-\cos A+\sqrt{3}\sin A=1=2\sin\left(A-\frac{\pi}{6}\right).
\]
因为 \(0<A<\pi\)，所以 \(-\frac{\pi}{6}<A-\frac{\pi}{6}<\frac{5\pi}{6}\). 在此范围内，方程 \(\sin\left(A-\frac{\pi}{6}\right)=\frac{1}{2}\) 只有解 \(A-\frac{\pi}{6}=\frac{\pi}{6}\)，故 \(A=\frac{\pi}{3}\).

\item 原等式有意义，所以 \(\cos 2B\ne 0\). 若 \(\cos B=0\)，则 \(B=\frac{\pi}{2}\)，原式等于 \(-1\)，与题设矛盾，故 \(\cos B\ne 0\). 令 \(t=\tan B\)，则
\[
-3=\frac{1+\sin 2B}{\cos 2B}=\frac{(1+t)^2}{1-t^2}=\frac{1+t}{1-t},
\]
解得 \(t=2\)，即 \(\tan B=2\). 由三角形内角和，\(C=\frac{2\pi}{3}-B\)，所以
\[
\tan C=\frac{\tan\frac{2\pi}{3}-\tan B}{1+\tan\frac{2\pi}{3}\tan B}
=\frac{-\sqrt{3}-2}{1-2\sqrt{3}}=\frac{8+5\sqrt{3}}{11}.
\]
\end{enumerate}
\end{solution}

\begin{problem}
某课程考核分理论和实验两部分进行，每部分考核成绩只记“合格”与“不合格”，两部分考核都“合格”则该课程考核“合格”. 甲，乙，丙三人在理论考核中合格的概率分别为 \(0.9\)，\(0.8\)，\(0.7\)；在实验考核中合格的概率分别为 \(0.8\)，\(0.7\)，\(0.9\). 所有考核是否合格互相之间没有影响.

\begin{enumerate}
\item 求甲，乙，丙三人在理论考核中至少要有两人合格的概率；
\item 求这三人该课程考核都合格的概率（结果保留三位小数）.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设理论考核中甲、乙、丙合格分别为事件 \(A\)，\(B\)，\(C\). 至少两人合格包括恰有两人合格和三人都合格，故所求概率为
\[
0.9\times0.8\times0.3+0.9\times0.2\times0.7+0.1\times0.8\times0.7+0.9\times0.8\times0.7=0.902.
\]

\item 甲、乙、丙课程考核合格的概率分别为 \(0.9\times0.8=0.72\)，\(0.8\times0.7=0.56\)，\(0.7\times0.9=0.63\). 由于所有考核是否合格互相之间没有影响，三人课程考核都合格的概率为
\[
0.72\times0.56\times0.63=0.254016\approx0.254.
\]
\end{enumerate}
\end{solution}

如图，在长方体 \(ABCD - A_{1}B_{1}C_{1}D_{1}\) 中，\(E\)，\(P\) 分别是 \(BC\)，\(A_{1}D_{1}\) 的中点，\(M\)，\(N\) 分别是 \(AE\)，\(CD_{1}\) 的中点，\(AD = AA_{1} = a\)，\(AB = 2a\).

\item 求二面角 \(P - AE - D\) 的大小.
\end{enumerate}

\bitmapfigure[max width=0.55\linewidth]{img/2006/sichuan_liberal/q20_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 以 \(A\) 为原点，分别以 \(\overrightarrow{AB}\)，\(\overrightarrow{AD}\)，\(\overrightarrow{AA_1}\) 所在直线为坐标轴，建立空间直角坐标系，则
\(A=(0,0,0)\)，\(B=(2a,0,0)\)，\(D=(0,a,0)\)，\(A_1=(0,0,a)\).
于是 \(C=(2a,a,0)\)，\(D_1=(0,a,a)\)，由中点条件得
\(E=\left(2a,\frac{a}{2},0\right)\)，\(P=\left(0,\frac{a}{2},a\right)\)，\(M=\left(a,\frac{a}{4},0\right)\)，\(N=\left(a,a,\frac{a}{2}\right)\).
从而 \(\overrightarrow{MN}=\left(0,\frac{3a}{4},\frac{a}{2}\right)\). 平面 \(ADD_1A_1\) 的方程为 \(x=0\)，而直线 \(MN\) 上各点的横坐标均为 \(a\)，故 \(MN\) 的方向平行于该平面且与该平面没有公共点，所以 \(MN\parallel\) 面 \(ADD_1A_1\).

\item 记二面角 \(P-AE-D\) 为 \(\theta\). 取平面 \(PAE\) 和平面 \(AED\) 的法向量，分别可取
\(\bs{u}=\overrightarrow{AE}\times\overrightarrow{AP}=\left(\frac{a^2}{2},-2a^2,a^2\right)\)，\(\bs{v}=\overrightarrow{AE}\times\overrightarrow{AD}=(0,0,2a^2)\).
于是 \(\bs{u}\cdot\bs{v}=2a^4\)，\(\lvert\bs{u}\rvert=\frac{a^2\sqrt{21}}{2}\)，\(\lvert\bs{v}\rvert=2a^2\). 因而
\[
\cos\theta=\frac{\bs{u}\cdot\bs{v}}{\lvert\bs{u}\rvert\lvert\bs{v}\rvert}=\frac{2}{\sqrt{21}}=\frac{2\sqrt{21}}{21}.
\]
所以二面角 \(P-AE-D\) 的大小为 \(\arccos\dfrac{2\sqrt{21}}{21}\).
\end{enumerate}
\end{solution}

\begin{problem}
已知函数 \(f(x) = x^3 + 3ax - 1\)，\(g(x) = f'(x) - ax - 5\)，其中 \(f'(x)\) 是 \(f(x)\) 的导函数.

\begin{enumerate}
\item 对满足 \(-1 \leqslant a \leqslant 1\) 的一切 \(a\) 的值，都有 \(g(x) < 0\)，求实数 \(x\) 的取值范围；
\item 设 \(a = -m^2\)，当实数 \(m\) 在什么范围内变化时，函数 \(y = f(x)\) 的图像与直线 \(y = 3\) 只有一个公共点.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由 \(f'(x)=3x^2+3a\)，得
\[
g(x)=3x^2-ax+3a-5=3x^2-5+a(3-x).
\]
当 \(-1\leqslant a\leqslant1\) 时，关于 \(a\) 的一次式 \(a(3-x)\) 的最大值为 \(\lvert 3-x\rvert\). 因而题设条件等价于
\[
3x^2-5+\lvert 3-x\rvert<0.
\]
若 \(x\leqslant3\)，则上式化为 \(3x^2-x-2<0\)，即 \(-\frac{2}{3}<x<1\). 若 \(x>3\)，则上式化为 \(3x^2+x-8<0\)，但其在 \(x>3\) 时不可能成立. 故 \(x\) 的取值范围为 \(\left(-\frac{2}{3},1\right)\).

\item 令 \(r=\lvert m\rvert\). 函数图像与直线 \(y=3\) 的公共点横坐标是方程
\[
h(x)=x^3-3m^2x-4=x^3-3r^2x-4=0
\]
的实根. 当 \(r=0\) 时，\(h'(x)=3x^2\geqslant0\)，且 \(h\) 只有一个实根.

当 \(r>0\) 时，\(h'(x)=3(x-r)(x+r)\)，所以 \(h\) 在 \((-\infty,-r)\) 和 \((r,+\infty)\) 上递增，在 \((-r,r)\) 上递减，并且
\(h(-r)=2r^3-4\)，\(h(r)=-2r^3-4<0\).
若 \(2r^3-4<0\)，两个极值点的函数值都小于零，方程只有一个实根；若 \(2r^3-4=0\)，\(-r\) 是一个重根且另有实根，故有两个不同实根；若 \(2r^3-4>0\)，方程有三个实根. 因而只有一个公共点的充要条件是 \(r^3<2\)，即
\[
m\in\left(-\sqrt[3]{2},\sqrt[3]{2}\right).
\]
\end{enumerate}
\end{solution}

\begin{problem}
已知两定点 \(F_{1}(-\sqrt{2},0)\)，\(F_{2}(\sqrt{2},0)\)，满足条件 \(\left|PF_{2}\right| - \left|PF_{1}\right| = 2\) 的点 \(P\) 的轨迹是曲线 \(E\)，直线 \(y = kx - 1\) 与曲线 \(E\) 交于 \(A\)，\(B\) 两点.

\begin{enumerate}
\item 求 \(k\) 的取值范围；
\item 如果 \(\left|AB\right|=6\sqrt{3}\)，且曲线 \(E\) 上存在点 \(C\)，使 \(\overrightarrow{OA}+\overrightarrow{OB}=m\overrightarrow{OC}\)，求 \(m\) 的值和 \(\triangle ABC\) 的面积 \(S\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设 \(P=(x,y)\). 由焦点坐标和距离差条件，曲线 \(E\) 是焦点为 \(F_1,F_2\)、实轴长为 \(2\) 的双曲线. 因为焦距的一半为 \(\sqrt{2}\)，故虚半轴的平方为 \(\left(\sqrt{2}\right)^2-1^2=1\)，双曲线方程为 \(x^2-y^2=1\). 条件 \(\lvert PF_2\rvert-\lvert PF_1\rvert=2>0\) 选取其左支，所以 \(E\) 满足 \(x\leqslant-1\).

将直线 \(y=kx-1\) 代入双曲线方程，得
\[
(1-k^2)x^2+2kx-2=0.
\]
要有两个不同的交点，首先需 \(4(2-k^2)>0\)，即 \(\lvert k\rvert<\sqrt{2}\). 当 \(-1<k<1\) 时，方程两根异号，不能同时在双曲线左支；当 \(1<k<\sqrt{2}\) 时，两根均为正，也不能在左支；当 \(-\sqrt{2}<k<-1\) 时，两根同为负，而双曲线上横坐标为负的点必在左支. \(k=\pm1\) 时方程退化为一次方程，\(k=\pm\sqrt{2}\) 时只有一个交点. 因此 \(k\) 的取值范围为 \(-\sqrt{2}<k<-1\).

\item 设直线与双曲线的两个交点横坐标为 \(x_1,x_2\). 由根的判别式和根之差，
\[
\lvert x_1-x_2\rvert=\frac{2\sqrt{2-k^2}}{k^2-1}.
\]
由于 \(A,B\) 在斜率为 \(k\) 的直线上，故
\[
\lvert AB\rvert=\sqrt{1+k^2}\lvert x_1-x_2\rvert=\frac{2\sqrt{(1+k^2)(2-k^2)}}{k^2-1}=6\sqrt{3}.
\]
令 \(t=k^2\)，平方并整理得
\[
(1+t)(2-t)=27(t-1)^2.
\]
所以 \(28t^2-55t+25=0\)，解得 \(t=\frac{5}{4}\) 或 \(t=\frac{5}{7}\). 结合 \(1<k^2<2\) 及 \(k<0\)，得 \(k=-\frac{\sqrt{5}}{2}\).

由韦达定理，
\(x_1+x_2=\frac{2k}{k^2-1}\)，\(y_1+y_2=k(x_1+x_2)-2=\frac{2}{k^2-1}\).
当 \(k=-\frac{\sqrt{5}}{2}\) 时，\(k^2-1=\frac14\)，所以
\[
\overrightarrow{OA}+\overrightarrow{OB}=(-4\sqrt{5},8).
\]
设 \(C=(x_C,y_C)\)，由 \(\overrightarrow{OA}+\overrightarrow{OB}=m\overrightarrow{OC}\)，得 \(C=\left(-\frac{4\sqrt{5}}{m},\frac{8}{m}\right)\). 又 \(C\in E\)，故
\[
\frac{(-4\sqrt{5})^2-8^2}{m^2}=1.
\]
所以 \(m^2=16\).
由于 \(C\) 必在双曲线左支，取 \(m=4\)，从而 \(C=(-\sqrt{5},2)\).

直线 \(AB\) 的方程为 \(y=-\frac{\sqrt{5}}{2}x-1\)，点 \(C\) 到该直线的距离为
\[
d=\frac{\left\lvert\frac{\sqrt{5}}{2}(-\sqrt{5})+2+1\right\rvert}{\sqrt{\frac54+1}}=\frac13.
\]
所以
\[
