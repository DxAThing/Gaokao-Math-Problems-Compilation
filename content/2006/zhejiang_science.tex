\chapter{2006年浙江卷（理）}
\section{单选题}
\begin{problem}
设集合 \(A = \{x \mid -1 \leqslant x \leqslant 2\}\)，\(B = \{x \mid 0 \leqslant x \leqslant 4\}\)，则 \(A \cap B =\)（\quad）

\choices
    {[0,2]}
    {\([1,2]\)}
    {\([0,4]\)}
    {\([1,4]\)}
\end{problem}

\begin{problem}
已知 \(\frac{m}{1 + \i} = 1 - mi\)，其中 \(m, n\) 是实数，1 是虚数单位，则 \(m + ni =\)（\quad）

\choices
    {\(1 + 2\i\)}
    {\(1 - 2\i\)}
    {\(2 + \i\)}
    {\(2 - \i\)}
\end{problem}

\begin{problem}
已知 \(0 < a < 1, \log_a m < \log_a n < 0\)，则（\quad）

\choices
    {\(1 < n < m\)}
    {\(1 < m < n\)}
    {\(m < n < 1\)}
    {\(n < m < 1\)}
\end{problem}

\begin{problem}
在平面直角坐标系中，不等式组 \( \left\{\begin{aligned}x+y-2&\geqslant0,\\ x-y+2&\geqslant0,\end{aligned}\right. \) 表示的平面区域的面积是（\quad）

\choices
    {\(4\sqrt{2}\)}
    {4}
    {\(2 \sqrt{2}\)}
    {2}
\end{problem}

\begin{problem}
若双曲线 \(\frac{x^2}{m} - y^2 = 1\) 上的点到左准线的距离是到左焦点距离的 \(\frac{1}{3}\)，则（\quad）

\choices
    {\(\frac{1}{2}\)}
    {\(\frac{3}{2}\)}
    {\(\frac{1}{8}\)}
    {\(\frac{9}{8}\)}
\end{problem}

\begin{problem}
函数 \( y = \frac{1}{2}\sin 2x + \sin^2 x, x \in \R \) 的值域是（\quad）

\choices
    {\(\left[-\frac{1}{2}, \frac{3}{2}\right]\)}
    {\(\left[\frac{3}{2}, \frac{1}{2}\right]\)}
    {\(\left[-\frac{\sqrt{2}}{2} + \frac{1}{2}, \frac{\sqrt{2}}{2} + \frac{1}{2}\right]\)}
    {\(\left[-\frac{\sqrt{2}}{2} - \frac{1}{2}, \frac{\sqrt{2}}{2} - \frac{1}{2}\right]\)}
\end{problem}

\begin{problem}
“a > b > 0”是“ab < \( \frac{a^{2} + b^{2}}{a} \) ”的（\quad）

\choices
    {充分而不必要条件}
    {必要而不充分条件}
    {充分必要条件}
    {既不充分也不必要条件}
\end{problem}

\begin{problem}
若多项式 \(x^{2} + x^{10} = a_{0} + a_{1}(x + 1) + \dots +a_{9}(x + 1)^{10} + a_{10}(x + 1)^{10}\)，则 \(a_0 =\)（\quad）

\choices
    {9}
    {10}
    {\(-9\)}
    {\(-10\)}
\end{problem}

\begin{problem}
如图，\(O\) 是半径为 1 的球心，点 \(A, B, C\) 在球面上，\(O, A, OB, OC\) 两两垂直，\(E, F\) 分别是大圆弧 \(\widehat{AB}\) 与 \(\widehat{AC}\) 的中点，则点 \(E, F\) 在该球面上的球面距离是（\quad）

\choices
    {\(\frac{\pi}{4}\)}
    {\(\frac{\pi}{3}\)}
    {\(\frac{\pi}{2}\)}
    {\(\frac{\sqrt{2}\pi}{4}\)}

\begin{problem}
函数 \( f: [1, 2, 3] \to [1, 2, 3] \) 满足 \( f(f(x)) = f(x) \)，则这样的函数个数共有（\quad）

\section{解答题}
\begin{problem}
如图，函数 \( y = 2\sin (\pi x + \varphi) \)，\( x \in \R \)，(其中 \( 0 \leqslant \varphi \leqslant \frac{\pi}{2} \)) 的图象与 \( y \) 轴交于点 (0,1). (1) 求 \( \varphi \) 的值: (2) 设 \(P\) 是图象上的最高点，\(M, N\) 是图象与 \(x\) 轴的交点，求 \(\overrightarrow{PM}\) 与 \(\overrightarrow{FN}\) 的夹角.
\end{problem}

\begin{problem}
设 \( f(x)=3ax^{2}+2bx+c \) . 若 \( a+b+c=0 \) ，\( f(0)>0 \) ，\( f(1)>0 \) ，求证: (1) \(a > 0\) 且 \(-2 < \frac{b}{a} < -1\); (2) 方程 \( f(x)=0 \) 在 \( (0,1) \) 内有两个实根.
\end{problem}

\begin{problem}
设 \( f(x)=3ax^{2}+2bx+c \) . 若 \( a+b+c=0 \) ，\( f(0)>0 \) ，\( f(1)>0 \) ，求证: (1) a > 0 且 \( -2 < \frac{b}{a} < -1 \) ; (2) 方程 \( f(x)=0 \) 在 \( (0,1) \) 内有两个实根.
\end{problem}

\begin{problem}
如图，在四棱锥 \(P-ABCD\) 中，底面为直角梯形，\(AD // BC\)，\(\angle BAD = 90^\circ\)，\(PA \perp\) 底面 \(ABCD\)，且 \(PA = AD = AB = 2BC\)，\(M\)，\(N\) 分别为 \(PC\)，\(PB\) 的中点. (1) 求证: \( PB \perp DM \) ; (2) 求 CD 与平面 ADMN 所成的角.
\end{problem}

\begin{problem}
甲，乙两袋装有大小相同的红球和白球，甲袋装有 2 个红球，2 个白球，乙袋装有 2 个红球，\(n\) 个白球. 现从甲，乙两袋中各任取 2 个球. (1) 若 \(n = 3\)，求取到的 4 个球全是红球的概率; (2) 若取到的 4 个球中至少有 2 个红球的概率为 \(\frac{3}{4}\)，求 \(n\).
\end{problem}

\begin{problem}
如图，椭圆 \(\frac{x^2}{a^2} + \frac{y^2}{b} = 1 (n > b > 0)\) 与过点 \(A(2,0), B(0,1)\) 的直线有且只有一个公共点 \(T\)，且椭圆的离心率 \(\epsilon = \frac{\sqrt{3}}{2}\). (1) 求椭圆方程; (2) 设 \(F_{1}, F_{2}\) 分别为椭圆的左、右焦点，\(M\) 为线段 \(AF_{2}\) 的中点. 求证: \(\angle ATM = \angle AF_{1}T\).
\end{problem}

\begin{problem}
已知函数 \( f(x) = x^2 + x^2 \)，数列 \( \{x_n\} (x_n > 0) \) 的第一项 \( x_1 = 1 \)，以后各项按如下方式取定：曲线 \( y = f(x) \) 在 \( (x_{n+1}, f(x_{n+1})) \) 处的切线与经过(0,0)和 \( (x_n, f(x_n)) \) 两点的直线平行（如图）. 求证：当 \( n \in \mathbf{N}^* \) 时， (1) 求椭圆方程; (2) 设 \(F_{1}, F_{2}\) 分别为椭圆的左、右焦点，\(M\) 为线段 \(AF_{2}\) 的中点. 求证: \(\angle ATM = \angle AF_{1}T\). (1) \(x_{n}^{2} + x_{n} = 3x_{n + 1}^{2} + 2x_{n + 1};\) (2) \(\left(\frac{1}{2}\right)^{n - 1}\leqslant x_n\leqslant \left(\frac{1}{2}\right)^{n - 2}\).
\end{problem}
