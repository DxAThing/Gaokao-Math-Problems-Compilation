\section{单选题}
\begin{problem}
已知向量 \(\bs{a} = (\sqrt{3}, 1)\)，\(\bs{b}\) 是不平行于 \(x\) 轴的单位向量，且 \(\bs{a} \cdot \bs{b} = \sqrt{3}\)，则 \(\bs{b} =\)（\quad）

\choices
    {\(\left(\frac{\sqrt{3}}{2}, \frac{1}{2}\right)\)}
    {\(\left(\frac{1}{2}, \frac{\sqrt{3}}{2}\right)\)}
    {\(\left(\frac{1}{4}, \frac{3\sqrt{3}}{4}\right)\)}
    {\(\left(1,0\right)\)}
\end{problem}

\begin{problem}
若互相不等的实数 \(a\)，\(b\)，\(c\) 成等差数列，\(c\)，\(a\)，\(b\) 成等比数列，且 \(a + 3b + c = 10\)，则 \(a =\)（\quad）

    {\(2\)}
    {\(-2\)}
    {\(-4\)}
\end{problem}

\begin{problem}
关于直线 \(m\)，\(n\) 与平面 \(\alpha\)，\(\beta\)，有以下四个命题：
\begin{circlelist}
\item 若 \(m \parallel \alpha\)，\(n \parallel \beta\) 且 \(\alpha \parallel \beta\)，则 \(m \parallel n\)；
\item 若 \(m \perp \alpha\)，\(n \perp \beta\) 且 \(\alpha \perp \beta\)，则 \(m \perp n\)；
\item 若 \(m \perp \alpha\)，\(n \parallel \beta\) 且 \(\alpha \parallel \beta\)，则 \(m \perp n\)；
\item 若 \(m \parallel \alpha\)，\(n \perp \beta\) 且 \(\alpha \perp \beta\)，
\end{circlelist}
则 \(m \parallel n\). 其中真命题的序号是（\quad）

\choices
    {\circled{1}\circled{2}}
    {\circled{3}\circled{4}}
    {\circled{1}\circled{4}}
    {\circled{2}\circled{3}}
\end{problem}

\begin{problem}
设过点 \(P(x, y)\) 的直线分别与 \(x\) 轴的正半轴和 \(y\) 轴的正半轴交于 \(A\)，\(B\) 两点，点 \(Q\) 与点 \(P\) 关于 \(y\) 轴对称，\(O\) 为坐标原点，若 \(\overrightarrow{BP} = 2\overrightarrow{PA}\) 且 \(\overrightarrow{OQ} \cdot \overrightarrow{AB} = 1\)，则点 \(P\) 的轨迹方程是（\quad）

\choices
    {\(3x^{2} + \frac{3}{2} y^{2} = 1\)（\(x > 0\)，\(y > 0\)）}
    {\(3x^{2} - \frac{3}{2} y^{2} = 1\)（\(x > 0\)，\(y > 0\)）}
    {\(\frac{3}{2} x^{2} - 3y^{2} = 1\)（\(x > 0\)，\(y > 0\)）}
    {\(\frac{3}{2} x^{2} + 3y^{2} = 1\)（\(x > 0\)，\(y > 0\)）}
\end{problem}

\begin{problem}
有限集合 \(S\) 中元素的个数记作 \(\operatorname{card}(S)\). 设 \(A\)，\(B\) 都为有限集合，给出下列命题：
\begin{circlelist}
\item \(A \cap B = \varnothing\) 的充要条件是 \(\operatorname{card}(A \cup B) = \operatorname{card}(A) + \operatorname{card}(B)\)；
\item \(A \subseteq B\) 的必要条件是 \(\operatorname{card}(A) \leqslant \operatorname{card}(B)\)；
\item \(A \nsubseteq B\) 的充分条件是 \(\operatorname{card}(A) \leqslant \operatorname{card}(B)\)；
\item \(A = B\) 的充要条件是 \(\operatorname{card}(A) = \operatorname{card}(B)\).
\end{circlelist}
其中真命题的序号是（\quad）

\choices
    {\circled{1}\circled{2}}
    {\circled{3}\circled{4}}
    {\circled{1}\circled{4}}
    {\circled{2}\circled{3}}
\end{problem}

\end{problem}

\begin{problem}
关于 \(x\) 的方程 \((x^{2} - 1)^{2} - |x^{2} - 1| + k = 0\). 给出下列四个命题：

\begin{problem}
设函数 \(f(x) = \bs{a} \cdot (\bs{b} + \bs{c})\)，其中向量 \(\bs{a} = (\sin x, -\cos x)\)，\(\bs{b} = (\sin x, -3\cos x)\)，\(\bs{c} = (-\cos x, \sin x)\)，\(x \in \R\).

\begin{enumerate}
\item 求函数 \(f(x)\) 的最大值和最小正周期；
\item 将函数 \(f(x)\) 的图像按向量 \(\bs{d}\) 平移，使平移后得到的图像关于坐标原点成中心对称，求长度最小的 \(\bs{d}\).
\end{enumerate}
\end{problem}

\item 设 \(b_n = \frac{3}{a_n a_{n+1}}\)，\(T_n\) 是数列 \(\{b_n\}\) 的前 \(n\) 项和，求使得 \(T_n < \frac{m}{20}\) 对所有 \(n \in \mathbf{N}^*\) 都成立的最小正整数 \(m\).
\end{enumerate}
\end{problem}

\begin{problem}
设 \(x = 3\) 是函数 \(f(x) = (x^2 + ax + b)\e^{3 - x}\)（\(x \in \R\)）的一个极值点.

\begin{enumerate}
\item 求 \(a\) 与 \(b\) 的关系式（用 \(a\) 表示 \(b\)），并求 \(f(x)\) 的单调区间；
\item 设 \(a > 0\)，\(g(x) = \left(a^2 + \frac{25}{4}\right)\e^x\). 若存在 \(\xi_1\)，\(\xi_2 \in [0,4]\) 使得 \(\left|f(\xi_1) - g(\xi_2)\right| < 1\) 成立，求 \(a\) 的取值范围.
\end{enumerate}
\end{problem}
