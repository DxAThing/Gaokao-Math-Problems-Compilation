\section{单选题}

\begin{problem}
设集合 \(M = \{x \mid x^2 - x < 0\}\)，\(N = \{x \mid |x| < 2\}\)，则（\quad）

\choices
    {\(M \cap N = \varnothing\)}
    {\(M \cap N = M\)}
    {\(M \cup N = M\)}
    {\(M \cup N = \R\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
由 \(M=(0,1)\)，\(N=(-2,2)\)，得 \(M\cap N=M\)，故选 B.
\end{solution}

\begin{problem}
已知函数 \(y = \e^x\) 的图象与函数 \(y = f(x)\) 的图象关于直线 \(y = x\) 对称，则（\quad）

\choices
    {\(f(2x) = \e^{2x}(x \in \R)\)}
    {\(f(2x) = \ln 2\cdot\ln x(x > 0)\)}
    {\(f(2x) = 2\e^x(x \in \R)\)}
    {\(f(2x) = \ln x + \ln 2(x > 0)\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
互为反函数，故 \(f(x)=\ln x\)（\(x>0\)），从而 \(f(2x)=\ln x+\ln2\)，故选 D.
\end{solution}

\begin{problem}
双曲线 \(mx^2 + y^2 = 1\) 的虚轴长是实轴长的 \(2\) 倍，则 \(m=\)（\quad）

\choices
    {\(-\frac{1}{4}\)}
    {\(-4\)}
    {4}
    {\(\frac{1}{4}\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
双曲线为 \(y^2-(-m)x^2=1\)，其实轴半长为 \(1\)，虚轴半长为 \(1/\sqrt{-m}\). 由 \(1/\sqrt{-m}=2\)，得 \(m=-\frac14\)，故选 A.
\end{solution}

\begin{problem}
如果复数 \((m^2 + \i)(1 + m\i)\) 是实数，则实数 \(m=\)（\quad）

\choices
    {1}
    {\(-1\)}
    {\(\sqrt{2}\)}
    {\(-\sqrt{2}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
展开得 \((m^2-m)+(m^3+1)\i\). 令虚部为零，得 \(m^3+1=0\)，所以 \(m=-1\)，故选 B.
\end{solution}

\begin{problem}
函数 \(f(x)=\tan\left(x+\frac{\pi}{4}\right)\) 的单调增区间为（\quad）

\choices
    {\(\left(k\pi-\frac{\pi}{2},k\pi+\frac{\pi}{2}\right),\ k\in\Z\)}
    {\((k\pi,(k+1)\pi),\ k\in\Z\)}
    {\(\left(k\pi-\frac{3\pi}{4},k\pi+\frac{\pi}{4}\right),\ k\in\Z\)}
    {\(\left(k\pi-\frac{\pi}{4},k\pi+\frac{3\pi}{4}\right),\ k\in\Z\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
令 \(-\frac\pi2< x+\frac\pi4-k\pi<\frac\pi2\)，得单调增区间 \(\left(k\pi-\frac{3\pi}{4},k\pi+\frac\pi4\right)\)，故选 C.
\end{solution}

\begin{problem}
\(\triangle ABC\) 的内角 \(A,B,C\) 的对边分别为 \(a,b,c\). 若 \(a,b,c\) 成等比数列，且 \(c=2a\)，则 \(\cos B=\)（\quad）

\choices
    {\(\frac14\)}
    {\(\frac34\)}
    {\(\frac{\sqrt2}{4}\)}
    {\(\frac{\sqrt2}{3}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
由 \(b^2=ac=2a^2\) 得 \(b^2=2a^2\). 余弦定理给出 \(\cos B=\frac{a^2+c^2-b^2}{2ac}=\frac{3}{4}\)，故选 B.
\end{solution}

\begin{problem}
已知各顶点都在一个球面上的正四棱柱高为 \(4\)，体积为 \(16\)，则这个球的表面积是（\quad）

\choices
    {\(16\pi\)}
    {\(20\pi\)}
    {\(24\pi\)}
    {\(32\pi\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
设底面边长为 \(s\)，则 \(4s^2=16\)，得 \(s=2\). 球半径为正四棱柱体对角线的一半，即 \(R=\frac12\sqrt{2^2+2^2+4^2}=\sqrt6\)，故表面积为 \(4\pi R^2=24\pi\)，选 C.
\end{solution}

\begin{problem}
抛物线 \(y=-x^2\) 上的点到直线 \(4x+3y-8=0\) 距离的最小值是（\quad）

\choices
    {\(\frac43\)}
    {\(\frac75\)}
    {\(\frac85\)}
    {3}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
距离为 \(\frac{|4x-3x^2-8|}{5}=\frac{3x^2-4x+8}{5}\). 当 \(x=\frac23\) 时取最小值 \(\frac43\)，故选 A.
\end{solution}

\begin{problem}
设平面向量 \(\bs{a}_1,\bs{a}_2,\bs{a}_3\) 的和为 \(\bs{a}_1+\bs{a}_2+\bs{a}_3=0\). 如果向量 \(\bs{b}_1,\bs{b}_2,\bs{b}_3\) 满足 \(|\bs{b}_i|=2|\bs{a}_i|\)，且 \(\bs{a}_i\) 顺时针旋转 \(30^{\circ}\) 后与 \(\bs{b}_i\) 同向，其中 \(i=1,2,3\)，则（\quad）

\choices
    {\(-\bs{b}_1+\bs{b}_2+\bs{b}_3=0\)}
    {\(\bs{b}_1-\bs{b}_2+\bs{b}_3=0\)}
    {\(\bs{b}_1+\bs{b}_2-\bs{b}_3=0\)}
    {\(\bs{b}_1+\bs{b}_2+\bs{b}_3=0\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
顺时针旋转并按相同比例放大是线性变换，故 \(\bs{b}_1+\bs{b}_2+\bs{b}_3=2R_{-30^{\circ}}(\bs{a}_1+\bs{a}_2+\bs{a}_3)=0\)，选 D.
\end{solution}

\begin{problem}
设 \(a_n\) 是公差为正数的等差数列，若 \(a_1+a_2+a_3=15\)，\(a_1a_2a_3=80\)，则 \(a_{11}+a_{12}+a_{13}=\)（\quad）

\choices
    {120}
    {105}
    {90}
    {75}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
设公差为 \(d>0\)，则 \(a_2=5\)，且 \(5(25-d^2)=80\)，得 \(d=3\). 故 \(a_{12}=35\)，所以 \(a_{11}+a_{12}+a_{13}=3a_{12}=105\)，选 B.
\end{solution}

\begin{problem}
用长度分别为 \(2,3,4,5,6\)（单位：\(\mathrm{cm}\)）的 \(5\) 根细木棒围成一个三角形（允许连接，但不允许折断），能够得到的三角形的最大面积为（\quad）

\choices
    {\(8\sqrt5\mathrm{~cm}^{2}\)}
    {\(6\sqrt{10}\mathrm{~cm}^{2}\)}
    {\(3\sqrt{55}\mathrm{~cm}^{2}\)}
    {\(20\mathrm{~cm}^{2}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
将木棒分成三组，三边之和为 \(20\). 逐一枚举分组后的非退化三边，最均衡且面积最大的是 \(6,7,7\)（分别由 \(6\)、\(2+5\)、\(3+4\) 组成）. 其高为 \(\sqrt{7^2-3^2}=2\sqrt{10}\)，面积为 \(\frac12\cdot6\cdot2\sqrt{10}=6\sqrt{10}\)，选 B.
\end{solution}

\begin{problem}
设集合 \(I=\{1,2,3,4,5\}\). 选择 \(I\) 的两个非空子集 \(A\) 和 \(B\)，要使 \(B\) 中最小的数大于 \(A\) 中最大的数，则不同的选择方法共有（\quad）

\choices
    {50 种}
    {49 种}
    {48 种}
    {47 种}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
设 \(\max A=k\)，则 \(k=1,2,3,4\)，对应方法数为 \(2^{k-1}(2^{5-k}-1)\). 总数为 \(15+14+12+8=49\)，选 B.
\end{solution}

\section{填空题}

\begin{problem}
已知正四棱锥的体积为 \(12\)，底面对角线长为 \(2\sqrt6\)，则侧面与底面所成的二面角等于\fillinblank{}.
\end{problem}

\begin{answer}
\(60^{\circ}\)
\end{answer}

\begin{solution}
底面面积为 \(12\)，高为 \(3\). 取底边中点作截面，截面直角三角形两直角边为 \(3\) 和 \(\sqrt3\)，故二面角 \(\theta\) 满足 \(\cos\theta=\frac{\sqrt3}{2\sqrt3}=\frac12\)，得 \(\theta=60^{\circ}\).
\end{solution}

\begin{problem}
设 \(z=2y-x\)，式中变量 \(x,y\) 满足 \(\begin{cases}2x-y\geqslant-1,\\3x+2y\leqslant23,\\y\geqslant1.\end{cases}\) 则 \(z\) 的最大值为\fillinblank{}.
\end{problem}

\begin{answer}
11
\end{answer}

\begin{solution}
可行域顶点为 \((0,1),(7,1),(3,7)\)，目标函数在 \((3,7)\) 处取最大值 \(z=2\cdot7-3=11\).
\end{solution}

\begin{problem}
安排 \(7\) 位工作人员在 \(5\) 月 1 日至 \(5\) 月 7 日值班，每人值班一天，其中甲、乙二人都不安排在 5 月 1 日和 2 日，不同的安排方法共有\fillinblank{}种.（用数字作答）
\end{problem}

\begin{answer}
2400
\end{answer}

\begin{solution}
甲、乙分别从后五天中选值班日，有 \(5\cdot4\) 种；其余五人任意排列，有 \(5!\) 种，共 \(5\cdot4\cdot5!=2400\) 种.
\end{solution}

\begin{problem}
函数 \(f(x)=\cos(\sqrt3x+\varphi)\)（\(0<\varphi<\pi\)）. 若 \(f(x)+f'(x)\) 是奇函数，则 \(\varphi=\)\fillinblank{}.
\end{problem}

\begin{answer}
\(\frac\pi6\)
\end{answer}

\begin{solution}
\(f(x)+f'(x)=2\cos\left(\sqrt3x+\varphi+\frac\pi3\right)\). 要为奇函数需 \(\varphi+\frac\pi3=\frac\pi2+k\pi\)，结合范围得 \(\varphi=\frac\pi6\).
\end{solution}

\section{解答题}

\begin{problem}
\(\triangle ABC\) 的三个内角为 \(A,B,C\)，求当 \(A\) 为何值时，\(\cos A+2\cos\frac{B+C}{2}\) 取得最大值，并求出这个最大值.
\end{problem}

\begin{solution}
因 \(B+C=\pi-A\)，原式为 \(\cos A+2\sin\frac A2=1-2\sin^2\frac A2+2\sin\frac A2=\frac32-2\left(\sin\frac A2-\frac12\right)^2\). 故当 \(\sin\frac A2=\frac12\)，即 \(A=\frac\pi3\) 时取得最大值 \(\frac32\).
\end{solution}

\begin{problem}
\(A\)，\(B\) 是治疗同一种疾病的两种药，用若干试验组进行对比试验. 每个试验组由 \(4\) 只小白鼠组成，其中 \(2\) 只服用 \(A\)，另 \(2\) 只服用 \(B\)，然后观察疗效. 若在一个试验组中，服用 \(A\) 有效的小白鼠的只数比服用 \(B\) 有效的多，就称该试验组为甲类组. 设每只小白鼠服用 \(A\) 有效的概率为 \(\frac23\)，服用 \(B\) 有效的概率为 \(\frac12\).

\begin{enumerate}
\item 求一个试验组为甲类组的概率.
\item 观察 \(3\) 个试验组，用 \(\xi\) 表示这 \(3\) 个试验组中甲类组的个数，求 \(\xi\) 的分布列和数学期望.
\end{enumerate}
\end{problem}

\begin{solution}
（1）设服用 \(A\)、\(B\) 有效的只数分别为 \(X,Y\)，则 \(X\sim B(2,\frac23)\)，\(Y\sim B(2,\frac12)\). 甲类组概率为
\(P(X>Y)=P(X=1,Y=0)+P(X=2,Y\leqslant1)=\frac19+\frac13=\frac49\).

（2）令 \(p=\frac49\)，则 \(\xi\sim B(3,p)\)，分布列为 \(P(\xi=k)=\mathrm C_3^k(\frac49)^k(\frac59)^{3-k}\)（\(k=0,1,2,3\)），即依次为 \(\frac{125}{729},\frac{100}{243},\frac{80}{243},\frac{64}{729}\)，且 \(E\xi=3p=\frac43\).
\end{solution}

\begin{problem}
如图，\(l_1,l_2\) 是互相垂直的异面直线，\(MN\) 是它们的公垂线段. 点 \(A,B\) 在 \(l_1\) 上，\(C\) 在 \(l_2\) 上，\(AM=MB=MN\).

\begin{enumerate}
\item 证明：\(AC\perp NB\).
\item 若 \(\angle ACB=60^{\circ}\)，求 \(NB\) 与平面 \(ABC\) 所成角的余弦值.
\end{enumerate}

\begin{center}
\bitmapfigure[max width=0.38\linewidth,max totalheight=4.8cm]{img/2006/national_paper_1_science/q19_fig1.png}
\end{center}
\end{problem}

\begin{solution}
（1）建立直角坐标系，令 \(M=(0,0,0)\)，\(N=(0,1,0)\)，\(A=(-1,0,0)\)，\(B=(1,0,0)\)，\(C=(0,1,t)\). 则 \(\overrightarrow{AC}=(1,1,t)\)，\(\overrightarrow{NB}=(1,-1,0)\)，故 \(\overrightarrow{AC}\cdot\overrightarrow{NB}=0\)，从而 \(AC\perp NB\).

（2）又 \(\cos\angle ACB=\frac{t^2}{t^2+2}=\frac12\)，得 \(t^2=2\). 平面 \(ABC\) 的法向量可取 \((0,-t,1)\)，故 \(\sin\theta=\frac{|\overrightarrow{NB}\cdot(0,-t,1)|}{|NB|\sqrt{t^2+1}}=\frac1{\sqrt3}\)，其中 \(\theta\) 为所成角，因而 \(\cos\theta=\sqrt{\frac23}=\frac{\sqrt6}{3}\).
\end{solution}

\begin{problem}
在平面直角坐标系 \(xOy\) 中，有一个以 \(F_1(0,-\sqrt3)\) 和 \(F_2(0,\sqrt3)\) 为焦点、离心率为 \(\frac{\sqrt3}{2}\) 的椭圆，设椭圆在第一象限的部分为曲线 \(C\)，动点 \(P\) 在 \(C\) 上，\(C\) 在点 \(P\) 处的切线与 \(x,y\) 轴的交点分别为 \(A,B\)，且向量 \(\overrightarrow{OM}=\overrightarrow{OA}+\overrightarrow{OB}\). 求：

\begin{enumerate}
\item 点 \(M\) 的轨迹方程.
\item \(|\overrightarrow{OM}|\) 的最小值.
\end{enumerate}
\end{problem}

\begin{solution}
（1）由 \(c=\sqrt3\)、\(e=\frac{c}{a}=\frac{\sqrt3}{2}\)，得 \(a=2,b=1\)，椭圆方程为 \(\frac{x^2}{4}+y^2=1\). 设 \(P=(x_P,y_P)\)（第一象限），切线为 \(\frac{xx_P}{4}+yy_P=1\)，故 \(A=(\frac4{x_P},0)\)，\(B=(0,\frac1{y_P})\). 令 \(M=(u,v)\)，消去 \(x_P,y_P\) 得
\(u^2v^2-u^2-4v^2=0\)，其中 \(u>2\)、\(v>1\).

（2）又 \(|OM|^2=\frac{16}{x_P^2}+\frac1{y_P^2}\)，令 \(t=x_P^2\)，则 \(|OM|^2=\frac{16}{t}+\frac4{4-t}\)，在 \(t=\frac83\) 时取得最小值 \(9\)，故 \(|OM|_{\min}=3\).
\end{solution}

\begin{problem}
已知函数 \(f(x)=\frac{1+x}{1-x}\e^{-ax}\).

\begin{enumerate}
\item 设 \(a>0\)，讨论 \(f(x)\) 的单调性.
\item 若对任意 \(x\in(0,1)\) 恒有 \(f(x)>1\)，求 \(a\) 的取值范围.
\end{enumerate}
\end{problem}

\begin{solution}
（1）函数定义域为 \(\R\setminus\{1\}\)，且
\(f'(x)=\frac{\e^{-ax}}{(1-x)^2}\bigl(ax^2+2-a\bigr)\).

当 \(0<a\leqslant2\) 时，\(f'(x)\geqslant0\)，故在 \((-\infty,1)\)、\((1,+\infty)\) 上均递增. 当 \(a>2\) 时，令 \(r=\sqrt{1-\frac2a}\)，则在 \((-\infty,-r)\)、\((r,1)\)、\((1,+\infty)\) 上递增，在 \((-r,r)\) 上递减.

（2）对 \(x\in(0,1)\)，不等式等价于 \(a<\frac1x\ln\frac{1+x}{1-x}\). 函数 \(\frac1x\ln\frac{1+x}{1-x}\) 在 \((0,1)\) 上递增且下确界为 \(2\)，故 \(0<a\leqslant2\).
\end{solution}

\begin{problem}
设数列 \(a_n\) 的前 \(n\) 项的和 \(S_n=\frac43a_n-\frac13\times2^{n+1}+\frac23\)，\(n=1,2,3,\ldots\).

\begin{enumerate}
\item 求首项 \(a_1\) 与通项 \(a_n\).
\item 设 \(T_n=\frac{2^n}{S_n}\)，\(n=1,2,3,\ldots\)，证明：\(\sum_{i=1}^nT_i<\frac32\).
\end{enumerate}
\end{problem}

\begin{solution}
（1）取 \(n=1\) 得 \(a_1=2\). 由 \(S_n-S_{n-1}=a_n\) 得 \(a_n=4a_{n-1}+2^n\)，解得 \(a_n=4^n-2^n\). 

（2）由此 \(S_n=\frac{2(2^n-1)(2^{n+1}-1)}{3}\)，故
\(T_n=\frac32\left(\frac1{2^n-1}-\frac1{2^{n+1}-1}\right)\). 于是
\(\sum_{i=1}^nT_i=\frac32\left(1-\frac1{2^{n+1}-1}\right)<\frac32\).
\end{solution}
