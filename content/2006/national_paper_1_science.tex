\section{单选题}

\begin{problem}
设集合 \(M = \{x \mid x^2 - x < 0\}\)，\(N = \{x \mid |x| < 2\}\)，则（\quad）

\choices
    {\(M \cap N = \varnothing\)}
    {\(M \cap N = M\)}
    {\(M \cup N = M\)}
    {\(M \cup N = \R\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
由 \(x^2-x<0\) 得 \(x(x-1)<0\)，所以 \(M=(0,1)\)；又 \(N=(-2,2)\)，从而 \(M\subset N\)，故 \(M\cap N=M\)，选 B.
\end{solution}

\begin{problem}
已知函数 \(y = \e^x\) 的图象与函数 \(y = f(x)\) 的图象关于直线 \(y = x\) 对称，则（\quad）

\choices
    {\(f(2x) = \e^{2x}\)（\(x \in \R\)）}
    {\(f(2x) = \ln 2\cdot\ln x\)（\(x > 0\)）}
    {\(f(2x) = 2\e^x\)（\(x \in \R\)）}
    {\(f(2x) = \ln x + \ln 2\)（\(x > 0\)）}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
关于直线 \(y=x\) 对称的两个函数图象互为反函数. 函数 \(y=\e^x\) 的反函数为 \(y=\ln x\)（\(x>0\)），所以 \(f(x)=\ln x\)（\(x>0\)）. 因此 \(f(2x)=\ln(2x)=\ln x+\ln2\)（\(x>0\)），选 D.
\end{solution}

\begin{problem}
双曲线 \(mx^2 + y^2 = 1\) 的虚轴长是实轴长的 \(2\) 倍，则 \(m=\)（\quad）

\choices
    {\(-\frac{1}{4}\)}
    {\(-4\)}
    {\(4\)}
    {\(\frac{1}{4}\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
因为曲线是双曲线，所以 \(m<0\)，原方程可写为 \(\dfrac{y^2}{1}-\dfrac{x^2}{1/(-m)}=1\). 因此实半轴长为 \(1\)，虚半轴长为 \(\dfrac{1}{\sqrt{-m}}\)，实轴长和虚轴长分别为 \(2\) 和 \(\dfrac{2}{\sqrt{-m}}\). 由题意，\(\dfrac{2}{\sqrt{-m}}=2\times2\)，所以 \(\sqrt{-m}=\dfrac12\)，从而 \(m=-\dfrac14\)，选 A.
\end{solution}

\begin{problem}
如果复数 \((m^2 + \i)(1 + m\i)\) 是实数，则实数 \(m=\)（\quad）

\choices
    {\(1\)}
    {\(-1\)}
    {\(\sqrt{2}\)}
    {\(-\sqrt{2}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
展开得 \((m^2+\i)(1+m\i)=m^2+m^3\i+\i+m\i^2=(m^2-m)+(m^3+1)\i\). 复数为实数，当且仅当虚部为零，即 \(m^3+1=0\)，所以 \(m=-1\)，选 B.
\end{solution}

\begin{problem}
函数 \(f(x)=\tan\left(x+\frac{\pi}{4}\right)\) 的单调增区间为（\quad）

\choices
    {\(\left(k\pi-\frac{\pi}{2},k\pi+\frac{\pi}{2}\right)\)（\(k\in\Z\)）}
    {\((k\pi,(k+1)\pi)\)（\(k\in\Z\)）}
    {\(\left(k\pi-\frac{3\pi}{4},k\pi+\frac{\pi}{4}\right)\)（\(k\in\Z\)）}
    {\(\left(k\pi-\frac{\pi}{4},k\pi+\frac{3\pi}{4}\right)\)（\(k\in\Z\)）}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
函数在有定义的每个区间上满足 \(f'(x)=\sec^2\left(x+\dfrac{\pi}{4}\right)>0\). 其定义区间由 \(x+\dfrac{\pi}{4}\in\left(k\pi-\dfrac{\pi}{2},k\pi+\dfrac{\pi}{2}\right)\) 给出，等价于 \(x\in\left(k\pi-\dfrac{3\pi}{4},k\pi+\dfrac{\pi}{4}\right)\)，其中 \(k\in\Z\). 故选 C.
\end{solution}

\begin{problem}
\(\triangle ABC\) 的内角 \(A\)，\(B\)，\(C\) 的对边分别为 \(a\)，\(b\)，\(c\). 若 \(a\)，\(b\)，\(c\) 成等比数列，且 \(c=2a\)，则 \(\cos B=\)（\quad）

\choices
    {\(\frac14\)}
    {\(\frac34\)}
    {\(\frac{\sqrt2}{4}\)}
    {\(\frac{\sqrt2}{3}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
因为 \(a\)，\(b\)，\(c\) 成等比数列，所以 \(b^2=ac=2a^2\). 由余弦定理，
\[
\cos B=\frac{a^2+c^2-b^2}{2ac}=\frac{a^2+4a^2-2a^2}{2a\cdot2a}=\frac34
\]
故选 B.
\end{solution}

\begin{problem}
已知各顶点都在一个球面上的正四棱柱高为 \(4\)，体积为 \(16\)，则这个球的表面积是（\quad）

\choices
    {\(16\pi\)}
    {\(20\pi\)}
    {\(24\pi\)}
    {\(32\pi\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
设正四棱柱的底面边长为 \(s\). 由体积等于底面积乘高，得 \(s^2\cdot4=16\)，所以 \(s=2\). 正四棱柱的外接球直径等于其体对角线长，因此球半径
\[
R=\frac12\sqrt{s^2+s^2+4^2}=\frac12\sqrt{4+4+16}=\sqrt6
\]
所以球的表面积为 \(4\pi R^2=4\pi\cdot6=24\pi\)，选 C.
\end{solution}

\begin{problem}
抛物线 \(y=-x^2\) 上的点到直线 \(4x+3y-8=0\) 距离的最小值是（\quad）

\choices
    {\(\frac43\)}
    {\(\frac75\)}
    {\(\frac85\)}
    {\(3\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
设抛物线上点的坐标为 \((x,-x^2)\). 它到直线的距离为
\[
d=\frac{|4x+3(-x^2)-8|}{\sqrt{4^2+3^2}}=\frac{|-3x^2+4x-8|}{5}
\]
由于 \(-3x^2+4x-8<0\)（其判别式为 \(4^2-4\cdot(-3)\cdot(-8)=-80<0\)，且二次项系数为负），所以 \(d=\dfrac{3x^2-4x+8}{5}=\dfrac{3\left(x-\frac23\right)^2+\frac{20}{3}}{5}\geq\dfrac43\). 故选 A.
\end{solution}

\begin{problem}
设平面向量 \(\bs{a}_1\)，\(\bs{a}_2\)，\(\bs{a}_3\) 的和为 \(\bs{a}_1+\bs{a}_2+\bs{a}_3=0\). 如果向量 \(\bs{b}_1\)，\(\bs{b}_2\)，\(\bs{b}_3\) 满足 \(|\bs{b}_i|=2|\bs{a}_i|\)，且 \(\bs{a}_i\) 顺时针旋转 \(30^{\circ}\) 后与 \(\bs{b}_i\) 同向，其中 \(i=1\)，\(2\)，\(3\)，则（\quad）

\choices
    {\(-\bs{b}_1+\bs{b}_2+\bs{b}_3=0\)}
    {\(\bs{b}_1-\bs{b}_2+\bs{b}_3=0\)}
    {\(\bs{b}_1+\bs{b}_2-\bs{b}_3=0\)}
    {\(\bs{b}_1+\bs{b}_2+\bs{b}_3=0\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
记“顺时针旋转 \(30^\circ\)”对应的旋转变换为 \(T\). 由长度关系和同向条件，\(\bs{b}_i=2T(\bs{a}_i)\)（\(i=1\)，\(2\)，\(3\)）. 旋转和数乘都是线性变换，所以
\[
\bs{b}_1+\bs{b}_2+\bs{b}_3=2T(\bs{a}_1)+2T(\bs{a}_2)+2T(\bs{a}_3)=2T(\bs{a}_1+\bs{a}_2+\bs{a}_3)=0
\]
故选 D.
\end{solution}

\begin{problem}
设 \(a_n\) 是公差为正数的等差数列，若 \(a_1+a_2+a_3=15\)，\(a_1a_2a_3=80\)，则 \(a_{11}+a_{12}+a_{13}=\)（\quad）

\choices
    {\(120\)}
    {\(105\)}
    {\(90\)}
    {\(75\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
设公差为 \(d>0\). 由 \(a_1+a_2+a_3=15\) 得 \(a_2=5\)，于是 \(a_1=5-d\)，\(a_3=5+d\). 由乘积条件得 \(5(5-d)(5+d)=80\)，即 \(25-d^2=16\)，从而 \(d=3\). 因此 \(a_{11}=5+9\cdot3=32\)，\(a_{12}=35\)，\(a_{13}=38\)，所以 \(a_{11}+a_{12}+a_{13}=105\)，选 B.
\end{solution}

\begin{problem}
用长度分别为 \(2\)，\(3\)，\(4\)，\(5\)，\(6\)（单位：\(\mathrm{cm}\)）的 \(5\) 根细木棒围成一个三角形（允许连接，但不允许折断），能够得到的三角形的最大面积为（\quad）

\choices
    {\(8\sqrt5\mathrm{~cm}^{2}\)}
    {\(6\sqrt{10}\mathrm{~cm}^{2}\)}
    {\(3\sqrt{55}\mathrm{~cm}^{2}\)}
    {\(20\mathrm{~cm}^{2}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
把木棒连接后，每根木棒完整地属于三角形的一条边，所以要把五根木棒分成三组，每组长度和就是一条边长. 五根木棒总长为 \(20\)，因而三角形成立的充要条件是最大边长小于 \(10\). 固定含有 \(6\mathrm{~cm}\) 木棒的一组，并把同一分组中三边的交换视为同一种分组，逐一枚举其余木棒后，满足最大边长小于 \(10\) 的分组恰为
\[
\begin{gathered}
2\mid(3+6)\mid(4+5),\\
(2+6)\mid3\mid(4+5),\\
(2+5)\mid(3+6)\mid4,\\
(2+6)\mid(3+5)\mid4,\\
(2+4)\mid(3+6)\mid5,\\
(2+6)\mid(3+4)\mid5,\\
(2+3)\mid(4+5)\mid6,\\
(2+4)\mid(3+5)\mid6,\\
(2+5)\mid(3+4)\mid6,\\
(2+3+4)\mid5\mid6.
\end{gathered}
\]
这里竖线表示不同的三条边. 上述枚举按含有 \(6\mathrm{~cm}\) 木棒的一组逐一列出，其他分组均有一条边长不小于 \(10\)，不能组成三角形. 去除得到相同三边长的重复分组后，所有可能的三边为
\[
\left\{\left(2,9,9\right),\left(3,8,9\right),\left(4,7,9\right),\left(4,8,8\right),\left(5,6,9\right),\left(5,7,8\right),\left(6,6,8\right),\left(6,7,7\right)\right\}
\]
令三边为 \(p\)，\(q\)，\(r\)，则半周长 \(s=10\)，由海伦公式 \(\Delta^2=s(s-p)(s-q)(s-r)\). 八组边长对应的面积平方分别为
\[
\begin{gathered}
10\cdot8\cdot1\cdot1=80,\\
10\cdot7\cdot2\cdot1=140,\\
10\cdot6\cdot3\cdot1=180,\\
10\cdot6\cdot2\cdot2=240,\\
10\cdot5\cdot4\cdot1=200,\\
10\cdot5\cdot3\cdot2=300,\\
10\cdot4\cdot4\cdot2=320,\\
10\cdot4\cdot3\cdot3=360.
\end{gathered}
\]
因此面积最大时 \(\Delta=\sqrt{360}=6\sqrt{10}\)，选 B.
\end{solution}

\begin{problem}
设集合 \(I=\{1,2,3,4,5\}\). 选择 \(I\) 的两个非空子集 \(A\) 和 \(B\)，要使 \(B\) 中最小的数大于 \(A\) 中最大的数，则不同的选择方法共有（\quad）

\choices
    {\(50\) 种}
    {\(49\) 种}
    {\(48\) 种}
    {\(47\) 种}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
设 \(k=\max A\). 由于 \(\min B>k\)，所以 \(k\) 只能取 \(1\)，\(2\)，\(3\)，\(4\). 当 \(\max A=k\) 时，\(A\) 中小于 \(k\) 的元素可任意选取，共有 \(2^{k-1}\) 种；\(B\) 是 \(\{k+1,\ldots,5\}\) 的非空子集，共有 \(2^{5-k}-1\) 种. 因此选择方法总数为
\[
\sum_{k=1}^{4}2^{k-1}\left(2^{5-k}-1\right)=15+14+12+8=49
\]
故选 B.
\end{solution}

\begin{problem}
已知正四棱锥的体积为 \(12\)，底面对角线长为 \(2\sqrt6\)，则侧面与底面所成的二面角等于\fillinblank{}.
\end{problem}

\begin{answer}
\(\dfrac{\pi}{3}\)
\end{answer}

\begin{solution}
设正四棱锥的底面边长为 \(s\)，高为 \(h\). 由底面对角线为 \(2\sqrt6\)，得 \(s\sqrt2=2\sqrt6\)，所以 \(s=2\sqrt3\)，底面积为 \(12\). 由体积公式 \(\dfrac13\cdot12\cdot h=12\)，得 \(h=3\). 设 \(S\) 为顶点，\(O\) 为底面中心，\(M\) 为底边中点，则 \(SO\perp\) 底面，且 \(OM=\dfrac{s}{2}=\sqrt3\). 在截面 \(SOM\) 中，\(SO=3\)，\(OM=\sqrt3\)，所以 \(SM=2\sqrt3\). 侧面与底面所成的二面角等于 \(\angle SMO\)，于是 \(\cos\angle SMO=\dfrac{OM}{SM}=\dfrac12\)，故该二面角为 \(\dfrac{\pi}{3}\).
\end{solution}

\begin{problem}
设 \(z=2y-x\)，式中变量 \(x\)，\(y\) 满足下列条件：\(\begin{cases}
2x-y\geqslant-1,\\
3x+2y\leqslant23,\\
y\geqslant1.
\end{cases}\) 则 \(z\) 的最大值为\fillinblank{}.
\end{problem}

\begin{answer}
\(11\)
\end{answer}

\begin{solution}
三个约束给出 \(y\leq2x+1\)，\(y\leq\dfrac{23-3x}{2}\)，\(y\geq1\). 由 \(1\leq2x+1\) 和 \(1\leq\dfrac{23-3x}{2}\)，可得 \(0\leq x\leq7\)，所以可行域是由直线 \(y=1\)、\(y=2x+1\) 和 \(y=\dfrac{23-3x}{2}\) 围成的三角形. 两条上界的交点满足 \(2x+1=\dfrac{23-3x}{2}\)，解得 \((x,y)=(3,7)\)；与 \(y=1\) 的两个交点分别为 \((0,1)\) 和 \((7,1)\). 目标函数 \(z=2y-x\) 在这三个顶点的值依次为 \(-1\)，\(11\)，\(-5\). 由于线性目标函数在有界多边形上的最大值必在顶点取得，所以 \(z\) 的最大值为 \(11\).
\end{solution}

\begin{problem}
安排 \(7\) 位工作人员在 \(5\) 月 1 日至 \(5\) 月 7 日值班，每人值班一天，其中甲，乙二人都不安排在 5 月 1 日和 2 日，不同的安排方法共有\fillinblank{}种. （用数字作答）
\end{problem}

\begin{answer}
\(2400\)
\end{answer}

\begin{solution}
甲、乙的值班日期都只能从 5 月 3 日至 5 月 7 日中选取. 先安排甲、乙，甲有 \(5\) 种选择，乙有 \(4\) 种选择；其余 \(5\) 位工作人员再安排到剩余的 \(5\) 天，共有 \(5!\) 种. 因此安排方法共有 \(5\cdot4\cdot5!=2400\) 种.
\end{solution}

\begin{problem}
函数 \(f(x)=\cos(\sqrt3x+\varphi)\)（\(0<\varphi<\pi\)）. 若 \(f(x)+f'(x)\) 是奇函数，则 \(\varphi=\)\fillinblank{}.
\end{problem}

\begin{answer}
\(\dfrac{\pi}{6}\)
\end{answer}

\begin{solution}
令 \(t=\sqrt3x+\varphi\)，则 \(f'(x)=-\sqrt3\sin t\)，从而
\[
f(x)+f'(x)=\cos t-\sqrt3\sin t=2\cos\left(\sqrt3x+\varphi+\frac{\pi}{3}\right)
\]
函数 \(2\cos(\sqrt3x+\alpha)\) 为奇函数，当且仅当 \(\cos\alpha=0\)，所以 \(\alpha=\dfrac{\pi}{2}+k\pi\)，即 \(\varphi=\dfrac{\pi}{6}+k\pi\). 结合 \(0<\varphi<\pi\)，只能取 \(\varphi=\dfrac{\pi}{6}\).
\end{solution}

\begin{problem}
\(\triangle ABC\) 的三个内角为 \(A\)，\(B\)，\(C\)，求当 \(A\) 为何值时，\(\cos A+2\cos\frac{B+C}{2}\) 取得最大值，并求出这个最大值.
\end{problem}

\begin{solution}
因为 \(A+B+C=\pi\)，所以
\[
\cos A+2\cos\frac{B+C}{2}=\cos A+2\cos\left(\frac{\pi-A}{2}\right)=\cos A+2\sin\frac A2
\]
令 \(u=\sin\dfrac A2\). 由于三角形内角满足 \(0<A<\pi\)，所以 \(0<u<1\)，且 \(\cos A=1-2u^2\). 于是原式
\[
1-2u^2+2u=\frac32-2\left(u-\frac12\right)^2\leq\frac32
\]
当且仅当 \(u=\dfrac12\) 时等号成立，即 \(\sin\dfrac A2=\dfrac12\). 由 \(0<\dfrac A2<\dfrac{\pi}{2}\)，得 \(\dfrac A2=\dfrac{\pi}{6}\)，所以 \(A=\dfrac{\pi}{3}\). 此时可取 \(B+C=\dfrac{2\pi}{3}\)，确为三角形内角，故最大值为 \(\dfrac32\).
\end{solution}

\begin{problem}
A，B 是治疗同一种疾病的两种药，用若干试验组进行对比试验. 每个试验组由 \(4\) 只小白鼠组成，其中 \(2\) 只服用 A，另 \(2\) 只服用 B，然后观察疗效. 若在一个试验组中，服用 A 有效的小白鼠的只数比服用 B 有效的多，就称该试验组为甲类组. 设每只小白鼠服用 A 有效的概率为 \(\frac23\)，服用 B 有效的概率为 \(\frac12\).

\begin{enumerate}
\item 求一个试验组为甲类组的概率.
\item 观察 \(3\) 个试验组，用 \(\xi\) 表示这 \(3\) 个试验组中甲类组的个数，求 \(\xi\) 的分布列和数学期望.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设 \(X\) 为服用药物 A 且有效的小白鼠只数，\(Y\) 为服用药物 B 且有效的小白鼠只数，则 \(X\sim B\left(2,\dfrac23\right)\)，\(Y\sim B\left(2,\dfrac12\right)\)，且 \(X\)，\(Y\) 相互独立. 甲类组的条件是 \(X>Y\)，所以
\[
P(X>Y)=P(Y=0)P(X\geq1)+P(Y=1)P(X=2)=\frac14\left(1-\frac19\right)+\frac12\cdot\frac49=\frac49
\]
\item 不同试验组之间相互独立，因此 \(\xi\sim B\left(3,\dfrac49\right)\). 当 \(k=0\)，\(1\)，\(2\)，\(3\) 时，
\[
P(\xi=k)=\mathrm C_3^k\left(\frac49\right)^k\left(\frac59\right)^{3-k}
\]
具体地，\(P(\xi=0)=\dfrac{125}{729}\)，\(P(\xi=1)=\dfrac{100}{243}\)，\(P(\xi=2)=\dfrac{80}{243}\)，\(P(\xi=3)=\dfrac{64}{729}\)，这些概率之和为 \(1\). 所以 \(E\xi=3\cdot\dfrac49=\dfrac43\).
\end{enumerate}
\end{solution}

\begin{problem}
如图，\(l_1\)，\(l_2\) 是互相垂直的异面直线，\(MN\) 是它们的公垂线段. 点 \(A\)，\(B\) 在 \(l_1\) 上，\(C\) 在 \(l_2\) 上，\(AM=MB=MN\).

\begin{enumerate}
\item 证明：\(AC\perp NB\).
\item 若 \(\angle ACB=60^{\circ}\)，求 \(NB\) 与平面 \(ABC\) 所成角的余弦值.
\end{enumerate}

\bitmapfigure[float=inline,width=5.4cm]{img/2006/national_paper_1_science/q19_fig1.png}
\end{problem}

\begin{solution}
取 \(MN\) 为 \(z\) 轴，令 \(M=(0,0,0)\)，\(N=(0,0,1)\)，令 \(l_1\) 为 \(x\) 轴，\(l_2\) 为过 \(N\) 且平行于 \(y\) 轴的直线. 由 \(AM=MB=MN=1\)，可取 \(A=(-1,0,0)\)，\(B=(1,0,0)\)；设 \(C=(0,t,1)\).

\begin{enumerate}
\item \(\overrightarrow{AC}=(1,t,1)\)，\(\overrightarrow{NB}=(1,0,-1)\)，所以 \(\overrightarrow{AC}\cdot\overrightarrow{NB}=1+0-1=0\)，从而 \(AC\perp NB\).
\item \(\overrightarrow{CA}=(-1,-t,-1)\)，\(\overrightarrow{CB}=(1,-t,-1)\)，于是
\[
\cos\angle ACB=\frac{\overrightarrow{CA}\cdot\overrightarrow{CB}}{|\overrightarrow{CA}||\overrightarrow{CB}|}=\frac{t^2}{t^2+2}
\]
由 \(\angle ACB=60^\circ\)，得 \(\dfrac{t^2}{t^2+2}=\dfrac12\)，所以 \(t^2=2\).

平面 \(ABC\) 的一个法向量可取
\[
\overrightarrow{AB}\times\overrightarrow{AC}=(2,0,0)\times(1,t,1)=(0,-2,2t)
\]
设 \(NB\) 与平面 \(ABC\) 所成的角为 \(\theta\)，则
\[
\sin\theta=\frac{\left|\overrightarrow{NB}\cdot(0,-2,2t)\right|}{|\overrightarrow{NB}|\sqrt{(-2)^2+(2t)^2}}=\frac{|2t|}{\sqrt2\sqrt{4+4t^2}}=\frac1{\sqrt3}
\]
因此 \(\cos\theta=\sqrt{1-\dfrac13}=\dfrac{\sqrt6}{3}\).
\end{enumerate}
\end{solution}

\begin{problem}
在平面直角坐标系 \(xOy\) 中，有一个以 \(F_1(0,-\sqrt3)\) 和 \(F_2(0,\sqrt3)\) 为焦点，离心率为 \(\frac{\sqrt3}{2}\) 的椭圆，设椭圆在第一象限的部分为曲线 \(C\)，动点 \(P\) 在 \(C\) 上，\(C\) 在点 \(P\) 处的切线与 \(x\)，\(y\) 轴的交点分别为 \(A\)，\(B\)，且向量 \(\overrightarrow{OM}=\overrightarrow{OA}+\overrightarrow{OB}\). 求：

\begin{enumerate}
\item 点 \(M\) 的轨迹方程.
\item \(|\overrightarrow{OM}|\) 的最小值.
\end{enumerate}
\end{problem}

\begin{solution}
由焦点坐标可知椭圆的半焦距为 \(c=\sqrt3\). 又离心率 \(e=\dfrac ca=\dfrac{\sqrt3}{2}\)，所以长半轴 \(a=2\)，短半轴满足 \(b^2=a^2-c^2=1\). 椭圆方程为 \(\dfrac{x^2}{4}+y^2=1\).

设 \(P=(x_0,y_0)\) 为第一象限内的点，则 \(0<x_0<2\)，\(0<y_0<1\). 在 \(P\) 处的切线方程为 \(\frac{x_0x}{4}+y_0y=1\).
所以切线与两坐标轴的交点分别为 \(A=\left(\dfrac4{x_0},0\right)\)，\(B=\left(0,\dfrac1{y_0}\right)\). 设 \(M=(u,v)\)，由 \(\overrightarrow{OM}=\overrightarrow{OA}+\overrightarrow{OB}\) 得 \(u=\dfrac4{x_0}\)，\(v=\dfrac1{y_0}\). 将 \(x_0=\dfrac4u\)，\(y_0=\dfrac1v\) 代入椭圆方程，得到
\[
\frac4{u^2}+\frac1{v^2}=1
\]
且 \(u>2\)，\(v>1\). 反之，若 \(u>2\)，\(v>1\) 且满足该方程，令 \(x_0=\dfrac4u\)，\(y_0=\dfrac1v\)，则 \(0<x_0<2\)，\(0<y_0<1\)，并且 \(\dfrac{x_0^2}{4}+y_0^2=1\)，故 \((x_0,y_0)\) 是曲线 \(C\) 上的点，且相应切线的两个截距之和正是 \((u,v)\). 因此点 \(M\) 的轨迹方程为上述方程.

令 \(p=u^2\)，\(q=v^2\)，则 \(p>4\)，\(q>1\)，且 \(\dfrac4p+\dfrac1q=1\)，从而 \(q=\dfrac p{p-4}\). 于是
\[
|\overrightarrow{OM}|^2=p+q=p+\frac p{p-4}=5+(p-4)+\frac4{p-4}\geq5+4=9
\]
其中等号当且仅当 \(p-4=2\)，即 \(p=6\)，此时 \(q=3\)，满足 \(u>2\)，\(v>1\). 所以 \(|\overrightarrow{OM}|\) 的最小值为 \(3\).
\end{solution}

\begin{problem}
已知函数 \(f(x)=\frac{1+x}{1-x}\e^{-ax}\).

\begin{enumerate}
\item 设 \(a>0\)，讨论 \(y=f(x)\) 的单调性.
\item 若对任意 \(x\in(0,1)\) 恒有 \(f(x)>1\)，求 \(a\) 的取值范围.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 函数定义域为 \(x\in\R\) 且 \(x\ne1\). 求导得
\[
f'(x)=\e^{-ax}\left[\frac2{(1-x)^2}-\frac{a(1+x)}{1-x}\right]=\frac{\e^{-ax}}{(1-x)^2}\left(2-a+ax^2\right)
\]
由于 \(\e^{-ax}>0\)，\((1-x)^2>0\)，故 \(f'(x)\) 的符号由 \(2-a+ax^2\) 决定.

当 \(0<a<2\) 时，\(2-a+ax^2>0\)；当 \(a=2\) 时，\(2-a+ax^2=2x^2\)，只在 \(x=0\) 处为零. 因此在任意两个属于同一连通定义区间的不同点之间，导数除至多一个点外均为正，\(f(x)\) 在 \((-\infty,1)\) 和 \((1,+\infty)\) 上均为增函数.

当 \(a>2\) 时，令 \(\alpha=\sqrt{1-\dfrac2a}\)，则 \(0<\alpha<1\)，并且
\(2-a+ax^2=a(x-\alpha)(x+\alpha)\). 因此 \(f(x)\) 在 \((-\infty,-\alpha)\)，\((\alpha,1)\)，\((1,+\infty)\) 上均为增函数，在 \((-\alpha,\alpha)\) 上为减函数.
\item 对 \(x\in(0,1)\)，原不等式等价于 \(\ln\frac{1+x}{1-x}-ax>0\).
令 \(h(x)=\dfrac1x\ln\dfrac{1+x}{1-x}\). 先证 \(h(x)>2\)：令 \(g(x)=\ln\dfrac{1+x}{1-x}-2x\)，则 \(g(0)=0\)，且
\[
g'(x)=\frac1{1+x}+\frac1{1-x}-2=\frac{2x^2}{1-x^2}>0
\]
所以 \(g(x)>0\)，即 \(h(x)>2\). 另一方面，\(\displaystyle\lim_{x\to0^+}h(x)=2\). 因此当 \(a\leq2\) 时，\(\ln\dfrac{1+x}{1-x}>2x\geq ax\)，原不等式对所有 \(x\in(0,1)\) 成立；当 \(a>2\) 时，由 \(h(x)\to2\)，可取充分小的正数 \(x\) 使 \(h(x)<a\)，此时原不等式不成立. 故 \(a\) 的取值范围为 \(a\leq2\).
\end{enumerate}
\end{solution}

\begin{problem}
设数列 \(a_n\) 的前 \(n\) 项的和 \(S_n=\frac43a_n-\frac13\times2^{n+1}+\frac23\)，\(n=1\)，\(2\)，\(3\)，\(\ldots\).

\begin{enumerate}
\item 求首项 \(a_1\) 与通项 \(a_n\).
\item 设 \(T_n=\frac{2^n}{S_n}\)，\(n=1\)，\(2\)，\(3\)，\(\ldots\)，证明：\(\sum_{i=1}^nT_i<\frac32\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 取 \(n=1\)，由 \(S_1=a_1\) 得
\[
a_1=\frac43a_1-\frac13\cdot2^2+\frac23=\frac43a_1-\frac23
\]
所以 \(a_1=2\). 当 \(n\geq2\) 时，\(a_n=S_n-S_{n-1}\)，从而
\[
a_n=\frac43(a_n-a_{n-1})-\frac13\left(2^{n+1}-2^n\right)=\frac43(a_n-a_{n-1})-\frac{2^n}{3}
\]
整理得 \(a_n=4a_{n-1}+2^n\). 对应齐次递推式的通项为 \(C4^{n-1}\). 设一个特解为 \(\lambda2^n\)，代入递推式得 \(\lambda2^n=4\lambda2^{n-1}+2^n=(2\lambda+1)2^n\)，所以 \(\lambda=-1\). 因此 \(a_n=C4^{n-1}-2^n\)，再由 \(a_1=2\) 得 \(C=4\)，所以
\[
a_n=4^n-2^n
\]
\item 将 \(a_n=4^n-2^n\) 代入已知关系，得
\[
S_n=\frac{4^{n+1}-3\cdot2^{n+1}+2}{3}=\frac23(2^{n+1}-1)(2^n-1)>0
\]
因此
\[
T_n=\frac{2^n}{S_n}=\frac32\left(\frac1{2^n-1}-\frac1{2^{n+1}-1}\right)
\]
于是
\[
\sum_{i=1}^nT_i=\frac32\left(\frac1{2^1-1}-\frac1{2^{n+1}-1}\right)=\frac32\left(1-\frac1{2^{n+1}-1}\right)<\frac32
\]
\end{enumerate}
\end{solution}
