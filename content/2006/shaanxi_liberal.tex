\chapter{2006年陕西卷（文）}
\section{单选题}
\begin{problem}
已知集合 \(P = \{x \in \N \mid 1 \leqslant x \leqslant 10\}\)，集合 \(Q = \{x \in \R \mid x^2 + x - 6 = 0\}\)，则 \(P \cap Q\) 等于（\quad）

\choices
    {\(\{2\}\)}
    {\(\{3\}\)}
    {\(\{-2, 3\}\)}
    {\(\{-3, 2\}\)}
\end{problem}

\answer{A}

\begin{solution}
由 \(x^{2}+x-6=(x-2)(x+3)\)，得 \(Q=\{-3,2\}\). 又 \(P\) 中的元素为 \(1,2,\ldots,10\)，所以 \(P\cap Q=\{2\}\)，故选 A.
\end{solution}

\begin{problem}
函数 \( f(x) = \frac{1}{1 + x^2} \)（\(x \in \R\)）的值域是（\quad）

\choices
    {\((0,1)\)}
    {\((0,1]\)}
    {\([0,1)\)}
    {\([0,1]\)}
\end{problem}

\answer{B}

\begin{solution}
因为 \(x\in\R\)，所以 \(x^{2}\geqslant0\)，从而 \(1+x^{2}\geqslant1\)，因此 \(0<f(x)\leqslant1\). 当 \(x=0\) 时 \(f(x)=1\)，而 \(f(x)\) 不可能等于 \(0\)，但当 \(\lvert x\rvert\) 无限增大时 \(f(x)\) 无限接近 \(0\). 所以函数的值域为 \((0,1]\)，故选 B.
\end{solution}

\begin{problem}
已知等差数列 \(\{a_{n}\}\) 中，\(a_{1} + a_{8} = 8\)，则该数列前 \(9\) 项和 \(S_{9}\) 等于（\quad）

\choices
    {\(18\)}
    {\(27\)}
    {\(36\)}
    {\(45\)}
\end{problem}

\answer{无唯一值}

\begin{solution}
设等差数列的首项为 \(a_{1}\)，公差为 \(d\)，则 \(a_{8}=a_{1}+7d\). 由题设 \(a_{1}+a_{8}=8\)，得
\[
2a_{1}+7d=8.
\]

另一方面，\(a_{9}=a_{1}+8d\)，所以
\[
S_{9}=\dfrac{9}{2}(a_{1}+a_{9})=\dfrac{9}{2}(2a_{1}+8d)=\dfrac{9}{2}(8+d)=36+\dfrac{9}{2}d.
\]

条件 \(2a_{1}+7d=8\) 只确定了 \(a_{1}\) 与 \(d\) 的一个关系，公差 \(d\) 仍可任意取值，因此 \(S_{9}\) 不能唯一确定. 例如，取 \(d=-4\)，则 \(a_{1}=18\)，此时 \(S_{9}=18\)；取 \(d=-2\)，则 \(a_{1}=11\)，此时 \(S_{9}=27\)；取 \(d=0\)，则 \(a_{1}=4\)，此时 \(S_{9}=36\)；取 \(d=2\)，则 \(a_{1}=-3\)，此时 \(S_{9}=45\). 四个选项中的数值均可能出现，所以按原题题面本题没有唯一答案.
\end{solution}

\begin{problem}
设函数 \( f(x) = \log_a(x + b) \)（\(a > 0\)，\(a \neq 1\)）的图象过点 \((0,0)\)，其反函数的图象过点 \((1,2)\)，则 \( a + b \) 等于（\quad）

\choices
    {\(6\)}
    {\(5\)}
    {\(4\)}
    {\(3\)}
\end{problem}

\answer{C}

\begin{solution}
由函数图象过点 \((0,0)\)，得 \(\log_{a}b=0\)，所以 \(b=1\). 反函数的图象过点 \((1,2)\)，等价于原函数图象过点 \((2,1)\)，故 \(\log_{a}(2+b)=1\)，从而 \(a=2+b=3\). 因此 \(a+b=4\)，故选 C.
\end{solution}

\begin{problem}
设直线过点 \((0, a)\)，其斜率为 \(1\)，且与圆 \(x^{2} + y^{2} = 2\) 相切，则 \(a\) 的值为（\quad）

\choices
    {\(\pm \sqrt{2}\)}
    {\(\pm 2\)}
    {\(\pm 2\sqrt{2}\)}
    {\(\pm 4\)}
\end{problem}

\answer{B}

\begin{solution}
直线的方程为 \(y=x+a\)，即 \(x-y+a=0\). 圆心为原点，半径为 \(\sqrt{2}\)，所以原点到切线的距离满足 \(\dfrac{\lvert a\rvert}{\sqrt{2}}=\sqrt{2}\). 因此 \(\lvert a\rvert=2\)，即 \(a=\pm2\)，故选 B.
\end{solution}

\begin{problem}
“\(\alpha\)，\(\beta\)，\(\gamma\) 成等差数列”是“等式 \(\sin (\alpha + \gamma) = \sin 2\beta\) 成立”的（\quad）

\choices
    {充分而不必要条件}
    {必要而不充分条件}
    {充分必要条件}
    {既不充分也不必要条件}
\end{problem}

\answer{A}

\begin{solution}
若 \(\alpha\)，\(\beta\)，\(\gamma\) 成等差数列，则 \(\alpha+\gamma=2\beta\)，从而 \(\sin(\alpha+\gamma)=\sin2\beta\). 因此“\(\alpha\)，\(\beta\)，\(\gamma\) 成等差数列”是该等式成立的充分条件.

反之不成立. 例如取 \(\alpha=0\)，\(\beta=0\)，\(\gamma=\pi\)，则 \(\sin(\alpha+\gamma)=\sin\pi=0=\sin0=\sin2\beta\)，但 \(\alpha+\gamma\neq2\beta\)，三数不成等差数列. 因此该条件是充分而不必要条件，故选 A.
\end{solution}

\begin{problem}
设 \(x\)，\(y\) 为正数，则 \((x + y)\left(\frac{1}{x} + \frac{4}{y}\right)\) 的最小值为（\quad）

\choices
    {\(6\)}
    {\(9\)}
    {\(12\)}
    {\(15\)}
\end{problem}

\answer{B}

\begin{solution}
展开得 \((x+y)\left(\dfrac{1}{x}+\dfrac{4}{y}\right)=5+\dfrac{4x}{y}+\dfrac{y}{x}\). 因为 \(x>0\)，\(y>0\)，由基本不等式，\(\dfrac{4x}{y}+\dfrac{y}{x}\geqslant2\sqrt{\dfrac{4x}{y}\cdot\dfrac{y}{x}}=4\). 所以原式的最小值为 \(5+4=9\)，当且仅当 \(\dfrac{4x}{y}=\dfrac{y}{x}\)，即 \(y=2x\) 时取得，故选 B.
\end{solution}

\begin{problem}
已知非零向量 \(\overrightarrow{AB}\) 与 \(\overrightarrow{AC}\) 满足 \(\left(\frac{\overrightarrow{AB}}{|\overrightarrow{AB}|} + \frac{\overrightarrow{AC}}{|\overrightarrow{AC}|}\right)\cdot \overrightarrow{BC} = 0\) 且 \(\frac{\overrightarrow{AB}}{|\overrightarrow{AB}|}\cdot \frac{\overrightarrow{AC}}{|\overrightarrow{AC}|} = \frac{1}{2}\)，则 \(\triangle ABC\) 为（\quad）

\choices
    {三边均不相等的三角形}
    {直角三角形}
    {等腰非等边三角形}
    {等边三角形}
\end{problem}

\answer{D}

\begin{solution}
令 \(\bs u=\dfrac{\overrightarrow{AB}}{\lvert\overrightarrow{AB}\rvert}\)，\(\bs v=\dfrac{\overrightarrow{AC}}{\lvert\overrightarrow{AC}\rvert}\)，\(b=\lvert\overrightarrow{AB}\rvert\)，\(c=\lvert\overrightarrow{AC}\rvert\). 由题意，\(\bs u\cdot\bs v=\dfrac{1}{2}\)，且 \(\overrightarrow{BC}=c\bs v-b\bs u\). 因此 \(0=(\bs u+\bs v)\cdot(c\bs v-b\bs u)=c\left(\bs u\cdot\bs v+\bs v\cdot\bs v\right)-b\left(\bs u\cdot\bs u+\bs v\cdot\bs u\right)=\dfrac{3}{2}(c-b)\). 所以 \(b=c\)，即 \(AB=AC\). 又 \(\bs u\cdot\bs v=\cos\angle BAC=\dfrac12\)，所以 \(\angle BAC=\dfrac\pi3\). 由余弦定理，\(BC^{2}=b^{2}+c^{2}-2bc\cos\dfrac\pi3=b^{2}\)，从而 \(BC=b=c\). 故三角形 \(ABC\) 为等边三角形，选 D.
\end{solution}

\begin{problem}
已知函数 \( f(x) = ax^2 + 2ax + 4 \)（\(a > 0\)）. 若 \(x_1 < x_2\)，\(x_1 + x_2 = 0\)，则（\quad）

\choices
    {\(f(x_{1}) < f(x_{2})\)}
    {\(f(x_{1}) = f(x_{2})\)}
    {\(f(x_{1}) > f(x_{2})\)}
    {\(f(x_{1})\) 与 \(f(x_{2})\) 的大小不能确定}
\end{problem}

\answer{A}

\begin{solution}
因为 \(x_{1}+x_{2}=0\)，所以 \(f(x_{1})-f(x_{2})=a(x_{1}^{2}-x_{2}^{2})+2a(x_{1}-x_{2})=a(x_{1}-x_{2})(x_{1}+x_{2}+2)=2a(x_{1}-x_{2})\). 又 \(a>0\)，\(x_{1}<x_{2}\)，故 \(f(x_{1})-f(x_{2})<0\)，即 \(f(x_{1})<f(x_{2})\)，选 A.
\end{solution}

\begin{problem}
已知双曲线 \(\frac{x^2}{a^2} - \frac{y^2}{2} = 1\)（\(a > \sqrt{2}\)）的两条渐近线的夹角为 \(\frac{\pi}{3}\)，则双曲线的离心率为（\quad）

\choices
    {\(2\)}
    {\(\sqrt{3}\)}
    {\(\frac{2\sqrt{6}}{3}\)}
    {\(\frac{2\sqrt{3}}{3}\)}
\end{problem}

\answer{D}

\begin{solution}
双曲线的渐近线为 \(y=\pm\dfrac{\sqrt{2}}{a}x\). 由 \(a>\sqrt{2}\)，令 \(m=\dfrac{\sqrt{2}}{a}\)，则 \(0<m<1\). 两条渐近线的锐角夹角为 \(\dfrac\pi3\)，故 \(\tan\dfrac\pi3=\dfrac{2m}{1-m^{2}}\)，即 \(\sqrt{3}m^{2}+2m-\sqrt{3}=0\). 解得 \(m=\dfrac{1}{\sqrt{3}}\)，从而 \(a=\sqrt{6}\). 双曲线的半焦距满足 \(c^{2}=a^{2}+2=8\)，所以 \(c=2\sqrt{2}\)，离心率 \(e=\dfrac{c}{a}=\dfrac{2\sqrt{2}}{\sqrt{6}}=\dfrac{2\sqrt{3}}{3}\). 故选 D.
\end{solution}

\begin{problem}
已知平面 \(\alpha\) 外不共线的三点 \(A\)，\(B\)，\(C\) 到 \(\alpha\) 的距离都相等. 则正确的结论是（\quad）

\choices
    {平面 \(ABC\) 必平行于 \(\alpha\)}
    {平面 \(ABC\) 必与 \(\alpha\) 相交}
    {平面 \(ABC\) 必不垂直于 \(\alpha\)}
    {存在 \(\triangle ABC\) 的一条中位线平行于 \(\alpha\) 或在 \(\alpha\) 内}
\end{problem}

\answer{D}

\begin{solution}
设三点到平面 \(\alpha\) 的距离为 \(d\). 若 \(d=0\)，则 \(A\)，\(B\)，\(C\) 均在 \(\alpha\) 内，三角形的中位线也在 \(\alpha\) 内.

若 \(d>0\)，以 \(\alpha\) 的一个法向量确定有向距离，则 \(A\)，\(B\)，\(C\) 的有向距离只能分别为 \(d\) 或 \(-d\). 由抽屉原理，三点中至少有两点在 \(\alpha\) 的同一侧. 不妨设 \(A\)，\(B\) 在同一侧，则线段 \(AB\) 平行于 \(\alpha\). 三角形 \(ABC\) 中，连接 \(AC\)，\(BC\) 的中点所得中位线平行于 \(AB\)，因而也平行于 \(\alpha\). 所以正确结论为 D.
\end{solution}

\begin{problem}
为确保信息安全，信息需加密传播，发送方由明文 \(\rightarrow\) 密文（加密），接收方由密文 \(\rightarrow\) 明文（解密）. 已知加密规则为：明文 \(a\)，\(b\)，\(c\)，\(d\) 对应密文 \(a + 2b\)，\(2b + c\)，\(2c + 3d\)，\(4d\). 例如，明文 \(1\)，\(2\)，\(3\)，\(4\) 对应密文 \(5\)，\(7\)，\(18\)，\(16\). 当接收方收到密文 \(14\)，\(9\)，\(23\)，\(28\) 时，则解密得到的明文为（\quad）

\choices
    {\(1\)，\(6\)，\(4\)，\(7\)}
    {\(4\)，\(6\)，\(1\)，\(7\)}
    {\(7\)，\(6\)，\(1\)，\(4\)}
    {\(6\)，\(4\)，\(1\)，\(7\)}
\end{problem}

\answer{D}

\begin{solution}
由最后一条加密关系 \(4d=28\)，得 \(d=7\). 代入第三条关系 \(2c+3d=23\)，得 \(2c+21=23\)，所以 \(c=1\). 再由第二条关系 \(2b+c=9\)，得 \(2b+1=9\)，所以 \(b=4\). 最后由第一条关系 \(a+2b=14\)，得 \(a+8=14\)，所以 \(a=6\). 因此解密得到的明文依次为 \(6\)，\(4\)，\(1\)，\(7\)，对应选项 D.
\end{solution}

\section{填空题}
\begin{problem}
\(\cos 43^{\circ}\cos 77^{\circ} + \sin 43^{\circ}\cos 167^{\circ}\) 的值为\fillinblank{}.
\end{problem}

\answer{\(-\frac{1}{2}\)}

\begin{solution}
因为 \(167^{\circ}=90^{\circ}+77^{\circ}\)，所以 \(\cos167^{\circ}=-\sin77^{\circ}\). 原式为 \(\cos43^{\circ}\cos77^{\circ}-\sin43^{\circ}\sin77^{\circ}=\cos(43^{\circ}+77^{\circ})=\cos120^{\circ}=-\dfrac12\).
\end{solution}

\begin{problem}
\(\left(2x - \frac{1}{\sqrt{x}}\right)^6\) 展开式中的常数项为\fillinblank{}. （用数字作答）
\end{problem}

\answer{60}

\begin{solution}
展开式的通项可写为 \(\mathrm C_{6}^{r}(2x)^{6-r}\left(-\dfrac{1}{\sqrt{x}}\right)^{r}=\mathrm C_{6}^{r}2^{6-r}(-1)^{r}x^{6-\frac32r}\). 要得到常数项，必须满足 \(6-\dfrac32r=0\)，所以 \(r=4\). 因此常数项为 \(\mathrm C_{6}^{4}2^{2}(-1)^{4}=15\times4=60\).
\end{solution}

\begin{problem}
某校从 \(8\) 名教师中选派 \(4\) 名教师同时去 \(4\) 个边远地区支教（每地 \(1\) 人），其中甲和乙不同去，则不同的选派方案共有\fillinblank{} 种.
\end{problem}

\answer{1320}

\begin{solution}
先不考虑甲、乙不能同时去的限制，从 \(8\) 名教师中选出 \(4\) 人并分配到 \(4\) 个不同地区，共有 \(\mathrm A_{8}^{4}=8\times7\times6\times5=1680\) 种方案.

若甲、乙同时去，则另外 \(2\) 名教师从其余 \(6\) 人中选取，再把 \(4\) 人分配到 \(4\) 个地区，共有 \(\mathrm C_{6}^{2}\mathrm A_{4}^{4}=15\times24=360\) 种方案. 因此符合条件的方案数为 \(1680-360=1320\).
\end{solution}

\begin{problem}
水平桌面 \(\alpha\) 上放有 \(4\) 个半径为 \(2R\) 的球. 且相邻的球都相切（球心的连线构成正方形）. 在这 \(4\) 个球的上面放 \(1\) 个半径为 \(R\) 的小球，它和下面的 \(4\) 个球恰好都相切，则小球的球心到水平桌面 \(\alpha\) 的距离是\fillinblank{}.
\end{problem}

\answer{\(3R\)}

\begin{solution}
四个大球的球心在同一水平面内，且相邻球相切，所以球心构成边长 \(4R\) 的正方形. 设正方形中心为 \(O\)，小球球心为 \(P\)，则小球球心在 \(O\) 的竖直线上. 大球球心到 \(O\) 的水平距离为正方形对角线的一半，即 \(2\sqrt{2}R\).

四个大球球心到桌面的距离均为 \(2R\)，设小球球心到桌面的距离为 \(h\). 由小球与大球相切，球心距为 \(3R\)，于是 \((h-2R)^{2}+(2\sqrt{2}R)^{2}=(3R)^{2}\). 解得 \((h-2R)^{2}=R^{2}\)，即 \(h=R\) 或 \(h=3R\). 由于小球放在四个大球的上面，取 \(h=3R\).
\end{solution}

\section{解答题}
\begin{problem}
甲，乙，丙 \(3\) 人投篮，投进的概率分别是 \(\frac{2}{5}\)，\(\frac{1}{2}\)，\(\frac{3}{5}\)，现 \(3\) 人各投篮 \(1\) 次，求：

\begin{enumerate}
\item \(3\) 人都投进的概率；
\item \(3\) 人中恰有 \(2\) 人投进的概率.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设甲、乙、丙投进分别为事件 \(A\)，\(B\)，\(C\). 按各次投篮相互独立处理，三人都投进的概率为 \(P(ABC)=\dfrac{2}{5}\times\dfrac{1}{2}\times\dfrac{3}{5}=\dfrac{3}{25}\).
\item 恰有两人投进包括甲、乙投进而丙未投进，甲、丙投进而乙未投进，以及乙、丙投进而甲未投进三种互斥情形. 因此所求概率为
\(P(AB\overline{C})+P(A\overline{B}C)+P(\overline{A}BC)=\dfrac{2}{5}\times\dfrac{1}{2}\times\dfrac{2}{5}+\dfrac{2}{5}\times\dfrac{1}{2}\times\dfrac{3}{5}+\dfrac{3}{5}\times\dfrac{1}{2}\times\dfrac{3}{5}=\dfrac{2}{25}+\dfrac{3}{25}+\dfrac{9}{50}=\dfrac{19}{50}\).
\end{enumerate}
\end{solution}

\begin{problem}
已知函数 \( f(x) = \sqrt{3}\sin \left(2x - \frac{\pi}{6}\right) + 2\sin^2\left(x - \frac{\pi}{12}\right) \)（\(x \in \R\)）.

\begin{enumerate}
\item 求函数 \( f(x) \) 的最小正周期；
\item 求使函数 \( f(x) \) 取得最大值的 \(x\) 的集合.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 令 \(u=2x-\dfrac{\pi}{6}\)，则 \(2\sin^{2}\left(x-\dfrac{\pi}{12}\right)=1-\cos\left(2x-\dfrac{\pi}{6}\right)=1-\cos u\)，所以 \(f(x)=1+\sqrt{3}\sin u-\cos u=1+2\sin\left(u-\dfrac{\pi}{6}\right)=1+2\sin\left(2x-\dfrac{\pi}{3}\right)\). 因此函数 \(f(x)\) 的最小正周期为 \(\pi\).
\item 当且仅当 \(\sin\left(2x-\dfrac{\pi}{3}\right)=1\) 时，\(f(x)\) 取得最大值，即 \(2x-\dfrac{\pi}{3}=\dfrac{\pi}{2}+2k\pi\)，其中 \(k\in\Z\). 解得 \(x=\dfrac{5\pi}{12}+k\pi\)，所以使函数取得最大值的 \(x\) 的集合为 \(\left\{x\mid x=\dfrac{5\pi}{12}+k\pi,\ k\in\Z\right\}\).
\end{enumerate}
\end{solution}

\begin{problem}
如图，\(\alpha \perp \beta\)，\(\alpha \cap \beta = l\)，\(A \in \alpha\)，\(B \in \beta\)，点 \(A\) 在直线 \(l\) 上的射影为 \(A_1\)，点 \(B\) 在 \(l\) 上的射影为 \(B_1\). 已知 \(AB = 2\)，\(AA_1 = 1\)，\(BB_1 = \sqrt{2}\). 求：

\begin{enumerate}
\item 直线 \(AB\) 分别与平面 \(\alpha\)，\(\beta\) 所成角的大小；
\item 二面角 \(A_{1} - AB - B_{1}\) 的大小.
\end{enumerate}

\bitmapfigure[max width=0.38\linewidth]{img/2006/shaanxi_liberal/q19_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 因为 \(\alpha\perp\beta\)，所以 \(BB_{1}\perp\alpha\)，\(AA_{1}\perp\beta\). 在直角三角形 \(ABB_{1}\) 中，\(AB_{1}=\sqrt{AB^{2}-BB_{1}^{2}}=\sqrt{4-2}=\sqrt{2}\)，故直线 \(AB\) 与平面 \(\alpha\) 所成的角为 \(\arcsin\dfrac{BB_{1}}{AB}=\arcsin\dfrac{\sqrt{2}}{2}=\dfrac{\pi}{4}\). 在直角三角形 \(AA_{1}B\) 中，\(A_{1}B=\sqrt{AB^{2}-AA_{1}^{2}}=\sqrt{4-1}=\sqrt{3}\)，故直线 \(AB\) 与平面 \(\beta\) 所成的角为 \(\arcsin\dfrac{AA_{1}}{AB}=\arcsin\dfrac{1}{2}=\dfrac{\pi}{6}\).
\item 由三条互相垂直的分量可得 \(A_{1}B_{1}=\sqrt{AB^{2}-AA_{1}^{2}-BB_{1}^{2}}=1\). 建立直角坐标系，使 \(l\) 为 \(x\) 轴，\(\alpha\) 为 \(xy\) 平面，\(\beta\) 为 \(xz\) 平面，并取 \(A_{1}=(0,0,0)\)，\(B_{1}=(1,0,0)\)，\(A=(0,1,0)\)，\(B=(1,0,\sqrt{2})\). 则平面 \(A_{1}AB\) 的一个法向量为 \(\bs n_{1}=(\sqrt{2},0,-1)\)，平面 \(B_{1}AB\) 的一个法向量为 \(\bs n_{2}=(\sqrt{2},\sqrt{2},0)\). 设所求二面角为 \(\theta\)，则 \(\cos\theta=\dfrac{\bs n_{1}\cdot\bs n_{2}}{\lvert\bs n_{1}\rvert\lvert\bs n_{2}\rvert}=\dfrac{2}{\sqrt{3}\times2}=\dfrac{\sqrt{3}}{3}\)，所以 \(\theta=\arccos\dfrac{\sqrt{3}}{3}\).
\end{enumerate}
\end{solution}

\begin{problem}
已知正项数列 \(\{a_{n}\}\)，其前 \(n\) 项和 \(S_{n}\) 满足 \(10S_{n} = a_{n}^{2} + 5a_{n} + 6\)，且 \(a_{1}\)，\(a_{3}\)，\(a_{15}\) 成等比数列，求数列 \(\{a_{n}\}\) 的通项 \(a_{n}\).
\end{problem}

\begin{solution}
取 \(n=1\)，由 \(S_{1}=a_{1}\) 得 \(10a_{1}=a_{1}^{2}+5a_{1}+6\)，即 \(a_{1}^{2}-5a_{1}+6=0\)，所以 \(a_{1}=2\) 或 \(a_{1}=3\).

当 \(n\geqslant2\) 时，将条件分别对 \(n\) 和 \(n-1\) 使用并相减，得 \(10a_{n}=a_{n}^{2}+5a_{n}+6-a_{n-1}^{2}-5a_{n-1}-6\)，即 \(10a_{n}=(a_{n}-a_{n-1})(a_{n}+a_{n-1}+5)\). 整理得 \((a_{n}+a_{n-1})(a_{n}-a_{n-1}-5)=0\). 由于数列各项为正，\(a_{n}+a_{n-1}>0\)，所以 \(a_{n}-a_{n-1}=5\). 从而 \(a_{n}=a_{1}+5(n-1)\).

若 \(a_{1}=2\)，则 \(a_{3}=12\)，\(a_{15}=72\)，且 \(a_{3}^{2}=144=a_{1}a_{15}\)，满足等比条件. 若 \(a_{1}=3\)，则 \(a_{3}=13\)，\(a_{15}=73\)，但 \(a_{3}^{2}=169\neq219=a_{1}a_{15}\)，不满足等比条件. 因此 \(a_{1}=2\)，数列的通项为 \(a_{n}=2+5(n-1)=5n-3\).
\end{solution}

\begin{problem}
如图，三定点 \(A(2,1)\)，\(B(0,-1)\)，\(C(-2,1)\)，三动点 \(D\)，\(E\)，\(M\) 满足 \(\overrightarrow{AD} = t\overrightarrow{AB}\)，\(\overrightarrow{BE} = t\overrightarrow{BC}\)，\(\overrightarrow{DM} = t\overrightarrow{DE}\)，\(t \in [0,1]\). 求：

\begin{enumerate}
\item 求动直线 \(DE\) 斜率的变化范围；
\item 求动点 \(M\) 的轨迹方程.
\end{enumerate}

\bitmapfigure[max width=0.40\linewidth]{img/2006/shaanxi_liberal/q21_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由 \(\overrightarrow{AB}=(-2,-2)\)，\(\overrightarrow{BC}=(-2,2)\)，得 \(D=(2-2t,1-2t)\)，\(E=(-2t,2t-1)\). 所以 \(\overrightarrow{DE}=(-2,4t-2)\)，动直线 \(DE\) 的斜率为 \(k_{DE}=\dfrac{4t-2}{-2}=1-2t\). 当 \(t\in[0,1]\) 时，\(k_{DE}\in[-1,1]\).
\item 由 \(\overrightarrow{DM}=t\overrightarrow{DE}\)，得 \(M=D+t(E-D)\)，即 \(M=(2-4t,1-4t+4t^{2})\). 记 \(M=(x,y)\)，则 \(x=2-4t\)，从而 \(t=\dfrac{2-x}{4}\)，且 \(t\in[0,1]\) 等价于 \(-2\leqslant x\leqslant2\). 代入 \(y\) 得 \(y=\dfrac{x^{2}}{4}\)，所以轨迹方程为 \(x^{2}=4y\)，其中 \(-2\leqslant x\leqslant2\).
\end{enumerate}
\end{solution}

\begin{problem}
设函数 \( f(x) = kx^{3} - 3x^{2} + 1 \)（\(k \geqslant 0\)）.

\begin{enumerate}
\item 求函数 \( f(x) \) 的单调区间；
\item 若函数 \( f(x) \) 的极小值大于 0，求 \(k\) 的取值范围.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item \(f'(x)=3kx^{2}-6x=3x(kx-2)\). 当 \(k=0\) 时，\(f'(x)=-6x\)，所以 \(f(x)\) 在 \((-\infty,0)\) 上单调递增，在 \((0,+\infty)\) 上单调递减. 当 \(k>0\) 时，\(f'(x)>0\) 的区间为 \((-\infty,0)\) 和 \((\frac{2}{k},+\infty)\)，\(f'(x)<0\) 的区间为 \((0,\frac{2}{k})\)，所以 \(f(x)\) 在 \((-\infty,0)\) 和 \((\frac{2}{k},+\infty)\) 上单调递增，在 \((0,\frac{2}{k})\) 上单调递减.
\item 当 \(k=0\) 时，\(f(x)=-3x^{2}+1\) 只有极大值而无极小值，不满足条件. 当 \(k>0\) 时，\(x=\frac{2}{k}\) 是极小值点，极小值为 \(f\left(\frac{2}{k}\right)=k\left(\frac{2}{k}\right)^{3}-3\left(\frac{2}{k}\right)^{2}+1=1-\frac{4}{k^{2}}\). 要求极小值大于 \(0\)，得 \(1-\frac{4}{k^{2}}>0\)，结合 \(k>0\)，可得 \(k>2\).
\end{enumerate}
\end{solution}
