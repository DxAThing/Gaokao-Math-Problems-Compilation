\section{单选题}

\begin{problem}
已知集合 \(A = \{x \mid 2 < x < 4\}\)，\(B = \{x \mid (x - 1)(x - 3) < 0\}\)，则 \(A \cap B =\)（\quad）

\choices
    {\((1,3)\)}
    {\((1, 4)\)}
    {\((2,3)\)}
    {\((2,4)\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
由 \((x-1)(x-3)<0\)，得 \(1<x<3\)，所以 \(B=(1,3)\). 因此
\(A\cap B=(2,4)\cap(1,3)=(2,3)\)，故选 C.
\end{solution}

\begin{problem}
若复数 \(z\) 满足 \(\frac{\overline{z}}{1 - \i} = \i\)，其中 \(\i\) 为虚数单位，则 \(z =\)（\quad）

\choices
    {\(1 - \i\)}
    {\(1 + \i\)}
    {\(-1 - \i\)}
    {\(-1 + \i\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
由题意，\(\overline{z}=\i(1-\i)=\i-\i^2=1+\i\). 两边取共轭，得 \(z=1-\i\)，故选 A.
\end{solution}

\begin{problem}
设 \(a = 0.6^{0.6}\)，\(b = 0.6^{1.5}\)，\(c = 1.5^{0.6}\)，则 \(a\)，\(b\)，\(c\) 的大小关系是（\quad）

\choices
    {\(a < b < c\)}
    {\(a < c < b\)}
    {\(b < a < c\)}
    {\(b < c < a\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
因为 \(0<0.6<1\)，指数越大，函数值越小，所以
\(b=0.6^{1.5}<a=0.6^{0.6}<1\). 又因为 \(1.5>1\)，所以 \(c=1.5^{0.6}>1\). 因而 \(b<a<c\)，故选 C.
\end{solution}

\begin{problem}
要得到函数 \( y = \sin \left(4x - \frac{\pi}{3}\right) \) 的图象，只需要将函数 \( y = \sin 4x \) 的图象（\quad）

\choices
    {向左平移 \(\frac{\pi}{12}\) 个单位}
    {向右平移 \(\frac{\pi}{12}\) 个单位}
    {向左平移 \(\frac{\pi}{3}\) 个单位}
    {向右平移 \(\frac{\pi}{3}\) 个单位}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
将 \(4x-\frac{\pi}{3}\) 写成 \(4\left(x-\frac{\pi}{12}\right)\)，可得
\(\sin\left(4x-\frac{\pi}{3}\right)=\sin 4\left(x-\frac{\pi}{12}\right)\). 因此将 \(y=\sin4x\) 的图象向右平移 \(\frac{\pi}{12}\) 个单位，故选 B.
\end{solution}

\begin{problem}
若 \(m \in \R\)，命题“若 \(m > 0\)，则方程 \(x^{2} + x - m = 0\) 有实数根”的逆否命题是（\quad）

\choices
    {若方程 \(x^{2} + x - m = 0\) 有实根，则 \(m > 0\)}
    {若方程 \(x^{2} + x - m = 0\) 有实根，则 \(m \leqslant 0\)}
    {若方程 \(x^{2} + x - m = 0\) 没有实根，则 \(m > 0\)}
    {若方程 \(x^{2} + x - m = 0\) 没有实根，则 \(m \leqslant 0\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
设原命题为“若 \(p\)，则 \(q\)”，其中 \(p\) 为 \(m>0\)，\(q\) 为方程有实数根. 逆否命题是“若非 \(q\)，则非 \(p\)”，即若方程没有实根，则 \(m\leqslant0\)，故选 D.
\end{solution}

\begin{problem}
为比较甲、乙两地某月14时的气温情况，随机选取该月中的5天，将这5天中14时的气温数据（单位：\(^{\circ}C\)）制成如图所示的茎叶图.

\begin{center}
\begin{tabular}{rrr|c|lll}
\multicolumn{3}{c|}{甲} & & \multicolumn{3}{c}{乙} \\
\hline
\(9\) & \(8\) & \(6\) & \(2\) & \(8\) & \(9\) & \\
\(1\) & \(1\) &        & \(3\) & \(0\) & \(1\) & \(2\)
\end{tabular}
\end{center}

考虑以下结论：

\begin{circlelist}
\item 甲地该月14时的平均气温低于乙地该月14时的平均气温；
\item 甲地该月14时的平均气温高于乙地该月14时的平均气温；
\item 甲地该月14时的气温的标准差小于乙地该月14时的气温的标准差；
\item 甲地该月14时的气温的标准差大于乙地该月14时的气温的标准差.
\end{circlelist}

其中根据茎叶图能得到的统计结论的标号为（\quad）

\choices
    {\circled{1}\circled{3}}
    {\circled{1}\circled{4}}
    {\circled{2}\circled{3}}
    {\circled{2}\circled{4}}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
由茎叶图，甲地的 5 个数据为 \(26\)，\(28\)，\(29\)，\(31\)，\(31\)，乙地的 5 个数据为 \(28\)，\(29\)，\(30\)，\(31\)，\(32\). 两地平均气温分别为
\(\overline{x}_{\text{甲}}=\frac{26+28+29+31+31}{5}=29\)，
\(\overline{x}_{\text{乙}}=\frac{28+29+30+31+32}{5}=30\)，所以结论1正确，结论2错误.

以各自平均数为基准，甲地数据的离差平方和为 \(3^2+1^2+0^2+2^2+2^2=18\)，乙地数据的离差平方和为 \(2^2+1^2+0^2+1^2+2^2=10\). 因而甲地的方差为 \(\frac{18}{5}\)，乙地的方差为 \(\frac{10}{5}\)，甲地标准差较大，结论4正确，结论3错误. 所以选 B.
\end{solution}

\begin{problem}
在区间 \([0,2]\) 上随机地取一个数 \(x\)，则事件“\(-1 \leqslant \log_{\frac{1}{2}}\left(x + \frac{1}{2}\right) \leqslant 1\)”发生的概率为（\quad）

\choices
    {\( \frac{3}{4} \)}
    {\( \frac{2}{3} \)}
    {\( \frac{1}{3} \)}
    {\( \frac{1}{4} \)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
令 \(t=x+\frac12\)，则 \(t>0\). 由于对数函数 \(\log_{1/2}t\) 的底数小于 \(1\)，它在正数范围内单调递减，因此
\[
-1\leqslant\log_{1/2}t\leqslant1
\quad\Longleftrightarrow\quad
\left(\frac12\right)^1\leqslant t\leqslant\left(\frac12\right)^{-1}
\]
所以 \(\frac12\leqslant x+\frac12\leqslant2\)，即 \(0\leqslant x\leqslant\frac32\). 事件长度为 \(\frac32\)，样本区间长度为 \(2\)，故概率为 \(\frac{3/2}{2}=\frac34\)，选 A.
\end{solution}

\begin{problem}
若函数 \( f(x) = \frac{2^x + 1}{2^x - a} \) 是奇函数，则使 \( f(x) > 3 \) 成立的 \( x \) 的取值范围为（\quad）

\choices
    {\((- \infty, -1)\)}
    {\((-1,0)\)}
    {\((0,1)\)}
    {\((1, + \infty)\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
令 \(t=2^x>0\). 奇函数要求定义域关于原点对称且 \(f(-x)=-f(x)\). 由
\(f(-x)=\frac{2^{-x}+1}{2^{-x}-a}=\frac{1+t}{1-at}\)，\(f(x)=\frac{1+t}{t-a}\)
并注意到 \(1+t>0\)，有 \(\frac{1}{1-at}=-\frac{1}{t-a}\)，从而
\((1-a)(t+1)=0\). 因此 \(a=1\). 此时 \(x\neq0\)，且
\[
f(x)=\frac{2^x+1}{2^x-1}>3
\quad\Longleftrightarrow\quad
\frac{2(2-2^x)}{2^x-1}>0
\]
令 \(t=2^x>0\)，且 \(t\ne1\). 当 \(0<t<1\) 时，分子为正、分母为负；当 \(1<t<2\) 时，分子和分母均为正；当 \(t>2\) 时，分子为负、分母为正. 因此不等式成立当且仅当 \(1<t<2\)，即 \(0<x<1\). 故选 C.
\end{solution}

\begin{problem}
已知等腰直角三角形的直角边的长为 2，将该三角形绕其斜边所在的直线旋转一周而形成的曲面所围成的几何体的体积为（\quad）

\choices
    {\(\frac{2\sqrt{2}}{3}\pi\)}
    {\(\frac{4\sqrt{2}\pi}{3}\)}
    {\(2\sqrt{2}\pi\)}
    {\(4\sqrt{2}\pi\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
斜边长为 \(2\sqrt2\)，直角顶点到斜边的高为
\(r=\frac{2\times2}{2\sqrt2}=\sqrt2\). 等腰直角三角形的高足为斜边中点，所以旋转后得到两个底面半径为 \(r\)、高之和为 \(2\sqrt2\) 的圆锥. 几何体体积为
\[
V=\frac13\pi r^2\cdot2\sqrt2
=\frac13\pi\cdot2\cdot2\sqrt2
=\frac{4\sqrt2\pi}{3}
\]
故选 B.
\end{solution}

\begin{problem}
设函数 \( f(x) = \begin{cases}
3x - b, & x < 1,\\
2^x, & x \geqslant 1,
\end{cases} \) 若 \( f\left(f\left(\frac{5}{6}\right)\right) = 4 \)，则 \( b = \)（\quad）

\choices
    {1}
    {\(\frac{7}{8}\)}
    {\(\frac{3}{4}\)}
    {\(\frac{1}{2}\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
令 \(u=f\left(\frac56\right)=\frac52-b\). 若 \(u\geqslant1\)，则
\(f(u)=2^u=4\)，从而 \(u=2\)，解得 \(b=\frac12\)，且 \(u=2\) 满足条件.

若 \(u<1\)，则 \(f(u)=3u-b=\frac{15}{2}-4b=4\)，解得 \(b=\frac78\). 但此时 \(u=\frac52-\frac78=\frac{13}{8}>1\)，与假设矛盾. 所以 \(b=\frac12\)，故选 D.
\end{solution}

执行如图所示的程序框图，若输入的 \(x\) 的值为 1，则输出的 \(y\) 的值是\fillinblank{}.
\texfigure[max width=0.42\linewidth]{img_repaint/2015/shandong_liberal/q11_fig1.tex}

\begin{answer}
13
\end{answer}

\begin{solution}
输入 \(x=1\) 后，判断 \(x<2\) 成立，执行 \(x=x+1\)，得到 \(x=2\). 再次判断时 \(x<2\) 不成立，执行 \(y=3x^2+1\)，所以
\(y=3\times2^2+1=13\).

\begin{problem}
若 \(x\)，\(y\) 满足约束条件 \(\begin{cases}
y - x \leqslant 1,\\
x + y \leqslant 3,\\
y \geqslant 1,
\end{cases}\) 则 \(z = x + 3y\) 的最大值为\fillinblank{}.
\end{problem}

\begin{answer}
7
\end{answer}

\begin{solution}
约束条件等价于 \(y\leqslant x+1\)、\(y\leqslant3-x\)、\(y\geqslant1\). 可行域的顶点为两直线 \(y=x+1\)、\(y=3-x\) 的交点 \((1,2)\)，以及与 \(y=1\) 的交点 \((0,1)\)、\((2,1)\). 目标函数在三个顶点处的值分别为
\(z(0,1)=3\)，\(z(2,1)=5\)，\(z(1,2)=7\).
因此 \(z\) 的最大值为 \(7\).
\end{solution}

\begin{problem}
过点 \(P(1, \sqrt{3})\) 作圆 \(x^2 + y^2 = 1\) 的两条切线，切点分别为 \(A\)，\(B\)，则 \(\overrightarrow{PA} \cdot \overrightarrow{PB} = \)\fillinblank{}.
\end{problem}

\begin{answer}
\(\frac{3}{2}\)
\end{answer}

\begin{solution}
设切点为 \(T(u,v)\). 圆 \(x^2+y^2=1\) 在 \(T\) 处的半径为 \(OT\)，而切线 \(PT\) 与半径垂直，所以 \(\overrightarrow{OP}\cdot\overrightarrow{OT}=1\). 对 \(P=(1,\sqrt3)\)，切点满足
\(u^2+v^2=1\)，\(u+\sqrt3v=1\).
解得两个切点为 \(A=(1,0)\)、\(B=\left(-\frac12,\frac{\sqrt3}{2}\right)\). 因而
\(\overrightarrow{PA}=(0,-\sqrt3)\)，\(\overrightarrow{PB}=\left(-\frac32,-\frac{\sqrt3}{2}\right)\)
所以 \(\overrightarrow{PA}\cdot\overrightarrow{PB}=\frac32\).
\end{solution}

\begin{problem}
定义运算“\(\otimes\)”： \( x \otimes y = \frac{x^{2} - y^{2}}{xy} \) （ \(x\)，\(y \in \R\)，\(xy \neq 0\) ）. 当 \(x > 0\)， \(y > 0\) 时， \( x \otimes y + (2y) \otimes x \) 的最小值为\fillinblank{}.
\end{problem}

\begin{answer}
\(\sqrt{2}\)
\end{answer}

\begin{solution}
由定义，
\(x\otimes y=\frac{x}{y}-\frac{y}{x}\)，\((2y)\otimes x=\frac{2y}{x}-\frac{x}{2y}\).
所以
\(x\otimes y+(2y)\otimes x=\frac{x}{2y}+\frac{y}{x}\). 令 \(t=\frac{x}{y}>0\)，原式为 \(\frac{t}{2}+\frac1t\). 由基本不等式，
\[
\frac{t}{2}+\frac1t\geqslant2\sqrt{\frac{t}{2}\cdot\frac1t}=\sqrt2
\]
当且仅当 \(\frac{t}{2}=\frac1t\)，即 \(t=\sqrt2\) 时取等号. 所以最小值为 \(\sqrt2\).
\end{solution}

\begin{problem}
过双曲线 \(C: \frac{x^2}{a^2} - \frac{y^2}{b^2} = 1\)（\(a > 0\)，\(b > 0\)）的右焦点作一条与其渐近线平行的直线，交 \(C\) 于点 \(P\). 若点 \(P\) 的横坐标为 \(2a\)，则 \(C\) 的离心率为\fillinblank{}.
\end{problem}

\begin{answer}
\(2 + \sqrt{3}\)
\end{answer}

\begin{solution}
设双曲线半焦距为 \(c\)，则 \(c^2=a^2+b^2\). 取与渐近线 \(y=\frac ba x\) 平行且过右焦点 \((c,0)\) 的直线，方程为
\(y=\frac ba(x-c)\). 由点 \(P\) 的横坐标为 \(2a\)，得 \(y=b\left(2-\frac ca\right)\). 代入双曲线方程，得到
\[
4-\frac{y^2}{b^2}=1,\qquad\text{即}\qquad
\left|2-\frac ca\right|=\sqrt3
\]
由于 \(c>a>0\)，不可能有 \(\frac ca=2-\sqrt3\)，故 \(\frac ca=2+\sqrt3\). 双曲线的离心率为 \(\frac ca=2+\sqrt3\).
\end{solution}

\begin{problem}
某中学调查了某班全部 45 名同学参加书法社团和演讲社团的情况，数据如下表：（单位：人）

\begin{center}
\begin{tabular}{|c|c|c|}
\hline
 & 参加书法社团 & 未参加书法社团 \\
\hline
参加演讲社团 & 8 & 5 \\
\hline
未参加演讲社团 & 2 & 30 \\
\hline
\end{tabular}
\end{center}

\begin{enumerate}
\item 从该班随机选 1 名同学，求该同学至少参加上述一个社团的概率；
\item 在既参加书法社团又参加演讲社团的 8 名同学中，有 5 名男同学 \(A_{1}\)，\(A_{2}\)，\(A_{3}\)，\(A_{4}\)，\(A_{5}\)，3 名女同学 \(B_{1}\)，\(B_{2}\)，\(B_{3}\). 现从这 5 名男同学和 3 名女同学中各随机选 1 人，求 \(A_{1}\) 被选中且 \(B_{1}\) 未被选中的概率.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由表中数据，未参加书法社团且未参加演讲社团的同学有 \(30\) 人，因此至少参加一个社团的同学有 \(45-30=15\) 人. 随机选取一名同学时，各人被选中的可能性相同，所以所求概率为
\[
P=\frac{15}{45}=\frac13
\]

\item 从 \(5\) 名男同学和 \(3\) 名女同学中各选取一人，基本事件总数为 \(5\times3=15\). 要求选中 \(A_1\) 且不选中 \(B_1\)，男生只能选 \(A_1\)，女生可选 \(B_2\) 或 \(B_3\)，所以符合条件的基本事件有 \(2\) 个. 因此所求概率为
\[
P=\frac{2}{15}
\]
\end{enumerate}
\end{solution}

\begin{problem}
\(\triangle ABC\) 中，角 \(A\)，\(B\)，\(C\) 所对的边分别为 \(a\)，\(b\)，\(c\)，已知 \(\cos B = \frac{\sqrt{3}}{3}\)，\(\sin (A + B) = \frac{\sqrt{6}}{9}\)，\(ac = 2\sqrt{3}\)，求 \(\sin A\) 和 \(c\) 的值.
\end{problem}

\begin{solution}
因为 \(A+B=\pi-C\)，所以
\[
\sin C=\sin(A+B)=\frac{\sqrt6}{9}
\]
又 \(B\) 是三角形内角且 \(\cos B=\frac{\sqrt3}{3}>0\)，故
\(\sin B=\sqrt{1-\cos^2B}=\frac{\sqrt6}{3}\). 由 \(\sin^2C+\cos^2C=1\)，得
\(\cos C=\pm\frac{5\sqrt3}{9}\). 因为 \(A=\pi-(B+C)\)，所以
\[
\sin A=\sin(B+C)=\sin B\cos C+\cos B\sin C
\]
若 \(\cos C=-\frac{5\sqrt3}{9}\)，则 \(\sin A=-\frac{4\sqrt2}{9}<0\)，不符合三角形内角的正弦为正，故 \(\cos C=\frac{5\sqrt3}{9}\). 从而
\[
\sin A=\frac{\sqrt6}{3}\cdot\frac{5\sqrt3}{9}+\frac{\sqrt3}{3}\cdot\frac{\sqrt6}{9}=\frac{2\sqrt2}{3}
\]
由正弦定理，\(\frac{a}{c}=\frac{\sin A}{\sin C}=2\sqrt3\)，即 \(a=2\sqrt3c\). 结合 \(ac=2\sqrt3\)，得 \(2\sqrt3c^2=2\sqrt3\). 由于 \(c>0\)，所以 \(c=1\).
\end{solution}

\begin{problem}
如图，在三棱台 \(DEF - ABC\) 中，\(AB = 2DE\)，\(G\)，\(H\) 分别为 \(AC\)，\(BC\) 的中点.

\begin{enumerate}
\item 求证：\(BD \parallel\) 平面 \(FGH\)；
\item 若 \(CF \perp BC\)，\(AB \perp BC\)，求证：平面 \(BCD \perp\) 平面 \(EGH\).
\end{enumerate}

\bitmapfigure[width=9.1cm]{img/2015/shandong_liberal/q18_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 连接 \(DG\)、\(CD\)，设 \(CD\) 与 \(FG\) 相交于 \(O\). 由三棱台的对应边关系，\(DF\parallel AC\)，且由 \(AB=2DE\) 得 \(DF=\frac12AC=GC\). 因而四边形 \(DFCG\) 是平行四边形，故其对角线 \(DC\)、\(FG\) 互相平分，\(O\) 是 \(CD\) 的中点. 又因为 \(H\) 是 \(BC\) 的中点，在三角形 \(BCD\) 中，\(OH\parallel BD\). 由于 \(O\)，\(H\) 都在平面 \(FGH\) 内，\(OH\subset\) 平面 \(FGH\)，所以 \(BD\parallel\) 平面 \(FGH\).

\item 在三角形 \(ABC\) 中，\(G\)，\(H\) 分别是 \(AC\)，\(BC\) 的中点，所以 \(GH\parallel AB\)，从而 \(GH\perp BC\). 又由三棱台的对应关系，\(EF\parallel BC\) 且 \(EF=\frac12BC=HC\)，因此四边形 \(EFCH\) 是平行四边形，得到 \(EH\parallel CF\). 由已知 \(CF\perp BC\)，得 \(EH\perp BC\). 直线 \(EH\)，\(GH\) 相交于 \(H\) 且都垂直于 \(BC\)，所以 \(BC\perp\) 平面 \(EGH\). 又 \(BC\subset\) 平面 \(BCD\)，根据平面与平面垂直的判定定理，得平面 \(BCD\perp\) 平面 \(EGH\).
\end{enumerate}
\end{solution}

\begin{problem}
已知数列 \(\{a_n\}\) 是首项为正数的等差数列，数列 \(\left\{\frac{1}{a_n \cdot a_{n+1}}\right\}\) 的前 \(n\) 项和为 \(\frac{n}{2n+1}\).

\begin{enumerate}
\item 求数列 \(\{a_n\}\) 的通项公式；
\item 设 \(b_n = (a_n + 1)\cdot 2^{a_n}\)，求数列 \(\{b_n\}\) 的前 \(n\) 项和 \(T_n\).
\end{enumerate}
\end{problem}

\begin{solution}
记 \(S_n=\displaystyle\sum_{k=1}^{n}\frac{1}{a_ka_{k+1}}\)，并令 \(S_0=0\). 由已知式，
\[
\frac{1}{a_na_{n+1}}=S_n-S_{n-1}
=\frac{n}{2n+1}-\frac{n-1}{2n-1}
=\frac{1}{(2n-1)(2n+1)}
\]
因此
\[
a_na_{n+1}=(2n-1)(2n+1)
\]
特别地，\(a_1a_2=3\)，\(a_2a_3=15\). 设等差数列公差为 \(d\)，则 \(a_2-a_1=a_3-a_2=d\)，从而
\[
a_2(a_3-a_1)=a_2\cdot2d=a_2a_3-a_1a_2=12
\]
所以 \(a_2d=6\). 由 \(a_1=a_2-d\) 及 \(a_1a_2=3\)，得
\[
\left(\frac6d-d\right)\frac6d=3
\]
即 \(d^2=4\). 因为 \(a_1>0\) 且 \(a_1a_2=3>0\)，有 \(a_2>0\)，再由 \(a_2d=6>0\) 得 \(d>0\)，所以 \(d=2\)，进而 \(a_2=3\)，\(a_1=1\). 因此
\[
a_n=a_1+(n-1)d=2n-1
\]

由 \(b_n=(a_n+1)2^{a_n}\)，得 \(b_n=2n\cdot2^{2n-1}=n4^n\). 设其前 \(n\) 项和为 \(T_n\)，则
\[
T_n=4+2\cdot4^2+\cdots+n4^n
\]
将等式乘以 \(4\) 后与原式相减，得
\[
3T_n=n4^{n+1}-\left(4+4^2+\cdots+4^n\right)
=n4^{n+1}-\frac{4(4^n-1)}{3}
\]
所以
\[
T_n=\frac{4+(3n-1)4^{n+1}}{9}
\]
\end{solution}

\begin{problem}
设函数 \(f(x) = (x + a)\ln x\)，\(g(x) = \frac{x^2}{\e^x}\). 已知曲线 \( y = f(x) \) 在点 \((1, f(1))\) 处的切线与直线 \( 2x - y = 0 \) 平行.

\begin{enumerate}
\item 求 \(a\) 的值；
\item 是否存在自然数 \(k\)，使得方程 \(f(x) = g(x)\) 在 \((k, k + 1)\) 内存在唯一的根？ 如果存在，求出 \(k\)；如果不存在，请说明理由；
\item 设函数 \(m(x) = \min \{f(x), g(x)\}\)（\(\min \{p, q\}\) 表示 \(p\)，\(q\) 中的较小值），求 \(m(x)\) 的最大值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item \(f(x)=(x+a)\ln x\) 的导数为 \(f'(x)=\ln x+1+\frac ax\)，所以 \(f'(1)=1+a\). 直线 \(2x-y=0\) 的斜率为 \(2\)，由切线平行条件得 \(1+a=2\)，即 \(a=1\).

\item 此时 \(f(x)=(x+1)\ln x\)，令
\[
h(x)=f(x)-g(x)=(x+1)\ln x-x^2\e^{-x}
\]
其中 \(x>0\).
当 \(0<x<1\) 时，\(f(x)<0<g(x)\)，所以没有根. 又
\(h(1)=-\e^{-1}<0\). 因为 \(\ln2>\frac12\) 且 \(\e>2\)，所以
\(h(2)=3\ln2-\frac4{\e^2}>\frac32-1>0\)，故在 \((1,2)\) 内至少有一个根.

对 \(x\geqslant1\)，有
\[
h'(x)=\ln x+1+\frac1x-x(2-x)\e^{-x}
\]
当 \(1\leqslant x\leqslant2\) 时，\(x(2-x)\leqslant1\)，故
\(h'(x)\geqslant1-\e^{-1}>0\)；当 \(x\geqslant2\) 时，\(x(2-x)\leqslant0\)，也有 \(h'(x)>0\). 因此 \(h(x)\) 在 \([1,+\infty)\) 上严格递增，方程在 \((1,2)\) 内恰有一个根，并且在其余 \((k,k+1)\)（\(k\geqslant2\)）内没有根. 所以存在的自然数为 \(k=1\).

\item 仍令 \(\alpha\) 为 \((1,2)\) 内唯一满足 \(f(\alpha)=g(\alpha)\) 的根. 当 \(0<x\leqslant1\) 时，\(m(x)=f(x)\leqslant0\). 当 \(1\leqslant x\leqslant\alpha\) 时，\(h(x)\leqslant0\)，故 \(m(x)=f(x)\). 令 \(\phi(x)=\ln x+\frac1x-1\)，则 \(\phi'(x)=\frac{x-1}{x^2}\)，所以 \(\phi(x)\) 在 \(x=1\) 处取得最小值 \(0\). 因此 \(f'(x)=\ln x+1+\frac1x=\phi(x)+2\geqslant2>0\)，此时 \(m(x)\) 递增. 当 \(x\geqslant\alpha\) 时，\(m(x)=g(x)=x^2\e^{-x}\)，且
\[
g'(x)=x(2-x)\e^{-x}
\]
所以 \(g(x)\) 在 \((0,2)\) 上递增、在 \((2,+\infty)\) 上递减. 因为 \(\alpha<2\)，\(m(x)\) 的最大值在 \(x=2\) 处取得，最大值为
\[
m(2)=g(2)=\frac4{\e^2}
\]
\end{enumerate}
\end{solution}

\begin{problem}
平面直角坐标系 \(xOy\) 中，已知椭圆 \(C: \frac{x^2}{a^2} + \frac{y^2}{b^2} = 1 \) （\(a > b > 0\)）的离心率为 \(\frac{\sqrt{3}}{2}\)，且点 \(\left(\sqrt{3}, \frac{1}{2}\right)\) 在椭圆 \(C\) 上.

\begin{enumerate}
\item 求椭圆 \(C\) 的方程；
\item 设椭圆 \(E: \frac{x^2}{4a^2} + \frac{y^2}{4b^2} = 1\)，\(P\) 为椭圆 \(C\) 上任意一点，过点 \(P\) 的直线 \(y = kx + m\) 交椭圆 \(E\) 于 \(A\)，\(B\) 两点，射线 \(PO\) 交椭圆 \(E\) 于点 \(Q\).
\begin{enumerate}
\item 求 \(\frac{|OQ|}{|OP|}\) 的值；
\item 求 \(\triangle ABQ\) 面积的最大值.
\end{enumerate}
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设椭圆半焦距为 \(c\). 由离心率，
\(\frac ca=\frac{\sqrt3}{2}\)，\(b^2=a^2-c^2=\frac{a^2}{4}\).
又点 \(\left(\sqrt3,\frac12\right)\) 在椭圆上，所以
\[
\frac{3}{a^2}+\frac{1/4}{a^2/4}=1
\]
即 \(\frac4{a^2}=1\). 因而 \(a^2=4\)，\(b^2=1\)，椭圆 \(C\) 的方程为
\[
\frac{x^2}{4}+y^2=1
\]

\item
\begin{enumerate}
\item 由上问，椭圆 \(E\) 的方程为
\[
\frac{x^2}{16}+\frac{y^2}{4}=1
\]
设 \(P=(x_0,y_0)\)，则 \(\frac{x_0^2}{4}+y_0^2=1\). 射线 \(PO\) 经过原点后与椭圆 \(E\) 相交，所以 \(Q=-\lambda P\)，其中 \(\lambda>0\). 代入椭圆 \(E\) 的方程，得
\[
\lambda^2\left(\frac{x_0^2}{16}+\frac{y_0^2}{4}\right)
=\frac{\lambda^2}{4}\left(\frac{x_0^2}{4}+y_0^2\right)=1
\]
所以 \(\lambda=2\)，从而 \(\frac{|OQ|}{|OP|}=2\).

\item 作线性变换 \(X=\frac{x}{4}\)、\(Y=\frac{y}{2}\). 在 \(X\)，\(Y\) 坐标系中，椭圆 \(E\) 变为单位圆，面积的伸缩倍数为 \(4\times2=8\). 设 \(P=(2\cos\theta,\sin\theta)\)，则变换后
\(P'=\left(\frac12\cos\theta,\frac12\sin\theta\right)\)，而由（i）知 \(Q' =(-\cos\theta,-\sin\theta)\).

设直线 \(AB\) 在变换后的坐标系中的单位法向量为 \(\bs{n}\)，记 \(u=|\bs{n}\cdot(\cos\theta,\sin\theta)|\)，则 \(0\leqslant u\leqslant1\). 直线 \(A'B'\) 到圆心的距离为 \(u/2\)，所以弦长为 \(2\sqrt{1-u^2/4}\). 又点 \(Q'\) 到直线 \(A'B'\) 的距离为 \(3u/2\). 因此
\[
S'_{\triangle A'B'Q'}
=\frac12\cdot2\sqrt{1-\frac{u^2}{4}}\cdot\frac{3u}{2}
=\frac{3u}{2}\sqrt{1-\frac{u^2}{4}}
\]
令 \(t=u^2\)，则 \(0\leqslant t\leqslant1\)，且
\[
\left(S'_{\triangle A'B'Q'}\right)^2
=\frac94t\left(1-\frac t4\right)
\]
右端关于 \(t\) 的导数为 \(\frac94(1-\frac t2)>0\)，所以在 \(t=1\)，即 \(u=1\) 时取得最大值 \(\frac{3\sqrt3}{4}\). 换回原坐标，三角形面积最大值为
\[
8\cdot\frac{3\sqrt3}{4}=6\sqrt3
\]
等号可以取得：取 \(P=(0,1)\)，直线 \(y=1\) 经过 \(P\)，在线性变换后的坐标系中有 \(u=1\). 因而上述最大值确实可达.
\end{enumerate}
\end{enumerate}
\end{solution}
