\section{单选题}

\begin{problem}
已知集合 \(A=\{-2,-1,0,1,2\}\)，\(B=\{x\mid (x-1)(x+2)<0\}\)，则 \(A\cap B\) 等于（\quad）

\choices
    {\(\{-1,0\}\)}
    {\(\{0,1\}\)}
    {\(\{-1,0,1\}\)}
    {\(\{0,1,2\}\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
由 \((x-1)(x+2)<0\) 得 \(-2<x<1\)，所以 \(B=(-2,1)\). 在集合 \(A\) 中属于该开区间的元素为 \(-1\)，\(0\)，故 \(A\cap B=\{-1,0\}\)，选 A.
\end{solution}

\begin{problem}
若 \(a\) 为实数，且 \((2+a\i)(a-2\i)=-4\i\)，则 \(a=\)（\quad）

\choices
    {\(-1\)}
    {\(0\)}
    {\(1\)}
    {\(2\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
展开左边得 \((2+a\i)(a-2\i)=4a+(a^2-4)\i\). 与 \(-4\i\) 比较实部和虚部，得到 \(4a=0\)，且 \(a^2-4=-4\). 因而 \(a=0\)，选 B.
\end{solution}

\begin{problem}
已知等比数列 \(\{a_n\}\) 满足 \(a_1=3\)，\(a_1+a_3+a_5=21\)，则 \(a_3+a_5+a_7=\)（\quad）

\choices
    {\(21\)}
    {\(42\)}
    {\(63\)}
    {\(84\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
设等比数列的公比为 \(q\). 由 \(a_1=3\) 及 \(a_1+a_3+a_5=21\)，得 \(3(1+q^2+q^4)=21\)，即 \(q^4+q^2-6=0\). 令 \(u=q^2\geqslant0\)，则 \((u+3)(u-2)=0\)，所以 \(q^2=2\). 因此 \(a_3+a_5+a_7=a_1q^2(1+q^2+q^4)=3\cdot2\cdot(1+2+4)=42\)，选 B.
\end{solution}

\begin{problem}
设函数 \(f(x)=\begin{cases}
1+\log_2(2-x), & x<1,\\
2^{x-1}, & x\geqslant 1,
\end{cases}\) 则 \(f(-2)+f(\log_2 12)=\)（\quad）

\choices
    {\(3\)}
    {\(6\)}
    {\(9\)}
    {\(12\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
因为 \(-2<1\)，所以 \(f(-2)=1+\log_2 4=3\). 又因为 \(\log_2 12>1\)，应取第二段，故 \(f(\log_2 12)=2^{\log_2 12-1}=\frac{12}{2}=6\). 因而 \(f(-2)+f(\log_2 12)=9\)，选 C.
\end{solution}

一个正方体被一个平面截去一部分后，剩余部分的三视图如图，则截去部分体积与剩余部分体积的比值为（\quad）

\begin{answer}
D
\end{answer}

\begin{problem}
过三点 \(A(1,3)\)，\(B(4,2)\)，\(C(1,-7)\) 的圆交 \(y\) 轴于 \(M\)，\(N\) 两点，则 \(|MN|=\)（\quad）

\choices
    {\(2\sqrt{6}\)}
    {\(8\)}
    {\(4\sqrt{6}\)}
    {\(10\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
设圆的方程为 \(x^2+y^2+Dx+Ey+F=0\). 代入 \(A(1,3)\)，\(B(4,2)\)，\(C(1,-7)\)，分别得到
\(10+D+3E+F=0\)，\(20+4D+2E+F=0\)，\(50+D-7E+F=0\).
相减得 \(E=4\)，\(D=-2\)，再代入得 \(F=-20\). 圆与 \(y\) 轴的交点满足 \(y^2+4y-20=0\)，其两根之差的绝对值为 \(\sqrt{16+80}=4\sqrt6\). 故 \(|MN|=4\sqrt6\)，选 C.
\end{solution}

下边程序框图的算法思路源于我国古代数学名著《九章算术》中的“更相减损术”. 执行该程序框图，若输入的 \(a\)，\(b\) 分别为 \(14\)，\(18\)，则输出的 \(a=\)（\quad）

\begin{problem}
已知 \(A\)，\(B\) 是球 \(O\) 的球面上两点，\(\angle AOB=90^\circ\)，\(C\) 为该球面上的动点，若三棱锥 \(O-ABC\) 体积的最大值为 \(36\)，则球 \(O\) 的表面积为（\quad）

\choices
    {\(36\pi\)}
    {\(64\pi\)}
    {\(144\pi\)}
    {\(256\pi\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
设球的半径为 \(R\). 取空间直角坐标系，使 \(O=(0,0,0)\)，\(A=(R,0,0)\)，\(B=(0,R,0)\). 设 \(C=(x,y,z)\)，则 \(x^2+y^2+z^2=R^2\). 三棱锥 \(O-ABC\) 的体积为
\(V=\frac16\left|\det\begin{pmatrix}R&0&0\\0&R&0\\x&y&z\end{pmatrix}\right|=\frac16R^2|z|\).
由于 \(|z|\leqslant R\)，且 \(C\) 可取为球的南、北极，体积最大值为 \(\frac16R^3\). 由 \(\frac16R^3=36\) 得 \(R=6\)，所以球面面积为 \(4\pi R^2=144\pi\)，选 C.
\end{solution}

如图，长方形 \(ABCD\) 的边 \(AB=2\)，\(BC=1\)，\(O\) 是 \(AB\) 的中点，点 \(P\) 沿着边 \(BC\)，\(CD\) 与 \(DA\) 运动，记 \(\angle BOP=x\). 将动点 \(P\) 到 \(A\)，\(B\) 两点距离之和表示为 \(x\) 的函数 \(f(x)\)，则 \(y=f(x)\) 的图象大致为（\quad）

\bitmapfigure[max width=0.92\linewidth]{img/2015/national_paper_2_science/q10_fig1.png}

\choices
    {\(\choicebitmap[width=0.15\paperwidth]{img/2015/national_paper_2_science/q10_optA.png}\)}
    {\(\choicebitmap[width=0.15\paperwidth]{img/2015/national_paper_2_science/q10_optB.png}\)}

\begin{answer}
B
\end{answer}

\begin{solution}
取 \(A=(-1,0)\)，\(O=(0,0)\)，\(B=(1,0)\)，\(C=(1,1)\)，\(D=(-1,1)\). 当 \(P\) 在 \(BC\) 上运动时，令 \(P=(1,y)\)，其中 \(0\leqslant y\leqslant1\)，则 \(x=\arctan y\)，且 \(f(x)=\sqrt{4+y^2}+y\)，故函数在 \(0\leqslant x\leqslant\frac\pi4\) 上递增.

\(f(x)=\sqrt{(u+1)^2+1}+\sqrt{(u-1)^2+1}\). 记右侧为 \(F(u)\)，则 \(F\) 是偶函数. 令 \(\varphi(t)=\frac{t}{\sqrt{t^2+1}}\)，则 \(\varphi'(t)=\frac{1}{(t^2+1)^{3/2}}>0\). 当 \(0\leqslant u\leqslant1\) 时，
\(F'(u)=\varphi(u+1)+\varphi(u-1)=\varphi(u+1)-\varphi(1-u)\geqslant0\)，所以 \(F\) 在 \([0,1]\) 上递增. 由于 \(u=\cot x\) 随 \(x\) 增大而减小，函数在 \(CD\) 上先减后增. 当 \(P\) 在 \(DA\) 上时，函数由 \(\sqrt5+1\) 下降到 \(2\). 因而图象依次呈现“上升、下降、上升、下降”，且关于 \(x=\frac\pi2\) 对称，符合图 B，选 B.

\begin{problem}
已知 \(A\)，\(B\) 为双曲线 \(E\) 的左，右顶点，点 \(M\) 在 \(E\) 上，\(\triangle ABM\) 为等腰三角形，且顶角为 \(120^\circ\)，则 \(E\) 的离心率为（\quad）

\choices
    {\(\sqrt{5}\)}
    {\(2\)}
    {\(\sqrt{3}\)}
    {\(\sqrt{2}\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
设双曲线方程为 \(\frac{x^2}{a^2}-\frac{y^2}{b^2}=1\)，其中 \(a,b>0\)，则 \(A=(-a,0)\)，\(B=(a,0)\). 若等腰三角形的顶角在 \(M\)，则 \(MA=MB\)，从而 \(M\) 在 \(AB\) 的垂直平分线上，即 \(M\) 的横坐标为 \(0\)，这与双曲线方程矛盾. 因此顶角只能在 \(A\) 或 \(B\)，由双曲线关于 \(y\) 轴对称，不妨设顶角在 \(B\)，于是 \(BA=BM=2a\)，且 \(\angle ABM=120^\circ\). 由对称性可取 \(M\) 在上方，则 \(M=(2a,\sqrt3a)\). 代入双曲线方程得 \(4-\frac{3a^2}{b^2}=1\)，从而 \(b^2=a^2\). 双曲线的离心率为 \(e=\frac{\sqrt{a^2+b^2}}{a}=\sqrt2\)，选 D.
\end{solution}

\begin{problem}
设函数 \(f'(x)\) 是奇函数 \(f(x)\)（\(x\in\R\)）的导函数，\(f(-1)=0\)，当 \(x>0\) 时，\(xf'(x)-f(x)<0\)，则使得 \(f(x)>0\) 成立的 \(x\) 的取值范围是（\quad）

\choices
    {\((-\infty,-1)\cup(0,1)\)}
    {\((-1,0)\cup(1,+\infty)\)}
    {\((-\infty,-1)\cup(-1,0)\)}
    {\((0,1)\cup(1,+\infty)\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
题设中的奇函数为 \(f(x)\)，所以 \(f\) 为奇函数，且 \(f(1)=-f(-1)=0\). 对 \(x>0\)，令 \(g(x)=\frac{f(x)}{x}\)，则
\(g'(x)=\frac{xf'(x)-f(x)}{x^2}<0\)，故 \(g\) 在 \((0,+\infty)\) 上递减. 由 \(g(1)=0\) 得，当 \(0<x<1\) 时 \(g(x)>0\)，从而 \(f(x)>0\)；当 \(x>1\) 时 \(g(x)<0\)，从而 \(f(x)<0\). 又 \(f\) 为奇函数，故 \(x<-1\) 时 \(f(x)>0\)，而 \(-1<x<0\) 时 \(f(x)<0\)，且 \(f(0)=0\). 所以使 \(f(x)>0\) 成立的 \(x\) 的取值范围为 \(\left(-\infty,-1\right)\cup\left(0,1\right)\)，选 A.
\end{solution}

\section{填空题}

\begin{problem}
设向量 \(\bs{a}\)，\(\bs{b}\) 不平行，向量 \(\lambda\bs{a}+\bs{b}\) 与 \(\bs{a}+2\bs{b}\) 平行，则实数 \(\lambda=\)\fillinblank{}.
\end{problem}

\begin{answer}
\(\frac{1}{2}\)
\end{answer}

\begin{solution}
由于 \(\lambda\bs{a}+\bs{b}\) 与 \(\bs{a}+2\bs{b}\) 平行，存在实数 \(k\)，使得 \(\lambda\bs{a}+\bs{b}=k\left(\bs{a}+2\bs{b}\right)\). 因为 \(\bs{a}\)，\(\bs{b}\) 不平行，比较两边关于这两个向量的系数，得 \(\lambda=k\)，\(1=2k\). 所以 \(\lambda=\frac12\).
\end{solution}

\begin{problem}
若 \(x\)，\(y\) 满足约束条件 \(\begin{cases}
x-y+1\geqslant 0,\\
x-2y\leqslant 0,\\
x+2y-2\leqslant 0,
\end{cases}\) 则 \(z=x+y\) 的最大值为\fillinblank{}.
\end{problem}

\begin{answer}
\(\frac{3}{2}\)
\end{answer}

\begin{solution}
三个约束边界分别为 \(y=x+1\)，\(y=\frac{x}{2}\)，\(y=1-\frac{x}{2}\). 可行域的三个顶点由两两联立得到：
\((-2,-1)\)，\((0,1)\)，\(\left(1,\frac12\right)\).
在这三个点处，\(z=x+y\) 的值分别为 \(-3\)，\(1\)，\(\frac32\). 因而 \(z\) 的最大值为 \(\frac32\).
\end{solution}

\begin{problem}
\((a+x)(1+x)^4\) 的展开式中 \(x\) 的奇数次幂项的系数之和为 \(32\)，则 \(a=\)\fillinblank{}.
\end{problem}

\begin{answer}
3
\end{answer}

\begin{solution}
设 \(P(x)=(a+x)(1+x)^4\). 奇数次幂项的系数之和等于 \(\frac{P(1)-P(-1)}{2}\). 计算得 \(P(1)=16(a+1)\)，\(P(-1)=0\)，所以 \(8(a+1)=32\)，解得 \(a=3\).
\end{solution}

\begin{problem}
设 \(S_n\) 是数列 \(\{a_n\}\) 的前 \(n\) 项和，且 \(a_1=-1\)，\(a_{n+1}=S_nS_{n+1}\)，则 \(S_n=\)\fillinblank{}.
\end{problem}

\begin{answer}
\(-\frac{1}{n}\)
\end{answer}

\begin{solution}
由 \(a_{n+1}=S_{n+1}-S_n\) 及 \(a_{n+1}=S_nS_{n+1}\)，得 \(S_{n+1}-S_n=S_nS_{n+1}\)，即 \(S_{n+1}(1-S_n)=S_n\). 初值 \(S_1=a_1=-1\). 若 \(S_n<0\)，则由上式得 \(S_{n+1}=\frac{S_n}{1-S_n}<0\)，所以由数学归纳法知 \(S_n<0\)，各项均非零，并且
\(\frac{1}{S_{n+1}}=\frac{1}{S_n}-1\).
递推 \(n-1\) 次，得 \(\frac{1}{S_n}=\frac{1}{S_1}-(n-1)=-1-(n-1)=-n\)，所以 \(S_n=-\frac1n\).
\end{solution}

\begin{problem}
在 \(\triangle ABC\) 中，\(D\) 是 \(BC\) 上的点，\(AD\) 平分 \(\angle BAC\)，\(\triangle ABD\) 面积是 \(\triangle ADC\) 面积的 \(2\) 倍.

\begin{enumerate}
\item 求 \(\frac{\sin B}{\sin C}\)；
\item 若 \(AD=1\)，\(DC=\frac{\sqrt{2}}{2}\)，求 \(BD\) 和 \(AC\) 的长.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由题意，\(AD\) 是同一高，所以 \(\frac{[\triangle ABD]}{[\triangle ADC]}=\frac{BD}{DC}=2\)，即 \(BD=2DC\). 又由角平分线定理，\(\frac{AB}{AC}=\frac{BD}{DC}=2\). 根据正弦定理，\(\frac{AB}{AC}=\frac{\sin C}{\sin B}\)，因此 \(\frac{\sin B}{\sin C}=\frac12\).
\item 由 \(BD=2DC\) 及 \(DC=\frac{\sqrt2}{2}\)，得 \(BD=\sqrt2\). 设 \(AC=x\)，则 \(AB=2x\). 由角平分线长度公式 \(AD^2=AB\cdot AC-BD\cdot DC\)，得 \(1=2x^2-\sqrt2\cdot\frac{\sqrt2}{2}=2x^2-1\)，所以 \(x=1\). 因而 \(BD=\sqrt2\)，\(AC=1\).
\end{enumerate}
\end{solution}

\begin{problem}
已知椭圆 \(C\)：\(9x^2+y^2=m^2\)（\(m>0\)），直线 \(l\) 不过原点 \(O\) 且不平行于坐标轴，\(l\) 与 \(C\) 有两个交点 \(A\)，\(B\)，线段 \(AB\) 的中点为 \(M\).

\begin{enumerate}
\item 证明：直线 \(OM\) 的斜率与 \(l\) 的斜率的乘积为定值；
\item 若 \(l\) 过点 \(\left(\frac{m}{3},m\right)\)，延长线段 \(OM\) 与 \(C\) 交于点 \(P\)，四边形 \(OAPB\) 能否为平行四边形？若能，求此时 \(l\) 的斜率；若不能，说明理由.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设直线 \(l\) 的斜率为 \(k\)，方程为 \(y=kx+b\). 由题设 \(k\ne0\)，\(b\ne0\). 将直线方程代入椭圆方程，得 \((9+k^2)x^2+2kbx+b^2-m^2=0\). 设交点 \(A\)，\(B\) 的横坐标为 \(x_1\)，\(x_2\)，则 \(x_M=\frac{x_1+x_2}{2}=-\frac{kb}{9+k^2}\)，\(y_M=kx_M+b=\frac{9b}{9+k^2}\). 因而直线 \(OM\) 的斜率为 \(\frac{y_M}{x_M}=-\frac9k\)，它与 \(l\) 的斜率之积为 \(-9\).
\item 由 \(l\) 过 \(\left(\frac{m}{3},m\right)\)，得 \(b=m\left(1-\frac{k}{3}\right)=\frac{m(3-k)}{3}\). 四边形 \(OAPB\) 为平行四边形时，其对角线 \(OP\) 与 \(AB\) 互相平分，所以 \(\overrightarrow{OP}=2\overrightarrow{OM}\)，即 \(P=(2x_M,2y_M)\). 由于 \(P\) 在椭圆上，得 \(9x_M^2+y_M^2=\frac{m^2}{4}\). 由上式中的 \(x_M,y_M\)，有 \(9x_M^2+y_M^2=\frac{9b^2}{9+k^2}\)，代入 \(b=\frac{m(3-k)}{3}\)，得到 \(\frac{(3-k)^2}{9+k^2}=\frac14\). 化简得 \(k^2-8k+9=0\)，所以 \(k=4-\sqrt7\) 或 \(k=4+\sqrt7\). 两个值均使直线不经过原点且不平行于坐标轴，故均可取.
\(\Delta=4\bigl((9+k^2)m^2-9b^2\bigr)=24km^2>0\)，故直线确与椭圆有两个交点；且两值均不为 \(0\) 或 \(3\)，分别保证直线不平行于坐标轴且不经过原点，故均可取.
\end{enumerate}
\end{solution}

\begin{problem}
设函数 \(f(x)=\e^{mx}+x^2-mx\).

\begin{enumerate}
\item 证明：\(f(x)\) 在 \((-\infty,0)\) 单调递减，在 \((0,+\infty)\) 单调递增；
\item 若对于任意 \(x_1,x_2\in[-1,1]\)，都有 \(\left|f(x_1)-f(x_2)\right|\leqslant \e-1\)，求 \(m\) 的取值范围.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item \(f'(x)=m\e^{mx}+2x-m=m\left(\e^{mx}-1\right)+2x\). 当 \(x>0\) 时，\(m\) 与 \(\e^{mx}-1\) 同号或其中一个为零，所以 \(m\left(\e^{mx}-1\right)\geqslant0\)，从而 \(f'(x)>0\). 当 \(x<0\) 时，\(m\left(\e^{mx}-1\right)\leqslant0\)，且 \(2x<0\)，从而 \(f'(x)<0\). 因此 \(f(x)\) 在 \((-\infty,0)\) 上单调递减，在 \((0,+\infty)\) 上单调递增.
\item 由（1）知，\(f\) 在 \([-1,1]\) 上以 \(f(0)=1\) 为最小值，因此题设条件等价于 \(\e^m-m\leqslant\e-1\)，\(\e^{-m}+m\leqslant\e-1\). 令 \(g(t)=\e^t-t\)，则 \(g'(t)=\e^t-1\)，所以 \(g\) 在 \((-\infty,0)\) 上递减，在 \((0,+\infty)\) 上递增. 当 \(-1\leqslant t\leqslant1\) 时，\(g(t)\leqslant g(1)=\e-1\)，且 \(g(-1)=\e^{-1}+1<\e-1\). 当 \(m>1\) 时 \(g(m)>g(1)\)，当 \(m<-1\) 时 \(g(-m)>g(1)\). 所以两个不等式同时成立的充要条件为 \(-1\leqslant m\leqslant1\).
\end{enumerate}
\end{solution}

\section{选考题：解答题}

\begin{problem}
如图，\(O\) 为等腰三角形 \(ABC\) 内一点，\(\odot O\) 与 \(\triangle ABC\) 的底边 \(BC\) 交于 \(M\)，\(N\) 两点，与底边上的高 \(AD\) 交于点 \(G\)，且与 \(AB\)，\(AC\) 分别相切于 \(E\)，\(F\) 两点.

\begin{enumerate}
\item 证明：\(EF\parallel BC\)；
\item 若 \(AG\) 等于 \(\odot O\) 的半径，且 \(AE=MN=2\sqrt{3}\)，求四边形 \(EBCF\) 的面积.
\end{enumerate}

\bitmapfigure[max width=0.48\linewidth]{img/2015/national_paper_2_science/q22_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 因为圆 \(\odot O\) 分别与 \(AB\)，\(AC\) 相切，所以圆心 \(O\) 到 \(AB\)，\(AC\) 的距离相等，\(O\) 在 \(\angle A\) 的平分线上. 等腰三角形的高 \(AD\) 也是 \(\angle A\) 的平分线，故 \(O\) 在 \(AD\) 上. 由切线与半径垂直，\(OE\perp AB\)，\(OF\perp AC\). 关于对称轴 \(AD\) 对称的两条切线的切点 \(E\)，\(F\) 的连线垂直于 \(AD\)，而 \(BC\perp AD\)，所以 \(EF\parallel BC\).
\item 设 \(\odot O\) 的半径为 \(r\). 由 \(G\) 是圆在 \(AD\) 上靠近 \(A\) 的交点，且 \(AG=r\)，\(OG=r\)，得 \(AO=2r\). 在直角三角形 \(AOE\) 中，\(OE=r\)，所以 \(AE^2=AO^2-OE^2=3r^2\). 由 \(AE=2\sqrt3\) 得 \(r=2\).

取 \(O\) 为原点，\(AD\) 为 \(y\) 轴，令 \(A=(0,4)\)，则圆方程为 \(x^2+y^2=4\). 设切点 \(E=(u,v)\)，由 \(OE\perp AE\) 得 \((u,v)\cdot(u,v-4)=0\)，即 \(u^2+v^2-4v=0\). 因为 \(u^2+v^2=4\)，故 \(v=1\)，从而 \(u=\sqrt3\). 对称地 \(F=(-\sqrt3,1)\)，所以 \(EF=2\sqrt3\).

设 \(BC\) 到 \(O\) 的距离为 \(h\). 弦 \(MN=2\sqrt3\) 满足 \(MN=2\sqrt{r^2-h^2}\)，所以 \(2\sqrt3=2\sqrt{4-h^2}\)，得 \(h=1\). 因而 \(BC\) 所在直线为 \(y=-1\)，而 \(EF\) 所在直线为 \(y=1\)，两平行线间距离为 \(2\). 直线 \(AB\) 经过 \(A=(0,4)\) 和 \(E=(\sqrt3,1)\)，方程为 \(y=4-\sqrt3x\). 与 \(y=-1\) 联立得 \(x=\frac5{\sqrt3}\)，所以 \(BC=2\cdot\frac5{\sqrt3}=\frac{10\sqrt3}{3}\). 因此四边形 \(EBCF\) 是以 \(EF\)，\(BC\) 为两底的梯形，面积为 \(\frac{EF+BC}{2}\cdot2=2\sqrt3+\frac{10\sqrt3}{3}=\frac{16\sqrt3}{3}\).
\end{enumerate}
\end{solution}

\begin{problem}
在直角坐标系 \(xOy\) 中，曲线 \(C_1\)：\(\begin{cases}
x=t\cos\alpha,\\
y=t\sin\alpha,
\end{cases}\)（\(t\) 为参数，\(t\neq 0\)），其中 \(0\leqslant\alpha<\pi\). 在以 \(O\) 为极点，\(x\) 轴正半轴为极轴的极坐标系中，曲线 \(C_2\)：\(\rho=2\sin\theta\)，\(C_3\)：\(\rho=2\sqrt{3}\cos\theta\).

\begin{enumerate}
\item 求 \(C_2\) 与 \(C_3\) 交点的直角坐标；
\item 若 \(C_1\) 与 \(C_2\) 相交于点 \(A\)，\(C_1\) 与 \(C_3\) 相交于点 \(B\)，求 \(|AB|\) 的最大值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由极坐标与直角坐标的关系，\(C_2\) 的方程化为 \(x^2+y^2=2y\)，\(C_3\) 的方程化为 \(x^2+y^2=2\sqrt3x\). 两式相减得 \(y=\sqrt3x\). 代回 \(x^2+y^2=2y\)，得 \(4x^2=2\sqrt3x\)，所以 \(x=0\) 或 \(x=\frac{\sqrt3}{2}\). 相应的 \(y\) 分别为 \(0\) 和 \(\frac32\)，故交点为 \((0,0)\) 和 \(\left(\frac{\sqrt3}{2},\frac32\right)\).
\item 由 \(C_1\) 的参数方程，代入 \(C_2\) 的直角坐标方程得 \(t^2=2t\sin\alpha\). 因 \(t\ne0\)，所以 \(t=2\sin\alpha\)，从而 \(A=(2\sin\alpha\cos\alpha,2\sin^2\alpha)\). 同理，代入 \(C_3\) 得 \(t=2\sqrt3\cos\alpha\)，从而 \(B=(2\sqrt3\cos^2\alpha,2\sqrt3\sin\alpha\cos\alpha)\). 于是
\(|AB|^2=4(\sin\alpha-\sqrt3\cos\alpha)^2\).
由于 \((\cos\alpha,\sin\alpha)\) 是单位向量，\(\sin\alpha-\sqrt3\cos\alpha\) 的绝对值不超过 \(\sqrt{1+3}=2\)，且在 \(\alpha=\frac{5\pi}{6}\) 时取到 \(2\). 因此 \(|AB|_{\max}=4\).
\end{enumerate}
\end{solution}

\begin{problem}
设 \(a\)，\(b\)，\(c\)，\(d\) 均为正数，且 \(a+b=c+d\). 证明：

\begin{enumerate}
\item 若 \(ab>cd\)，则 \(\sqrt{a}+\sqrt{b}>\sqrt{c}+\sqrt{d}\)；
\item \(\sqrt{a}+\sqrt{b}>\sqrt{c}+\sqrt{d}\) 是 \(\left|a-b\right|<\left|c-d\right|\) 的充要条件.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由 \(a+b=c+d\)，得 \((\sqrt a+\sqrt b)^2=a+b+2\sqrt{ab}\)，\((\sqrt c+\sqrt d)^2=c+d+2\sqrt{cd}\). 当 \(ab>cd\) 时，\(\sqrt{ab}>\sqrt{cd}\)，结合 \(a+b=c+d\)，有 \((\sqrt a+\sqrt b)^2>(\sqrt c+\sqrt d)^2\). 两边均为正数，所以 \(\sqrt a+\sqrt b>\sqrt c+\sqrt d\).
\item 记 \(S=a+b=c+d\). 由 \((a-b)^2=S^2-4ab\)，\((c-d)^2=S^2-4cd\)，可得 \(|a-b|<|c-d|\Longleftrightarrow S^2-4ab<S^2-4cd\Longleftrightarrow ab>cd\). 又因为两边均为正数，结合 \(a+b=c+d\)，有
\(\sqrt a+\sqrt b>\sqrt c+\sqrt d\Longleftrightarrow(\sqrt a+\sqrt b)^2>(\sqrt c+\sqrt d)^2\Longleftrightarrow ab>cd\). 因而 \(\sqrt a+\sqrt b>\sqrt c+\sqrt d\) 是 \(|a-b|<|c-d|\) 的充要条件.
\end{enumerate}
\end{solution}
