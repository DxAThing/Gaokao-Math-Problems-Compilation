\section{单选题}

\begin{problem}
设 \(i\) 是虚数单位，则复数 \( (1 - \i)(1 + 2\i) = \)（\quad）

\choices
    {\(3 + 3\i\)}
    {\(-1 + 3\i\)}
    {\(3 + \i\)}
    {\(-1 + \i\)}

\begin{answer}
C
\end{answer}

\begin{solution}
利用 \(\i^2=-1\)，直接展开得
\[
(1-\i)(1+2\i)=1+2\i-\i-2\i^2=1+\i+2=3+\i
\]
所以应选 C.
\end{solution}
\end{problem}

\begin{problem}
设全集 \(U = \{1, 2, 3, 4, 5, 6\}\)，\(A = \{1, 2\}\)，\(B = \{2, 3, 4\}\)，则 \(A \cap (\complement_U B) =\)（\quad）

\choices
    {\(\{1, 2, 5, 6\}\)}
    {\(\{1\}\)}
    {\(\{2\}\)}
    {\(\{1, 2, 3, 4\}\)}

\begin{answer}
B
\end{answer}

\begin{solution}
在全集 \(U\) 中去掉集合 \(B\) 的元素，得到 \(\complement_U B=\{1,5,6\}\). 因而
\[
A\cap(\complement_U B)=\{1,2\}\cap\{1,5,6\}=\{1\}
\]
所以应选 B.
\end{solution}
\end{problem}

\begin{problem}
设 \(p: x < 3\)，\(q: -1 < x < 3\)，则 \(p\) 是 \(q\) 成立的（\quad）

\choices
    {充分必要条件}
    {充分不必要条件}
    {必要不充分条件}
    {既不充分也不必要条件}

\begin{answer}
C
\end{answer}

\begin{solution}
若 \(q\) 成立，则 \(-1<x<3\)，必有 \(p\) 成立，因此 \(p\) 是 \(q\) 成立的必要条件. 但 \(p\) 成立不一定有 \(q\) 成立，例如 \(x=-2\) 时 \(p\) 成立而 \(q\) 不成立，所以 \(p\) 不是充分条件. 因此 \(p\) 是 \(q\) 成立的必要不充分条件，应选 C.
\end{solution}
\end{problem}

\begin{problem}
下列函数中，既是偶函数又存在零点的是（\quad）

\choices
    {\(y = \cos x\)}
    {\(y = \sin x\)}
    {\(y = \ln x\)}
    {\(y = x^{2} + 1\)}

\begin{answer}
A
\end{answer}

\begin{solution}
函数 \(y=\cos x\) 的定义域为 \(\R\)，且 \(\cos(-x)=\cos x\)，所以它是偶函数；又 \(\cos\frac{\pi}{2}=0\)，所以它存在零点. 函数 \(y=\sin x\) 满足 \(\sin(-x)=-\sin x\)，是奇函数；函数 \(y=\ln x\) 的定义域为 \((0,+\infty)\)，不是关于原点对称的集合，不是偶函数；函数 \(y=x^2+1\) 虽为偶函数，但对任意实数 \(x\) 都有 \(x^2+1>0\)，没有零点. 所以应选 A.
\end{solution}
\end{problem}

\begin{problem}
已知 \(x\)，\(y\) 满足约束条件 \(\begin{cases}
x - y \geqslant 0,\\
x + y - 4 \leqslant 0,\\
y \geqslant 1,
\end{cases}\)，则 \(z = -2x + y\) 的最大值是（\quad）

\choices
    {\(-1\)}
    {\(-2\)}
    {\(-5\)}
    {1}

\begin{answer}
A
\end{answer}

\begin{solution}
由约束条件得 \(x\geqslant y\) 且 \(y\geqslant1\). 因为 \(z=-2x+y\)，对固定的 \(y\)，当 \(x\) 取允许的最小值时 \(z\) 最大；而 \(x\geqslant y\)，所以
\[
z=-2x+y\leqslant-2y+y=-y\leqslant-1
\]
取 \(x=y=1\) 时，三个约束条件分别取值为 \(0\)、\(-2\)、\(1\)，均满足，且 \(z=-1\). 因而最大值为 \(-1\)，应选 A.
\end{solution}
\end{problem}

\begin{problem}
下列双曲线中，渐近线方程为 \( y = \pm 2x \) 的是（\quad）

\choices
    {\(x^{2} - \frac{y^{2}}{4} = 1\)}
    {\(\frac{x^2}{4} - y^2 = 1\)}
    {\(x^{2} - \frac{y^{2}}{2} = 1\)}
    {\(\frac{x^2}{2} - y^2 = 1\)}

\begin{answer}
A
\end{answer}

\begin{solution}
双曲线 \(\frac{x^2}{a^2}-\frac{y^2}{b^2}=1\) 的渐近线为 \(y=\pm\frac{b}{a}x\). 选项 A 可写成 \(\frac{x^2}{1^2}-\frac{y^2}{2^2}=1\)，其渐近线为 \(y=\pm2x\). 其余选项对应的 \(\frac{b}{a}\) 分别为 \(\frac12\)、\(\sqrt2\)、\(\frac{1}{\sqrt2}\)，均不等于 \(2\). 所以应选 A.
\end{solution}
\end{problem}

\begin{problem}
执行如图所示的程序框图（算法流程图），输出的 \(n\) 为（\quad）

\texfigure[max width=0.52\linewidth]{img_repaint/2015/anhui_liberal/q7_fig1.tex}

\choices
    {\(3\)}
    {\(4\)}
    {\(5\)}
    {\(6\)}

\begin{answer}
B
\end{answer}

\begin{solution}
程序开始时 \(a=1\)，\(n=1\). 第一次判断中，\(|1-1.414|=0.414\geqslant0.005\)，所以更新得 \(a=1+\frac{1}{1+1}=1.5\)，并令 \(n=2\).

第二次判断中，\(|1.5-1.414|=0.086\geqslant0.005\)，所以更新得 \(a=1+\frac{1}{1+1.5}=1.4\)，并令 \(n=3\).

第三次判断中，\(|1.4-1.414|=0.014\geqslant0.005\)，所以更新得 \(a=1+\frac{1}{1+1.4}=1+\frac{5}{12}=\frac{17}{12}\)，并令 \(n=4\).

此时
\[
\left|\frac{17}{12}-1.414\right|=\left|\frac{17}{12}-\frac{707}{500}\right|=\frac{1}{375}<0.005
\]
满足停止条件，程序输出 \(n=4\). 所以应选 B.
\end{solution}
\end{problem}

\begin{problem}
直线 \(3x + 4y = b\) 与圆 \(x^{2} + y^{2} - 2x - 2y + 1 = 0\) 相切，则 \(b\) 的值是（\quad）

\choices
    {\(-2\) 或 \(12\)}
    {\(2\) 或 \(-12\)}
    {\(-2\) 或 \(-12\)}
    {\(2\) 或 \(12\)}

\begin{answer}
D
\end{answer}

\begin{solution}
将圆的方程配方，得
\[
(x-1)^2+(y-1)^2=1
\]
所以圆心为 \((1,1)\)，半径为 \(1\). 圆心到直线 \(3x+4y-b=0\) 的距离为
\[
\frac{|3\cdot1+4\cdot1-b|}{\sqrt{3^2+4^2}}=\frac{|7-b|}{5}
\]
相切时该距离等于半径，故 \(|7-b|=5\)，从而 \(b=2\) 或 \(b=12\). 所以应选 D.
\end{solution}
\end{problem}

\begin{problem}
一个四面体的三视图如图所示，则该四面体的表面积是（\quad）

\bitmapfigure[max width=0.52\linewidth]{img/2015/anhui_liberal/q9_fig1.png}

\choices
    {\(1 + \sqrt{3}\)}
    {\(2 + \sqrt{3}\)}
    {\(1 + 2\sqrt{2}\)}
    {\(2\sqrt{2}\)}

\begin{answer}
B
\end{answer}

\begin{solution}
根据三视图建立直角坐标系，可取四面体的四个顶点为
\(A=(-1,0,0)\)、\(B=(1,0,0)\)、\(C=(0,1,0)\)、\(P=(0,0,1)\).
在正视图中，\(A\)，\(B\) 的投影为底边两端，\(P\) 的投影为顶点，\(C\) 的投影为底边中点；在侧视图中，\(A\)，\(B\) 的投影重合，\(C\) 与该投影的水平距离为 \(1\)，\(P\) 与该投影的竖直距离为 \(1\)；俯视图也与上述坐标投影一致，因此该坐标模型与三视图一致.

由坐标可得 \(AB=2\)，且 \(AC=BC=AP=BP=CP=\sqrt2\). 三角形 \(ABC\) 和 \(PAB\) 的底边都为 \(2\)，对应高都为 \(1\)，所以
\[
S_{ABC}=S_{PAB}=\frac12\times2\times1=1
\]
三角形 \(PAC\) 和 \(PBC\) 的三边都为 \(\sqrt2\)，均为等边三角形，故
\[
S_{PAC}=S_{PBC}=\frac{\sqrt3}{4}(\sqrt2)^2=\frac{\sqrt3}{2}
\]
所以四面体的表面积为
\[
S=1+1+\frac{\sqrt3}{2}+\frac{\sqrt3}{2}=2+\sqrt3
\]
因此应选 B.
\end{solution}
\end{problem}

\begin{problem}
函数 \( f(x) = ax^3 + bx^2 + cx + d \) 的图象如图所示，则下列结论成立的是（\quad）

\bitmapfigure[max width=0.42\linewidth]{img/2015/anhui_liberal/q10_fig1.png}

\choices
    {\(a > 0\)，\(b < 0\)，\(c > 0\)，\(d > 0\)}
    {\(a > 0\)，\(b < 0\)，\(c < 0\)，\(d > 0\)}
    {\(a < 0\)，\(b < 0\)，\(c > 0\)，\(d > 0\)}
    {\(a > 0\)，\(b > 0\)，\(c > 0\)，\(d < 0\)}

\begin{answer}
A
\end{answer}

\begin{solution}
由图象右端向上，知三次项系数 \(a>0\). 设两个极值点的横坐标为 \(x_1\)、\(x_2\)，由图可知 \(0<x_1<x_2\). 因为
\[
f'(x)=3ax^2+2bx+c
\]
且 \(x_1\)，\(x_2\) 是方程 \(f'(x)=0\) 的两个根，所以
\[
x_1+x_2=-\frac{2b}{3a}>0
\]
结合 \(a>0\)，得 \(b<0\). 图象在 \(x=0\) 处递增，所以 \(c=f'(0)>0\)；又图象与 \(y\) 轴的交点在 \(x\) 轴上方，所以 \(d=f(0)>0\). 因此只有选项 A 成立.
\end{solution}
\end{problem}

\begin{problem}
计算： \( \lg\frac{5}{2}+2\lg2-\left(\frac{1}{2}\right)^{-1}=\)\fillinblank{}.

\begin{answer}
\(-1\)
\end{answer}

\begin{solution}
由对数运算性质，
\[
\lg\frac52+2\lg2=\lg\frac52+\lg4=\lg10=1
\]
又 \(\left(\frac12\right)^{-1}=2\)，所以原式的值为 \(1-2=-1\).
\end{solution}
\end{problem}

\begin{problem}
在 \(\triangle ABC\) 中，\(AB = \sqrt{6}\)，\(\angle A = 75^{\circ}\)，\(\angle B = 45^{\circ}\)，则 \(AC =\)\fillinblank{}.

\begin{answer}
\(2\)
\end{answer}

\begin{solution}
三角形内角和给出 \(\angle C=180^{\circ}-75^{\circ}-45^{\circ}=60^{\circ}\). 由正弦定理，
\[
\frac{AC}{\sin45^{\circ}}=\frac{AB}{\sin60^{\circ}}
\]
所以
\[
AC=\sqrt6\frac{\sin45^{\circ}}{\sin60^{\circ}}
=\sqrt6\frac{\sqrt2/2}{\sqrt3/2}=2
\]
\end{solution}
\end{problem}

\begin{problem}
已知数列 \( \{a_{n}\} \) 中，\(a_{1}=1\)，\(a_{n}=a_{n-1}+\frac{1}{2}\left(n\geqslant2\right)\)，则数列 \( \{a_{n}\} \) 的前 \(9\) 项和等于\fillinblank{}.

\begin{answer}
\(27\)
\end{answer}

\begin{solution}
由递推式可知 \(a_n-a_{n-1}=\frac12\)，所以 \(\{a_n\}\) 是首项为 \(1\)、公差为 \(\frac12\) 的等差数列. 因而
\[
a_n=1+(n-1)\frac12=\frac{n+1}{2}
\]
特别地，\(a_9=5\). 于是前 \(9\) 项和为
\[
S_9=\frac{9(a_1+a_9)}{2}=\frac{9(1+5)}{2}=27
\]
\end{solution}
\end{problem}

\begin{problem}
在平面直角坐标系 \(xOy\) 中，若直线 \(y = 2a\) 与函数 \(y = |x - a| - 1\) 的图象只有一个交点，则 \(a\) 的值为\fillinblank{}.

\begin{answer}
\(-\frac{1}{2}\)
\end{answer}

\begin{solution}
联立直线与函数图象的方程，得
\[
2a=|x-a|-1
\]
即 \(|x-a|=2a+1\). 若 \(2a+1>0\)，则 \(x-a=\pm(2a+1)\)，有两个不同的交点；若 \(2a+1<0\)，则绝对值方程无解；只有 \(2a+1=0\) 时，方程有唯一解 \(x=a\). 因此
\[
a=-\frac12
\]
\end{solution}
\end{problem}

\begin{problem}
\(\triangle ABC\) 是边长为 \(2\) 的等边三角形，已知向量 \(\bs{a}\)，\(\bs{b}\) 满足 \(\overrightarrow{AB} = 2\bs{a}\)，\(\overrightarrow{AC} = 2\bs{a} + \bs{b}\). 则下列结论中正确的是\fillinblank{}. （写出所有正确结论的序号）

\begin{circlelist}
\item \(\bs{a}\) 为单位向量；
\item \(\bs{b}\) 为单位向量；
\item \(\bs{a}\perp\bs{b}\)；
\item \(\bs{b}\parallel\overrightarrow{BC}\)；
\item \(\left(4\bs{a}+\bs{b}\right)\perp\overrightarrow{BC}\).
\end{circlelist}

\begin{answer}
\(\circled{1}\circled{4}\circled{5}\)
\end{answer}

\begin{solution}
因为 \(ABC\) 是边长为 \(2\) 的等边三角形，所以 \(|\overrightarrow{AB}|=|\overrightarrow{AC}|=|\overrightarrow{BC}|=2\). 由 \(\overrightarrow{AB}=2\bs a\)，得 \(|\bs a|=1\)，结论\(\circled{1}\)正确.

又
\[
\overrightarrow{BC}=\overrightarrow{AC}-\overrightarrow{AB}=\bs b
\]
所以 \(|\bs b|=2\)，结论\(\circled{2}\)错误，且 \(\bs b\parallel\overrightarrow{BC}\)，结论\(\circled{4}\)正确.

由 \(|2\bs a+\bs b|=|\overrightarrow{AC}|=2\)，平方得
\[
4|\bs a|^2+4\bs a\cdot\bs b+|\bs b|^2=4
\]
代入 \(|\bs a|=1\)、\(|\bs b|=2\)，得到 \(\bs a\cdot\bs b=-1\ne0\)，所以结论\(\circled{3}\)错误. 最后
\[
\left(4\bs a+\bs b\right)\cdot\overrightarrow{BC}
=\left(4\bs a+\bs b\right)\cdot\bs b
=4(-1)+2^2=0
\]
故结论\(\circled{5}\)正确. 所有正确结论的序号为 \(\circled{1}\)、\(\circled{4}\)、\(\circled{5}\).
\end{solution}
\end{problem}

\section{解答题}

\begin{problem}
已知函数 \( f(x) = (\sin x + \cos x)^2 + \cos 2x \).
\begin{enumerate}
\item 求 \( f(x) \) 最小正周期；
\item 求 \( f(x) \) 在区间 \( \left[0, \frac{\pi}{2}\right] \) 上的最大值和最小值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 利用二倍角公式，
\[
(\sin x+\cos x)^2=\sin^2x+2\sin x\cos x+\cos^2x=1+\sin2x
\]
因此
\[
f(x)=1+\sin2x+\cos2x=1+\sqrt2\sin\left(2x+\frac{\pi}{4}\right)
\]
正弦函数的最小正周期为 \(2\pi\)，故上式中因变量的系数为 \(2\)，从而 \(f(x)\) 的最小正周期为 \(T=\frac{2\pi}{2}=\pi\).
\item 当 \(x\in\left[0,\frac\pi2\right]\) 时，令 \(t=2x+\frac\pi4\)，则 \(t\in\left[\frac\pi4,\frac{5\pi}4\right]\). 在此区间内，正弦函数的最大值为 \(1\)，在 \(t=\frac\pi2\)，即 \(x=\frac\pi8\) 时取得；最小值为 \(-\frac{\sqrt2}{2}\)，在 \(t=\frac{5\pi}4\)，即 \(x=\frac\pi2\) 时取得. 所以 \(f_{\max}=1+\sqrt2\)，且
\[
f_{\min}=1+\sqrt2\left(-\frac{\sqrt2}{2}\right)=0
\]
\end{enumerate}
\end{solution}

\begin{problem}
某企业为了了解下属某部门对本企业职工的服务情况，随机访问 \(50\text{ 名}\) 职工，根据这 \(50\text{ 名}\) 职工对该部门的评分，绘制频率分布直方图（如图所示），其中样本数据分组区间为：\(\left[40,50\right)\)，\(\left[50,60\right)\)，…，\(\left[80,90\right)\)，\(\left[90,100\right]\).

\begin{enumerate}
\item 求频率分布直方图中 \(a\) 的值；
\item 估计该企业的职工对该部门评分不低于 \(80\) 的概率；
\item 从评分在 \(\left[40,60\right)\) 的受访职工中，随机抽取 \(2\text{ 人}\)，求此 \(2\) 人评分都在 \(\left[40,50\right)\) 的概率.
\end{enumerate}

\texfigure[float=inline,max width=0.42\linewidth]{img_repaint/2015/anhui_liberal/q17_fig1.tex}
\end{problem}

\FloatBarrier

\begin{solution}
\begin{enumerate}
\item 每个小矩形的面积表示该组频率，所有组频率之和为 \(1\). 组距为 \(10\)，故
\[
10\left(0.004+a+0.022+0.028+0.022+0.018\right)=1
\]
解得 \(a=0.006\).
\item 评分不低于 \(80\) 的频率为
\[
10\times0.022+10\times0.018=0.4
\]
因此估计该企业职工评分不低于 \(80\) 的概率为 \(0.4\).
\item 评分在 \(\left[40,50\right)\) 内的职工人数为 \(50\times10\times0.004=2\)，评分在 \(\left[50,60\right)\) 内的职工人数为 \(50\times10\times0.006=3\). 因此评分在 \(\left[40,60\right)\) 内共有 \(5\) 人. 从这 \(5\) 人中抽取 \(2\) 人，所有等可能的结果数为 \(\mathrm C_5^2\)，其中两人都来自 \(\left[40,50\right)\) 的结果数为 \(\mathrm C_2^2\). 所求概率为
\[
P=\frac{\mathrm C_2^2}{\mathrm C_5^2}=\frac{1}{10}
\]
\end{enumerate}
\end{solution}

\begin{problem}
已知数列 \(\{a_{n}\}\) 是递增的等比数列，且 \(a_{1} + a_{4} = 9\)，\(a_{2}a_{3} = 8\).
\begin{enumerate}
\item 求数列 \( \{a_{n}\} \) 的通项公式；
\item 设 \(S_{n}\) 为数列 \(\{a_{n}\}\) 的前 \(n\) 项和，\(b_{n} = \frac{a_{n+1}}{S_{n} S_{n+1}}\)，求数列 \(\{b_{n}\}\) 的前 \(n\) 项和 \(T_{n}\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设等比数列的公比为 \(q\). 由等比数列性质，
\[
a_2a_3=a_1q\cdot a_1q^2=a_1^2q^3=a_1a_4=8
\]
因此 \(a_1\) 和 \(a_4\) 是方程
\[
t^2-(a_1+a_4)t+a_1a_4=t^2-9t+8=0
\]
的两个根，解得 \(\{a_1,a_4\}=\{1,8\}\). 数列递增，所以 \(a_1<a_4\)，从而 \(a_1=1\)、\(a_4=8\). 于是 \(q^3=\frac{a_4}{a_1}=8\)，得到 \(q=2\)，故
\[
a_n=a_1q^{n-1}=2^{n-1}
\]
\item 由通项公式，
\[
S_n=1+2+\cdots+2^{n-1}=2^n-1
\]
又因为 \(a_{n+1}=S_{n+1}-S_n\)，所以
\[
b_n=\frac{S_{n+1}-S_n}{S_nS_{n+1}}=\frac1{S_n}-\frac1{S_{n+1}}
\]
裂项相消得
\[
T_n=\sum_{k=1}^n b_k
=\left(\frac1{S_1}-\frac1{S_2}\right)+\cdots+\left(\frac1{S_n}-\frac1{S_{n+1}}\right)
=1-\frac1{S_{n+1}}
=1-\frac1{2^{n+1}-1}
\]
\end{enumerate}
\end{solution}

\begin{problem}
如图，三棱锥 \(P-ABC\) 中，\(PA \perp\) 平面 \(ABC\)，\(PA = 1\)，\(AB = 1\)，\(AC = 2\)，\(\angle BAC = 60^\circ\).
\begin{enumerate}
\item 求三棱锥 \(P-ABC\) 的体积；
\item 证明：在线段 \(PC\) 上存在点 \(M\)，使得 \(AC \perp BM\)，并求 \(\frac{PM}{MC}\) 的值.
\end{enumerate}

\bitmapfigure[float=inline,max width=0.42\linewidth]{img/2015/anhui_liberal/q19_fig1.png}
\end{problem}

\FloatBarrier

\begin{solution}
\begin{enumerate}
\item 底面三角形 \(ABC\) 的面积为
\[
S_{\triangle ABC}=\frac12\times AB\times AC\times\sin\angle BAC
=\frac12\times1\times2\times\sin60^\circ=\frac{\sqrt3}{2}
\]
由 \(PA\perp\) 平面 \(ABC\)，知 \(PA=1\) 是三棱锥的高，所以
\[
V=\frac13S_{\triangle ABC}\cdot PA=\frac13\times\frac{\sqrt3}{2}\times1=\frac{\sqrt3}{6}
\]
\item 建立空间直角坐标系，取 \(A\) 为原点，\(AC\) 所在直线为 \(x\) 轴，平面 \(ABC\) 为 \(xOy\) 平面，且 \(P\) 在 \(z\) 轴正半轴上. 则
\[
C=(2,0,0)
\]
并且 \(P=(0,0,1)\).
由 \(AB=1\)、\(\angle BAC=60^\circ\)，可取
\[
B=\left(\frac12,\frac{\sqrt3}{2},0\right)
\]
设 \(M\) 在线段 \(PC\) 上，令 \(t=\frac{PM}{PC}\)，则 \(0\leq t\leq1\)，且
\[
M=P+t(C-P)=(2t,0,1-t)
\]
于是 \(\overrightarrow{AC}=(2,0,0)\)，
\(\overrightarrow{BM}=\left(2t-\frac12,-\frac{\sqrt3}{2},1-t\right)\).
因此
\[
\overrightarrow{AC}\cdot\overrightarrow{BM}=2\left(2t-\frac12\right)=4t-1
\]
取 \(t=\frac14\)，则 \(M\) 在线段 \(PC\) 上且 \(\overrightarrow{AC}\cdot\overrightarrow{BM}=0\)，所以 \(AC\perp BM\). 此时
\[
\frac{PM}{MC}=\frac{t}{1-t}=\frac{1/4}{3/4}=\frac13
\]
\end{enumerate}
\end{solution}

\begin{problem}
设椭圆 \(E\) 的方程为 \(\frac{x^2}{a^2} + \frac{y^2}{b^2} = 1\)（\(a > b > 0\)），点 \(O\) 为坐标原点，点 \(A\) 的坐标为 \((a, 0)\)，点 \(B\) 的坐标为 \((0, b)\)，点 \(M\) 在线段 \(AB\) 上，满足 \(|BM| = 2|MA|\)，直线 \(OM\) 的斜率为 \(\frac{\sqrt{5}}{10}\).
\begin{enumerate}
\item 求 \(E\) 的离心率 \(e\)；
\item 设点 \(C\) 的坐标为 \((0, -b)\)，\(N\) 为线段 \(AC\) 的中点，证明：\(MN \perp AB\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 因为 \(|BM|=2|MA|\)，所以 \(AM:MB=1:2\). 由内分点公式，
\[
M=A+\frac13(B-A)=\left(\frac{2a}{3},\frac{b}{3}\right)
\]
故直线 \(OM\) 的斜率为 \(\frac{b/3}{2a/3}=\frac{b}{2a}\). 由题意
\(\frac{b}{2a}=\frac{\sqrt5}{10}\)，\(\frac ba=\frac{\sqrt5}{5}\)，\(\frac{b^2}{a^2}=\frac15\).
椭圆的半焦距满足 \(c^2=a^2-b^2\)，所以
\[
e=\frac ca=\sqrt{1-\frac{b^2}{a^2}}=\sqrt{1-\frac15}=\frac{2\sqrt5}{5}
\]
\item 由 \(C=(0,-b)\) 且 \(N\) 为 \(AC\) 的中点，得
\[
N=\left(\frac a2,-\frac b2\right)
\]
因此 \(\overrightarrow{MN}=N-M=\left(-\frac a6,-\frac{5b}{6}\right)\).
又 \(\overrightarrow{AB}=B-A=(-a,b)\). 因而
\[
\overrightarrow{MN}\cdot\overrightarrow{AB}
=\frac{a^2}{6}-\frac{5b^2}{6}
=\frac{a^2-5b^2}{6}=0
\]
其中最后一步使用了 \(b^2=\frac{a^2}{5}\). 所以 \(MN\perp AB\).
\end{enumerate}
\end{solution}

\begin{problem}
已知函数 \(f(x) = \frac{ax}{(x + r)^2}\)（\(a > 0\)，\(r > 0\)）.
\begin{enumerate}
\item 求 \( f(x) \) 的定义域，并讨论 \( f(x) \) 的单调性；
\item 若 \(\frac{a}{r} = 400\)，求 \(f(x)\) 在 \((0, +\infty)\) 内的极值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 分母不能为零，所以定义域为 \(\R\setminus\{-r\}=(-\infty,-r)\cup(-r,+\infty)\). 求导得
\[
f'(x)=\frac{a(x+r)^2-2ax(x+r)}{(x+r)^4}=\frac{a(r-x)}{(x+r)^3}
\]
因为 \(a>0\)，分类讨论符号：当 \(x\in(-\infty,-r)\) 时，\(r-x>0\)、\((x+r)^3<0\)，故 \(f'(x)<0\)；当 \(x\in(-r,r)\) 时，二者均为正，故 \(f'(x)>0\)；当 \(x\in(r,+\infty)\) 时，\(r-x<0\)、\((x+r)^3>0\)，故 \(f'(x)<0\). 因此 \(f(x)\) 在 \((-\infty,-r)\) 和 \((r,+\infty)\) 上单调递减，在 \((-r,r)\) 上单调递增.
\item 在 \((0,+\infty)\) 内，函数在 \((0,r)\) 上递增，在 \((r,+\infty)\) 上递减，所以在 \(x=r\) 处取得极大值. 该极大值为
\[
f(r)=\frac{ar}{(2r)^2}=\frac{a}{4r}=\frac14\frac ar=100
\]
又因为 \(x\to0^+\) 或 \(x\to+\infty\) 时 \(f(x)\to0\)，但 \(0\notin(0,+\infty)\)，所以在该开区间内没有最小值.
\end{enumerate}
\end{solution}
