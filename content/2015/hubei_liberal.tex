\section{单选题}

\begin{problem}
\(\i\) 为虚数单位，\(\i^{607}=\)（\quad）

\choices
    {\(\i\)}
    {\(-\i\)}
    {\(1\)}
    {\(-1\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
因为 \(607=4\times151+3\)，所以 \(\i^{607}=\i^3=-\i\)，故选 B.
\end{solution}

\begin{problem}
我国古代数学名著《数书九章》有“米谷粒分”题：粮仓开仓收粮，有人送来米 \(1534\text{ 石}\)，验得米内夹谷，抽样取米一把，数得 \(254\text{ 粒}\) 内夹谷 \(28\text{ 粒}\)，则这批米内夹谷约为（\quad）

\choices
    {\(134\text{ 石}\)}
    {\(169\text{ 石}\)}
    {\(338\text{ 石}\)}
    {\(1365\text{ 石}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
样本中夹谷米所占比例为 \(\dfrac{28}{254}\)，因此这批米内夹谷约为 \(1534\times\dfrac{28}{254}=\dfrac{21476}{127}\approx169\text{ 石}\)，故选 B.
\end{solution}

\begin{problem}
命题“\(\exists x_0\in (0,+\infty)\)，\(\ln x_0=x_0-1\)”的否定是（\quad）

\choices
    {\(\exists x_0\in (0,+\infty)\)，\(\ln x_0\neq x_0-1\)}
    {\(\exists x_0\notin (0,+\infty)\)，\(\ln x_0=x_0-1\)}
    {\(\forall x\in (0,+\infty)\)，\(\ln x\neq x-1\)}
    {\(\forall x\notin (0,+\infty)\)，\(\ln x=x-1\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
否定特称命题时，将“存在”改为“任意”，并将等式改为不等式，同时保留原定义域. 因此原命题的否定为“\(\forall x\in(0,+\infty)\)，\(\ln x\neq x-1\)”，故选 C.
\end{solution}

\begin{problem}
已知变量 \(x\) 和 \(y\) 满足关系 \(y=-0.1x+1\)，变量 \(y\) 与 \(z\) 正相关. 下列结论中正确的是（\quad）

\choices
    {\(x\) 与 \(y\) 正相关，\(x\) 与 \(z\) 负相关}
    {\(x\) 与 \(y\) 正相关，\(x\) 与 \(z\) 正相关}
    {\(x\) 与 \(y\) 负相关，\(x\) 与 \(z\) 负相关}
    {\(x\) 与 \(y\) 负相关，\(x\) 与 \(z\) 正相关}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
由 \(y=-0.1x+1\) 可知，\(x\) 增大时 \(y\) 减小，所以 \(x\) 与 \(y\) 负相关. 又因 \(y\) 与 \(z\) 正相关，\(y\) 减小时 \(z\) 也有减小趋势，所以 \(x\) 与 \(z\) 负相关，故选 C.
\end{solution}

\begin{problem}
\(l_1\)，\(l_2\) 表示空间中的两条直线，若 \(p\)：\(l_1\)，\(l_2\) 是异面直线；\(q\)：\(l_1\)，\(l_2\) 不相交，则（\quad）

\choices
    {\(p\) 是 \(q\) 的充分条件，但不是 \(q\) 的必要条件}
    {\(p\) 是 \(q\) 的必要条件，但不是 \(q\) 的充分条件}
    {\(p\) 是 \(q\) 的充分必要条件}
    {\(p\) 既不是 \(q\) 的充分条件，也不是 \(q\) 的必要条件}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
异面直线一定不相交，所以由 \(p\) 可推出 \(q\)，即 \(p\) 是 \(q\) 的充分条件. 两条不相交的空间直线也可能是平行直线，平行直线不是异面直线，所以由 \(q\) 不能推出 \(p\)，即 \(p\) 不是 \(q\) 的必要条件，故选 A.
\end{solution}

\begin{problem}
函数 \(f(x)=\sqrt{4-|x|}+\lg\dfrac{x^2-5x+6}{x-3}\) 的定义域为（\quad）

\choices
    {\((2,3)\)}
    {\((2,4]\)}
    {\((2,3)\cup(3,4]\)}
    {\((-1,3)\cup(3,6]\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
根式有意义要求 \(4-|x|\geqslant0\)，即 \(-4\leqslant x\leqslant4\). 对数真数须为正且分母不为零，且
\(\dfrac{x^2-5x+6}{x-3}=\dfrac{(x-2)(x-3)}{x-3}=x-2\)，其中 \(x\neq3\). 因此还需 \(x>2\) 且 \(x\neq3\)，与前一条件交集为 \((2,3)\cup(3,4]\)，故选 C.
\end{solution}

\begin{problem}
已知符号函数 \(\operatorname{sgn}x=\begin{cases}
1, & x>0,\\
0, & x=0,\\
-1, & x<0,
\end{cases}\) 则（\quad）

\choices
    {\(|x|=x|\operatorname{sgn}x|\)}
    {\(|x|=x\operatorname{sgn}|x|\)}
    {\(|x|=|x|\operatorname{sgn}x\)}
    {\(|x|=x\operatorname{sgn}x\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
当 \(x>0\) 时，\(x\operatorname{sgn}x=x=|x|\)；当 \(x=0\) 时，两边均为 \(0\)；当 \(x<0\) 时，\(x\operatorname{sgn}x=x(-1)=-x=|x|\). 所以对任意实数 \(x\)，都有 \(|x|=x\operatorname{sgn}x\)，故选 D.
\end{solution}

\begin{problem}
在区间 \([0,1]\) 上随机取两个数 \(x\)，\(y\)，记 \(p_1\) 为事件“\(x+y\leqslant \frac{1}{2}\)”的概率，\(p_2\) 为事件“\(xy\leqslant \frac{1}{2}\)”的概率，则（\quad）

\choices
    {\(p_1<p_2<\frac{1}{2}\)}
    {\(p_2<\frac{1}{2}<p_1\)}
    {\(\frac{1}{2}<p_2<p_1\)}
    {\(p_1<\frac{1}{2}<p_2\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
把 \((x,y)\) 看作单位正方形 \([0,1]^2\) 中的随机点. 事件 \(x+y\leqslant\frac12\) 对应直角三角形，直角边均为 \(\frac12\)，故 \(p_1=\dfrac{\frac12\times\frac12}{2}=\frac18<\frac12\). 事件 \(xy\leqslant\frac12\) 至少包含两个矩形的并集：\(0\leqslant x\leqslant\frac12\) 或 \(0\leqslant y\leqslant\frac12\)，其面积为 \(\frac12+\frac12-\frac14=\frac34>\frac12\)，所以 \(p_2>\frac12\). 因此 \(p_1<\frac12<p_2\)，故选 D.
\end{solution}

\begin{problem}
将离心率为 \(e_1\) 的双曲线 \(C_1\) 的实半轴长 \(a\) 和虚半轴长 \(b\)（\(a\neq b\)）同时增加 \(m\)（\(m>0\)）个单位长度，得到离心率为 \(e_2\) 的双曲线 \(C_2\)，则（\quad）

\choices
    {对任意的 \(a\)，\(b\)，\(e_1>e_2\)}
    {当 \(a>b\) 时，\(e_1>e_2\)；当 \(a<b\) 时，\(e_1<e_2\)}
    {对任意的 \(a\)，\(b\)，\(e_1<e_2\)}
    {当 \(a>b\) 时，\(e_1<e_2\)；当 \(a<b\) 时，\(e_1>e_2\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
双曲线的离心率满足 \(e^2=\dfrac{a^2+b^2}{a^2}=1+\dfrac{b^2}{a^2}\). 增加后
\(e_2^2=1+\dfrac{(b+m)^2}{(a+m)^2}\). 比较两个分式时，因 \(a\)，\(b\)，\(m>0\)，有
\[
\frac{(b+m)^2}{(a+m)^2}-\frac{b^2}{a^2}
=\frac{\bigl(a(b+m)-b(a+m)\bigr)\bigl(a(b+m)+b(a+m)\bigr)}{a^2(a+m)^2}
\]
其中除 \(a(b+m)-b(a+m)=m(a-b)\) 外的因子均为正，所以该差的符号与 \(a-b\) 相同. 因此当 \(a>b\) 时，\(e_2>e_1\)；当 \(a<b\) 时，\(e_2<e_1\)，故选 D.
\end{solution}

\begin{problem}
已知集合 \(A=\{(x,y)\mid x^2+y^2\leqslant 1,\ x,y\in\Z\}\)，\(B=\{(x,y)\mid |x|\leqslant 2,\ |y|\leqslant 2,\ x,y\in\Z\}\)，定义集合 \(A\oplus B=\{(x_1+x_2,y_1+y_2)\mid (x_1,y_1)\in A,\ (x_2,y_2)\in B\}\)，则 \(A\oplus B\) 中元素的个数为（\quad）

\choices
    {77}
    {49}
    {45}
    {30}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
集合 \(A\) 中的整点只有 \((0,0)\)，\((1,0)\)，\((-1,0)\)，\((0,1)\)，\((0,-1)\). 集合 \(B\) 是包含 \(25\) 个整点的方格. 与 \((0,0)\) 相加得到原方格 \(B\)；与 \((1,0)\)，\((-1,0)\) 相加分别在左右各增加一列 \(5\) 个点；与 \((0,1)\)，\((0,-1)\) 相加分别在上下各增加一行 \(5\) 个点. 这四列、行与原方格及彼此均不重合，因此
\(\lvert A\oplus B\rvert=25+4\times5=45\)，故选 C.
\end{solution}

\section{填空题}

\begin{problem}
已知向量 \(\overrightarrow{OA}\perp\overrightarrow{AB}\)，\(|\overrightarrow{OA}|=3\)，则 \(\overrightarrow{OA}\cdot\overrightarrow{OB}=\)\fillinblank{}.
\end{problem}

\begin{answer}
\(9\)
\end{answer}

\begin{solution}
由向量加法，\(\overrightarrow{OB}=\overrightarrow{OA}+\overrightarrow{AB}\). 因此 \(\overrightarrow{OA}\cdot\overrightarrow{OB}=\overrightarrow{OA}\cdot\overrightarrow{OA}+\overrightarrow{OA}\cdot\overrightarrow{AB}=|\overrightarrow{OA}|^2+0=3^2=9\).
\end{solution}

\begin{problem}
若变量 \(x\)，\(y\) 满足约束条件 \(\begin{cases}
x+y\leqslant 4,\\
x-y\leqslant 2,\\
3x-y\geqslant 0,
\end{cases}\) 则 \(3x+y\) 的最大值是\fillinblank{}.
\end{problem}

\begin{answer}
\(10\)
\end{answer}

\begin{solution}
将前两条约束相加，得 \(2x\leqslant6\)，所以 \(x\leqslant3\). 于是 \(3x+y=(x+y)+2x\leqslant4+2\times3=10\). 取 \((x,y)=(3,1)\) 时，三条约束分别为 \(4\leqslant4\)，\(2\leqslant2\)，\(8\geqslant0\)，且 \(3x+y=10\)，所以最大值为 \(10\).
\end{solution}

\begin{problem}
函数 \(f(x)=2\sin x\sin\left(x+\frac{\pi}{2}\right)-x^2\) 的零点个数为\fillinblank{}.
\end{problem}

\begin{answer}
\(2\)
\end{answer}

\begin{solution}
因为 \(\sin\left(x+\frac\pi2\right)=\cos x\)，所以 \(f(x)=2\sin x\cos x-x^2=\sin2x-x^2\). 若 \(f(x)=0\)，则 \(x^2=\sin2x\leqslant1\)，从而 \(-1\leqslant x\leqslant1\). 当 \(-1\leqslant x<0\) 时，\(2x\in[-2,0)\)，故 \(\sin2x<0\)，而 \(x^2>0\)，不可能成立；\(x=0\) 是一个零点.

在 \(0<x\leqslant1\) 上，令 \(h(x)=\sin2x-x^2\)，则 \(h(0)=0\)，\(h(1)=\sin2-1<0\)，且 \(h''(x)=-4\sin2x-2<0\). 又 \(h'(0)=2>0\)，\(h'(1)=2\cos2-2<0\)，所以 \(h'(x)\) 严格递减并且只在一个位置变号，\(h(x)\) 先增后减；结合 \(h(0)=0\) 和 \(h(1)<0\)，故在 \((0,1]\) 内恰有一个零点. 综上，零点共有 \(2\) 个.
\end{solution}

\begin{problem}
某电子商务公司对 \(10000\) 名网络购物者 \(2014\) 年度的消费情况进行统计，发现消费金额（单位：万元）都在区间 \([0.3,0.9]\) 内，其频率分布直方图如图所示.

\begin{enumerate}
\item 直方图中的 \(a=\)\fillinblank{}.
\item 在这些购物者中，消费金额在区间 \([0.5,0.9]\) 内的购物者的人数为\fillinblank{}.
\end{enumerate}

\bitmapfigure[max width=0.52\linewidth]{img/2015/hubei_liberal/q14_fig1.png}
\end{problem}

\begin{answer}
\(3\)，\(6000\)
\end{answer}

\begin{solution}
直方图的各组组距均为 \(0.1\)，频率等于“组距乘频率密度”，所有组频率之和为 \(1\). 因此 \(0.1(1.5+2.5+a+2.0+0.8+0.2)=1\)，解得 \(a=3\). 消费金额在 \([0.5,0.9]\) 内的频率为 \(0.1(3+2.0+0.8+0.2)=0.6\)，所以人数为 \(10000\times0.6=6000\).
\end{solution}

\begin{problem}
如图，一辆汽车在一条水平的公路上向正西行驶，到 \(A\) 处时测得公路北侧一山顶 \(D\) 在西偏北 \(30^\circ\) 的方向上，行驶 \(600\mathrm{~m}\) 后到达 \(B\) 处，测得此山顶在西偏北 \(75^\circ\) 的方向上，仰角为 \(30^\circ\)，则此山的高度 \(CD=\)\fillinblank{}\(\mathrm{~m}\).

\bitmapfigure[max width=0.46\linewidth]{img/2015/hubei_liberal/q15_fig1.png}
\end{problem}

\begin{answer}
\(100\sqrt{6}\)
\end{answer}

\begin{solution}
设山顶 \(D\) 在水平面上的投影为 \(C\). 由两个方向角，地面三角形 \(ABC\) 中有 \(\angle BAC=30^\circ\)，\(\angle ABC=180^\circ-75^\circ=105^\circ\)，所以 \(\angle ACB=45^\circ\). 在三角形 \(ABC\) 中应用正弦定理，得 \(\dfrac{BC}{\sin30^\circ}=\dfrac{AB}{\sin45^\circ}\)，即 \(BC=\dfrac{600\times\frac12}{\frac{\sqrt2}{2}}=300\sqrt2\).

在直角三角形 \(BCD\) 中，\(\angle DBC=30^\circ\)，所以 \(CD=BC\tan30^\circ=300\sqrt2\times\dfrac1{\sqrt3}=100\sqrt6\).
\end{solution}

\begin{problem}
如图，圆 \(C\) 与 \(x\) 轴相切于点 \(T(1,0)\)，与 \(y\) 轴正半轴交于两点 \(A\)，\(B\)（\(B\) 在 \(A\) 的上方），且 \(|AB|=2\).

\begin{enumerate}
\item 圆 \(C\) 的标准方程为\fillinblank{}.
\item 圆 \(C\) 在点 \(B\) 处的切线在 \(x\) 轴上的截距为\fillinblank{}.
\end{enumerate}

\bitmapfigure[max width=0.30\linewidth]{img/2015/hubei_liberal/q16_fig1.png}
\end{problem}

\begin{answer}
\((x-1)^2+(y-\sqrt{2})^2=2\)，\(-1-\sqrt{2}\)
\end{answer}

\begin{solution}
设圆心为 \(C=(1,r)\)，其中 \(r>0\)，则圆的方程为 \((x-1)^2+(y-r)^2=r^2\). 令 \(x=0\)，与 \(y\) 轴的两个交点的纵坐标满足 \((y-r)^2=r^2-1\)，所以弦长 \(AB=2\sqrt{r^2-1}=2\)，从而 \(r^2=2\)，即 \(r=\sqrt2\). 因此圆的标准方程为 \((x-1)^2+(y-\sqrt2)^2=2\).

此时 \(B=(0,\sqrt2+1)\)，圆心为 \(C=(1,\sqrt2)\)，所以半径 \(CB\) 的斜率为 \(-1\)，点 \(B\) 处切线的斜率为 \(1\). 切线方程为 \(y-(\sqrt2+1)=x\)，令 \(y=0\)，得其在 \(x\) 轴上的截距为 \(-1-\sqrt2\).
\end{solution}

\begin{problem}
\(a\) 为实数，函数 \(f(x)=|x^2-ax|\) 在区间 \([0,1]\) 上的最大值记为 \(g(a)\). 当 \(a=\)\fillinblank{} 时，\(g(a)\) 的值最小.
\end{problem}

\begin{answer}
\(2\sqrt{2}-2\)
\end{answer}

\begin{solution}
在 \([0,1]\) 上讨论 \(f(x)=|x(x-a)|\). 当 \(a\leqslant0\) 时，\(x-a\geqslant0\)，且 \(x^2-ax\) 在 \([0,1]\) 上递增，所以 \(g(a)=1-a\). 当 \(0<a<1\) 时，函数在 \(x=a\) 处变号：在 \([0,a]\) 上为 \(ax-x^2\)，其最大值为 \(a^2/4\)；在 \([a,1]\) 上为 \(x^2-ax\)，最大值为 \(1-a\). 当 \(1\leqslant a<2\) 时，整个区间内 \(x-a\leqslant0\)，函数为 \(ax-x^2\)，最大值为 \(a^2/4\)；当 \(a\geqslant2\) 时，该函数在 \([0,1]\) 上递增，最大值为 \(a-1\). 因此
\[
g(a)=
\begin{cases}
1-a, & a\leqslant0,\\
\max\left\{1-a,\dfrac{a^2}{4}\right\}, & 0<a<1,\\
\dfrac{a^2}{4}, & 1\leqslant a<2,\\
a-1, & a\geqslant2.
\end{cases}
\]
在 \(0<a<1\) 内令两部分相等，得 \(1-a=\dfrac{a^2}{4}\)，即 \(a^2+4a-4=0\)，所以正根为 \(a=2\sqrt2-2\). 在此前 \(g(a)=1-a\) 递减，在此后 \(g(a)=a^2/4\) 或 \(a-1\) 递增，故 \(g(a)\) 取得最小值时 \(a=2\sqrt2-2\).
\end{solution}

\begin{problem}
某同学用“五点法”画函数 \(f(x)=A\sin(\omega x+\varphi)\)（\(\omega>0\)，\(|\varphi|<\frac{\pi}{2}\)）在某一个周期内的图象时，列表并填入了部分数据，如下表：

\begin{center}
\renewcommand{\arraystretch}{1.25}
\begin{tabular}{|c|c|c|c|c|c|}
\hline
\(\omega x+\varphi\) & \(0\) & \(\frac{\pi}{2}\) & \(\pi\) & \(\frac{3\pi}{2}\) & \(2\pi\) \\
\hline
\(x\) &  & \(\frac{\pi}{3}\) &  & \(\frac{5\pi}{6}\) &  \\
\hline
\(A\sin(\omega x+\varphi)\) & \(0\) & \(5\) &  & \(-5\) & \(0\) \\
\hline
\end{tabular}
\end{center}

\begin{enumerate}
\item 请将上表数据补充完整，并直接写出函数 \(f(x)\) 的解析式；
\item 将 \(y=f(x)\) 图象上所有点向左平行移动 \(\frac{\pi}{6}\) 个单位长度，得到 \(y=g(x)\) 的图象. 求 \(y=g(x)\) 的图象离原点 \(O\) 最近的对称中心.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由表中两组已知数据，横坐标之差与相位之差满足
\(\omega\left(\dfrac{5\pi}{6}-\dfrac{\pi}{3}\right)=\dfrac{3\pi}{2}-\dfrac{\pi}{2}\)，即 \(\omega\cdot\dfrac{\pi}{2}=\pi\)，所以 \(\omega=2\). 又由 \(2\cdot\dfrac{\pi}{3}+\varphi=\dfrac{\pi}{2}\)，得 \(\varphi=-\dfrac{\pi}{6}\). 由 \(A\sin\dfrac{\pi}{2}=5\)，得 \(A=5\). 对于相位为 \(\theta\) 的表格列，横坐标为 \(x=\dfrac{\theta-\varphi}{\omega}\)，故表中 \(x\) 一行依次为 \(\dfrac{\pi}{12}\)，\(\dfrac{\pi}{3}\)，\(\dfrac{7\pi}{12}\)，\(\dfrac{5\pi}{6}\)，\(\dfrac{13\pi}{12}\)，函数解析式为
\[
f(x)=5\sin\left(2x-\dfrac{\pi}{6}\right)
\]

\item 图象向左平移 \(\dfrac{\pi}{6}\) 个单位后，横坐标变换为 \(x+\dfrac{\pi}{6}\)，所以
\[
g(x)=f\left(x+\dfrac{\pi}{6}\right)=5\sin\left(2x+\dfrac{\pi}{6}\right)
\]
正弦曲线的对称中心位于其与 \(x\) 轴的交点处，因此所有对称中心为
\(\left(\dfrac{k\pi}{2}-\dfrac{\pi}{12},0\right)\)，\(k\in\Z\).
当 \(k=0\) 时，该点到原点的距离为 \(\dfrac{\pi}{12}\)，小于其他整数 \(k\) 对应的距离，故最近的对称中心为 \(\left(-\dfrac{\pi}{12},0\right)\).
\end{enumerate}
\end{solution}

\begin{problem}
设等差数列 \(\{a_n\}\) 的公差为 \(d\)，前 \(n\) 项和为 \(S_n\)，等比数列 \(\{b_n\}\) 的公比为 \(q\). 已知 \(b_1=a_1\)，\(b_2=2\)，\(q=d\)，\(S_{10}=100\).

\begin{enumerate}
\item 求数列 \(\{a_n\}\)，\(\{b_n\}\) 的通项公式；
\item 当 \(d>1\) 时，记 \(c_n=\dfrac{a_n}{b_n}\)，求数列 \(\{c_n\}\) 的前 \(n\) 项和 \(T_n\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由 \(b_2=b_1q\)、\(b_1=a_1\) 和 \(q=d\)，得
\(a_1d=2\)，所以 \(d\neq0\) 且 \(a_1=\dfrac{2}{d}\). 又
\[
100=S_{10}=\frac{10}{2}\left(2a_1+9d\right)
\]
从而 \(2a_1+9d=20\). 代入 \(a_1=\dfrac{2}{d}\)，得
\(\frac{4}{d}+9d=20\)，\(9d^2-20d+4=0\)
所以 \(d=2\) 或 \(d=\dfrac{2}{9}\).

当 \(d=2\) 时，\(a_1=1\)，故
\(a_n=a_1+(n-1)d=2n-1\)，\(b_n=b_1q^{n-1}=2^{n-1}\).
当 \(d=\dfrac{2}{9}\) 时，\(a_1=9\)，故
\(a_n=9+(n-1)\frac{2}{9}=\frac{2n+79}{9}\)，\(b_n=9\left(\frac{2}{9}\right)^{n-1}\).

\item 当 \(d>1\) 时只能取 \(d=2\)，于是
\[
c_n=\frac{a_n}{b_n}=\frac{2n-1}{2^{n-1}}
\]
设 \(T_n=6-\dfrac{2n+3}{2^{n-1}}\). 当 \(n=1\) 时，\(T_1=1=c_1\). 且
\[
T_{n+1}-T_n
=\frac{2n+3}{2^{n-1}}-\frac{2n+5}{2^n}
=\frac{2n+1}{2^n}=c_{n+1}
\]
因此由数学归纳法，数列 \(\{c_n\}\) 的前 \(n\) 项和为
\[
T_n=6-\frac{2n+3}{2^{n-1}}
\]
\end{enumerate}
\end{solution}

\begin{problem}
《九章算术》中，将底面为长方形且有一条侧棱与底面垂直的四棱锥称之为阳马，将四个面都为直角三角形的四面体称之为鳖臑.

如图，在阳马 \(P-ABCD\) 中，侧棱 \(PD\perp\) 底面 \(ABCD\)，且 \(PD=CD\)，点 \(E\) 是 \(PC\) 的中点，连接 \(DE\)，\(BD\)，\(BE\).

\begin{enumerate}
\item 证明：\(DE\perp\) 平面 \(PBC\). 试判断四面体 \(EBCD\) 是否为鳖臑，若是，写出其每个面的直角（只需写出结论）；若不是，请说明理由；
\item 记阳马 \(P-ABCD\) 的体积为 \(V_1\)，四面体 \(EBCD\) 的体积为 \(V_2\)，求 \(\dfrac{V_1}{V_2}\) 的值.
\end{enumerate}

\problemresetlayout

\bitmapfigure[max width=0.40\linewidth]{img/2015/hubei_liberal/q20_fig1.png}
\FloatBarrier
\end{problem}

\begin{solution}
建立如图所示的空间直角坐标系，令
\(D=(0,0,0)\)，\(C=(c,0,0)\)，\(A=(0,b,0)\)，\(B=(c,b,0)\)，其中 \(c=CD>0\)，\(b=AD>0\). 由 \(PD\perp\) 平面 \(ABCD\) 且 \(PD=CD=c\)，可取 \(P=(0,0,c)\). 因为 \(E\) 是 \(PC\) 的中点，所以 \(E=\left(\dfrac{c}{2},0,\dfrac{c}{2}\right)\). 从而
\(\overrightarrow{DE}=\left(\frac{c}{2},0,\frac{c}{2}\right)\)，\(\overrightarrow{BC}=(0,-b,0)\)，\(\overrightarrow{PB}=(c,b,-c)\).
于是 \(\overrightarrow{DE}\cdot\overrightarrow{BC}=0\)，\(\overrightarrow{DE}\cdot\overrightarrow{PB}=0\)，即 \(DE\perp BC\)，\(DE\perp PB\). 直线 \(BC\) 和 \(PB\) 是平面 \(PBC\) 内相交的两条直线，故 \(DE\perp\) 平面 \(PBC\).

由于 \(ABCD\) 是长方形，\(BC\perp CD\)，所以 \(\angle BCD=90^\circ\). 又 \(BC\perp CD\)，且 \(BC\perp PD\)，从而 \(BC\perp\) 平面 \(PCD\)，故 \(BC\perp CE\)，即 \(\angle BCE=90^\circ\). 由 \(DE\perp\) 平面 \(PBC\)，且 \(EC\)、\(EB\) 都在平面 \(PBC\) 内，得 \(DE\perp EC\)、\(DE\perp EB\)，即 \(\angle DEC=90^\circ\)、\(\angle DEB=90^\circ\). 因此四面体 \(EBCD\) 是鳖臑，四个面的直角分别为 \(\angle BCD\)，\(\angle BCE\)，\(\angle DEC\)，\(\angle DEB\).

设长方形 \(ABCD\) 的面积为 \(S=bc\). 则
\[
V_1=\frac{1}{3}S\cdot PD=\frac{1}{3}bc^2
\]
以三角形 \(BCD\) 为四面体 \(EBCD\) 的底面，则 \(\triangle BCD\) 的面积为 \(\dfrac{1}{2}bc\). 点 \(E\) 到平面 \(BCD\) 的距离为 \(\dfrac{1}{2}PD=\dfrac{c}{2}\)，所以
\[
V_2=\frac{1}{3}\cdot\frac{bc}{2}\cdot\frac{c}{2}=\frac{1}{12}bc^2
\]
故
\[
\frac{V_1}{V_2}=\frac{bc^2/3}{bc^2/12}=4
\]
\end{solution}

\begin{problem}
设函数 \(f(x)\)，\(g(x)\) 的定义域均为 \(\R\)，且 \(f(x)\) 是奇函数，\(g(x)\) 是偶函数，\(f(x)+g(x)=\e^x\)，其中 \(\e\) 为自然对数的底数.

\begin{enumerate}
\item 求 \(f(x)\)，\(g(x)\) 的解析式，并证明：当 \(x>0\) 时，\(f(x)>0\)，\(g(x)>1\)；
\item 设 \(a\leqslant 0\)，\(b\geqslant 1\)，证明：当 \(x>0\) 时，\(ag(x)+(1-a)<\dfrac{f(x)}{x}<bg(x)+(1-b)\).
\end{enumerate}
\end{problem}

\begin{solution}
由 \(f(x)\) 为奇函数、\(g(x)\) 为偶函数，分别在原式中令 \(x\) 换为 \(-x\)，得
\(f(x)+g(x)=\e^x\)，\(-f(x)+g(x)=\e^{-x}\).
两式相加、相减，得到
\(f(x)=\frac{\e^x-\e^{-x}}{2}\)，\(g(x)=\frac{\e^x+\e^{-x}}{2}\).
当 \(x>0\) 时，\(\e^x>\e^{-x}\)，所以 \(f(x)>0\). 又由均值不等式
\(\e^x+\e^{-x}>2\sqrt{\e^x\e^{-x}}=2\)，得 \(g(x)>1\).

令 \(x>0\). 由上述解析式可得 \(f'(x)=g(x)\)，且 \(f(0)=0\). 因而
\[
f(x)=\int_0^x g(t)\,\mathrm{d}t>x
\]
从而 \(\dfrac{f(x)}{x}>1\). 再令 \(h(x)=xg(x)-f(x)\)，则 \(h(0)=0\)，并且利用 \(g'(x)=f(x)\)、\(f'(x)=g(x)\)，有
\[
h'(x)=g(x)+xg'(x)-f'(x)=xf(x)>0
\]
所以 \(h(x)>0\)，即 \(\dfrac{f(x)}{x}<g(x)\). 综上，
\[
1<\frac{f(x)}{x}<g(x)
\]

因为 \(a\leqslant0\) 且 \(g(x)-1>0\)，所以
\[
ag(x)+(1-a)=1+a\bigl(g(x)-1\bigr)\leqslant1<\frac{f(x)}{x}
\]
因为 \(b\geqslant1\)，所以
\[
bg(x)+(1-b)=1+b\bigl(g(x)-1\bigr)\geqslant g(x)>\frac{f(x)}{x}
\]
故当 \(x>0\) 时，原不等式成立.
\end{solution}

\begin{problem}
一种作图工具如左图所示. \(O\) 是滑槽 \(AB\) 的中点，短杆 \(ON\) 可绕 \(O\) 转动，长杆 \(MN\) 通过 \(N\) 处铰链与 \(ON\) 连接，\(MN\) 上的栓子 \(D\) 可沿滑槽 \(AB\) 滑动，且 \(DN=ON=1\)，\(MN=3\). 当栓子 \(D\) 在滑槽 \(AB\) 内作往复运动时，带动 \(N\) 绕 \(O\) 转动一周（\(D\) 不动时，\(N\) 也不动），\(M\) 处的笔尖画出的曲线记为 \(C\). 以 \(O\) 为原点，\(AB\) 所在的直线为 \(x\) 轴建立如右图所示的平面直角坐标系.

\begin{enumerate}
\item 求曲线 \(C\) 的方程；
\item 设动直线 \(l\) 与两定直线 \(l_1\colon x-2y=0\) 和 \(l_2\colon x+2y=0\) 分别交于 \(P\)，\(Q\) 两点. 若直线 \(l\) 总与曲线 \(C\) 有且只有一个公共点，试探究：\(\triangle OPQ\) 的面积是否存在最小值？若存在，求出该最小值；若不存在，说明理由.
\end{enumerate}

\FigureLayoutDeclare{img/2015/hubei_liberal/q22_fig1.png}{14.0cm}{109bp 217bp 1bp 3bp}
\bitmapfigure[max width=0.68\linewidth]{img/2015/hubei_liberal/q22_fig1.png}
\FloatBarrier
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设 \(N=(u,v)\)，\(D=(t,0)\). 由 \(ON=DN=1\)，得
\(u^2+v^2=1\)，\((t-u)^2+v^2=1\).
两式相减，得到 \(t^2-2tu=0\). 在机构运动的支路上，除去静止的退化位置 \(D=O\) 外有 \(t=2u\)，由连续性在退化位置也成立. 又因为 \(D\) 在线段 \(MN\) 上，且 \(DN=1\)、\(MN=3\)，所以
\[
\overrightarrow{NM}=3\overrightarrow{ND}
\]
若 \(M=(X,Y)\)，则
\[
M=N+3(D-N)=(3t-2u,-2v)=(4u,-2v)
\]
于是 \(X=4u\)，\(Y=-2v\)，代入 \(u^2+v^2=1\)，得曲线 \(C\) 的方程
\[
\frac{X^2}{16}+\frac{Y^2}{4}=1
\]
将曲线坐标仍记为 \((x,y)\)，即
\(C:\)，\(\frac{x^2}{16}+\frac{y^2}{4}=1\).

\item 直线 \(l\) 与椭圆只有一个公共点，所以 \(l\) 是椭圆的切线. 设
\(P=(2p,p)\in l_1\)，\(Q=(-2q,q)\in l_2\).
过 \(P\)、\(Q\) 的直线方程为
\[
(p-q)x-2(p+q)y+4pq=0
\]
一般直线 \(\alpha x+\beta y+\gamma=0\) 与椭圆相切的条件为
\(\gamma^2=16\alpha^2+4\beta^2\). 事实上，将椭圆参数化为 \(x=4\cos\theta\)、\(y=2\sin\theta\)，直线与椭圆的交点满足
\(4\alpha\cos\theta+2\beta\sin\theta+\gamma=0\)，恰有一个解的切线情形等价于 \(\lvert\gamma\rvert=\sqrt{16\alpha^2+4\beta^2}\).
因此
\[
16p^2q^2=16(p-q)^2+16(p+q)^2=32(p^2+q^2)
\]
即
\[
p^2q^2=2(p^2+q^2)
\]
若 \(p=0\) 或 \(q=0\)，则上式推出 \(p=q=0\)，此时直线 \(l\) 经过原点，而过原点的直线与中心在原点的椭圆有两个公共点，不符合题设，故 \(pq\neq0\). 记 \(r=\lvert pq\rvert>0\). 由基本不等式
\(p^2+q^2\geqslant2\lvert pq\rvert=2r\)，得
\[
r^2=2(p^2+q^2)\geqslant4r
\]
所以 \(r\geqslant4\). 又
\[
S_{\triangle OPQ}=\frac{1}{2}\left|\begin{vmatrix}2p&p\\-2q&q\end{vmatrix}\right|=2\lvert pq\rvert=2r\geqslant8
\]
当 \(p=q=2\) 时，\(P=(4,2)\)、\(Q=(-4,2)\)，直线 \(l\) 为 \(y=2\)，它与椭圆相切，且三角形面积为 \(8\). 因此面积存在最小值，最小值为 \(8\).
\end{enumerate}
\end{solution}
