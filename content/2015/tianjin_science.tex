\section{单选题}

\begin{problem}
已知全集 \(U = \{1, 2, 3, 4, 5, 6, 7, 8\}\)，集合 \(A = \{2, 3, 5, 6\}\)，集合 \(B = \{1, 3, 4, 6, 7\}\)，则集合 \(A \cap \complement_U B =\)（\quad）

\choices
    {\(\{2,5\}\)}
    {\(\{3,6\}\)}
    {\(\{2,5,6\}\)}
    {\(\{2,3,5,6,8\}\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
由全集和集合 \(B\) 可得 \(\complement_U B=\{2,5,8\}\). 因此 \(A\cap\complement_U B=\{2,3,5,6\}\cap\{2,5,8\}=\{2,5\}\)，故选 A.
\end{solution}

\begin{problem}
设变量 \(x\)，\(y\) 满足约束条件 \(\begin{cases}
x + 2 \geqslant 0,\\
x - y + 3 \geqslant 0,\\
2x + y - 3 \leqslant 0,
\end{cases}\) 则目标函数 \(z = x + 6y\) 的最大值为（\quad）

\choices
    {3}
    {4}
    {18}
    {40}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
约束条件等价于 \(x\geqslant-2\)，\(y\leqslant x+3\)，\(y\leqslant3-2x\). 当 \(x\leqslant0\) 时，\(x+3\leqslant3-2x\)，所以 \(z\leqslant x+6(x+3)=7x+18\leqslant18\). 当 \(x\geqslant0\) 时，\(3-2x\leqslant x+3\)，所以 \(z\leqslant x+6(3-2x)=18-11x\leqslant18\). 在 \((x,y)=(0,3)\) 处等号成立，故选 C.
\end{solution}

阅读程序框图，运行相应的程序，则输出 \(S\) 的值为（\quad）

\texfigure[max width=6.0cm]{img_repaint/2015/tianjin_science/q3_fig1.tex}

\choices

    {6}

\begin{answer}
B

程序开始时 \(S=20\)，\(i=1\). 第一次循环后 \(i=2\)，\(S=20-2=18\)，此时 \(i\leqslant5\)；第二次循环后 \(i=4\)，\(S=18-4=14\)，此时仍有 \(i\leqslant5\)；第三次循环后 \(i=8\)，\(S=14-8=6\)，此时 \(i>5\)，程序输出 \(S=6\)，故选 B.
\end{solution}

\begin{problem}
设 \(x \in \R\)，则“\(|x - 2| < 1\)”是“\(x^2 + x - 2 > 0\)”的（\quad）

\choices
    {充分而不必要条件}
    {必要而不充分条件}
    {充要条件}
    {既不充分也不必要条件}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
由 \(|x-2|<1\) 得 \(1<x<3\). 又 \(x^2+x-2=(x+2)(x-1)\)，所以 \(x^2+x-2>0\) 的解集为 \((-\infty,-2)\cup(1,+\infty)\). 因此 \((1,3)\) 是后一个解集的子集，前一个条件能推出后一个条件；但取 \(x=4\) 时后一个不等式成立而前一个不成立，所以前一个条件不是必要条件，故选 A.
\end{solution}

\begin{problem}
如图，在圆 \(O\) 中，\(M\)，\(N\) 是弦 \(AB\) 的三等分点，弦 \(CD\)，\(CE\) 分别经过点 \(M\)，\(N\)，若 \(CM = 2\)，\(MD = 4\)，\(CN = 3\)，则线段 \(NE\) 的长为（\quad）

\bitmapfigure[max width=0.42\linewidth]{img/2015/tianjin_science/q5_fig1.png}

\choices
    {\(\frac{8}{3}\)}
    {3}
    {\(\frac{10}{3}\)}
    {\(\frac{5}{2}\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
设 \(AM=MN=NB=t\). 由相交弦定理，在点 \(M\) 处有 \(MA\cdot MB=MC\cdot MD\)，即 \(t\cdot2t=2\cdot4\)，所以 \(t=2\). 于是 \(NA=AM+MN=4\)，\(NB=2\). 再次由相交弦定理，在点 \(N\) 处有 \(NA\cdot NB=NC\cdot NE\)，故 \(4\cdot2=3\cdot NE\)，解得 \(NE=\frac{8}{3}\)，故选 A.
\end{solution}

\begin{problem}
已知双曲线 \(\frac{x^2}{a^2} - \frac{y^2}{b^2} = 1\)（\(a > 0\)，\(b > 0\)）的一条渐近线过点 \((2, \sqrt{3})\)，且双曲线的一个焦点在抛物线 \(y^2 = 4\sqrt{7} x\) 的准线上，则双曲线的方程为（\quad）

\choices
    {\(\frac{x^2}{21} -\frac{y^2}{28} = 1\)}
    {\(\frac{x^2}{28} -\frac{y^2}{21} = 1\)}
    {\(\frac{x^2}{9} -\frac{y^2}{4} = 1\)}
    {\(\frac{x^2}{4} -\frac{y^2}{3} = 1\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
双曲线的渐近线为 \(y=\pm\frac{b}{a}x\). 由渐近线经过 \((2,\sqrt3)\)，得 \(\frac{b}{a}=\frac{\sqrt3}{2}\)，即 \(b^2=\frac34a^2\). 抛物线可写成 \(y^2=4px\)，其中 \(p=\sqrt7\)，所以其准线为 \(x=-\sqrt7\). 双曲线的一个焦点在此准线上，故焦距参数满足 \(c=\sqrt7\). 又 \(c^2=a^2+b^2\)，于是 \(7=a^2+\frac34a^2\)，解得 \(a^2=4\)，\(b^2=3\). 因此双曲线方程为 \(\frac{x^2}{4}-\frac{y^2}{3}=1\)，故选 D.
\end{solution}

\begin{problem}
已知定义在 \(\R\) 上的函数 \(f(x) = 2^{|x - m|} - 1\) （\(m\) 为实数）为偶函数，记 \(a = f(\log_{0.5} 3)\)，\(b = f(\log_{2} 5)\)，\(c = f(2m)\)，则 \(a\)，\(b\)，\(c\) 的大小关系为（\quad）

\choices
    {\(a < b < c\)}
    {\(a < c < b\)}
    {\(c < a < b\)}
    {\(c < b < a\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
函数为偶函数，所以对任意 \(x\) 有 \(f(-x)=f(x)\). 因指数函数 \(2^t\) 在 \(\R\) 上严格递增，故 \(|x+m|=|x-m|\). 两边平方得 \((x+m)^2=(x-m)^2\)，即 \(4mx=0\) 对任意 \(x\) 成立，因此 \(m=0\). 于是 \(f(x)=2^{|x|}-1\). 因为 \(\log_{0.5}3=-\log_2 3\)，所以 \(a=2^{\log_2 3}-1=2\)；\(b=2^{\log_2 5}-1=4\)；\(c=f(0)=0\). 故 \(c<a<b\)，选 C.
\end{solution}

\begin{problem}
已知函数 \( f(x) = \begin{cases}
2 - |x|, & x \leqslant 2,\\
(x - 2)^2, & x > 2,
\end{cases} \) 函数 \( g(x) = b - f(2 - x)\)，其中 \( b \in \R\). 若函数 \( y = f(x) - g(x) \) 恰有4个零点，则 \( b \) 的取值范围是（\quad）

\choices
    {\(\left(\frac{7}{4}, +\infty\right)\)}
    {\(\left(-\infty, \frac{7}{4}\right)\)}
    {\(\left(0, \frac{7}{4}\right)\)}
    {\(\left(\frac{7}{4}, 2\right)\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
方程 \(f(x)-g(x)=0\) 等价于 \(f(x)+f(2-x)=b\). 记 \(h(x)=f(x)+f(2-x)\)，按 \(x<0\)，\(0\leqslant x\leqslant2\)，\(x>2\) 分段计算得
\[
h(x)=\begin{cases}
x^2+x+2=\left(x+\frac12\right)^2+\frac74, & x<0,\\
2, & 0\leqslant x\leqslant2,\\
x^2-5x+8=\left(x-\frac52\right)^2+\frac74, & x>2.
\end{cases}
\]
当 \(\frac74<b<2\) 时，左、右两个开区间内的二次函数各有两个解，且中间区间无解，共有4个零点. 当 \(b=\frac74\) 时只有两个切点；当 \(b=2\) 时中间整个区间都是解；当 \(b>2\) 时左右各只有一个解；当 \(b<\frac74\) 时无解. 因此 \(b\in\left(\frac74,2\right)\)，故选 D.
\end{solution}

\section{填空题}

\begin{problem}
\(\i\) 是虚数单位，若复数 \((1 - 2\i)(a + \i)\) 是纯虚数，则实数 \(a\) 的值为\fillinblank{}.
\end{problem}

\begin{answer}
\(-2\)
\end{answer}

\begin{solution}
展开得 \((1-2\i)(a+\i)=(a+2)+(1-2a)\i\). 纯虚数的实部为零，所以 \(a+2=0\)，解得 \(a=-2\). 此时虚部为 \(5\ne0\)，确为纯虚数.
\end{solution}

\begin{problem}
一个几何体的三视图如图所示（单位：\(\mathrm{m}\)），则该几何体的体积为\fillinblank{}\(\mathrm{m}^3\).
\bitmapfigure[max width=0.58\linewidth]{img/2015/tianjin_science/q10_fig1.png}
\end{problem}

\begin{answer}
\(\frac{8\pi}{3}\)
\end{answer}

\begin{solution}
由三视图可知，侧视图是半径为 \(1\mathrm{~m}\) 的圆，正视图显示中间圆柱的高为 \(2\mathrm{~m}\)，两侧各有高为 \(1\mathrm{~m}\) 的圆锥. 故该几何体由一个底面半径为 \(1\mathrm{~m}\)、高为 \(2\mathrm{~m}\) 的圆柱，以及圆柱两端两个底面半径为 \(1\mathrm{~m}\)、高为 \(1\mathrm{~m}\) 的圆锥组成. 因此体积为
\[
V=\pi\cdot1^2\cdot2+2\cdot\frac13\pi\cdot1^2\cdot1=\frac{8\pi}{3}\mathrm{~m}^3
\]
所以填 \(\frac{8\pi}{3}\).
\end{solution}

\begin{problem}
曲线 \( y = x^{2} \) 与直线 \( y = x \) 所围成的封闭图形的面积为\fillinblank{}.
\end{problem}

\begin{answer}
\(\frac{1}{6}\)
\end{answer}

\begin{solution}
令 \(x^2=x\)，得两个交点的横坐标为 \(x=0\) 和 \(x=1\). 在 \([0,1]\) 上有 \(x\geqslant x^2\)，所以所围面积为
\[
S=\int_0^1(x-x^2)\,\mathrm{d}x=\left(\frac{x^2}{2}-\frac{x^3}{3}\right)\Big|_0^1=\frac16
\]
\end{solution}

\begin{problem}
在 \(\left(x - \frac{1}{4x}\right)^6\) 的展开式中，\(x^2\) 的系数为\fillinblank{}.
\end{problem}

\begin{answer}
\(\frac{15}{16}\)
\end{answer}

\begin{solution}
通项中从第二项取 \(k\) 次的项为 \(\mathrm C_6^k x^{6-k}\left(-\frac{1}{4x}\right)^k=\mathrm C_6^k\frac{(-1)^k}{4^k}x^{6-2k}\). 令 \(6-2k=2\)，得 \(k=2\). 故 \(x^2\) 的系数为 \(\mathrm C_6^2\frac{1}{4^2}=\frac{15}{16}\).
\end{solution}

\begin{problem}
在 \(\triangle ABC\) 中，内角 \(A\)，\(B\)，\(C\) 所对的边分别为 \(a\)，\(b\)，\(c\). 已知 \(\triangle ABC\) 的面积为 \(3\sqrt{15}\)，\(b - c = 2\)，\(\cos A = -\frac{1}{4}\)，则 \(a\) 的值为\fillinblank{}.
\end{problem}

\begin{answer}
\(8\)
\end{answer}

\begin{solution}
因为 \(A\) 是三角形内角，所以 \(\sin A=\sqrt{1-\cos^2A}=\frac{\sqrt{15}}4\). 由面积公式
\[
\frac12bc\sin A=3\sqrt{15}
\]
得 \(bc=24\). 结合 \(b-c=2\)，有 \(c(c+2)=24\)，由边长为正得 \(c=4\)，从而 \(b=6\). 由余弦定理，\(a^2=b^2+c^2-2bc\cos A=36+16-2\cdot24\cdot\left(-\frac14\right)=64\)，所以 \(a=8\).
\end{solution}

\begin{problem}
在等腰梯形 \(ABCD\) 中，已知 \(AB \parallel DC\)，\(AB = 2\)，\(BC = 1\)，\(\angle ABC = 60^\circ\). 动点 \(E\) 和 \(F\) 分别在线段 \(BC\) 和 \(DC\) 上，且 \(\overrightarrow{BE} = \lambda \overrightarrow{BC}\)，\(\overrightarrow{DF} = \frac{1}{9\lambda}\overrightarrow{DC}\)，则 \(\overrightarrow{AE} \cdot \overrightarrow{AF}\) 的最小值为\fillinblank{}.
\end{problem}

\begin{answer}
\(\frac{29}{18}\)
\end{answer}

\begin{solution}
建立平面直角坐标系，取 \(B=(0,0)\)，\(A=(2,0)\). 由 \(BC=1\) 和 \(\angle ABC=60^\circ\)，可取 \(C=\left(\frac12,\frac{\sqrt3}{2}\right)\). 等腰梯形只有一组平行对边，故排除平行四边形构型，结合 \(AD=BC=1\) 得 \(D=\left(\frac32,\frac{\sqrt3}{2}\right)\)，从而 \(\overrightarrow{DC}=(-1,0)\).

由于 \(E\)，\(F\) 在线段上，参数满足 \(\frac19\leqslant\lambda\leqslant1\). 由题设
\(E=\left(\frac\lambda2,\frac{\sqrt3\lambda}{2}\right)\)，\(F=\left(\frac32-\frac{1}{9\lambda},\frac{\sqrt3}{2}\right)\).
所以
\(\overrightarrow{AE}=\left(\frac\lambda2-2,\frac{\sqrt3\lambda}{2}\right)\)，\(\overrightarrow{AF}=\left(-\frac12-\frac{1}{9\lambda},\frac{\sqrt3}{2}\right)\)
从而
\[
\overrightarrow{AE}\cdot\overrightarrow{AF}=\frac\lambda2+\frac{2}{9\lambda}+\frac{17}{18}
\]
由基本不等式，\(\frac\lambda2+\frac{2}{9\lambda}\geqslant2\sqrt{\frac\lambda2\cdot\frac{2}{9\lambda}}=\frac23\)，等号当且仅当 \(\lambda=\frac23\)，该值在取值范围内. 因此最小值为 \(\frac23+\frac{17}{18}=\frac{29}{18}\).
\end{solution}

\begin{problem}
已知函数 \(f(x) = \sin^2 x - \sin^2 \left( x - \frac{\pi}{6} \right)\)，\(x \in \R\).

\begin{enumerate}
\item 求 \( f(x) \) 的最小正周期.
\item 求 \( f(x) \) 在区间 \( \left[-\frac{\pi}{3}, \frac{\pi}{4}\right] \) 上的最大值和最小值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由降幂公式，
\[
f(x)=\frac{1-\cos2x}{2}-\frac{1-\cos\left(2x-\frac\pi3\right)}{2}=\frac12\sin\left(2x-\frac\pi6\right)
\]
函数 \(\sin\left(2x-\frac\pi6\right)\) 的最小正周期为 \(\pi\)，故 \(f(x)\) 的最小正周期为 \(T=\pi\).
\item 当 \(x\in\left[-\frac\pi3,\frac\pi4\right]\) 时，令 \(t=2x-\frac\pi6\)，则 \(t\in\left[-\frac{5\pi}{6},\frac\pi3\right]\). 在 \(\left[-\frac{5\pi}{6},-\frac\pi2\right]\) 上，\(\sin t\) 递减；在 \(\left[-\frac\pi2,\frac\pi3\right]\) 上，\(\sin t\) 递增，所以 \(\sin t\) 在 \(t=-\frac\pi2\) 处取得最小值 \(-1\)，在右端点 \(t=\frac\pi3\) 处取得最大值 \(\frac{\sqrt3}{2}\). 因此 \(f(x)\) 的最小值为 \(-\frac12\)，最大值为 \(\frac{\sqrt3}{4}\).
\end{enumerate}
\end{solution}

\begin{problem}
为推动乒乓球运动的发展，某乒乓球比赛允许不同协会的运动员组队参加. 现有来自甲协会的运动员3名，其中种子选手2名；乙协会的运动员5名，其中种子选手3名. 从这8名运动员中随机选择4人参加比赛.

\begin{enumerate}
\item 设 A 为事件“选出的 4 人中恰有 2 名种子选手，且这 2 名种子选手来自同一个协会”，求事件 A 发生的概率；
\item 设 X 为选出的 4 人中种子选手的人数，求随机变量 X 的分布列和数学期望.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 从 8 人中任取 4 人的基本事件数为 \(\mathrm C_8^4=70\). 甲、乙两个协会共有 \(1+2=3\) 名非种子选手. 若两名种子选手来自甲协会，则取出甲协会的 2 名种子选手，并从 3 名非种子选手中取 2 名，有 \(\mathrm C_2^2\mathrm C_3^2\) 种；若两名种子选手来自乙协会，则从乙协会的 3 名种子选手中取 2 名，并从 3 名非种子选手中取 2 名，有 \(\mathrm C_3^2\mathrm C_3^2\) 种. 这两类情况互斥且恰好覆盖事件 \(A\)，因此
\[
P(A)=\frac{\mathrm C_2^2\mathrm C_3^2+\mathrm C_3^2\mathrm C_3^2}{\mathrm C_8^4}=\frac{3+9}{70}=\frac{12}{70}=\frac6{35}
\]
\item 共有 5 名种子选手和 3 名非种子选手，所以 \(X\) 的可能取值为 \(1\)，\(2\)，\(3\)，\(4\). 对 \(k=1\)，\(2\)，\(3\)，\(4\)，
\[
P(X=k)=\frac{\mathrm C_5^k\mathrm C_3^{4-k}}{\mathrm C_8^4}
\]
于是 \(P(X=1)=\frac1{14}\)，\(P(X=2)=\frac37\)，\(P(X=3)=\frac37\)，\(P(X=4)=\frac1{14}\)，这就给出了 \(X\) 的分布列. 其数学期望为
\[
E(X)=1\cdot\frac1{14}+2\cdot\frac37+3\cdot\frac37+4\cdot\frac1{14}=\frac52
\]
\end{enumerate}
\end{solution}

\begin{problem}
已知数列 \(\{a_{n}\}\) 满足 \(a_{n + 2} = qa_{n}\)（\(q\) 为实数，且 \(q \neq 1\)），\(n \in \mathbf{N}^{*}\)，\(a_{1} = 1\)，\(a_{2} = 2\)，且 \(a_{2} + a_{3}\)，\(a_{3} + a_{4}\)，\(a_{4} + a_{5}\) 成等差数列.

\begin{enumerate}
\item 求 \(q\) 的值和 \( \{a_{n}\} \) 的通项公式；
\item 设 \(b_{n}=\frac{\log_{2}a_{2n}}{a_{2n-1}}\)，\(n\in \mathbf{N}^{*}\)，求数列 \( \{b_{n}\} \) 的前 \(n\) 项和.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由递推式得 \(a_3=q\)，\(a_4=2q\)，\(a_5=q^2\). 三个数成等差数列，所以
\[
2(a_3+a_4)=(a_2+a_3)+(a_4+a_5)
\]
即 \(6q=2+3q+q^2\)，从而 \((q-1)(q-2)=0\). 由 \(q\ne1\)，得 \(q=2\). 因此
\(a_{2k-1}=2^{k-1}\)，\(a_{2k}=2^k\)，\((k\in\mathbf N^*)\)
即
\[
a_n=\begin{cases}
2^{\frac{n-1}{2}},&n\text{为奇数},\\
2^{\frac n2},&n\text{为偶数}.
\end{cases}
\]
\item 由通项公式，\(a_{2n}=2^n\)，\(a_{2n-1}=2^{n-1}\)，所以 \(b_n=\frac{n}{2^{n-1}}\). 设前 \(n\) 项和为 \(S_n\)，则
\[
S_n=\sum_{k=1}^n\frac{k}{2^{k-1}}
\]
将此式乘以 \(\frac12\) 后错位相减，得
\[
\frac12S_n=1+\sum_{k=2}^{n}\frac{1}{2^{k-1}}-\frac{n}{2^n}
 =2-\frac{n+2}{2^n}
\]
故 \(S_n=4-\frac{n+2}{2^{n-1}}\).
\end{enumerate}
\end{solution}

\begin{problem}
已知椭圆 \(\frac{x^2}{a^2} + \frac{y^2}{b^2} = 1\)（\(a > b > 0\)）的左焦点为 \(F(-c, 0)\)，离心率为 \(\frac{\sqrt{3}}{3}\)，点 \(M\) 在椭圆上且位于第一象限，直线 \(FM\) 被圆 \(x^2 + y^2 = \frac{b^2}{4}\) 截得的线段的长为 \(c\)，\(|FM| = \frac{4\sqrt{3}}{3}\).

\begin{enumerate}
\item 求直线 \(FM\) 的斜率；
\item 求椭圆的方程；
\item 设动点 \(P\) 在椭圆上，若直线 \(FP\) 的斜率大于 \(\sqrt{2}\)，求直线 \(OP\)（\(O\) 为原点）的斜率的取值范围.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由离心率得 \(c/a=\sqrt3/3\)，即 \(a^2=3c^2\)，再由 \(a^2=b^2+c^2\) 得 \(b^2=2c^2\). 设直线 \(FM\) 的斜率为 \(k\)，其方程为 \(y=k(x+c)\). 圆心 \(O\) 到该直线的距离为 \(d=\frac{|kc|}{\sqrt{k^2+1}}\). 由弦长公式，
\[
2\sqrt{\frac{b^2}{4}-d^2}=c
\]
代入 \(b^2=2c^2\) 并平方，得 \(\frac12-\frac{k^2}{k^2+1}=\frac14\)，所以 \(3k^2=1\). 由于 \(M\) 在第一象限而 \(F\) 在 \(x\) 轴负半轴，直线斜率为正，故 \(k=\frac{\sqrt3}{3}\).
\item 由上问，直线为 \(y=\frac{x+c}{\sqrt3}\). 联立椭圆方程 \(\frac{x^2}{3c^2}+\frac{y^2}{2c^2}=1\)，得
\[
2x^2+(x+c)^2=6c^2
\]
即 \((x-c)(3x+5c)=0\). 第一象限交点为 \(M=\left(c,\frac{2c}{\sqrt3}\right)\)，所以
\[
|FM|=\sqrt{(2c)^2+\left(\frac{2c}{\sqrt3}\right)^2}=\frac{4c}{\sqrt3}
\]
由题设 \(|FM|=\frac{4\sqrt3}{3}=\frac4{\sqrt3}\)，得 \(c=1\). 于是 \(a^2=3\)，\(b^2=2\)，椭圆方程为 \(\frac{x^2}{3}+\frac{y^2}{2}=1\).
\item 设 \(P=(x,y)\)，则直线 \(FP\) 的斜率为 \(\frac{y}{x+1}>\sqrt2\). 因为该斜率为正，所以 \(x+1\ne0\)，且 \(y\) 与 \(x+1\) 同号，因而可以平方；又由椭圆方程得 \(y^2=2-\frac{2x^2}{3}\). 因此
\[
2-\frac{2x^2}{3}>2(x+1)^2
\]
化简得 \(x(2x+3)<0\)，所以 \(-\frac32<x<0\). 由于斜率为正，\(y\) 与 \(x+1\) 同号.

当 \(-\frac32<x<-1\) 时，\(y<0\)，故 \(\frac yx>0\)，且
\[
\left(\frac yx\right)^2=\frac{2}{x^2}-\frac23
\]
随着 \(x\) 从 \(-\frac32\) 增大到 \(-1\)，\(\frac yx\) 从 \(\frac{\sqrt2}{3}\) 增大到 \(\frac{2\sqrt3}{3}\)，得到 \(\frac{\sqrt2}{3}<\frac yx<\frac{2\sqrt3}{3}\).

当 \(-1<x<0\) 时，\(y>0\)，故 \(\frac yx<0\)，同一式表明 \(\frac yx\) 的取值为 \(\left(-\infty,-\frac{2\sqrt3}{3}\right)\). 点 \(x=-1\) 时直线 \(FP\) 为竖直线，不具有有限斜率. 综上，直线 \(OP\) 的斜率取值范围为
\[
\left(-\infty,-\frac{2\sqrt3}{3}\right)\cup\left(\frac{\sqrt2}{3},\frac{2\sqrt3}{3}\right)
\]
\end{enumerate}
\end{solution}

\begin{problem}
已知函数 \(f(x) = nx - x^n\)，\(x \in \R\). 其中 \( n \in \mathbf{N}^* \)，且 \( n \geqslant 2 \).

\begin{enumerate}
\item 讨论 \( f(x) \) 的单调性；
\item 设曲线 \( y = f(x) \) 与 \( x \) 轴正半轴的交点为 \(P\)，曲线在点 \(P\) 处的切线方程为 \( y = g(x) \). 求证：对于任意的正实数 \( x\)，都有 \( f(x) \leqslant g(x) \)；
\item 若关于 \(x\) 的方程 \(f(x)=a\)（\(a\) 为实数）有两个正实数根 \(x_{1}\)，\(x_{2}\)，求证： \(|x_{2} - x_{1}| < \frac{a}{1 - n} + 2\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item \(f'(x)=n-nx^{n-1}=n(1-x^{n-1})\). 当 \(n\) 为奇数时，\(n-1\) 为偶数，故在 \((-\infty,-1)\) 和 \((1,+\infty)\) 上 \(f'(x)<0\)，在 \((-1,1)\) 上 \(f'(x)>0\). 所以 \(f(x)\) 在这两个区间上分别单调递减，在 \((-1,1)\) 上单调递增. 当 \(n\) 为偶数时，\(n-1\) 为奇数，故 \(x<1\) 时 \(f'(x)>0\)，\(x>1\) 时 \(f'(x)<0\)，所以 \(f(x)\) 在 \((-\infty,1)\) 上单调递增，在 \((1,+\infty)\) 上单调递减.
\item 设 \(p=n^{\frac1{n-1}}\)，则正半轴上的交点为 \(P=(p,0)\)，且 \(f'(p)=n(1-n)\). 所以切线函数为 \(g(x)=n(1-n)(x-p)\). 令 \(h(x)=g(x)-f(x)\)，则 \(h(p)=0\)，且
\[
h'(x)=n(1-n)-n(1-x^{n-1})=n(x^{n-1}-n)
\]
当 \(0<x<p\) 时，\(x^{n-1}<p^{n-1}=n\)，故 \(h'(x)<0\)；当 \(x>p\) 时，\(h'(x)>0\). 因此 \(h\) 在 \(p\) 的左侧递减、右侧递增，且在 \(p\) 处取得最小值 \(0\). 所以对任意正实数 \(x\)，有 \(h(x)\geqslant0\)，即 \(f(x)\leqslant g(x)\).
\item 不妨设 \(0<x_1<x_2\). 由 \(f(0)=f(p)=0\)，以及 \(f\) 在 \((0,1)\) 上递增、在 \((1,p)\) 上递减可知，两个正根满足 \(0<x_1<1<x_2<p\)，且 \(a>0\). 由上问在 \(x_2\) 处应用不等式，得
\[
a=f(x_2)\leqslant g(x_2)=n(n-1)(p-x_2)
\]
所以 \(x_2\leqslant p-\frac{a}{n(n-1)}\). 又因为 \(a=f(x_1)=nx_1-x_1^n\)，于是
\[
\begin{aligned}
x_2-x_1+\frac{a}{n-1}
&\leqslant p-x_1-\frac{a}{n(n-1)}+\frac{a}{n-1}\\
&=p-x_1+\frac an
=p-\frac{x_1^n}{n}<p.
\end{aligned}
\]
对 \(n\geqslant2\)，有 \(n\leqslant2^{n-1}\)，从而 \(p=n^{1/(n-1)}\leqslant2\). 因此
\[
x_2-x_1+\frac{a}{n-1}<2
\]
即 \(|x_2-x_1|<2-\frac{a}{n-1}=\frac{a}{1-n}+2\).
\end{enumerate}
\end{solution}
