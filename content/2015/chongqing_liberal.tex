\section{单选题}

\begin{problem}
已知集合 \(A = \{1, 2, 3\}\)，\(B = \{1, 3\}\)，则 \(A \cap B =\)（\quad）

\choices
    {\(\{2\}\)}
    {\(\{1,2\}\)}
    {\(\{1,3\}\)}
    {\(\{1,2,3\}\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
交集中的元素必须同时属于集合 \(A\) 和集合 \(B\). 集合 \(A\) 与集合 \(B\) 的公共元素为 \(1,3\)，所以 \(A\cap B=\{1,3\}\)，故选 C.
\end{solution}

\begin{problem}
“ \(x = 1\) ”是“ \(x^{2} - 2x + 1 = 0\) ”的（\quad）

\choices
    {充要条件}
    {充分而不必要条件}
    {必要而不充分条件}
    {既不充分也不必要条件}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
将方程因式分解，得
\[
x^{2}-2x+1=(x-1)^{2}=0
\]
因此方程成立当且仅当 \(x=1\). 也就是说，“\(x=1\)”既是该方程成立的充分条件，也是必要条件，故选 A.
\end{solution}

\begin{problem}
函数 \(f(x) = \log_2 (x^2 + 2x - 3)\) 的定义域是（\quad）

\choices
    {\([-3, 1]\)}
    {\((-3, 1)\)}
    {\((-\infty, -3] \cup [1, +\infty)\)}
    {\((-\infty, -3) \cup (1, +\infty)\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
因为对数的真数必须为正数，函数的定义域满足
\[
x^{2}+2x-3=(x+3)(x-1)>0
\]
由两个一次因子的符号可知，当 \(x<-3\) 或 \(x>1\) 时乘积为正，所以定义域为 \((-\infty,-3)\cup(1,+\infty)\)，故选 D.
\end{solution}

重庆市 2013 年各月的平均气温（\(^{\circ}\mathrm{C}\)）数据的茎叶图如下：

\begin{problem}
若 \(\tan \alpha = \frac{1}{3}\)，\(\tan (\alpha + \beta) = \frac{1}{2}\)，则 \(\tan \beta =\)（\quad）

\choices
    {\(\frac{1}{7}\)}
    {\(\frac{1}{6}\)}
    {\(\frac{5}{7}\)}
    {\(\frac{5}{6}\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
设 \(t=\tan\beta\). 由正切的和角公式，得
\[
\frac{\frac13+t}{1-\frac13t}=\frac12
\]
解得
\(\frac23+2t=1-\frac13t\)，\(\frac73t=\frac13\)，\(t=\frac17\).
所以 \(\tan\beta=\frac17\)，故选 A.
\end{solution}

\begin{problem}
已知非零向量 \(\bs{a}\)，\(\bs{b}\) 满足 \(\left|\bs{b}\right| = 4\left|\bs{a}\right|\)，且 \(\bs{a} \perp (2\bs{a} + \bs{b})\)，则 \(\bs{a}\) 与 \(\bs{b}\) 的夹角为（\quad）

\choices
    {\(\frac{\pi}{3}\)}
    {\(\frac{\pi}{2}\)}
    {\(\frac{2\pi}{3}\)}
    {\(\frac{5\pi}{6}\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
设 \(\bs{a}\) 与 \(\bs{b}\) 的夹角为 \(\theta\). 由垂直关系，
\[
\bs{a}\cdot(2\bs{a}+\bs{b})=0
\]
所以
\(2|\bs{a}|^{2}+\bs{a}\cdot\bs{b}=0\)，\(\bs{a}\cdot\bs{b}=-2|\bs{a}|^{2}\).
又因为 \(|\bs{b}|=4|\bs{a}|\)，故
\[
\cos\theta=\frac{\bs{a}\cdot\bs{b}}{|\bs{a}||\bs{b}|}
 =\frac{-2|\bs{a}|^{2}}{4|\bs{a}|^{2}}=-\frac12
\]
由于 \(0\leqslant\theta\leqslant\pi\)，所以 \(\theta=\frac{2\pi}{3}\)，故选 C.
\end{solution}

\begin{problem}
设双曲线 \(\frac{x^2}{a^2} - \frac{y^2}{b^2} = 1 \) （\(a > 0\)，\(b > 0\)）的右焦点是 \(F\)，左、右顶点分别是 \(A_1\)，\(A_2\)，过 \(F\) 作 \(A_1A_2\) 的垂线与双曲线交于 \(B\)，\(C\) 两点. 若 \(A_1B \perp A_2C\)，则该双曲线的渐近线的斜率为（\quad）

\choices
    {\(\pm \frac{1}{2}\)}
    {\(\pm \frac{\sqrt{2}}{2}\)}
    {\(\pm 1\)}
    {\(\pm \sqrt{2}\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
设双曲线的焦距参数满足 \(c^{2}=a^{2}+b^{2}\). 取右焦点 \(F=(c,0)\)，则过 \(F\) 且垂直于 \(A_{1}A_{2}\) 的直线为 \(x=c\). 设其与双曲线的交点为 \(B=(c,y)\)，\(C=(c,-y)\)，则
\[
\frac{c^{2}}{a^{2}}-\frac{y^{2}}{b^{2}}=1
\]
从而
\[
y^{2}=b^{2}\left(\frac{c^{2}}{a^{2}}-1\right)=\frac{b^{4}}{a^{2}}
\]
因此可取
\(\overrightarrow{A_{1}B}=\left(a+c,\frac{b^{2}}a\right)\)，\(\overrightarrow{A_{2}C}=\left(c-a,-\frac{b^{2}}a\right)\).
由 \(A_{1}B\perp A_{2}C\)，得
\[
0=\overrightarrow{A_{1}B}\cdot\overrightarrow{A_{2}C}
=(a+c)(c-a)-\frac{b^{4}}{a^{2}}
=c^{2}-a^{2}-\frac{b^{4}}{a^{2}}
=b^{2}-\frac{b^{4}}{a^{2}}
\]
由于 \(b>0\)，所以 \(b^{2}=a^{2}\). 双曲线渐近线为 \(y=\pm\frac ba x\)，故其斜率为 \(\pm1\)，选 C.
\end{solution}

\begin{problem}
若不等式组
\(
\begin{cases}
x + y - 2 \leqslant 0,\\
x + 2y - 2 \geqslant 0,\\
x - y + 2m \geqslant 0,
\end{cases}
\)
表示的平面区域为三角形，且其面积等于 \(\frac{4}{3}\)，则 \(m\) 的值为（\quad）

\choices
    {\(-3\)}
    {\(1\)}
    {\(\frac{4}{3}\)}
    {\(3\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
三条边界直线分别为
\(L_{1}:x+y=2\)，\(L_{2}:x+2y=2\)，\(L_{3}:x-y+2m=0\).
它们两两相交于
\[
P=L_{1}\cap L_{2}=(2,0)
\]
\[
Q=L_{1}\cap L_{3}=(1-m,1+m)
\]
\[
R=L_{2}\cap L_{3}=\left(\frac{2(1-2m)}3,\frac{2(1+m)}3\right)
\]
令 \(u=1+m\). 三条半平面的交集在顶点 \(P\) 处的两条边界射线分别沿着 \(\overrightarrow{PQ}=(-u,u)\) 和 \(\overrightarrow{PR}=\left(-\frac{4u}{3},\frac{2u}{3}\right)\) 延伸；只有 \(u>0\) 时，第三条边界直线截断该可行角并形成有界非退化三角形. 因此 \(m>-1\).
所以三角形面积为
\[
S=\frac12\left|\det\begin{pmatrix}-u&u\\-\frac{4u}{3}&\frac{2u}{3}\end{pmatrix}\right|
=\frac{u^{2}}3=\frac{(1+m)^{2}}3
\]
由题设 \(S=\frac43\)，得
\((1+m)^{2}=4\)，\(m=1\ \text{或}\ m=-3\).
结合三角形条件 \(m>-1\)，只能取 \(m=1\)，故选 B.
\end{solution}

\begin{problem}
复数 \((1 + 2\i)\i\) 的实部为\fillinblank{}.
\end{problem}

\begin{answer}
\(-2\)
\end{answer}

\begin{solution}
利用 \(\i^{2}=-1\)，有
\[
(1+2\i)\i=\i+2\i^{2}=-2+\i
\]
所以该复数的实部为 \(-2\).
\end{solution}

\begin{problem}
若点 \(P(1,2)\) 在以坐标原点为圆心的圆上，则该圆在点 \(P\) 处的切线方程为\fillinblank{}.
\end{problem}

\begin{answer}
\(x+2y-5=0\)
\end{answer}

\begin{solution}
设圆的半径为 \(r\). 因为圆心为原点且点 \(P(1,2)\) 在圆上，所以
\[
r^{2}=1^{2}+2^{2}=5
\]
圆的方程为 \(x^{2}+y^{2}=5\). 圆在 \(P\) 处的切线垂直于半径 \(OP\)，而 \(\overrightarrow{OP}=(1,2)\) 是切线的一个法向量，因此切线方程为
\[
1(x-1)+2(y-2)=0
\]
即 \(x+2y-5=0\).
\end{solution}

\begin{problem}
设 \(\triangle ABC\) 的内角 \(A\)，\(B\)，\(C\) 的对边分别为 \(a\)，\(b\)，\(c\)，且 \(a = 2\)，\(\cos C = -\frac{1}{4}\)，\(3\sin A = 2\sin B\)，则 \(c =\fillinblank{}\).
\end{problem}

\begin{answer}
\(4\)
\end{answer}

\begin{solution}
由正弦定理，存在同一个正数 \(2R\)，使得
\(\sin A=\frac{a}{2R}\)，\(\sin B=\frac{b}{2R}\).
代入 \(3\sin A=2\sin B\) 和 \(a=2\)，得
\(3a=2b\)，\(b=3\).
在三角形 \(ABC\) 中，对角 \(C\) 使用余弦定理，得
\[
c^{2}=a^{2}+b^{2}-2ab\cos C
=2^{2}+3^{2}-2\cdot2\cdot3\left(-\frac14\right)=16
\]
因为 \(c>0\)，所以 \(c=4\).
\end{solution}

\begin{problem}
设 \(a\)，\(b > 0\)，\(a + b = 5\)，则 \(\sqrt{a + 1} + \sqrt{b + 3}\) 的最大值为\fillinblank{}.
\end{problem}

\begin{answer}
\(3\sqrt{2}\)
\end{answer}

\begin{solution}
令 \(x=a+1\)，\(y=b+3\). 则 \(x+y=a+b+4=9\)，并且 \(x\)，\(y>0\). 由柯西不等式，
\[
\left(\sqrt{a+1}+\sqrt{b+3}\right)^{2}
\leqslant 2\bigl((a+1)+(b+3)\bigr)=18
\]
所以原式不超过 \(3\sqrt2\). 当且仅当 \(a+1=b+3\) 时等号成立，此时结合 \(a+b=5\) 得
\(a=\frac72\)，\(b=\frac32\)
满足 \(a\)，\(b>0\). 因此最大值为 \(3\sqrt2\).
\end{solution}

\begin{problem}
在区间 \([0, 5]\) 上随机地选择一个数 \(p\)，则方程 \(x^2 + 2px + 3p - 2 = 0\) 有两个负根的概率为\fillinblank{}.
\end{problem}

\begin{answer}
\(\frac{2}{3}\)
\end{answer}

\begin{solution}
设方程的两个根为 \(x_{1}\)，\(x_{2}\). 要使两个根均为负数，首先由韦达定理必须有
\(x_{1}+x_{2}=-2p<0\)，\(x_{1}x_{2}=3p-2>0\).
这给出 \(p>0\) 和 \(p>\frac23\). 另外，方程有实根还需
\[
\Delta=(2p)^{2}-4(3p-2)=4(p-1)(p-2)\geqslant0
\]
即 \(p\leqslant1\) 或 \(p\geqslant2\). 结合 \(p\in[0,5]\)，可行范围为
\[
p\in\left(\frac23,1\right]\cup[2,5]
\]
当 \(p=1\) 或 \(p=2\) 时为重根，是否把重根计入“两个负根”不影响连续均匀选择的概率；按 \(\Delta\geqslant0\) 计，可行集合长度为
\[
\left(1-\frac23\right)+(5-2)=\frac{10}{3}
\]
因此所求概率为
\[
\frac{\frac{10}{3}}5=\frac23
\]
\end{solution}

\begin{problem}
已知等差数列 \(\{a_{n}\}\) 满足 \(a_3 = 2\)，前 3 项和 \(S_3 = \frac{9}{2}\).

\begin{enumerate}
\item 求 \(\{a_{n}\}\) 的通项公式；
\item 设等比数列 \(\{b_{n}\}\) 满足 \(b_1 = a_1\)，\(b_4 = a_{15}\)，求 \(\{b_{n}\}\) 的前 \(n\) 项和 \(T_n\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设等差数列 \(\{a_n\}\) 的公差为 \(d\)，则 \(a_1=a_3-2d=2-2d\). 由前 \(3\) 项和得
\[
S_3=\frac{3}{2}\left(a_1+a_3\right)=\frac{9}{2}
\]
所以 \(a_1+a_3=3\)，从而 \(a_1=1\)，再由 \(a_3-a_1=2d\) 得 \(d=\frac{1}{2}\). 因此
\[
a_n=a_1+(n-1)d=1+\frac{n-1}{2}=\frac{n+1}{2}
\]
\item 由第（1）问，\(b_1=a_1=1\)，且 \(b_4=a_{15}=\frac{15+1}{2}=8\). 设等比数列 \(\{b_n\}\) 的公比为 \(q\)，则 \(b_4=b_1q^3\)，所以 \(q^3=8\)，得 \(q=2\). 因而 \(b_n=2^{n-1}\)，其前 \(n\) 项和为
\[
T_n=\frac{b_1\left(q^n-1\right)}{q-1}=2^n-1
\]
\end{enumerate}
\end{solution}

\begin{problem}
随着我国经济的发展，居民的储蓄存款逐年增长. 设某地区城乡居民人民币储蓄存款（年底余额）如下表：

\begin{center}
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{|c|ccccc|}
\hline
年份 & 2010 & 2011 & 2012 & 2013 & 2014 \\
\hline
时间代号 \(t\) & 1 & 2 & 3 & 4 & 5 \\
\hline
储蓄存款 \(y\)（千亿元） & 5 & 6 & 7 & 8 & 10 \\
\hline
\end{tabular}
\end{center}

\begin{enumerate}
\item 求 \(y\) 关于 \(t\) 的回归方程 \(\hat{y} = \hat{b} t + \hat{a}\)；
\item 用所求回归方程预测该地区 2015 年（\(t = 6\)）的人民币储蓄存款. 附：回归方程 \(\hat{y} = \hat{b} t + \hat{a}\) 中，\(\hat{b} = \frac{\sum_{i=1}^{n} t_{i} y_{i} - n \overline{t} \, \overline{y}}{\sum_{i=1}^{n} t_{i}^{2} - n \overline{t}^{2}}\)，\(\hat{a} = \overline{y} - \hat{b} \overline{t}\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由表中数据得 \(n=5\)，\(\overline{t}=3\)，\(\overline{y}=\frac{5+6+7+8+10}{5}=\frac{36}{5}\). 又
\[
\sum_{i=1}^{5}t_i y_i=1\cdot5+2\cdot6+3\cdot7+4\cdot8+5\cdot10=120
\]
\[
\sum_{i=1}^{5}t_i^2=1^2+2^2+3^2+4^2+5^2=55
\]
代入回归系数公式，得
\[
\hat{b}=\frac{120-5\cdot3\cdot\frac{36}{5}}{55-5\cdot3^2}=\frac{6}{5}
\]
\[
\hat{a}=\frac{36}{5}-\frac{6}{5}\cdot3=\frac{18}{5}
\]
所以 \(y\) 关于 \(t\) 的回归方程为 \(\hat{y}=\frac{6}{5}t+\frac{18}{5}\).
\item 当 \(t=6\) 时，\(\hat{y}=\frac{6}{5}\cdot6+\frac{18}{5}=\frac{54}{5}=10.8\). 因此预测该地区 \(2015\) 年的人民币储蓄存款为 \(10.8\) 千亿元.
\end{enumerate}
\end{solution}

\begin{problem}
已知函数 \(f(x) = \frac{1}{2}\sin 2x - \sqrt{3}\cos^2 x\).

\begin{enumerate}
\item 求 \(f(x)\) 的最小正周期和最小值；
\item 将函数 \(f(x)\) 的图象上每一点的横坐标伸长到原来的两倍，纵坐标不变，得到函数 \(g(x)\) 的图象，当 \(x \in \left[\frac{\pi}{2}, \pi\right]\) 时，求 \(g(x)\) 的值域.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由 \(\cos^2x=\frac{1+\cos2x}{2}\)，得
\[
f(x)=\frac{1}{2}\sin2x-\frac{\sqrt{3}}{2}\cos2x-\frac{\sqrt{3}}{2}=\sin\left(2x-\frac{\pi}{3}\right)-\frac{\sqrt{3}}{2}
\]
因为 \(\sin\left(2x-\frac{\pi}{3}\right)\) 的最小正周期为 \(\pi\)，所以 \(f(x)\) 的最小正周期为 \(\pi\)，且最小值为 \(-1-\frac{\sqrt{3}}{2}=-\frac{2+\sqrt{3}}{2}\).
\item 将图象上每一点的横坐标伸长到原来的两倍，相当于把自变量 \(x\) 换成 \(\frac{x}{2}\)，故 \(g(x)=f\left(\frac{x}{2}\right)\). 从而
\[
g(x)=\frac{1}{2}\sin x-\sqrt{3}\cos^2\frac{x}{2}=\sin\left(x-\frac{\pi}{3}\right)-\frac{\sqrt{3}}{2}
\]
当 \(x\in\left[\frac{\pi}{2},\pi\right]\) 时，\(x-\frac{\pi}{3}\in\left[\frac{\pi}{6},\frac{2\pi}{3}\right]\). 在此区间上，正弦函数的最小值为 \(\frac{1}{2}\)，最大值为 \(1\)，所以 \(g(x)\) 的值域为
\[
\left[\frac{1-\sqrt{3}}{2},1-\frac{\sqrt{3}}{2}\right]
\]
\end{enumerate}
\end{solution}

\begin{problem}
已知函数 \(f(x) = ax^3 + x^2\)（\(a \in \R\)）在 \(x = -\frac{4}{3}\) 处取得极值.

\begin{enumerate}
\item 确定 \(a\) 的值；
\item 若 \(g(x) = f(x)\e^{x}\)，讨论 \(g(x)\) 的单调性.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 因为函数在 \(x=-\frac{4}{3}\) 处取得极值，所以 \(f'\left(-\frac{4}{3}\right)=0\). 又
\[
f'(x)=3ax^2+2x=x(3ax+2)
\]
故
\[
0=f'\left(-\frac{4}{3}\right)=-\frac{4}{3}\left(-4a+2\right)
\]
解得 \(a=\frac{1}{2}\). 此时 \(f'(x)=\frac{1}{2}x(3x+4)\)，在 \(-\frac{4}{3}\) 的两侧符号由正变负，确实取得极大值.
\item 由 \(a=\frac{1}{2}\)，得 \(g(x)=\left(\frac{1}{2}x^3+x^2\right)\e^x\). 因为 \(\e^x>0\)，所以
\[
g'(x)=\e^x\left(f'(x)+f(x)\right)=\frac{1}{2}\e^x x(x+1)(x+4)
\]
临界点为 \(-4\)，\(-1\)，\(0\). 由各因子的符号可知，\(g'(x)<0\) 在 \((-\infty,-4)\) 和 \((-1,0)\) 上成立，\(g'(x)>0\) 在 \((-4,-1)\) 和 \((0,+\infty)\) 上成立. 因此 \(g(x)\) 在 \((-\infty,-4)\)、\((-1,0)\) 上单调递减，在 \((-4,-1)\)、\((0,+\infty)\) 上单调递增.
\end{enumerate}
\end{solution}

\begin{problem}
如图，三棱锥 \(P - ABC\) 中，平面 \(PAC \perp\) 平面 \(ABC\)，\(\angle ABC = \frac{\pi}{2}\)，点 \(D\)，\(E\) 在线段 \(AC\) 上，且 \(AD = DE = EC = 2\)，\(PD = PC = 4\)，点 \(F\) 在线段 \(AB\) 上，且 \(EF \parallel BC\).

\begin{enumerate}
\item 证明：\(AB \perp\) 平面 \(PFE\)；
\item 若四棱锥 \(P - DFBC\) 的体积为 7，求线段 \(BC\) 的长.
\end{enumerate}

\bitmapfigure[max width=0.55\linewidth]{img/2015/chongqing_liberal/q20_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 因为 \(DE=EC\)，且 \(PD=PC\)，所以在等腰三角形 \(PDC\) 中，\(E\) 是底边 \(DC\) 的中点，从而 \(PE\perp DC\)，即 \(PE\perp AC\). 又平面 \(PAC\perp\) 平面 \(ABC\)，且两平面的交线为 \(AC\)，所以由面面垂直的性质，得 \(PE\perp\) 平面 \(ABC\)，从而 \(PE\perp AB\). 另一方面，\(\angle ABC=\frac{\pi}{2}\)，故 \(AB\perp BC\). 由 \(EF\parallel BC\)，得 \(AB\perp EF\). 因为 \(PE\) 与 \(EF\) 是平面 \(PFE\) 内相交的两条直线，且 \(AB\) 同时垂直于它们，所以 \(AB\perp\) 平面 \(PFE\).
\item 由 \(AD=DE=EC=2\) 得 \(AC=6\). 设 \(BC=x\)，则在直角三角形 \(ABC\) 中，
\[
AB=\sqrt{AC^2-BC^2}=\sqrt{36-x^2}
\]
由 \(EF\parallel BC\)，得 \(\triangle AEF\sim\triangle ABC\)，且 \(\frac{AE}{AC}=\frac{4}{6}=\frac{2}{3}\)，所以 \(\frac{AF}{AB}=\frac{2}{3}\). 因此
\[
\frac{S_{\triangle ADF}}{S_{\triangle ABC}}=\frac{AD}{AC}\cdot\frac{AF}{AB}=\frac{1}{3}\cdot\frac{2}{3}=\frac{2}{9}
\]
从而 \(S_{DFBC}=\frac{7}{9}S_{\triangle ABC}\). 在等腰三角形 \(PDC\) 中，\(DE=2\)，\(PD=4\)，所以
\[
PE=\sqrt{PD^2-DE^2}=\sqrt{4^2-2^2}=2\sqrt{3}
\]
由第（1）问，\(PE\) 是四棱锥 \(P-DFBC\) 的高，于是
\[
7=\frac{1}{3}S_{DFBC}\cdot PE=\frac{1}{3}\cdot\frac{7}{9}\cdot\frac{1}{2}x\sqrt{36-x^2}\cdot2\sqrt{3}
\]
化简得 \(x\sqrt{36-x^2}=9\sqrt{3}\). 两边平方，得
\[
x^2(36-x^2)=243
\]
\[
(x^2)^2-36x^2+243=0
\]
所以 \((x^2-9)(x^2-27)=0\). 由于 \(x>0\)，得 \(x=3\) 或 \(x=3\sqrt{3}\). 因此 \(BC=3\) 或 \(BC=3\sqrt{3}\).
\end{enumerate}
\end{solution}

\begin{problem}
如图，椭圆 \(\frac{x^2}{a^2} + \frac{y^2}{b^2} = 1 \) （\(a > b > 0\)）的左、右焦点分别为 \(F_1\)，\(F_2\)，过 \(F_2\) 的直线交椭圆于 \(P\)，\(Q\) 两点，且 \(PQ \perp PF_1\).

\begin{enumerate}
\item 若 \(|PF_1| = 2 + \sqrt{2}\)，\(|PF_2| = 2 - \sqrt{2}\)，求椭圆标准方程；
\item 若 \(|PQ| = \lambda |PF_1|\)，且 \(\frac{3}{4} \leqslant \lambda < \frac{4}{3}\)，试确定椭圆离心率 \(e\) 的取值范围.
\end{enumerate}

\bitmapfigure[max width=0.42\linewidth]{img/2015/chongqing_liberal/q21_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设 \(u=|PF_1|=2+\sqrt{2}\)，\(v=|PF_2|=2-\sqrt{2}\)，焦距参数为 \(c\). 因为 \(P\)，\(F_2\)，\(Q\) 共线，且 \(PQ\perp PF_1\)，所以三角形 \(PF_1F_2\) 在 \(P\) 处是直角三角形. 由椭圆定义，
\[
2a=u+v=4
\]
所以 \(a=2\). 又由勾股定理，
\[
(2c)^2=F_1F_2^2=u^2+v^2=(2+\sqrt{2})^2+(2-\sqrt{2})^2=12
\]
故 \(c=\sqrt{3}\). 因而 \(b^2=a^2-c^2=4-3=1\)，椭圆标准方程为
\[
\frac{x^2}{4}+y^2=1
\]
\item 设 \(u=|PF_1|\)，\(v=|PF_2|\)，并令 \(r=\frac{v}{u}>0\). 由椭圆定义和直角三角形 \(PF_1F_2\) 得
\[
u+v=2a
\]
\[
u^2+v^2=4c^2
\]
于是
\[
uv=\frac{(u+v)^2-(u^2+v^2)}{2}=2b^2
\]
由于 \(e=\frac{c}{a}\)，所以
\[
\frac{r}{(1+r)^2}=\frac{uv}{(u+v)^2}=\frac{b^2}{2a^2}=\frac{1-e^2}{2}
\]
即
\[
e^2=1-\frac{2r}{(1+r)^2}
\]
下面求 \(\lambda\) 与 \(r\) 的关系. 建立直角坐标系，令 \(F_1=(-c,0)\)，\(F_2=(c,0)\)，\(P=(x_0,y_0)\). 由距离公式得
\[
u^2-v^2=4cx_0
\]
在直线 \(F_2P\) 上取参数点 \(F_2+t(P-F_2)\)，其中 \(t=1\) 对应点 \(P\)，设对应点 \(Q\) 的参数为 \(t_Q\). 将参数点代入椭圆方程，得到关于 \(t\) 的方程
\[
\left(\frac{(x_0-c)^2}{a^2}+\frac{y_0^2}{b^2}\right)t^2+\frac{2c(x_0-c)}{a^2}t+\frac{c^2}{a^2}-1=0
\]
其中 \(t=1\) 是点 \(P\) 对应的根，且由 \(P\) 在椭圆上，二次项系数可写为 \(\frac{a^2+c^2-2cx_0}{a^2}\)，常数项为 \(-\frac{b^2}{a^2}\). 由韦达定理，两个根的乘积为
\[
t_Q=-\frac{b^2}{a^2+c^2-2cx_0}
\]
由于焦点 \(F_2\) 在椭圆内部，弦 \(PQ\) 的两个端点位于 \(F_2\) 的两侧，所以 \(t_Q<0\)，从而 \(|PQ|=(1-t_Q)v\). 再由
\[
a=\frac{u+v}{2}
\]
\[
c^2=\frac{u^2+v^2}{4}
\]
\[
2cx_0=\frac{u^2-v^2}{2}
\]
有
\[
a^2+c^2-2cx_0=\frac{v(u+2v)}{2}
\]
\[
a^2+c^2-2cx_0+b^2=2av
\]
因此
\[
|PQ|=v\frac{2av}{v(u+2v)/2}=\frac{2v(u+v)}{u+2v}
\]
从而
\[
\lambda=\frac{|PQ|}{|PF_1|}=\frac{2r(1+r)}{1+2r}
\]
函数 \(\lambda(r)=\frac{2r(1+r)}{1+2r}\) 满足
\[
\lambda'(r)=\frac{2(2r^{2}+2r+1)}{(1+2r)^{2}}>0
\]
因此它在 \(r>0\) 上单调递增，且其在 \(r=\frac{1}{2}\) 和 \(r=1\) 时的值分别为 \(\frac{3}{4}\) 和 \(\frac{4}{3}\). 由题设 \(\frac{3}{4}\leqslant\lambda<\frac{4}{3}\)，得
\[
\frac{1}{2}\leqslant r<1
\]
又
\(\left(1-\frac{2r}{(1+r)^2}\right)'=\frac{2(r-1)}{(1+r)^3}<0\)，\(\left(\frac{1}{2}\leqslant r<1\right)\)
所以在 \(\frac{1}{2}\leqslant r<1\) 上，\(1-\frac{2r}{(1+r)^2}\) 单调递减，从而
\[
\frac{1}{2}<e^2\leqslant\frac{5}{9}
\]
又因为 \(e>0\)，故离心率的取值范围为
\[
\frac{\sqrt{2}}{2}<e\leqslant\frac{\sqrt{5}}{3}
\]
\end{enumerate}
\end{solution}
