\section{解答题}

\begin{problem}
已知等差数列 \(\{a_{n}\}\) 满足 \(a_3 = 2\)，前 3 项和 \(S_3 = \frac{9}{2}\).

\begin{enumerate}
\item 求 \(\{a_{n}\}\) 的通项公式；
\item 设等比数列 \(\{b_{n}\}\) 满足 \(b_1 = a_1\)，\(b_4 = a_{15}\)，求 \(\{b_{n}\}\) 的前 \(n\) 项和 \(T_n\).
\end{enumerate}
\end{problem}

\begin{problem}
随着我国经济的发展，居民的储蓄存款逐年增长. 设某地区城乡居民人民币储蓄存款（年底余额）如下表：

\begin{center}
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{|c|ccccc|}
\hline
年份 & 2010 & 2011 & 2012 & 2013 & 2014 \\
\hline
时间代号 \(t\) & 1 & 2 & 3 & 4 & 5 \\
\hline
储蓄存款 \(y\)（千亿元） & 5 & 6 & 7 & 8 & 10 \\
\hline
\end{tabular}
\end{center}

\begin{enumerate}
\item 求 \(y\) 关于 \(t\) 的回归方程 \(\hat{y} = \hat{b} t + \hat{a}\)；
\item 用所求回归方程预测该地区 2015 年（\(t = 6\)）的人民币储蓄存款. 附：回归方程 \(\hat{y} = \hat{b} t + \hat{a}\) 中，\(\hat{b} = \frac{\sum_{i=1}^{n} t_{i} y_{i} - n \overline{t} \, \overline{y}}{\sum_{i=1}^{n} t_{i}^{2} - n \overline{t}^{2}}\)，\(\hat{a} = \overline{y} - \hat{b} \overline{t}\).
\end{enumerate}
\end{problem}

\begin{problem}
已知函数 \(f(x) = \frac{1}{2}\sin 2x - \sqrt{3}\cos^2 x\).

\begin{enumerate}
\item 求 \(f(x)\) 的最小正周期和最小值；
\item 将函数 \(f(x)\) 的图象上每一点的横坐标伸长到原来的两倍，纵坐标不变，得到函数 \(g(x)\) 的图象，当 \(x \in \left[\frac{\pi}{2}, \pi\right]\) 时，求 \(g(x)\) 的值域.
\end{enumerate}
\end{problem}

\end{problem}
