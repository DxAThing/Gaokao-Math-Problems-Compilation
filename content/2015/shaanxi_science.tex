\begin{problem}
设 \( f_{n}(x) \) 是等比数列 1, \(x\)，\(x^{2}\)，\(\cdots\)，\(x^{n}\) 的各项和，其中 \(x > 0\)，\(n \in \mathbf{N}\)，\( n \geqslant 2 \).

\begin{enumerate}
\item 证明：函数 \(F_{n}(x) = f_{n}(x) - 2\) 在 \(\left(\frac{1}{2}, 1\right)\) 内有且仅有一个零点（记为 \(x_{n}\)），且 \(x_{n} = \frac{1}{2} + \frac{1}{2}x_{n}^{n+1}\)；
\item 设有一个与上述等比数列的首项，末项，项数分别相同的等差数列，其各项和为 \(g_{n}(x)\)，比较 \(f_{n}(x)\) 和 \(g_{n}(x)\) 的大小，并加以证明.
\end{enumerate}
\end{problem}
