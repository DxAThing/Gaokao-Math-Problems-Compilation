\chapter{1962年全国卷}

\begin{problem}
某工厂第三年的产量比第一年的产量增长 \( 21\% \). 问平均每年比上一年增长百分之几？又第一年的产量是第三年的产量的百分之几？（精确到 \( 1\% \)）

\end{problem}

\begin{solution}
设每年平均增长 \( x\% \)，则
\[
\left( 1+\frac{x}{100} \right)^2=\frac{121}{100}
\]
所以 \( 1+\frac{x}{100}=\frac{11}{10} \)，\( x=10 \).

答：每年平均增长 \( 10\% \).

第一年产量是第三年的 \( \frac{100}{121}\approx 0.83 \)，所以约为 \( 83\% \).
\end{solution}

\begin{problem}
求 \( \left( 1-2\i \right)^5 \) 的实部.

\end{problem}

\begin{solution}
因为
\[
\left( 1-2\i \right)^5=1-\mathrm C_5^1 2\i+\mathrm C_5^2 \left( 2\i \right)^2-\mathrm C_5^3 \left( 2\i \right)^3+\mathrm C_5^4 \left( 2\i \right)^4-\left( 2\i \right)^5
\]
所以 \( \left( 1-2\i \right)^5 \) 的实部是 \( 1+10\left( 2\i \right)^2+5\left( 2\i \right)^4=1-40+80=41 \).
\end{solution}

\begin{problem}
解方程：\( \log \left( x-5 \right)+\log \left( x+3 \right)-2\log 2=\log \left( 2x-9 \right) \).

\end{problem}

\begin{solution}
原方程可以写成
\[
\log \frac{\left( x-5 \right)\left( x+3 \right)}{4}=\log \left( 2x-9 \right)
\]
或者 \( \left( x-5 \right)\left( x+3 \right)=4\left( 2x-9 \right) \). 化简得 \( x^2-10x+21=0 \)，即 \( \left( x-3 \right)\left( x-7 \right)=0 \). 所以 \( x_1=3 \)，\( x_2=7 \). 当 \( x=3 \) 时，\( x-5<0 \)，而 \( \log \left( x-5 \right) \) 无意义，所以 \( 3 \) 不是原方程的解. 经检验，\( x=7 \) 是原方程的解.
\end{solution}

\begin{problem}
求 \( \sin \left( 2\arcsin \frac{4}{5} \right) \) 的值.

\end{problem}

\begin{solution}
设 \( \arcsin \frac{4}{5}=a \)（\( 0<a<\frac{\pi}{2} \)），则 \( \sin a=\frac{4}{5} \)，\( \cos a=\sqrt{1-\left( \frac{4}{5} \right)^2}=\frac{3}{5} \).

所以
\begin{align*}
\sin \left( 2\arcsin \frac{4}{5} \right)
&=\sin 2a\\
&=2\sin a\cos a\\
&=2\cdot \frac{4}{5}\cdot \frac{3}{5}\\
&=\frac{24}{25}
\end{align*}
\end{solution}

\begin{problem}
求证：（1）圆内接平行四边形是矩形；（2）圆外切平行四边形是菱形.

\end{problem}

\begin{solution}
（1）设 \( ABCD \) 为圆内接平行四边形. 因为平行四边形的对角相等，所以 \( \angle A=\angle C \). 又因圆内接四边形的对角互补，得 \( \angle A+\angle C=180^\circ \)，所以 \( \angle A=\angle C=90^\circ \). 因此 \( ABCD \) 是矩形.

\bitmapfigure[max width=3.0cm,max totalheight=3.0cm]{img/1962/national/q5_solution_fig1.png}

（2）设平行四边形 \( ABCD \) 切圆于 \( E \)、\( F \)、\( G \)、\( H \). 因从圆外一点所作的两条切线等长，得
\[
\begin{cases}
AE=AH,\\
BE=BF,\\
CG=CF,\\
DG=DH.
\end{cases}
\]

\bitmapfigure[max width=3.0cm,max totalheight=3.0cm]{img/1962/national/q5_solution_fig2.png}

四式相加，得 \( AB+CD=AD+BC \). 又因平行四边形的对边相等，得 \( AB=CD \)，\( AD=BC \). 所以 \( AB=BC \). 因此 \( ABCD \) 是菱形.
\end{solution}

\begin{problem}
解方程组：\( y^2-4x-2y+1=0 \)，\( y=x+a \). 并讨论：\( a \) 取哪些实数值时，这个方程组（1）有不同的两组实数解；（2）有相同的两组实数解；（3）没有实数解.

\end{problem}

\begin{solution}
由 \( y=x+a \)，得 \( x=y-a \). 代入 \( y^2-4x-2y+1=0 \)，得 \( y^2-4\left( y-a \right)-2y+1=0 \)，即 \( y^2-6y+4a+1=0 \).

所以
\[
y=3\pm \sqrt{9-\left( 4a+1 \right)}=3\pm 2\sqrt{2-a}
\]
代入 \( x=y-a \)，得
\[
x=3-a\pm 2\sqrt{2-a}
\]
方程组的解为
\[
\begin{cases}
x_1=3-a+2\sqrt{2-a},\\
y_1=3+2\sqrt{2-a},
\end{cases}
\]
\[
\begin{cases}
x_2=3-a-2\sqrt{2-a},\\
y_2=3-2\sqrt{2-a}.
\end{cases}
\]

讨论：

（1）当 \( a<2 \) 时，方程组有不同的两组实数解.

（2）当 \( a=2 \) 时，方程组有相同的两组实数解.

（3）当 \( a>2 \) 时，方程组没有实数解.
\end{solution}

\begin{problem}
\( ABCD \) 和 \( A'B'C'D' \) 都是正方形，而 \( A' \)、\( B' \)、\( C' \)、\( D' \) 顺次分 \( AB \)、\( BC \)、\( CD \)、\( DA \) 成 \( m:n \)，并设 \( AB=1 \). （1）求正方形 \( A'B'C'D' \) 的面积；（2）证明：正方形 \( A'B'C'D' \) 的面积不小于 \( \frac{1}{2} \).

\bitmapfigure[max width=3.4cm,max totalheight=3.4cm]{img/1962/national/q7_fig1.png}

\end{problem}

\begin{solution}
（1）由于 \( AB=1 \)，\( AA':A'B=m:n \)，得 \( AA'=\frac{m}{m+n} \)，\( A'B=\frac{n}{m+n} \).

又知 \( \triangle BA'B' \) 为直角三角形，所以
\begin{align*}
A'B'
&=\sqrt{A'B^2+B'B^2}\\
&=\sqrt{A'B^2+A'A^2}\\
&=\sqrt{\frac{m^2+n^2}{\left( m+n \right)^2}}
\end{align*}
故正方形 \( A'B'C'D' \) 的面积为
\[
A'B'^2=\frac{m^2+n^2}{\left( m+n \right)^2}
\]

（2）因为
\begin{align*}
\frac{m^2+n^2}{\left( m+n \right)^2}-\frac{1}{2}
&=\frac{2\left( m^2+n^2 \right)-\left( m+n \right)^2}{2\left( m+n \right)^2}\\
&=\frac{\left( m-n \right)^2}{2\left( m+n \right)^2}\\
&\ge 0
\end{align*}
所以 \( \frac{m^2+n^2}{\left( m+n \right)^2}\ge \frac{1}{2} \). 因此正方形 \( A'B'C'D' \) 的面积不小于 \( \frac{1}{2} \).
\end{solution}

\begin{problem}
\( D \) 是 \( \triangle ABC \) 内的一点，已知 \( AB=AC=1 \)，\( \angle CAB=63^\circ \)，\( \angle DAB=33^\circ \)，\( \angle DBA=27^\circ \)，求 \( CD \). （\( \sin 27^\circ=0.4540 \). 最后结果计算到小数点后两位.）

\end{problem}

\begin{solution}
在 \( \triangle ABD \) 内，\( \angle BDA=180^\circ-33^\circ-27^\circ=120^\circ \). 故根据正弦定理，
\[
\frac{AD}{\sin 27^\circ}=\frac{1}{\sin 120^\circ}
\]
所以 \( AD=\frac{2}{\sqrt{3}}\sin 27^\circ \).

\bitmapfigure[max width=3.0cm,max totalheight=3.0cm]{img/1962/national/q8_solution_fig1.png}

又 \( \angle DAC=63^\circ-33^\circ=30^\circ \). 故根据余弦定理，在 \( \triangle ADC \) 内，
\begin{align*}
CD^2
&=1+AD^2-2AD\cos 30^\circ\\
&=1+\frac{4}{3}\sin^2 27^\circ-2\sin 27^\circ\\
&=1+2\sin 27^\circ \left( \frac{2}{3}\sin 27^\circ-1 \right)\\
&=1+0.9080\times \left( 0.3027-1 \right)\\
&=1-0.9080\times 0.6973\\
&=1-0.6331\\
&=0.3669
\end{align*}
所以 \( CD=0.60 \).
\end{solution}

\begin{problem}
由正方体 \( ABCD-A_1B_1C_1D_1 \) 的顶点 \( A \) 作这个正方体的对角线 \( A_1C \) 的垂线，垂足为 \( E \). 证明：\( A_1E:EC=1:2 \). （要求画图）

\end{problem}

\begin{solution}
作 \( AC \).

\bitmapfigure[max width=4.0cm,max totalheight=4.0cm]{img/1962/national/q9_fig1.png}

则 \( A_1A\perp AC \). 故 \( \triangle A_1AC \) 为直角三角形. 设正方体的棱长为 \( 1 \)，则 \( AC=\sqrt{2} \).

在直角三角形 \( A_1AC \) 中，\( AE\perp A_1C \)，所以 \( A_1A^2=A_1E\cdot A_1C \)，即 \( 1=A_1E\cdot A_1C \). 同理 \( AC^2=EC\cdot A_1C \)，即 \( 2=EC\cdot A_1C \). 由此得 \( A_1E:EC=1:2 \).
\end{solution}

\begin{problem}
求证：两两相交而不通过同一点的四条直线必在同一平面内.

\end{problem}

\begin{solution}
在这四条直线中，任取两条直线 \( a \)、\( b \)，设其交点为 \( P \). 因为四条直线不通过同一点，所以在另外两条直线中，至少有一条直线 \( c \) 不通过 \( P \)，直线 \( c \) 必与 \( a \)、\( b \) 分别交于不同的两点. 因此，\( c \) 必在 \( a \)、\( b \) 所决定的平面内.

第四条直线 \( d \) 与 \( a \)、\( b \)、\( c \) 至少交于两个不同的点，所以 \( d \) 也在 \( a \)、\( b \) 所决定的平面内.
\end{solution}
