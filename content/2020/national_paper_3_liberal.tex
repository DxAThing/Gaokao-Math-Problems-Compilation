\begin{problem}
设等比数列 \(\{a_n\}\) 满足 \(a_1+a_2=4\)，\(a_3-a_1=8\).

\begin{enumerate}
    \item 求 \(\{a_n\}\) 的通项公式；
    \item 记 \(S_n\) 为数列 \(\{\log_3 a_n\}\) 的前 \(n\) 项和. 若 \(S_m+S_{m+1}=S_{m+3}\)，求 \(m\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
    \item 设等比数列 \(\{a_n\}\) 的首项为 \(a_1\)，公比为 \(q\). 由题设得 \(a_1(1+q)=4\)，\(a_1(q^2-1)=a_1(q-1)(q+1)=8\). 两式相除得 \(q-1=2\)，所以 \(q=3\). 代入 \(a_1(1+3)=4\) 得 \(a_1=1\). 故 \(\{a_n\}\) 的通项公式为 \(a_n=3^{n-1}\).
    \item 由（1）知 \(\log_3 a_n=\log_3 3^{n-1}=n-1\)，所以数列 \(\{\log_3 a_n\}\) 的前 \(n\) 项和为 \(S_n=\dfrac{n(n-1)}{2}\). 由 \(S_m+S_{m+1}=S_{m+3}\) 得 \(\dfrac{m(m-1)}{2}+\dfrac{(m+1)m}{2}=\dfrac{(m+3)(m+2)}{2}\)，整理得 \(m^2-5m-6=0\)，解得 \(m=6\) 或 \(m=-1\)（舍去）. 故 \(m=6\).
\end{enumerate}
\end{solution}

\begin{problem}
已知函数 \(f(x)=x^3-kx+k^2\).

\begin{enumerate}
    \item 讨论 \(f(x)\) 的单调性；
    \item 若 \(f(x)\) 有三个零点，求 \(k\) 的取值范围.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
    \item \(f'(x)=3x^2-k\). 当 \(k\le 0\) 时，对任意 \(x\in\R\)，\(f'(x)=3x^2-k\ge0\)，所以 \(f(x)\) 在 \(\R\) 上单调递增；当 \(k>0\) 时，令 \(f'(x)=0\) 得 \(x=\pm\dfrac{\sqrt{3k}}{3}\). 当 \(x\in\left(-\infty,-\dfrac{\sqrt{3k}}{3}\right)\) 和 \(x\in\left(\dfrac{\sqrt{3k}}{3},+\infty\right)\) 时，\(f'(x)>0\)，\(f(x)\) 单调递增；当 \(x\in\left(-\dfrac{\sqrt{3k}}{3},\dfrac{\sqrt{3k}}{3}\right)\) 时，\(f'(x)<0\)，\(f(x)\) 单调递减. 综上，当 \(k\le 0\) 时，\(f(x)\) 在 \(\R\) 上单调递增；当 \(k>0\) 时，\(f(x)\) 在 \(\left(-\infty,-\dfrac{\sqrt{3k}}{3}\right)\) 和 \(\left(\dfrac{\sqrt{3k}}{3},+\infty\right)\) 上单调递增，在 \(\left(-\dfrac{\sqrt{3k}}{3},\dfrac{\sqrt{3k}}{3}\right)\) 上单调递减.
    \item 若 \(f(x)\) 有三个零点，由（1）知必有 \(k>0\)，且极大值与极小值异号. 此时极大值为 \(f\left(-\dfrac{\sqrt{3k}}{3}\right)=\dfrac{2k\sqrt{3k}}{9}+k^2>0\)，极小值为 \(f\left(\dfrac{\sqrt{3k}}{3}\right)=k^2-\dfrac{2k\sqrt{3k}}{9}\). 所以只需 \(k^2-\dfrac{2k\sqrt{3k}}{9}<0\). 因为 \(k>0\)，所以 \(k<\dfrac{2\sqrt{3k}}{9}\)，两边平方得 \(k^2<\dfrac{4k}{27}\)，即 \(k<\dfrac{4}{27}\). 故 \(k\) 的取值范围为 \(\left(0,\dfrac{4}{27}\right)\).
\end{enumerate}
\end{solution}

\begin{problem}
已知椭圆 \(C:\dfrac{x^2}{25}+\dfrac{y^2}{m^2}=1\)（\(0<m<5\)）的离心率为 \(\dfrac{\sqrt{15}}{4}\)，\(A\)，\(B\) 分别为 \(C\) 的左，右顶点.

\begin{enumerate}
    \item 求 \(C\) 的方程；
    \item 若点 \(P\) 在 \(C\) 上，点 \(Q\) 在直线 \(x=6\) 上，且 \(|BP|=|BQ|\)，\(BP\perp BQ\)，求 \(\triangle APQ\) 的面积.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
    \item 由题设知 \(a=5\)，\(b=m\). 离心率 \(e=\dfrac ca=\dfrac{\sqrt{15}}{4}\)，所以 \(c=\dfrac{5\sqrt{15}}{4}\)，\(m^2=b^2=a^2-c^2=25-\dfrac{375}{16}=\dfrac{25}{16}\). 故 \(C\) 的方程为 \(\dfrac{x^2}{25}+\dfrac{16y^2}{25}=1\).
    \item 由（1）知 \(A(-5,0)\)，\(B(5,0)\). 设 \(Q(6,t)\)，则 \(\overrightarrow{BQ}=(1,t)\). 因为 \(BP\perp BQ\) 且 \(|BP|=|BQ|\)，所以 \(\overrightarrow{BP}=(-t,1)\) 或 \(\overrightarrow{BP}=(t,-1)\). 由对称性，不妨设 \(\overrightarrow{BP}=(-t,1)\)，则 \(P(5-t,1)\). 将 \(P\) 代入椭圆方程得 \(\dfrac{(5-t)^2}{25}+\dfrac{16}{25}=1\)，即 \((5-t)^2=9\)，解得 \(t=2\) 或 \(t=8\). 当 \(t=2\) 时，\(P(3,1)\)，\(Q(6,2)\)，\(\overrightarrow{AP}=(8,1)\)，\(\overrightarrow{AQ}=(11,2)\)，\(\triangle APQ\) 的面积为 \(\dfrac12|8\times2-11\times1|=\dfrac52\)；当 \(t=8\) 时，\(P(-3,1)\)，\(Q(6,8)\)，\(\overrightarrow{AP}=(2,1)\)，\(\overrightarrow{AQ}=(11,8)\)，\(\triangle APQ\) 的面积为 \(\dfrac12|2\times8-11\times1|=\dfrac52\). 同理，当 \(\overrightarrow{BP}=(t,-1)\) 时，面积也为 \(\dfrac52\). 综上，\(\triangle APQ\) 的面积为 \(\dfrac52\).
\end{enumerate}
\end{solution}

\begin{problem}
在直角坐标系 \(xOy\) 中，曲线 \(C\) 的参数方程为
\(\begin{cases}
x=2-t-t^2,\\
y=2-3t+t^2
\end{cases}\)
（\(t\) 为参数且 \(t\neq 1\)），\(C\) 与坐标轴交于 \(A\)，\(B\) 两点.

\begin{enumerate}
    \item 求 \(|AB|\)；
    \item 以坐标原点为极点，\(x\) 轴正半轴为极轴建立极坐标系，求直线 \(AB\) 的极坐标方程.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
    \item 令 \(y=0\)，得 \(t^2-3t+2=0\)，由 \(t\ne1\) 得 \(t=2\)，代入得 \(x=-4\)，所以 \(A(-4,0)\). 令 \(x=0\)，得 \(t^2+t-2=0\)，由 \(t\ne1\) 得 \(t=-2\)，代入得 \(y=12\)，所以 \(B(0,12)\). 于是 \(|AB|=\sqrt{(-4)^2+(-12)^2}=4\sqrt{10}\).
    \item 由（1）知直线 \(AB\) 经过点 \(A(-4,0)\) 和 \(B(0,12)\)，其直角坐标方程为 \(\dfrac{x}{-4}+\dfrac{y}{12}=1\)，即 \(3x-y+12=0\). 代入 \(x=\rho\cos\theta\)，\(y=\rho\sin\theta\)，得直线 \(AB\) 的极坐标方程为 \(3\rho\cos\theta-\rho\sin\theta+12=0\).
\end{enumerate}
\end{solution}

\begin{problem}
设 \(a\)，\(b\)，\(c\in \R\)，\(a+b+c=0\)，\(abc=1\).

\begin{enumerate}
    \item 证明：\(ab+bc+ca<0\)；
    \item 用 \(\max\{a,b,c\}\) 表示 \(a\)，\(b\)，\(c\) 中的最大值，证明：\(\max\{a,b,c\}\ge \sqrt[3]{4}\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
    \item 由 \(a+b+c=0\) 两边平方得 \(a^2+b^2+c^2+2(ab+bc+ca)=0\)，所以 \(ab+bc+ca=-\dfrac12(a^2+b^2+c^2)\). 因为 \(abc=1\ne0\)，所以 \(a\)，\(b\)，\(c\) 均不为 \(0\)，故 \(a^2+b^2+c^2>0\). 从而 \(ab+bc+ca<0\).
    \item 因为 \(abc=1>0\) 且 \(a+b+c=0\)，所以 \(a\)，\(b\)，\(c\) 必为一正两负. 不妨设 \(a>0\)，\(b<0\)，\(c<0\)，则 \(\max\{a,b,c\}=a\). 由 \(a+b+c=0\) 得 \((-b)+(-c)=a\)，其中 \(-b>0\)，\(-c>0\). 由基本不等式，\(a=(-b)+(-c)\ge2\sqrt{(-b)(-c)}=2\sqrt{bc}=2\sqrt{\dfrac1a}=\dfrac{2}{\sqrt a}\). 两边平方得 \(a^2\ge\dfrac4a\)，即 \(a^3\ge4\)，所以 \(a\ge\sqrt[3]4\). 故 \(\max\{a,b,c\}\ge\sqrt[3]4\).
\end{enumerate}
\end{solution}
