\begin{problem}
已知函数 \( f(x) = (a + 2\cos^2 x)\cos (2x + \theta) \) 为奇函数，且 \( f\left(\frac{\pi}{4}\right) = 0 \)，其中 \(a \in \R\)，\(\theta \in (0, \pi)\).
\begin{enumerate}
\item 求 \(a\)，\(\theta\) 的值；
\item 若 \(f\left(\frac{\alpha}{4}\right) = -\frac{2}{5}\)，\(\alpha \in \left(\frac{\pi}{2}, \pi\right)\)，求 \( \sin \left(\alpha + \frac{\pi}{3}\right) \) 的值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 因为 \(f(x)\) 是奇函数，所以对任意 \(x\) 都有 \(f(x)+f(-x)=0\). 由 \(\cos(2x+\theta)+\cos(\theta-2x)=2\cos\theta\cos2x\)，得 \(f(x)+f(-x)=2(a+2\cos^2x)\cos\theta\cos2x\). 因为 \((a+2\cos^2x)\cos2x\) 不恒为 \(0\)，所以 \(\cos\theta=0\). 又 \(\theta\in(0,\pi)\)，故 \(\theta=\frac{\pi}{2}\). 此时 \(f\left(\frac{\pi}{4}\right)=(a+1)\cos\pi=-(a+1)=0\)，所以 \(a=-1\). 因而 \(f(x)=(-1+2\cos^2x)\cos\left(2x+\frac{\pi}{2}\right)=-\cos2x\sin2x=-\frac{1}{2}\sin4x\).
\item 由上面的函数表达式以及题设，得 \(-\frac{1}{2}\sin\alpha=f\left(\frac{\alpha}{4}\right)=-\frac{2}{5}\)，所以 \(\sin\alpha=\frac{4}{5}\). 因为 \(\alpha\in\left(\frac{\pi}{2},\pi\right)\)，故 \(\cos\alpha=-\frac{3}{5}\). 因此 \(\sin\left(\alpha+\frac{\pi}{3}\right)=\sin\alpha\cos\frac{\pi}{3}+\cos\alpha\sin\frac{\pi}{3}=\frac{4}{5}\cdot\frac{1}{2}-\frac{3}{5}\cdot\frac{\sqrt{3}}{2}=\frac{4-3\sqrt{3}}{10}\).
\end{enumerate}
\end{solution}

\begin{problem}
已知数列 \(\{a_{n}\}\) 的前 \(n\) 项和 \(S_{n} = \frac{3n^{2} - n}{2}\)，\(n \in \N^*\).
\begin{enumerate}
\item 求数列 \( \{a_{n}\} \) 的通项公式；
\item 证明：对任意 \(n > 1\)，都有 \(m \in \N^*\)，使得 \(a_1\)，\(a_n\)，\(a_m\) 成等比数列.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 当 \(n\geqslant2\) 时，\(a_n=S_n-S_{n-1}=\frac{3n^2-n-\left(3(n-1)^2-(n-1)\right)}{2}=3n-2\). 又 \(a_1=S_1=1=3\times1-2\)，所以对任意 \(n\in\N^*\)，都有 \(a_n=3n-2\).
\item 对任意 \(n>1\)，取 \(m=3n^2-4n+2\). 因为 \(n\geqslant2\)，所以 \(m=n(3n-4)+2\in\N^*\). 由通项公式，\(a_1=1\)，\(a_n=3n-2\)，且 \(a_m=3m-2=9n^2-12n+4=(3n-2)^2\). 从而 \(a_n^2=a_1a_m\)，即 \(a_1\)，\(a_n\)，\(a_m\) 成等比数列.
\end{enumerate}
\end{solution}

\begin{problem}
已知函数 \( f(x) = (4x^{2} + 4ax + a^{2})\sqrt{x} \)，其中 \( a < 0 \).
\begin{enumerate}
\item 当 \(a = -4\) 时，求 \(f(x)\) 的单调递增区间；
\item 若 \( f(x) \) 在区间 \( [1,4] \) 上的最小值为 8，求 \(a\) 的值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 当 \(a=-4\) 时，\(f(x)=(2x-4)^2\sqrt{x}\)，定义域为 \(x\geqslant0\). 当 \(x>0\) 时，\(f'(x)=\frac{20x^2-48x+16}{2\sqrt{x}}=\frac{2(5x-2)(x-2)}{\sqrt{x}}\). 因此在 \(0<x<\frac{2}{5}\) 时 \(f'(x)>0\)，在 \(\frac{2}{5}<x<2\) 时 \(f'(x)<0\)，在 \(x>2\) 时 \(f'(x)>0\). 又 \(f(0)=0\)，所以 \(f(x)\) 的单调递增区间为 \(\left[0,\frac{2}{5}\right]\) 和 \([2,+\infty)\).
\item 将函数写成 \(f(x)=(2x+a)^2\sqrt{x}\). 当 \(-8\leqslant a\leqslant-2\) 时，\(x=-\frac{a}{2}\in[1,4]\)，从而 \(f\left(-\frac{a}{2}\right)=0\)，最小值不可能为 \(8\). 当 \(-2<a<0\) 时，\(f'(x)=\frac{q(x)}{2\sqrt{x}}\)，其中 \(q(x)=20x^2+12ax+a^2\). 此时 \(q(1)=(a+2)(a+10)>0\)，且 \(q'(x)=40x+12a>0\)（\(x\geqslant1\)），所以 \(f\) 在 \([1,4]\) 上递增，最小值为 \(f(1)=(a+2)^2<4\)，也不符合题意.

当 \(-10<a<-8\) 时，\(q(x)=20\left(x+\frac{a}{10}\right)\left(x+\frac{a}{2}\right)\)，且 \(-\frac{a}{10}<1<4<-\frac{a}{2}\)，故 \(f\) 在 \([1,4]\) 上递减，最小值为 \(f(4)=2(a+8)^2<8\). 当 \(a=-10\) 时，\(f\) 在 \([1,4]\) 上递减，且 \(f(4)=2(8-10)^2=8\). 当 \(a<-10\) 时，\(f\) 在区间内部的驻点 \(x=-\frac{a}{10}\) 处由增转减，故最小值只能在端点处取得；但 \(f(1)=(a+2)^2>64\)，\(f(4)=2(a+8)^2>8\)，不可能等于 \(8\). 综上，\(a=-10\).
\end{enumerate}
\end{solution}

\begin{problem}
如图，三棱柱 \(ABC - A_{1}B_{1}C_{1}\) 中，\(AA_{1} \perp BC\)，\(A_{1}B \perp BB_{1}\).
\begin{enumerate}
\item 求证：\(A_{1}C \perp CC_{1}\)；
\item 若 \(AB = 2\)，\(AC = \sqrt{3}\)，\(BC = \sqrt{7}\)，问 \(AA_{1}\) 为何值时，三棱柱 \(ABC-A_{1}B_{1}C_{1}\) 体积最大，并求此最大值.
\end{enumerate}
\bitmapfigure[max width=0.72\linewidth,float=inline]{img/2014/jiangxi_liberal/q19_fig1.png}
\FloatBarrier
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设 \(\bs{v}=\overrightarrow{AA_1}=\overrightarrow{BB_1}=\overrightarrow{CC_1}\)，\(\bs{b}=\overrightarrow{AB}\)，\(\bs{c}=\overrightarrow{AC}\). 则 \(\overrightarrow{BC}=\bs{c}-\bs{b}\)，\(\overrightarrow{A_1B}=\bs{b}-\bs{v}\). 由 \(AA_1\perp BC\) 和 \(A_1B\perp BB_1\)，分别得到 \(\bs{v}\mathbin{\cdot}(\bs{c}-\bs{b})=0\) 和 \((\bs{b}-\bs{v})\mathbin{\cdot}\bs{v}=0\). 因而 \(\bs{v}\mathbin{\cdot}\bs{c}=\bs{v}\mathbin{\cdot}\bs{b}=|\bs{v}|^2\). 又 \(\overrightarrow{A_1C}=\bs{c}-\bs{v}\)，\(\overrightarrow{CC_1}=\bs{v}\)，所以 \(\overrightarrow{A_1C}\mathbin{\cdot}\overrightarrow{CC_1}=(\bs{c}-\bs{v})\mathbin{\cdot}\bs{v}=0\)，即 \(A_1C\perp CC_1\).
\item 因为 \(AB^2+AC^2=2^2+(\sqrt{3})^2=7=BC^2\)，所以 \(\triangle ABC\) 在 \(A\) 处为直角三角形，底面积为 \(\frac{1}{2}AB\cdot AC=\sqrt{3}\). 取 \(A=(0,0,0)\)，\(B=(2,0,0)\)，\(C=(0,\sqrt{3},0)\)，设 \(\bs{v}=(r,s,t)\)，并记 \(h=AA_1=|\bs{v}|\). 由 \(AA_1\perp BC\)，有 \((r,s,t)\mathbin{\cdot}(-2,\sqrt{3},0)=0\)，即 \(s=\frac{2r}{\sqrt{3}}\). 由 \(A_1B\perp BB_1\)，有 \((2-r,-s,-t)\mathbin{\cdot}(r,s,t)=0\)，即 \(2r=h^2\). 因而 \(r=\frac{h^2}{2}\)，\(s=\frac{h^2}{\sqrt{3}}\)，从而 \(t^2=h^2-r^2-s^2=h^2-\frac{7h^4}{12}=h^2\left(1-\frac{7h^2}{12}\right)\). 三棱柱两底面间的距离为 \(|t|\)，所以体积 \(V=\sqrt{3}|t|\)，进而 \(V^2=3h^2-\frac{7}{4}h^4\). 令 \(u=h^2\)，则 \(V^2=3u-\frac{7}{4}u^2=\frac{9}{7}-\frac{7}{4}\left(u-\frac{6}{7}\right)^2\leqslant\frac{9}{7}\). 当且仅当 \(u=\frac{6}{7}\)，即 \(AA_1=\sqrt{\frac{6}{7}}\) 时取等号，此时 \(V_{\max}=\frac{3}{\sqrt{7}}\).
\end{enumerate}
\end{solution}

\begin{problem}
如图，已知抛物线 \(C: x^2 = 4y\)，过点 \(M(0,2)\) 任作一直线与 \(C\) 相交于 \(A\)，\(B\) 两点，过点 \(B\) 作 \(y\) 轴的平行线与直线 \(AO\) 相交于点 \(D\) （\(O\) 为坐标原点）.
\begin{enumerate}
\item 证明：动点 \(D\) 在定直线上；
\item 作 \(C\) 的任意一条切线 \(l\)（不含 \(x\) 轴）与直线 \(y = 2\) 相交于点 \(N_{1}\)；与第一问中的定直线相交于点 \(N_{2}\)，证明：\(|MN_{2}|^{2} - |MN_{1}|^{2}\) 为定值，并求此定值.

\bitmapfigure[max width=0.62\linewidth,float=inline]{img/2014/jiangxi_liberal/q20_fig1.png}
\end{problem}


\begin{solution}
\begin{enumerate}
\item 设抛物线上的点用参数表示为 \(P(t)=(2t,t^2)\)，并设 \(A=P(u)\)，\(B=P(v)\). 过 \(A\)，\(B\) 的直线方程为 \(y=\frac{u+v}{2}x-uv\). 因为该直线过 \(M(0,2)\)，所以 \(-uv=2\)，即 \(uv=-2\). 直线 \(AO\) 的方程为 \(y=\frac{u}{2}x\)，过 \(B\) 作的竖线为 \(x=2v\)，故 \(D=(2v,uv)=(2v,-2)\). 因此动点 \(D\) 在定直线 \(y=-2\) 上.

\end{enumerate}
\end{solution}

\begin{problem}
将连续正整数 \(1\)，\(2\)，\(\cdots\)，\(n\) （\(n \in \N^*\)） 从小到大排列构成一个数 \(123\cdots n\)，\(F(n)\) 为这个数的位数 （如 \(n = 12\) 时，此数为 \(123456789101112\)，共有 15 个数字，\(F(12) = 15\)），现从这个数中随机取一个数字，\(p(n)\) 为恰好取到 0 的概率.
\begin{enumerate}
\item 求 \(p(100)\)；
\item 当 \(n \leqslant 2014\) 时，求 \(F(n)\) 的表达式；
\item 令 \(g(n)\) 为这个数中数字 0 的个数，\(f(n)\) 为这个数中数字 9 的个数，\(h(n) = f(n) - g(n)\)，\(S = \{n \mid h(n) = 1, n \leqslant 100, n \in \N^*\}\)，求当 \(n \in S\) 时 \(p(n)\) 的最大值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 从 \(1\) 到 \(100\) 的连接数中，数字 \(0\) 出现在 \(10,20,\ldots,90\) 中各一次，在 \(100\) 中出现两次，所以共有 \(9+2=11\) 个数字 \(0\). 又 \(F(100)=9+90\times2+3=192\)，故 \(p(100)=\frac{11}{192}\).
\item 当 \(1\leqslant n\leqslant9\) 时，全部是 \(1\) 位数，所以 \(F(n)=n\). 当 \(10\leqslant n\leqslant99\) 时，前面有 \(9\) 个 \(1\) 位数和 \(n-9\) 个 \(2\) 位数，所以 \(F(n)=9+2(n-9)=2n-9\). 当 \(100\leqslant n\leqslant999\) 时，前面共有 \(9+90\times2=189\) 位，另有 \(n-99\) 个 \(3\) 位数，所以 \(F(n)=189+3(n-99)=3n-108\). 当 \(1000\leqslant n\leqslant2014\) 时，前面共有 \(9+90\times2+900\times3=2889\) 位，另有 \(n-999\) 个 \(4\) 位数，所以 \(F(n)=2889+4(n-999)=4n-1107\). 因而

\(F(n)=\begin{cases}
n,&1\leqslant n\leqslant9,\\
2n-9,&10\leqslant n\leqslant99,\\
3n-108,&100\leqslant n\leqslant999,\\
4n-1107,&1000\leqslant n\leqslant2014.
\end{cases}\)
\item 先看 \(n\leqslant9\). 只有 \(n=9\) 时 \(h(n)=1\). 对 \(10\leqslant n\leqslant99\)，写成 \(n=10k+r\)，其中 \(1\leqslant k\leqslant9\)，\(0\leqslant r\leqslant9\). 从 \(10\) 到 \(10k-1\) 的每个完整十位组中，个位数字 \(0\) 和 \(9\) 各出现一次，贡献相互抵消；当 \(k\leqslant8\) 时，当前十位数字不是 \(9\)，所以 \(r\leqslant8\) 时当前组个位的 \(0\) 使 \(h(10k+r)=0\)，而 \(r=9\) 时再增加一个 \(9\)，使 \(h(10k+9)=1\). 当 \(k=9\) 时，数字 \(9\) 作为十位数字出现 \(r+1\) 次，因此 \(h(90+r)=r+1\)（\(0\leqslant r\leqslant8\)），而 \(h(99)=11\). 于是 \(S=\{9,19,29,39,49,59,69,79,89,90\}\).

对 \(n=10k+9\)（\(1\leqslant k\leqslant8\)），数字 \(0\) 恰在 \(10,20,\ldots,10k\) 中出现，所以 \(g(n)=k\)，且 \(F(n)=20k+9\)，从而 \(p(n)=\frac{k}{20k+9}\). 因为

\(\frac{k+1}{20k+29}-\frac{k}{20k+9}=\frac{9}{(20k+29)(20k+9)}>0\)，所以这些数中的最大值为 \(p(89)=\frac{8}{169}\). 另有 \(g(90)=9\)，\(F(90)=171\)，故 \(p(90)=\frac{9}{171}=\frac{1}{19}>\frac{8}{169}\)，而 \(p(9)=0\). 因此当 \(n\in S\) 时，\(p(n)\) 的最大值为 \(\frac{1}{19}\).
