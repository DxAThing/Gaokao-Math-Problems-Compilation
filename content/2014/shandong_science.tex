\section{单选题}

\begin{problem}
已知 \(a, b \in \R\)，i 是虚数单位. 若 \(a - \i\) 与 \(2 + bi\) 互为共轭复数，则 \((a + bi)^2 =\)（\quad）

\choices
    {\(5 - 4\i\)}
    {\(5 + 4\i\)}
    {\(3 - 4\i\)}
    {\(3 + 4\i\)}
\end{problem}

\begin{problem}
设集合 \(A = \{x \mid |x - 1| < 2\}\)，\(B = \{y \mid y = 2^x, x \in [0,2]\}\)，则 \(A \cap B =\)（\quad）

\choices
    {\([0,2]\)}
    {\((1,3)\)}
    {\([1,3]\)}
    {\((1,4)\)}
\end{problem}

\begin{problem}
函数 \( f(x) = \frac{1}{\sqrt{(\log_2x)^2 - 1}} \) 的定义域为（\quad）

\choices
    {\(\left(0, \frac{1}{2}\right)\)}
    {\((2, +\infty)\)}
    {\(\left(0, \frac{1}{2}\right) \cup (2, +\infty)\)}
    {\(\left(0, \frac{1}{2}\right] \cup [2, +\infty)\)}
\end{problem}

\begin{problem}
用反证法证明命题“设 \(a, b\) 为实数，则方程 \(x^{3} + ax + b = 0\) 至少有一个实根”时，要做的假设是（\quad）

\choices
    {方程 \( x^{3} + ax + b = 0 \) 没有实根}
    {方程 \( x^{3} + ax + b = 0 \) 至多有一个实根}
    {方程 \( x^{3} + ax + b = 0 \) 至多有两个实根}
    {方程 \( x^{3} + ax + b = 0 \) 恰好有两个实根}
\end{problem}

\begin{problem}
已知实数 \(x, y\) 满足 \(a^x < a^y\) (\(0 < a < 1\))，则下列关系式恒成立的是（\quad）

\choices

    {\(\ln (x^2 + 1) > \ln (y^2 + 1)\)}
    {\(\sin x > \sin y\)}
    {\(x^{3} > y^{3}\)}

\begin{problem}

\begin{problem}
已知函数 \( f(x) = |x - 2| + 1 \)，\( g(x) = kx \)，若方程 \( f(x) = g(x) \) 有两个不相等的实根，则实数 \( k \) 的取值范围是（\quad）

\begin{problem}
已知 \(x, y\) 满足约束条件 \(\left\{ \begin{array}{l} x - y - 1 \leqslant 0, \\ 2x - y - 3 \geqslant 0, \end{array} \right.\) 当目标函数 \(z = ax + by (a > 0, b > 0)\) 在该约束条件下取到最小值 \(2\sqrt{5}\) 时，\(a^2 + b^2\) 的最小值为（\quad）

\choices
    {\(5\)}
    {\(4\)}
    {\(\sqrt{5}\)}
    {\(2\)}
\end{problem}

\begin{problem}
已知 \(a > b > 0\)，椭圆 \(C_1\) 的方程为 \(\frac{x^2}{a^2} + \frac{y^2}{b^2} = 1\)，双曲线 \(C_2\) 的方程为 \(\frac{x^2}{a^2} - \frac{y^2}{b^2} = 1, C_1\) 与 \(C_2\) 的离心率之积为 \(\frac{\sqrt{3}}{2}\)，则 \(C_2\) 的渐近线方程为（\quad）

\choices
    {\(x \pm \sqrt{2}y = 0\)}
    {\(\sqrt{2}x \pm y = 0\)}
    {\(x \pm 2y = 0\)}
    {\(2x \pm y = 0\)}
\end{problem}

\begin{problem}

\begin{enumerate}
\item 求 \(m, n\) 的值;
\item 将 \( y = f(x) \) 的图象向左平移 \( \varphi (0 < \varphi < \pi) \) 个单位后得到函数 \( y = g(x) \) 的图象. 若 \( y = g(x) \) 的图象上各最高点到点(0,3)的距离的最小值为1，求 \( y = g(x) \) 的单调增区间.
\end{enumerate}
\end{problem}

\begin{problem}
设函数 \( f(x) = \frac{\e^x}{x^2} - k\left(\frac{2}{x} + \ln x\right) \) (\( k \) 为常数，\( \e = 2.71828 \cdots \) 是自然对数的底数).

\end{enumerate}
\end{problem}

\begin{enumerate}
\item 求 \(C\) 的方程;
\item 若直线 \( l_{1} // l \) ，且 \( l_{1} \) 和 C 有且只有一个公共点 E，
\begin{enumerate}[label=\circled{\arabic*}]
\item 证明直线 AE 过定点，并求出定点坐标；
\item \(\triangle ABE\) 的面积是否存在最小值？若存在，请求出最小值；若不存在，请说明理由.
\end{enumerate}
\end{enumerate}
\end{problem}
