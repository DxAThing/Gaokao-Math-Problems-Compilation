\section{单选题}

\begin{problem}
设 \(z = \dfrac{10\i}{\left(3+\i\right)}\)，则 \(z\) 的共轭复数为（\quad）

\choices
    {\(-1 + 3\i\)}
    {\(-1 - 3\i\)}
    {\(1 + 3\i\)}
    {\(1 - 3\i\)}
\answer{D}

\begin{solution}
有理化得
\[
z=\frac{10\i(3-\i)}{(3+\i)(3-\i)}
=\frac{10\i(3-\i)}{10}=1+3\i
\]
所以 \(\overline{z}=1-3\i\)，故选 D.
\end{solution}
\end{problem}

\begin{problem}
设集合 \(M = \{x \mid x^{2} - 3x - 4 < 0\}\)，\(N = \{x \mid 0 \leqslant x \leqslant 5\}\)，则 \(M \cap N =\)（\quad）

\choices
    {\(\left(0, 4\right]\)}
    {\(\left[0, 4\right)\)}
    {\(\left[-1, 0\right)\)}
    {\(\left(-1, 0\right]\)}
\answer{B}

\begin{solution}
因式分解得 \(x^{2}-3x-4=(x-4)(x+1)\)，所以
\(M=\left(-1,4\right)\). 与 \(N=\left[0,5\right]\) 相交，得
\[
M\cap N=\left[0,4\right)
\]
故选 B.
\end{solution}
\end{problem}

\begin{problem}
设 \(a = \sin 33^{\circ}\)，\(b = \cos 55^{\circ}\)，\(c = \tan 35^{\circ}\)，则（\quad）

\choices
    {\(a > b > c\)}
    {\(b > c > a\)}
    {\(c > b > a\)}
    {\(c > a > b\)}
\answer{C}

\begin{solution}
由诱导公式，\(b=\cos55^{\circ}=\sin35^{\circ}\). 因为正弦函数在
\(\left(0^{\circ},90^{\circ}\right)\) 上递增，且 \(33^{\circ}<35^{\circ}\)，所以
\(a<b\). 又 \(0<\cos35^{\circ}<1\)，故
\[
c=\tan35^{\circ}=\frac{\sin35^{\circ}}{\cos35^{\circ}}>\sin35^{\circ}=b
\]
因此 \(c>b>a\)，故选 C.
\end{solution}
\end{problem}

\begin{problem}
若向量 \(\bs{a}\)，\(\bs{b}\) 满足：\(|\bs{a}| = 1\)，\((\bs{a} + \bs{b}) \perp \bs{a}\)，\((2\bs{a} + \bs{b}) \perp \bs{b}\)，则 \(|\bs{b}| =\)（\quad）

\choices
    {\(2\)}
    {\(\sqrt{2}\)}
    {\(1\)}
    {\(\dfrac{\sqrt{2}}{2}\)}
\answer{B}

\begin{solution}
由 \((\bs{a}+\bs{b})\perp\bs{a}\)，有
\[
0=(\bs{a}+\bs{b})\cdot\bs{a}=|\bs{a}|^{2}+\bs{a}\cdot\bs{b}
=1+\bs{a}\cdot\bs{b}
\]
所以 \(\bs{a}\cdot\bs{b}=-1\). 再由
\((2\bs{a}+\bs{b})\perp\bs{b}\)，得
\[
0=(2\bs{a}+\bs{b})\cdot\bs{b}=2\bs{a}\cdot\bs{b}+|\bs{b}|^{2}
=-2+|\bs{b}|^{2}
\]
因此 \(|\bs{b}|=\sqrt{2}\)，故选 B.
\end{solution}
\end{problem}

\begin{problem}
有 \(6\text{ 名}\)男医生，\(5\text{ 名}\)女医生，从中选出 \(2\text{ 名}\)男医生，\(1\text{ 名}\)女医生组成一个医疗小组，则不同的选法共有（\quad）

\choices
    {\(60\text{ 种}\)}
    {\(70\text{ 种}\)}
    {\(75\text{ 种}\)}
    {\(150\text{ 种}\)}
\answer{C}

\begin{solution}
先从男医生中选 \(2\) 人，再从女医生中选 \(1\) 人，选法数为
\[
\mathrm C_{6}^{2}\mathrm C_{5}^{1}=15\times5=75
\]
故选 C.
\end{solution}
\end{problem}

\begin{problem}
已知椭圆 \(C: \dfrac{x^{2}}{a^{2}}+\dfrac{y^{2}}{b^{2}}=1\left(a>b>0\right)\) 的左，右焦点分别为 \(F_{1}\)，\(F_{2}\)，离心率为 \(\dfrac{\sqrt{3}}{3}\)，过 \(F_{2}\) 的直线 \(l\) 交 \(C\) 于 \(A\)，\(B\) 两点，若 \(\triangle AF_{1}B\) 的周长为 \(4\sqrt{3}\)，则 \(C\) 的方程为（\quad）

\choices
    {\(\dfrac{x^{2}}{3}+\dfrac{y^{2}}{2}=1\)}
    {\(\dfrac{x^{2}}{3}+y^{2}=1\)}
    {\(\dfrac{x^{2}}{12}+\dfrac{y^{2}}{8}=1\)}
    {\(\dfrac{x^{2}}{12}+\dfrac{y^{2}}{4}=1\)}
\answer{A}

\begin{solution}
设椭圆的焦距的一半为 \(c\). 由离心率
\(\dfrac{c}{a}=\dfrac{\sqrt{3}}{3}\)，得
\(c=\frac{a}{\sqrt{3}}\)，\(b^{2}=a^{2}-c^{2}=\frac{2a^{2}}{3}\).
焦点 \(F_{2}\) 在椭圆内部，故过 \(F_{2}\) 的弦的两个端点满足
\(F_{2}\) 在线段 \(AB\) 上，从而 \(AB=AF_{2}+BF_{2}\). 由椭圆的定义，
\(AF_{1}+AF_{2}=2a\)，\(BF_{1}+BF_{2}=2a\).
于是
\[
AF_{1}+AB+BF_{1}=4a
\]
由三角形周长为 \(4\sqrt{3}\)，得 \(a=\sqrt{3}\)，进而
\(c=1\)，\(b^{2}=2\). 所以椭圆方程为
\[
\frac{x^{2}}{3}+\frac{y^{2}}{2}=1
\]
故选 A.
\end{solution}
\end{problem}

\begin{problem}
曲线 \(y=x\e^{x-1}\) 在点 \(\left(1,1\right)\) 处切线的斜率等于（\quad）

\choices
    {\(2\e\)}
    {\(\e\)}
    {\(2\)}
    {\(1\)}
\answer{C}

\begin{solution}
求导得
\[
y'=\e^{x-1}+x\e^{x-1}=(x+1)\e^{x-1}
\]
因此在 \(x=1\) 处的切线斜率为
\(y'(1)=2\)，故选 C.
\end{solution}
\end{problem}

\begin{problem}
正四棱锥的顶点都在同一球面上，若该棱锥的高为 \(4\)，底面边长为 \(2\)，则该球的表面积为（\quad）

\choices
    {\(\dfrac{81\pi}{4}\)}
    {\(16\pi\)}
    {\(9\pi\)}
    {\(\dfrac{27\pi}{4}\)}
\answer{A}

\begin{solution}
由正四棱锥的对称性，球心 \(O'\) 在高 \(SO\) 所在直线上. 设底面中心为 \(O\)，顶点为 \(S\)，并设
\(OO'=t\). 正方形底面边长为 \(2\)，所以底面顶点 \(V\) 满足
\(OV=\sqrt{2}\). 由球面经过 \(V\) 和 \(S\)，有
\(O'V^{2}=t^{2}+2\)，\(O'S^{2}=(4-t)^{2}\).
故
\(t^{2}+2=(4-t)^{2}\)，\(t=\frac{7}{4}\).
球半径为 \(R=4-t=\dfrac{9}{4}\)，所以球的表面积为
\[
4\pi R^{2}=4\pi\left(\frac{9}{4}\right)^{2}=\frac{81\pi}{4}
\]
故选 A.
\end{solution}
\end{problem}

\begin{problem}
已知双曲线 \(C\) 的离心率为 \(2\)，焦点为 \(F_{1}\)，\(F_{2}\)，点 \(A\) 在 \(C\) 上，若 \(|F_{1}A|=2|F_{2}A|\)，则 \(\cos \angle AF_{2}F_{1}=\)（\quad）

\choices
    {\(\dfrac{1}{4}\)}
    {\(\dfrac{1}{3}\)}
    {\(\dfrac{\sqrt{2}}{4}\)}
    {\(\dfrac{\sqrt{2}}{3}\)}
\answer{A}

\begin{solution}
双曲线的离心率为 \(2\)，所以 \(c=2a\)，且
\(F_{1}F_{2}=2c=4a\). 由双曲线定义以及
\(|F_{1}A|=2|F_{2}A|\)，得
\(|F_{1}A|-|F_{2}A|=2a\)，\(|F_{2}A|=2a\)，\(|F_{1}A|=4a\).
在三角形 \(AF_{2}F_{1}\) 中，利用余弦定理，
\[
\cos\angle AF_{2}F_{1}
=\frac{|AF_{2}|^{2}+|F_{2}F_{1}|^{2}-|AF_{1}|^{2}}
{2|AF_{2}|\,|F_{2}F_{1}|}
=\frac{4a^{2}+16a^{2}-16a^{2}}{16a^{2}}=\frac14
\]
故选 A.
\end{solution}
\end{problem}

\begin{problem}
等比数列 \(\{a_{n}\}\) 中，\(a_{4}=2\)，\(a_{5}=5\)，则数列 \(\{\lg a_{n}\}\) 的前 \(8\) 项和等于（\quad）

\choices
    {6}
    {5}
    {4}
    {3}
\answer{C}

\begin{solution}
设 \(u_{n}=\lg a_{n}\). 等比数列的公比为
\(q=\dfrac{a_{5}}{a_{4}}=\dfrac52\)，故 \(\{u_{n}\}\) 是等差数列.
对称项的和相等，因此
\[
\sum_{n=1}^{8}u_{n}=4(u_{1}+u_{8})=4(u_{4}+u_{5})
\]
又 \(u_{4}=\lg2\)，\(u_{5}=\lg5\)，所以
\[
\sum_{n=1}^{8}\lg a_{n}=4\lg(2\times5)=4
\]
故选 C.
\end{solution}
\end{problem}

\begin{problem}
已知二面角 \(\alpha-l-\beta\) 为 \(60^{\circ}\)，\(AB\subset \alpha\)，\(AB\perp l\)，\(A\) 为垂足，\(CD\subset \beta\)，\(C\in l\)，\(\angle ACD=135^{\circ}\)，则异面直线 \(AB\) 与 \(CD\) 所成角的余弦值为（\quad）

\choices
    {\(\dfrac{1}{4}\)}
    {\(\dfrac{\sqrt{2}}{4}\)}
    {\(\dfrac{\sqrt{3}}{4}\)}
    {\(\dfrac{1}{2}\)}
\answer{B}

\begin{solution}
取 \(l\) 为 \(x\) 轴，取 \(\alpha\) 内垂直于 \(l\) 的单位向量为
\(\bs{e}_{y}\)，并取垂直于 \(\alpha\) 的单位向量为
\(\bs{e}_{z}\). 由二面角为 \(60^{\circ}\)，\(\beta\) 内垂直于
\(l\) 的单位向量可取
\[
\bs{u}=\frac12\bs{e}_{y}+\frac{\sqrt3}{2}\bs{e}_{z}
\]
设从 \(C\) 指向 \(D\) 的单位向量为 \(\bs{v}\). 因为射线 \(CA\) 的方向为
\(-\bs{e}_{x}\)，且 \(\angle ACD=135^{\circ}\)，有
\[
(-\bs{e}_{x})\cdot\bs{v}=-\frac{\sqrt2}{2}
\]
于是
\[
\bs{v}=\frac{\sqrt2}{2}\bs{e}_{x}
\pm\frac{\sqrt2}{2}\bs{u}
\]
从而
\[
\left|\bs{e}_{y}\cdot\bs{v}\right|=\frac{\sqrt2}{4}
\]
异面直线所成角取锐角，故其余弦为 \(\dfrac{\sqrt2}{4}\)，故选 B.
\end{solution}
\end{problem}

\begin{problem}
函数 \(y=f(x)\) 的图象与函数 \(y=g(x)\) 的图象关于直线 \(x+y=0\) 对称，则 \(y=f(x)\) 的反函数是（\quad）

\choices
    {\(y=g(x)\)}
    {\(y=g(-x)\)}
    {\(y=-g(x)\)}
    {\(y=-g(-x)\)}
\answer{D}

\begin{solution}
关于直线 \(x+y=0\) 的对称把点 \((x,y)\) 变为 \((-y,-x)\). 因此图象
\(y=f(x)\) 上的点 \((x,f(x))\) 对称后为
\((-f(x),-x)\)，且该点在 \(y=g(x)\) 的图象上，即
\[
g(-f(x))=-x
\]
令 \(t=f(x)\)，则 \(x=f^{-1}(t)\)，上式化为
\(g(-t)=-f^{-1}(t)\)，\(f^{-1}(t)=-g(-t)\).
所以反函数为 \(y=-g(-x)\)，故选 D.
\end{solution}
\end{problem}

\section{填空题}

\begin{problem}
\(\left(\dfrac{x}{\sqrt{y}}-\dfrac{y}{\sqrt{x}}\right)^{8}\) 的展开式中 \(x^{2}y^{2}\) 的系数为\fillinblank{}.
\answer{70}

\begin{solution}
通项中取第二项 \(k\) 次，所得项为
\[
(-1)^{k}\mathrm C_{8}^{k}
x^{8-k-\frac{k}{2}}y^{-\frac{8-k}{2}+k}
=(-1)^{k}\mathrm C_{8}^{k}x^{8-\frac32k}y^{-4+\frac32k}
\]
要得到 \(x^{2}y^{2}\)，需满足
\(8-\dfrac32k=2\)，解得 \(k=4\)，此时符号为正. 所以所求系数为
\[
\mathrm C_{8}^{4}=70
\]
\end{solution}
\end{problem}

\begin{problem}
设 \(x\)，\(y\) 满足约束条件 \(\begin{cases}
x-y\geqslant 0,\\
x+2y\leqslant 3,\\
x-2y\leqslant 1,
\end{cases}\) 则 \(z=x+4y\) 的最大值为\fillinblank{}.
\answer{5}

\begin{solution}
三个约束边界的交点分别为
\(y=x\)，\(\ x-2y=1\Longrightarrow(-1,-1)\)，
\(y=x\)，\(\ x+2y=3\Longrightarrow(1,1)\)，以及
\(x+2y=3\)，\(\ x-2y=1\Longrightarrow\left(2,\frac12\right)\).
这三个点均满足其余约束，构成可行域的三个顶点；线性目标函数在该多边形上的最大值必在顶点处取得. 在顶点处计算
\(z=x+4y\)，分别得到 \(-5\)，\(5\)，\(4\). 因而最大值为 \(5\).
\end{solution}
\end{problem}

\begin{problem}
直线 \(l_{1}\) 和 \(l_{2}\) 是圆 \(x^{2}+y^{2}=2\) 的两条切线. 若 \(l_{1}\) 与 \(l_{2}\) 的交点为 \(\left(1,3\right)\)，则 \(l_{1}\) 与 \(l_{2}\) 的夹角的正切值等于\fillinblank{}.
\answer{\(\dfrac{4}{3}\)}

\begin{solution}
设两切线的交点为 \(P=(1,3)\)，圆心为 \(O=(0,0)\)，切点为 \(T\). 则
\(OP=\sqrt{1^{2}+3^{2}}=\sqrt{10}\)，圆半径为 \(OT=\sqrt2\)，且
\(PT\perp OT\). 设切线与 \(PO\) 的夹角为 \(\theta\)，则在直角三角形
\(OPT\) 中
\[
\tan\theta=\frac{OT}{PT}
=\frac{\sqrt2}{\sqrt{OP^{2}-OT^{2}}}
=\frac{\sqrt2}{\sqrt8}=\frac12
\]
两条切线关于 \(PO\) 对称，故两切线的锐角夹角为 \(2\theta\). 因此
\[
\tan 2\theta=\frac{2\tan\theta}{1-\tan^{2}\theta}
=\frac{1}{1-\frac14}=\frac43
\]
\end{solution}
\end{problem}

\begin{problem}
若函数 \(f(x)=\cos 2x+a\sin x\) 在区间 \(\left(\dfrac{\pi}{6},\dfrac{\pi}{2}\right)\) 上是减函数，则 \(a\) 的取值范围是\fillinblank{}.
\answer{\(\left(-\infty,2\right]\)}

\begin{solution}
求导得
\[
f'(x)=-2\sin2x+a\cos x=\cos x\left(a-4\sin x\right)
\]
在 \(\left(\dfrac{\pi}{6},\dfrac{\pi}{2}\right)\) 上，\(\cos x>0\)，且
\(\sin x>\dfrac12\). 若 \(a\leqslant2\)，则
\(a-4\sin x<0\)，所以函数在该区间上递减.

若 \(a>2\)，由 \(\sin x\) 在该区间左端附近趋近于 \(\dfrac12\)，可取区间内的
\(x\) 使 \(4\sin x<a\)，此时 \(f'(x)>0\)，不可能在整个区间上递减.
因此 \(a\) 的取值范围为 \(\left(-\infty,2\right]\).
\end{solution}
\end{problem}

\begin{problem}
\(\triangle ABC\) 的内角 \(A\)，\(B\)，\(C\) 的对边分别为 \(a\)，\(b\)，\(c\)，已知 \(3a\cos C=2c\cos A\)，\(\tan A=\dfrac{1}{3}\)，求 \(B\).
\begin{solution}
由正弦定理，存在正数 \(k\)，使
\(a=k\sin A\)，\(c=k\sin C\). 代入已知条件得
\[
3\sin A\cos C=2\sin C\cos A
\]
因 \(\tan A=\dfrac13>0\) 且 \(A\in\left(0,\pi\right)\)，所以 \(A\) 为锐角；上式也说明
\(\cos C\neq0\). 因此可除以 \(\cos A\cos C\)，得到
\(3\tan A=2\tan C\)，\(\tan C=\frac12\).
\(C\in\left(0,\pi\right)\) 且正切值为正，所以 \(C\) 也是锐角. 于是
\[
\tan(A+C)=\frac{\tan A+\tan C}{1-\tan A\tan C}
=\frac{\frac13+\frac12}{1-\frac16}=1
\]
又 \(A+C\in\left(0,\pi\right)\)，且其正切为正，故
\(A+C=45^{\circ}\). 因而
\[
B=180^{\circ}-(A+C)=135^{\circ}
\]
\end{solution}
\end{problem}

\begin{problem}
等差数列 \(\{a_{n}\}\) 的前 \(n\) 项和为 \(S_{n}\)，已知 \(a_{1}=10\)，\(a_{2}\) 为整数，且 \(S_{n}\leqslant S_{4}\).

\begin{enumerate}
\item 求 \(\{a_{n}\}\) 的通项公式；
\item 设 \(b_{n}=\dfrac{1}{a_{n}a_{n+1}}\)，求数列 \(\{b_{n}\}\) 的前 \(n\) 项和 \(T_{n}\).
\end{enumerate}
\begin{solution}
\begin{enumerate}
\item 设等差数列的公差为 \(d\)，则 \(a_{n}=10+(n-1)d\). 由题设
\(S_{3}\leqslant S_{4}\) 和 \(S_{5}\leqslant S_{4}\)，分别得到
\(a_{4}=10+3d\geqslant0\)，\(a_{5}=10+4d\leqslant0\).
因为 \(d\) 为整数，所以 \(-\dfrac{10}{3}\leqslant d\leqslant-\dfrac52\) 只能给出
\(d=-3\). 此时
\(S_{n}=\frac{n}{2}\left(20-3(n-1)\right)=\frac{n(23-3n)}{2}\)，\(S_{4}=22\)
且对任意正整数 \(n\)，
\[
S_{n}-S_{4}=-\frac{(3n-11)(n-4)}{2}\leqslant0
\]
其中，\(n=1\)，\(2\)，\(3\) 时两个因式均为负，\(n\geqslant4\) 时两个因式均为非负.
所以所得数列确实满足题设条件，且
\[
a_{n}=10-3(n-1)=13-3n
\]
\item 由 \(a_{k}-a_{k+1}=3\)，有
\[
b_{k}=\frac{1}{a_{k}a_{k+1}}
=\frac13\left(\frac1{a_{k+1}}-\frac1{a_{k}}\right)
\]
所以
\[
T_{n}=\sum_{k=1}^{n}b_{k}
=\frac13\left(\frac1{a_{n+1}}-\frac1{a_{1}}\right)
=\frac13\left(\frac1{10-3n}-\frac1{10}\right)
=\frac{n}{10(10-3n)}
\]
\end{enumerate}
\end{solution}
\end{problem}

\begin{problem}
如图，三棱柱 \(ABC-A_{1}B_{1}C_{1}\) 中，点 \(A_{1}\) 在平面 \(ABC\) 内的射影 \(D\) 在 \(AC\) 上，\(\angle ACB=90^{\circ}\)，\(BC=1\)，\(AC=CC_{1}=2\).

\begin{enumerate}
\item 证明：\(AC_{1}\perp A_{1}B\)；
\item 设直线 \(AA_{1}\) 与平面 \(BCC_{1}B_{1}\) 的距离为 \(\sqrt{3}\)，求二面角 \(A_{1}-AB-C\) 的大小.
\end{enumerate}

\bitmapfigure[max width=0.35\linewidth]{img/2014/syllabus_science/q19_fig1.png}
\begin{solution}
\begin{enumerate}
\item 建立空间直角坐标系，使 \(C\) 为原点，\(CA\) 和 \(CB\) 分别为
\(x\) 轴和 \(y\) 轴正向. 则
\(A=(2,0,0)\)，\(B=(0,1,0)\)，\(C=(0,0,0)\).
设 \(D=(d,0,0)\)，由射影条件可设 \(A_{1}=(d,0,h)\)，其中 \(h>0\).
棱柱的平移向量为 \(\overrightarrow{AA_{1}}=\overrightarrow{CC_{1}}\)，故
\(C_{1}=(d-2,0,h)\)，\((d-2)^{2}+h^{2}=|CC_{1}|^{2}=4\).
于是
\(\overrightarrow{AC_{1}}=(d-4,0,h)\)，\(\overrightarrow{A_{1}B}=(-d,1,-h)\)
且
\[
\overrightarrow{AC_{1}}\cdot\overrightarrow{A_{1}B}
=-d(d-4)-h^{2}=-(d^{2}-4d+h^{2})=0
\]
其中最后一步使用了 \((d-2)^{2}+h^{2}=4\). 因而
\(AC_{1}\perp A_{1}B\).
\item 平面 \(BCC_{1}B_{1}\) 经过原点，包含方向向量
\((0,1,0)\) 和 \((d-2,0,h)\)，其方程为
\[
h x-(d-2)z=0
\]
直线 \(AA_{1}\) 的方向向量为 \((d-2,0,h)\)，在该平面内，所以
\(AA_{1}\) 平行于平面 \(BCC_{1}B_{1}\). 于是其与该平面的距离等于点
\(A\) 到平面的距离，即
\[
\frac{2h}{\sqrt{h^{2}+(d-2)^{2}}}=\frac{2h}{2}=h
\]
由题设距离为 \(\sqrt3\)，得 \(h=\sqrt3\)，再由
\((d-2)^{2}+h^{2}=4\) 得 \(d=1\) 或 \(d=3\). 因为 \(D\) 在线段
\(AC\) 上，故 \(0\leqslant d\leqslant2\)，只能取 \(d=1\).

平面 \(A_{1}AB\) 的一个法向量为
\[
\overrightarrow{AB}\times\overrightarrow{AA_{1}}
=\left(\sqrt3,2\sqrt3,2-d\right)
\]
其长度为 \(4\). 平面 \(ABC\) 的法向量为 \((0,0,1)\). 设二面角
\(A_{1}-AB-C\) 为 \(\theta\)，则
\[
\cos\theta=\frac{|2-d|}{4}=\frac14
\]
由于 \(\theta\) 为锐角，故
\(\tan\theta=\sqrt{\frac{1}{\cos^{2}\theta}-1}=\sqrt{15}\)，\(\theta=\arctan\sqrt{15}\).
\end{enumerate}
\end{solution}
\end{problem}

\begin{problem}
设每个工作日甲，乙，丙，丁 \(4\) 人需使用某种设备的概率分别为 \(0.6\)，\(0.5\)，\(0.5\)，\(0.4\)，各人是否需使用设备相互独立.

\begin{enumerate}
\item 求同一工作日至少 \(3\) 人需使用设备的概率；
\item \(X\) 表示同一工作日需使用设备的人数，求 \(X\) 的数学期望.
\end{enumerate}
\begin{solution}
\begin{enumerate}
\item 恰有 \(3\) 人需使用设备的概率为
\[
0.4\times0.5\times0.5\times0.4
+0.6\times0.5\times0.5\times0.4
+0.6\times0.5\times0.5\times0.4
+0.6\times0.5\times0.5\times0.6=0.25
\]
四人都需使用设备的概率为
\[
0.6\times0.5\times0.5\times0.4=0.06
\]
所以同一工作日至少 \(3\) 人需使用设备的概率为
\[
0.25+0.06=0.31
\]
\item 设第 \(i\) 个人需使用设备的指示变量为 \(I_{i}\)，则
\(X=I_{1}+I_{2}+I_{3}+I_{4}\). 由期望的线性性质，
\[
E(X)=E(I_{1})+E(I_{2})+E(I_{3})+E(I_{4})
=0.6+0.5+0.5+0.4=2
\]
\end{enumerate}
\end{solution}
\end{problem}

\begin{problem}
已知抛物线 \(C: y^{2} = 2px\) \(\left(p > 0\right)\) 的焦点为 \(F\)，直线 \(y = 4\) 与 \(y\) 轴的交点为 \(P\)，与 \(C\) 的交点为 \(Q\)，且 \(|QF| = \dfrac{5}{4}|PQ|\).

\begin{enumerate}
\item 求 \(C\) 的方程；
\item 过 \(F\) 的直线 \(l\) 与 \(C\) 相交于 \(A\)，\(B\) 两点，若 \(AB\) 的垂直平分线 \(l'\) 与 \(C\) 相交于 \(M\)，\(N\) 两点，且 \(A\)，\(M\)，\(B\)，\(N\) 四点在同一圆上，求 \(l\) 的方程.
\end{enumerate}
\begin{solution}
\begin{enumerate}
\item 抛物线 \(y^{2}=2px\) 的焦点为 \(F=\left(\dfrac p2,0\right)\). 由
\(y=4\) 与抛物线的交点条件，得
\(Q=\left(\frac8p,4\right)\)，\(P=(0,4)\)，\(|PQ|=\frac8p\).
因此
\[
|QF|^{2}=\left(\frac8p-\frac p2\right)^{2}+16
=\frac{100}{p^{2}}
\]
整理得
\[
p^{4}+32p^{2}-144=0
\]
令 \(u=p^{2}\)，则 \(u^{2}+32u-144=0\)，所以
\(u=4\) 或 \(u=-36\). 由 \(p>0\)，得 \(p=2\)，故
\[
C:y^{2}=4x
\]
\item 由上可知 \(F=(1,0)\). 直线 \(y=0\) 与抛物线只有一个交点
\((0,0)\)，不满足 \(l\) 与 \(C\) 相交于 \(A\)，\(B\) 两点，故先考虑
\(k\neq0\) 的非竖直直线，设过 \(F\) 的直线为
\(l:y=k(x-1)\)，\(k\neq0\).
它与抛物线的交点横坐标为 \(x_{1}\)，\(x_{2}\)，则
\(k^{2}(x-1)^{2}=4x\)，\(x_{1}+x_{2}=2+\frac4{k^{2}}\)，\(x_{1}x_{2}=1\).
两交点的中点为 \(\left(1+\dfrac2{k^{2}},\dfrac2k\right)\)，而弦 \(AB\) 的斜率为
\(k\)，故其垂直平分线为
\[
l':x+ky=3+\frac2{k^{2}}
\]
设 \(A\)，\(B\)，\(M\)，\(N\) 在抛物线上的参数分别为其纵坐标
\(y_{1}\)，\(y_{2}\)，\(y_{3}\)，\(y_{4}\). 由韦达定理，
\(y_{1}+y_{2}=k(x_{1}+x_{2}-2)=\frac4k\)，\(y_{3}+y_{4}=-4k\)
其中第二式来自将 \(x=3+\dfrac2{k^{2}}-ky\) 代入 \(y^{2}=4x\).

抛物线上的点可写为 \(\left(\dfrac{y^{2}}4,y\right)\). 若四点共圆，设圆方程为
\(x^{2}+y^{2}+ux+vy+w=0\)，代入抛物线后得到关于参数 \(y\) 的四次方程，且其三次项系数为零，所以四个参数之和为零. 反过来，四个参数之和为零时，以这四个参数为根的首一四次式也无三次项，能够写成上述圆与抛物线联立所得的形式，故该条件也是充分的. 因而
\(\frac4k-4k=0\)，\(k^{2}=1\).
所以 \(k=1\) 或 \(k=-1\)，从而
\[
l:y=x-1\quad\text{或}\quad l:y=-x+1
\]
即 \(l\) 的方程为
\[
x-y-1=0\quad\text{或}\quad x+y-1=0
\]
竖直直线 \(x=1\) 的垂直平分线为 \(y=0\)，只与抛物线有一个交点，不满足题设的两个交点条件，故不再取.
\end{enumerate}
\end{solution}
\end{problem}

\section{选考题：解答题}

\begin{problem}
函数 \(f(x)=\ln(x+1)-\dfrac{ax}{x+a}\) \(\left(a>1\right)\).

\begin{enumerate}
\item 讨论 \(f(x)\) 的单调性；
\item 设 \(a_{1}=1\)，\(a_{n+1}=\ln(a_{n}+1)\)，证明：\(\dfrac{2}{n+2}<a_{n}\leqslant \dfrac{3}{n+2}\).
\end{enumerate}
\begin{solution}
\begin{enumerate}
\item 函数定义域为 \((-1,+\infty)\). 求导得
\[
f'(x)=\frac1{x+1}-\frac{a^{2}}{(x+a)^{2}}
=\frac{x\left(x-a^{2}+2a\right)}{(x+1)(x+a)^{2}}
=\frac{x\left(x-(a^{2}-2a)\right)}{(x+1)(x+a)^{2}}
\]
在定义域内分母恒为正，因此只需判断
\(x\left(x-(a^{2}-2a)\right)\) 的符号.

当 \(1<a<2\) 时，\(-1<a^{2}-2a<0\)，故
\[
f(x)\text{ 在 }(-1,a^{2}-2a)\text{ 上递增，在 }(a^{2}-2a,0)\text{ 上递减，}
\]
且在 \((0,+\infty)\) 上递增.

当 \(a=2\) 时，\(a^{2}-2a=0\)，有 \(f'(x)\geqslant0\)，且除 \(x=0\) 外为正，故
\(f(x)\) 在 \((-1,+\infty)\) 上递增.

当 \(a>2\) 时，\(a^{2}-2a>0\)，故
\[
f(x)\text{ 在 }(-1,0)\text{ 上递增，在 }(0,a^{2}-2a)\text{ 上递减，}
\]
且在 \((a^{2}-2a,+\infty)\) 上递增.
\item 先证明对任意 \(x>0\)，有
\[
\frac{2x}{2+x}<\ln(1+x)\leqslant\frac{x(2+x)}{2(1+x)}
\]
左侧不等式可令
\(G(x)=\ln(1+x)-\dfrac{2x}{2+x}\)，则
\[
G'(x)=\frac1{1+x}-\frac4{(2+x)^{2}}
=\frac{x^{2}}{(1+x)(2+x)^{2}}>0
\]
且 \(G(0)=0\). 右侧不等式可令
\(H(x)=\dfrac{x(2+x)}{2(1+x)}-\ln(1+x)\)，则
\[
H'(x)=\frac{x^{2}}{2(1+x)^{2}}\geqslant0
\]
且 \(H(0)=0\).

用归纳法证明所给不等式. 当 \(n=1\) 时，
\(\dfrac23<a_{1}=1\leqslant1=\dfrac33\). 若 \(a_n>0\)，则
\(a_{n+1}=\ln(1+a_n)>0\)，所以归纳过程中各项均为正. 假设
\[
\frac2{n+2}<a_{n}\leqslant\frac3{n+2}
\]
由 \(a_{n}>0\) 及左侧不等式，
\[
a_{n+1}=\ln(1+a_{n})
>\frac{2a_{n}}{2+a_{n}}
>\frac{2\cdot\frac2{n+2}}{2+\frac2{n+2}}
=\frac2{n+3}
\]
令 \(t=n+2\)，则 \(t\geqslant3\). 由对数不等式的右侧，
\[
a_{n+1}\leqslant\ln\left(1+\frac3t\right)
\leqslant\frac{3(2t+3)}{2t(t+3)}
\leqslant\frac3{t+1}=\frac3{n+3}
\]
最后一个不等式等价于 \(t\geqslant3\). 归纳完成，故
\[
\frac2{n+2}<a_{n}\leqslant\frac3{n+2}
\]
\end{enumerate}
\end{solution}
\end{problem}
