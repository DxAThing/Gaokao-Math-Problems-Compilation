\section{选择题}

\begin{problem}
设 \(z = \dfrac{10\i}{3+\i}\)，则 \(z\) 的共轭复数为（\quad）

\choices
    {\(-1 + 3\i\)}
    {\(-1 - 3\i\)}
    {\(1 + 3\i\)}
    {\(1 - 3\i\)}
\end{problem}

\begin{problem}
设集合 \(M = \{x \mid x^{2} - 3x - 4 < 0\}\)，\(N = \{x \mid 0 \leqslant x \leqslant 5\}\)，则 \(M \cap N =\)（\quad）

\choices
    {\((0, 4]\)}
    {\([0, 4)\)}
    {\([-1, 0)\)}
    {\((-1, 0]\)}
\end{problem}

\begin{problem}
设 \(a = \sin 33^{\circ}\)，\(b = \cos 55^{\circ}\)，\(c = \tan 35^{\circ}\)，则（\quad）

\choices
    {\(a > b > c\)}
    {\(b > c > a\)}
    {\(c > b > a\)}
    {\(c > a > b\)}
\end{problem}

\begin{problem}
若向量 \(a\)，\(b\) 满足：\(|a| = 1\)，\((a + b) \perp a\)，\((2a + b) \perp b\)，则 \(|b| =\)（\quad）

\choices
    {\(2\)}
    {\(\sqrt{2}\)}
    {\(1\)}
    {\(\dfrac{\sqrt{2}}{2}\)}
\end{problem}

\begin{problem}
有 6 名男医生，5 名女医生，从中选出 2 名男医生，1 名女医生组成一个医疗小组，则不同的选法共有（\quad）

\choices
    {60 种}
    {70 种}
    {75 种}
    {150 种}
\end{problem}

\begin{problem}
已知椭圆 \(C: \dfrac{x^{2}}{a^{2}}+\dfrac{y^{2}}{b^{2}}=1\,(a>b>0)\) 的左，右焦点分别为 \(F_{1}\)，\(F_{2}\)，离心率为 \(\dfrac{\sqrt{3}}{3}\)，过 \(F_{2}\) 的直线 \(l\) 交 \(C\) 于 \(A\)，\(B\) 两点，若 \(\triangle AF_{1}B\) 的周长为 \(4\sqrt{3}\)，则 \(C\) 的方程为（\quad）

\choices
    {\(\dfrac{x^{2}}{3}+\dfrac{y^{2}}{2}=1\)}
    {\(\dfrac{x^{2}}{3}+y^{2}=1\)}
    {\(\dfrac{x^{2}}{12}+\dfrac{y^{2}}{8}=1\)}
    {\(\dfrac{x^{2}}{12}+\dfrac{y^{2}}{4}=1\)}
\end{problem}

\begin{problem}
曲线 \(y=x\e^{x-1}\) 在点 \((1,1)\) 处切线的斜率等于（\quad）

\choices
    {\(2e\)}
    {\(e\)}
    {\(2\)}
    {\(1\)}
\end{problem}

\begin{problem}
正四棱锥的顶点都在同一球面上，若该棱锥的高为 \(4\)，底面边长为 \(2\)，则该球的表面积为（\quad）

\choices
    {\(\dfrac{81\pi}{4}\)}
    {\(16\pi\)}
    {\(9\pi\)}
    {\(\dfrac{27\pi}{4}\)}
\end{problem}

\begin{problem}
已知双曲线 \(C\) 的离心率为 \(2\)，焦点为 \(F_{1}\)，\(F_{2}\)，点 \(A\) 在 \(C\) 上，若 \(|F_{1}A|=2|F_{2}A|\)，则 \(\cos \angle AF_{2}F_{1}=\)（\quad）

\choices
    {\(\dfrac{1}{4}\)}
    {\(\dfrac{1}{3}\)}
    {\(\dfrac{\sqrt{2}}{4}\)}
    {\(\dfrac{\sqrt{2}}{3}\)}
\end{problem}

\begin{problem}
等比数列 \(\{a_{n}\}\) 中，\(a_{4}=2\)，\(a_{5}=5\)，则数列 \(\{\lg a_{n}\}\) 的前 \(8\) 项和等于（\quad）

\choices
    {6}
    {5}
    {4}
    {3}
\end{problem}

\begin{problem}
已知二面角 \(\alpha-l-\beta\) 为 \(60^{\circ}\)，\(AB\subset \alpha\)，\(AB\perp l\)，\(A\) 为垂足，\(CD\subset \beta\)，\(C\in l\)，\(\angle ACD=135^{\circ}\)，则异面直线 \(AB\) 与 \(CD\) 所成角的余弦值为（\quad）

\choices
    {\(\dfrac{1}{4}\)}
    {\(\dfrac{\sqrt{2}}{4}\)}
    {\(\dfrac{\sqrt{3}}{4}\)}
    {\(\dfrac{1}{2}\)}
\end{problem}

\begin{problem}
函数 \(y=f(x)\) 的图象与函数 \(y=g(x)\) 的图象关于直线 \(x+y=0\) 对称，则 \(y=f(x)\) 的反函数是（\quad）

\choices
    {\(y=g(x)\)}
    {\(y=g(-x)\)}
    {\(y=-g(x)\)}
    {\(y=-g(-x)\)}
\end{problem}

x+2y\leqslant 3,\\

\section{解答题}

\begin{problem}
\(\triangle ABC\) 的内角 \(A\)，\(B\)，\(C\) 的对边分别为 \(a\)，\(b\)，\(c\)，已知 \(3a\cos C=2c\cos A\)，\(\tan A=\dfrac{1}{3}\)，求 \(B\).
\end{problem}

\begin{problem}
等差数列 \(\{a_{n}\}\) 的前 \(n\) 项和为 \(S_{n}\)，已知 \(a_{1}=10\)，\(a_{2}\) 为整数，且 \(S_{n}\leqslant S_{4}\).

\begin{enumerate}
\item 求 \(\{a_{n}\}\) 的通项公式；
\item 设 \(b_{n}=\dfrac{1}{a_{n}a_{n+1}}\)，求数列 \(\{b_{n}\}\) 的前 \(n\) 项和 \(T_{n}\).
\end{enumerate}
\end{problem}

\begin{enumerate}
\item 证明：\(AC_{1}\perp A_{1}B\)；
\item 设直线 \(AA_{1}\) 与平面 \(BCC_{1}B_{1}\) 的距离为 \(\sqrt{3}\)，求二面角 \(A_{1}-AB-C\) 的大小.
\end{enumerate}

\begin{problem}
设每个工作日甲，乙，丙，丁 \(4\) 人需使用某种设备的概率分别为 \(0.6\)，\(0.5\)，\(0.5\)，\(0.4\)，各人是否需使用设备相互独立.

\begin{enumerate}
\item 求同一工作日至少 \(3\) 人需使用设备的概率；
\item \(X\) 表示同一工作日需使用设备的人数，求 \(X\) 的数学期望.
\end{enumerate}
\end{problem}

\begin{problem}
已知抛物线 \(C: y^{2} = 2px\) （\(p > 0\)）的焦点为 \(F\)，直线 \(y = 4\) 与 \(y\) 轴的交点为 \(P\)，与 \(C\) 的交点为 \(Q\)，且 \(|QF| = \dfrac{5}{4}|PQ|\).

\begin{enumerate}
\item 求 \(C\) 的方程；
\item 过 \(F\) 的直线 \(l\) 与 \(C\) 相交于 \(A\)，\(B\) 两点，若 \(AB\) 的垂直平分线 \(l'\) 与 \(C\) 相交于 \(M\)，\(N\) 两点，且 \(A\)，\(M\)，\(B\)，\(N\) 四点在同一圆上，求 \(l\) 的方程.
\end{enumerate}
\end{problem}

\begin{problem}
函数 \(f(x)=\ln(x+1)-\dfrac{ax}{x+a}\)（\(a>1\)）.

\begin{enumerate}
\item 讨论 \(f(x)\) 的单调性；
\item 设 \(a_{1}=1\)，\(\ a_{n+1}=\ln(a_{n}+1)\)，证明：\(\dfrac{2}{n+2}<a_{n}\leqslant \dfrac{3}{n+2}\).
\end{enumerate}
\end{problem}
