\section{单选题}

\begin{problem}
设 \(\i\) 是虚数单位，复数 \(\i^{3} + \frac{2\i}{1+\i} =\)（\quad）

\choices
    {\(-\i\)}
    {\(\i\)}
    {\(-1\)}
    {\(1\)}

\begin{answer}
D
\end{answer}
\end{problem}

\begin{solution}
因为
\[
\i^{3}=-\i,
\qquad
\frac{2\i}{1+\i}=\frac{2\i(1-\i)}{(1+\i)(1-\i)}=\i(1-\i)=1+\i.
\]
所以
\[
\i^{3}+\frac{2\i}{1+\i}=-\i+(1+\i)=1.
\]
故选 D.
\end{solution}

\begin{problem}
命题“\(\forall x\in \R\)，\(|x| + x^2 \geqslant 0\)”的否定是（\quad）

\choices
    {\(\forall x\in \R\)，\(|x| + x^2 < 0\)}
    {\(\forall x\in \R\)，\(|x| + x^2 \leqslant 0\)}
    {\(\exists x_0\in \R\)，\(|x_0| + x_0^2 < 0\)}
    {\(\exists x_0\in \R\)，\(|x_0| + x_0^2 \geqslant 0\)}

\begin{answer}
C
\end{answer}
\end{problem}

\begin{solution}
原命题是“对任意 \(x\in\R\)，都有 \(|x|+x^2\geqslant0\)”. 否定全称命题时，把“对任意”改为“存在”，并把不等式 \(\geqslant0\) 改为 \(<0\)，因此否定为“存在 \(x_0\in\R\)，使 \(|x_0|+x_0^2<0\)”.
故选 C.
\end{solution}

\begin{problem}
抛物线 \(y = \frac{1}{4}x^2\) 的准线方程是（\quad）

\choices
    {\(y = -1\)}
    {\(y = -2\)}
    {\(x = -1\)}
    {\(x = -2\)}

\begin{answer}
A
\end{answer}
\end{problem}

\begin{solution}
抛物线的标准方程为 \(x^2=4py\). 由
\[
y=\frac14x^2\Longleftrightarrow x^2=4y
\]
可得 \(p=1\). 因此抛物线的准线为 \(y=-p=-1\).
故选 A.
\end{solution}

如图所示，程序框图（算法流程图）的输出结果是（\quad）

\begin{problem}
设 \(a = \log_{3} 7\)，\(b = 2^{1.1}\)，\(c = 0.8^{3.1}\)，则（\quad）

\choices
    {\(b < a < c\)}
    {\(c < a < b\)}
    {\(c < b < a\)}
    {\(a < c < b\)}

\begin{answer}
B
\end{answer}
\end{problem}

\begin{solution}

因此
\[

\]

\begin{problem}
过点 \(P(-\sqrt{3}, -1)\) 的直线 \(l\) 与圆 \(x^{2} + y^{2} = 1\) 有公共点，则直线 \(l\) 的倾斜角的取值范围是（\quad）

\choices
    {\(\left(0, \frac{\pi}{6}\right]\)}
    {\(\left(0, \frac{\pi}{3}\right]\)}
    {\(\left[0, \frac{\pi}{6}\right]\)}
    {\(\left[0, \frac{\pi}{3}\right]\)}

\begin{answer}
D
\end{answer}
\end{problem}

\begin{solution}
设直线 \(l\) 的倾斜角为 \(\alpha\)，其中 \(0\leqslant\alpha<\pi\). 取直线的单位方向向量为 \((\cos\alpha,\sin\alpha)\)，则点 \(P(-\sqrt{3},-1)\) 到直线 \(l\) 的距离为
\[
d=\left|(-\sqrt{3})\sin\alpha-(-1)\cos\alpha\right|=\left|\cos\alpha-\sqrt{3}\sin\alpha\right|=2\left|\cos\left(\alpha+\frac{\pi}{3}\right)\right|.
\]
直线与单位圆有公共点，当且仅当圆心到直线的距离不超过半径，即 \(d\leqslant1\). 因此
\[
\left|\cos\left(\alpha+\frac{\pi}{3}\right)\right|\leqslant\frac12.
\]
由于 \(\alpha+\frac{\pi}{3}\in\left[\frac{\pi}{3},\frac{4\pi}{3}\right)\)，上式等价于 \(\alpha+\frac{\pi}{3}\in\left[\frac{\pi}{3},\frac{2\pi}{3}\right]\)，所以 \(\alpha\in\left[0,\frac{\pi}{3}\right]\).
故选 D.
\end{solution}

\begin{problem}
若将函数 \(f(x) = \sin 2x + \cos 2x\) 的图象向右平移 \(\varphi\) 个单位，所得图象关于 \(y\) 轴对称，则 \(\varphi\) 的最小正值是（\quad）

\choices
    {\(\frac{\pi}{8}\)}
    {\(\frac{\pi}{4}\)}
    {\(\frac{3\pi}{8}\)}
    {\(\frac{3\pi}{4}\)}

\begin{answer}
C
\end{answer}
\end{problem}

\begin{solution}
向右平移 \(\varphi\) 个单位后，所得函数为
\[
g(x)=\sin 2(x-\varphi)+\cos 2(x-\varphi)=\sqrt{2}\sin\left(2x-2\varphi+\frac{\pi}{4}\right).
\]
记 \(\delta=\frac{\pi}{4}-2\varphi\)，则
\[
g(x)=\sqrt{2}\sin(2x+\delta)=\sqrt{2}\left(\sin2x\cos\delta+\cos2x\sin\delta\right).
\]
要使 \(g(x)\) 关于 \(y\) 轴对称，必须是偶函数，因此 \(\sin2x\) 的系数为零，即 \(\cos\delta=0\). 于是
\[
\frac{\pi}{4}-2\varphi=\frac{\pi}{2}+k\pi\quad(k\in\Z),
\]
从而
\[
\varphi=-\frac{\pi}{8}-\frac{k\pi}{2}.
\]
取使 \(\varphi>0\) 的最小整数 \(k=-1\)，得 \(\varphi=\frac{3\pi}{8}\).
故选 C.
\end{solution}

\begin{problem}
一个多面体的三视图如图所示，则该多面体的体积为（\quad）

\bitmapfigure[max width=0.46\linewidth]{img/2014/anhui_liberal/q8_fig1.png}

\choices
    {\(\frac{23}{3}\)}
    {\(\frac{47}{6}\)}
    {\(6\)}
    {\(7\)}

\begin{answer}
A
\end{answer}
\end{problem}

\begin{solution}
由三个视图可知，该多面体可看作一个棱长为 \(2\) 的正方体截去两个全等的三棱锥. 每个被截去的三棱锥的底面是直角边长分别为 \(1\) 和 \(1\) 的直角三角形，高为 \(1\)，故其体积为
\[
\frac13\times\left(\frac12\times1\times1\right)\times1=\frac16.
\]
正方体的体积为 \(2^3=8\)，所以多面体的体积为
\[
8-2\times\frac16=\frac{23}{3}.
\]
故选 A.
\end{solution}

\begin{problem}
若函数 \(f(x) = |x + 1| + |2x + a|\) 的最小值为 \(3\)，则实数 \(a\) 的值为（\quad）

\choices
    {\(5\) 或 \(8\)}
    {\(-1\) 或 \(5\)}
    {\(-1\) 或 \(-4\)}
    {\(-4\) 或 \(8\)}

\begin{answer}
D
\end{answer}
\end{problem}

\begin{solution}
将函数改写为
\[
f(x)=|x+1|+2\left|x+\frac{a}{2}\right|.
\]
由三角不等式，
\[
f(x)\geqslant\left|x+1\right|+\left|x+\frac{a}{2}\right|\geqslant\left|1-\frac{a}{2}\right|.
\]
当 \(x=-\frac{a}{2}\) 时，原式取值为 \(\left|1-\frac{a}{2}\right|\)，所以
\[
\min f(x)=\left|1-\frac{a}{2}\right|.
\]
由题意，
\[
\left|1-\frac{a}{2}\right|=3\Longleftrightarrow|a-2|=6,
\]
解得 \(a=8\) 或 \(a=-4\).
故选 D.
\end{solution}

\begin{problem}
设 \(\bs{a}\)，\(\bs{b}\) 为非零向量，\(\left|\bs{b}\right| = 2\left|\bs{a}\right|\)，两组向量 \(\bs{x}_1\)，\(\bs{x}_2\)，\(\bs{x}_3\)，\(\bs{x}_4\) 和 \(\bs{y}_1\)，\(\bs{y}_2\)，\(\bs{y}_3\)，\(\bs{y}_4\) 均由 \(2\text{ 个}\) \(\bs{a}\) 和 \(2\text{ 个}\) \(\bs{b}\) 排列而成，若 \(\bs{x}_1 \cdot \bs{y}_1 + \bs{x}_2 \cdot \bs{y}_2 + \bs{x}_3 \cdot \bs{y}_3 + \bs{x}_4 \cdot \bs{y}_4\) 所有可能取值中的最小值为 \(4\left|\bs{a}\right|^2\)，则 \(\bs{a}\) 与 \(\bs{b}\) 的夹角为（\quad）

\choices
    {\(\frac{2}{3}\pi\)}
    {\(\frac{\pi}{3}\)}
    {\(\frac{\pi}{6}\)}
    {\(0\)}

\begin{answer}
B
\end{answer}
\end{problem}

\begin{solution}
设 \(s=\left|\bs{a}\right|^2>0\)，并记 \(t=\cos\theta\)，其中 \(\theta\) 为 \(\bs{a}\) 与 \(\bs{b}\) 的夹角. 由 \(\left|\bs{b}\right|=2\left|\bs{a}\right|\)，得
 \(\bs{a}\cdot\bs{a}=s\)，\(\bs{b}\cdot\bs{b}=4s\)，\(\bs{a}\cdot\bs{b}=2st\).
固定 \(\bs{x}_1,\bs{x}_2,\bs{x}_3,\bs{x}_4\) 的排列，设对应位置上 \(\bs{x}_i\) 与 \(\bs{y}_i\) 同为 \(\bs{a}\) 的位置数为 \(k\). 因为两组中各有两个 \(\bs{a}\) 和两个 \(\bs{b}\)，同为 \(\bs{b}\) 的位置数也为 \(k\)，而混合位置各有 \(2-k\) 个，其中 \(k=0,1,2\).

因此内积和的可能值分别为 \(8st\)，\(5s+4st\)，\(10s\).
题设说明这三个数的最小值为 \(4s\)，于是
\[
\min\{8t,5+4t,10\}=4.
\]
由于 \(10>4\)，要使最小值恰为 \(4\)，必须有 \(8t=4\)，且另外两个数不小于 \(4\). 这给出 \(t=\frac12\)，并且此时 \(5+4t=7>4\).
所以 \(\cos\theta=\frac12\)，\(\theta=\frac{\pi}{3}\).
故选 B.
\end{solution}

\begin{problem}
\(\left(\frac{16}{81}\right)^{-\frac{3}{4}} + \log_{3}\frac{5}{4} + \log_{3}\frac{4}{5} =\)\fillinblank{}.

\begin{answer}
\(\frac{27}{8}\)
\end{answer}
\end{problem}

\begin{solution}
因为
\[
\left(\frac{16}{81}\right)^{-\frac34}
=\left(\frac{2^4}{3^4}\right)^{-\frac34}
=\left(\frac23\right)^{-3}
=\left(\frac32\right)^3
=\frac{27}{8}.
\]
又因为
\[
\log_3\frac54+\log_3\frac45=\log_3 1=0,
\]
所以原式的值为 \(\frac{27}{8}\).
\end{solution}

\begin{problem}
如图，在等腰直角三角形 \(ABC\) 中，斜边 \(BC = 2\sqrt{2}\)，过点 \(A\) 作 \(BC\) 的垂线，垂足为 \(A_1\)；过点 \(A_1\) 作 \(AC\) 的垂线，垂足为 \(A_2\)；过点 \(A_2\) 作 \(A_1C\) 的垂线，垂足为 \(A_3\)；\(\cdots\)，依此类推，设 \(BA = a_1\)，\(AA_1 = a_2\)，\(A_1A_2 = a_3\)，\(\cdots\)，\(A_5A_6 = a_7\)，则 \(a_7 =\)\fillinblank{}.

\bitmapfigure[max width=0.62\linewidth]{img/2014/anhui_liberal/q12_fig1.png}

\begin{answer}
\(\frac{1}{4}\)
\end{answer}
\end{problem}

\begin{solution}
由 \(BC=2\sqrt2\) 且 \(\triangle ABC\) 为等腰直角三角形，得
\[
BA=AC=2.
\]
斜边上的高也是中线，所以
\[
CA_1=BA_1=\sqrt2,\qquad AA_1=\sqrt2.
\]
在直角三角形 \(CA_1A_2\) 中，\(\angle A_1CA_2=45^\circ\)，故
\[
CA_2=CA_1\cos45^\circ=1,\qquad A_1A_2=CA_1\sin45^\circ=1.
\]
对后续每一个直角三角形 \(CA_{k-1}A_k\)，位于 \(C\) 点的锐角均为 \(45^\circ\)，所以新得到的垂线段等于前一条 \(C\) 到垂足的距离乘以 \(\sin45^\circ\). 于是
\[
CA_3=CA_2\cos45^\circ=\frac{\sqrt2}{2},\qquad
A_2A_3=CA_2\sin45^\circ=\frac{\sqrt2}{2},
\]
\[
CA_4=CA_3\cos45^\circ=\frac12,\qquad
A_3A_4=CA_3\sin45^\circ=\frac12,
\]
\[
CA_5=CA_4\cos45^\circ=\frac{\sqrt2}{4},\qquad
A_4A_5=CA_4\sin45^\circ=\frac{\sqrt2}{4}.
\]
再由直角三角形 \(CA_5A_6\)，得
\[
a_7=A_5A_6=CA_5\sin45^\circ
=\frac{\sqrt2}{4}\cdot\frac{\sqrt2}{2}
=\frac14.
\]
\end{solution}

\begin{problem}
不等式组 \(\begin{cases}
x + y - 2 \geqslant 0,\\
x + 2y - 4 \leqslant 0,\\
x + 3y - 2 \geqslant 0.
\end{cases}\) 表示的平面区域的面积为\fillinblank{}.

\begin{answer}
\(4\)
\end{answer}
\end{problem}

\begin{solution}
记三条边界直线分别为
\[
l_1:x+y=2,\qquad l_2:x+2y=4,\qquad l_3:x+3y=2.
\]
它们两两相交于
\[
l_1\cap l_2=(0,2),\qquad
l_1\cap l_3=(2,0),\qquad
l_2\cap l_3=(8,-2).
\]
这三个交点都满足原不等式组，所以所表示的区域是以它们为顶点的三角形. 取 \((0,2)\) 为参考点，另外两个顶点对应的向量为 \((2,-2)\) 和 \((8,-4)\)，因此区域面积为
\[
\frac12\left|2\times(-4)-(-2)\times8\right|=4.
\]
\end{solution}

\begin{problem}
若函数 \(f(x)\)（\(x \in \R\)）是周期为 \(4\) 的奇函数，且在 \([0, 2]\) 上的解析式为 \(f(x) = \begin{cases}
x(1 - x), & 0 \leqslant x \leqslant 1,\\
\sin \pi x, & 1 < x \leqslant 2.
\end{cases}\) 则 \(f\left(\frac{29}{4}\right) + f\left(\frac{41}{6}\right) =\)\fillinblank{}.

\begin{answer}
\(\frac{5}{16}\)
\end{answer}
\end{problem}

\begin{solution}
由周期性，
\[
f\left(\frac{29}{4}\right)=f\left(\frac{13}{4}\right)=f\left(-\frac34\right).
\]
由奇函数性质，
\[
f\left(-\frac34\right)=-f\left(\frac34\right)
=-\frac34\left(1-\frac34\right)
=-\frac{3}{16}.
\]
同样由周期性，
\[
f\left(\frac{41}{6}\right)=f\left(\frac{17}{6}\right)=f\left(-\frac76\right)
=-f\left(\frac76\right).
\]
因为 \(\frac76\in(1,2]\)，所以
\[
f\left(\frac76\right)=\sin\frac{7\pi}{6}=-\frac12,
\]
从而 \(f\left(\frac{41}{6}\right)=\frac12\). 故
\[
f\left(\frac{29}{4}\right)+f\left(\frac{41}{6}\right)
=-\frac{3}{16}+\frac12=\frac{5}{16}.
\]
\end{solution}

\begin{problem}
若直线 \(l\) 与曲线 \(C\) 满足下列两个条件：

\begin{circlelist}
\item 直线 \(l\) 在点 \(P(x_0, y_0)\) 处与曲线 \(C\) 相切；
\item 曲线 \(C\) 在点 \(P\) 附近位于直线 \(l\) 的两侧，
\end{circlelist}

则称直线 \(l\) 在点 \(P\) 处“切过”曲线 \(C\).

下列命题正确的是\fillinblank{}.（写出所有正确命题的编号）

\begin{circlelist}
\item 直线 \(l: y = 0\) 在点 \(P(0,0)\) 处“切过”曲线 \(C: y = x^3\).
\item 直线 \(l: x = -1\) 在点 \(P(-1,0)\) 处“切过”曲线 \(C: y = (x + 1)^2\).
\item 直线 \(l: y = x\) 在点 \(P(0,0)\) 处“切过”曲线 \(C: y = \sin x\).
\item 直线 \(l: y = x\) 在点 \(P(0,0)\) 处“切过”曲线 \(C: y = \tan x\).
\item 直线 \(l: y = x - 1\) 在点 \(P(1,0)\) 处“切过”曲线 \(C: y = \ln x\).
\end{circlelist}

\begin{answer}
\(\circled{1}\circled{3}\circled{4}\)
\end{answer}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 对曲线 \(y=x^3\)，在 \(x=0\) 处的导数为 \(0\)，故切线为 \(y=0\). 又当 \(x<0\) 时 \(x^3<0\)，当 \(x>0\) 时 \(x^3>0\)，曲线在切点附近位于直线两侧，所以命题 \(\circled{1}\) 正确.
\item 曲线 \(y=(x+1)^2\) 在 \(x=-1\) 处的导数为 \(0\)，切线为 \(y=0\)，不是直线 \(x=-1\). 并且曲线始终位于直线 \(x=-1\) 的右侧，所以命题 \(\circled{2}\) 错误.
\item 曲线 \(y=\sin x\) 在 \(x=0\) 处的导数为 \(1\)，切线为 \(y=x\). 令 \(h(x)=\sin x-x\)，则 \(h'(x)=\cos x-1\leqslant0\)，且 \(h(0)=0\). 因此在 \(0\) 的右侧 \(h(x)<0\)，在左侧 \(h(x)>0\)，曲线位于直线两侧，所以命题 \(\circled{3}\) 正确.
\item 曲线 \(y=\tan x\) 在 \(x=0\) 处的导数为 \(1\)，切线为 \(y=x\). 令 \(h(x)=\tan x-x\)，则 \(h'(x)=\tan^2x\geqslant0\)，且 \(h(0)=0\). 因此在 \(0\) 的右侧 \(h(x)>0\)，在左侧 \(h(x)<0\)，曲线位于直线两侧，所以命题 \(\circled{4}\) 正确.
\item 曲线 \(y=\ln x\) 在 \(x=1\) 处的导数为 \(1\)，切线为 \(y=x-1\). 令 \(h(x)=\ln x-x+1\)，则 \(h'(x)=\frac1x-1\)，所以 \(h(x)\) 在 \(x=1\) 处取得最大值 \(0\)，在 \(x=1\) 附近除切点外均有 \(h(x)<0\). 曲线只位于直线的一侧，所以命题 \(\circled{5}\) 错误.
\end{enumerate}
因此正确命题的编号为 \(\circled{1}\)、\(\circled{3}\)、\(\circled{4}\).
\end{solution}

\begin{problem}
设 \(\triangle ABC\) 的内角 \(A\)，\(B\)，\(C\) 所对边的长分别是 \(a\)，\(b\)，\(c\)，且 \(b = 3\)，\(c = 1\)，\(\triangle ABC\) 的面积为 \(\sqrt{2}\)，求 \(\cos A\) 与 \(a\) 的值.
\end{problem}

\begin{solution}
由三角形面积公式，得
\[
\frac{1}{2}bc\sin A=\sqrt{2}.
\]
代入 \(b=3\)，\(c=1\)，可得
\[
\sin A=\frac{2\sqrt{2}}{3}.
\]
因为 \(0<A<\pi\)，所以
\[
\cos A=\pm\sqrt{1-\sin^2 A}=\pm\frac{1}{3}.
\]

由余弦定理，得
\[
a^2=b^2+c^2-2bc\cos A=10-6\cos A.
\]
当 \(\cos A=\frac{1}{3}\) 时，\(a^2=8\)，所以 \(a=2\sqrt{2}\)；当 \(\cos A=-\frac{1}{3}\) 时，\(a^2=12\)，所以 \(a=2\sqrt{3}\).
当 \(a=2\sqrt{2}\) 或 \(a=2\sqrt{3}\) 时，都有
\[
|b-c|=2<a<4=b+c,
\]
所以两组值均满足三角形的边长条件，故
\[
(\cos A,a)=\left(\frac{1}{3},2\sqrt{2}\right)\quad\text{或}\quad\left(-\frac{1}{3},2\sqrt{3}\right).
\]
\end{solution}

\begin{problem}
某高校共有学生 \(15000\text{ 人}\)，其中男生 \(10500\text{ 人}\)，女生 \(4500\text{ 人}\)，为调查该校学生每周平均体育运动时间的情况，采用分层抽样的方法，收集 \(300\text{ 位}\) 学生每周平均体育运动时间的样本数据（单位：\(\text{小时}\)）.

\begin{enumerate}
\item 应收集多少位女生样本数据？
\item 根据这 \(300\text{ 个}\)样本数据，得到学生每周平均体育运动时间的频率分布直方图（如图所示），其中样本数据分组区间为：\([0,2]\)，\((2,4]\)，\((4,6]\)，\((6,8]\)，\((8,10]\)，\((10,12]\). 估计该校学生每周平均体育运动时间超过 \(4\text{ 个小时}\) 的概率.

300\times0.75=225.

\begin{problem}
数列 \(\{a_{n}\}\) 满足 \(a_1 = 1\)，\(na_{n+1} = (n + 1)a_n + n(n + 1)\)，\(n \in \N^*\).

\begin{enumerate}
\item 证明：数列 \(\left\{\frac{a_{n}}{n}\right\}\) 是等差数列；
\item 设 \(b_n = 3^n \cdot \sqrt{a_n}\)，求数列 \(\{b_n\}\) 的前 \(n\) 项和 \(S_n\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 将递推式两边同除以 \(n(n+1)\)，得
\[
\frac{a_{n+1}}{n+1}=\frac{a_n}{n}+1.
\]
令 \(c_n=\frac{a_n}{n}\)，则
\[
c_{n+1}-c_n=1.
\]
所以数列 \(\{c_n\}\)，即数列 \(\left\{\frac{a_n}{n}\right\}\)，是以 \(c_1=\frac{a_1}{1}=1\) 为首项、以 \(1\) 为公差的等差数列.

\item 由第（1）问，得
\[
\frac{a_n}{n}=1+(n-1)=n,
\]
所以 \(a_n=n^2\). 因为 \(n\in\N^*\)，故
\[
b_n=3^n\sqrt{a_n}=3^n\sqrt{n^2}=n3^n.
\]
于是
\[
S_n=\sum_{k=1}^{n}k3^k.
\]
将上式乘以 \(3\)，得
\[
3S_n=\sum_{k=1}^{n}k3^{k+1}.
\]
两式相减，得
\[
2S_n=n3^{n+1}-3-\sum_{k=2}^{n}3^k.
\]
又
\[
\sum_{k=2}^{n}3^k=\frac{3^{n+1}-9}{2},
\]
所以
\[
2S_n=n3^{n+1}-3-\frac{3^{n+1}-9}{2}
=\frac{(2n-1)3^{n+1}+3}{2}.
\]
因此
\[
S_n=\frac{(2n-1)3^{n+1}+3}{4}.
\]
\end{enumerate}
\end{solution}


\begin{problem}
如图，四棱锥 \(P - ABCD\) 的底面是边长为 \(8\) 的正方形，四条侧棱长均为 \(2\sqrt{17}\)，点 \(G\)，\(E\)，\(F\)，\(H\) 分别是棱 \(PB\)，\(AB\)，\(CD\)，\(PC\) 上共面的四点，平面 \(GEFH \perp\) 平面 \(ABCD\)，\(BC \parallel\) 平面 \(GEFH\).

\begin{enumerate}
\item 证明：\(GH \parallel EF\)；
\item 若 \(EB = 2\)，求四边形 \(GEFH\) 的面积.
\end{enumerate}

\bitmapfigure[max width=0.62\linewidth]{img/2014/anhui_liberal/q19_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 建立空间直角坐标系，令
\[
A(0,0,0),\quad B(8,0,0),\quad C(8,8,0),\quad D(0,8,0).
\]
设正方形中心为 \(O\). 因为 \(PA=PB=PC=PD=2\sqrt{17}\)，所以 \(P\) 在过 \(O\) 且垂直于底面的直线上. 又 \(OA=4\sqrt{2}\)，故
\[
PO=\sqrt{PA^2-OA^2}=\sqrt{68-32}=6.
\]
取 \(P(4,4,6)\).

记平面 \(GEFH\) 为 \(\alpha\). 平面 \(\alpha\) 垂直于底面，且 \(BC\parallel\alpha\)，所以 \(\alpha\) 是一个与 \(y\) 轴方向平行的竖直平面，可设其方程为 \(x=t\). 于是
\[
E(t,0,0),\qquad F(t,8,0).
\]
设 \(G\)，\(H\) 分别为直线段 \(PB\)，\(PC\) 与平面 \(\alpha\) 的交点. 由
\[
G=P+\lambda(B-P),\qquad H=P+\mu(C-P),
\]
以及 \(x_G=x_H=t\)，得
\[
4+4\lambda=4+4\mu=t,
\]
所以 \(\lambda=\mu\). 因而
\[
\overrightarrow{GH}
=\bigl(0,8\lambda,0\bigr),\qquad
\overrightarrow{EF}=(0,8,0).
\]
故 \(GH\parallel EF\).

\item 由 \(E(t,0,0)\) 和 \(B(8,0,0)\)，得
\[
EB=8-t=2,
\]
所以 \(t=6\)，从而
\[
\lambda=\frac{t-4}{4}=\frac{1}{2}.
\]
于是
\[
E(6,0,0),\quad F(6,8,0),\quad G(6,2,3),\quad H(6,6,3).
\]
在平面 \(x=6\) 内，四边形 \(GEFH\) 是以上底 \(GH=4\)，下底 \(EF=8\)，高为 \(3\) 的梯形，所以
\[
S_{GEFH}=\frac{4+8}{2}\times3=18.
\]
\end{enumerate}
\end{solution}

\begin{problem}
设函数 \(f(x) = 1 + (1 + a)x - x^2 - x^3\)，其中 \(a > 0\).

\begin{enumerate}
\item 讨论 \(f(x)\) 在其定义域上的单调性；
\item 当 \(x \in [0,1]\) 时，求 \(f(x)\) 取得最大值和最小值时的 \(x\) 的值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 函数的定义域为 \(\R\)，且
\[
f'(x)=1+a-2x-3x^2=-3x^2-2x+1+a.
\]
令 \(f'(x)=0\)，得
\[
3x^2+2x-(1+a)=0.
\]
因为 \(a>0\)，所以该方程的两个根为
\[
\alpha=\frac{-1-\sqrt{4+3a}}{3}<0,\qquad
\beta=\frac{-1+\sqrt{4+3a}}{3}>0.
\]
于是
\[
f'(x)=-3(x-\alpha)(x-\beta).
\]
因此，\(f'(x)\) 在 \((-\infty,\alpha)\) 上为负，在 \((\alpha,\beta)\) 上为正，在 \((\beta,+\infty)\) 上为负. 所以 \(f(x)\) 的单调性为：在 \((-\infty,\alpha]\) 上单调递减，在 \([\alpha,\beta]\) 上单调递增，在 \([\beta,+\infty)\) 上单调递减.

\item 因为 \(\alpha<0\)，只需考察 \(\beta\) 与 \(1\) 的大小关系. 由
\[
\beta\leqslant1\Longleftrightarrow\sqrt{4+3a}\leqslant4\Longleftrightarrow a\leqslant4,
\]
可知：

当 \(0<a<4\) 时，\(f(x)\) 在 \([0,\beta]\) 上递增，在 \([\beta,1]\) 上递减，所以最大值在 \(x=\beta\) 时取得；当 \(a\geqslant4\) 时，\(f(x)\) 在 \([0,1]\) 上递增，所以最大值在 \(x=1\) 时取得.

再比较两个端点的函数值：
\[
f(0)=1,\qquad f(1)=a.
\]
所以最小值取得时的 \(x\) 为
\[
\begin{cases}
1, & 0<a<1,\\
0\text{ 或 }1, & a=1,\\
0, & a>1.
\end{cases}
\]
综上，最大值取得时的 \(x\) 为
\[
\begin{cases}
\dfrac{-1+\sqrt{4+3a}}{3}, & 0<a<4,\\
1, & a\geqslant4.
\end{cases}
\]
\end{enumerate}
\end{solution}

\begin{problem}
设 \(F_1\)，\(F_2\) 分别是椭圆 \(E: \frac{x^2}{a^2} + \frac{y^2}{b^2} = 1\) （\(a > b > 0\)）的左，右焦点，过点 \(F_1\) 的直线交椭圆 \(E\) 于 \(A\)，\(B\) 两点，\(\left|AF_1\right| = 3\left|BF_1\right|\).

\begin{enumerate}
\item 若 \(\left|AB\right| = 4\)，\(\triangle ABF_2\) 的周长为 \(16\)，求 \(\left|AF_2\right|\)；
\item 若 \(\cos \angle AF_2B = \frac{3}{5}\)，求椭圆 \(E\) 的离心率.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 焦点 \(F_1\) 在椭圆内部，所以直线 \(AB\) 与椭圆交于 \(A\)，\(B\) 两点时，\(F_1\) 在线段 \(AB\) 上. 设
\[
|BF_1|=x,
\]
则
\[
|AF_1|=3x,\qquad |AB|=|AF_1|+|BF_1|=4x.
\]
由 \(|AB|=4\)，得 \(x=1\)，所以
\[
|AF_1|=3,\qquad |BF_1|=1.
\]
由椭圆定义，得
\[
|AF_1|+|AF_2|=2a,\qquad |BF_1|+|BF_2|=2a.
\]
所以
\[
|AF_2|=2a-3,\qquad |BF_2|=2a-1.
\]
由 \(\triangle ABF_2\) 的周长为 \(16\)，得
\[
4+(2a-3)+(2a-1)=16,
\]
即 \(4a=16\)，所以 \(a=4\). 因此
\[
|AF_2|=2a-3=5.
\]

\item 设 \(|BF_1|=k\)，则
\[
|AF_1|=3k,\qquad |AB|=4k.
\]
由椭圆定义，得
\[
|AF_2|=2a-3k,\qquad |BF_2|=2a-k.
\]
在 \(\triangle ABF_2\) 中，由余弦定理和 \(\cos\angle AF_2B=\frac{3}{5}\)，得
\[
(4k)^2=(2a-3k)^2+(2a-k)^2-2(2a-3k)(2a-k)\frac{3}{5}.
\]
化简得
\[
(a-3k)(a+k)=0.
\]
因为 \(a>0\)，\(k>0\)，所以 \(a=3k\). 于是
\[
|AF_1|=3k=a,\qquad |AF_2|=2a-3k=3k=a,
\]
且
\[
|BF_2|=2a-k=5k,\qquad |AB|=4k.
\]
因此
\[
|BF_2|^2=|AF_2|^2+|AB|^2,
\]
由勾股定理的逆定理，得 \(AF_2\perp AB\). 又 \(F_1\) 在线段 \(AB\) 上，所以 \(AF_2\perp AF_1\). 在直角三角形 \(AF_1F_2\) 中，
\[
|AF_1|=|AF_2|=a,\qquad |F_1F_2|=2c.
\]
故
\[
(2c)^2=a^2+a^2,
\]
从而
\[
\frac{c}{a}=\frac{\sqrt{2}}{2}.
\]
所以椭圆 \(E\) 的离心率为 \(\frac{\sqrt{2}}{2}\).
\end{enumerate}
\end{solution}
