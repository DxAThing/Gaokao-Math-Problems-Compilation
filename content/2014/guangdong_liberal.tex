\section{单选题}

\begin{problem}
已知集合 \(M = \{2, 3, 4\}\)，\(N = \{0, 2, 3, 5\}\)，则 \(M \cap N\)（\quad）

\choices
    {\(\{0, 2\}\)}
    {\(\{2, 3\}\)}
    {\(\{3, 4\}\)}
    {\(\{3, 5\}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
交集中的元素必须同时属于两个集合.集合 \(M\) 的元素为 \(2\)，\(3\)，\(4\)，其中 \(2,3\) 也属于 \(N\)，而 \(4\notin N\)，所以
\[
M\cap N=\{2,3\}
\]
因此选 B.
\end{solution}

\begin{problem}
已知复数 \(z\) 满足 \((3 - 4\i)z = 25\)，则 \(z =\)（\quad）

\choices
    {\(-3 - 4\i\)}
    {\(-3 + 4\i\)}
    {\(3 - 4\i\)}
    {\(3 + 4\i\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
由题设
\[
z=\frac{25}{3-4\i}=\frac{25(3+4\i)}{(3-4\i)(3+4\i)}
 =\frac{25(3+4\i)}{3^2+4^2}=3+4\i
\]
因此选 D.
\end{solution}

已知向量 \(\bs{a} = (1, 2)\)，\(\bs{b} = (3, 1)\)，则 \(\bs{b} - \bs{a} =\)（\quad）

    {\((4, 3)\)}
\end{problem}

B
\end{answer}

\begin{solution}
按坐标相减，得

\bs{b}-\bs{a}=(3-1,1-2)=(2,-1)
\]
因此选 B.
\end{solution}

\begin{problem}
若变量 \(x\)，\(y\) 满足约束条件 \(\begin{cases}
x + 2y \leqslant 8,\\
0 \leqslant x \leqslant 4,\\
0 \leqslant y \leqslant 3,
\end{cases}\) 则 \(z = 2x + y\) 的最大值等于（\quad）

\choices
    {7}
    {8}
    {10}
    {11}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
可行域是由题设不等式确定的多边形，其顶点为
\((0,0)\)，\(\ (4,0)\)，\(\ (4,2)\)，\(\ (2,3)\)，\(\ (0,3)\).
由于 \(z=2x+y\) 是线性函数，只需比较各顶点处的值：
\(0\)，\(8\)，\(10\)，\(7\)，\(3\).
最大值为 \(10\)，因此选 C.
\end{solution}

\begin{problem}
下列函数为奇函数的是（\quad）

\choices
    {\(2^{x} - \frac{1}{2^{x}}\)}
    {\(x^{3}\sin x\)}
    {\(2\cos x + 1\)}
    {\(x^{2} + 2^{x}\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
逐项检验奇偶性.对选项 A，
\[
f(-x)=2^{-x}-2^x=-(2^x-2^{-x})=-f(x)
\]
且定义域为 \(\R\)，所以它是奇函数.

选项 B 中，\(x^3\) 与 \(\sin x\) 都是奇函数，乘积是偶函数；选项 C 有常数项，且
\(2\cos(-x)+1=2\cos x+1\)，是偶函数；选项 D 既不是奇函数也不是偶函数.因此选 A.
\end{solution}

\begin{problem}
为了解 1000 名学生的学习情况，采用系统抽样的方法，从中抽取容量为 40 的样本，则分段的间隔为（\quad）

\choices

\begin{answer}

\begin{problem}
在 \(\triangle ABC\) 中，角 \(A\)，\(B\)，\(C\) 所对应的边分别为 \(a\)，\(b\)，\(c\)，则 “\(a \leqslant b\)” 是 “\(\sin A \leqslant \sin B\)” 的（\quad）

\choices
    {充分必要条件}
    {充分非必要条件}
    {必要非充分条件}
    {非充分非必要条件}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
由正弦定理，存在同一个正数 \(2R\)，使得
\(a=2R\sin A\)，\(b=2R\sin B\).
因为 \(2R>0\)，所以
\[
a\leqslant b\iff 2R\sin A\leqslant 2R\sin B
\iff \sin A\leqslant\sin B
\]
两个不等式互相推出，故这是充分必要条件，选 A.
\end{solution}

\begin{problem}
若实数 \(k\) 满足 \(0 < k < 5\)，则曲线 \(\frac{x^{2}}{16} - \frac{y^{2}}{5-k} = 1\) 与曲线 \(\frac{x^{2}}{16-k} - \frac{y^{2}}{5} = 1\) 的（\quad）

\choices
    {实半轴长相等}
    {虚半轴长相等}
    {离心率相等}
    {焦距相等}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
两条曲线都是焦点在 \(x\) 轴上的双曲线.第一条双曲线满足
\(a_1^2=16\)，\(b_1^2=5-k\)
所以
\[
c_1^2=a_1^2+b_1^2=16+5-k=21-k
\]
第二条双曲线满足
\(a_2^2=16-k\)，\(b_2^2=5\)
从而
\[
c_2^2=a_2^2+b_2^2=16-k+5=21-k
\]
因此 \(c_1=c_2\)，两条双曲线的焦距 \(2c\) 相等，选 D.
\end{solution}

\begin{problem}
若空间中四条两两不同的直线 \(l_{1}\)，\(l_{2}\)，\(l_{3}\)，\(l_{4}\)，满足 \(l_{1} \perp l_{2}\)，\(l_{2} \parallel l_{3}\)，\(l_{3} \perp l_{4}\)，则下列结论一定正确的是（\quad）

\choices
    {\(l_{1} \perp l_{4}\)}
    {\(l_{1} \parallel l_{4}\)}
    {\(l_{1}\)，\(l_{4}\) 既不垂直也不平行}
    {\(l_{1}\)，\(l_{4}\) 的位置关系不确定}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
由 \(l_2\parallel l_3\) 及 \(l_1\perp l_2\)，可知 \(l_1\perp l_3\).但两条都垂直于 \(l_3\) 的直线，在空间中的位置关系仍可能不同.

例如取 \(l_2\) 与 \(l_3\) 为两条不同的平行 \(x\) 轴，取 \(l_1\) 为 \(y\) 轴；可以取 \(l_4\) 为与 \(l_1\) 平行且与 \(l_3\) 相交的直线，此时 \(l_1\parallel l_4\).也可以取 \(l_4\) 为与 \(l_3\) 相交的 \(z\) 轴，此时 \(l_1\perp l_4\).因此 \(l_1\) 与 \(l_4\) 的位置关系不确定，选 D.
\end{solution}

\begin{problem}
对任意复数 \(w_{1}\)，\(w_{2}\)，定义 \(w_{1} * w_{2} = w_{1} \overline{w}_{2}\). 其中 \(\overline{w}_{2}\) 是 \(w_{2}\) 的共轭复数. 对任意复数 \(z_{1}\)，\(z_{2}\)，\(z_{3}\) 有如下四个命题：

\begin{circlelist}
\item \((z_{1} + z_{2}) * z_{3} = (z_{1} * z_{3}) + (z_{2} * z_{3})\)；
\item \(z_{1} * (z_{2} + z_{3}) = (z_{1} * z_{2}) + (z_{1} * z_{3})\)；
\item \((z_{1} * z_{2}) * z_{3} = z_{1} * (z_{2} * z_{3})\)；
\item \(z_{1} * z_{2} = z_{2} * z_{1}\).
\end{circlelist}

则真命题的个数是（\quad）

\choices
    {1}
    {2}
    {3}
    {4}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
由共轭运算的加法性质，
\[
(z_1+z_2)*z_3=(z_1+z_2)\overline{z_3}
 =z_1\overline{z_3}+z_2\overline{z_3}
 =(z_1*z_3)+(z_2*z_3)
\]
所以命题 1 正确.

同理，
\[
z_1*(z_2+z_3)=z_1\overline{z_2+z_3}
 =z_1(\overline{z_2}+\overline{z_3})
 =(z_1*z_2)+(z_1*z_3)
\]
所以命题 2 正确.

对命题 3，左边和右边分别为
\((z_1*z_2)*z_3=z_1\overline{z_2}\,\overline{z_3}\)，\(z_1*(z_2*z_3)=z_1\overline{z_2\overline{z_3}}=z_1\overline{z_2}z_3\).
取 \(z_1=z_2=1\)，\(z_3=\i\)，两式分别为 \(-\i\) 和 \(\i\)，故命题 3 不正确.

对命题 4，取 \(z_1=1\)，\(z_2=\i\)，则
\(z_1*z_2=-\i\)，\(z_2*z_1=\i\)
故命题 4 不正确.真命题共有 2 个，选 B.
\end{solution}

\begin{problem}
曲线 \(y = -5\e^{x} + 3\) 在点 \((0, -2)\) 处的切线方程为\fillinblank{}.
\end{problem}

\begin{answer}
\(5x + y + 2 = 0\)
\end{answer}

\begin{solution}
函数的导数为
\[
y'=-5\e^x
\]
在 \(x=0\) 处，切线斜率为 \(y'(0)=-5\)，且切点为 \((0,-2)\).由点斜式，切线方程为
\[
y+2=-5(x-0)
\]
即
\[
5x+y+2=0
\]
\end{solution}

\begin{problem}
从字母 \(a\)，\(b\)，\(c\)，\(d\)，\(e\) 中任取两个不同字母，则取到字母 \(a\) 的概率为\fillinblank{}.
\end{problem}

\begin{answer}
\(\dfrac{2}{5}\)
\end{answer}

\begin{solution}
任取两个不同字母的基本结果数为
\[
\mathrm C_5^2=10
\]
取到字母 \(a\) 时，另一个字母可从其余 4 个字母中任选一个，所以有
\[
\mathrm C_4^1=4
\]
个结果.故所求概率为
\[
\frac{4}{10}=\frac{2}{5}
\]
\end{solution}

\begin{problem}
等比数列 \(\{a_{n}\}\) 的各项均为正数，且 \(a_{1}a_{5} = 4\)，则 \(\log_{2} a_{1} + \log_{2} a_{2} + \log_{2} a_{3} + \log_{2} a_{4} + \log_{2} a_{5} =\fillinblank{}\).
\end{problem}

\begin{answer}
5
\end{answer}

\begin{solution}
设公比为 \(q>0\).等比数列的对称项满足
\[
a_1a_5=a_3^2=4
\]
由于各项为正数，故 \(a_3=2\).另外
\[
a_2a_4=a_3^2=4
\]
于是
\[
a_1a_2a_3a_4a_5=(a_1a_5)(a_2a_4)a_3=4\cdot4\cdot2=32
\]
因此
\[
\sum_{k=1}^{5}\log_2a_k
=\log_2\left(a_1a_2a_3a_4a_5\right)
=\log_2 32=5
\]
\end{solution}

\begin{problem}
在极坐标系中，曲线 \(C_{1}\) 和 \(C_{2}\) 的方程分别为 \(2\rho \cos^{2}\theta = \sin\theta\) 和 \(\rho \cos\theta = 1\)，以极点为平面直角坐标系的原点，极轴为 \(x\) 轴的正半轴，建立平面直角坐标系，则曲线 \(C_{1}\) 和 \(C_{2}\) 的交点的直角坐标为\fillinblank{}.
\end{problem}

\begin{answer}
\((1, 2)\)
\end{answer}

\begin{solution}
由极坐标与直角坐标的关系
\(x=\rho\cos\theta\)，\(y=\rho\sin\theta\).
由第二条曲线 \(\rho\cos\theta=1\) 可知 \(\rho\ne0\)，所以可以将第一条曲线的方程乘以 \(\rho\)，得
\[
2\rho^2\cos^2\theta=\rho\sin\theta
\]
即
\[
2x^2=y
\]
第二条曲线的方程就是
\[
x=\rho\cos\theta=1
\]
联立得 \(x=1\)，\(y=2\)，所以交点的直角坐标为 \((1,2)\).
\end{solution}

如图，在平行四边形 \(ABCD\) 中，点 \(E\) 在 \(AB\) 上，且 \(EB = 2AE\)，\(AC\) 与 \(DE\) 交于点 \(F\)，则 \(\triangle CDF\) 的周长与 \(\triangle AEF\) 的周长之比为\fillinblank{}.

\bitmapfigure[max width=0.34\linewidth]{img/2014/guangdong_liberal/q15_fig1.png}
\end{problem}

\begin{answer}
3
\end{answer}

\begin{solution}
由 \(EB=2AE\)，可得
\[
AB=AE+EB=3AE
\]
在平行四边形中，\(CD\parallel AB\)，且 \(C\)，\(F\)，\(A\) 共线、\(D\)，\(F\)，\(E\) 共线.因此

\frac{CD}{AE}=\frac{AB}{AE}=3

\frac{CD+DF+FC}{AE+EF+FA}=3
\]
\end{solution}

\begin{problem}
已知函数 \(f(x) = A \sin \left( x + \frac{\pi}{3} \right)\)，\(x \in \R\)，且 \(f\left( \frac{5\pi}{12} \right) = \frac{3\sqrt{2}}{2}\).

\begin{enumerate}
\item 求 \(A\) 的值；
\item 若 \(f(\theta) - f(-\theta) = \sqrt{3}\)，\(\theta \in \left(0, \frac{\pi}{2}\right)\)，求 \(f\left(\frac{\pi}{6} - \theta\right)\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由
\[
f\left(\frac{5\pi}{12}\right)
=A\sin\left(\frac{5\pi}{12}+\frac{\pi}{3}\right)
=A\sin\frac{3\pi}{4}=\frac{A\sqrt{2}}{2}
\]
以及题设值 \(\frac{3\sqrt{2}}{2}\)，得 \(A=3\).

\item 由和差化积公式，
\[
\begin{aligned}
f(\theta)-f(-\theta)
&=3\left[\sin\left(\theta+\frac{\pi}{3}\right)-\sin\left(\frac{\pi}{3}-\theta\right)\right]\\
&=3\cdot2\cos\frac{\pi}{3}\sin\theta
=3\sin\theta.
\end{aligned}
\]
所以 \(3\sin\theta=\sqrt{3}\)，即
\[
\sin\theta=\frac{\sqrt{3}}{3}
\]
由于 \(0<\theta<\frac{\pi}{2}\)，有
\[
\cos\theta=\sqrt{1-\sin^2\theta}=\frac{\sqrt{6}}{3}
\]
于是
\[
f\left(\frac{\pi}{6}-\theta\right)
=3\sin\left(\frac{\pi}{6}-\theta+\frac{\pi}{3}\right)
=3\sin\left(\frac{\pi}{2}-\theta\right)
=3\cos\theta=\sqrt{6}
\]
\end{enumerate}
\end{solution}

\begin{problem}
如图 1，四边形 \(ABCD\) 为矩形，\(PD \perp\) 平面 \(ABCD\)，\(AB = 1\)，\(BC = PC = 2\)，作如图 2 折叠，折痕 \(EF \parallel DC\). 其中点 \(E\)，\(F\) 分别在线段 \(PD\)，\(PC\) 上，沿 \(EF\) 折叠后点 \(P\) 叠在线段 \(AD\) 上的点记为 \(M\)，并且 \(MF \perp CF\).

\begin{enumerate}
\item 证明：\(CF \perp\) 平面 \(MDF\)；
\item 求三棱锥 \(M-CDE\) 的体积.
\end{enumerate}

\end{problem}

\begin{solution}
\begin{enumerate}
\item 因为 \(PD\perp\) 平面 \(ABCD\)，所以 \(MD\perp PD\).又因为 \(ABCD\) 是矩形，\(AD\perp DC\)，而 \(MD\subset AD\)，所以 \(MD\perp DC\).直线 \(PD\) 和 \(DC\) 是平面 \(PCD\) 内的相交直线，故
\[
MD\perp\text{平面 }PCD
\]
由于 \(CF\subset\text{平面 }PCD\)，所以 \(CF\perp MD\).结合已知的 \(MF\perp CF\)，且 \(MD\) 与 \(MF\) 是平面 \(MDF\) 内相交的两条直线，得到
\[
CF\perp\text{平面 }MDF
\]

\[
\angle DCF=\angle DCP=60^\circ
\]
在直角三角形 \(CDF\) 中，\(CD=1\)，于是
\(CF=CD\cos60^\circ=\frac12\)，\(DF=CD\sin60^\circ=\frac{\sqrt3}{2}\).
因为 \(EF\parallel DC\)，所以 \(\triangle PEF\sim\triangle PDC\).由 \(PF=PC-CF=\frac32\)，得
\(\frac{PE}{PD}=\frac{PF}{PC}=\frac34\)，\(PE=\frac{3\sqrt3}{4}\)，\(DE=PD-PE=\frac{\sqrt3}{4}\).
折叠保持折痕上点 \(E\) 到点 \(P\) 的距离不变，故 \(ME=PE=\frac{3\sqrt3}{4}\).又 \(MD\subset AD\)，\(DE\subset PD\)，且 \(PD\perp\) 平面 \(ABCD\)，所以 \(MD\perp DE\).在直角三角形 \(MDE\) 中，
\[
MD=\sqrt{ME^2-DE^2}
=\sqrt{\left(\frac{3\sqrt3}{4}\right)^2-\left(\frac{\sqrt3}{4}\right)^2}
=\sqrt{\frac32}=\frac{\sqrt6}{2}
\]
由 \(CD\perp DE\)，三角形 \(CDE\) 的面积为
\[
S_{\triangle CDE}=\frac12\,CD\cdot DE=\frac{\sqrt3}{8}
\]
第（1）问已证 \(MD\perp\) 平面 \(PCD\)，而三角形 \(CDE\) 在平面 \(PCD\) 内，故 \(MD\) 是三棱锥 \(M-CDE\) 的高.因此
\[
V_{M-CDE}=\frac13S_{\triangle CDE}\cdot MD
=\frac13\cdot\frac{\sqrt3}{8}\cdot\frac{\sqrt6}{2}
=\frac{\sqrt2}{16}
\]
\end{enumerate}
\end{solution}

\begin{problem}
设各项均为正数的数列 \(\{a_{n}\}\) 的前 \(n\) 项和为 \(S_{n}\)，且 \(S_{n}\) 满足 \(S_{n}^{2} - (n^{2} + n - 3)S_{n} - 3(n^{2} + n) = 0\)，\(n \in \N^{*}\).

\begin{enumerate}
\item 求 \(a_{1}\) 的值；
\item 求数列 \(\{a_{n}\}\) 的通项公式；
\item 证明：对一切正整数 \(n\)，有 \(\frac{1}{a_1(a_1 + 1)} + \frac{1}{a_2(a_2 + 1)} + \cdots + \frac{1}{a_n(a_n + 1)} < \frac{1}{3}\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 令
\[
Q_n(S)=S^2-(n^2+n-3)S-3(n^2+n)
\]
注意到
\[
Q_n\bigl(n(n+1)\bigr)=0
\]
并且将二次方程因式分解为
\[
\left(S_n-n(n+1)\right)(S_n+3)=0
\]
因为各项均为正数，所以 \(S_n>0\)，不能取 \(S_n=-3\)，故
\[
S_n=n(n+1)
\]
取 \(n=1\)，得 \(S_1=2\)，所以 \(a_1=2\).

\item 当 \(n\geqslant2\) 时，
\[
a_n=S_n-S_{n-1}=n(n+1)-(n-1)n=2n
\]
而 \(a_1=2=2\times1\)，所以对所有正整数 \(n\)，
\[
a_n=2n
\]

\item 由上式，需证明
\[
\sum_{k=1}^{n}\frac{1}{2k(2k+1)}<\frac13
\]
当 \(n=1\) 时，左边为 \(\frac16<\frac13\).当 \(n\geqslant2\) 时，对每个 \(k\geqslant2\)，有
\[
\frac{1}{2k(2k+1)}
<\frac{1}{(2k-1)(2k+1)}
=\frac12\left(\frac{1}{2k-1}-\frac{1}{2k+1}\right)
\]
因此
\[
\begin{aligned}
\sum_{k=1}^{n}\frac{1}{2k(2k+1)}
&<\frac16+\frac12\sum_{k=2}^{n}\left(\frac{1}{2k-1}-\frac{1}{2k+1}\right)\\
&=\frac16+\frac12\left(\frac13-\frac{1}{2n+1}\right)\\
&<\frac16+\frac16=\frac13.
\end{aligned}
\]
所以不等式对一切正整数 \(n\) 成立.
\end{enumerate}
\end{solution}

\begin{problem}
已知椭圆 \(C: \frac{x^2}{a^2} + \frac{y^2}{b^2} = 1\)（\(a > b > 0\)）的一个焦点为 \((\sqrt{5}, 0)\)，离心率为 \(\frac{\sqrt{5}}{3}\).

\begin{enumerate}
\item 求椭圆 \(C\) 的标准方程；
\item 若动点 \(P(x_0, y_0)\) 为椭圆 \(C\) 外一点，且点 \(P\) 到椭圆 \(C\) 的两条切线互相垂直，求点 \(P\) 的轨迹方程.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设椭圆的半焦距为 \(c\).由焦点坐标得 \(c=\sqrt{5}\)，再由离心率
\[
e=\frac{c}{a}=\frac{\sqrt{5}}{3}
\]
得 \(a=3\).由椭圆关系 \(b^2=a^2-c^2\)，得
\[
b^2=9-5=4
\]
所以椭圆的标准方程为
\[
\frac{x^2}{9}+\frac{y^2}{4}=1
\]

\item 设过点 \(P(x_0,y_0)\) 的一条非竖直直线为
\(y=kx+b\)，\(b=y_0-kx_0\).
将其代入椭圆方程，得到关于 \(x\) 的二次方程
\[
\frac{x^2}{9}+\frac{(kx+b)^2}{4}=1
\]
直线与椭圆相切的充要条件是该方程的判别式为零，化简得
\[
b^2=9k^2+4
\]
所以两条切线的斜率 \(k_1\)，\(k_2\) 是方程
\[
(x_0^2-9)k^2-2x_0y_0k+(y_0^2-4)=0
\]
的两个根.当 \(x_0^2\ne9\) 时，由两切线垂直，\(k_1k_2=-1\)，由韦达定理得
\[
\frac{y_0^2-4}{x_0^2-9}=-1
\]
即
\[
x_0^2+y_0^2=13
\]

当 \(x_0^2=9\) 时，一条切线为竖直线 \(x=x_0\).要与另一条切线垂直，另一条必须是水平切线 \(y=2\) 或 \(y=-2\)，于是 \(y_0^2=4\)，仍满足 \(x_0^2+y_0^2=13\).反过来，圆 \(x_0^2+y_0^2=13\) 上的点均在椭圆外；除上述竖直切线的特殊情形外，由上面的斜率积为 \(-1\) 可知两切线互相垂直.因此轨迹方程为
\[
x_0^2+y_0^2=13
\]
\end{enumerate}
\end{solution}

\begin{problem}
已知函数 \(f(x) = \frac{1}{3}x^3 + x^2 + ax + 1\)（\(a \in \R\)）.

\begin{enumerate}
\item 求函数 \(f(x)\) 的单调区间；
\item 当 \(a < 0\) 时，试讨论是否存在 \(x_0 \in \left(0, \frac{1}{2}\right) \cup \left(\frac{1}{2}, 1\right)\)，使得 \(f(x_0) = f\left(\frac{1}{2}\right)\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 求导得
\[
f'(x)=x^2+2x+a=(x+1)^2+a-1
\]
当 \(a\geqslant1\) 时，\(f'(x)\geqslant0\)，且导数不在任何区间上恒为零，所以函数在 \(\R\) 上单调递增.

当 \(a<1\) 时，令 \(f'(x)=0\)，得两个根
\(x_1=-1-\sqrt{1-a}\)，\(x_2=-1+\sqrt{1-a}\).
由于 \(f'(x)\) 的二次项系数为正，故
\[
f'(x)>0\quad\text{在 }(-\infty,x_1)\text{ 和 }(x_2,+\infty)
\]
\[
f'(x)<0\quad\text{在 }(x_1,x_2)
\]
因此函数的单调递增区间为
\[
\left(-\infty,-1-\sqrt{1-a}\right)\quad\text{和}\quad
\left(-1+\sqrt{1-a},+\infty\right)
\]
单调递减区间为
\[
\left(-1-\sqrt{1-a},-1+\sqrt{1-a}\right)
\]

\item 因为 \(x_0\ne\frac12\)，先将等式移项并因式分解：
\[
\begin{aligned}
f(x_0)-f\left(\frac12\right)
&=\frac{1}{3}x_0^3+x_0^2+ax_0+1
 -\left(\frac{1}{24}+\frac14+\frac a2+1\right)\\
&=\frac{x_0-\frac12}{12}\left(4x_0^2+14x_0+7+12a\right).
\end{aligned}
\]
所以所求条件等价于
\[
a=-\frac{4x_0^2+14x_0+7}{12}
\]
记右端为 \(h(x_0)\).在 \(0<x_0<1\) 上，
\[
h'(x_0)=-\frac{8x_0+14}{12}<0
\]
所以 \(h\) 严格递减.分别考察两个区间：
\[
h(0)=-\frac{7}{12}
\]
\[
h\left(\frac12\right)=-\frac54
\]
\[
h(1)=-\frac{25}{12}
\]
由于两个端点 \(0\)，\(\frac12\)，\(1\) 均不属于相应的开区间，存在这样的 \(x_0\) 当且仅当
\[
a\in\left(-\frac{25}{12},-\frac54\right)\cup
\left(-\frac54,-\frac{7}{12}\right)
\]
这些参数均满足 \(a<0\)，其余负数参数均不存在这样的 \(x_0\).
\end{enumerate}
\end{solution}
