\section{单选题}

\begin{problem}
已知全集 \(U = \R\)，\(A = \{x \mid x \leqslant 0\}\)，\(B = \{x \mid x \geqslant 1\}\)，则集合 \(\complement_{U}(A \cup B)=\)（\quad）

\choices
    {\( \{x \mid x \geqslant 0\} \)}
    {\(\{x\mid x\leqslant 1\}\)}
    {\(\{x\mid 0\leqslant x\leqslant 1\}\)}
    {\(\{x\mid 0 < x < 1\}\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
因为 \(A=\{x\mid x\leqslant0\}\)，\(B=\{x\mid x\geqslant1\}\)，所以
\(A\cup B=\{x\mid x\leqslant0\text{ 或 }x\geqslant1\}\). 在全集 \(U=\R\) 中取补集，得到
\(\complement_U(A\cup B)=\{x\mid0<x<1\}\)，故选择 D.
\end{solution}

\begin{problem}
设复数 \(z\) 满足 \((z - 2\i)(2 - \i) = 5\)，则 \(z =\)（\quad）

\choices
    {\(2 + 3\i\)}
    {\(2 - 3\i\)}
    {\(3 + 2\i\)}
    {\(3 - 2\i\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
由 \((z-2\i)(2-\i)=5\)，得
\(z-2\i=\frac{5}{2-\i}=\frac{5(2+\i)}{(2-\i)(2+\i)}=2+\i\).
因此 \(z=2+3\i\)，故选择 A.
\end{solution}

\begin{problem}
已知 \(a = 2^{-\frac{1}{3}}\)，\(b = \log_{2} \frac{1}{3}\)，\(c = \log_{\frac{1}{2}} \frac{1}{3}\)，则（\quad）

\choices
    {\(a > b > c\)}
    {\(a > c > b\)}
    {\(c > a > b\)}
    {\(c > b > a\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
因为 \(b=\log_{2}\frac{1}{3}<0\). 又 \(3>2\)，所以
\(c=\log_{1/2}\frac{1}{3}=\log_{2}3>1\)；而
\(a=2^{-1/3}<1\). 因此 \(c>a>b\)，故选择 C.
\end{solution}

\begin{problem}
已知 \(m\)，\(n\) 表示两条不同直线，\(\alpha\) 表示平面，下列说法正确的是（\quad）

\choices
    {若 \(m \parallel \alpha, n \parallel \alpha\)，则 \(m \parallel n\)}
    {若 \(m \perp \alpha\)，\(n \subset \alpha\)，则 \(m \perp n\)}
    {若 \(m \perp \alpha\)，\(m \perp n\)，则 \(n \parallel \alpha\)}
    {若 \(m \parallel \alpha\)，\(m \perp n\)，则 \(n \perp \alpha\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
若 \(m\perp\alpha\)，则 \(m\) 垂直于平面 \(\alpha\) 内的每一条直线，所以当 \(n\subset\alpha\) 时必有 \(m\perp n\)，选项 B 正确.

为排除其余选项，取 \(\alpha:z=0\). 对于 A，可取
\(m:(x,y,z)=(t,0,1)\) 和 \(n:(x,y,z)=(0,t,1)\)，两条直线都平行于 \(\alpha\)，但方向不同，故不平行. 对于 C，可取 \(m\) 为 \(z\) 轴、\(n\) 为 \(x\) 轴，此时 \(m\perp\alpha\) 且 \(m\perp n\)，但 \(n\subset\alpha\)，故 \(n\) 不平行于 \(\alpha\). 对于 D，可取
\(m:(x,y,z)=(t,0,1)\)，\(n:(x,y,z)=(0,t,1+t)\)，两线相交且方向向量的数量积为 \(0\)，所以 \(m\perp n\)；但 \(n\) 的方向既不垂直于 \(\alpha\)，也不平行于 \(\alpha\)，故 \(n\not\perp\alpha\). 因此只有 B 正确.
\end{solution}

\begin{problem}
设 \(a\)，\(b\)，\(c\) 是非零向量，已知命题 \(p\)：若 \(a \cdot b = 0\)，\(b \cdot c = 0\)，则 \(a \cdot c = 0\)；命题 \(q\)：若 \(a \parallel b\)，\(b \parallel c\)，则 \(a \parallel c\)，则下列命题中的真命题是（\quad）

\choices
    {\(p \vee q\)}
    {\(p \wedge q\)}
    {\((\neg p) \wedge (\neg q)\)}
    {\(p \vee (\neg q)\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
命题 \(p\) 不一定成立. 例如取
\(\bs a=(1,0,0)\)，\(\bs b=(0,1,0)\)，\(\bs c=(1,0,0)\)，则
\(\bs a\cdot\bs b=\bs b\cdot\bs c=0\)，但 \(\bs a\cdot\bs c=1\neq0\).

命题 \(q\) 恒成立：由 \(\bs a\parallel\bs b\) 和 \(\bs b\parallel\bs c\)，存在非零实数 \(\lambda\)，\(\mu\) 使 \(\bs a=\lambda\bs b\)，\(\bs b=\mu\bs c\)，于是 \(\bs a=\lambda\mu\bs c\)，所以 \(\bs a\parallel\bs c\). 因而 \(p\) 为假、\(q\) 为真，只有 \(p\vee q\) 为真，故选择 A.
\end{solution}

\begin{problem}
若将一个质点随机投入如图所示的长方形 \(ABCD\) 中，其中 \(AB = 2\)，\(BC = 1\)，则质点落在以 \(AB\) 为直径的半圆内的概率是（\quad）

\bitmapfigure[max width=0.32\linewidth]{img/2014/liaoning_liberal/q6_fig1.png}

\choices
    {\(\frac{\pi}{2}\)}
    {\(\frac{\pi}{4}\)}
    {\(\frac{\pi}{6}\)}
    {\(\frac{\pi}{8}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
矩形 \(ABCD\) 的面积为
\[
S_{\text{矩形}}=AB\cdot BC=2\times1=2
\]
以 \(AB\) 为直径的半圆半径为 \(1\)，其面积为
\[
S_{\text{半圆}}=\frac{1}{2}\pi\cdot1^{2}=\frac{\pi}{2}
\]
由于质点在矩形内均匀随机投入，所求概率为
\[
\frac{S_{\text{半圆}}}{S_{\text{矩形}}}=\frac{\pi/2}{2}=\frac{\pi}{4}
\]
故选择 B.
\end{solution}

\begin{problem}
某几何体三视图如图所示，则该几何体的体积为（\quad）

\bitmapfigure[max width=0.56\linewidth]{img/2014/liaoning_liberal/q7_fig1.png}

\choices
    {\(8 - \frac{\pi}{4}\)}
    {\(8 - \frac{\pi}{2}\)}
    {\(8 - \pi\)}
    {\(8 - 2\pi\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
由主视图和左视图可知，几何体的高为 \(2\)，且俯视图给出的水平截面在整个高度内不变. 俯视图是一个边长为 \(2\) 的正方形去掉两个半径为 \(1\) 的四分之一圆，因此底面积为
\[
S=2\times2-2\times\frac{1}{4}\pi\cdot1^{2}=4-\frac{\pi}{2}
\]
所以该几何体是以此为底、以 \(2\) 为高的直棱柱，体积为
\[
V=Sh=\left(4-\frac{\pi}{2}\right)\times2=8-\pi
\]
故选择 C.
\end{solution}

\begin{problem}
已知点 \(A(-2,3)\) 在抛物线 \(C: y^{2} = 2px\) 的准线上，记 \(C\) 的焦点为 \(F\)，则直线 \(AF\) 的斜率为（\quad）

\choices
    {\(-\frac{4}{3}\)}
    {\(-1\)}
    {\(-\frac{3}{4}\)}
    {\(-\frac{1}{2}\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
抛物线 \(y^{2}=2px\) 的准线方程为 \(x=-\dfrac{p}{2}\). 由点 \(A(-2,3)\) 在准线上，得
\[
-\frac{p}{2}=-2
\]
所以 \(p=4\). 抛物线焦点为 \(F\left(\dfrac{p}{2},0\right)=(2,0)\). 因而直线 \(AF\) 的斜率为
\[
k_{AF}=\frac{0-3}{2-(-2)}=-\frac{3}{4}
\]
故选择 C.
\end{solution}

\begin{problem}
设等差数列 \(\{a_{n}\}\) 的公差为 \(d\)，若数列 \(\{2^{a_{1}a_{n}}\}\) 为递减数列，则（\quad）

\choices
    {\(d < 0\)}
    {\(d > 0\)}
    {\(a_{1}d > 0\)}
    {\(a_{1}d < 0\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
等差数列的通项为 \(a_n=a_1+(n-1)d\)，所以
\[
a_1a_n=a_1^{2}+(n-1)a_1d
\]
因为底数 \(2>1\)，数列 \(\{2^{a_1a_n}\}\) 递减等价于指数数列 \(\{a_1a_n\}\) 递减，即相邻两项之差
\[
a_1a_{n+1}-a_1a_n=a_1d<0
\]
故选择 D.
\end{solution}

\begin{problem}
已知 \( f(x) \) 为偶函数，当 \( x \geqslant 0 \) 时， \( f(x) = \begin{cases}
\cos \pi x, & x \in \left[0, \frac{1}{2}\right],\\
2x - 1, & x \in \left(\frac{1}{2}, +\infty\right).
\end{cases} \) 则不等式 \(f(x - 1)\leqslant \frac{1}{2}\) 的解集为（\quad）

\choices
    {\(\left[\frac{1}{4}, \frac{2}{3}\right] \cup \left[\frac{4}{3}, \frac{7}{4}\right]\)}
    {\(\left[-\frac{3}{4}, -\frac{1}{3}\right] \cup \left[\frac{1}{4}, \frac{2}{3}\right]\)}
    {\(\left[\frac{1}{3}, \frac{3}{4}\right] \cup \left[\frac{4}{3}, \frac{7}{4}\right]\)}
    {\(\left[-\frac{3}{4}, -\frac{1}{3}\right] \cup \left[\frac{1}{3}, \frac{3}{4}\right]\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
令 \(t=|x-1|\)，由 \(f\) 为偶函数可得 \(f(x-1)=f(t)\)，其中 \(t\geqslant0\). 当 \(0\leqslant t\leqslant\dfrac12\) 时，
\[
f(t)=\cos\pi t\leqslant\frac12
\]
由于 \(\cos\pi t\) 在 \(\left[0,\dfrac12\right]\) 上递减，故此时
\[
t\geqslant\frac13
\]
当 \(t>\dfrac12\) 时，
\[
2t-1\leqslant\frac12\iff t\leqslant\frac34
\]
合并两种情形，得 \(\dfrac13\leqslant t\leqslant\dfrac34\)，即
\[
\frac13\leqslant|x-1|\leqslant\frac34
\]
所以
\[
x\in\left[\frac14,\frac23\right]\cup\left[\frac43,\frac74\right]
\]
故选择 A.
\end{solution}

\begin{problem}
将函数 \( y = 3\sin \left(2x + \frac{\pi}{3}\right) \) 的图象向右平移 \( \frac{\pi}{2} \) 个单位长度，所得图象对应的函数（\quad）

\choices
    {在区间 \(\left[\frac{\pi}{12}, \frac{7\pi}{12}\right]\) 上单调递减}
    {在区间 \(\left[\frac{\pi}{12}, \frac{7\pi}{12}\right]\) 上单调递增}
    {在区间 \(\left[-\frac{\pi}{6}, \frac{\pi}{3}\right]\) 上单调递减}
    {在区间 \(\left[-\frac{\pi}{6}, \frac{\pi}{3}\right]\) 上单调递增}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
函数图象向右平移 \(\dfrac{\pi}{2}\) 后，对应函数为
\[
y=3\sin\left(2\left(x-\frac{\pi}{2}\right)+\frac{\pi}{3}\right)=3\sin\left(2x-\frac{2\pi}{3}\right)
\]
其导数为
\[
y'=6\cos\left(2x-\frac{2\pi}{3}\right)
\]
当 \(x\in\left[\dfrac{\pi}{12},\dfrac{7\pi}{12}\right]\) 时，
\[
2x-\frac{2\pi}{3}\in\left[-\frac{\pi}{2},\frac{\pi}{2}\right]
\]
从而 \(y'\geqslant0\)，且在区间内部除有限个端点外均为正，所以函数在该区间上单调递增. 故选择 B.
\end{solution}

\begin{problem}
当 \(x \in [-2, 1]\) 时，不等式 \(ax^3 - x^2 + 4x + 3 \geqslant 0\) 恒成立，则实数 \(a\) 的取值范围是（\quad）

\choices
    {\([-5, - 3]\)}
    {\(\left[-6, -\frac{9}{8}\right]\)}
    {\([-6, - 2]\)}
    {\([-4, - 3]\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
当 \(x\neq0\) 时，原不等式等价于
\(a\geqslant\frac{x^{2}-4x-3}{x^{3}}\)，\((x>0)\)，\(a\leqslant\frac{x^{2}-4x-3}{x^{3}}\)，\((x<0)\).
令 \(\varphi(x)=\frac{x^{2}-4x-3}{x^{3}}=\frac1x-\frac4{x^{2}}-\frac3{x^{3}}\)，则
\(\varphi'(x)=\frac{-x^{2}+8x+9}{x^{4}}=-\frac{(x-9)(x+1)}{x^{4}}\).
在 \(0<x\leqslant1\) 上，\(\varphi'(x)>0\)，故
\[
\sup_{0<x\leqslant1}\varphi(x)=\varphi(1)=-6
\]
在 \([-2,0)\) 上，\(\varphi\) 在 \([-2,-1]\) 上递减、在 \([-1,0)\) 上递增，因此
\[
\inf_{-2\leqslant x<0}\varphi(x)=\varphi(-1)=-2
\]
当 \(x=0\) 时左端为 \(3>0\)，不产生额外限制. 所以必须且只需满足
\[
-6\leqslant a\leqslant-2
\]
故选择 C.
\end{solution}

\begin{problem}
已知 \(x\)，\(y\) 满足条件 \(\begin{cases}
2x + y - 2 \geqslant 0,\\
x - 2y + 4 \geqslant 0,\\
3x - y - 3 \leqslant 0,
\end{cases}\) 则目标函数 \(z = 3x + 4y\) 的最大值为\fillinblank{}.
\end{problem}

\begin{answer}
18
\end{answer}

\begin{solution}
三条边界直线分别为
\(y=2-2x\)，\(y=\frac{x}{2}+2\)，\(y=3x-3\).
它们的交点为
\((0,2)\)，\((1,0)\)，\((2,3)\).
直接代入三组约束可知，这三个点均满足全部不等式，且可行域正是由它们围成的三角形. 线性目标函数在有界多边形上的最大值必在顶点取得，计算得
\(z(0,2)=8\)，\(z(1,0)=3\)，\(z(2,3)=3\times2+4\times3=18\).
因此最大值为 \(18\).
\end{solution}

\begin{problem}
已知椭圆 \(C: \frac{x^2}{9} + \frac{y^2}{4} = 1\)，点 \(M\) 与 \(C\) 的焦点不重合，若 \(M\) 关于 \(C\) 的焦点的对称点分别为 \(A\)，\(B\)，线段 \(MN\) 的中点在 \(C\) 上，则 \(|AN| + |BN| =\fillinblank{}\).
\end{problem}

\begin{answer}
12
\end{answer}

\begin{solution}
椭圆的半长轴为 \(a=3\)，焦半距为
\[
c=\sqrt{3^{2}-2^{2}}=\sqrt5
\]
设两焦点为 \(F_1\)，\(F_2\)，点 \(M\) 关于它们的对称点分别为 \(A\)，\(B\). 设 \(P\) 为线段 \(MN\) 的中点，则
\(\overrightarrow{AN}=2\overrightarrow{F_1P}\)，\(\overrightarrow{BN}=2\overrightarrow{F_2P}\)
从而
\[
|AN|+|BN|=2\bigl(|F_1P|+|F_2P|\bigr)
\]
因为 \(P\) 在椭圆 \(C\) 上，根据椭圆定义，
\[
|F_1P|+|F_2P|=2a=6
\]
所以
\[
|AN|+|BN|=2\times6=12
\]
\end{solution}

\begin{problem}
对于 \(c > 0\)，当非零实数 \(a\)，\(b\) 满足 \(4a^2 - 2ab + b^2 - c = 0\) 且使 \(|2a + b|\) 最大时，\(\frac{1}{a} + \frac{2}{b} + \frac{4}{c}\) 的最小值为\fillinblank{}.
\end{problem}

\begin{answer}
-1
\end{answer}

\begin{solution}
令 \(u=2a+b\). 将条件配方为
\[
4a^{2}-2ab+b^{2}=u^{2}-3ub+3b^{2}
=\frac{u^{2}}{4}+3\left(b-\frac{u}{2}\right)^{2}=c
\]
因此 \(u^{2}\leqslant4c\)，且当且仅当 \(b=\dfrac{u}{2}\) 时，\(|u|=|2a+b|\) 取得最大值 \(2\sqrt c\). 此时
\(b=2a\)，\(a=\pm\frac{\sqrt c}{2}\)，\(b=\pm\sqrt c\)
其中 \(a\)，\(b\) 取同号.

因此在同时允许 \(c>0\) 变化时，代入所求式，两个符号分别给出
\[
\frac1a+\frac2b+\frac4c=\frac4{\sqrt c}+\frac4c
\quad\text{或}\quad
-\frac4{\sqrt c}+\frac4c
\]
取较小者，并令 \(t=\dfrac1{\sqrt c}>0\)，有
\[
\frac4c-\frac4{\sqrt c}=4t^{2}-4t=4\left(t-\frac12\right)^{2}-1\geqslant-1
\]
当 \(t=\dfrac12\)，即 \(c=4\)，取 \(a=-1\)，\(b=-2\) 时等号成立. 所以最小值为 \(-1\).
\end{solution}

\begin{problem}
在 \(\triangle ABC\) 中，内角 \(A\)，\(B\)，\(C\) 的对边分别为 \(a\)，\(b\)，\(c\)，且 \(a > c\)，已知 \(\overrightarrow{BA} \cdot \overrightarrow{BC} = 2\)，\(\cos B = \frac{1}{3}\)，\(b = 3\). 求：
\begin{enumerate}
\item \(a\) 和 \(c\) 的值；
\item \(\cos(B-C)\) 的值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由向量数量积的定义，\(\overrightarrow{BA} \cdot \overrightarrow{BC}=ac\cos B\)，所以 \(ac\cdot\frac{1}{3}=2\)，即 \(ac=6\). 由余弦定理，\(b^{2}=a^{2}+c^{2}-2ac\cos B=a^{2}+c^{2}-4\)，从而 \(a^{2}+c^{2}=13\). 又因为 \(a>c>0\)，\((a+c)^{2}=a^{2}+c^{2}+2ac=25\)，所以 \(a+c=5\). 因此 \(a\)，\(c\) 是方程 \(t^{2}-5t+6=0\) 的两个正根，结合 \(a>c\) 得 \(a=3\)，\(c=2\).
\item 由正弦定理，\(\sin C=\dfrac{c\sin B}{b}=\dfrac{2\cdot\frac{2\sqrt{2}}{3}}{3}=\dfrac{4\sqrt{2}}{9}\). 又由余弦定理，\(\cos C=\dfrac{a^{2}+b^{2}-c^{2}}{2ab}=\dfrac{9+9-4}{18}=\dfrac{7}{9}\). 因而
\[
\cos(B-C)=\cos B\cos C+\sin B\sin C=\frac{1}{3}\cdot\frac{7}{9}+\frac{2\sqrt{2}}{3}\cdot\frac{4\sqrt{2}}{9}=\frac{23}{27}
\]
\end{enumerate}
\end{solution}

\begin{problem}
某大学餐饮中心为了了解新生的饮食习惯，在全校一年级学生中进行了抽样调查，调查结果如下表所示：

\begin{center}
\begin{tabular}{c|ccc}
 & 喜欢甜品 & 不喜欢甜品 & 合计 \\ \hline
南方学生 & 60 & 20 & 80 \\
北方学生 & 10 & 10 & 20 \\
合计 & 70 & 30 & 100
\end{tabular}
\end{center}

\begin{enumerate}
\item 根据表中数据，问是否有 95\% 的把握认为“南方学生和北方学生在选用甜品的饮食习惯方面有差异”；
\item 已知在被调查的北方学生中有 5 名数学系的学生，其中 2 名喜欢甜品，现在从这 5 名学生中随机抽取 3 人，求至多有 1 人喜欢甜品的概率.
\end{enumerate}

附：
\begin{center}
\(\displaystyle \chi^2 = \frac{n\left(n_{11}n_{22} - n_{12}n_{21}\right)^2}{n_{1+}n_{2+}n_{+1}n_{+2}}\)，
\qquad
\begin{tabular}{c|ccc}
\(P(\chi^2 \geqslant k)\) & \(0.100\) & \(0.050\) & \(0.010\) \\ \hline
\(k\) & \(2.706\) & \(3.841\) & \(6.635\)
\end{tabular}
\end{center}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由题中给出的公式，取 \(n=100\)，\(n_{11}=60\)，\(n_{12}=20\)，\(n_{21}=10\)，\(n_{22}=10\)，以及 \(n_{1+}=80\)，\(n_{2+}=20\)，\(n_{+1}=70\)，\(n_{+2}=30\)，得
\[
\chi^{2}=\frac{100(60\times10-20\times10)^{2}}{80\times20\times70\times30}=\frac{100}{21}\approx4.762
\]
因为 \(4.762>3.841\)，所以有 \(95\%\) 的把握认为南方学生和北方学生在选用甜品的饮食习惯方面有差异.
\item 从 5 名学生中抽取 3 人的所有等可能结果数为 \(\mathrm C_{5}^{3}\). 至多有 1 人喜欢甜品包括抽到 0 人或 1 人喜欢甜品两种情况，因此所求概率为
\[
\frac{\mathrm C_{2}^{0}\mathrm C_{3}^{3}+\mathrm C_{2}^{1}\mathrm C_{3}^{2}}{\mathrm C_{5}^{3}}=\frac{1+6}{10}=\frac{7}{10}
\]
\end{enumerate}
\end{solution}

\begin{problem}
如图，\(\triangle ABC\) 和 \(\triangle BCD\) 所在平面互相垂直，且 \(AB = BC = BD = 2\)，\(\angle ABC = \angle DBC = 120^{\circ}\)，\(E\)，\(F\)，\(G\) 分别为 \(AC\)，\(DC\)，\(AD\) 的中点. 附：锥体的体积公式 \( V = \frac{1}{3} Sh \)，其中 \(S\) 为底面面积，\(h\) 为高.
\begin{enumerate}
\item 求证：\(EF \perp\) 平面 \(BCG\)；
\item 求三棱锥 \(D - BCG\) 的体积.
\end{enumerate}

\bitmapfigure[max width=0.34\linewidth]{img/2014/liaoning_liberal/q19_fig1.png}
\end{problem}

\begin{solution}
由于两平面互相垂直且交线为 \(BC\)，可建立空间直角坐标系：取 \(B\) 为原点，\(BC\) 所在直线为 \(x\) 轴，分别在平面 \(ABC\) 和平面 \(BCD\) 内取互相垂直的 \(y\) 轴和 \(z\) 轴. 由题意可取
\(B=(0,0,0)\)，\(C=(2,0,0)\)，\(A=(-1,\sqrt{3},0)\)，\(D=(-1,0,\sqrt{3})\).
例如，\(\overrightarrow{BA}\cdot\overrightarrow{BC}=-2\)，且 \(\lvert\overrightarrow{BA}\rvert=\lvert\overrightarrow{BC}\rvert=2\)，所以 \(\angle ABC=120^{\circ}\)，点 \(D\) 同理满足题设条件.
\begin{enumerate}
\item 由中点坐标公式，
\(E=\left(\frac{1}{2},\frac{\sqrt{3}}{2},0\right)\)，\(F=\left(\frac{1}{2},0,\frac{\sqrt{3}}{2}\right)\)，\(G=\left(-1,\frac{\sqrt{3}}{2},\frac{\sqrt{3}}{2}\right)\).
所以
\(\overrightarrow{EF}=\left(0,-\frac{\sqrt{3}}{2},\frac{\sqrt{3}}{2}\right)\)，\(\overrightarrow{BC}=(2,0,0)\)
从而 \(\overrightarrow{EF}\cdot\overrightarrow{BC}=0\). 又
\[
\overrightarrow{CG}=\left(-3,\frac{\sqrt{3}}{2},\frac{\sqrt{3}}{2}\right)
\]
故 \(\overrightarrow{EF}\cdot\overrightarrow{CG}=0\). 设 \(H\) 为 \(EF\) 的中点，则
\(H=\left(\frac{1}{2},\frac{\sqrt{3}}{4},\frac{\sqrt{3}}{4}\right)\)，且 \(H\) 也是 \(CG\) 的中点，所以 \(EF\) 与 \(CG\) 相交于 \(H\). 由于 \(BC\) 与 \(CG\) 是平面 \(BCG\) 内相交的两条直线，且
\(\overrightarrow{EF}\cdot\overrightarrow{BC}=\overrightarrow{EF}\cdot\overrightarrow{CG}=0\)，故 \(EF\perp\) 平面 \(BCG\).
\item
\(\overrightarrow{BG}=\left(-1,\frac{\sqrt{3}}{2},\frac{\sqrt{3}}{2}\right)\)，\(\overrightarrow{BD}=(-1,0,\sqrt{3})\).
三棱锥 \(D-BCG\) 的体积为
\[
V=\frac{1}{6}\left|\overrightarrow{BC}\cdot\left(\overrightarrow{BG}\times\overrightarrow{BD}\right)\right|
\]
计算得
\[
\overrightarrow{BG}\times\overrightarrow{BD}=\left(\frac{3}{2},\frac{\sqrt{3}}{2},\frac{\sqrt{3}}{2}\right)
\]
所以
\[
V=\frac{1}{6}\left|(2,0,0)\cdot\left(\frac{3}{2},\frac{\sqrt{3}}{2},\frac{\sqrt{3}}{2}\right)\right|=\frac{1}{2}
\]
\end{enumerate}
\end{solution}

\begin{problem}
圆 \(x^{2} + y^{2} = 4\) 的切线与 \(x\) 轴正半轴，\(y\) 轴正半轴围成一个三角形，当该三角形面积最小时，切点为 \(P\)（如图）.
\begin{enumerate}
\item 求点 \(P\) 的坐标；
\item 焦点在 \(x\) 轴上的椭圆 \(C\) 过点 \(P\)，且与直线 \(l: y = x + \sqrt{3}\) 交于 \(A\)，\(B\) 两点，若 \(\triangle PAB\) 的面积为 \(2\)，求 \(C\) 的标准方程.
\end{enumerate}

\bitmapfigure[max width=0.34\linewidth]{img/2014/liaoning_liberal/q20_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设第一象限内的切点为 \(P=(x_{0},y_{0})\). 因为 \(P\) 在圆上，所以 \(x_{0}^{2}+y_{0}^{2}=4\)，且 \(x_{0}>0\)，\(y_{0}>0\). 圆在 \(P\) 处的切线方程为 \(x_{0}x+y_{0}y=4\)，它在两坐标轴正半轴上的截距分别为 \(\dfrac{4}{x_{0}}\) 和 \(\dfrac{4}{y_{0}}\). 因此三角形面积
\[
S=\frac{1}{2}\cdot\frac{4}{x_{0}}\cdot\frac{4}{y_{0}}=\frac{8}{x_{0}y_{0}}
\]
要使 \(S\) 最小，只需使 \(x_{0}y_{0}\) 最大. 由 \(2x_{0}y_{0}\leqslant x_{0}^{2}+y_{0}^{2}=4\)，得 \(x_{0}y_{0}\leqslant2\)，等号成立当且仅当 \(x_{0}=y_{0}\). 所以 \(x_{0}=y_{0}=\sqrt{2}\)，即 \(P=(\sqrt{2},\sqrt{2})\).
\item 设椭圆 \(C\) 的标准方程为
\(\frac{x^{2}}{a^{2}}+\frac{y^{2}}{b^{2}}=1\)，\(a>b>0\).
因为 \(P\) 在椭圆上，所以
\[
\frac{2}{a^{2}}+\frac{2}{b^{2}}=1,\quad \text{即}\quad \frac{1}{a^{2}}+\frac{1}{b^{2}}=\frac{1}{2}
\]
令 \(u=\dfrac{1}{b^{2}}\). 将直线 \(y=x+\sqrt{3}\) 代入椭圆方程，得
\[
\left(\frac{1}{a^{2}}+\frac{1}{b^{2}}\right)x^{2}+\frac{2\sqrt{3}}{b^{2}}x+\frac{3}{b^{2}}-1=0
\]
即 \(\frac{1}{2}x^{2}+2\sqrt{3}ux+3u-1=0\).
设其两根为 \(x_{1}\)，\(x_{2}\)，则根的判别式为
\[
\Delta=(2\sqrt{3}u)^{2}-4\cdot\frac{1}{2}(3u-1)=12u^{2}-6u+2
\]
从而 \(\lvert x_{1}-x_{2}\rvert=2\sqrt{\Delta}\). 由于直线斜率为 \(1\)，
\[
AB=\sqrt{(x_{1}-x_{2})^{2}+(y_{1}-y_{2})^{2}}=\sqrt{2}\lvert x_{1}-x_{2}\rvert=2\sqrt{2\Delta}
\]
点 \(P\) 到直线 \(l\) 的距离为
\[
\frac{\left|\sqrt{2}-\sqrt{2}-\sqrt{3}\right|}{\sqrt{2}}=\sqrt{\frac{3}{2}}
\]
由 \(\triangle PAB\) 的面积为 \(2\)，得
\[
\frac{1}{2}\cdot2\sqrt{2\Delta}\cdot\sqrt{\frac{3}{2}}=2
\]
所以 \(\Delta=\dfrac{4}{3}\). 因而 \(12u^{2}-6u+2=\dfrac{4}{3}\)，即 \(18u^{2}-9u+1=0\)，解得 \(u=\dfrac{1}{3}\) 或 \(u=\dfrac{1}{6}\). 当 \(u=\dfrac{1}{3}\) 时，\(b^{2}=3\)，\(a^{2}=6\)；当 \(u=\dfrac{1}{6}\) 时，\(b^{2}=6\)，\(a^{2}=3\)，与 \(a>b\) 矛盾. 所以椭圆的标准方程为
\[
\frac{x^{2}}{6}+\frac{y^{2}}{3}=1
\]
\end{enumerate}
\end{solution}

\begin{problem}
已知函数 \(f(x) = \pi (x - \cos x) - 2\sin x - 2\)，\(g(x) = (x - \pi)\sqrt{\frac{1 - \sin x}{1 + \sin x}} + \frac{2x}{\pi} - 1\).
\begin{enumerate}
\item 证明：存在唯一 \(x_0 \in \left(0, \frac{\pi}{2}\right)\)，使 \(f(x_0) = 0\)；
\item 证明：存在唯一 \(x_1 \in \left(\frac{\pi}{2}, \pi\right)\)，使 \(g(x_1) = 0\)，且对上一问中的 \(x_0\)，有 \(x_0 + x_1 > \pi\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 函数 \(f\) 在 \(\left[0,\frac{\pi}{2}\right]\) 上连续，且
\[
f'(x)=\pi(1+\sin x)-2\cos x
\]
当 \(0<x<\dfrac{\pi}{2}\) 时，\(\pi(1+\sin x)>\pi>2\geqslant2\cos x\)，所以 \(f'(x)>0\)，即 \(f\) 在该区间上单调递增. 又
\(f(0)=-\pi-2<0\)，\(f\left(\frac{\pi}{2}\right)=\frac{\pi^{2}}{2}-4>0\)
（因为 \(\pi>3\)），由零点存在定理，存在唯一 \(x_{0}\in\left(0,\dfrac{\pi}{2}\right)\) 使 \(f(x_{0})=0\).
\item 令
\(r(t)=\sqrt{\frac{1-\sin t}{1+\sin t}}\)，\(h(t)=1-\frac{2t}{\pi}-tr(t)\)，\(0\leqslant t\leqslant\frac{\pi}{2}\).
当 \(x=\pi-t\) 时，
\[
g(\pi-t)=1-\frac{2t}{\pi}-t\sqrt{\frac{1-\sin t}{1+\sin t}}=h(t)
\]
在 \(0<t<\dfrac{\pi}{2}\) 时，\(r(t)=\tan\left(\dfrac{\pi}{4}-\dfrac{t}{2}\right)\)，故 \(r'(t)=-\dfrac{1+r(t)^{2}}{2}\). 另外，
\(r(t)=\frac{\cos t}{1+\sin t}\)，\(\frac{1+r(t)^{2}}{2}=\frac{1}{1+\sin t}\).
因此
\[
h'(t)=-\frac{2}{\pi}-r(t)+\frac{t(1+r(t)^{2})}{2}=-\frac{2}{\pi}-\frac{\cos t}{1+\sin t}+\frac{t}{1+\sin t}=\frac{f(t)}{\pi(1+\sin t)}
\]
由上一问，\(f(t)<0\)（\(0<t<x_{0}\)），\(f(t)>0\)（\(x_{0}<t<\dfrac{\pi}{2}\)），所以 \(h\) 在 \((0,x_{0})\) 上单调递减，在 \(\left(x_{0},\dfrac{\pi}{2}\right)\) 上单调递增. 又
\(h(0)=1>0\)，\(h\left(\frac{\pi}{2}\right)=0\).
故 \(h(x_{0})<0\)，且 \(h\) 在 \((0,x_{0})\) 内恰有一个零点，记为 \(t_{1}\). 在 \(\left[x_{0},\dfrac{\pi}{2}\right)\) 上 \(h(t)<0\)，所以没有其他零点. 令 \(x_{1}=\pi-t_{1}\)，则 \(x_{1}\in\left(\dfrac{\pi}{2},\pi\right)\)，且 \(g(x_{1})=h(t_{1})=0\)，这也是 \(g\) 在该区间内的唯一零点. 最后，由 \(t_{1}<x_{0}\)，得
\[
x_{0}+x_{1}=x_{0}+\pi-t_{1}=\pi+(x_{0}-t_{1})>\pi
\]
\end{enumerate}
\end{solution}

\begin{problem}
如图，\(EP\) 交圆于 \(E\)，\(C\) 两点，\(PD\) 切圆于 \(D\)，\(G\) 为 \(CE\) 上一点且 \(PG = PD\)，连接 \(DG\) 并延长交圆于点 \(A\)，作弦 \(AB\) 垂直 \(EP\)，垂足为 \(F\).
\begin{enumerate}
\item 求证：\(AB\) 为圆的直径；
\item 若 \(AC = BD\)，求证：\(AB = ED\).
\end{enumerate}

\bitmapfigure[max width=0.34\linewidth]{img/2014/liaoning_liberal/q22_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 建立直角坐标系，取 \(P\) 为原点，直线 \(EP\) 为 \(x\) 轴. 设 \(E=(e,0)\)，\(C=(c,0)\)，\(G=(g,0)\)，圆的方程写成
\[
x^{2}+y^{2}+ux+vy+w=0
\]
因为 \(E\)，\(C\) 是圆与 \(x\) 轴的两个交点，所以 \(w=ec\). 由切割线定理，\(PD^{2}=PE\cdot PC=ec\)，又因为 \(PG=PD\)，故 \(g^{2}=ec=w\).

设 \(D=(d_{1},d_{2})\). 圆心为 \(\left(-\dfrac{u}{2},-\dfrac{v}{2}\right)\)，且 \(PD\) 是切线，所以半径 \(OD\) 垂直于 \(PD\). 于是
\[
d_{1}\left(d_{1}+\frac{u}{2}\right)+d_{2}\left(d_{2}+\frac{v}{2}\right)=0
\]
结合 \(D\) 在圆上，即 \(d_{1}^{2}+d_{2}^{2}+ud_{1}+vd_{2}+w=0\)，可得
\[
ud_{1}+vd_{2}=-2w=-2g^{2}
\]

令从 \(G\) 出发的直线 \(GD\) 上的点为 \(X(t)=G+t(D-G)\). 代入圆方程，所得关于 \(t\) 的二次式的二次项系数为 \(L=(d_{1}-g)^{2}+d_{2}^{2}\). 其中一个根为 \(t=1\)，对应点 \(D\). 另外，\(t=0\) 时该二次式的值为
\[
Q(G)=g^{2}+ug+w=g(u+2g)
\]
又由圆方程和上面的切线关系，
\[
L=d_{1}^{2}+d_{2}^{2}-2gd_{1}+g^{2}=-(u+2g)d_{1}-vd_{2}=2g(g-d_{1})
\]
所以另一个交点 \(A\) 对应的参数为
\[
t_{A}=\frac{Q(G)}{L}=\frac{u+2g}{2(g-d_{1})}
\]
其横坐标为
\[
x_{A}=g+t_{A}(d_{1}-g)=g-\frac{u+2g}{2}=-\frac{u}{2}
\]
因此 \(A\) 在直线 \(x=-\dfrac{u}{2}\) 上. 由于 \(AB\perp EP\)，直线 \(AB\) 也是该直线；而圆心的横坐标正是 \(-\dfrac{u}{2}\)，所以圆心在 \(AB\) 上. 故 \(AB\) 是圆的直径.
\item 由上问，\(AB\) 是直径，且 \(AB\perp EC\). 因此圆心到弦 \(EC\) 的垂线平分弦 \(EC\)，垂足 \(F\) 是 \(EC\) 的中点. 从而 \(AE=AC\)，\(BE=BC\). 当 \(AC=BD\) 时，得到 \(AE=BD\).

又因为 \(AB\) 是直径，\(\angle AEB=\angle ADB=90^{\circ}\). 在直角三角形 \(AEB\) 和 \(BDA\) 中，斜边 \(AB\) 相等，且一条直角边 \(AE=BD\)，所以两三角形全等，对应顶点为 \(A\leftrightarrow B\)、\(E\leftrightarrow D\)、\(B\leftrightarrow A\)，从而 \(\angle EAB=\angle DBA\). 在直角三角形 \(DBA\) 中，\(\angle DBA+\angle BAD=90^{\circ}\)，故
\[
\angle EAD=\angle EAB+\angle BAD=90^{\circ}
\]
由圆周角定理的逆定理，\(ED\) 是圆的直径，所以 \(ED=AB\).
\end{enumerate}
\end{solution}

\begin{problem}
将圆 \(x^{2} + y^{2} = 1\) 上每一点的横坐标保持不变，纵坐标变为原来的 2 倍，得曲线 \(C\).
\begin{enumerate}
\item 写出 \(C\) 的参数方程；
\item 设直线 \(l: 2x + y - 2 = 0\) 与 \(C\) 的交点为 \(P_1\)，\(P_2\)，以坐标原点为极点，\(x\) 轴正半轴为极轴建立极坐标系，求过线段 \(P_1P_2\) 的中点且与 \(l\) 垂直的直线的极坐标方程.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设原圆上的点为 \((\cos t,\sin t)\)，其中 \(t\in[0,2\pi)\). 变换后横坐标不变，纵坐标变为原来的 2 倍，所以曲线 \(C\) 的参数方程为
\[
\begin{cases}
x=\cos t,\\
y=2\sin t,
\end{cases}\quad t\in[0,2\pi)
\]
\item 由参数方程或消去参数可知 \(C\) 的方程为 \(x^{2}+\dfrac{y^{2}}{4}=1\). 将直线 \(l:2x+y-2=0\)，即 \(y=2-2x\)，代入得
\[
x^{2}+\frac{(2-2x)^{2}}{4}=1
\]
即 \(2x^{2}-2x=0\). 所以两交点为 \(P_{1}=(0,2)\)，\(P_{2}=(1,0)\)，其中两点次序不影响后续结论. 线段 \(P_{1}P_{2}\) 的中点为
\[
M=\left(\frac{1}{2},1\right)
\]
直线 \(l\) 的斜率为 \(-2\)，所以过 \(M\) 且与 \(l\) 垂直的直线斜率为 \(\dfrac{1}{2}\)，其直角坐标方程为
\[
y-1=\frac{1}{2}\left(x-\frac{1}{2}\right)
\]
即 \(x-2y+\dfrac{3}{2}=0\). 代入 \(x=\rho\cos\theta\)，\(y=\rho\sin\theta\)，得所求极坐标方程
\[
\rho\left(\cos\theta-2\sin\theta\right)+\frac{3}{2}=0
\]
\end{enumerate}
\end{solution}

\begin{problem}
设函数 \( f(x) = 2|x - 1| + x - 1 \)，\( g(x) = 16x^{2} - 8x + 1 \)，记 \( f(x) \leqslant 1 \) 的解集为 \(M\)，\( g(x) \leqslant 4 \) 的解集为 \(N\).
\begin{enumerate}
\item 求 \(M\)；
\item 当 \(x \in M \cap N\) 时，证明：\(x^2 f(x) + x[f(x)]^2 \leqslant \frac{1}{4}\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 当 \(x\geqslant1\) 时，\(f(x)=3x-3\)，由 \(3x-3\leqslant1\) 得 \(1\leqslant x\leqslant\dfrac{4}{3}\). 当 \(x<1\) 时，\(f(x)=1-x\)，由 \(1-x\leqslant1\) 得 \(x\geqslant0\)，结合 \(x<1\) 得 \(0\leqslant x<1\). 因此
\[
M=\left[0,\frac{4}{3}\right]
\]
\item 先求 \(N\). 因为
\[
g(x)=(4x-1)^{2}
\]
所以
\[
g(x)\leqslant4\iff -2\leqslant4x-1\leqslant2\iff -\frac{1}{4}\leqslant x\leqslant\frac{3}{4}
\]
故 \(M\cap N=\left[0,\dfrac{3}{4}\right]\). 在此区间内 \(x\leqslant1\)，所以 \(f(x)=1-x\). 从而
\[
x^{2}f(x)+x[f(x)]^{2}=x^{2}(1-x)+x(1-x)^{2}=x(1-x)
\]
又由 \(x(1-x)=\frac{1}{4}-\left(x-\frac{1}{2}\right)^{2}\leqslant\frac{1}{4}\)，
故原不等式成立.
\end{enumerate}
\end{solution}
