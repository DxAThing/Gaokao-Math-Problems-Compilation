\section{单选题}

\begin{problem}
已知集合 \(M=\{2,4,6,8,10\}\)，\(N=\{x\mid -1<x<6\}\)，则 \(M\cap N=\)（\quad）

\choices
    {\(\{2,4\}\)}
    {\(\{2,4,6\}\)}
    {\(\{2,4,6,8\}\)}
    {\(\{2,4,6,8,10\}\)}

\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
因为 \(M=\{2,4,6,8,10\}\)，\(N=\{x\mid -1<x<6\}\)，所以
\(M\cap N=\{2,4\}\).
故选 A.
\end{solution}

\begin{problem}
设 \((1+2\i)a+b=2\i\)，其中 \(a\)，\(b\in\R\)，则（\quad）

\choices
    {\(a=1\)，\(b=-1\)}
    {\(a=1\)，\(b=1\)}
    {\(a=-1\)，\(b=1\)}
    {\(a=-1\)，\(b=-1\)}

\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
因为
\((1+2\i)a+b=2\i\)，
所以
\(a+b+2a\i=2\i\).
比较实部与虚部，得 \(\begin{cases}a+b=0,\\2a=2,\end{cases}\)，解得 \(\begin{cases}a=1,\\b=-1.\end{cases}\).
故选 A.
\end{solution}

\begin{problem}
已知向量 \(\bs{a}=(2,1)\)，\(\bs{b}=(-2,4)\)，则 \(\left|\bs{a}-\bs{b}\right|=\)（\quad）

\choices
    {2}
    {3}
    {4}
    {5}

\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
因为
\(\bs{a}-\bs{b}=(2,1)-(-2,4)=(4,-3)\)，
所以
\(\left|\bs{a}-\bs{b}\right|=\sqrt{4^2+(-3)^2}=5\).
故选 D.
\end{solution}

若 \(x\)，\(y\) 满足约束条件 \(\begin{cases}
x+y\ge2\\
x+2y\le4\\
y\ge0
\end{cases}\)，则 \(z=2x-y\) 的最大值是（\quad）

\choices
    {\(-2\)}
    {4}
    {8}
    {12}

\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
由题意作出可行域，如图阴影部分所示.
转化目标函数 \(z=2x-y\) 为
\(y=2x-z\).
上下平移直线 \(y=2x-z\)，可得当直线过点 \((4,0)\) 时，直线截距最小，\(z\) 最大，
\(z_{\max}=2\times4-0=8\).
故选 C.
\end{solution}

\begin{problem}
设抛物线 \(C:y^2=4x\) 的焦点为 \(F\)，点 \(A\) 在 \(C\) 上，点 \(B(3,0)\). 若 \(AF=BF\)，则 \(AB=\)（\quad）

\choices
    {2}
    {\(2\sqrt2\)}
    {3}
    {\(3\sqrt2\)}

\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
由题意得 \(F(1,0)\)，则
\(AF=BF=2\).
即点 \(A\) 到准线 \(x=-1\) 的距离为 \(2\)，所以点 \(A\) 的横坐标为
\(-1+2=1\).
不妨设点 \(A\) 在 \(x\) 轴上方，代入得
\(A(1,2)\).
所以
\(AB=\sqrt{(3-1)^2+(0-2)^2}=2\sqrt2\).
故选 B.
\end{solution}

\texfigure[max width=6.0cm]{img_repaint/2022/national_paper_b_liberal/q7_fig1.tex}

\begin{problem}
在正方体 \(ABCD-A_1B_1C_1D_1\) 中，\(E\)，\(F\) 分别为 \(AB\)，\(BC\) 的中点，则（\quad）

\choices
    {平面 \(B_1EF\perp\) 平面 \(BDD_1\)}
    {平面 \(B_1EF\perp\) 平面 \(A_1BD\)}
    {平面 \(B_1EF//\) 平面 \(A_1AC\)}
    {平面 \(B_1EF//\) 平面 \(A_1C_1D\)}

\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
在正方体 \(ABCD-A_1B_1C_1D_1\) 中，
\(AC\perp BD\)，\(DD_1\perp \text{平面 }ABCD\).
又 \(EF\subset \text{平面 }ABCD\)，所以
\(EF\perp DD_1\).
因为 \(E\)，\(F\) 分别为 \(AB\)，\(BC\) 的中点，所以
\(EF\parallel AC\)，
所以
\(EF\perp BD\).
又
\(BD\cap DD_1=D\)，
所以
\(EF\perp \text{平面 }BDD_1\).
又 \(EF\subset \text{平面 }B_1EF\)，所以
\(\text{平面 }B_1EF\perp \text{平面 }BDD_1\).
故选 A.
\end{solution}

\begin{problem}
已知等比数列 \(\{a_n\}\) 的前 \(3\) 项和为 \(168\)，\(a_2-a_5=42\)，则 \(a_6=\)（\quad）

\choices
    {14}
    {12}
    {6}
    {3}

\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
设等比数列 \(\{a_n\}\) 的首项为 \(a_1\)，公比为 \(q\)，则 \(q\ne0\)，且由题意知 \(q\ne1\).
因为前 \(3\) 项和为 \(168\)，所以
\(a_1+a_2+a_3=a_1+a_1q+a_1q^2=168\).
又
\(a_2-a_5=a_1q-a_1q^4=a_1q(1-q^3)=42\).
解得
\(q=\frac13\)，\(a_1=96\).
所以
\(a_6=a_1q^5=96\times\left(\frac13\right)^5=3\).
故选 D.
\end{solution}

\begin{problem}
函数 \(f(x)=\cos x+(x+1)\sin x+1\) 在区间 \([0,2\pi]\) 上的最小值，最大值分别为（\quad）

\choices
    {\(-\dfrac{\pi}{2}\)，\(\dfrac{\pi}{2}\)}
    {\(-\dfrac{3\pi}{2}\)，\(\dfrac{\pi}{2}\)}
    {\(-\dfrac{\pi}{2}\)，\(\dfrac{\pi}{2}+2\)}
    {\(-\dfrac{3\pi}{2}\)，\(\dfrac{\pi}{2}+2\)}

\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
因为
\(f'(x)=-\sin x+\sin x+(x+1)\cos x=(x+1)\cos x\).
所以 \(f(x)\) 在区间 \(\left(0,\frac{\pi}{2}\right)\) 和 \(\left(\frac{3\pi}{2},2\pi\right)\) 上单调递增，
在区间 \(\left(\frac{\pi}{2},\frac{3\pi}{2}\right)\) 上单调递减.
又
\(f(0)=f(2\pi)=2\)，\(f\left(\frac{\pi}{2}\right)=\frac{\pi}{2}+2\)，
\(f\left(\frac{3\pi}{2}\right)=-\left(\frac{3\pi}{2}+1\right)+1=-\frac{3\pi}{2}\).
所以 \(f(x)\) 在区间 \([0,2\pi]\) 上的最小值为 \(-\dfrac{3\pi}{2}\)，最大值为 \(\dfrac{\pi}{2}+2\).
故选 D.
\end{solution}

\begin{problem}
已知球 \(O\) 的半径为 \(1\)，四棱锥的顶点为 \(O\)，底面的四个顶点均在球 \(O\) 的球面上，则当该四棱锥的体积最大时，其高为（\quad）

\choices
    {\(\dfrac{1}{3}\)}
    {\(\dfrac{1}{2}\)}
    {\(\dfrac{\sqrt3}{3}\)}
    {\(\dfrac{\sqrt2}{2}\)}

\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
设该四棱锥底面所在小圆半径为 \(r\)，高为 \(h\)，则
\(r^2+h^2=1\).
当四棱锥底面为正方形时，底面积最大，且最大值为
\(2r^2\).
所以四棱锥体积
\(V=\frac13\cdot2r^2\cdot h=\frac23r^2h=\frac23(1-h^2)h\).
令
\(g(h)=(1-h^2)h\)，
则
\(g'(h)=1-3h^2\).
当
\(1-3h^2=0\)
时，体积取得最大值，解得
\(h=\frac{\sqrt3}{3}\).
故选 C.
\end{solution}

\begin{problem}
记 \(S_n\) 为等差数列 \(\{a_n\}\) 的前 \(n\) 项和. 若 \(2S_3=3S_2+6\)，则公差 \(d=\fillinblank{}\).

\end{problem}

\begin{answer}
2
\end{answer}

\begin{solution}
由
\(2S_3=3S_2+6\)
可得
\(2(a_1+a_2+a_3)=3(a_1+a_2)+6\).
化简得
\(2a_3=a_1+a_2+6\).
即
\(2(a_1+2d)=2a_1+d+6\)，
解得
\(d=2\).
故答案为：2.
\end{solution}

\begin{problem}
从甲，乙等 \(5\) 名同学中随机选 \(3\) 名参加社区服务工作，则甲，乙都入选的概率为 \(\fillinblank{}\).

\end{problem}

\begin{answer}
\(\dfrac{3}{10}\)
\end{answer}

\begin{solution}
从 \(5\) 名同学中随机选 \(3\) 名的方法数为
\(\binom{5}{3}=10\).
甲，乙都入选的方法数为
\(\binom{3}{1}=3\).
所以甲，乙都入选的概率
\(P=\frac{\binom{3}{1}}{\binom{5}{3}}=\frac{3}{10}\).
故答案为：\(\dfrac{3}{10}\).
\end{solution}

\begin{problem}
过四点 \((0,0)\)，\((4,0)\)，\((-1,1)\)，\((4,2)\) 中任意三点可得的一个圆的方程为 \(\fillinblank{}\).

\end{problem}

\begin{answer}
\((x-2)^2+(y-3)^2=13\) 或 \((x-2)^2+(y-1)^2=5\) 或 \(\left(x-\dfrac{4}{3}\right)^2+\left(y-\dfrac{7}{3}\right)^2=\dfrac{65}{9}\) 或 \(\left(x-\dfrac{8}{5}\right)^2+(y-1)^2=\dfrac{169}{25}\)
\end{answer}

\begin{solution}
设四点分别为 \(A(0,0)\)，\(B(4,0)\)，\(C(-1,1)\)，\(D(4,2)\).

\circled{1} 若圆过 \(A\)，\(B\)，\(C\) 三点，则线段 \(AB\) 的中垂线为 \(x=2\). 设圆心为 \((2,a)\)，则
\(4+a^2=9+(a-1)^2\)，
解得
\(a=3\).
所以圆的方程为
\((x-2)^2+(y-3)^2=13\).

\circled{2} 若圆过 \(A\)，\(B\)，\(D\) 三点，则线段 \(AB\) 的中垂线仍为 \(x=2\). 设圆心为 \((2,a)\)，则
\(4+a^2=4+(a-2)^2\)，
解得
\(a=1\).
所以圆的方程为
\((x-2)^2+(y-1)^2=5\).

\circled{3} 若圆过 \(A\)，\(C\)，\(D\) 三点，则线段 \(AC\) 的中垂线方程为
\(y=x+1\)，
线段 \(AD\) 的中垂线方程为
\(y=-2x+5\).
联立得圆心为
\(\left(\frac43,\frac73\right)\)，
从而圆的方程为
\(\left(x-\frac43\right)^2+\left(y-\frac73\right)^2=\frac{65}{9}\).

\circled{4} 若圆过 \(B\)，\(C\)，\(D\) 三点，则线段 \(BD\) 的中垂线方程为
\(y=1\)，
线段 \(BC\) 的中垂线方程为
\(y=5x-7\).
联立得圆心为
\(\left(\frac85,1\right)\)，
从而圆的方程为
\(\left(x-\frac85\right)^2+(y-1)^2=\frac{169}{25}\).

故答案为：
\((x-2)^2+(y-3)^2=13\)
或
\((x-2)^2+(y-1)^2=5\)
或
\(\left(x-\frac43\right)^2+\left(y-\frac73\right)^2=\frac{65}{9}\)
或
\(\left(x-\frac85\right)^2+(y-1)^2=\frac{169}{25}\).
\end{solution}

\begin{problem}
若 \(f(x)=\ln\left|a+\dfrac{1}{1-x}\right|+b\) 是奇函数，则 \(a=\fillinblank{}\)，\(b=\fillinblank{}\).

\end{problem}

\begin{answer}
\(a=-\dfrac12\)，\(b=\ln2\)
\end{answer}

\begin{solution}
若 \(a=0\)，则 \(f(x)\) 的定义域为 \(\{x\mid x\ne1\}\)，不关于原点对称，所以
\(a\ne0\).
因为
\(a+\frac{1}{1-x}\ne0\)，
所以
\(x\ne1\)，\(\quad x\ne1+\frac1a\).
函数 \(f(x)\) 为奇函数，定义域关于原点对称，因此
\(1+\frac1a=-1\)，
解得
\(a=-\frac12\).
又因为奇函数满足
\(f(0)=0\)，
所以
\(\ln\left|-\frac12+1\right|+b=0\)，
即
\(\ln\frac12+b=0\).
解得
\(b=\ln2\).
故答案为：\(a=-\dfrac12\)，\(b=\ln2\).
\end{solution}
\section{解答题}

\begin{problem}
已知 \(\triangle ABC\) 的内角 \(A\)，\(B\)，\(C\) 的对边分别为 \(a\)，\(b\)，\(c\)，且 \(\sin C\sin\left(A-B\right)=\sin B\sin\left(C-A\right)\).

\begin{enumerate}
\item 若 \(A=2B\)，求 \(C\)；
\item 证明：\(2a^2=b^2+c^2\).
\end{enumerate}

\end{problem}

\begin{solution}
（1） 由 \(A=2B\) 及
\(\sin C\sin\left(A-B\right)=\sin B\sin\left(C-A\right)\)
可得
\(\sin C\sin B=\sin B\sin\left(C-A\right)\).
而
\(0<B<\frac{\pi}{2}\)，
所以 \(\sin B>0\)，于是
\(\sin C=\sin\left(C-A\right)\).
又
\(0<C<\pi\)，\(0<C-A<\pi\)，
且若 \(C=C-A\)，则 \(A=0\)，这与三角形内角满足 \(0<A<\pi\) 矛盾，所以
\(C+\left(C-A\right)=\pi\).
再由
\(A=2B\)，\(A+B+C=\pi\)，
得
\(2C-A=\pi\)，\(3A+2C=2\pi\).
解得
\(C=\frac{5\pi}{8}\).

（2） 由已知条件得
\(\sin C\left(\sin A\cos B-\cos A\sin B\right)=\sin B\left(\sin C\cos A-\cos C\sin A\right)\).
再由正弦定理可得
\(ac\cos B-bc\cos A=bc\cos A-ab\cos C\).
根据余弦定理，\(ac\cos B=\frac{a^2+c^2-b^2}{2}\)，\(bc\cos A=\frac{b^2+c^2-a^2}{2}\)，\(ab\cos C=\frac{a^2+b^2-c^2}{2}\). 所以 \(\frac{a^2+c^2-b^2}{2}-\frac{b^2+c^2-a^2}{2}=\frac{b^2+c^2-a^2}{2}-\frac{a^2+b^2-c^2}{2}\).
化简得
\(2a^2=b^2+c^2\).
故原命题得证.
\end{solution}

\begin{problem}
如图，四面体 \(ABCD\) 中，\(AD\perp CD\)，\(AD=CD\)，\(\angle ADB=\angle CDB\)，\(E\) 为 \(AC\) 的中点.

\begin{center}
\bitmapfigure[max width=6.8cm]{img/2022/national_paper_B_liberal/q18_fig1.png}
\end{center}

\circled{1} 证明：平面 \(BED\perp\) 平面 \(ACD\)；

\circled{2} 设 \(AB=BD=2\)，\(\angle ACB=60^\circ\)，点 \(F\) 在 \(BD\) 上，当 \(\triangle AFC\) 的面积最小时，求三棱锥 \(F-ABC\) 的体积.

\end{problem}

\begin{solution}
\circled{1} 由于 \(AD=CD\)，\(E\) 是 \(AC\) 的中点，所以
\(AC\perp DE\).
又因为 \(AD=CD\)，\(BD=BD\)，\(\angle ADB=\angle CDB\)，所以 \(\triangle ADB\cong\triangle CDB\)，从而 \(AB=CB\).
故
\(AC\perp BE\).
由于
\(DE\cap BE=E\)，\(\quad DE\)，\(BE\subset \text{平面 }BED\)，
所以
\(AC\perp \text{平面 }BED\).
又 \(AC\subset \text{平面 }ACD\)，因此
\(\text{平面 }BED\perp \text{平面 }ACD\).

\circled{2} 由题意
\(AB=BD=2\)，\(AB=CB\)，\(\angle ACB=60^\circ\)，
可知 \(\triangle ABC\) 是边长为 \(2\) 的等边三角形，所以
\(AC=2\)，\(AE=CE=1\)，\(BE=\sqrt3\).
又因为 \(AD=CD\)，\(AD\perp CD\)，所以 \(\triangle ACD\) 是等腰直角三角形，从而
\(DE=1\).
在直角三角形 \(BED\) 中，
\(DE^2+BE^2=BD^2\)，
故
\(1^2+(\sqrt3)^2=2^2\)，
所以
\(DE\perp BE\).
又因为 \(AC\cap BE=E\)，\(AC\)，\(BE\subset \text{平面 }ABC\)，故
\(DE\perp \text{平面 }ABC\).
由于 \(\triangle ADB\cong\triangle CDB\)，所以
\(\angle FBA=\angle FBC\).
再由
\(BF=BF\)，\(AB=CB\)，
得
\(\triangle FBA\cong\triangle FBC\)，
从而
\(AF=CF\).
所以
\(EF\perp AC\).
因为
\(S_{\triangle AFC}=\frac12\cdot AC\cdot EF\)，
所以当 \(EF\) 最短时，\(\triangle AFC\) 的面积最小.
过 \(E\) 作
\(EF\perp BD\)，
垂足为 \(F\). 在直角三角形 \(BED\) 中，
\(\frac12\cdot BE\cdot DE=\frac12\cdot BD\cdot EF\)，
所以
\(EF=\frac{BE\cdot DE}{BD}=\frac{\sqrt3}{2}\).

\begin{center}
\bitmapfigure[max width=5.6cm]{img/2022/national_paper_B_liberal/q18_solution_fig1.png}
\end{center}

设过 \(F\) 作 \(FH\perp BE\)，垂足为 \(H\). 因为 \(FH//DE\)，所以
\(FH\perp \text{平面 }ABC\).
又
\(\frac{FH}{DE}=\frac{BF}{BD}\).
在直角三角形 \(BED\) 中，
\(BF=\frac{BE^2}{BD}=\frac{(\sqrt3)^2}{2}=\frac32\)，
故
\(FH=\frac{BF}{BD}\cdot DE=\frac{3/2}{2}\cdot1=\frac34\).
所以三棱锥 \(F-ABC\) 的体积为 \(V_{F-ABC}=\frac13\cdot S_{\triangle ABC}\cdot FH=\frac13\cdot\frac12\cdot2\cdot\sqrt3\cdot\frac34=\frac{\sqrt3}{4}\).
\end{solution}

\begin{problem}
某地经过多年的环境治理，已将荒山改造成了绿水青山. 为估计一林区某种树木的总材积量，随机选取了 \(10\) 棵这种树木，测量每棵树的根部横截面积（单位：\(\text{m}^2\)）和材积量（单位：\(\text{m}^3\)），得到如下数据：

\begin{center}
\begin{tabular}{c|ccccc}
样本号 \(i\) & 1 & 2 & 3 & 4 & 5 \\\hline
根部横截面积 \(x_i\) & 0.04 & 0.06 & 0.04 & 0.08 & 0.08 \\
材积量 \(y_i\) & 0.25 & 0.40 & 0.22 & 0.54 & 0.51 \\
\end{tabular}

\vspace{0.5em}

\begin{tabular}{c|cccc|c}
样本号 \(i\) & 6 & 7 & 8 & 9 & 10 \\\hline
根部横截面积 \(x_i\) & 0.05 & 0.05 & 0.07 & 0.07 & 0.06 \\
材积量 \(y_i\) & 0.34 & 0.36 & 0.46 & 0.42 & 0.40 \\
\end{tabular}
\end{center}

并计算得 \(\sum_{i=1}^{10}x_i=0.6\)，\(\sum_{i=1}^{10}y_i=3.9\)，\(\sum_{i=1}^{10}x_i^2=0.038\)；
\(\sum_{i=1}^{10}y_i^2=1.6158\)，\(\sum_{i=1}^{10}x_iy_i=0.2474\).

\begin{enumerate}
\item 估计该林区这种树木平均一棵的根部横截面积与平均一棵的材积量；
\item 求该林区这种树木的根部横截面积与材积量的样本相关系数（精确到 \(0.01\)）；
\item 现测量了该林区所有这种树木的根部横截面积，并得到所有这种树木的根部横截面积总和为 \(186\text{ m}^2\). 已知树木的材积量与其根部横截面积近似成正比，利用以上数据给出该林区这种树木的总材积量的估计值.
\end{enumerate}

附：相关系数 \(r=\dfrac{\sum_{i=1}^n(x_i-\bar{x})(y_i-\bar{y})}{\sqrt{\sum_{i=1}^n(x_i-\bar{x})^2\sum_{i=1}^n(y_i-\bar{y})^2}}\)，且 \(\sqrt{1.896}\approx1.377\).

\end{problem}

\begin{solution}
（1）样本中 \(10\) 棵这种树木的根部横截面积的平均值
\(\bar{x}=\frac{0.6}{10}=0.06\).
样本中 \(10\) 棵这种树木的材积量的平均值
\(\bar{y}=\frac{3.9}{10}=0.39\).
据此可估计该林区这种树木平均一棵的根部横截面积为 \(0.06\text{ m}^2\)，平均一棵的材积量为 \(0.39\text{ m}^3\).

（2） 由题给公式，
\(r=\frac{\sum_{i=1}^{10}(x_i-\bar{x})(y_i-\bar{y})}{\sqrt{\sum_{i=1}^{10}(x_i-\bar{x})^2\sum_{i=1}^{10}(y_i-\bar{y})^2}}=\frac{\sum_{i=1}^{10}x_iy_i-10\bar{x}\bar{y}}{\sqrt{\left(\sum_{i=1}^{10}x_i^2-10\bar{x}^2\right)\left(\sum_{i=1}^{10}y_i^2-10\bar{y}^2\right)}}=\frac{0.2474-10\times0.06\times0.39}{\sqrt{\left(0.038-10\times0.06^2\right)\left(1.6158-10\times0.39^2\right)}}=\frac{0.0134}{\sqrt{0.0001896}} \approx\frac{0.0134}{0.01377} \approx0.97\).
所以样本相关系数约为 \(0.97\).

（3） 设该林区这种树木的总材积量的估计值为 \(Y\text{ m}^3\).
已知树木的材积量与其根部横截面积近似成正比，所以
\(\frac{0.06}{0.39}=\frac{186}{Y}\).
解得
\(Y=1209\).
所以该林区这种树木的总材积量估计为 \(1209\text{ m}^3\).

\end{solution}

\begin{problem}
已知函数 \(f(x)=ax-\dfrac{1}{x}-(a+1)\ln x\)，\(x>0\).

\begin{enumerate}
    \item 当 \(a=0\) 时，求 \(f(x)\) 的最大值；
    \item 若 \(f(x)\) 恰有一个零点，求 \(a\) 的取值范围.
\end{enumerate}

\end{problem}

\begin{solution}
（1） 当 \(a=0\) 时，
\(f(x)=-\frac1x-\ln x\)，\(x>0\).
则
\(f'(x)=\frac{1}{x^2}-\frac1x=\frac{1-x}{x^2}\).
当 \(x\in(0,1)\) 时，\(f'(x)>0\)，\(f(x)\) 单调递增；
当 \(x\in(1,+\infty)\) 时，\(f'(x)<0\)，\(f(x)\) 单调递减.
所以
\(f(x)_{\max}=f(1)=-1\).

（2） 因为
\(f(x)=ax-\frac1x-(a+1)\ln x\)，\(x>0\)，
所以
\(f'(x)=a+\frac1{x^2}-\frac{a+1}{x}=\frac{(ax-1)(x-1)}{x^2}\).
当 \(a\le0\) 时，\(ax-1<0\). 所以当 \(x\in(0,1)\) 时，\(f'(x)>0\)；当 \(x\in(1,+\infty)\) 时，\(f'(x)<0\). 于是
\(f(x)_{\max}=f(1)=a-1<0\)，
此时函数无零点，不合题意.

当 \(0<a<1\) 时，
\(\frac1a>1\).
在 \((0,1)\) 与 \(\left(\frac1a,+\infty\right)\) 上，\(f'(x)>0\)，\(f(x)\) 单调递增；
在 \(\left(1,\frac1a\right)\) 上，\(f'(x)<0\)，\(f(x)\) 单调递减.
又
\(f(1)=a-1<0\).
由 \(\circled{1}\) 可得
\(\frac1x+\ln x\ge1\)，
于是当 \(x>1\) 时，\(f(x)\) 最终趋于 \(+\infty\)，从而 \(f(x)\) 仅在 \(\left(\frac1a,+\infty\right)\) 内有唯一零点，符合题意.

当 \(a=1\) 时，
\(f'(x)=\frac{(x-1)^2}{x^2}\ge0\)，
所以 \(f(x)\) 单调递增. 又
\(f(1)=a-1=0\)，
所以 \(f(x)\) 有唯一零点，符合题意.

当 \(a>1\) 时，
\(\frac1a<1\).
在 \(\left(0,\frac1a\right)\) 与 \((1,+\infty)\) 上，\(f'(x)>0\)，\(f(x)\) 单调递增；
在 \(\left(\frac1a,1\right)\) 上，\(f'(x)<0\)，\(f(x)\) 单调递减.
此时
\(f(1)=a-1>0\).
由 \(\circled{1}\) 得当 \(0<x<1\) 时，
\(\ln x>1-\frac1x\)，
故在足够靠近 \(0\) 的某点 \(n\in\left(0,\frac1a\right)\) 处有 \(f(n)<0\). 又当 \(x>\frac1a\) 且趋于 \(+\infty\) 时，
\(f(x)\to+\infty\)，
因此 \(f(x)\) 在 \(\left(0,\frac1a\right)\) 内有一个零点，在 \(\left(\frac1a,+\infty\right)\) 内无零点，所以 \(f(x)\) 有唯一零点，符合题意.
综上，\(a\) 的取值范围为
\((0,+\infty)\).
\end{solution}

\begin{problem}
已知椭圆 \(E\) 的中心为坐标原点，对称轴为 \(x\) 轴，\(y\) 轴，且过 \(A(0,-2)\)，\(B\left(\dfrac32,-1\right)\) 两点.

\begin{enumerate}
\item 求 \(E\) 的方程；

\item 设过点 \(P(1,-2)\) 的直线交 \(E\) 于 \(M\)，\(N\) 两点，过 \(M\) 且平行于 \(x\) 轴的直线与线段 \(AB\) 交于点 \(T\)，点 \(H\) 满足 \(MT=TH\). 证明：直线 \(HN\) 过定点.
\end{enumerate}

\end{problem}

\begin{solution}
（1）设椭圆 \(E\) 的方程为
\(mx^2+ny^2=1\).
由点 \(A(0,-2)\)，\(B\left(\frac32,-1\right)\) 在椭圆上，得
\(4n=1\)，\(\frac94m+n=1\).
解得
\(m=\frac13\)，\(n=\frac14\).
所以椭圆 \(E\) 的方程为
\(\frac{x^2}{3}+\frac{y^2}{4}=1\).

（2）因为
\(A(0,-2)\)，\(B\left(\frac32,-1\right)\)，
所以直线 \(AB\) 的方程为
\(y+2=\frac23x\).

若过点 \(P(1,-2)\) 的直线斜率不存在，则直线为
\(x=1\).
代入
\(\frac{x^2}{3}+\frac{y^2}{4}=1\)
可得
\(M\left(1,-\frac{2\sqrt6}{3}\right)\)，\(N\left(1,\frac{2\sqrt6}{3}\right)\).
由 \(AB\) 的方程得
\(T\left(-\sqrt6+3,-\frac{2\sqrt6}{3}\right)\).

由 \(MT=TH\) 得
\(H\left(-2\sqrt6+5,-\frac{2\sqrt6}{3}\right)\).
于是直线 \(HN\) 过点 \((0,-2)\).

若过点 \(P(1,-2)\) 的直线斜率存在，设直线方程为 \(kx-y-(k+2)=0\)，设 \(M(x_1,y_1)\)，\(N(x_2,y_2)\). 联立 \(\begin{cases}kx-y-(k+2)=0,\\\dfrac{x^2}{3}+\dfrac{y^2}{4}=1,\end{cases}\)，得 \((3k^2+4)x^2-6k(k+2)x+3k(k+4)=0\). 于是 \(x_1+x_2=\dfrac{6k(k+2)}{3k^2+4}\)，\(x_1x_2=\dfrac{3k(k+4)}{3k^2+4}\)；\(y_1+y_2=\dfrac{-8(k+2)}{3k^2+4}\)，\(y_1y_2=\dfrac{4(4+4k-2k^2)}{3k^2+4}\)，且 \(x_1y_2+x_2y_1=\frac{-24k}{3k^2+4}\).

联立 \(y=y_1\) 和 \(y=\dfrac23x-2\)，可得
\(T\left(\frac{3y_1}{2}+3,y_1\right)\)，\(H(3y_1+6-x_1,y_1)\).
于是 \(HN:\ y-y_2=\frac{y_1-y_2}{3y_1+6-x_1-x_2}(x-x_2)\). 将 \((0,-2)\) 代入并整理，需证 \(2(x_1+x_2)-6(y_1+y_2)+x_1y_2+x_2y_1-3y_1y_2-12=0\).
把上面的各式代入，恒成立.
综上，直线 \(HN\) 过定点
\((0,-2)\).
\end{solution}

\begin{problem}
在直角坐标系 \(xOy\) 中，曲线 \(C\) 的参数方程为 \(x=\sqrt3\cos2t\)，\(y=2\sin t\)（\(t\) 为参数），以坐标原点为极点，\(x\) 轴正半轴为极轴建立极坐标系，已知直线 \(l\) 的极坐标方程为 \(\rho\sin\left(\theta+\frac{\pi}{3}\right)+m=0\).

\begin{enumerate}
\item 写出 \(l\) 的直角坐标方程；
\item 若 \(l\) 与 \(C\) 有公共点，求 \(m\) 的取值范围.
\end{enumerate}

\end{problem}

\begin{solution}
（1）因为
\(\rho\sin\left(\theta+\frac{\pi}{3}\right)+m=0\)，
所以
\(\frac12\rho\sin\theta+\frac{\sqrt3}{2}\rho\cos\theta+m=0\).
又因为
\(\rho\sin\theta=y\)，\(\rho\cos\theta=x\)，
所以化简为
\(\frac12y+\frac{\sqrt3}{2}x+m=0\).
整理得 \(l\) 的直角坐标方程为
\(\sqrt3x+y+2m=0\).

（2）联立 \(l\) 与 \(C\) 的方程，即将
\(x=\sqrt3\cos2t\)，\(y=2\sin t\)
代入
\(\sqrt3x+y+2m=0\)
中，可得
\(3\cos2t+2\sin t+2m=0\).
由
\(\cos2t=1-2\sin^2t\)
得
\(6\sin^2t-2\sin t-3=2m\).
令
\(a=\sin t\)，\(a\in[-1,1]\)，
则
\(2m=6a^2-2a-3\).
设
\(f(a)=6a^2-2a-3\)，\(-1\le a\le1\).
因为 \(f(a)\) 的对称轴为
\(a=\frac16\)，
所以
\(f(a)_{\min}=f\left(\frac16\right)=\frac16-\frac13-3=-\frac{19}{6}\)，
\(f(a)_{\max}=f(-1)=6+2-3=5\).
故
\(-\frac{19}{6}\le2m\le5\)，
即
\(-\frac{19}{12}\le m\le\frac52\).
所以 \(m\) 的取值范围为
\(\left[-\frac{19}{12},\frac52\right]\).
\end{solution}

\begin{problem}
已知 \(a\)，\(b\)，\(c>0\)，且 \(a^{\frac32}+b^{\frac32}+c^{\frac32}=1\). 证明：

\begin{enumerate}
\item \(abc\le\dfrac19\)；
\item \(\dfrac{a}{b+c}+\dfrac{b}{c+a}+\dfrac{c}{a+b}\le\dfrac{1}{2\sqrt{abc}}\).
\end{enumerate}

\end{problem}

\begin{solution}
（1） 因为
\(a^{\frac32}>0\)，\(b^{\frac32}>0\)，\(c^{\frac32}>0\)，
由三元均值不等式可得 \(\frac{a^{\frac32}+b^{\frac32}+c^{\frac32}}{3}\ge\sqrt[3]{a^{\frac32}b^{\frac32}c^{\frac32}}=\sqrt{abc}\).
又
\(a^{\frac32}+b^{\frac32}+c^{\frac32}=1\)，
所以
\(\sqrt{abc}\le\frac13\).
故
\(abc\le\frac19\).

（2） 因为
\(b+c\ge2\sqrt{bc}\)，\(c+a\ge2\sqrt{ca}\)，\(a+b\ge2\sqrt{ab}\)，
所以
\(\frac{a}{b+c}\le\frac{a}{2\sqrt{bc}}=\frac{a^{\frac32}}{2\sqrt{abc}}\)，
\(\frac{b}{c+a}\le\frac{b}{2\sqrt{ca}}=\frac{b^{\frac32}}{2\sqrt{abc}}\)，
\(\frac{c}{a+b}\le\frac{c}{2\sqrt{ab}}=\frac{c^{\frac32}}{2\sqrt{abc}}\).
三式相加，得 \(\frac{a}{b+c}+\frac{b}{c+a}+\frac{c}{a+b}\le\frac{a^{\frac32}+b^{\frac32}+c^{\frac32}}{2\sqrt{abc}}=\frac{1}{2\sqrt{abc}}\).
故原命题得证.
\end{solution}
