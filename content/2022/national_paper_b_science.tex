\begin{problem}
设全集 \(U=\{1,2,3,4,5\}\)，集合 \(M\) 满足 \(\complement_U M=\{1,3\}\)，则（\quad）

\choices
    {\(2\in M\)}
    {\(3\in M\)}
    {\(4\notin M\)}
    {\(5\notin M\)}

\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
由 \(U=\{1,2,3,4,5\}\)，\(\complement_U M=\{1,3\}\)，所以 \(M=\{2,4,5\}\).
所以 \(2\in M\)，\(3\notin M\)，\(4\in M\)，\(5\in M\).
\(A\) 正确，\(B\)，\(C\)，\(D\) 错误.
故选 A.
\end{solution}

\begin{problem}
已知 \(z=1-2\i\)，且 \(z+a\overline z+b=0\)，其中 \(a\)，\(b\) 为实数，则（\quad）

\choices
    {\(a=1\)，\(b=-2\)}
    {\(a=-1\)，\(b=2\)}
    {\(a=1\)，\(b=2\)}
    {\(a=-1\)，\(b=-2\)}

\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
因为 \(z=1-2\i\)，所以 \(\overline z=1+2\i\).
由 \(z+a\overline z+b=0\)，得
\(1-2\i+a\left( 1+2\i \right)+b=0\)，
即
\(\left( 1+a+b \right)+\left( 2a-2 \right)\i=0\).
所以由上式得 \(1+a+b=0\)，\(2a-2=0\).
解得 \(a=1\)，\(b=-2\).
故选 A.
\end{solution}

\begin{problem}
已知向量 \(\bs{a}\)，\(\bs{b}\) 满足 \(|\bs{a}|=1\)，\(|\bs{b}|=\sqrt{3}\)，\(|\bs{a}-2\bs{b}|=3\)，则 \(\bs{a}\cdot\bs{b}=\)（\quad）

\choices
    {\(-2\)}
    {\(-1\)}
    {\(1\)}
    {\(2\)}

\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
因为
\(|\bs{a}-2\bs{b}|^2=|\bs{a}|^2-4\bs{a}\cdot\bs{b}+4|\bs{b}|^2\)，
又 \(|\bs{a}|=1\)，\(|\bs{b}|=\sqrt{3}\)，\(|\bs{a}-2\bs{b}|=3\)，所以
\(9=1-4\bs{a}\cdot\bs{b}+4\times 3=13-4\bs{a}\cdot\bs{b}\).
因此
\(\bs{a}\cdot\bs{b}=1\).
故选 C.
\end{solution}

\begin{problem}
嫦娥二号卫星在完成探月任务后，继续进行深空探测，成为我国第一颗环绕太阳飞行的人造行星，为研究嫦娥二号绕日周期与地球绕日周期的比值，用到数列 \(\{b_n\}\)：\(b_1=1+\frac{1}{\alpha_1}\)，\(b_2=1+\frac{1}{\alpha_1+\frac{1}{\alpha_2}}\)，\(b_3=1+\frac{1}{\alpha_1+\frac{1}{\alpha_2+\frac{1}{\alpha_3}}}\)，…，依此类推，其中 \(\alpha_k\in\N^*\)（\(k=1\)，\(2\)，\(\dots\)），则（\quad）

\choices
    {\(b_1<b_5\)}
    {\(b_3<b_8\)}
    {\(b_6<b_2\)}
    {\(b_4<b_7\)}

\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
因为 \(\alpha_k\in\N^*\)（\(k=1\)，\(2\)，\(\dots\)），所以
\(\alpha_1<\alpha_1+\frac{1}{\alpha_2}\)，
从而
\(\frac{1}{\alpha_1}>\frac{1}{\alpha_1+\frac{1}{\alpha_2}}\)，
即 \(b_1>b_2\).
又因为
\(\frac{1}{\alpha_2}>\frac{1}{\alpha_2+\frac{1}{\alpha_3}}\)，
所以
\(\alpha_1+\frac{1}{\alpha_2}>\alpha_1+\frac{1}{\alpha_2+\frac{1}{\alpha_3}}\)，
从而 \(b_2<b_3\)，且 \(b_1>b_3\).
\(\alpha_1+\frac{1}{\alpha_2+\frac{1}{\alpha_3+\frac{1}{\alpha_4+\frac{1}{\alpha_5}}}}>\alpha_1\)，
所以 \(b_1>b_5\).
又因为 \(\frac{1}{\alpha_3}>\frac{1}{\alpha_3+\frac{1}{\alpha_4+\frac{1}{\alpha_5+\frac{1}{\alpha_6+\frac{1}{\alpha_7+\frac{1}{\alpha_8}}}}}}\)，所以 \(\alpha_2+\frac{1}{\alpha_3}>\alpha_2+\frac{1}{\alpha_3+\frac{1}{\alpha_4+\frac{1}{\alpha_5+\frac{1}{\alpha_6+\frac{1}{\alpha_7+\frac{1}{\alpha_8}}}}}}\)，从而 \(\frac{1}{\alpha_2+\frac{1}{\alpha_3}}<\frac{1}{\alpha_2+\frac{1}{\alpha_3+\frac{1}{\alpha_4+\frac{1}{\alpha_5+\frac{1}{\alpha_6+\frac{1}{\alpha_7+\frac{1}{\alpha_8}}}}}}}\)，即 \(b_3>b_8\).
又因为 \(\alpha_2<\alpha_2+\frac{1}{\alpha_3+\frac{1}{\alpha_4+\frac{1}{\alpha_5+\frac{1}{\alpha_6}}}}\)，所以 \(\frac{1}{\alpha_2}>\frac{1}{\alpha_2+\frac{1}{\alpha_3+\frac{1}{\alpha_4+\frac{1}{\alpha_5+\frac{1}{\alpha_6}}}}}\)，即 \(b_2<b_6\).
又因为 \(\alpha_4< \alpha_4+\frac{1}{\alpha_5+\frac{1}{\alpha_6+\frac{1}{\alpha_7}}}\)，所以 \(\frac{1}{\alpha_4}>\frac{1}{\alpha_4+\frac{1}{\alpha_5+\frac{1}{\alpha_6+\frac{1}{\alpha_7}}}}\)，从而 \(\alpha_3+\frac{1}{\alpha_4}>\alpha_3+\frac{1}{\alpha_4+\frac{1}{\alpha_5+\frac{1}{\alpha_6+\frac{1}{\alpha_7}}}}\)，于是 \(\frac{1}{\alpha_3+\frac{1}{\alpha_4}}<\frac{1}{\alpha_3+\frac{1}{\alpha_4+\frac{1}{\alpha_5+\frac{1}{\alpha_6+\frac{1}{\alpha_7}}}}}\)，所以 \(\alpha_2+\frac{1}{\alpha_3+\frac{1}{\alpha_4}}<\alpha_2+\frac{1}{\alpha_3+\frac{1}{\alpha_4+\frac{1}{\alpha_5+\frac{1}{\alpha_6+\frac{1}{\alpha_7}}}}}\)，从而 \(\frac{1}{\alpha_2+\frac{1}{\alpha_3+\frac{1}{\alpha_4}}}>\frac{1}{\alpha_2+\frac{1}{\alpha_3+\frac{1}{\alpha_4+\frac{1}{\alpha_5+\frac{1}{\alpha_6+\frac{1}{\alpha_7}}}}}}\)，即 \(b_4<b_7\).
\(b_1>b_3>b_5>b_7>\cdots\)，
\(b_2<b_4<b_6<b_8<\cdots\)，
并且任意奇数项都大于任意偶数项.
\(b_1>b_5\)，\(b_3>b_8\)，\(b_2<b_6\)，\(b_4<b_7\).
故选 D.
\end{solution}

\begin{problem}
设 \(F\) 为抛物线 \(C:y^2=4x\) 的焦点，点 \(A\) 在 \(C\) 上，点 \(B(3,0)\)，若 \(AF=BF\)，则 \(AB=\)（\quad）

\choices
    {\(2\)}
    {\(2\sqrt{2}\)}
    {\(3\)}
    {\(3\sqrt{2}\)}

\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
由题意得 \(F(1,0)\)，则
\(BF=\sqrt{\left( 3-1 \right)^2+0^2}=2\).
因为 \(AF=BF\)，所以 \(AF=2\).
抛物线上的点到焦点与准线的距离相等，所以点 \(A\) 到准线 \(x=-1\) 的距离为 \(2\)，故点 \(A\) 的横坐标为 \(1\).
不妨设点 \(A\) 在 \(x\) 轴上方，代入 \(y^2=4x\) 得 \(A(1,2)\).
所以
\(AB=\sqrt{\left( 3-1 \right)^2+\left( 0-2 \right)^2}=2\sqrt{2}\).
故选 B.
\end{solution}

\begin{problem}
在正方体 \(ABCD-A_1B_1C_1D_1\) 中，\(E\)，\(F\) 分别为 \(AB\)，\(BC\) 的中点，则（\quad）

\choices
    {\(\text{平面 }B_1EF\perp \text{平面 }BDD_1\)}
    {\(\text{平面 }B_1EF\perp \text{平面 }A_1BD\)}
    {\(\text{平面 }B_1EF // \text{平面 }A_1AC\)}
    {\(\text{平面 }B_1EF // \text{平面 }A_1C_1D\)}

\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
在正方体 \(ABCD-A_1B_1C_1D_1\) 中，\(AC\perp BD\)，且 \(DD_1\perp \text{平面 }ABCD\).
又因为 \(EF\subset \text{平面 }ABCD\)，所以 \(EF\perp DD_1\).
因为 \(E\)，\(F\) 分别为 \(AB\)，\(BC\) 的中点，所以 \(EF // AC\)，从而 \(EF\perp BD\).
又 \(BD\cap DD_1=D\)，所以 \(EF\perp \text{平面 }BDD_1\).
因为 \(EF\subset \text{平面 }B_1EF\)，所以 \(\text{平面 }B_1EF\perp \text{平面 }BDD_1\).
故选 A.
\end{solution}

\begin{problem}
已知等比数列 \(\{a_n\}\) 的前 \(3\) 项和为 \(168\)，\(a_2-a_5=42\)，则 \(a_6=\)（\quad）

\choices
    {\(14\)}
    {\(12\)}
    {\(6\)}
    {\(3\)}

\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
设等比数列 \(\{a_n\}\) 的首项为 \(a_1\)，公比为 \(q\).
若 \(q=1\)，则 \(a_2-a_5=0\)，与题意矛盾，所以 \(q\neq 1\).
由题意得
\(a_1+a_2+a_3=\frac{a_1\left( 1-q^3 \right)}{1-q}=168\)，
\(a_2-a_5=a_1q-a_1q^4=42\).
解得
\(a_1=96\)，\(q=\frac{1}{2}\).
所以
\(a_6=a_1q^5=96\times \left( \frac{1}{2} \right)^5=3\).
故选 D.
\end{solution}

\begin{problem}
已知球 \(O\) 的半径为 \(1\)，四棱锥的顶点为 \(O\)，底面的四个顶点均在球 \(O\) 的球面上，则当该四棱锥的体积最大时，其高为（\quad）

\choices
    {\(\frac{1}{3}\)}
    {\(\frac{1}{2}\)}
    {\(\frac{\sqrt{3}}{3}\)}
    {\(\frac{\sqrt{2}}{2}\)}

\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
设该四棱锥底面为四边形 \(ABCD\)，四边形 \(ABCD\) 所在小圆半径为 \(r\)，四棱锥的高为 \(h\).
因为四边形 \(ABCD\) 的四个顶点都在同一圆上，设其对角线夹角为 \(\alpha\)，则 \(S_{ABCD}=\frac{1}{2}\cdot AC\cdot BD\cdot \sin \alpha\le \frac{1}{2}\cdot AC\cdot BD\le \frac{1}{2}\cdot 2r\cdot 2r=2r^2\).
当且仅当四边形 \(ABCD\) 为正方形时等号成立.
又因为球半径为 \(1\)，所以
\(r^2+h^2=1\).
于是四棱锥体积
\(V=\frac{1}{3}\cdot S_{ABCD}\cdot h \le \frac{1}{3}\cdot 2r^2h\).
由 \(r^2+h^2=1\)，得
\(V\le \frac{2}{3}r^2h\).
当 \(r^2=2h^2\) 时取到最大值，因此
\(1=r^2+h^2=2h^2+h^2=3h^2\)，
所以
\(h=\frac{\sqrt{3}}{3}\).
故选 C.
\end{solution}

\begin{problem}
某棋手与甲，乙，丙三位棋手各比赛一盘，各盘比赛结果相互独立. 已知该棋手与甲，乙，丙比赛获胜的概率分别为 \(p_1\)，\(p_2\)，\(p_3\)，且 \(p_3>p_2>p_1>0\). 记该棋手连胜两盘的概率为 \(p\)，则（\quad）

\choices
    {\(p\) 与该棋手和甲，乙，丙的比赛次序无关}
    {该棋手在第二盘与甲比赛，\(p\) 最大}
    {该棋手在第二盘与乙比赛，\(p\) 最大}
    {该棋手在第二盘与丙比赛，\(p\) 最大}

\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
该棋手连胜两盘，则第二盘为必胜盘.
记该棋手在第二盘与甲比赛且连胜两盘的概率为 \(p_{\text{甲}}\). 此时比赛顺序可能为乙甲丙或丙甲乙，两种顺序概率相同，所以 \(p_{\text{甲}}=\frac{1}{2}\left[ \left( 1-p_2 \right)p_1p_3+p_2p_1\left( 1-p_3 \right) \right]=\frac{1}{2}\left[ \left( 1-p_3 \right)p_1p_2+p_3p_1\left( 1-p_2 \right) \right]=p_1\left( p_2+p_3 \right)-2p_1p_2p_3\).
记该棋手在第二盘与乙比赛且连胜两盘的概率为 \(p_{\text{乙}}\). 此时比赛顺序可能为甲乙丙或丙乙甲，两种顺序下连胜两盘的概率分别为 \(\left( 1-p_1 \right)p_2p_3+p_1p_2\left( 1-p_3 \right)\) 和 \(\left( 1-p_3 \right)p_2p_1+p_3p_2\left( 1-p_1 \right)\). 这两个式子相等，所以 \(p_{\text{乙}}=\left( 1-p_1 \right)p_2p_3+p_1p_2\left( 1-p_3 \right)=p_2\left( p_1+p_3 \right)-2p_1p_2p_3\).
记该棋手在第二盘与丙比赛且连胜两盘的概率为 \(p_{\text{丙}}\). 此时比赛顺序可能为甲丙乙或乙丙甲，两种顺序下连胜两盘的概率分别为 \(\left( 1-p_1 \right)p_3p_2+p_1p_3\left( 1-p_2 \right)\) 和 \(\left( 1-p_2 \right)p_3p_1+p_2p_3\left( 1-p_1 \right)\). 这两个式子也相等，所以 \(p_{\text{丙}}=\left( 1-p_1 \right)p_3p_2+p_1p_3\left( 1-p_2 \right)=p_3\left( p_1+p_2 \right)-2p_1p_2p_3\).
于是
\(p_{\text{甲}}-p_{\text{乙}} =\left( p_1-p_2 \right)p_3<0\)，
\(p_{\text{乙}}-p_{\text{丙}} =\left( p_2-p_3 \right)p_1<0\).
所以
\(p_{\text{甲}}<p_{\text{乙}}<p_{\text{丙}}\).
因此该棋手在第二盘与丙比赛时，\(p\) 最大.
故选 D.
\end{solution}

\begin{problem}
双曲线 \(C\) 的两个焦点为 \(F_1\)，\(F_2\)，以 \(C\) 的实轴为直径的圆记为 \(D\)，过 \(F_1\) 作 \(D\) 的切线与 \(C\) 交于 \(M\)，\(N\) 两点，且 \(\cos \angle F_1NF_2=\frac{3}{5}\)，则 \(C\) 的离心率为（\quad）

\choices
    {\(\frac{\sqrt{5}}{2}\)}
    {\(\frac{3}{2}\)}
    {\(\frac{\sqrt{13}}{2}\)}
    {\(\frac{\sqrt{17}}{2}\)}

\end{problem}

\begin{answer}
AC
\end{answer}

\begin{solution}
依题意不妨设双曲线焦点在 \(x\) 轴，设过 \(F_1\) 作圆 \(D\) 的切线切点为 \(B\).
设 \(\angle F_1NF_2=\alpha\)，由题意
\(\cos \alpha=\frac{3}{5}\)，
所以
\(\sin \alpha=\frac{4}{5}\).
分两种情况讨论.

当 \(M\)，\(N\) 在双曲线的同一支上时，因为 \(\cos \angle F_1NF_2>0\)，所以 \(N\) 在双曲线的左支.
设 \(|OB|=a\)，\(|OF_1|=c\)，\(|F_1B|=b\)，则
\(|NA|=\frac{3}{2}a\)，\(|NF_2|=\frac{5}{2}a\).
又因为
\(|NF_2|-|NF_1|=2a\)，
所以
\(\frac{5}{2}a-\left( \frac{3}{2}a-2b \right)=2a\)，
即
\(2b=a\).
于是 \(e=\frac{c}{a}=\sqrt{1+\left( \frac{b}{a} \right)^2}=\sqrt{1+\left( \frac{1}{2} \right)^2}=\frac{\sqrt{5}}{2}\).
故此时选 A.

当 \(M\)，\(N\) 在双曲线的两支上时，因为 \(\cos \angle F_1NF_2>0\)，所以 \(N\) 在双曲线的右支.
设 \(\angle F_1NF_2=\alpha\)，由
\(\cos \angle F_1NF_2=\frac{3}{5}\)，
即
\(\cos \alpha=\frac{3}{5}\)，
则
\(\sin \alpha=\frac{4}{5}\).
因此
\(|NA|=\frac{3}{2}a\)，\(|NF_2|=\frac{5}{2}a\)，
又因为
\(|NF_1|-|NF_2|=2a\)，
所以
\(\frac{3}{2}a+2b-\frac{5}{2}a=2a\)，
即
\(2b=3a\).
于是 \(e=\frac{c}{a}=\sqrt{1+\left( \frac{b}{a} \right)^2}=\sqrt{1+\left( \frac{3}{2} \right)^2}=\frac{\sqrt{13}}{2}\).
故此时选 C.
综上，故选 AC.
\end{solution}

\begin{problem}
已知函数 \(f(x)\)，\(g(x)\) 的定义域均为 \(\R\)，且 \(f(x)+g(2-x)=5\)，\(g(x)-f(x-4)=7\). 若 \(y=g(x)\) 的图像关于直线 \(x=2\) 对称，\(g(2)=4\)，则 \(\sum_{k=1}^{22}f(k)=\)（\quad）

\choices
    {\(-21\)}
    {\(-22\)}
    {\(-23\)}
    {\(-24\)}

\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
因为 \(y=g(x)\) 的图像关于直线 \(x=2\) 对称，所以
\(g(2-x)=g(x+2)\).
又因为
\(g(x)-f(x-4)=7\)，
所以
\(g(x+2)-f(x-2)=7\)，
即
\(g(x+2)=7+f(x-2)\).
由
\(f(x)+g(2-x)=5\)
和 \(g(2-x)=g(x+2)\)，得
\(f(x)+g(x+2)=5\).
代入上式，得
\(f(x)+\left[ 7+f(x-2) \right]=5\)，
即
\(f(x)+f(x-2)=-2\).
于是
\(f(3)+f(5)+\cdots+f(21)=-10\)，
\(f(4)+f(6)+\cdots+f(22)=-10\).
又因为
\(f(x)+g(2-x)=5\)，
取 \(x=0\)，得
\(f(0)+g(2)=5\)，
所以
\(f(0)=1\).
再由
\(f(2)+f(0)=-2\)，
得
\(f(2)=-3\).
又因为
\(g(x)-f(x-4)=7\)，
所以
\(g(x+4)-f(x)=7\).
联立
\(f(x)+g(2-x)=5\)，
得
\(g(2-x)+g(x+4)=12\).
因此 \(y=g(x)\) 的图像关于点 \((3,6)\) 中心对称.
因为函数 \(g(x)\) 的定义域为 \(\R\)，所以
\(g(3)=6\).
再由
\(f(x)+g(x+2)=5\)，
取 \(x=1\)，得
\(f(1)=5-g(3)=-1\).
所以 \(\sum_{k=1}^{22}f(k) =f(1)+f(2)+\left[ f(3)+f(5)+\cdots+f(21) \right]+\left[ f(4)+f(6)+\cdots+f(22) \right]=-1-3-10-10=-24\).
故选 D.
\end{solution}

\begin{problem}
从甲，乙等 \(5\) 名同学中随机选 \(3\) 名参加社区服务工作，则甲，乙都入选的概率为\fillinblank{}.

\end{problem}

\begin{answer}
\(\frac{3}{10}\) 或 \(0.3\)
\end{answer}

\begin{solution}
从 \(5\) 名同学中随机选 \(3\) 名，共有
\(\binom{5}{3}=10\)
种选法.
其中甲，乙都入选时，还需从其余 \(3\) 名同学中再选 \(1\) 名，共有
\(\binom{3}{1}=3\)
种选法.
所以所求概率
\(P=\frac{3}{10}\).
故答案为 \(\frac{3}{10}\) 或 \(0.3\).
\end{solution}

\begin{problem}
过四点 \((0,0)\)，\((4,0)\)，\((-1,1)\)，\((4,2)\) 中的三点的一个圆的方程为\fillinblank{}.

\end{problem}

\begin{answer}
\((x-2)^2+(y-3)^2=13\) 或 \((x-2)^2+(y-1)^2=5\) 或 \(\left( x-\frac{4}{3} \right)^2+\left( y-\frac{7}{3} \right)^2=\frac{65}{9}\) 或 \(\left( x-\frac{8}{5} \right)^2+(y-1)^2=\frac{169}{25}\)
\end{answer}

\begin{solution}
设圆的方程为 \(x^2+y^2+Dx+Ey+F=0\).

\begin{problem}
已知 \(x=x_1\) 和 \(x=x_2\) 分别是函数 \(f(x)=2a^x-\e x^2\)（\(a>0\) 且 \(a\neq 1\)）的极小值点和极大值点. 若 \(x_1<x_2\)，则 \(a\) 的取值范围是\fillinblank{}.

\end{problem}

\begin{answer}
\(\left( \frac{1}{\e},1 \right)\)
\end{answer}

\begin{solution}
因为
\(f'(x)=2\ln a\cdot a^x-2\e x\)，
所以方程
\(2\ln a\cdot a^x-2\e x=0\)
的两个根为 \(x_1\)，\(x_2\).
即方程
\(\ln a\cdot a^x=\e x\)
有两个不同的实根，也就是函数 \(y=\ln a\cdot a^x\) 与函数 \(y=\e x\) 的图像有两个不同的交点.
因为 \(x_1\)，\(x_2\) 分别是函数 \(f(x)=2a^x-\e x^2\) 的极小值点和极大值点，且 \(x_1<x_2\)，所以函数 \(f(x)\) 在 \((-\infty,x_1)\) 和 \((x_2,+\infty)\) 上递减，在 \((x_1,x_2)\) 上递增.
因此当 \(x\in (-\infty,x_1)\cup (x_2,+\infty)\) 时，\(f'(x)<0\)，即直线 \(y=\e x\) 在曲线 \(y=\ln a\cdot a^x\) 上方；当 \(x\in (x_1,x_2)\) 时，\(f'(x)>0\)，即直线 \(y=\e x\) 在曲线 \(y=\ln a\cdot a^x\) 下方.
若 \(a>1\)，则 \(\ln a>0\)，从而对任意 \(x\) 都有 \(\ln a\cdot a^x>0\)；而当 \(x<0\) 时，\(\e x<0\). 这说明在 \(x<0\) 的范围内，直线 \(y=\e x\) 不可能在曲线 \(y=\ln a\cdot a^x\) 上方，更不可能与它相交，这与在 \((-\infty,x_1)\) 上恒有 \(f'(x)<0\) 矛盾. 所以
\(0<a<1\).
令
\(g(x)=\ln a\cdot a^x\)，
则
\(g'(x)=\ln^2 a\cdot a^x\).
设过原点且与函数 \(y=g(x)\) 的图像相切的直线的切点为 \(\left( x_0,\ln a\cdot a^{x_0} \right)\)，则该切线斜率为
\(g'(x_0)=\ln^2 a\cdot a^{x_0}\).
切线方程为
\(y-\ln a\cdot a^{x_0}=\ln^2 a\cdot a^{x_0}\left( x-x_0 \right)\).
因为该切线过原点，所以
\(0-\ln a\cdot a^{x_0}=\ln^2 a\cdot a^{x_0}\left( 0-x_0 \right)\)，
解得
\(x_0=\frac{1}{\ln a}\).
于是切线斜率
\(g'(x_0)=\ln^2 a\cdot a^{\frac{1}{\ln a}}=\e \ln^2 a\).
因为函数 \(y=\ln a\cdot a^x\) 与函数 \(y=\e x\) 的图像有两个不同的交点，所以
\(\e \ln^2 a<\e\)，
即
\(\ln^2 a<1\).
解得
\(\frac{1}{\e}<a<\e\).
再结合 \(0<a<1\)，得
\(\frac{1}{\e}<a<1\).
故答案为 \(\left( \frac{1}{\e},1 \right)\).
\end{solution}

\begin{problem}
记 \(\triangle ABC\) 的内角 \(A\)，\(B\)，\(C\) 的对边分别为 \(a\)，\(b\)，\(c\)，已知 \(\sin C\sin \left( A-B \right)=\sin B\sin \left( C-A \right)\).
\begin{enumerate}
\item 证明：\(2a^2=b^2+c^2\)；
\item 若 \(a=5\)，\(\cos A=\frac{25}{31}\)，求 \(\triangle ABC\) 的周长.
\end{enumerate}

\end{problem}

\begin{solution}
\begin{enumerate}
\item 证明：因为
\(\sin C\sin \left( A-B \right)=\sin B\sin \left( C-A \right)\)，
所以
\(\sin C\sin A\cos B-\sin C\sin B\cos A =\sin B\sin C\cos A-\sin B\sin A\cos C\).
由正弦定理与余弦定理可得 \(ac\cdot \frac{a^2+c^2-b^2}{2ac}-2bc\cdot \frac{b^2+c^2-a^2}{2bc}=-ab\cdot \frac{a^2+b^2-c^2}{2ab}\).
化简得
\(\frac{a^2+c^2-b^2}{2}-\left( b^2+c^2-a^2 \right) =-\frac{a^2+b^2-c^2}{2}\)，
所以
\(2a^2=b^2+c^2\).

\item 因为 \(a=5\)，\(\cos A=\frac{25}{31}\)，由（1）得
\(b^2+c^2=2a^2=50\).
由余弦定理
\(a^2=b^2+c^2-2bc\cos A\)，
得
\(25=50-2bc\cdot \frac{25}{31}\)，
所以
\(bc=\frac{31}{2}\).
于是
\(\left( b+c \right)^2=b^2+c^2+2bc=50+31=81\)，
所以
\(b+c=9\).
因此 \(\triangle ABC\) 的周长为
\(a+b+c=5+9=14\).
\end{enumerate}
\end{solution}

\item 若 \(a>0\)，当 \(x\in(-1,0)\) 时，
\(g(x)=\e^x+a(1-x^2)>0\)，
所以 \(f'(x)>0\).
故 \(f(x)\) 在 \((-1,0)\) 上单调递增，而 \(f(x)<f(0)=0\)，所以 \(f(x)\) 在 \((-1,0)\) 上没有零点，不合题意.

\item 若 \(-1\le a\le 0\)，当 \(x\in(0,+\infty)\) 时，
\(g'(x)=\e^x-2ax>0\)，
所以 \(g(x)\) 在 \((0,+\infty)\) 上单调递增，从而
\(g(x)>g(0)=1+a\ge 0\).
于是 \(f'(x)>0\)，故 \(f(x)\) 在 \((0,+\infty)\) 上单调递增，而 \(f(x)>f(0)=0\)，所以 \(f(x)\) 在 \((0,+\infty)\) 上没有零点，不合题意.

\item 若 \(a<-1\).
先看区间 \((0,+\infty)\). 因为
\(g'(x)=\e^x-2ax>0\)，
所以 \(g(x)\) 在 \((0,+\infty)\) 上单调递增.
又
\(g(0)=1+a<0\)，\(g(1)=\e>0\)，
所以存在 \(m\in(0,1)\)，使得
\(g(m)=0\)，
即 \(f'(m)=0\).
当 \(x\in(0,m)\) 时，\(f'(x)<0\)，\(f(x)\) 单调递减；当 \(x\in(m,+\infty)\) 时，\(f'(x)>0\)，\(f(x)\) 单调递增.
于是
\(f(x)<f(0)=0\)（\(x\in(0,m)\)），
且
\(\lim\limits_{x\to+\infty}f(x)=+\infty\)，
故 \(f(x)\) 在 \((0,+\infty)\) 上恰有一个零点.
再看区间 \((-1,0)\). 此时
\(g'(x)=\e^x-2ax\).
设
\(h(x)=g'(x)=\e^x-2ax\)，
则
\(h'(x)=\e^x-2a>0\).
所以 \(g'(x)\) 在 \((-1,0)\) 上单调递增.
又
\(g'(-1)=\frac{1}{\e}+2a<0\)，\(g'(0)=1>0\)，
所以存在 \(n\in(-1,0)\)，使得
\(g'(n)=0\).
当 \(x\in(-1,n)\) 时，\(g'(x)<0\)，\(g(x)\) 单调递减；当 \(x\in(n,0)\) 时，\(g'(x)>0\)，\(g(x)\) 单调递增.
又
\(g(0)=1+a<0\)，\(g(-1)=\frac{1}{\e}>0\)，
所以存在 \(t\in(-1,n)\)，使得
\(g(t)=0\)，
即 \(f'(t)=0\).
当 \(x\in(-1,t)\) 时，\(f(x)\) 单调递增；当 \(x\in(t,0)\) 时，\(f(x)\) 单调递减.
因为
\(\lim\limits_{x\to-1^+}f(x)=-\infty\)，\(f(0)=0\)，
所以当 \(x\in(t,0)\) 时，\(f(x)>0\).
故 \(f(x)\) 在 \((-1,t)\) 上有唯一零点，在 \((t,0)\) 上无零点，从而 \(f(x)\) 在 \((-1,0)\) 上恰有一个零点.
综上，\(a\) 的取值范围为
\((-\infty,-1)\).
\end{enumerate}
\end{enumerate}
\end{solution}

\begin{problem}
在直角坐标系 \(xOy\) 中，曲线 \(C\) 的参数方程为 \(x=\sqrt{3}\cos 2t\)，\(y=2\sin t\)（\(t\) 为参数），以坐标原点为极点，\(x\) 轴正半轴为极轴建立极坐标系，已知直线 \(l\) 的极坐标方程为 \( \rho \sin \left( \theta+\frac{\pi}{3} \right)+m=0 \).
\begin{enumerate}
\item 写出 \(l\) 的直角坐标方程；
\item 若 \(l\) 与 C 有公共点，求 \(m\) 的取值范围.
\end{enumerate}

\end{problem}

\begin{solution}
\begin{enumerate}
\item 因为
\(\rho \sin \left( \theta+\frac{\pi}{3} \right)+m=0\)，
所以
\(\frac{1}{2}\rho \sin \theta+\frac{\sqrt{3}}{2}\rho \cos \theta+m=0\).
又因为
\(\rho \sin \theta=y\)，\(\rho \cos \theta=x\)，
所以
\(\frac{1}{2}y+\frac{\sqrt{3}}{2}x+m=0\).
整理得 \(l\) 的直角坐标方程为
\(\sqrt{3}x+y+2m=0\).

\item 【法一】把
\(x=\sqrt{3}\cos 2t\)，\(y=2\sin t\)
代入
\(\sqrt{3}x+y+2m=0\)，
得
\(3\cos 2t+2\sin t+2m=0\).
由
\(\cos 2t=1-2\sin^2 t\)，
得
\(-6\sin^2 t+2\sin t+3+2m=0\).
令
\(\sin t=a\)，\(a\in[-1,1]\)，
则
\(2m=6a^2-2a-3\).
设
\(f(a)=6a^2-2a-3\)，\(-1\le a\le 1\).
因为抛物线开口向上，对称轴为
\(a=\frac{1}{6}\)，
所以
\(f(a)_{\max}=f(-1)=5\)，\(f(a)_{\min}=f\left( \frac{1}{6} \right)=-\frac{19}{6}\).
因此
\(-\frac{19}{6}\le 2m\le 5\)，
即
\(-\frac{19}{12}\le m\le \frac{5}{2}\).
故 \(m\) 的取值范围为
\(\left[ -\frac{19}{12},\frac{5}{2} \right]\).
\end{enumerate}
\end{solution}

\begin{problem}
已知 \(a\)，\(b\)，\(c\) 都是正数，且 \(a^{\frac{3}{2}}+b^{\frac{3}{2}}+c^{\frac{3}{2}}=1\)，证明：
\begin{enumerate}
\item \(abc\le \frac{1}{9}\)；
\item \(\frac{a}{b+c}+\frac{b}{a+c}+\frac{c}{a+b}\le \frac{1}{2\sqrt{abc}}\).
\end{enumerate}

\end{problem}

\begin{solution}
\begin{enumerate}
\item 证明：因为 \(a>0\)，\(b>0\)，\(c>0\)，所以 \(a^{\frac{3}{2}}>0\)，\(b^{\frac{3}{2}}>0\)，\(c^{\frac{3}{2}}>0\).
由均值不等式得 \(\frac{a^{\frac{3}{2}}+b^{\frac{3}{2}}+c^{\frac{3}{2}}}{3}\ge \sqrt[3]{a^{\frac{3}{2}}b^{\frac{3}{2}}c^{\frac{3}{2}}}\).
因为
\(a^{\frac{3}{2}}+b^{\frac{3}{2}}+c^{\frac{3}{2}}=1\)，
所以
\(\frac{1}{3}\ge (abc)^{\frac{1}{2}}\).
于是
\(abc\le \frac{1}{9}\).
当且仅当
\(a^{\frac{3}{2}}=b^{\frac{3}{2}}=c^{\frac{3}{2}}\)
时取等号.

\item 证明：因为
\(b+c\ge 2\sqrt{bc}\)，\(a+c\ge 2\sqrt{ac}\)，\(a+b\ge 2\sqrt{ab}\)，
所以
\(\frac{a}{b+c}\le \frac{a}{2\sqrt{bc}}=\frac{a^{\frac{3}{2}}}{2\sqrt{abc}}\)，
\(\frac{b}{a+c}\le \frac{b}{2\sqrt{ac}}=\frac{b^{\frac{3}{2}}}{2\sqrt{abc}}\)，
\(\frac{c}{a+b}\le \frac{c}{2\sqrt{ab}}=\frac{c^{\frac{3}{2}}}{2\sqrt{abc}}\).
三式相加得
所以 \(\frac{a}{b+c}+\frac{b}{a+c}+\frac{c}{a+b}\le \frac{a^{\frac{3}{2}}+b^{\frac{3}{2}}+c^{\frac{3}{2}}}{2\sqrt{abc}}=\frac{1}{2\sqrt{abc}}\).
当且仅当 \(a=b=c\) 时取等号.
\end{enumerate}
\end{solution}
