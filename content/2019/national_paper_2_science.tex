\section{单选题}

\begin{problem}
设集合 \(A=\{x\mid x^2-5x+6>0\}\)，\(B=\{x\mid x-1<0\}\)，则 \(A\cap B=\)（\quad）

\choices
    {\((-\infty,1)\)}
    {\((-2,1)\)}
    {\((-3,-1)\)}
    {\((3,+\infty)\)}

\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
由 \(A=\{x\mid x^2-5x+6>0\}=\{x\mid x>3\text{\ 或\ }x<2\}\)，\(B=\{x\mid x<1\}\)，所以 \(A\cap B=\{x\mid x<1\}=(-\infty,1)\). 故选 A.
\end{solution}

\choices

    {第三象限}
    {第四象限}

\end{problem}

\begin{answer}

\end{answer}

\begin{solution}

\end{solution}

\begin{problem}
已知 \(\overrightarrow{AB}=(2,3)\)，\(\overrightarrow{AC}=(3,t)\)，\(|\overrightarrow{BC}|=1\)，则 \(\overrightarrow{AB}\cdot\overrightarrow{BC}=\)（\quad）

\choices
    {\(-3\)}
    {\(-2\)}
    {\(2\)}
    {\(3\)}

\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
由 \(\overrightarrow{BC}=\overrightarrow{AC}-\overrightarrow{AB}=(1,t-3)\). 因为 \(|\overrightarrow{BC}|=1\)，所以 \(1+(t-3)^2=1\)，得 \(t=3\). 于是 \(\overrightarrow{BC}=(1,0)\)，从而 \(\overrightarrow{AB}\cdot\overrightarrow{BC}=(2,3)\cdot(1,0)=2\). 故选 C.
\end{solution}

\begin{problem}
2019 年 1 月 3 日嫦娥四号探测器成功实现人类历史上首次月球背面软着陆，我国航天事业取得又一重大成就. 实现月球背面软着陆需要解决的一个关键技术问题是地面与探测器的通讯联系. 为解决这个问题，发射了嫦娥四号中继星“鹊桥”，鹊桥沿着围绕地月拉格朗日 \(L_2\) 点的轨道运行. \(L_2\) 点是平衡点，位于地月连线的延长线上. 设地球质量为 \(M_1\)，月球质量为 \(M_2\)，地月距离为 \(R\)，\(L_2\) 点到月球的距离为 \(r\)，根据牛顿运动定律和万有引力定律，\(r\) 满足方程：\(\dfrac{M_1}{(R+r)^2}+\dfrac{M_2}{r^2}=(R+r)\dfrac{M_1}{R^3}\).

设 \(\alpha=\dfrac rR\). 由于 \(\alpha\) 的值很小，因此在近似计算中 \(\dfrac{3\alpha^3+3\alpha^4+\alpha^5}{(1+\alpha)^2}\approx 3\alpha^3\)，则 \(r\) 的近似值为（\quad）

\choices
    {\(R\sqrt{\dfrac{M_2}{M_1}}\)}
    {\(R\sqrt{\dfrac{M_2}{2M_1}}\)}
    {\(R\sqrt[3]{\dfrac{3M_2}{M_1}}\)}
    {\(R\sqrt[3]{\dfrac{M_2}{3M_1}}\)}

\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
设 \(\alpha=\dfrac rR\)，则原方程化为 \(\frac{M_1}{R^2(1+\alpha)^2}+\frac{M_2}{R^2\alpha^2}= \frac{M_1}{R^2}(1+\alpha)\). 整理得 \(\frac{3\alpha^3+3\alpha^4+\alpha^5}{(1+\alpha)^2}=\frac{M_2}{M_1}\). 由题设近似 \(3\alpha^3\approx \frac{M_2}{M_1}\)，所以 \(\alpha\approx \sqrt[3]{\frac{M_2}{3M_1}}\). 因此 \(r=\alpha R\approx R\sqrt[3]{\frac{M_2}{3M_1}}\). 故选 D.
\end{solution}

演讲比赛共有 \(9\) 位评委分别给出某选手的原始评分，评定该选手的成绩时，从 \(9\) 个原始评分中去掉 \(1\) 个最高分，\(1\) 个最低分，得到 \(7\) 个有效评分. \(7\) 个有效评分与 \(9\) 个原始评分相比，不变的数字特征是（\quad）

\begin{answer}

\end{answer}

\begin{problem}
若 \(a>b\)，则（\quad）

\choices
    {\(\ln(a-b)>0\)}
    {\(3^a<3^b\)}
    {\(a^3-b^3>0\)}
    {\(|a|>|b|\)}

\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
取 \(a=0\)，\(b=-1\)，则 \(a>b\) 成立，而 \(\ln(a-b)=\ln1=0\)，\(3^a=1>3^{-1}\)，\(|a|=0<1=|b|\). 只有 \(a^3-b^3=0-(-1)=1>0\) 成立，故选 C.
\end{solution}

\begin{problem}
设 \(\alpha\)，\(\beta\) 为两个平面，则 \(\alpha//\beta\) 的充要条件是（\quad）

\choices
    {\(\alpha\) 内有无数条直线与 \(\beta\) 平行}
    {\(\alpha\) 内有两条相交直线与 \(\beta\) 平行}
    {\(\alpha\)，\(\beta\) 平行于同一条直线}

\end{answer}

\end{solution}

\begin{problem}
若抛物线 \(y^2=2px\)（\(p>0\)）的焦点是椭圆 \(\dfrac{x^2}{3p}+\dfrac{y^2}{p}=1\) 的一个焦点，则 \(p=\)（\quad）

\choices
    {\(2\)}
    {\(3\)}
    {\(4\)}
    {\(8\)}

\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
抛物线 \(y^2=2px\) 的焦点为 \(\left(\dfrac p2,0\right)\). 椭圆 \(\dfrac{x^2}{3p}+\dfrac{y^2}{p}=1\) 的焦点为 \(\left(\pm\sqrt{3p-p},0\right)=\left(\pm\sqrt{2p},0\right)\). 由焦点重合得 \(\frac p2=\sqrt{2p}\)，解得 \(p=8\). 故选 D.
\end{solution}

\begin{problem}
下列函数中，以 \(\dfrac{\pi}{2}\) 为周期且在区间 \(\left(\dfrac{\pi}{4},\dfrac{\pi}{2}\right)\) 单调递增的是（\quad）

\choices
    {\(f(x)=|\cos 2x|\)}
    {\(f(x)=|\sin 2x|\)}
    {\(f(x)=\cos|x|\)}
    {\(f(x)=\sin|x|\)}

\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
\(f(x)=|\sin 2x|\) 的周期为 \(\dfrac{\pi}{2}\)，但在 \(\left(\dfrac{\pi}{4},\dfrac{\pi}{2}\right)\) 上单调递减；\(f(x)=\cos|x|\) 的周期不是 \(\dfrac{\pi}{2}\)；\(f(x)=\sin|x|\) 也不是周期函数. 对 \(f(x)=|\cos 2x|\)，当 \(x\in\left(\dfrac{\pi}{4},\dfrac{\pi}{2}\right)\) 时，\(\cos 2x<0\)，所以 \(f(x)=-\cos 2x\)，且它在该区间单调递增. 故选 A.
\end{solution}

\begin{problem}
已知 \(\alpha\in\left(0,\dfrac{\pi}{2}\right)\)，\(2\sin2\alpha=\cos2\alpha+1\)，则 \(\sin\alpha=\)（\quad）

\choices
    {\(\dfrac15\)}
    {\(\dfrac{\sqrt5}{5}\)}
    {\(\dfrac{\sqrt3}{3}\)}
    {\(\dfrac{2\sqrt5}{5}\)}

\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
由 \(2\sin2\alpha=\cos2\alpha+1\) 及 \(\sin2\alpha=2\sin\alpha\cos\alpha\)，\(\cos2\alpha+1=2\cos^2\alpha\)，得 \(4\sin\alpha\cos\alpha=2\cos^2\alpha\). 因为 \(\alpha\in\left(0,\dfrac{\pi}{2}\right)\)，所以 \(\cos\alpha>0\)，于是 \(2\sin\alpha=\cos\alpha\). 两边平方，得 \(4\sin^2\alpha=1-\sin^2\alpha\)，所以 \(5\sin^2\alpha=1\)，\(\sin\alpha=\frac{\sqrt5}{5}\). 故选 B.
\end{solution}

\begin{problem}
设 \(F\) 为双曲线 \(C:\dfrac{x^2}{a^2}-\dfrac{y^2}{b^2}=1\)（\(a>0\)，\(b>0\)）的右焦点，\(O\) 为坐标原点，以 \(OF\) 为直径的圆与圆 \(x^2+y^2=a^2\) 交于 \(P\)，\(Q\) 两点. 若 \(|PQ|=|OF|\)，则 \(C\) 的离心率为（\quad）

\choices
    {\(\sqrt2\)}
    {\(\sqrt3\)}
    {\(2\)}
    {\(\sqrt5\)}

\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
设双曲线的焦距参数为 \(c\)，则 \(F=(c,0)\)，离心率 \(e=\dfrac ca\). 圆 \(OF\) 的方程为 \(\left(x-\frac c2\right)^2+y^2=\frac{c^2}{4}\). 它与圆 \(x^2+y^2=a^2\) 联立消去 \(y\)，得交弦 \(PQ\) 所在直线为 \(cx=a^2\). 所以 \(|PQ|=2\sqrt{a^2-\frac{a^4}{c^2}}\). 由 \(|PQ|=|OF|=c\) 得 \(4\left(a^2-\frac{a^4}{c^2}\right)=c^2\). 令 \(e=\dfrac ca\)，化为 \(4\left(1-\frac1{e^2}\right)=e^2\)，整理得 \((e^2-2)^2=0\)，故 \(e=\sqrt2\). 故选 A.
\end{solution}

\begin{problem}
设函数 \(f(x)\) 的定义域为 \(\R\)，满足 \(f(x+1)=2f(x)\)，且当 \(x\in(0,1]\) 时，\(f(x)=x(x-1)\). 若对任意 \(x\in(-\infty,m]\)，都有 \(f(x)\ge -\dfrac89\)，则 \(m\) 的取值范围是（\quad）

\choices
    {\((-\infty,\dfrac94]\)}
    {\((-\infty,\dfrac73]\)}
    {\((-\infty,\dfrac52]\)}
    {\((-\infty,\dfrac83]\)}

\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
由 \(f(x+1)=2f(x)\) 可知，若 \(x=n+t\)（\(n\in\Z\)，\(t\in(0,1]\)），则 \(f(x)=2^n t(t-1)\). 在区间 \((0,1]\) 上，\(f(x)=x(x-1)\in\left[-\dfrac14,0\right)\)；在 \((1,2]\) 上，\(f(x)\in\left[-\dfrac12,0\right)\)；在 \((2,3]\) 上，\(f(x)=4t(t-1)\)（\(t=x-2\in(0,1]\)），其最小值为 \(-1\)，并且 \(4t(t-1)=-\frac89\) 时，\(t=\dfrac13\) 或 \(\dfrac23\)，对应 \(x=\dfrac73\) 或 \(\dfrac83\). 由于题目要求对任意 \(x\le m\) 都有 \(f(x)\ge-\dfrac89\)，所以 \(m\) 不能超过第一次跌破 \(-\dfrac89\) 的位置，即 \(m=\frac73\). 故选 B.
\end{solution}

\begin{problem}
11 分制乒乓球比赛，每赢一球得 \(1\) 分，当某局打成 \(10:10\) 平后，每球交换发球权，先多得 \(2\) 分的一方获胜，该局比赛结束. 甲，乙两位同学进行单打比赛，假设甲发球时甲得分的概率为 \(0.5\)，乙发球时甲得分的概率为 \(0.4\)，各球的结果相互独立. 在某局双方 \(10:10\) 平后，甲先发球，两人又打了 \(X\) 个球该局比赛结束.

\item 求 \(P(X=2)\)；
\item 求事件“\(X=4\) 且甲获胜”的概率.
\end{enumerate}

\end{problem}

\begin{solution}
设双方 \(10:10\) 平后的第 \(k\) 个球甲获胜为事件 \(A_k\)（\(k=1\)，\(2\)，\(3\)，\(\cdots\)）.

（1）有
\examdisplaychain{
P(X=2)&=P(A_1A_2)+P(\overline{A_1}\,\overline{A_2})\\
&=P(A_1)P(A_2)+P(\overline{A_1})P(\overline{A_2})\\
&=0.5\times0.4+0.5\times0.6=0.5
}

（2）有
\examdisplaychain{
P(X=4\text{ 且甲获胜})&=P(\overline{A_1}A_2A_3A_4)+P(A_1\overline{A_2}A_3A_4)\\
&=P(\overline{A_1})P(A_2)P(A_3)P(A_4)+P(A_1)P(\overline{A_2})P(A_3)P(A_4)\\
&=(0.5\times0.4+0.5\times0.6)\times0.5\times0.4=0.1
}
\end{solution}

\begin{problem}
已知数列 \(\{a_n\}\) 和 \(\{b_n\}\) 满足 \(a_1=1\)，\(b_1=0\)，\(4a_{n+1}=3a_n-b_n+4\)，\(4b_{n+1}=3b_n-a_n-4\).

\begin{enumerate}
\item 证明：\(\{a_n+b_n\}\) 是等比数列，\(\{a_n-b_n\}\) 是等差数列；
\item 求 \(\{a_n\}\) 和 \(\{b_n\}\) 的通项公式.
\end{enumerate}

\end{problem}

\begin{solution}
（1）由 \(4a_{n+1}=3a_n-b_n+4\)，\(4b_{n+1}=3b_n-a_n-4\) 相加得 \(4(a_{n+1}+b_{n+1})=2(a_n+b_n)\)，所以 \(a_{n+1}+b_{n+1}=\frac12(a_n+b_n)\). 又 \(a_1+b_1=1\)，故 \(\{a_n+b_n\}\) 是首项为 \(1\)，公比为 \(\dfrac12\) 的等比数列.

相减得 \(4(a_{n+1}-b_{n+1})=4(a_n-b_n)+8\)，所以 \(a_{n+1}-b_{n+1}=a_n-b_n+2\). 又 \(a_1-b_1=1\)，故 \(\{a_n-b_n\}\) 是首项为 \(1\)，公差为 \(2\) 的等差数列.

（2）由（1）可得 \(a_n+b_n=\left(\frac12\right)^{n-1}\)，\(a_n-b_n=1+2(n-1)=2n-1\). 联立解得 \(a_n=\frac12\left[\left(\frac12\right)^{n-1}+2n-1\right]=\frac{1}{2^n}+n-\frac12\)，\(b_n=\frac12\left[\left(\frac12\right)^{n-1}-(2n-1)\right]=\frac{1}{2^n}-n+\frac12\).
\end{solution}

\begin{problem}
已知函数 \(f(x)=\ln x-\dfrac{x+1}{x-1}\).

\begin{enumerate}
\item 讨论 \(f(x)\) 的单调性，并证明 \(f(x)\) 有且仅有两个零点；
\item 设 \(x_0\) 是 \(f(x)\) 的一个零点，证明曲线 \(y=\ln x\) 在点 \(A(x_0,\ln x_0)\) 处的切线也是曲线 \(y=\e^x\) 的切线.
\end{enumerate}

\end{problem}

\begin{solution}
（1）函数 \(f(x)=\ln x-\dfrac{x+1}{x-1}\) 的定义域为 \((0,1)\cup(1,+\infty)\). 其导数为 \(f'(x)=\frac1x+\frac{2}{(x-1)^2}>0\)（\(x>0\)，\(\ x\ne1\)），所以 \(f(x)\) 在 \((0,1)\) 和 \((1,+\infty)\) 上都单调递增.

在 \((0,1)\) 内取 \(x=\dfrac1{e^2}\)，\(\dfrac1e\)，有 \(f\!\left(\frac1{e^2}\right)<0\)，\(f\!\left(\frac1e\right)>0\)，故在 \((0,1)\) 内有且仅有一个零点.

在 \((1,+\infty)\) 内取 \(x=e\)，\(e^2\)，有 \(f(e)<0\)，\(f(e^2)>0\)，故在 \((1,+\infty)\) 内有且仅有一个零点.

因此 \(f(x)\) 在定义域内有且仅有两个零点.

（2）设 \(x_0\) 是 \(f(x)\) 的一个零点，则 \(\ln x_0=\frac{x_0+1}{x_0-1}\). 曲线 \(y=\ln x\) 在点 \(A(x_0,\ln x_0)\) 处的切线方程为 \(y-\ln x_0=\frac1{x_0}(x-x_0)\)，即 \(y=\frac1{x_0}x-1+\ln x_0=\frac1{x_0}x+\frac{2}{x_0-1}\). 而曲线 \(y=\e^x\) 在点 \(\left(\ln\dfrac1{x_0},\dfrac1{x_0}\right)\) 处的切线方程为 \(y-\frac1{x_0}=\frac1{x_0}\left(x-\ln\frac1{x_0}\right)\)，即 \(y=\frac1{x_0}x+\frac{2}{x_0-1}\). 所以曲线 \(y=\ln x\) 在点 \(A(x_0,\ln x_0)\) 处的切线也是曲线 \(y=\e^x\) 的切线.
\end{solution}

\begin{problem}
已知点 \(A(-2,0)\)，\(B(2,0)\)，动点 \(M(x,y)\) 满足直线 \(AM\) 与 \(BM\) 的斜率之积为 \(-\dfrac12\). 记 \(M\) 的轨迹为曲线 \(C\).

\begin{enumerate}
\item 求 \(C\) 的方程，并说明 \(C\) 是什么曲线；
\item 过坐标原点的直线交 \(C\) 于 \(P\)，\(Q\) 两点，点 \(P\) 在第一象限，\(PE\perp x\) 轴，垂足为 \(E\)，连结 \(QE\) 并延长交 \(C\) 于点 \(G\).
\begin{enumerate}
\item 证明：\(\triangle PQG\) 是直角三角形；
\item 求 \(\triangle PQG\) 面积的最大值.
\end{enumerate}
\end{enumerate}

\end{problem}

\begin{solution}
（1）由题意，\(\frac{y}{x+2}\cdot\frac{y}{x-2}=-\frac12\)，整理得 \(\frac{x^2}{4}+\frac{y^2}{2}=1\)（\(y\ne0\)）. 所以曲线 \(C\) 是焦点在 \(x\) 轴上不含长轴端点的椭圆.

（2）（1）设 \(P(x_0,y_0)\)，则 \(Q(-x_0,-y_0)\)，\(E(x_0,0)\)，\(G(x_G,y_G)\).
直线 \(QE\) 的方程为 \(y=\frac{y_0}{2x_0}(x-x_0)\). 与椭圆方程联立，消去 \(y\)，得 \(G\) 点坐标满足 \((2x_0^2+y_0^2)x^2-2x_0y_0^2x+x_0^2y_0^2-8x_0^2=0\).
因为点 \(Q\) 在椭圆上，所以 \(x=-x_0\) 是上述方程的一个根，从而
\(x_0x_G=\frac{8x_0^2-x_0^2y_0^2}{2x_0^2+y_0^2}\)，即 \(x_G=\frac{(8-y_0^2)x_0}{2x_0^2+y_0^2}\).
于是
\(y_G=\frac{y_0}{2x_0}(x_G-x_0)=\frac{y_0(4-x_0^2-y_0^2)}{2x_0^2+y_0^2}\).
故
\(k_{PG}=\frac{y_G-y_0}{x_G-x_0}=\frac{y_0(4-3x_0^2-2y_0^2)}{2x_0(4-x_0^2-y_0^2)}\).
又因为点 \(P\) 在椭圆 \(C\) 上，所以
\(x_0^2+2y_0^2=4\).
代入上式，得
\(k_{PG}=\frac{y_0(4-3x_0^2-4+2x_0^2)}{2x_0(4-4+2y_0^2)}=-\frac{x_0}{y_0}\).
而 \(k_{PQ}=\frac{y_0-(-y_0)}{x_0-(-x_0)}=\frac{y_0}{x_0}\)，
所以
\(k_{PQ}\cdot k_{PG}=\frac{y_0}{x_0}\cdot\left(-\frac{x_0}{y_0}\right)=-1\).
故 \(PQ\perp PG\)，所以 \(\triangle PQG\) 为直角三角形.

（2）设
\(t=\frac{x_0}{y_0}+\frac{y_0}{x_0}\)，
则 \(t\ge2\)，且
\examdisplaychain{
S_{\triangle PQG}&=\frac12|PE|\,(x_G-x_Q)=\frac12y_0(x_G+x_0)\\
&=\frac12y_0\left[\frac{(8-y_0^2)x_0}{2x_0^2+y_0^2}+x_0\right]=\frac{y_0x_0(4+x_0^2)}{2x_0^2+y_0^2}
}
又由 \(x_0^2+2y_0^2=4\)，得
\examdisplaychain{
S_{\triangle PQG}&=\frac{8y_0x_0(x_0^2+y_0^2)}{(2x_0^2+y_0^2)(x_0^2+2y_0^2)}=\frac{8t}{2t^2+1}\\
&=\frac{8}{2t+\frac1t}
}
令
\(f(t)=2t+\frac1t\)，
在 \([2,+\infty)\) 上单调递增，因此
\(f(t)\ge f(2)=\frac92\). 所以 \(S_{\triangle PQG}\le \frac{8}{9/2}=\frac{16}{9}\)，
当 \(t=2\) 即 \(x_0=y_0=\dfrac{2\sqrt3}{3}\) 时取等号.
故 \(\triangle PQG\) 面积的最大值为 \(\dfrac{16}{9}\).
\end{solution}

\begin{problem}
在极坐标系中，\(O\) 为极点，点 \(M(\rho_0,\theta_0)\)（\(\rho_0>0\)）在曲线 \(C:\rho=4\sin\theta\) 上，直线 \(l\) 过点 \(A(4,0)\) 且与 \(OM\) 垂直，垂足为 \(P\).

\begin{enumerate}
\item 当 \(\theta_0=\dfrac{\pi}{3}\) 时，求 \(\rho_0\) 及 \(l\) 的极坐标方程；
\item 当 \(M\) 在 \(C\) 上运动且 \(P\) 在线段 \(OM\) 上时，求 \(P\) 点轨迹的极坐标方程.
\end{enumerate}

\end{problem}

\begin{solution}
（1）当 \(\theta_0=\dfrac{\pi}{3}\) 时，\(\rho_0=4\sin\frac{\pi}{3}=2\sqrt3\). 在直线 \(l\) 上任取一点 \((\rho,\theta)\)，由 \(l\perp OM\) 得 \(\rho\cos\left(\theta-\frac{\pi}{3}\right)=2\).

（2）设 \(P(\rho,\theta)\). 在直角三角形 \(OAP\) 中，由边与角的关系得 \(\rho=4\cos\theta\). 又因为 \(P\) 在线段 \(OM\) 上，所以 \(\theta\in\left[\frac{\pi}{4},\frac{\pi}{2}\right]\). 故 \(P\) 点轨迹的极坐标方程为 \(\rho=4\cos\theta\)，\(\theta\in\left[\frac{\pi}{4},\frac{\pi}{2}\right]\).

\end{solution}

\begin{problem}
已知 \(f(x)=|x-a|x+|x-2|(x-a)\).

\begin{enumerate}
\item 当 \(a=1\) 时，求不等式 \(f(x)<0\) 的解集；
\item 当 \(x\in(-\infty,1)\) 时，\(f(x)<0\)，求 \(a\) 的取值范围.
\end{enumerate}

\end{problem}

\begin{solution}
（1）当 \(a=1\) 时，\(f(x)=|x-1|x+|x-2|(x-1)\). 当 \(x<1\) 时，\(f(x)=-(x-1)^2\cdot2<0\)；当 \(x\ge1\) 时，\(f(x)=(x-1)(x+|x-2|)\ge0\). 所以不等式 \(f(x)<0\) 的解集为 \((-\infty,1)\).

（2）当 \(a\ge1\) 时，对任意 \(x\in(-\infty,1)\)，都有 \(f(x)=2(a-x)(x-1)<0\)，满足题意.

当 \(a<1\) 时，取 \(x\in(a,1)\)，有 \(f(x)=2(x-a)>0\)，不满足题意.

故 \(a\) 的取值范围为 \([1,+\infty)\).
\end{solution}
