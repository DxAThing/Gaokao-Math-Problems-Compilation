\begin{problem}
设函数 \(f(x)=\sin x\)，\(x\in\R\).

\begin{enumerate}
  \item 已知 \(\theta\in[0,2\pi)\)，函数 \(f(x+\theta)\) 是偶函数，求 \(\theta\) 的值；
  \item 求函数 \(y=\left[f\left(x+\dfrac{\pi}{12}\right)\right]^2+\left[f\left(x+\dfrac{\pi}{4}\right)\right]^2\) 的值域.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 因为函数 \(f(x+\theta)=\sin(x+\theta)\) 是偶函数，所以对任意 \(x\in\R\)，有
\[
\sin(\theta-x)=\sin(\theta+x).
\]
利用和差化积，得 \(2\cos\theta\sin(-x)=0\)，即 \(\cos\theta\sin x=0\) 对任意 \(x\in\R\) 成立，从而 \(\cos\theta=0\). 又 \(\theta\in[0,2\pi)\)，所以 \(\theta=\dfrac{\pi}{2}\) 或 \(\theta=\dfrac{3\pi}{2}\).

\item 利用 \(\sin^2u=\dfrac{1-\cos2u}{2}\)，得
\[
y=1-\frac12\left[\cos\left(2x+\frac{\pi}{6}\right)+\cos\left(2x+\frac{\pi}{2}\right)\right]=1-\frac{\sqrt3}{2}\cos\left(2x+\frac{\pi}{3}\right).
\]
因为 \(-1\le\cos\left(2x+\dfrac{\pi}{3}\right)\le1\)，所以函数值域为 \(\left[1-\dfrac{\sqrt3}{2},1+\dfrac{\sqrt3}{2}\right]\).
\end{enumerate}
\end{solution}

\begin{problem}
如图，已知三棱柱 \(ABC-A_1B_1C_1\)，平面 \(A_1AC_1C\perp\) 平面 \(ABC\)，\(\angle ABC=90^\circ\)，\(\angle BAC=30^\circ\)，\(A_1A=A_1C=AC\)，\(E\)，\(F\) 分别是 \(AC\)，\(A_1B_1\) 的中点.

\begin{enumerate}
  \item 证明：\(EF\perp BC\)；
  \item 求直线 \(EF\) 与平面 \(A_1BC\) 所成角的余弦值.
\end{enumerate}

\bitmapfigure[max width=0.28\textwidth]{img/2019/zhejiang/q19_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 取 \(AC=4\)，建立空间直角坐标系，使
\[
A(0,0,0),\qquad C(4,0,0),\qquad B(3,\sqrt3,0).
\]
此时 \(AB=2\sqrt3\)，\(BC=2\)，且 \(\angle ABC=90^\circ\)，\(\angle BAC=30^\circ\)，符合题设.

平面 \(A_1AC_1C\) 与平面 \(ABC\) 垂直，且其交线为 \(AC\)，故可取 \(A_1=(2,0,h)\). 由 \(A_1A=A_1C=AC=4\)，得
\[
2^2+h^2=4^2,
\]
所以 \(h=2\sqrt3\)，即 \(A_1=(2,0,2\sqrt3)\). 由于是三棱柱，\(\overrightarrow{AA_1}=\overrightarrow{BB_1}\)，从而 \(B_1=(5,\sqrt3,2\sqrt3)\). 因为 \(E\)，\(F\) 分别是 \(AC\)，\(A_1B_1\) 的中点，所以
\[
E=(2,0,0),\qquad F=\left(\frac72,\frac{\sqrt3}{2},2\sqrt3\right).
\]
于是
\[
\overrightarrow{EF}=\left(\frac32,\frac{\sqrt3}{2},2\sqrt3\right),\qquad \overrightarrow{BC}=(1,-\sqrt3,0),
\]
且 \(\overrightarrow{EF}\cdot\overrightarrow{BC}=\dfrac32-\dfrac32=0\). 因此 \(EF\perp BC\).

\item 取平面 \(A_1BC\) 内的两个相交向量
\[
u=\overrightarrow{BC}=(1,-\sqrt3,0),\qquad v=\overrightarrow{BA_1}=(-1,-\sqrt3,2\sqrt3).
\]
其法向量可取
\[
 n=u\times v=(-6,-2\sqrt3,-2\sqrt3),
\]
所以也可取 \(n=(3,\sqrt3,\sqrt3)\). 设直线 \(EF\) 与平面 \(A_1BC\) 所成的角为 \(\varphi\)，则
\[
\sin\varphi=\frac{|\overrightarrow{EF}\cdot n|}{|\overrightarrow{EF}|\,|n|}.
\]
由 \(\overrightarrow{EF}\cdot n=12\)，\(|\overrightarrow{EF}|^2=15\)，\(|n|^2=15\)，得 \(\sin\varphi=\dfrac{12}{15}=\dfrac45\). 因为 \(\varphi\) 是直线与平面的锐角，所以
\[
\cos\varphi=\sqrt{1-\sin^2\varphi}=\frac35.
\]
故所求余弦值为 \(\dfrac35\).
\end{enumerate}
\end{solution}

\begin{problem}
设等差数列 \(\{a_n\}\) 的前 \(n\) 项和为 \(S_n\)，\(a_3=4\)，\(a_4=S_3\)，数列 \(\{b_n\}\) 满足：对每个 \(n\in\N^*\)，\(S_n+b_n\)，\(S_{n+1}+b_n\)，\(S_{n+2}+b_n\) 成等比数列.

\begin{enumerate}
  \item 求数列 \(\{a_n\}\)，\(\{b_n\}\) 的通项公式；
  \item 记 \(C_n=\sqrt{\dfrac{a_n}{2b_n}}\)（\(n\in\N^*\)），证明：\(C_1+C_2+\cdots+C_n<2\sqrt n\)，\(n\in\N^*\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设等差数列 \(\{a_n\}\) 的首项为 \(a_1\)，公差为 \(d\). 由 \(a_3=4\)，得
\[
a_1+2d=4.
\]
又由 \(a_4=S_3\)，得
\[
a_1+3d=\frac32\left(2a_1+2d\right)=3a_1+3d,
\]
所以 \(a_1=0\)，进而 \(d=2\). 因此
\[
a_n=2n-2,\qquad S_n=\frac n2\left(a_1+a_n\right)=n(n-1).
\]

令 \(X_n=S_n+b_n\). 由于
\[
S_{n+1}-S_n=a_{n+1}=2n,\qquad S_{n+2}-S_n=a_{n+1}+a_{n+2}=4n+2,
\]
题设三个数成等比数列等价于
\[
(X_n+2n)^2=X_n(X_n+4n+2).
\]
展开并化简，得 \(4n^2=2X_n\)，所以 \(X_n=2n^2\). 从而
\[
b_n=X_n-S_n=2n^2-n(n-1)=n^2+n.
\]
故 \(a_n=2n-2\)，\(b_n=n^2+n\)，\(n\in\N^*\).

\item 由上面的通项公式，得
\[
C_k=\sqrt{\frac{a_k}{2b_k}}=\sqrt{\frac{2(k-1)}{2k(k+1)}}=\sqrt{\frac{k-1}{k(k+1)}}<\frac1{\sqrt{k}},
\]
其中 \(k\in\N^*\). 因此
\[
C_1+C_2+\cdots+C_n<\sum_{k=1}^n\frac1{\sqrt{k}}\le1+\int_1^n\frac{1}{\sqrt{x}}\,\mathrm dx=2\sqrt n-1<2\sqrt n.
\]
所以对任意 \(n\in\N^*\)，均有 \(C_1+C_2+\cdots+C_n<2\sqrt n\).
\end{enumerate}
\end{solution}

\begin{problem}
如图，已知点 \(F(1,0)\) 为抛物线 \(y^2=2px\)（\(p>0\)）的焦点，过点 \(F\) 的直线交抛物线于 \(A\)，\(B\) 两点，点 \(C\) 在抛物线上，使得 \(\triangle ABC\) 的重心 \(G\) 在 \(x\) 轴上，直线 \(AC\) 交 \(x\) 轴于点 \(Q\)，且 \(Q\) 在点 \(F\) 右侧. 记 \(\triangle AFG\)，\(\triangle CQG\) 的面积分别为 \(S_1\)，\(S_2\).

\begin{enumerate}
  \item 求 \(p\) 的值及抛物线的准线方程；
  \item 求 \(\dfrac{S_1}{S_2}\) 的最小值及此时点 \(G\) 的坐标.
\end{enumerate}

\bitmapfigure[max width=0.32\textwidth]{img/2019/zhejiang/q21_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 抛物线 \(y^2=2px\) 的焦点为 \(\left(\dfrac p2,0\right)\). 由焦点为 \(F(1,0)\)，得 \(\dfrac p2=1\)，所以 \(p=2\). 因而抛物线方程为 \(y^2=4x\)，其准线方程为
\[
x=-\frac p2=-1.
\]

\item 对抛物线 \(y^2=4x\)，用参数表示其上的点为 \((t^2,2t)\). 设
\[
A=(u^2,2u),\qquad B=(v^2,2v),\qquad C=(w^2,2w).
\]
过参数为 \(u\)，\(v\) 的两点的弦与 \(x\) 轴交点的横坐标为 \(-uv\). 因为直线 \(AB\) 过 \(F(1,0)\)，所以 \(-uv=1\)，即 \(uv=-1\). 又因为重心 \(G\) 在 \(x\) 轴上，所以
\[
u+v+w=0.
\]
按图取 \(A\) 在 \(x\) 轴上方，故可令 \(u>0\). 则 \(v=-\dfrac1u\)，\(w=\dfrac1u-u\). 令 \(s=u^2\)，则直线 \(AC\) 与 \(x\) 轴交点 \(Q\) 的横坐标为
\[
q=-uw=u^2-1=s-1.
\]
由点 \(Q\) 在点 \(F\) 的右侧，得 \(q>1\)，所以 \(s>2\).

重心坐标为
\[
G=\left(\frac{u^2+v^2+w^2}{3},0\right)=\left(\frac23\left(s+\frac1s-1\right),0\right).
\]
记其横坐标为 \(g\). 直接计算得
\[
g-1=\frac{(2s-1)(s-2)}{3s}>0,\qquad q-g=\frac{(s-2)(s+1)}{3s}>0.
\]
又因为 \(w<0\)，所以
\[
S_1=\frac12(g-1)\cdot2u=u(g-1),\qquad S_2=\frac12(q-g)\cdot(-2w)=(-w)(q-g).
\]
利用 \(-w=\dfrac{s-1}{u}\)，得
\[
\frac{S_1}{S_2}=\frac{u(g-1)}{\dfrac{s-1}{u}(q-g)}=\frac{s(2s-1)}{(s-1)(s+1)}=:R(s),\qquad s>2.
\]
于是
\[
R'(s)=\frac{(4s-1)(s^2-1)-2s(2s^2-s)}{(s^2-1)^2}=\frac{s^2-4s+1}{(s^2-1)^2}.
\]
在 \(s>2\) 时，导数在 \(2<s<2+\sqrt3\) 上为负，在 \(s>2+\sqrt3\) 上为正，故当 \(s=2+\sqrt3\) 时，\(R(s)\) 取得最小值. 由于此时 \(s^2-4s+1=0\)，有
\[
R(s)=\frac{s(2s-1)}{s^2-1}=\frac{s(2s-1)}{2(2s-1)}=\frac s2=\frac{2+\sqrt3}{2}.
\]
又 \(s=2+\sqrt3\) 时 \(\dfrac1s=2-\sqrt3\)，所以
\[
g=\frac23\left(2+\sqrt3+2-\sqrt3-1\right)=2.
\]
故 \(\dfrac{S_1}{S_2}\) 的最小值为 \(\dfrac{2+\sqrt3}{2}\)，此时 \(G=(2,0)\).
\end{enumerate}
\end{solution}

\begin{problem}
已知实数 \(a\neq0\)，设函数 \(f(x)=a\ln x+\sqrt{x+1}\)，\(x>0\).

\begin{enumerate}
  \item 当 \(a=-\dfrac34\) 时，求函数 \(f(x)\) 的单调区间；
  \item 对任意 \(x\in\left[\dfrac{1}{\e^2},+\infty\right)\) 均有 \(f(x)\le\dfrac{\sqrt x}{2a}\)，求 \(a\) 的取值范围.
\end{enumerate}

注：\(\e=2.71828\ldots\) 为自然对数的底数.
\end{problem}

\begin{solution}
\begin{enumerate}
\item 当 \(a=-\dfrac34\) 时，
\[
f'(x)=-\frac{3}{4x}+\frac{1}{2\sqrt{x+1}}=\frac{2x-3\sqrt{x+1}}{4x\sqrt{x+1}}.
\]
令 \(f'(x)=0\)，得 \(2x=3\sqrt{x+1}\). 因为 \(x>0\)，两边平方得
\[
4x^2=9(x+1),
\]
即 \((4x+3)(x-3)=0\)，所以定义域内的唯一驻点为 \(x=3\). 由 \(f'(1)<0\)，\(f'(4)>0\)，可知函数在 \((0,3)\) 上单调递减，在 \((3,+\infty)\) 上单调递增.

\item 先取题设区间内的 \(x=1\)，得
\[
\sqrt2=f(1)\le\frac{1}{2a}.
\]
因此 \(a>0\)，且 \(a\le\dfrac{1}{2\sqrt2}=\dfrac{\sqrt2}{4}\).

下面证明这个范围内的 \(a\) 均满足题设不等式. 记
\[
D_a(x)=\frac{\sqrt{x}}{2a}-a\ln x-\sqrt{x+1},\qquad a_0=\frac{1}{2\sqrt2}.
\]
当 \(x\ge1\) 时，由 \(0<a\le a_0\) 得
\[
D_a(x)\ge\sqrt{2x}-\frac{\ln x}{2\sqrt2}-\sqrt{x+1}=D_{a_0}(x).
\]
并且
\[
D_{a_0}'(x)=\frac{1}{\sqrt{2x}}-\frac{1}{2\sqrt2x}-\frac{1}{2\sqrt{x+1}}.
\]
通分后，导数符号等同于判断
\[
2\sqrt{x(x+1)}-\sqrt2x-\sqrt{x+1}
\]
的符号. 记 \(P=2x^2+3x-1\). 若 \(P\le0\)，则上式平方后左边与右边之差为负；若 \(P>0\)，则 \(x>\dfrac14\)，且
\[
P^2-8x^2(x+1)=(x-1)(4x^3+8x^2+5x-1),
\]
其中多项式 \(4x^3+8x^2+5x-1\) 在 \(x>\dfrac14\) 时为正. 因此上式的符号与 \(x-1\) 相同，从而 \(D_{a_0}'(x)\) 在 \(x\ge1\) 时非负，在 \(\dfrac1{\e^2}\le x\le1\) 时非正. 因而 \(D_{a_0}(x)\) 在 \(x\ge1\) 上递增，在 \(\left[\dfrac1{\e^2},1\right]\) 上递减，且 \(D_{a_0}(1)=0\)，所以在上述两个区间内均有 \(D_{a_0}(x)\ge0\).

当 \(\dfrac1{\e^2}\le x\le1\) 时，令 \(L=-\ln x\)，则 \(0\le L\le2\)，并且
\[
D_a(x)=\frac{\sqrt{x}}{2a}+aL-\sqrt{x+1}.
\]
若 \(\dfrac1{\e^2}\le x\le\dfrac14\)，由基本不等式得
\[
\frac{\sqrt{x}}{2a}+aL\ge\sqrt{2\sqrt{x}L}.
\]
又令 \(K(x)=2\sqrt{x}(-\ln x)-x-1\)，则
\[
K'(x)=\frac{-\ln x-2}{\sqrt{x}}-1<0
\]
在 \(\left[\dfrac1{\e^2},1\right]\) 上成立，所以
\[
K(x)\ge K\left(\frac14\right)=\ln4-\frac54>0.
\]
从而 \(2\sqrt{x}L\ge x+1\)，即 \(D_a(x)\ge0\).

若 \(\dfrac14\le x\le1\)，则
\[
\frac{\partial D_a(x)}{\partial a}=L-\frac{\sqrt{x}}{2a^2}\le L-4\sqrt{x}\le0.
\]
这里用到了函数 \(\dfrac{-\ln x}{\sqrt{x}}\) 在 \(\left[\dfrac14,1\right]\) 上递减，故 \(\dfrac{L}{\sqrt{x}}\le2\ln4<4\)，从而 \(L<4\sqrt{x}\). 所以 \(D_a(x)\ge D_{a_0}(x)\ge0\).

综上，题设不等式对任意 \(x\in\left[\dfrac1{\e^2},+\infty\right)\) 成立的充要条件为
\[
a\in\left(0,\frac{\sqrt2}{4}\right].
\]
\end{enumerate}
\end{solution}
