\begin{problem}
在 \( \triangle ABC \) 中，内角 \(A\)，\(B\)，\(C\) 所对的边分别为 \(a\)，\(b\)，\(c\). 已知 \( b+c=2a \)，\( 3c\sin B=4a\sin C \).

\begin{enumerate}
\item 求 \( \cos B \) 的值；
\item 求 \( \sin \left( 2B+\dfrac{\pi}{6} \right) \) 的值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由正弦定理 \(\dfrac{b}{\sin B}=\dfrac{c}{\sin C}\) 得 \(c\sin B=b\sin C\)，结合 \(3c\sin B=4a\sin C\) 得 \(3b\sin C=4a\sin C\). 因为 \(\sin C>0\)，所以 \(3b=4a\)，即 \(b=\dfrac{4a}{3}\). 又 \(b+c=2a\)，所以 \(c=2a-b=\dfrac{2a}{3}\). 由余弦定理，
\[
\cos B=\frac{a^2+c^2-b^2}{2ac}=\frac{a^2+\frac{4a^2}{9}-\frac{16a^2}{9}}{2\cdot a\cdot\frac{2a}{3}}=\frac{-\frac{a^2}{3}}{\frac{4a^2}{3}}=-\frac14
\]
\item 因为 \(0<B<\pi\)，由 \(\cos B=-\dfrac14\) 得 \(\sin B=\sqrt{1-\cos^2B}=\dfrac{\sqrt{15}}{4}\). 从而 \(\sin 2B=2\sin B\cos B=-\dfrac{\sqrt{15}}{8}\)，\(\cos 2B=2\cos^2B-1=-\dfrac78\). 所以
\[
\sin\left(2B+\frac\pi6\right)=\sin 2B\cos\frac\pi6+\cos 2B\sin\frac\pi6=-\frac{\sqrt{15}}{8}\times\frac{\sqrt3}{2}-\frac78\times\frac12=-\frac{3\sqrt5+7}{16}
\]
\end{enumerate}
\end{solution}


\begin{problem}
设甲，乙两位同学上学期间，每天 \( 7{:}30 \) 之前到校的概率均为 \( \dfrac{2}{3} \). 假定甲，乙两位同学到校情况互不影响，且任一同学每天到校情况相互独立.

\begin{enumerate}
\item 用 \(X\) 表示甲同学上学期间的三天中 \( 7{:}30 \) 之前到校的天数，求随机变量 \(X\) 的分布列和数学期望；
\item 设 \(M\) 为事件“上学期间的三天中，甲同学在 \( 7{:}30 \) 之前到校的天数比乙同学在 \( 7{:}30 \) 之前到校的天数恰好多 \( 2 \)”，求事件 \(M\) 发生的概率.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 每天 \(7{:}30\) 之前到校的概率为 \(\dfrac23\)，三天到校情况相互独立，所以 \(X\sim B\left(3,\dfrac23\right)\)，且 \(P(X=k)=\mathrm C_3^k\left(\dfrac23\right)^k\left(\dfrac13\right)^{3-k}\)（\(k=0\)，\(1\)，\(2\)，\(3\)）. 于是 \(P(X=0)=\dfrac1{27}\)，\(P(X=1)=\dfrac29\)，\(P(X=2)=\dfrac49\)，\(P(X=3)=\dfrac8{27}\). 所以 \(E(X)=0\times\dfrac1{27}+1\times\dfrac29+2\times\dfrac49+3\times\dfrac8{27}=2\)（也可由 \(E(X)=3\times\dfrac23=2\) 直接得到）.
\item 设乙同学在 \(7{:}30\) 之前到校的天数为 \(Y\)，则 \(X\) 与 \(Y\) 独立同分布. 事件 \(M\) 即 \(X=Y+2\)，包含 \(X=2\)，\(Y=0\) 与 \(X=3\)，\(Y=1\) 两种情形，所以 \(P(M)=P(X=2)P(Y=0)+P(X=3)P(Y=1)=\dfrac49\times\dfrac1{27}+\dfrac8{27}\times\dfrac29=\dfrac4{243}+\dfrac{16}{243}=\dfrac{20}{243}\).
\end{enumerate}
\end{solution}


\begin{problem}
如图，\( AE\perp \)平面 \(ABCD\)，\( CF\parallel AE \)，\( AD\parallel BC \)，\( AD\perp AB \)，\( AB=AD=1 \)，\( AE=BC=2 \).

\begin{enumerate}
\item 求证：\( BF\parallel \)平面 \(ADE\)；
\item 求直线 \(CE\) 与平面 \(BDE\) 所成角的正弦值；
\item 若二面角 \( E-BD-F \) 的余弦值为 \( \dfrac{1}{3} \)，求线段 \(CF\) 的长.
\end{enumerate}

\FigureLayoutDeclare{img/2019/tianjin_science/q17_fig1.png}{10.5cm}{37bp 45bp 42bp 27bp}
\begingroup
\bitmapfigure[float=inline,max width=8.5cm]{img/2019/tianjin_science/q17_fig1.png}
\endgroup
\FloatBarrier
\end{problem}

\begin{solution}
\begin{enumerate}
\item 因为 \(AE\perp\) 平面 \(ABCD\)，\(CF\parallel AE\)，所以 \(CF\perp\) 平面 \(ABCD\). 由 \(AD\parallel BC\)，\(AD\perp AB\) 及 \(AB=AD=1\)，\(AE=BC=2\)，以 \(A\) 为原点，\(\overrightarrow{AB}\)，\(\overrightarrow{AD}\)，\(\overrightarrow{AE}\) 分别为 \(x\) 轴、\(y\) 轴、\(z\) 轴建立空间直角坐标系，则 \(B(1,0,0)\)，\(D(0,1,0)\)，\(E(0,0,2)\)，\(C(1,2,0)\). 设 \(CF=h\)，则 \(F(1,2,h)\)，\(\overrightarrow{BF}=(0,2,h)\). 平面 \(ADE\) 即 \(x=0\) 平面，其法向量可取 \(\overrightarrow{AB}=(1,0,0)\)，由 \(\overrightarrow{BF}\cdot\overrightarrow{AB}=0\) 知 \(\overrightarrow{BF}\) 与平面 \(ADE\) 平行. 又 \(B\notin\) 平面 \(ADE\)，所以 \(BF\parallel\) 平面 \(ADE\).
\item 由（1）得 \(\overrightarrow{CE}=E-C=(-1,-2,2)\). 设平面 \(BDE\) 的法向量为 \(\bs n=(x,y,z)\)，由 \(\overrightarrow{BD}=(-1,1,0)\)，\(\overrightarrow{BE}=(-1,0,2)\)，得 \(-x+y=0\) 且 \(-x+2z=0\). 取 \(x=2\)，得 \(y=2\)，\(z=1\)，即 \(\bs n=(2,2,1)\). 设直线 \(CE\) 与平面 \(BDE\) 所成的角为 \(\theta\)，则 \(\sin\theta=\dfrac{|\overrightarrow{CE}\cdot\bs n|}{|\overrightarrow{CE}||\bs n|}=\dfrac{|-2-4+2|}{3\times3}=\dfrac49\).
\item 由（1）得 \(\overrightarrow{BD}=(-1,1,0)\)，\(\overrightarrow{BF}=(0,2,h)\). 设平面 \(BDF\) 的法向量为 \(\bs m=(x,y,z)\)，由 \(\bs m\cdot\overrightarrow{BD}=0\)，\(\bs m\cdot\overrightarrow{BF}=0\) 得 \(-x+y=0\) 且 \(2y+hz=0\). 取 \(y=h\)，得 \(x=h\)，\(z=-2\)，即 \(\bs m=(h,h,-2)\). 因为平面 \(BDE\) 的法向量为 \(\bs n=(2,2,1)\)，所以 \(\dfrac{|\bs n\cdot\bs m|}{|\bs n||\bs m|}=\dfrac{|2h+2h-2|}{3\sqrt{2h^2+4}}=\dfrac{|4h-2|}{3\sqrt{2h^2+4}}=\dfrac13\). 于是 \(|4h-2|=\sqrt{2h^2+4}\)，两边平方得 \(16h^2-16h+4=2h^2+4\)，即 \(14h^2-16h=0\)，解得 \(h=0\) 或 \(h=\dfrac87\). 因为 \(CF\) 是线段的长，\(h>0\)，所以 \(CF=\dfrac87\).
\end{enumerate}
\end{solution}


\begin{problem}
设椭圆 \( \dfrac{x^2}{a^2}+\dfrac{y^2}{b^2}=1 \)（\( a>b>0 \)）的左焦点为 \(F\)，上顶点为 \(B\). 已知椭圆的短轴长为 \( 4 \)，离心率为 \( \dfrac{\sqrt5}{5} \).

\begin{enumerate}
\item 求椭圆的方程；
\item 设点 \(P\) 在椭圆上，且异于椭圆的上、下顶点，点 \(M\) 为直线 \(PB\) 与 \( x \) 轴的交点，点 \(N\) 在 \( y \) 轴的负半轴上. 若 \( |ON|=|OF| \)（\(O\) 为原点），且 \( OP\perp MN \)，求直线 \(PB\) 的斜率.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由短轴长为 \(4\) 得 \(2b=4\)，即 \(b=2\). 又离心率 \(e=\dfrac ca=\dfrac{\sqrt5}{5}\)，且 \(c^2=a^2-b^2\)，所以 \(e^2=\dfrac{a^2-b^2}{a^2}=1-\dfrac{b^2}{a^2}=\dfrac15\)，解得 \(a^2=\dfrac{5b^2}{4}=5\). 故椭圆方程为 \(\dfrac{x^2}{5}+\dfrac{y^2}{4}=1\).
\item 由（1）知 \(c=1\)，左焦点 \(F(-1,0)\)，上顶点 \(B(0,2)\). 因为 \(N\) 在 \(y\) 轴负半轴上且 \(|ON|=|OF|=1\)，所以 \(N(0,-1)\). 设直线 \(PB\) 的方程为 \(y=kx+2\)（\(k\ne0\)），代入椭圆方程得 \(4x^2+5(kx+2)^2=20\)，即 \((4+5k^2)x^2+20kx=0\). 解得 \(x=0\)（对应点 \(B\)）或 \(x_P=-\dfrac{20k}{4+5k^2}\)，从而 \(y_P=kx_P+2=\dfrac{8-10k^2}{4+5k^2}\). 令 \(y=0\) 得 \(M\left(-\dfrac2k,0\right)\)，所以 \(\overrightarrow{MN}=\left(\dfrac2k,-1\right)\)，\(\overrightarrow{OP}=\left(-\dfrac{20k}{4+5k^2},\dfrac{8-10k^2}{4+5k^2}\right)\). 由 \(OP\perp MN\) 得 \(\overrightarrow{OP}\cdot\overrightarrow{MN}=x_P\cdot\dfrac2k-y_P=0\)，即 \(2x_P=ky_P\). 代入得 \(-\dfrac{40k}{4+5k^2}=\dfrac{8k-10k^3}{4+5k^2}\)，约去 \(k\ne0\) 得 \(-40=8-10k^2\)，解得 \(k^2=\dfrac{24}{5}\)，即 \(k=\pm\dfrac{2\sqrt{30}}5\). 经检验 \(k^2=\dfrac{24}{5}\) 时 \(P\) 异于椭圆的上、下顶点，故直线 \(PB\) 的斜率为 \(\dfrac{2\sqrt{30}}5\) 或 \(-\dfrac{2\sqrt{30}}5\).
\end{enumerate}
\end{solution}


\begin{problem}
设 \( \{a_n\} \) 是等差数列，\( \{b_n\} \) 是等比数列. 已知 \( a_1=4 \)，\( b_1=6 \)，\( b_2=2a_2-2 \)，\( b_3=2a_3+4 \).

\begin{enumerate}
\item 求 \( \{a_n\} \) 和 \( \{b_n\} \) 的通项公式；
\item 设数列 \( \{c_n\} \) 满足 \( c_1=1 \)，\( c_n=\begin{cases}
1, & 2^k<n<2^{k+1},\\
b_k, & n=2^k,
\end{cases} \) 其中 \( k\in \N^* \).
\begin{enumerate}
\item 求数列 \( \{a_{2^n}(c_{2^n}-1)\} \) 的通项公式；
\item 求 \( \sum_{i=1}^{2^n} a_ic_i \)（\( n\in \N^* \)）.
\end{enumerate}
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设 \(\{a_n\}\) 的公差为 \(d\)，\(\{b_n\}\) 的公比为 \(q\). 由 \(a_1=4\)，\(b_1=6\) 得 \(a_2=4+d\)，\(a_3=4+2d\)，\(b_2=6q\)，\(b_3=6q^2\). 由 \(b_2=2a_2-2\) 得 \(6q=2d+6\)，即 \(d=3q-3\)；由 \(b_3=2a_3+4\) 得 \(6q^2=4d+12\). 将 \(d=3q-3\) 代入得 \(6q^2=12q\)，即 \(q^2=2q\). 因为 \(q\ne0\)，所以 \(q=2\)，从而 \(d=3\). 故 \(a_n=4+3(n-1)=3n+1\)，\(b_n=6\times2^{n-1}=3\times2^n\).
\item
\begin{enumerate}
\item 由 \(c_n\) 的定义，\(c_{2^n}=b_n=3\times2^n\)，又 \(a_{2^n}=3\times2^n+1\)，故 \(a_{2^n}(c_{2^n}-1)=(3\times2^n+1)(3\times2^n-1)=9\times4^n-1\).
\item 当 \(i=2^k\)（\(1\le k\le n\)）时 \(c_i=b_k=3\times2^k\)，其余 \(i\le2^n\) 时 \(c_i=1\)，所以
\[
\sum_{i=1}^{2^n}a_ic_i=\sum_{i=1}^{2^n}(3i+1)+\sum_{k=1}^{n}a_{2^k}(c_{2^k}-1)=\frac32\times2^n(2^n+1)+2^n+\sum_{k=1}^{n}(9\times4^k-1)
\]
其中 \(\displaystyle\sum_{k=1}^{n}4^k=\dfrac{4(4^n-1)}{3}\)，所以 \(\displaystyle\sum_{k=1}^{n}(9\times4^k-1)=12(4^n-1)-n\). 从而 \(\displaystyle\sum_{i=1}^{2^n}a_ic_i=\dfrac32\times4^n+\dfrac52\times2^n+12\times4^n-12-n=27\times2^{2n-1}+5\times2^{n-1}-n-12\).
\end{enumerate}
\end{enumerate}
\end{solution}


\begin{problem}
设函数 \( f(x)=\e^x\cos x \)，\( g(x) \) 为 \( f(x) \) 的导函数.

\begin{enumerate}
\item 求 \( f(x) \) 的单调区间；
\item 当 \( x\in \left[ \dfrac{\pi}{4},\dfrac{\pi}{2} \right] \) 时，证明 \( f(x)+g(x)\left( \dfrac{\pi}{2}-x \right)\ge 0 \)；
\item 设 \( x_n \) 为函数 \( u(x)=f(x)-1 \) 在区间 \( \left( 2n\pi+\dfrac{\pi}{4},2n\pi+\dfrac{\pi}{2} \right) \) 内的零点，其中 \( n\in \N \)，证明
\(2n\pi+\dfrac{\pi}{2}-x_n<\dfrac{\e^{-2n\pi}}{\sin x_0-\cos x_0}\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item \(f'(x)=\e^x\cos x-\e^x\sin x=\e^x(\cos x-\sin x)=\sqrt2\,\e^x\cos\left(x+\dfrac\pi4\right)\). 由 \(f'(x)>0\) 得 \(\cos\left(x+\dfrac\pi4\right)>0\)，即 \(2k\pi-\dfrac\pi2<x+\dfrac\pi4<2k\pi+\dfrac\pi2\)，解得 \(2k\pi-\dfrac{3\pi}{4}<x<2k\pi+\dfrac\pi4\)（\(k\in\Z\)）. 所以 \(f(x)\) 的单调递增区间为 \(\left[2k\pi-\dfrac{3\pi}{4},2k\pi+\dfrac\pi4\right]\)，单调递减区间为 \(\left[2k\pi+\dfrac\pi4,2k\pi+\dfrac{5\pi}{4}\right]\)（\(k\in\Z\)）.
\item 因为 \(g(x)=f'(x)=\e^x(\cos x-\sin x)\)，所以 \(f(x)+g(x)\left(\dfrac\pi2-x\right)=\e^x\left[\cos x+(\cos x-\sin x)\left(\dfrac\pi2-x\right)\right]\). 令 \(t=\dfrac\pi2-x\)，则 \(t\in\left[0,\dfrac\pi4\right]\)，且 \(\cos x=\sin t\)，\(\sin x=\cos t\)，于是上式括号内为 \(h(t)=\sin t+t(\sin t-\cos t)\)，且 \(h(0)=0\). 因为 \(h'(t)=\cos t+(\sin t-\cos t)+t(\cos t+\sin t)=\sin t+t(\cos t+\sin t)>0\)（\(0<t\le\dfrac\pi4\)），所以 \(h(t)\) 在 \(\left[0,\dfrac\pi4\right]\) 上单调递增，故 \(h(t)\ge h(0)=0\). 又 \(\e^x>0\)，所以 \(f(x)+g(x)\left(\dfrac\pi2-x\right)\ge0\).
\item 当 \(x\in\left(2n\pi+\dfrac\pi4,2n\pi+\dfrac\pi2\right)\) 时，\(f'(x)=\sqrt2\,\e^x\cos\left(x+\dfrac\pi4\right)<0\)，所以 \(f(x)\) 在该区间上单调递减. 又 \(u\left(2n\pi+\dfrac\pi4\right)=\e^{2n\pi+\frac\pi4}\cdot\dfrac{\sqrt2}{2}-1>0\)，\(u\left(2n\pi+\dfrac\pi2\right)=-1<0\)，所以 \(x_n\) 存在且唯一. 记 \(x_0\) 为 \(u(x)\) 在 \(\left(\dfrac\pi4,\dfrac\pi2\right)\) 内的零点，则 \(\e^{x_0}\cos x_0=1\). 令 \(y=x_n-2n\pi\)，则 \(y\in\left(\dfrac\pi4,\dfrac\pi2\right)\) 且 \(\e^y\cos y=\e^{-2n\pi}\). 因为 \(\e^t\cos t\) 在 \(\left(\dfrac\pi4,\dfrac\pi2\right)\) 上单调递减，且 \(\e^y\cos y=\e^{-2n\pi}<1=\e^{x_0}\cos x_0\)，所以 \(y>x_0\). 设 \(H(t)=\left(\dfrac\pi2-t\right)(\sin x_0-\cos x_0)-\e^t\cos t\)（\(x_0\le t<\dfrac\pi2\)）. 因为 \(\dfrac\pi2-x_0\in\left(0,\dfrac\pi4\right)\)，所以 \(H(x_0)=\left(\dfrac\pi2-x_0\right)(\sin x_0-\cos x_0)-1<\dfrac\pi4\times1-1<0\)，且 \(H\left(\dfrac\pi2\right)=0\). 又 \(H''(t)=2\e^t\sin t>0\)，故 \(H(t)\) 为凸函数，凸函数在两点间的图象位于弦下方，因此当 \(x_0<t<\dfrac\pi2\) 时 \(H(t)<0\). 取 \(t=y\)，得 \(\left(\dfrac\pi2-y\right)(\sin x_0-\cos x_0)<\e^y\cos y=\e^{-2n\pi}\)，即 \(2n\pi+\dfrac\pi2-x_n<\dfrac{\e^{-2n\pi}}{\sin x_0-\cos x_0}\).
\end{enumerate}
\end{solution}
