\begin{problem}
已知函数 \(f(x)=x\left( 1-\ln x \right)\).

\begin{enumerate}
\item 讨论 \(f(x)\) 的单调性；
\item 设 \(a\)，\(b\) 为两个不相等的正数，且 \(b\ln a-a\ln b=a-b\)，证明 \(2<\frac{1}{a}+\frac{1}{b}<\e\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 短答案：\(f(x)\) 在 \((0,1)\) 上递增，在 \((1,+\infty)\) 上递减.
\item 短答案：结论为 \(2<\dfrac{1}{a}+\dfrac{1}{b}<\e\).
\end{enumerate}
\end{solution}
