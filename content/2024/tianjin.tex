\begin{problem}
下列函数是偶函数的是（\quad）

\choices
    {\(y=\dfrac{\e^x-x^2}{\e^x+x^2}\)}
    {\(y=\dfrac{\cos x-x^2}{x^2+1}\)}
    {\(y=\dfrac{\e^x-x}{\e^x+x}\)}
    {\(y=\dfrac{\sin x-x}{x^2+1}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
A 中，设
\(f(x)=\frac{\e^x-x^2}{\e^x+x^2}\)，
函数定义域为 \(\R\)，但
\(\frac{\e^{-1}-1}{\e^{-1}+1}\ne \frac{\e-1}{\e+1}\)，
所以 \(f(-1)\ne f(1)\)，A 错误.

B 中，设
\(g(x)=\frac{\cos x-x^2}{x^2+1}\)，
函数定义域为 \(\R\)，且
\(g(-x)=\frac{\cos(-x)-(-x)^2}{(-x)^2+1}=\frac{\cos x-x^2}{x^2+1}=g(x)\)，
所以 \(g(x)\) 为偶函数，B 正确.

C 中，设
\(h(x)=\frac{\e^x-x}{\e^x+x}\)，
由 \(h(1)\ne h(-1)\) 可知 \(h(x)\) 不是偶函数，C 错误.

D 中，设
\(\phi(x)=\frac{\sin x-x}{x^2+1}\)，
函数定义域为 \(\R\). 因为
\(\phi(1)=\frac{\sin 1-1}{2}\)，\(\phi(-1)=\frac{-\sin 1+1}{2}\)，
所以 \(\phi(1)\ne \phi(-1)\)，故 \(\phi(x)\) 不是偶函数，D 错误.
故选 B.
\end{solution}

\begin{problem}
若 \(a=4.2^{-0.2}\)，\(b=4.2^{0.2}\)，\(c=\log_{4.2}0.2\)，则 \(a\)，\(b\)，\(c\) 的大小关系为（\quad）

\choices
    {\(a<b<c\)}
    {\(a<c<b\)}
    {\(c<b<a\)}
    {\(c<a<b\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
因为 \(4.2>1\)，所以指数函数 \(y=4.2^x\) 单调递增，从而
\[
0<4.2^{-0.2}<1<4.2^{0.2}
\]
即 \(0<a<1<b\). 又因为 \(0<0.2<1\)，故
\(c=\log_{4.2}0.2<0\). 于是 \(c<a<b\). 故选 D.
\end{solution}

\begin{problem}
若 \(m\)，\(n\) 为两条不同的直线，\(\alpha\) 为一个平面，则下列结论中正确的是（\quad）

\choices
    {若 \(m\parallel \alpha\)，\(m\perp n\)，则 \(n\perp \alpha\)}
    {若 \(m\perp \alpha\)，\(m\perp n\)，则 \(n\perp \alpha\)}
    {若 \(m\parallel \alpha\)，\(n\perp \alpha\)，则 \(m\perp n\)}
    {若 \(m\perp \alpha\)，\(n\perp \alpha\)，则 \(m\perp n\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
A 中，若 \(m\parallel \alpha\)，\(m\perp n\)，则 \(n\) 不一定垂直于 \(\alpha\)，所以 A 错误.

B 中，若 \(m\perp \alpha\)，\(m\perp n\)，则 \(n\) 不一定垂直于 \(\alpha\)，所以 B 错误.

C 中，若 \(m\parallel \alpha\)，\(n\perp \alpha\)，过 \(m\) 作平面 \(\beta\)，使得 \(\beta\cap \alpha=s\). 因为 \(m\subset \beta\)，所以 \(m\parallel s\)；又 \(s\subset \alpha\)，所以 \(n\perp s\)，于是 \(m\perp n\)，C 正确.

D 中，若 \(m\perp \alpha\)，\(n\perp \alpha\)，则 \(m\parallel n\)，所以 D 错误.
故选 C.
\end{solution}

\begin{problem}
双曲线 \(\frac{x^2}{a^2}-\frac{y^2}{b^2}=1\)（\(a>0\)，\(b>0\)）的左，右焦点分别为 \(F_1\)，\(F_2\). \(P\) 是双曲线右支上一点，且直线 \(PF_2\) 的斜率为 2. \(\triangle PF_1F_2\) 是面积为 8 的直角三角形，则双曲线的方程为（\quad）

\choices
    {\(\dfrac{x^2}{2}-\dfrac{y^2}{8}=1\)}
    {\(\dfrac{x^2}{4}-\dfrac{y^2}{8}=1\)}
    {\(\dfrac{x^2}{8}-\dfrac{y^2}{2}=1\)}
    {\(\dfrac{x^2}{8}-\dfrac{y^2}{4}=1\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
如图，点 \(P\) 在第四象限，且 \(\angle F_1PF_2=90^\circ\).
\solutionfigure{\bitmapfigure[max width=4.6cm]{img/2024/tianjin/q8_solution_fig1.png}}
设 \(|PF_2|=m\). 由斜率 \(k_{PF_2}=2\) 可得
\(\sin\angle PF_2F_1=\frac{2}{\sqrt5}\).
又因为 \(PF_1\perp PF_2\)，所以另一锐角的正弦为
\(\frac{1}{\sqrt5}\).
由正弦定理可得
\(|PF_1|:|PF_2|:|F_1F_2|=2:1:\sqrt5\).
故
\(|PF_1|=2m\)，\(2c=|F_1F_2|=\sqrt5\,m\).
又
\[
S_{\triangle PF_1F_2}=\frac12|PF_1|\cdot |PF_2| =\frac12\cdot 2m\cdot m=8
\]
解得
\(m=2\sqrt2\).
于是
\(|PF_2|=2\sqrt2\)，\(|PF_1|=4\sqrt2\)，\(2c=2\sqrt{10}\)，
所以
\(c=\sqrt{10}\).
根据双曲线定义，
\(|PF_1|-|PF_2|=2a=2\sqrt2\)，
故
\(a=\sqrt2\)，\(b=\sqrt{c^2-a^2}=\sqrt8\).
因此双曲线方程为
\(\frac{x^2}{2}-\frac{y^2}{8}=1\).
故选 A.
\end{solution}

圆 \((x-1)^2+y^2=25\) 的圆心与抛物线 \(y^2=2px\)（\(p>0\)）的焦点 \(F\) 重合，且两曲线在第一象限的交点为 \(A\)，则原点到直线 \(AF\) 的距离为\fillinblank{}.

\begin{problem}
若函数 \(f(x)=2\sqrt{x^2-ax}-|ax-2|+1\) 恰有一个零点，则 \(a\) 的取值范围为\fillinblank{}.
\end{problem}

\begin{answer}
\((-\sqrt{3},-1)\cup(1,\sqrt{3})\)
\end{answer}

\begin{solution}
令 \(f(x)=0\)，即
\(2\sqrt{x^2-ax}=|ax-2|-1\).
由根号意义有
\(x^2-ax\ge 0\).

先讨论 \(a=0\). 此时 \(x\in \R\)，原方程化为
\(2\sqrt{x^2}=|-2|-1=1\)，
所以 \(x=\pm \frac{\sqrt{2}}{2}\)，不符合题意，舍去.

当 \(a>0\) 时，
\[
2\sqrt{x^2-ax}=|ax-2|-1=
\begin{cases}
ax-3,& x\ge \dfrac{2}{a},\\[4pt]
1-ax,& x<\dfrac{2}{a}.
\end{cases}
\]
记
\[
g(x)=2\sqrt{x^2-ax},\qquad
h(x)=
\begin{cases}
ax-3,& x\ge \dfrac{2}{a},\\[4pt]
1-ax,& x<\dfrac{2}{a}.
\end{cases}
\]
由 \(x^2-ax\ge 0\) 得 \(x\ge a\) 或 \(x\le 0\).
当 \(x\le 0\) 时，\(ax-2<0\)，所以
\(2\sqrt{x^2-ax}=1-ax\).
两边平方，得
\(4x^2-4ax=(1-ax)^2\)，
即
\((4-a^2)x^2-2ax-1=\bigl[(2+a)x+1\bigr]\bigl[(2-a)x-1\bigr]=0\).
当 \(a=2\) 时，解得 \(x=-\frac14\)；当 \(a\in(0,2)\) 时，\(x=-\frac{1}{2+a}<0\) 且 \(x=\frac{1}{2-a}>0\)；当 \(a>2\) 时，\(x=-\frac{1}{2+a}<0\) 且 \(x=\frac{1}{2-a}<0\). 所以当且仅当 \(a\in(0,2]\) 时，方程在 \(x\le 0\) 上恰有一个解.

接着讨论 \(x\ge a\). 在 \(a\in(0,2]\) 时，为使原方程恰有一个零点，这一部分必须无解. 函数 \(h(x)\) 关于 \(x=\frac{2}{a}\) 对称，且
\(h(x)=0\)
的两个解是
\(x=\frac{1}{a}\)，\(x=\frac{3}{a}\).
又 \(h(x)\) 在 \(\left(\frac{1}{a},\frac{2}{a}\right]\) 上单调递减，在 \(\left[\frac{2}{a},\frac{3}{a}\right)\) 上单调递增. 另一方面，
\(g(x)=2\sqrt{x^2-ax}=0\)
时 \(x=a\)，且 \(g(x)\) 在 \((a,+\infty)\) 上单调递增. 于是只需
\(\frac{1}{a}<a\)，\(\frac{3}{a}>a\)，
解得
\(1<a<\sqrt{3}\).

当 \(a<0\) 时，
\[
2\sqrt{x^2-ax}=|ax-2|-1=
\begin{cases}
ax-3,& x\le \dfrac{2}{a},\\[4pt]
1-ax,& x>\dfrac{2}{a}.
\end{cases}
\]
仍记
\[
g(x)=2\sqrt{x^2-ax},\qquad
h(x)=
\begin{cases}
ax-3,& x\le \dfrac{2}{a},\\[4pt]
1-ax,& x>\dfrac{2}{a}.
\end{cases}
\]
由 \(x^2-ax\ge 0\) 得 \(x\ge 0\) 或 \(x\le a\).

当 \(x\ge 0\) 时，\(ax-2<0\)，所以
\(2\sqrt{x^2-ax}=1-ax\).
同样平方后得到
\(\bigl[(2+a)x+1\bigr]\bigl[(2-a)x-1\bigr]=0\).
当 \(a=-2\) 时，解得 \(x=\frac14\)；当 \(a\in(-2,0)\) 时，\(x=-\frac{1}{2+a}<0\)（舍去），\(x=\frac{1}{2-a}>0\)；当 \(a<-2\) 时，两个根都为正数. 所以当且仅当 \(a\in[-2,0)\) 时，方程在 \(x\ge 0\) 上恰有一个解.

为了使原方程恰有一个零点，在 \(a\in[-2,0)\) 时，方程在 \(x\le a\) 上必须无解. 此时 \(h(x)\) 关于 \(x=\frac{2}{a}\) 对称，零点仍为
\(x=\frac{1}{a}\)，\(x=\frac{3}{a}\).
又 \(g(x)=0\) 时 \(x=a\)，并且 \(g(x)\) 在 \((-\infty,a)\) 上单调递减，所以只需
\(\frac{1}{a}>a\)，\(\frac{3}{a}<a\)，
解得
\(-\sqrt{3}<a<-1\).
综上，
\(a\in(-\sqrt{3},-1)\cup(1,\sqrt{3})\).
故答案为 \((-\sqrt{3},-1)\cup(1,\sqrt{3})\).
\end{solution}

\begin{problem}
在 \(\triangle ABC\) 中，角 \(A\)，\(B\)，\(C\) 所对的边分别为 \(a\)，\(b\)，\(c\)，已知 \(\cos B=\frac{9}{16}\)，\(b=5\)，\(\frac{a}{c}=\frac23\).
\begin{enumerate}
    \item 求 \(a\)；
    \item 求 \(\sin A\)；
    \item 求 \(\cos(B-2A)\) 的值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
    \item 设
\(a=2t\)，\(c=3t\)（\(t>0\)）.
由余弦定理，
\(b^2=a^2+c^2-2ac\cos B\)，
即
\(25=4t^2+9t^2-2\cdot 2t\cdot 3t\cdot \frac{9}{16}\).
解得
\(t=2\)，
故
\(a=4\)，\(c=6\).

    \item 因为
\[
\sin B=\sqrt{1-\cos^2 B} =\sqrt{1-\left(\frac{9}{16}\right)^2} =\frac{5\sqrt7}{16}
\]
由正弦定理
\(\frac{a}{\sin A}=\frac{b}{\sin B}\)
得
\[
\sin A=\frac{a\sin B}{b} =\frac{4}{5}\cdot \frac{5\sqrt7}{16} =\frac{\sqrt7}{4}
\]

    \item 因为 \(a<b\)，所以 \(A<B\)，从而 \(A\) 为锐角，于是
\(\cos A=\sqrt{1-\sin^2 A} =\sqrt{1-\frac{7}{16}} =\frac34\).
因此
\[
\sin 2A=2\sin A\cos A =2\cdot \frac{\sqrt7}{4}\cdot \frac34 =\frac{3\sqrt7}{8}
\]
\[
\cos 2A=2\cos^2 A-1 =2\cdot \left(\frac34\right)^2-1 =\frac18
\]
所以
\[
\cos(B-2A)=\cos B\cos 2A+\sin B\sin 2A =\frac{9}{16}\cdot \frac18+\frac{5\sqrt7}{16}\cdot \frac{3\sqrt7}{8} =\frac{57}{64}
\]
\end{enumerate}
\end{solution}

\begin{problem}
已知四棱柱 \(ABCD-A_1B_1C_1D_1\) 中，底面 \(ABCD\) 为梯形，\(AB\parallel CD\)，\(A_1A\perp\) 平面 \(ABCD\)，\(AD\perp AB\)，其中 \(AB=A_1A=2\)，\(AD=DC=1\).
\(N\) 是 \(B_1C_1\) 的中点，\(M\) 是 \(DD_1\) 的中点.
\begin{enumerate}
    \item 求证：\(D_1N\parallel\) 平面 \(CB_1M\)；
    \item 求平面 \(CB_1M\) 与平面 \(BB_1CC_1\) 的夹角余弦值；
    \item 求点 \(B\) 到平面 \(CB_1M\) 的距离.
\end{enumerate}

\bitmapfigure[max width=6.0cm]{img/2024/tianjin/q17_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
    \item 取 \(CB_1\) 的中点 \(P\)，连接 \(NP\)，\(MP\). 因为 \(N\) 是 \(B_1C_1\) 的中点，所以
\(NP\parallel CC_1\)，\(NP=\frac12CC_1\).
因为 \(M\) 是 \(DD_1\) 的中点，所以
\(D_1M\parallel CC_1\)，\(D_1M=\frac12DD_1=\frac12CC_1\).
故
\(D_1M\parallel NP\)，\(D_1M=NP\)，
所以四边形 \(D_1MPN\) 是平行四边形，从而
\(D_1N\parallel MP\).
又 \(MP\subset\) 平面 \(CB_1M\)，故
\(D_1N\parallel \text{平面 }CB_1M\).

    \item 以 \(A\) 为原点建立空间直角坐标系，则
\(A(0,0,0)\)，\(B(2,0,0)\)，\(B_1(2,0,2)\)，\(C(1,1,0)\)，\(C_1(1,1,2)\)，\(M(0,1,1)\).
于是
\(\overrightarrow{CB_1}=(1,-1,2)\)，\(\overrightarrow{CM}=(-1,0,1)\)，\(\overrightarrow{BB_1}=(0,0,2)\).
\solutionfigure{\bitmapfigure[max width=4.6cm]{img/2024/tianjin/q17_solution_fig1.png}}
设平面 \(CB_1M\) 的法向量为 \(\bs m=(x_1,y_1,z_1)\)，则
\(\bs m\cdot \overrightarrow{CB_1}=x_1-y_1+2z_1=0\)，\(\bs m\cdot \overrightarrow{CM}=-x_1+z_1=0\).
取 \(x_1=1\)，得
\(\bs m=(1,3,1)\).
平面 \(BB_1CC_1\) 的法向量可取
\(\bs n=(1,1,0)\).
因此两平面的夹角余弦为
\[
\cos\theta=\frac{|\bs m\cdot \bs n|}{|\bs m|\,|\bs n|} =\frac{1+3}{\sqrt{1+9+1}\sqrt{1+1}} =\frac{2\sqrt{22}}{11}
\]

    \item 点 \(B\) 到平面 \(CB_1M\) 的距离等于向量 \(\overrightarrow{BB_1}\) 在法向量方向上的投影长度，即
\[
d=\frac{|\overrightarrow{BB_1}\cdot \bs m|}{|\bs m|} =\frac{2}{\sqrt{11}} =\frac{2\sqrt{11}}{11}
\]
\end{enumerate}
\end{solution}

\begin{problem}
已知椭圆 \(\frac{x^2}{a^2}+\frac{y^2}{b^2}=1\)（\(a>b>0\)）的离心率 \(e=\frac12\). 左顶点为 \(A\)，下顶点为 \(B\)，\(C\) 是线段 \(OB\) 的中点，其中 \(S_{\triangle ABC}=\frac{3\sqrt3}{2}\).
\begin{enumerate}
    \item 求椭圆方程；
    \item 过点 \(\left(0,-\frac32\right)\) 的动直线与椭圆有两个交点 \(P\)，\(Q\). 在 \(y\) 轴上是否存在点 \(T\) 使得 \(\overrightarrow{TP}\cdot \overrightarrow{TQ}\le 0\)？若存在，求出这个 \(T\) 点纵坐标的取值范围；若不存在，请说明理由.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
    \item 设半焦距为 \(c\). 由离心率
\(e=\frac{c}{a}=\frac12\)
得
\(a=2c\).
又
\(b^2=a^2-c^2=3c^2\)，
故
\(b=\sqrt3\,c\).
于是
\(A(-2c,0)\)，\(B(0,-\sqrt3\,c)\)，\(C\left(0,-\frac{\sqrt3\,c}{2}\right)\).
所以
\(S_{\triangle ABC} =\frac12\cdot 2c\cdot \frac{\sqrt3\,c}{2} =\frac{\sqrt3}{2}c^2 =\frac{3\sqrt3}{2}\).
解得
\(c=\sqrt3\).
从而
\(a=2\sqrt3\)，\(b=3\)，
故椭圆方程为
\(\frac{x^2}{12}+\frac{y^2}{9}=1\).

    \item 设过点 \(\left(0,-\frac32\right)\) 的动直线方程为
\(y=kx-\frac32\).
设
\(P(x_1,y_1)\)，\(Q(x_2,y_2)\)，\(T(0,t)\).
联立 \(3x^2+4y^2=36\)，\(y=kx-\frac32\) 得
\((3+4k^2)x^2-12kx-27=0\).
因此
\(x_1+x_2=\frac{12k}{3+4k^2}\)，\(x_1x_2=-\frac{27}{3+4k^2}\).
又
\(\overrightarrow{TP}=(x_1,y_1-t)\)，\(\overrightarrow{TQ}=(x_2,y_2-t)\)，
故
\(\overrightarrow{TP}\cdot \overrightarrow{TQ} =x_1x_2+(y_1-t)(y_2-t)\).
代入 \(y_i=kx_i-\frac32\) 并化简可得
\(\overrightarrow{TP}\cdot \overrightarrow{TQ} =\frac{\bigl[(3+2t)^2-12t-45\bigr]k^2+3\left(\frac32+t\right)^2-27}{3+4k^2}\).
\solutionfigure{\bitmapfigure[max width=4.2cm]{img/2024/tianjin/q18_solution_fig1.png}}
要使其对任意 \(k\) 都不大于 0，只需 \((3+2t)^2-12t-45\le 0\)，\(3\left(\frac32+t\right)^2-27\le 0\).
解得
\(-3\le t\le \frac32\).
若直线斜率不存在，则动直线为
\(x=0\).
此时它与椭圆交于
\(P(0,3)\)，\(Q(0,-3)\).
所以
\(\overrightarrow{TP}\cdot \overrightarrow{TQ} =(0,3-t)\cdot(0,-3-t) =t^2-9\).
要使 \(\overrightarrow{TP}\cdot \overrightarrow{TQ}\le 0\)，需
\(-3\le t\le 3\).
与前面的 \(-3\le t\le \frac32\) 结合后，仍得
\(-3\le t\le \frac32\).
故存在点
\(T(0,t)\)（\(-3\le t\le \frac32\)）
使得 \(\overrightarrow{TP}\cdot \overrightarrow{TQ}\le 0\).
\end{enumerate}
\end{solution}

\begin{problem}
已知数列 \(\{a_n\}\) 是公比大于 0 的等比数列，其前 \(n\) 项和为 \(S_n\). 若 \(a_1=1\)，\(S_2=a_3-1\).
\begin{enumerate}
    \item 求数列 \(\{a_n\}\) 前 \(n\) 项和 \(S_n\)；
    \item 设 \(b_n=k\)（\(n=a_k\)），\(b_n=b_{n-1}+2k\)（\(a_k<n<a_{k+1}\)），其中 \(k\in \mathbb N^*\).
    \begin{enumerate}
        \item 当 \(k\ge 2\)，\(n=a_{k+1}\) 时，求证：\(b_{n-1}\ge a_k b_n\)；
        \item 求 \(\displaystyle \sum_{i=1}^{S_n}b_i\).
    \end{enumerate}
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
    \item 设等比数列公比为 \(q>0\). 由
\(S_2=a_3-1\)
得
\(a_1+a_2=a_3-1\).
代入 \(a_1=1\)，\(a_2=q\)，\(a_3=q^2\)，有
\(1+q=q^2-1\).
解得
\(q=2\)
（\(q=-1\) 舍去）.
因此
\(a_n=2^{n-1}\)，\(S_n=\frac{1-2^n}{1-2}=2^n-1\).

    \item 由（1）知
    \begin{enumerate}
        \item \(a_k=2^{k-1}\)，\(a_{k+1}=2^k\).
当 \(n=a_{k+1}=2^k\) 时，
\(a_k<n-1<a_{k+1}\).
由定义得
\(b_n=k+1\)，
\[
b_{n-1}=b_{a_k}+(a_{k+1}-a_k-1)\cdot 2k =k+2k(2^{k-1}-1) =k(2^k-1)
\]
于是
\(b_{n-1}-a_k b_n =k(2^k-1)-(k+1)2^{k-1} =(k-1)2^{k-1}-k\ge 0\)（\(k\ge 2\)）.
故
\(b_{n-1}\ge a_k b_n\).

        \item 因为 \(a_n=2^{n-1}\)，所以
\(S_n=2^n-1=a_{n+1}-1\).
当
\(2^{k-1}\le i\le 2^k-1\)
时，\(\{b_i\}\) 构成公差为 \(2k\) 的等差数列，且首项为 \(k\). 分段求和可得
\(\sum_{i=2^{k-1}}^{2^k-1} b_i =\frac19\Bigl[(3k-1)4^k-(3k-4)4^{k-1}\Bigr]\).
将各段自 \(k=1\) 到 \(k=n\) 累加，整理得
\(\sum_{i=1}^{S_n}b_i=\frac{(3n-1)4^n+1}{9}\).
    \end{enumerate}
\end{enumerate}
\end{solution}

\begin{problem}
设函数 \(f(x)=x\ln x\).
\begin{enumerate}
    \item 求 \(f(x)\) 图象上点 \((1,f(1))\) 处的切线方程；
    \item 若 \(f(x)\ge a(x-\sqrt{x})\) 在 \(x\in(0,+\infty)\) 时恒成立，求 \(a\) 的值；
    \item 若 \(x_1\)，\(x_2\in(0,1)\)，证明 \(|f(x_1)-f(x_2)|\le |x_1-x_2|^{1/2}\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
    \item 因为
\(f'(x)=\ln x+1\)，
所以
\(f(1)=0\)，\(f'(1)=1\).
故所求切线方程为
\(y=x-1\).

    \item 设
\(h(t)=t-1-\ln t\)，
则
\(h'(t)=1-\frac{1}{t}=\frac{t-1}{t}\).
当 \(0<t<1\) 时，\(h'(t)<0\)；当 \(t>1\) 时，\(h'(t)>0\). 所以 \(h(t)\) 在 \((0,1]\) 上递减，在 \([1,+\infty)\) 上递增，从而
\(h(t)\ge h(1)=0\)，
即
\(t-1\ge \ln t\)
且等号成立当且仅当 \(t=1\).
再令
\(t=\frac{1}{\sqrt{x}}>0\).
设
\(g(t)=a(t-1)-2\ln t\)，
则
\(f(x)-a(x-\sqrt{x}) =x\left[a(t-1)-2\ln t\right]\).
于是题设等价于对任意 \(t>0\)，恒有
\(g(t)=a(t-1)-2\ln t\ge 0\).
一方面，若对任意 \(t>0\) 都有 \(g(t)\ge 0\)，则
\[
0\le g(t)=a(t-1)-2\ln t=a(t-1)+2\ln \frac{1}{t}\le a(t-1)+2\left(\frac{1}{t}-1\right)=at+\frac{2}{t}-a-2
\]
取 \(t=2\)，得
\(0\le a-1\)，
故 \(a\ge 1>0\).
再取
\(t=\sqrt{\frac{2}{a}}\)，
则
\[
0\le a\sqrt{\frac{2}{a}}+\frac{2}{\sqrt{2/a}}-a-2=2\sqrt{2a}-a-2=-\left(\sqrt{a}-\sqrt{2}\right)^2
\]
故只能有
\(a=2\).
反过来，当 \(a=2\) 时，
\(g(t)=2(t-1)-2\ln t =2(t-1-\ln t)\ge 0\).
因为 \(t-1\ge \ln t\) 对一切 \(t>0\) 成立，所以 \(a=2\) 确实满足条件.
综上，\(a=2\).

    \item 先证：对 \(0<a<b\)，有
\(\ln a+1<\frac{f(b)-f(a)}{b-a}<\ln b+1\)
因为
\(f'(x)=\ln x+1\)
在 \((0,1)\) 上单调递增，所以由拉格朗日中值定理可得上式成立.
又由
\(f'(x)=\ln x+1\)
知 \(f(x)\) 在 \(\left(0,\frac1e\right]\) 上递减，在 \(\left[\frac1e,+\infty\right)\) 上递增.
不妨设 \(0<x_1\le x_2<1\).

若
\(\frac1e\le x_1\le x_2<1\)，
则 \(f(x)\) 在该区间上单调递增，所以
\[
|f(x_1)-f(x_2)| =f(x_2)-f(x_1) <(\ln x_2+1)(x_2-x_1) <x_2-x_1 <\sqrt{x_2-x_1}
\]

若
\(0<x_1\le x_2\le \frac1e\)，
则 \(f(x)\) 在该区间上单调递减，所以
\(|f(x_1)-f(x_2)| =f(x_1)-f(x_2) =x_1\ln x_1-x_2\ln x_2\).
对任意 \(c\in \left(0,\frac1e\right]\)，设
\(\varphi(x)=x\ln x-c\ln c-\sqrt{c-x}\)（\(0<x\le c\)）.
则
\(\varphi''(x)=\frac{1}{x}+\frac{1}{4(c-x)^{3/2}}>0\)，
所以 \(\varphi(x)\) 在 \((0,c)\) 上是凸函数.
又
\(\varphi(c)=0\).
设 \(u=\sqrt{c}\)，则 \(0<u\le \e^{-1/2}\)，并且
\[
-c\ln c=-2u^2\ln u=2u(-u\ln u)\le \frac{2u}{\e}<u=\sqrt{c}
\]
故
\(\varphi(0)=-c\ln c-\sqrt{c}<0\).
由凸函数的性质可知，对任意 \(x\in (0,c)\)，都有
\(\varphi(x)\le \left(1-\frac{x}{c}\right)\varphi(0)+\frac{x}{c}\varphi(c)<0\).
因此
\(x\ln x-c\ln c\le \sqrt{c-x}\).
令 \(c=x_2\)，\(x=x_1\)，便得
\(|f(x_1)-f(x_2)| =f(x_1)-f(x_2) \le \sqrt{x_2-x_1}\).

若
\(0<x_1\le \frac1e\le x_2<1\)，
则由前两种情形，分别作用于区间 \(\left[x_1,\frac1e\right]\) 与 \(\left[\frac1e,x_2\right]\)，可得
\(f(x_1)-f\left(\frac1e\right)\le \sqrt{\frac1e-x_1}\le \sqrt{x_2-x_1}\)，
\(f(x_2)-f\left(\frac1e\right)\le \sqrt{x_2-\frac1e}\le \sqrt{x_2-x_1}\).
又因为 \(f\left(\frac1e\right)\) 是 \(f(x)\) 在 \((0,1)\) 上的最小值，所以
\(|f(x_1)-f(x_2)| \le \max\left\{f(x_1)-f\left(\frac1e\right),\,f(x_2)-f\left(\frac1e\right)\right\}\).
因此
\(|f(x_1)-f(x_2)|\le \sqrt{x_2-x_1}\).
综上，
\(|f(x_1)-f(x_2)|\le |x_1-x_2|^{1/2}\).
\end{enumerate}
\end{solution}
