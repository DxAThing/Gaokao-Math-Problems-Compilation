\section{单选题}

\begin{problem}
已知集合 \(A = \{x \in \R \mid |x| \leqslant 2\}\)，\(B = \{x \in \R \mid x \leqslant 1\}\)，则 \(A \cap B =\)（\quad）

\choices
    {\((-\infty, 2]\)}
    {\([1,2]\)}
    {\([-2, 2]\)}
    {\([-2, 1]\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
由 \(A= [-2,2]\)，\(B=(-\infty,1]\)，所以 \(A\cap B=[-2,1]\)，故选 D.
\end{solution}

\begin{problem}
设变量 \(x\)，\(y\) 满足约束条件 \(\begin{cases}
3x + y - 6 \geqslant 0,\\
x - y - 2 \leqslant 0,\\
y - 3 \leqslant 0,
\end{cases}\) 则目标函数 \(z = y - 2x\) 的最小值为（\quad）

\choices
    {\(-7\)}
    {\(-4\)}
    {1}
    {2}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
三个约束的边界直线分别为 \(3x+y-6=0\)，\(x-y-2=0\) 和 \(y=3\). 可行域的顶点为 \((1,3)\)，\((5,3)\) 和 \((2,0)\)，其中 \((1,3)\) 是第一条边界直线与 \(y=3\) 的交点，\((5,3)\) 是第二条边界直线与 \(y=3\) 的交点，\((2,0)\) 是前两条边界直线的交点. 线性目标函数在可行域的顶点处取得最小值，且

\(z(1,3)=3-2\cdot1=1\)，\(z(5,3)=3-2\cdot5=-7\)，\(z(2,0)=0-2\cdot2=-4\).

因此 \(z\) 的最小值为 \(-7\)，故选 A.
\end{solution}

\begin{problem}
已知下列三个命题：

\begin{circlelist}
\item 若一个球的半径缩小到原来的 \(\frac{1}{2}\)，则其体积缩小到原来的 \(\frac{1}{8}\)；
\item 若两组数据的平均数相等，则它们的标准差也相等；
\item 直线 \(x + y + 1 = 0\) 与圆 \(x^{2} + y^{2} = \frac{1}{2}\) 相切.
\end{circlelist}

其中真命题的序号是（\quad）

\choices
    {\(\circled{1}\circled{2}\circled{3}\)}
    {\(\circled{1}\circled{2}\)}
    {\(\circled{1}\circled{3}\)}
    {\(\circled{2}\circled{3}\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
命题\(\circled{1}\)中，球的体积与半径的三次方成正比，半径变为原来的 \(\frac{1}{2}\) 后，体积变为原来的 \(\left(\frac{1}{2}\right)^3=\frac{1}{8}\)，命题\(\circled{1}\)为真.

命题\(\circled{2}\)不成立. 例如两组数据 \(0,2\) 和 \(1,1\) 的平均数都为 \(1\)，但第一组数据的标准差不为零，而第二组数据的标准差为零.

命题\(\circled{3}\)中，圆心为原点，半径为 \(\sqrt{\frac{1}{2}}=\frac{1}{\sqrt{2}}\)，原点到直线 \(x+y+1=0\) 的距离为 \(\frac{1}{\sqrt{1^2+1^2}}=\frac{1}{\sqrt{2}}\)，所以直线与圆相切，命题\(\circled{3}\)为真. 因而真命题的序号为 \(\circled{1}\)，\(\circled{3}\)，故选 C.
\end{solution}

\begin{problem}
已知双曲线 \(\frac{x^2}{a^2} - \frac{y^2}{b^2} = 1 \) （\(a > 0\)，\(b > 0\)）的两条渐近线与抛物线 \(y^2 = 2px\) （\(p > 0\)）的准线分别交于 \(A\)，\(B\) 两点，\(O\) 为坐标原点. 若双曲线的离心率为 2，\(\triangle AOB\) 的面积为 \(\sqrt{3}\)，则 \(p =\)（\quad）

\choices
    {1}
    {\(\frac{3}{2}\)}
    {2}
    {3}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
双曲线的渐近线为 \(y=\pm\frac{b}{a}x\)，抛物线 \(y^2=2px\) 的准线为 \(x=-\frac{p}{2}\). 因此两交点的纵坐标的绝对值均为 \(\frac{bp}{2a}\)，从而 \(AB=\frac{bp}{a}\)，原点到直线 \(AB\) 的距离为 \(\frac{p}{2}\). 所以

\[
[\triangle AOB]=\frac{1}{2}\cdot\frac{bp}{a}\cdot\frac{p}{2}=\frac{bp^2}{4a}
\]

由双曲线离心率为 \(2\)，得 \(\frac{c}{a}=2\)，即 \(c=2a\). 又 \(c^2=a^2+b^2\)，所以 \(b^2=3a^2\)，即 \(\frac{b}{a}=\sqrt{3}\). 代入面积条件得 \(\frac{\sqrt{3}p^2}{4}=\sqrt{3}\)，故 \(p^2=4\). 因为 \(p>0\)，所以 \(p=2\)，故选 C.
\end{solution}

\begin{problem}
在 \(\triangle ABC\) 中，\(\angle ABC = \frac{\pi}{4}\)，\(AB = \sqrt{2}\)，\(BC = 3\). 则 \(\sin \angle BAC =\)（\quad）

\choices
    {\(\frac{\sqrt{10}}{10}\)}
    {\(\frac{\sqrt{10}}{5}\)}
    {\(\frac{3\sqrt{10}}{10}\)}
    {\(\frac{\sqrt{5}}{5}\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
由余弦定理，

\[
AC^2=AB^2+BC^2-2AB\cdot BC\cos\angle ABC
=2+9-2\cdot\sqrt{2}\cdot3\cdot\frac{\sqrt{2}}{2}=5
\]

所以 \(AC=\sqrt{5}\). 三角形 \(ABC\) 的面积既可表示为

\[
\frac{1}{2}AB\cdot BC\sin\angle ABC
=\frac{1}{2}\cdot\sqrt{2}\cdot3\cdot\frac{\sqrt{2}}{2}=\frac{3}{2}
\]

也可表示为 \(\frac{1}{2}AB\cdot AC\sin\angle BAC\). 因此

\[
\sin\angle BAC=\frac{2\cdot\frac{3}{2}}{\sqrt{2}\cdot\sqrt{5}}
=\frac{3\sqrt{10}}{10}
\]

故选 C.
\end{solution}

\begin{problem}
函数 \( f(x) = 2^{x} |\log_{0.5} x| - 1 \) 的零点个数为（\quad）

\choices
    {1}
    {2}
    {3}
    {4}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
定义域为 \(x>0\). 方程 \(f(x)=0\) 等价于

\(2^x|\log_{0.5}x|=1\).

当 \(0<x<1\) 时，\(|\log_{0.5}x|=-\log_{2}x\). 令 \(\phi(x)=2^x(-\log_2x)\)，则

\(\phi'(x)=2^x\left(-\ln x-\frac{1}{x\ln 2}\right)<0\)，因为 \(-x\ln x\leqslant\frac{1}{\e}<\frac{1}{\ln 2}\). 又 \(\phi(0^+)=+\infty\)，\(\phi(1^-)=0\)，所以 \(\phi(x)=1\) 在 \((0,1)\) 内有且只有一个根.

当 \(x>1\) 时，令 \(\psi(x)=2^x\log_2x\)，则

\(\psi'(x)=2^x\left(\ln2\cdot\log_2x+\frac{1}{x\ln2}\right)>0\).

又 \(\psi(1^+)=0\)，\(\psi(+\infty)=+\infty\)，所以 \(\psi(x)=1\) 在 \((1,+\infty)\) 内有且只有一个根. 因此零点共有 \(2\) 个，故选 B.
\end{solution}

\begin{problem}
已知函数 \( f(x) = x(1 + a|x|) \). 设关于 \( x \) 的不等式 \( f(x + a) < f(x) \) 的解集为 \(A\). 若 \( \left[-\frac{1}{2}, \frac{1}{2}\right] \subseteq A \)，则实数 \( a \) 的取值范围是（\quad）

\choices
    {\(\left(\frac{1 - \sqrt{5}}{2}, 0\right)\)}
    {\(\left(\frac{1 - \sqrt{3}}{2}, 0\right)\)}
    {\(\left(\frac{1 - \sqrt{5}}{2}, 0\right) \cup \left(0, \frac{1 + \sqrt{3}}{2}\right)\)}
    {\(\left(-\infty, \frac{1 - \sqrt{5}}{2}\right)\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
取 \(x=0\). 若 \(a>0\)，则 \(f(a)=a(1+a^2)>0=f(0)\)，不满足题设；若 \(a=0\)，不等式也不成立. 因而设 \(a=-t\)，其中 \(t>0\).

对 \(x\in\left[-\frac{1}{2},\frac{1}{2}\right]\)，有

\[
f(x-t)-f(x)=-t\left[1+(x-t)|x-t|-x|x|\right]
\]

由于 \(-t<0\)，原不等式对所有这些 \(x\) 成立，当且仅当

\(D(x):=x|x|-(x-t)|x-t|<1\).

当 \(-\frac{1}{2}\leqslant x\leqslant0\) 时，\(D(x)=t^2-2tx\)，故最大值为 \(t^2+t\)，在 \(x=-\frac{1}{2}\) 处取得. 当 \(0\leqslant x\leqslant\min\{t,\frac{1}{2}\}\) 时，\(D(x)=x^2+(x-t)^2\) 是开口向上的二次函数，其最大值在端点处取得，且不超过 \(t^2+t\). 当 \(t\leqslant x\leqslant\frac{1}{2}\) 时（此区间存在时），\(D(x)=2tx-t^2\)，其最大值为 \(t-t^2<t^2+t\). 因此整个区间上 \(D(x)\) 的最大值为 \(t^2+t\).

故所需条件为 \(t^2+t<1\). 结合 \(t>0\)，得 \(0<t<\frac{\sqrt{5}-1}{2}\)，即

\(\frac{1-\sqrt{5}}{2}<a<0\).

故选 A.
\end{solution}

\section{填空题}

\begin{problem}
已知 \(a\)，\(b\in\R\)，\(\i\) 是虚数单位. 若 \((a + \i)(1 + \i) = b\i\)，则 \(a + b\i =\fillinblank{}\).
\end{problem}

\begin{answer}
\(1+2\i\)
\end{answer}

\begin{solution}
展开等式得 \((a+\i)(1+\i)=(a-1)+(a+1)\i=b\i\). 比较实部和虚部，得到 \(a-1=0\)，\(a+1=b\)，所以 \(a=1\)，\(b=2\)，从而 \(a+b\i=1+2\i\).
\end{solution}

\begin{problem}
\(\left(x - \frac{1}{\sqrt{x}}\right)^6\) 的二项展开式中的常数项为\fillinblank{}.
\end{problem}

\begin{answer}
\(15\)
\end{answer}

\begin{solution}
二项展开式的一般项为

\(\mathrm C_6^k x^{6-k}\left(-\frac{1}{\sqrt{x}}\right)^k=(-1)^k\mathrm C_6^k x^{6-\frac{3k}{2}}\).

要取得常数项，需满足 \(6-\frac{3k}{2}=0\)，所以 \(k=4\). 此时常数项为 \((-1)^4\mathrm C_6^4=15\).
\end{solution}

\begin{problem}
已知圆的极坐标方程为 \(\rho = 4\cos \theta\)，圆心为 \(C\)，点 \(P\) 的极坐标为 \(\left(4, \frac{\pi}{3}\right)\)，则 \(|CP| = \)\fillinblank{}.
\end{problem}

\begin{answer}
\(2\sqrt{3}\)
\end{answer}

\begin{solution}
由极坐标方程 \(\rho=4\cos\theta\)，得 \(\rho^2=4\rho\cos\theta\)，即 \(x^2+y^2=4x\)，所以圆心为 \(C=(2,0)\). 点 \(P\) 的直角坐标为 \(\left(4\cos\frac{\pi}{3},4\sin\frac{\pi}{3}\right)=(2,2\sqrt{3})\). 因此 \(|CP|=2\sqrt{3}\).
\end{solution}

\begin{problem}
在平行四边形 \(ABCD\) 中，\(AD = 1\)，\(\angle BAD = 60^{\circ}\)，\(E\) 为 \(CD\) 的中点. 若 \(\overrightarrow{AC} \cdot \overrightarrow{BE} = 1\)，则 \(AB\) 的长为\fillinblank{}.
\end{problem}

\begin{answer}
\(\frac{1}{2}\)
\end{answer}

\begin{solution}
设 \(\overrightarrow{AB}=\bs{u}\)，\(\overrightarrow{AD}=\bs{v}\)，则 \(|\bs{v}|=1\)，\(|\bs{u}|=AB\)，且 \(\bs{u}\cdot\bs{v}=AB\cos60^\circ=\frac{AB}{2}\). 取 \(A\) 为原点，则 \(\overrightarrow{AC}=\bs{u}+\bs{v}\). 因为 \(E\) 为 \(CD\) 的中点，所以 \(\overrightarrow{BE}=\bs{v}-\frac{1}{2}\bs{u}\). 于是

\[
\overrightarrow{AC}\cdot\overrightarrow{BE}
=(\bs{u}+\bs{v})\cdot\left(\bs{v}-\frac{1}{2}\bs{u}\right)
=1+\frac{AB}{4}-\frac{AB^2}{2}
\]

由题意该内积等于 \(1\)，故 \(AB\left(\frac{1}{4}-\frac{AB}{2}\right)=0\). 因平行四边形非退化，\(AB>0\)，所以 \(AB=\frac{1}{2}\).
\end{solution}

\begin{problem}
如图，\(\triangle ABC\) 为圆的内接三角形，\(BD\) 为圆的弦，且 \(BD \parallel AC\). 过点 \(A\) 作圆的切线与 \(DB\) 的延长线交于点 \(E\)，\(AD\) 与 \(BC\) 交于点 \(F\). 若 \(AB = AC\)，\(AE = 6\)，\(BD = 5\)，则线段 \(CF\) 的长为\fillinblank{}.
\bitmapfigure[max width=0.42\linewidth]{img/2013/tianjin_science/q13_fig1.png}
\end{problem}

\begin{answer}
\(\frac{8}{3}\)
\end{answer}

\begin{solution}
由点 \(E\) 的切线与割线定理，\(EA^2=EB\cdot ED\). 设 \(EB=x\)，则 \(ED=x+BD=x+5\)，所以 \(x(x+5)=36\)，解得 \(x=4\)（舍去负根），即 \(EB=4\).

由切线弦定理和 \(BD\parallel AC\)，有 \(\angle EAB=\angle DAC\)，又有 \(\angle EBA=\angle ACD\)，故 \(\triangle EAB\sim\triangle DAC\). 因 \(AB=AC\)，相似比为 \(1\)，于是 \(AD=AE=6\)，\(CD=EB=4\)，且 \(AB=AC\).

因 \(A\)，\(B\)，\(D\)，\(C\) 四点共圆且 \(AC\parallel BD\)，四边形 \(ABDC\) 是圆内接梯形，故为等腰梯形，从而 \(AB=CD=4\). 结合 \(AB=AC\)，得 \(AC=4\). 等腰梯形的两条对角线相等，所以 \(BC=AD=6\). 又因为 \(AC\parallel BD\)，且 \(F\in AD\)、\(F\in BC\)，所以 \(\triangle AFC\sim\triangle FDB\)，从而

\(\frac{CF}{FB}=\frac{AC}{BD}=\frac{4}{5}\).

由 \(BC=CF+FB=6\)，得 \(CF=\frac{4}{4+5}\cdot6=\frac{8}{3}\).
\end{solution}

\begin{problem}
设 \(a + b = 2\)，\(b > 0\)，则当 \(a =\fillinblank{}\) 时，\(\frac{1}{2|a|} + \frac{|a|}{b}\) 取得最小值.
\end{problem}

\begin{answer}
\(-2\)
\end{answer}

\begin{solution}
因式子中含有 \(|a|\)，且 \(b=2-a>0\). 先讨论 \(a>0\). 此时 \(0<a<2\)，令

\(g(a)=\frac{1}{2a}+\frac{a}{2-a}\).

有 \(g'(a)=-\frac{1}{2a^2}+\frac{2}{(2-a)^2}\). 当 \(0<a<\frac{2}{3}\) 时 \(g'(a)<0\)，当 \(\frac{2}{3}<a<2\) 时 \(g'(a)>0\). 令 \(g'(a)=0\)，得 \(2a=2-a\)，即 \(a=\frac{2}{3}\)，此时 \(g(a)=\frac{5}{4}\)，这是正半轴上的最小值.

再讨论 \(a<0\). 令 \(t=-a>0\)，则 \(b=2+t\)，且

\(g=\frac{1}{2t}+\frac{t}{2+t}\).

有 \(g'(t)=-\frac{1}{2t^2}+\frac{2}{(2+t)^2}\). 当 \(0<t<2\) 时 \(g'(t)<0\)，当 \(t>2\) 时 \(g'(t)>0\). 令 \(g'(t)=0\)，得 \(2t=2+t\)，即 \(t=2\)，所以 \(a=-2\)，此时 \(g=\frac{1}{4}+\frac{1}{2}=\frac{3}{4}\). 当 \(t\to0^+\) 时 \(g(t)\to+\infty\)，当 \(t\to+\infty\) 时 \(g(t)\to1>\frac{3}{4}\)，结合上述导数符号可知这是负半轴上的最小值.

比较两个分支的最小值 \(\frac{3}{4}<\frac{5}{4}\)，所以当 \(a=-2\) 时原式取得最小值.
\end{solution}

\begin{problem}
已知函数 \(f(x) = -\sqrt{2} \sin \left(2x + \frac{\pi}{4}\right) + 6 \sin x \cos x - 2 \cos^2 x + 1\)，\(x \in \R\).
\begin{enumerate}
\item 求 \( f(x) \) 的最小正周期；
\item 求 \( f(x) \) 在区间 \([0, \frac{\pi}{2}]\) 上的最大值和最小值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由和差公式、二倍角公式，

\[
f(x)=-\left(\sin2x+\cos2x\right)+3\sin2x-\left(1+\cos2x\right)+1=2\sin2x-2\cos2x=2\sqrt{2}\sin\left(2x-\frac{\pi}{4}\right)
\]

因为正弦函数的最小正周期为 \(2\pi\)，所以 \(f(x)\) 的最小正周期为 \(\pi\).

\item 当 \(x\in\left[0,\frac{\pi}{2}\right]\) 时，令 \(u=2x\)，则 \(u\in[0,\pi]\)，且 \(f(x)=2(\sin u-\cos u)\). 记 \(h(u)=2(\sin u-\cos u)\)，则 \(h'(u)=2(\sin u+\cos u)\). 区间内的驻点为 \(u=\frac{3\pi}{4}\). 比较端点和驻点的函数值：

\(h(0)=-2\)，\(h(\pi)=2\)，\(h\left(\frac{3\pi}{4}\right)=2\sqrt{2}\). 因此最大值为 \(2\sqrt{2}\)，最小值为 \(-2\).
\end{enumerate}
\end{solution}

\begin{problem}
一个盒子里装有7张卡片，其中有红色卡片4张，编号分别为1， 2， 3， 4；白色卡片3张，编号分别为2， 3， 4. 从盒子中任取4张卡片（假设取到任何一张卡片的可能性相同）.
\begin{enumerate}
\item 求取出的 4 张卡片中，含有编号为 3 的卡片的概率.
\item 在取出的 4 张卡片中，红色卡片编号的最大值设为 \(X\)，求随机变量 \(X\) 的分布列和数学期望.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 从 7 张卡片中任取 4 张，共有 \(\mathrm C_7^4=35\) 种等可能结果. 编号为 3 的卡片有红、白两张，不含编号为 3 的卡片时只能从其余 5 张中取 4 张，因此

\(P(\text{含有编号为 3 的卡片})=1-\frac{\mathrm C_5^4}{\mathrm C_7^4}=1-\frac{5}{35}=\frac{6}{7}\).

\item 设红色卡片编号的最大值为 \(X\). 事件 \(X\leqslant m\) 表示所取卡片中没有编号大于 \(m\) 的红色卡片. 因此

\(P(X=1)=\frac{\mathrm C_3^3}{\mathrm C_7^4}=\frac{1}{35}\)，
\(P(X=2)=\frac{\mathrm C_5^4-\mathrm C_3^3}{\mathrm C_7^4}=\frac{4}{35}\)，
\(P(X=3)=\frac{\mathrm C_6^4-\mathrm C_5^4}{\mathrm C_7^4}=\frac{10}{35}=\frac{2}{7}\)，
\(P(X=4)=\frac{\mathrm C_7^4-\mathrm C_6^4}{\mathrm C_7^4}=\frac{20}{35}=\frac{4}{7}\).

故 \(X\) 的分布列为

\begin{center}
\begin{tabular}{c|cccc}
\(X\) & 1 & 2 & 3 & 4 \\
\hline
\(P\) & \(\frac{1}{35}\) & \(\frac{4}{35}\) & \(\frac{2}{7}\) & \(\frac{4}{7}\)
\end{tabular}
\end{center}

数学期望为 \(E(X)=1\cdot\frac{1}{35}+2\cdot\frac{4}{35}+3\cdot\frac{2}{7}+4\cdot\frac{4}{7}=\frac{17}{5}\).
\end{enumerate}
\end{solution}

\begin{problem}
如图所示，在四棱柱 \(ABCD - A_{1}B_{1}C_{1}D_{1}\) 中，侧棱 \(A_{1}A \perp\) 底面 \(ABCD\)，\(AB \parallel DC\)，\(AB \perp AD\)，\(AD = CD = 1\)，\(AA_{1} = AB = 2\)，\(E\) 为棱 \(AA_{1}\) 的中点.
\begin{enumerate}
\item 证明： \( B_{1}C_{1}\perp CE \)；
\item 求二面角 \(B_{1} - CE - C_{1}\) 的平面角的正弦值；
\item 设点 \(M\) 在线段 \(C_{1}E\) 上，且直线 \(AM\) 与平面 \(ADD_{1}A_{1}\) 所成角的正弦值为 \( \frac{\sqrt{2}}{6} \)，求线段 \(AM\) 的长.
\end{enumerate}
\bitmapfigure[max width=0.34\linewidth]{img/2013/tianjin_science/q17_fig1.png}
\end{problem}

\begin{solution}
建立直角坐标系，令 \(A=(0,0,0)\)，\(B=(2,0,0)\)，\(D=(0,1,0)\)，\(C=(1,1,0)\)，并令棱柱的高沿 \(z\) 轴方向. 则

\(A_{1}=(0,0,2)\)，\(B_{1}=(2,0,2)\)，\(C_{1}=(1,1,2)\)，\(D_{1}=(0,1,2)\)，\(E=(0,0,1)\).

\begin{enumerate}
\item \(\overrightarrow{B_{1}C_{1}}=(-1,1,0)\)，\(\overrightarrow{CE}=(-1,-1,1)\)，所以

\(\overrightarrow{B_{1}C_{1}}\cdot\overrightarrow{CE}=1-1+0=0\).

故 \(B_{1}C_{1}\perp CE\).

\item 取平面 \(B_{1}CE\) 和平面 \(C_{1}CE\) 的法向量

\(\bs{n}_{1}=\overrightarrow{CB_{1}}\times\overrightarrow{CE}=(1,-3,-2)\)，\(\bs{n}_{2}=\overrightarrow{CC_{1}}\times\overrightarrow{CE}=(2,-2,0)\).

两平面夹角的余弦值为

\[
\cos\alpha=\frac{|\bs{n}_{1}\cdot\bs{n}_{2}|}{|\bs{n}_{1}|\,|\bs{n}_{2}|}
=\frac{8}{\sqrt{14}\cdot2\sqrt{2}}=\frac{2}{\sqrt{7}}
\]

因此

\(\sin\alpha=\sqrt{1-\frac{4}{7}}=\frac{\sqrt{21}}{7}\).

\item 设 \(M\) 在线段 \(EC_{1}\) 上，令 \(M=E+\lambda(C_{1}-E)\)，则 \(0\leqslant\lambda\leqslant1\)，且 \(M=(\lambda,\lambda,1+\lambda)\). 平面 \(ADD_{1}A_{1}\) 的方程为 \(x=0\)，其法向量可取 \(\bs{n}=(1,0,0)\). 设直线 \(AM\) 与该平面所成的角为 \(\beta\)，则

\[
\sin\beta=\frac{|\overrightarrow{AM}\cdot\bs{n}|}{|\overrightarrow{AM}|}
=\frac{\lambda}{\sqrt{\lambda^2+\lambda^2+(1+\lambda)^2}}
=\frac{\lambda}{\sqrt{3\lambda^2+2\lambda+1}}
\]

由题意 \(\frac{\lambda}{\sqrt{3\lambda^2+2\lambda+1}}=\frac{\sqrt{2}}{6}\)，平方并整理得 \(15\lambda^2-2\lambda-1=0\). 解得 \(\lambda=\frac{1}{3}\) 或 \(\lambda=-\frac{1}{5}\)，结合 \(0\leqslant\lambda\leqslant1\)，得 \(\lambda=\frac{1}{3}\). 所以

\(AM=\sqrt{\left(\frac{1}{3}\right)^2+\left(\frac{1}{3}\right)^2+\left(\frac{4}{3}\right)^2}=\sqrt{2}\).
\end{enumerate}
\end{solution}

\begin{problem}
设椭圆 \(\frac{x^2}{a^2} + \frac{y^2}{b^2} = 1 \) （\(a > b > 0\)）的左焦点为 \(F\)，离心率为 \(\frac{\sqrt{3}}{3}\)，过点 \(F\) 且与 \(x\) 轴垂直的直线被椭圆截得的线段长为 \(\frac{4\sqrt{3}}{3}\).
\begin{enumerate}
\item 求椭圆的方程；
\item 设 \(A\)，\(B\) 分别为椭圆的左，右顶点，过点 \(F\) 且斜率为 \(k\) 的直线与椭圆交于 \(C\)，\(D\) 两点. 若 \(\overrightarrow{AC} \cdot \overrightarrow{DB} + \overrightarrow{AD} \cdot \overrightarrow{CB} = 8\)，求 \(k\) 的值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设椭圆的半焦距为 \(c\)，则离心率 \(e=\frac{c}{a}=\frac{\sqrt{3}}{3}\)，左焦点为 \(F=(-c,0)\). 直线 \(x=-c\) 与椭圆交点的纵坐标满足

\(\frac{c^2}{a^2}+\frac{y^2}{b^2}=1\)，所以弦长为

\[
2b\sqrt{1-\frac{c^2}{a^2}}=2b\sqrt{1-e^2}=2b\sqrt{\frac{2}{3}}=\frac{4\sqrt{3}}{3}
\]

由此得 \(b^2=2\). 又 \(c^2=a^2-b^2\)，且 \(c^2=\frac{a^2}{3}\)，所以 \(\frac{a^2}{3}=a^2-2\)，解得 \(a^2=3\). 因而椭圆方程为

\(\frac{x^2}{3}+\frac{y^2}{2}=1\).

\item 此时 \(A=(-\sqrt{3},0)\)，\(B=(\sqrt{3},0)\)，\(F=(-1,0)\). 直线过 \(F\) 且斜率为 \(k\)，故交点 \(C\)，\(D\) 的横坐标 \(x_1\)，\(x_2\) 满足 \(y=k(x+1)\) 以及

\[
2x^2+3k^2(x+1)^2=6
\]

由韦达定理，

\(x_1+x_2=-\frac{6k^2}{2+3k^2}\)，\(x_1x_2=\frac{3k^2-6}{2+3k^2}\).

又 \(y_i=k(x_i+1)\)，所以

\[
y_1y_2=k^2(x_1+1)(x_2+1)=-\frac{4k^2}{2+3k^2}
\]

由
\(\overrightarrow{AC}=(x_1+\sqrt{3},y_1)\)，
\(\overrightarrow{DB}=(\sqrt{3}-x_2,-y_2)\)，
\(\overrightarrow{AD}=(x_2+\sqrt{3},y_2)\)，
\(\overrightarrow{CB}=(\sqrt{3}-x_1,-y_1)\)，展开向量内积得

\[
\overrightarrow{AC}\cdot\overrightarrow{DB}+\overrightarrow{AD}\cdot\overrightarrow{CB}=6-2(x_1x_2+y_1y_2)
\]

题设等式给出 \(x_1x_2+y_1y_2=-1\). 又

\(x_1x_2+y_1y_2=-\frac{k^2+6}{2+3k^2}\)，故 \(-\frac{k^2+6}{2+3k^2}=-1\)，解得 \(k^2=2\). 因此 \(k=\pm\sqrt{2}\).
\end{enumerate}
\end{solution}

\begin{problem}
已知首项为 \(\frac{3}{2}\) 的等比数列 \(\{a_{n}\}\) 不是递减数列. 其前 \(n\) 项和为 \(S_{n}\) （\(n \in \N^{*}\)），且 \(S_{3} + a_{3}\)，\(S_{5} + a_{5}\)，\(S_{4} + a_{4}\) 成等差数列.
\begin{enumerate}
\item 求数列 \( \{a_{n}\} \) 的通项公式；
\item 设 \(T_{n} = S_{n} - \frac{1}{S_{n}}\) （\(n \in \N^{*}\)），求数列 \(\{T_{n}\}\) 的最大项的值与最小项的值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设等比数列的公比为 \(q\)，并记 \(U_n=S_n+a_n\). 则

\(U_3=\frac{3}{2}(1+q+2q^2)\)，\(U_4=\frac{3}{2}(1+q+q^2+2q^3)\)，\(U_5=\frac{3}{2}(1+q+q^2+q^3+2q^4)\).

题目按 \(U_3\)，\(U_5\)，\(U_4\) 的顺序说明三项成等差数列，所以 \(2U_5=U_3+U_4\)，即

\(q^2(4q^2-1)=0\).

当 \(q=0\) 或 \(q=\frac{1}{2}\) 时，数列为递减数列，与题设矛盾，故 \(q=-\frac{1}{2}\). 因此

\(a_n=\frac{3}{2}\left(-\frac{1}{2}\right)^{n-1}\).

由等比数列求和公式，

\(S_n=\frac{\frac{3}{2}\left[1-\left(-\frac{1}{2}\right)^n\right]}{1+\frac{1}{2}}=1-\left(-\frac{1}{2}\right)^n\).

\item 令 \(h(u)=u-\frac{1}{u}\)，则 \(h'(u)=1+\frac{1}{u^2}>0\). 当 \(n\) 为奇数时，\(S_n=1+2^{-n}>1\)，且随 \(n\) 增大而减小，所以 \(T_n=h(S_n)\) 的最大值在 \(n=1\) 时取得，为

\(T_1=\frac{3}{2}-\frac{2}{3}=\frac{5}{6}\).

当 \(n\) 为偶数时，\(S_n=1-2^{-n}<1\)，且随 \(n\) 增大而增大，所以 \(T_n=h(S_n)\) 的最小值在 \(n=2\) 时取得，为

\(T_2=\frac{3}{4}-\frac{4}{3}=-\frac{7}{12}\).

奇数项的 \(T_n\) 为正，偶数项的 \(T_n\) 为负，故数列 \(\{T_n\}\) 的最大项的值为 \(\frac{5}{6}\)，最小项的值为 \(-\frac{7}{12}\).
\end{enumerate}
\end{solution}

\begin{problem}
已知函数 \( f(x) = x^2 \ln x \).
\begin{enumerate}
\item 求函数 \( f(x) \) 的单调区间；
\item 证明：对任意的 \(t > 0\)，存在唯一的 \(s\)，使 \(t = f(s)\).
\item 设（2）中所确定的 \(s\) 关于 \(t\) 的函数为 \(s = g(t)\)，证明：当 \(t > \e^2\) 时，有 \( \frac {2}{5} < \frac {\ln g(t)}{\ln t} < \frac {1}{2} \).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 函数定义域为 \((0,+\infty)\)，且 \(f'(x)=2x\ln x+x=x(2\ln x+1)\). 因为 \(x>0\)，所以当 \(0<x<\frac{1}{\sqrt{\e}}\) 时 \(f'(x)<0\)，当 \(x>\frac{1}{\sqrt{\e}}\) 时 \(f'(x)>0\). 故 \(f(x)\) 在 \(\left(0,\frac{1}{\sqrt{\e}}\right)\) 上单调递减，在 \(\left(\frac{1}{\sqrt{\e}},+\infty\right)\) 上单调递增.

\item 当 \(t>0\) 时，若 \(0<s\leqslant1\)，则 \(f(s)\leqslant0\)，所以满足 \(t=f(s)\) 的 \(s\) 必须大于 1. 函数 \(f\) 在 \([1,+\infty)\) 上连续且严格递增，且 \(f(1)=0\)，\(\lim_{s\to+\infty}f(s)=+\infty\). 因此由连续性存在 \(s>1\) 使 \(f(s)=t\)，由严格单调性知该 \(s\) 唯一.

\item 由 \(t>\e^2=f(\e)\) 及 \(f\) 在 \((1,+\infty)\) 上严格递增，得 \(g(t)>\e\). 令 \(u=\ln g(t)\)，则 \(u>1\)，且

\(\ln t=\ln\left(g(t)^2\ln g(t)\right)=2u+\ln u\).

因为 \(u>1\)，所以 \(2u+\ln u>2u\)，从而 \(\frac{\ln g(t)}{\ln t}=\frac{u}{2u+\ln u}<\frac{1}{2}\).

另一方面，令 \(r(u)=u-2\ln u\)，则 \(r'(u)=1-\frac{2}{u}\)，故 \(r(u)\) 在 \((0,2)\) 上递减、在 \((2,+\infty)\) 上递增，从而 \(r(u)\geqslant r(2)=2-2\ln2>0\). 因此 \(u>2\ln u\)，即 \(2\ln u<u\). 于是

\(5u>4u+2\ln u=2(2u+\ln u)\)，所以 \(\frac{\ln g(t)}{\ln t}=\frac{u}{2u+\ln u}>\frac{2}{5}\).

综上，\(\frac{2}{5}<\frac{\ln g(t)}{\ln t}<\frac{1}{2}\).
\end{enumerate}
\end{solution}
