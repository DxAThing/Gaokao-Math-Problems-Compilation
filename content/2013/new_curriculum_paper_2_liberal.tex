\begin{problem}
已知等差数列 \(\{a_{n}\}\) 的公差不为零，\(a_{1} = 25\)，且 \(a_{1}\)，\(a_{11}\)，\(a_{13}\) 成等比数列.
\begin{enumerate}
\item 求 \(\{a_{n}\}\) 的通项公式；
\item 求 \(a_{1} + a_{4} + a_{7} + \cdots + a_{3n - 2}\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设等差数列的公差为 \(d\)，则 \(a_{11}=25+10d\)，\(a_{13}=25+12d\). 由 \(a_1\)，\(a_{11}\)，\(a_{13}\) 成等比数列，得
\[
(25+10d)^2=25(25+12d).
\]
化简得 \(100d(d+2)=0\). 因为公差不为零，所以 \(d=-2\)，从而
\[
a_n=25+(n-1)(-2)=27-2n.
\]
\item 由上式，对于 \(k=1,2,\ldots,n\)，有
\[
a_{3k-2}=27-2(3k-2)=31-6k.
\]
这些项构成首项为 \(25\)，末项为 \(31-6n\) 的等差数列，因此
\[
a_1+a_4+a_7+\cdots+a_{3n-2}
=\frac{n\left[25+(31-6n)\right]}{2}=n(28-3n).
\]
\end{enumerate}
\end{solution}

\begin{problem}
如图，直三棱柱 \(ABC - A_{1}B_{1}C_{1}\) 中，\(D\)，\(E\) 分别是 \(AB\)，\(BB_{1}\) 的中点.

\begin{enumerate}
\item 证明： \(BC_{1} \parallel\) 平面 \(A_{1}CD\)；
\item 设 \(AA_{1} = AC = CB = 2\)，\(AB = 2\sqrt{2}\)，求三棱锥 \(C - A_{1}DE\) 的体积.
\end{enumerate}

\bitmapfigure[width=0.40\linewidth]{img/2013/new_curriculum_paper_2_liberal/q18_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由 \(\overrightarrow{BC_1}=\overrightarrow{DC}+\overrightarrow{DA_1}\)，且 \(\overrightarrow{DC}\) 与 \(\overrightarrow{DA_1}\) 都是平面 \(A_1CD\) 内的向量，知 \(\overrightarrow{BC_1}\) 平行于平面 \(A_1CD\). 又 \(B\notin\) 平面 \(A_1CD\)，所以 \(BC_1\parallel\) 平面 \(A_1CD\).
\item 由 \(AC=CB=2\)，\(AB=2\sqrt{2}\)，知 \(\angle ACB=\frac{\pi}{2}\)，底面三角形 \(ABC\) 的面积为
\[
S_{\triangle ABC}=\frac12 AC\cdot CB=2.
\]
取 \(C\) 为原点，令 \(A=(2,0,0)\)，\(B=(0,2,0)\)，并令棱柱的高方向为 \(z\) 轴. 因为 \(AA_1=2\)，所以
\[
A_1=(2,0,2),\qquad D=(1,1,0),\qquad E=(0,2,1).
\]
于是三棱锥 \(C-A_1DE\) 的体积为
\[
V=\frac16\left|\begin{vmatrix}
2&0&2\\
1&1&0\\
0&2&1
\end{vmatrix}\right|=1.
\]
\end{enumerate}
\end{solution}

\begin{problem}
经销商经销某种农产品，在一个销售季度内，每售出 \(1\mathrm{~t}\) 该产品获利润 \(500\text{ 元}\). 未售出的产品，每 \(1\mathrm{~t}\) 亏损 \(300\text{ 元}\). 根据历史资料，得到销售季度内市场需求量的频率分布直方图，如图所示. 经销商为下一个销售季度购进了 \(130\mathrm{~t}\) 该农产品. 以 \(X\)（单位：\(\mathrm{t}\)，\(100 \leqslant X \leqslant 150\)）表示下一个销售季度的市场需求量，\(T\)（单位：\(\text{元}\)）表示下一个销售季度内经销该农产品的利润.

\begin{enumerate}
\item 将 \(T\) 表示为 \(X\) 的函数；
\item 根据直方图估计利润 \(T\) 不少于 57000 元的概率.
\end{enumerate}

\bitmapfigure[width=0.46\linewidth,float=inline]{img/2013/new_curriculum_paper_2_liberal/q19_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 当 \(100\leqslant X\leqslant 130\) 时，售出 \(X\mathrm{~t}\)，未售出 \(130-X\mathrm{~t}\)，所以
\[
T=500X-300(130-X)=800X-39000.
\]
当 \(130<X\leqslant 150\) 时，购进的产品全部售出，所以 \(T=500\times130=65000\). 因此
\[
T=\begin{cases}
800X-39000,&100\leqslant X\leqslant130,\\
65000,&130<X\leqslant150.
\end{cases}
\]
\item 当 \(100\leqslant X\leqslant130\) 时，\(T\geqslant57000\) 等价于
\[
800X-39000\geqslant57000,
\]
即 \(X\geqslant120\). 当 \(130<X\leqslant150\) 时，\(T=65000\geqslant57000\)，所以所求事件为 \(X\geqslant120\).

由直方图可知，五个组的组距均为 \(10\mathrm{~t}\)，各组频率分别为 \(0.10\)，\(0.20\)，\(0.30\)，\(0.25\)，\(0.15\). 因而
\[
P(X\geqslant120)=0.30+0.25+0.15=0.70.
\]
所以利润 \(T\) 不少于 \(57000\) 元的概率约为 \(0.70\).
\end{enumerate}
\end{solution}

\begin{problem}
在平面直角坐标系 \(xOy\) 中，已知圆 \(P\) 在 \(x\) 轴上截得线段长为 \(2\sqrt{2}\)，在 \(y\) 轴上截得线段长为 \(2\sqrt{3}\).
\begin{enumerate}
\item 求圆心 \(P\) 的轨迹方程；
\item 若 \(P\) 点到直线 \(y = x\) 的距离为 \(\frac{\sqrt{3}}{2}\)，求圆 \(P\) 的方程.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设圆心 \(P=(u,v)\)，半径为 \(r\). 圆在 \(x\) 轴上的弦长为 \(2\sqrt{2}\)，故圆心到 \(x\) 轴的距离为 \(|v|\)，从而
\[
r^2-v^2=2.
\]
同理，圆在 \(y\) 轴上的弦长为 \(2\sqrt{3}\)，所以
\[
r^2-u^2=3.
\]
两式相减得 \(v^2-u^2=1\)，即圆心 \(P\) 的轨迹方程为
\[
y^2-x^2=1.
\]
反之，轨迹上任意一点都可取 \(r^2=u^2+3=v^2+2\)，能够得到题设两条弦长，故这就是圆心的完整轨迹.
\item 由点 \(P=(u,v)\) 到直线 \(y=x\) 的距离条件，得
\[
\frac{|v-u|}{\sqrt{2}}=\frac{\sqrt{3}}{2},
\]
所以 \((v-u)^2=\frac{3}{2}\). 又因为
\[
(v-u)(v+u)=v^2-u^2=1,
\]
令 \(s=v-u\)，\(t=v+u\)，则
\[
(s,t)=\left(\sqrt{\frac{3}{2}},\sqrt{\frac{2}{3}}\right)\quad\text{或}\quad
(s,t)=\left(-\sqrt{\frac{3}{2}},-\sqrt{\frac{2}{3}}\right).
\]
由 \(u=\frac{t-s}{2}\)，\(v=\frac{s+t}{2}\)，得到两个圆心
\[
P_1=\left(-\frac{\sqrt{6}}{12},\frac{5\sqrt{6}}{12}\right),\qquad
P_2=\left(\frac{\sqrt{6}}{12},-\frac{5\sqrt{6}}{12}\right).
\]
由 \(r^2=u^2+3\)，两圆的半径平方均为
\[
r^2=\frac{1}{24}+3=\frac{73}{24}.
\]
因此圆 \(P\) 的方程为
\[
\left(x+\frac{\sqrt{6}}{12}\right)^2+\left(y-\frac{5\sqrt{6}}{12}\right)^2=\frac{73}{24}
\]
或
\[
\left(x-\frac{\sqrt{6}}{12}\right)^2+\left(y+\frac{5\sqrt{6}}{12}\right)^2=\frac{73}{24}.
\]
\end{enumerate}
\end{solution}

\begin{problem}
已知函数 \( f(x) = x^{2}\e^{-x} \).
\begin{enumerate}
\item 求 \( f(x) \) 的极小值和极大值；
\item 当曲线 \( y = f(x) \) 的切线 \( l \) 的斜率为负数时，求 \( l \) 在 \( x \) 轴上截距的取值范围.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 因为 \(\e^{-x}>0\)，所以
\[
f'(x)=2x\e^{-x}-x^2\e^{-x}=x(2-x)\e^{-x}.
\]
当 \(x<0\) 时，\(f'(x)<0\)；当 \(0<x<2\) 时，\(f'(x)>0\)；当 \(x>2\) 时，\(f'(x)<0\). 因此 \(f(x)\) 在 \((-\infty,0)\) 上单调递减，在 \((0,2)\) 上单调递增，在 \((2,+\infty)\) 上单调递减.

又因为 \(f(x)=x^2\e^{-x}\geqslant0\)，且 \(f(0)=0\)，所以 \(f(x)\) 的极小值为 \(0\). 在 \(x=2\) 处取得极大值
\[
f(2)=4\e^{-2}=\frac{4}{\e^2}.
\]
\item 设切线 \(l\) 与曲线相切于点 \(\left(t,f(t)\right)\). 切线斜率为负数等价于 \(f'(t)<0\)，由上面的符号分析知 \(t<0\) 或 \(t>2\). 因为此时 \(t\ne0,2\)，切线与 \(x\) 轴的交点横坐标 \(x_l\) 满足
\[
0=f(t)+f'(t)(x_l-t),
\]
所以
\[
x_l=t-\frac{f(t)}{f'(t)}=t-\frac{t}{2-t}=\frac{t(1-t)}{2-t}.
\]
记 \(\phi(t)=\frac{t(1-t)}{2-t}\). 则
\[
\phi'(t)=\frac{t^2-4t+2}{(2-t)^2}.
\]
当 \(t<0\) 时，\(\phi'(t)>0\)，且 \(\phi(t)\) 从 \(-\infty\) 增至 \(0\)，故 \(x_l\in(-\infty,0)\).

当 \(t>2\) 时，\(\phi'(t)=0\) 的点为 \(t=2+\sqrt{2}\)，函数在 \((2,2+\sqrt{2})\) 上递减，在 \((2+\sqrt{2},+\infty)\) 上递增. 又
\[
\lim_{t\to2^+}\phi(t)=+\infty,\qquad
\phi(2+\sqrt{2})=3+2\sqrt{2},\qquad
\lim_{t\to+\infty}\phi(t)=+\infty.
\]
所以 \(x_l\in[3+2\sqrt{2},+\infty)\). 综上，截距的取值范围为
\[
(-\infty,0)\cup[3+2\sqrt{2},+\infty).
\]
\end{enumerate}
\end{solution}

\begin{problem}
如图，\(CD\) 为 \(\triangle ABC\) 外接圆的切线，\(AB\) 的延长线交直线 \(CD\) 于点 \(D\)，\(E\)，\(F\) 分别为弦 \(AB\) 与弦 \(AC\) 上的点，且 \(BC \cdot AE = DC \cdot AF\)，\(B\)，\(E\)，\(F\)，\(C\) 四点共圆.

\begin{enumerate}
\item 证明： \(CA\) 是 \(\triangle ABC\) 外接圆的直径；
\item 若 \(DB = BE = EA\)，求过 \(B\)，\(E\)，\(F\)，\(C\) 四点的圆的面积与 \(\triangle ABC\) 外接圆面积的比值.
\end{enumerate}

\bitmapfigure[width=0.42\linewidth]{img/2013/new_curriculum_paper_2_liberal/q22_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设三角形 \(ABC\) 的内角 \(A\)，\(B\)，\(C\) 分别为 \(\angle CAB\)，\(\angle ABC\)，\(\angle ACB\). 由 \(B\)，\(E\)，\(F\)，\(C\) 四点共圆，点 \(A\) 对该圆的两条割线给出
\[
AB\cdot AE=AC\cdot AF.
\]
结合题设 \(BC\cdot AE=DC\cdot AF\)，可得
\[
\frac{DC}{AC}=\frac{BC}{AB}.
\]

由切线与弦 \(CB\) 所成的角等于弦 \(CB\) 所对的圆周角，得 \(\angle DCB=A\). 又因 \(D\)，\(B\)，\(A\) 三点依次共线，\(\angle DBC=\pi-B\)，所以在三角形 \(DBC\) 中
\[
\angle BDC=B-A.
\]
由正弦定理以及三角形 \(ABC\) 的正弦定理，分别有
\[
\frac{DC}{BC}=\frac{\sin B}{\sin(B-A)},\qquad
\frac{AC}{BC}=\frac{\sin B}{\sin A}.
\]
因此
\[
\frac{DC}{AC}=\frac{\sin A}{\sin(B-A)}.
\]
另一方面，\(\frac{BC}{AB}=\frac{\sin A}{\sin C}\)，结合前面的长度关系可得
\[
\sin(B-A)=\sin C.
\]
由于 \(B-A=\angle BDC\in(0,\pi)\)，且 \(C\in(0,\pi)\)，所以或者 \(B-A=C\)，或者 \(B-A+C=\pi\). 后一种情形与 \(A+B+C=\pi\) 联立会得到 \(A=0\)，不合题意，故 \(B-A=C\). 从而 \(2B=\pi\)，即 \(B=\frac{\pi}{2}\). 因此 \(\angle CBA=\frac{\pi}{2}\)，由圆周角为直角的弦是直径，得 \(CA\) 是三角形 \(ABC\) 外接圆的直径.
\item 由相似变换不改变所求面积比，令 \(DB=BE=EA=1\). 取 \(AB\) 为 \(x\) 轴，设
\[
D=(-1,0),\quad B=(0,0),\quad E=(1,0),\quad A=(2,0).
\]
由第（1）问，\(\angle CBA=\frac{\pi}{2}\)，故设 \(C=(0,h)\)，其中 \(h>0\). 三角形 \(ABC\) 外接圆以 \(AC\) 为直径，圆心为 \(O=(1,\frac h2)\). 因为 \(CD\) 是该圆在 \(C\) 处的切线，所以半径 \(OC\) 垂直于 \(CD\)，即
\[
\overrightarrow{OC}\cdot\overrightarrow{DC}=\left(-1,\frac h2\right)\cdot\left(-1,-h\right)=1-\frac{h^2}{2}=0.
\]
于是 \(h^2=2\).

圆 \(BEFC\) 经过 \(B\)，\(E\)，\(C\)，其圆心在 \(BE\) 和 \(BC\) 的垂直平分线上，因此圆心为
\[
O_1=\left(\frac12,\frac h2\right)=\left(\frac12,\frac{\sqrt2}{2}\right),
\]
半径平方为
\[
R_1^2=O_1B^2=\frac14+\frac{h^2}{4}=\frac34.
\]
三角形 \(ABC\) 外接圆的半径平方为
\[
R^2=OA^2=1+\frac{h^2}{4}=\frac32.
\]
所以两圆面积之比为
\[
\frac{\pi R_1^2}{\pi R^2}=\frac{3/4}{3/2}=\frac12.
\]
\end{enumerate}
\end{solution}

\begin{problem}
已知动点 \(P\)，\(Q\) 都在曲线 \(C: \begin{cases}
x = 2\cos t,\\
y = 2\sin t,
\end{cases}\)（\(t\) 为参数）上，对应参数分别为 \(t = \alpha\) 与 \(t = 2\alpha\)（\(0 < \alpha < 2\pi\)），\(M\) 为 \(PQ\) 的中点.
\begin{enumerate}
\item 求 \(M\) 的轨迹的参数方程；
\item 将 \(M\) 到坐标原点的距离 \(d\) 表示为 \(\alpha\) 的函数，并判断 \(M\) 的轨迹是否过坐标原点.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由参数条件，点 \(P\)，\(Q\) 的坐标分别为
\[
P=(2\cos\alpha,2\sin\alpha).
\]
\(Q=(2\cos2\alpha,2\sin2\alpha)\).
设 \(M=(x,y)\)，因为 \(M\) 是 \(PQ\) 的中点，所以
\[
x=\cos\alpha+\cos2\alpha.
\]
\(y=\sin\alpha+\sin2\alpha\)，且 \(0<\alpha<2\pi\).
这就是 \(M\) 的轨迹的参数方程. 也可利用和差化积写成
\[
x=2\cos\frac{3\alpha}{2}\cos\frac\alpha2.
\]
\(y=2\sin\frac{3\alpha}{2}\cos\frac\alpha2\)，且 \(0<\alpha<2\pi\).
\item 由上式
\[
d^2=x^2+y^2=2+2\cos\alpha=4\cos^2\frac\alpha2.
\]
因为 \(d\geqslant0\)，所以 \(d=2\left|\cos\frac\alpha2\right|\).
当 \(0<\alpha<2\pi\) 时，\(d=0\) 等价于 \(\cos\frac\alpha2=0\)，从而 \(\alpha=\pi\). 此时 \(P=(-2,0)\)，\(Q=(2,0)\)，\(M=(0,0)\)，所以 \(M\) 的轨迹经过坐标原点.
\end{enumerate}
\end{solution}

\begin{problem}
设 \(a\)，\(b\)，\(c\) 均为正数，且 \(a + b + c = 1\)，证明：
\begin{enumerate}
\item \(ab + bc + ca \leqslant \frac{1}{3}\)；
\item \(\frac{a^2}{b} + \frac{b^2}{c} + \frac{c^2}{a} \geqslant 1\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 因为
\[
(a-b)^2+(b-c)^2+(c-a)^2\geqslant0,
\]
所以 \(a^2+b^2+c^2\geqslant ab+bc+ca\). 于是
\[
1=(a+b+c)^2=a^2+b^2+c^2+2ab+2bc+2ca\geqslant3(ab+bc+ca),
\]
从而
\[
ab+bc+ca\leqslant\frac13.
\]
\item 由柯西不等式
\[
\left(\frac{a^2}{b}+\frac{b^2}{c}+\frac{c^2}{a}\right)(b+c+a)\geqslant(a+b+c)^2.
\]
由于 \(a\)，\(b\)，\(c\) 均为正数，\(a+b+c=1\)，且各分母均有意义，故两边除以 \(b+c+a=1\)，得到
\[
\frac{a^2}{b}+\frac{b^2}{c}+\frac{c^2}{a}\geqslant1.
\]
\end{enumerate}
\end{solution}
