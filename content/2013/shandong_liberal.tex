\section{单选题}

\begin{problem}
复数 \(z = \frac{(2 - \i)^2}{\i}\)（\(\i\) 为虚数单位），则 \(|z| =\)（\quad）

\choices
    {\(25\)}
    {\(\sqrt{41}\)}
    {\(5\)}
    {\(\sqrt{5}\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
\((2-\i)^2=4-4\i+\i^2=3-4\i\)，所以 \(z=\dfrac{3-4\i}{\i}=-4-3\i\)，从而 \(|z|=\sqrt{(-4)^2+(-3)^2}=5\). 因此选 C.
\end{solution}

\begin{problem}
已知集合 \(A\)，\(B\) 均为全集 \(U = \{1, 2, 3, 4\}\) 的子集，且 \(\complement_U (A \cup B) = \{4\}\)，\(B = \{1, 2\}\)，则 \(A \cap \complement_U B =\)（\quad）

\choices
    {\(\{3\}\)}
    {\(\{4\}\)}
    {\(\{3, 4\}\)}
    {\(\varnothing\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
由 \(\complement_U(A\cup B)=\{4\}\) 得 \(A\cup B=\{1,2,3\}\). 又 \(B=\{1,2\}\)，故 \(3\in A\)，且 \(4\notin A\). 因为 \(\complement_U B=\{3,4\}\)，所以 \(A\cap\complement_U B=\{3\}\). 因此选 A.
\end{solution}

\begin{problem}
已知函数 \( f(x) \) 为奇函数，且当 \( x > 0 \) 时，\( f(x) = x^2 + \frac{1}{x} \)，则 \( f(-1) =\)（\quad）

\choices
    {\(2\)}
    {\(1\)}
    {\(0\)}
    {\(-2\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
因为 \(f\) 是奇函数，所以 \(f(-1)=-f(1)\). 又 \(f(1)=1^2+\dfrac{1}{1}=2\)，故 \(f(-1)=-2\). 因此选 D.
\end{solution}

\begin{problem}
一个四棱锥的侧棱长都相等，底面是正方形，其正（主）视图如图所示，则该四棱锥侧面积和体积分别是（\quad）

\bitmapfigure[max width=0.32\linewidth]{img/2013/shandong_liberal/q4_fig1.png}

\choices
    {\(4\sqrt{5}\)，\(8\)}
    {\(4\sqrt{5}\)，\(\frac{8}{3}\)}
    {\(4(\sqrt{5} + 1)\)，\(\frac{8}{3}\)}
    {\(8\)，\(8\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
设正方形底面的边长为 \(2\)，四棱锥的高为 \(2\). 设顶点在底面上的射影为 \(H\)，则由勾股定理，\(PA^2=PH^2+HA^2\)，\(PB^2=PH^2+HB^2\)，\(PC^2=PH^2+HC^2\)，\(PD^2=PH^2+HD^2\). 由于四条侧棱相等，得 \(HA=HB=HC=HD\)，所以 \(H\) 是正方形中心. 设底边中点为 \(K\)，则 \(HK=1\)，故每个侧面的高为 \(PK=\sqrt{PH^2+HK^2}=\sqrt{2^2+1^2}=\sqrt{5}\). 所以侧面积为 \(4\times\dfrac{1}{2}\times2\times\sqrt{5}=4\sqrt{5}\)，体积为 \(\dfrac{1}{3}\times2^2\times2=\dfrac{8}{3}\). 因此选 B.
\end{solution}

\begin{problem}
函数 \( f(x) = \sqrt{1 - 2^x} + \frac{1}{\sqrt{x + 3}} \) 的定义域为（\quad）

\choices
    {\((-3,0]\)}
    {\((-3, 1]\)}
    {\((-\infty, -3) \cup (-3, 0]\)}
    {\((-\infty, -3) \cup (-3, 1]\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
根式 \(\sqrt{1-2^x}\) 要求 \(1-2^x\geqslant0\)，即 \(x\leqslant0\). 分母 \(\sqrt{x+3}\) 不能为零，且根式有意义，所以 \(x+3>0\)，即 \(x>-3\). 两个条件取交集得定义域为 \((-3,0]\). 因此选 A.
\end{solution}

执行两次如图所示的程序框图，若第一次输入的 \(a\) 的值为 \(-1.2\)，第二次输入的 \(a\) 的值为 \(1.2\)，则第一次，第二次输出的 \(a\) 的值分别为（\quad）

\begin{problem}
\(\triangle ABC\) 的内角 \(A\)，\(B\)，\(C\) 所对的边分别是 \(a\)，\(b\)，\(c\). 若 \(B = 2A\)，\(a = 1\)，\(b = \sqrt{3}\)，则 \(c =\)（\quad）

\choices
    {\(2 \sqrt{3}\)}
    {\(2\)}
    {\(\sqrt{2}\)}
    {\(1\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
由正弦定理，\(\dfrac{b}{a}=\dfrac{\sin B}{\sin A}=\dfrac{\sin2A}{\sin A}=2\cos A\). 因而 \(\sqrt{3}=2\cos A\)，得 \(A=\dfrac{\pi}{6}\)，从而 \(B=\dfrac{\pi}{3}\)，\(C=\dfrac{\pi}{2}\). 再由正弦定理，\(\dfrac{c}{\sin C}=\dfrac{a}{\sin A}\)，所以 \(c=\dfrac{1}{1/2}=2\). 因此选 B.
\end{solution}

\begin{problem}
给定两个命题 \( p \)，\( q \)，若 \( \neg p \) 是 \( q \) 的必要而不充分条件，则 \( p \) 是 \( \neg q \) 的（\quad）

\choices
    {充分而不必要条件}
    {必要而不充分条件}
    {充要条件}
    {既不充分也不必要条件}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
“\(\neg p\) 是 \(q\) 的必要条件”表示 \(q\Rightarrow\neg p\)，其逆否命题为 \(p\Rightarrow\neg q\)，所以 \(p\) 是 \(\neg q\) 的充分条件. 又“\(\neg p\) 不是 \(q\) 的充分条件”表示 \(\neg p\not\Rightarrow q\)，其逆否关系说明 \(\neg q\not\Rightarrow p\)，所以 \(p\) 不是 \(\neg q\) 的必要条件. 因此选 A.
\end{solution}

\begin{problem}
函数 \( y = x \cos x + \sin x \) 的图象大致为（\quad）

\choices
    {\choicebitmap[width=0.15\paperwidth]{img/2013/shandong_liberal/q9_choice_a.png}}
    {\choicebitmap[width=0.15\paperwidth]{img/2013/shandong_liberal/q9_choice_b.png}}
    {\choicebitmap[width=0.15\paperwidth]{img/2013/shandong_liberal/q9_choice_c.png}}
    {\choicebitmap[width=0.15\paperwidth]{img/2013/shandong_liberal/q9_choice_d.png}}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
令 \(f(x)=x\cos x+\sin x\)，则 \(f(-x)=-f(x)\)，所以图象关于原点对称. 又 \(f(0)=0\)，\(f'(x)=2\cos x-x\sin x\)，故 \(f'(0)=2>0\)，图象经过原点时向右上方穿过. 在 \(0<x<\dfrac{\pi}{2}\) 时，\(f'(x)=0\) 等价于 \(x\tan x=2\). 函数 \(x\tan x\) 在该区间严格递增，并从 \(0\) 增至 \(+\infty\)，故存在唯一的 \(\alpha\in\left(0,\dfrac{\pi}{2}\right)\) 使 \(f'(\alpha)=0\)；于是 \(f\) 在 \((0,\alpha)\) 上递增. 在 \(\left[\dfrac{\pi}{2},\pi\right]\) 上有 \(f'(x)=2\cos x-x\sin x<0\)，故 \(f\) 在 \(\left[\alpha,\pi\right]\) 上递减. 另外 \(f\left(\dfrac{\pi}{2}\right)=1>0\)，而 \(f(\pi)=-\pi<0\)，所以正半轴上只有一个位于 \(\dfrac{\pi}{2}\) 与 \(\pi\) 之间的非零零点. 结合奇对称性，只有图 D 符合. 因此选 D.
\end{solution}

\begin{problem}
将某选手的 \(9\) 个得分去掉 \(1\) 个最高分，去掉 \(1\) 个最低分，\(7\) 个剩余分数的平均分为 \(91\)，现场作的 \(9\) 个分数的茎叶图后来有 \(1\) 个数据模糊，无法辨认，在图中以 \(x\) 表示：

\begin{center}
\begin{tabular}{r|l}
8 & \(7\quad 7\) \\
9 & \(4\)，\(0\)，\(1\)，\(0\)，\(x\)，\(9\)，\(1\)
\end{tabular}
\end{center}

则 7 个剩余分数的方差为（\quad）

\choices
    {\(\frac{116}{9}\)}
    {\(\frac{36}{7}\)}
    {\(36\)}
    {\(\frac{6\sqrt{7}}{7}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
设模糊叶为 \(x\)，则 \(x\in\{0,1,\ldots,9\}\)，模糊数据为 \(90+x\)，已知的另外八个数据之和为
\(87+87+94+90+91+90+99+91=729\). 因为最高分为 \(99\)，最低分为 \(87\)，去掉二者后，剩余七个数据的和为 \(729+(90+x)-99-87=633+x\). 由平均分为 \(91\)，得 \(\dfrac{633+x}{7}=91\)，所以 \(x=4\). 七个剩余分数为 \(87\)，\(94\)，\(90\)，\(91\)，\(90\)，\(94\)，\(91\)，其平均数为 \(91\)，故方差为
\[
\frac{(-4)^2+3^2+(-1)^2+0^2+(-1)^2+3^2+0^2}{7}=\frac{36}{7}
\]
因此选 B.
\end{solution}

\begin{problem}
抛物线 \(C_1: y = \frac{1}{2p} x^2\)（\(p > 0\)）的焦点与双曲线 \(C_2: \frac{x^2}{3} - y^2 = 1\) 的右焦点的连线交 \(C_1\) 于第一象限的点 \(M\). 若 \(C_1\) 在点 \(M\) 处的切线平行于 \(C_2\) 的一条渐近线，则 \(p =\)（\quad）

\choices
    {\(\frac{\sqrt{3}}{16}\)}
    {\(\frac{\sqrt{3}}{8}\)}
    {\(\frac{2\sqrt{3}}{3}\)}
    {\(\frac{4\sqrt{3}}{3}\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
由于 \(C_1\) 可写为 \(x^2=2py\)，其焦点为 \(F\left(0,\dfrac{p}{2}\right)\). 双曲线 \(C_2\) 的半实轴、半虚轴分别为 \(\sqrt{3}\)，\(1\)，故其右焦点为 \(G(2,0)\)，渐近线斜率为 \(\pm\dfrac{1}{\sqrt{3}}\). 设 \(M=(x,y)\)，则 \(x>0\)，且 \(C_1\) 在 \(M\) 处的切线斜率为 \(\dfrac{x}{p}>0\)，所以切线只能平行于斜率为 \(\dfrac{1}{\sqrt{3}}\) 的渐近线，从而 \(\dfrac{x}{p}=\dfrac{1}{\sqrt{3}}\)，即 \(x=\dfrac{p}{\sqrt{3}}\). 直线 \(FG\) 的方程为 \(y=\dfrac{p}{2}-\dfrac{p}{4}x\)，而 \(M\in C_1\)，故
\[
\frac{x^2}{2p}=\frac{p}{2}-\frac{p}{4}x
\]
代入 \(x=\dfrac{p}{\sqrt{3}}\) 得 \(\dfrac{p}{6}=\dfrac{p}{2}-\dfrac{p^2}{4\sqrt{3}}\). 因为 \(p>0\)，所以 \(p=\dfrac{4\sqrt{3}}{3}\). 因此选 D.
\end{solution}

\begin{problem}
设正实数 \(x\)，\(y\)，\(z\) 满足 \(x^2 - 3xy + 4y^2 - z = 0\)，则当 \(\frac{z}{xy}\) 取得最小值时，\(x + 2y - z\) 的最大值为（\quad）

\choices
    {\(0\)}
    {\(\frac{9}{8}\)}
    {\(2\)}
    {\(\frac{9}{4}\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
由已知 \(z=x^2-3xy+4y^2\)，令 \(t=\dfrac{x}{y}>0\)，则
\[
\frac{z}{xy}=\frac{x}{y}-3+4\frac{y}{x}=t+\frac{4}{t}-3\geqslant2\sqrt{4}-3=1
\]
等号当且仅当 \(t=2\)，即 \(x=2y\). 此时 \(z=2y^2\)，所以
\(x+2y-z=4y-2y^2=2-2(y-1)^2\leqslant2\). 当 \(y=1\) 时取等号，因此最大值为 \(2\). 因此选 C.
\end{solution}

\begin{problem}
过点 \((3,1)\) 作圆 \( (x-2)^{2}+(y-2)^{2}=4 \) 的弦，其中最短弦的长为\fillinblank{}.
\end{problem}

\begin{answer}
\(2\sqrt{2}\)
\end{answer}

\begin{solution}
圆心为 \(O(2,2)\)，半径为 \(2\)，且点 \(P(3,1)\) 到圆心的距离为 \(OP=\sqrt{2}<2\). 设过 \(P\) 的弦所在直线到圆心的距离为 \(d\)，则 \(d\leqslant OP\)，弦长为 \(2\sqrt{2^2-d^2}\). 要使弦长最短，应使 \(d\) 最大，即取过 \(P\) 且垂直于 \(OP\) 的直线，此时 \(d=OP=\sqrt{2}\). 所以最短弦长为 \(2\sqrt{4-2}=2\sqrt{2}\).
\end{solution}

\begin{problem}
在平面直角坐标系 \(xOy\) 中，\(M\) 为不等式组 \(\begin{cases}
2x + 3y - 6 \leqslant 0,\\
x + y - 2 \geqslant 0,\\
y \geqslant 0,
\end{cases}\) 所表示的区域上一动点，则 \(|OM|\) 的最小值是\fillinblank{}.
\end{problem}

\begin{answer}
\(\sqrt{2}\)
\end{answer}

\begin{solution}
由 \(x+y\geqslant2\) 及柯西不等式，\(x+y\leqslant\sqrt{2}\sqrt{x^2+y^2}\)，得 \(\sqrt{x^2+y^2}\geqslant\sqrt{2}\). 点 \((1,1)\) 满足 \(2+3-6=-1\leqslant0\)、\(1+1-2=0\) 及 \(1\geqslant0\)，属于该区域，且在此点 \(|OM|=\sqrt{1^2+1^2}=\sqrt{2}\). 因此最小值为 \(\sqrt{2}\).
\end{solution}

\begin{problem}
在平面直角坐标系 \(xOy\) 中，已知 \(\overrightarrow{OA} = (-1, t)\)，\(\overrightarrow{OB} = (2, 2)\). 若 \(\angle ABO = 90^\circ\)，则实数 \(t\) 的值为\fillinblank{}.
\end{problem}

\begin{answer}
5
\end{answer}

\begin{solution}
由向量坐标得 \(\overrightarrow{BA}=\overrightarrow{OA}-\overrightarrow{OB}=(-3,t-2)\)，\(\overrightarrow{BO}=-\overrightarrow{OB}=(-2,-2)\). 因为 \(\angle ABO=90^\circ\)，所以 \(\overrightarrow{BA}\cdot\overrightarrow{BO}=0\)，即
\[
(-3)(-2)+(t-2)(-2)=0
\]
解得 \(10-2t=0\)，所以 \(t=5\).
\end{solution}

\begin{problem}
定义“正对数”： \( \ln^{+} x=\begin{cases}
0,&0<x<1,\\
\ln x,&x\geqslant1
\end{cases} \).
现有四个命题：
\begin{circlelist}
\item 若 \(a > 0\)，\(b > 0\)，则 \( \ln^{+}(a^{b}) = b\ln^{+} a \)；
\item 若 \(a > 0\)，\(b > 0\)，则 \( \ln^{+}(ab) = \ln^{+}a + \ln^{+}b \)；
\item 若 \(a > 0\)，\(b > 0\)，则 \( \ln^{+}\left(\frac{a}{b}\right) \geqslant \ln^{+} a - \ln^{+} b \)；
\item 若 \(a > 0\)，\(b > 0\)，则 \( \ln^{+}(a + b) \leqslant \ln^{+} a + \ln^{+} b + \ln 2 \).
\end{circlelist}

其中真命题有\fillinblank{}.（写出所有真命题的编号）
\end{problem}

\begin{answer}
\(\circled{1}\circled{3}\circled{4}\)
\end{answer}

\begin{solution}
记 \(u=\ln a\)，则 \(\ln^{+}a=\max\{u,0\}\). 对命题（1），当 \(a\geqslant1\) 时有 \(a^b\geqslant1\)，等式由对数运算直接成立；当 \(0<a<1\) 时 \(0<a^b<1\)，等式两边均为 \(0\)，故命题（1）为真.

命题（2）不一定成立，例如取 \(a=2\)，\(b=\dfrac12\)，则 \(\ln^{+}(ab)=0\)，而 \(\ln^{+}a+\ln^{+}b=\ln2>0\)，故命题（2）为假.

对命题（3），分情况验证：若 \(u\leqslant0\)，则 \(\max\{u,0\}-\max\{v,0\}\leqslant0\leqslant\max\{u-v,0\}\)；若 \(u>0\)、\(v\leqslant0\)，则 \(\max\{u-v,0\}=u-v\geqslant u=\max\{u,0\}-\max\{v,0\}\)；若 \(u>0\)、\(v>0\)，则 \(\max\{u-v,0\}\geqslant u-v=\max\{u,0\}-\max\{v,0\}\). 因而对任意实数 \(u\)，\(v\) 都有 \(\max\{u-v,0\}\geqslant\max\{u,0\}-\max\{v,0\}\). 取 \(u=\ln a\)，\(v=\ln b\)，即得 \(\ln^{+}\left(\dfrac{a}{b}\right)\geqslant\ln^{+}a-\ln^{+}b\)，故命题（3）为真.

对命题（4）分情况讨论：若 \(0<a<1\)、\(0<b<1\)，则 \(a+b<2\)，故 \(\ln^{+}(a+b)\leqslant\ln2\)；若恰有一个数不小于 \(1\)，不妨设 \(a\geqslant1\)，则 \(a+b\leqslant2a\)，故 \(\ln^{+}(a+b)\leqslant\ln a+\ln2\)；若 \(a\geqslant1\)、\(b\geqslant1\)，则 \(a+b\leqslant2ab\)，故 \(\ln^{+}(a+b)\leqslant\ln a+\ln b+\ln2\). 因此命题（4）为真. 所以真命题的编号为 \(1\)，\(3\)，\(4\).
\end{solution}

\section{解答题}

\begin{problem}
某小组共有 A，B，C，D，E 五位同学，他们的身高（单位：\(\text{米}\)）及体重指标（单位：\(\text{千克}/\text{米}^{2}\)）如下表所示：

\begin{center}
\begin{tabular}{|c|c|c|c|c|c|}
\hline
 & A & B & C & D & E \\
\hline
身高 & 1.69 & 1.73 & 1.75 & 1.79 & 1.82 \\
\hline
体重指标 & 19.2 & 25.1 & 18.5 & 23.3 & 20.9 \\
\hline
\end{tabular}
\end{center}

\begin{enumerate}
\item 从该小组身高低于 \(1.80\) 的同学中任选 \(2\) 人，求选到的 \(2\) 人身高都在 \(1.78\) 以下的概率；
\item 从该小组同学中任选 \(2\) 人，求选到的 \(2\) 人的身高都在 \(1.70\) 以上且体重指标都在 \([18.5, 23.9)\) 中的概率.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 身高低于 \(1.80\) 的同学为 A，B，C，D，共有 \(\mathrm C_4^2=6\) 种等可能的选法. 其中身高都在 \(1.78\) 以下的同学为 A，B，C，共有 \(\mathrm C_3^2=3\) 种选法. 所以所求概率为 \(\dfrac{\mathrm C_3^2}{\mathrm C_4^2}=\dfrac{1}{2}\).
\item 身高不低于 \(1.70\) 的同学为 B，C，D，E；其中体重指标属于 \([18.5,23.9)\) 的为 C，D，E，因为 \(18.5\) 取到而 \(23.9\) 不取到. 因此满足两个条件的同学共有 C，D，E 三人. 从五人中任选两人共有 \(\mathrm C_5^2=10\) 种选法，有利选法为 \(\mathrm C_3^2=3\) 种，故所求概率为 \(\dfrac{\mathrm C_3^2}{\mathrm C_5^2}=\dfrac{3}{10}\).
\end{enumerate}
\end{solution}

\begin{problem}
设函数 \(f(x) = \frac{\sqrt{3}}{2} - \sqrt{3}\sin^2\omega x - \sin \omega x\cos \omega x \)（\(\omega > 0\)），且 \( y = f(x) \) 图象的一个对称中心到最近的对称轴的距离为 \( \frac{\pi}{4} \).

\begin{enumerate}
\item 求 \(\omega\) 的值；
\item 求 \( f(x) \) 在区间 \( \left[\pi, \frac{3\pi}{2}\right] \) 上的最大值和最小值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 令 \(u=\omega x\)，利用 \(\sin^2u=\dfrac{1-\cos2u}{2}\) 和 \(\sin u\cos u=\dfrac{\sin2u}{2}\)，得
\[
f(x)=\frac{\sqrt{3}}{2}\cos(2\omega x)-\frac{1}{2}\sin(2\omega x)=\cos\left(2\omega x+\frac{\pi}{6}\right)
\]
余弦函数图象的相邻对称轴与对称中心在相位上相差 \(\dfrac{\pi}{2}\)，因此在 \(x\) 轴方向上的最近距离为 \(\dfrac{\pi}{4\omega}\). 由题意 \(\dfrac{\pi}{4\omega}=\dfrac{\pi}{4}\)，故 \(\omega=1\).
\item 此时 \(f(x)=\cos\left(2x+\dfrac{\pi}{6}\right)\). 在 \(x\in\left[\pi,\dfrac{3\pi}{2}\right]\) 时，相位 \(2x+\dfrac{\pi}{6}\in\left[\dfrac{13\pi}{6},\dfrac{19\pi}{6}\right]\). 导数为 \(f'(x)=-2\sin\left(2x+\dfrac{\pi}{6}\right)\)，区间内唯一的驻点满足 \(2x+\dfrac{\pi}{6}=3\pi\)，即 \(x=\dfrac{17\pi}{12}\). 比较端点和驻点的函数值：
\(f(\pi)=\frac{\sqrt{3}}{2}\)，\(f\left(\frac{3\pi}{2}\right)=-\frac{\sqrt{3}}{2}\)，\(f\left(\frac{17\pi}{12}\right)=-1\).
所以最大值为 \(\dfrac{\sqrt{3}}{2}\)，最小值为 \(-1\).
\end{enumerate}
\end{solution}

\begin{problem}
如图，四棱锥 \(P - ABCD\) 中，\(AB \perp AC\)，\(AB \perp PA\)，\(AB \parallel CD\)，\(AB = 2CD\)，\(E\)，\(F\)，\(G\)，\(M\)，\(N\) 分别为 \(PB\)，\(AB\)，\(BC\)，\(PD\)，\(PC\) 的中点.

\begin{enumerate}
\item 求证：\(CE \parallel\) 平面 \(PAD\)；
\item 求证：平面 \(EFG \perp\) 平面 \(EMN\).
\end{enumerate}

\FigureLayoutDeclare{img/2013/shandong_liberal/q19_fig1.png}{11.0cm}{24bp 141bp 172bp 28bp}
\bitmapfigure[max width=0.52\linewidth]{img/2013/shandong_liberal/q19_fig1.png}
\end{problem}

\begin{solution}
建立直角坐标系，使 \(A=(0,0,0)\)，\(B=(2,0,0)\)，\(C=(0,c,0)\)，\(D=(-1,c,0)\)，其中 \(c\ne0\). 由 \(AB\perp PA\)，可设 \(P=(0,u,v)\)，其中 \(v\ne0\). 于是
\(E=\left(1,\frac{u}{2},\frac{v}{2}\right)\)，\(F=(1,0,0)\)，\(G=\left(1,\frac{c}{2},0\right)\)，\(M=\left(-\frac12,\frac{u+c}{2},\frac{v}{2}\right)\)，\(N=\left(0,\frac{u+c}{2},\frac{v}{2}\right)\).
\begin{enumerate}
\item \(\overrightarrow{CE}=\left(1,\dfrac{u}{2}-c,\dfrac{v}{2}\right)\)，而
\[
\frac12\overrightarrow{AP}-\overrightarrow{AD}=\frac12(0,u,v)-(-1,c,0)=\left(1,\frac{u}{2}-c,\frac{v}{2}\right)=\overrightarrow{CE}
\]
因此 \(\overrightarrow{CE}\) 是平面 \(PAD\) 内两相交向量 \(\overrightarrow{AP}\)，\(\overrightarrow{AD}\) 的线性组合. 又 \(C\) 不在平面 \(PAD\) 内：若 \(C=\alpha\overrightarrow{AP}+\beta\overrightarrow{AD}\)，由 \(x\) 坐标得 \(\beta=0\)，再由 \(z\) 坐标得 \(\alpha=0\)，这与 \(C=(0,c,0)\)、\(c\ne0\) 矛盾. 故 \(CE\parallel\) 平面 \(PAD\).
\item \(\overrightarrow{FE}=\dfrac12\overrightarrow{AP}\)，\(\overrightarrow{FG}=\dfrac12\overrightarrow{AC}\)，所以平面 \(EFG\parallel\) 平面 \(PAC\). 又因为 \(AB\perp PA\)，\(AB\perp AC\)，且 \(PA\) 与 \(AC\) 是平面 \(PAC\) 内的两条相交直线，故 \(AB\perp\) 平面 \(PAC\)，从而 \(AB\perp\) 平面 \(EFG\). 另一方面，\(\overrightarrow{MN}=\left(\dfrac12,0,0\right)\parallel AB\)，且 \(MN\subset\) 平面 \(EMN\)，故 \(MN\perp\) 平面 \(EFG\). 因此平面 \(EMN\perp\) 平面 \(EFG\).
\end{enumerate}
\end{solution}

\begin{problem}
设等差数列 \(\{a_{n}\}\) 的前 \(n\) 项和为 \(S_{n}\)，且 \(S_{4} = 4S_{2}\)，\(a_{2n} = 2a_{n} + 1\).

\begin{enumerate}
\item 求数列 \( \{a_{n}\} \) 的通项公式；
\item 若数列 \( \{b_{n}\} \) 满足 \(\frac{b_{1}}{a_{1}} + \frac{b_{2}}{a_{2}} + \cdots + \frac{b_{n}}{a_{n}} = 1 - \frac{1}{2^{n}}\)，\(n \in \N^{*}\)，求 \( \{b_{n}\} \) 的前 \(n\) 项和 \( T_{n} \).
\end{enumerate}
\end{problem}

\begin{solution}
设等差数列首项为 \(a_1\)，公差为 \(d\). 由 \(S_4=4a_1+6d\)，\(S_2=2a_1+d\)，条件 \(S_4=4S_2\) 得 \(4a_1+6d=8a_1+4d\)，即 \(d=2a_1\). 又
\[
a_{2n}=a_1+(2n-1)d=2\bigl(a_1+(n-1)d\bigr)+1=2a_n+1
\]
故 \(d-a_1=1\). 联立得 \(a_1=1\)，\(d=2\)，所以 \(a_n=1+2(n-1)=2n-1\).

当 \(n=1\) 时，由已知等式直接得 \(\dfrac{b_1}{a_1}=\dfrac12\). 当 \(n\geqslant2\) 时，将已知关于 \(b_n\) 的等式分别写成 \(n\) 项和与 \(n-1\) 项和并相减，得
\[
\frac{b_n}{a_n}=\left(1-\frac{1}{2^n}\right)-\left(1-\frac{1}{2^{n-1}}\right)=\frac{1}{2^n}
\]
因此 \(b_n=\dfrac{a_n}{2^n}=\dfrac{2n-1}{2^n}\)，从而
\[
T_n=\sum_{k=1}^{n}\frac{2k-1}{2^k}=2\sum_{k=1}^{n}\frac{k}{2^k}-\sum_{k=1}^{n}\frac{1}{2^k}
\]
其中 \(\sum_{k=1}^{n}\dfrac{1}{2^k}=1-\dfrac{1}{2^n}\)，且由错位相减法
\[
\sum_{k=1}^{n}\frac{k}{2^k}=2-\frac{n+2}{2^n}
\]
所以 \(T_n=3-\dfrac{2n+3}{2^n}\).
\end{solution}

\begin{problem}
已知函数 \( f(x) = ax^2 + bx - \ln x  \)（\(a\)，\(b \in \R\)）.

\begin{enumerate}
\item 设 \(a \geqslant 0\)，求 \(f(x)\) 的单调区间；
\item 设 \(a > 0\)，且对任意 \(x > 0\)，\(f(x) \geqslant f(1)\)，试比较 \(\ln a\) 与 \(-2b\) 的大小.
\end{enumerate}
\end{problem}

\begin{solution}
函数定义域为 \((0,+\infty)\)，且
\[
f'(x)=2ax+b-\frac{1}{x}=\frac{2ax^2+bx-1}{x}
\]
\begin{enumerate}
\item 当 \(a=0\) 时，\(f'(x)=b-\dfrac1x\). 若 \(b\leqslant0\)，则 \(f'(x)<0\)，所以 \(f(x)\) 在 \((0,+\infty)\) 上单调递减；若 \(b>0\)，则 \(f'(x)<0\) 当且仅当 \(0<x<\dfrac1b\)，而 \(f'(x)>0\) 当且仅当 \(x>\dfrac1b\)，所以 \(f(x)\) 在 \(\left(0,\dfrac1b\right)\) 上单调递减，在 \(\left(\dfrac1b,+\infty\right)\) 上单调递增.

当 \(a>0\) 时，方程 \(2ax^2+bx-1=0\) 的两个根乘积为 \(-\dfrac{1}{2a}<0\)，故只有一个正根
\[
x_0=\frac{-b+\sqrt{b^2+8a}}{4a}
\]
由于二次函数开口向上，且分母 \(x>0\)，所以 \(f'(x)<0\) 在 \((0,x_0)\) 上成立，\(f'(x)>0\) 在 \((x_0,+\infty)\) 上成立. 因此 \(f(x)\) 在 \(\left(0,x_0\right)\) 上单调递减，在 \(\left(x_0,+\infty\right)\) 上单调递增.
\item 因为对任意 \(x>0\) 有 \(f(x)\geqslant f(1)\)，所以 \(x=1\) 是内点处的极小值点，必有 \(f'(1)=0\). 因而 \(2a+b-1=0\)，即 \(b=1-2a\). 于是 \(-2b=4a-2\).

令 \(g(a)=4a-2-\ln a\)（\(a>0\)），则 \(g'(a)=4-\dfrac1a\)，故 \(g\) 在 \(a=\dfrac14\) 处取得最小值，且
\[
g\left(\frac14\right)=\ln4-1>0
\]
所以 \(4a-2-\ln a>0\)，即 \(\ln a<-2b\).
\end{enumerate}
\end{solution}

\begin{problem}
在平面直角坐标系 \(xOy\) 中，已知椭圆 \(C\) 的中心在原点 \(O\)，焦点在 \(x\) 轴上，短轴长为 2，离心率为 \(\frac{\sqrt{2}}{2}\).

\begin{enumerate}
\item 求椭圆 \(C\) 的方程；
\item \(A\)，\(B\) 为椭圆 \(C\) 上满足 \(\triangle AOB\) 的面积为 \(\frac{\sqrt{6}}{4}\) 的任意两点，\(E\) 为线段 \(AB\) 的中点，射线 \(OE\) 交椭圆 \(C\) 于点 \(P\). 设 \(\overrightarrow{OP} = t\overrightarrow{OE}\)，求实数 \(t\) 的值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设椭圆方程为 \(\dfrac{x^2}{a^2}+\dfrac{y^2}{b^2}=1\)（\(a>b>0\)），焦距的一半为 \(c\). 由短轴长为 \(2\) 得 \(b=1\)，由离心率得 \(\dfrac{c}{a}=\dfrac{\sqrt2}{2}\). 又 \(a^2=b^2+c^2\)，所以 \(a^2=1+\dfrac{a^2}{2}\)，即 \(a^2=2\). 因此椭圆方程为 \(\dfrac{x^2}{2}+y^2=1\).
\item 作线性变换 \(X=\dfrac{x}{\sqrt2}\)，\(Y=y\)，则椭圆变为单位圆 \(X^2+Y^2=1\). 记变换后的点为 \(A'\)，\(B'\)，中点为 \(E'\). 由面积条件 \(\dfrac12\left|\det(A,B)\right|=\dfrac{\sqrt6}{4}\)，得 \(\left|\det(A,B)\right|=\dfrac{\sqrt6}{2}\). 线性变换的行列式为 \(\dfrac{1}{\sqrt2}\)，所以
\[
\left|\det(A',B')\right|=\frac{1}{\sqrt2}\left|\det(A,B)\right|=\frac{1}{\sqrt2}\times\frac{\sqrt6}{2}=\frac{\sqrt3}{2}
\]
令 \(D'=\dfrac{A'-B'}{2}\)，则 \(A'=E'+D'\)，\(B'=E'-D'\). 因为 \(|A'|=|B'|=1\)，有 \(E'\cdot D'=0\) 及 \(|E'|^2+|D'|^2=1\). 又
\[
\left|\det(A',B')\right|=2\left|\det(E',D')\right|=2|E'||D'|
\]
所以若令 \(r=|E'|^2\)，则 \(4r(1-r)=\dfrac34\)，即
\(r(1-r)=\frac{3}{16}\)，\(r=\frac14\ \text{或}\ r=\frac34\).
由 \(\overrightarrow{OP}=t\overrightarrow{OE}\)，变换后仍有 \(P'=tE'\). 因为 \(P'\) 在单位圆上，\(t^2|E'|^2=1\). 又 \(P\) 在射线 \(OE\) 上，故 \(t>0\)，于是
\[
t=\frac{1}{\sqrt r}=2\ \text{或}\ \frac{2\sqrt3}{3}
\]
两种情形分别对应圆上两点所夹的圆心角为 \(\dfrac{2\pi}{3}\) 或 \(\dfrac{\pi}{3}\)，均能满足题设面积条件，故两值都应保留.
\end{enumerate}
\end{solution}
