\chapter{1989年全国卷（文）}

\section{单选题}

\begin{problem}
如果 \(I = \{a, b, c, d, e\}\)，\(M = \{a, c, d\}\)，\(N = \{b, d, e\}\)，其中 \(I\) 是全集，那么 \(\overline{M} \cap \overline{N}\) 等于（\(\quad\)）

\choices
    {\(\varnothing\)}
    {\(\{d\}\)}
    {\(\{a, c\}\)}
    {\(\{b, e\}\)}
\end{problem}

\begin{problem}
与函数 \(y = x\) 有相同图象的一个函数是（\(\quad\)）

\choices
    {\(y = \sqrt{x^2}\)}
    {\(y = \frac{x^2}{x}\)}
    {\(y = a^{\log_a x}\)（\(a > 0\)，\(a \neq 1\)）}
    {\(y = \log_a a^x\)（\(a > 0\)，\(a \neq 1\)）}
\end{problem}

\begin{problem}
如果圆锥的底面半径为 \(\sqrt{2}\)，高为 \(2\)，那么它的侧面积是（\(\quad\)）

\choices
    {\(4\sqrt{3}\pi\)}
    {\(2\sqrt{2}\pi\)}
    {\(2\sqrt{3}\pi\)}
    {\(4\sqrt{2}\pi\)}
\end{problem}

\begin{problem}
已知 \(\{a_n\}\) 是等比数列，如果 \(a_1 + a_2 = 12\)，\(a_2 + a_3 = -6\)，且 \(S_n = a_1 + a_2 + \cdots + a_n\)，那么 \(\lim\limits_{n \to \infty} S_n\) 的值等于（\(\quad\)）

\choices
    {\(8\)}
    {\(16\)}
    {\(32\)}
    {\(48\)}
\end{problem}

\begin{problem}
如果 \((1 - 2x)^7 = a_0 + a_1x + a_2x^2 + \cdots + a_7x^7\)，那么 \(a_1 + a_2 + \cdots + a_7\) 的值等于（\(\quad\)）

\choices
    {\(-2\)}
    {\(-1\)}
    {\(0\)}
    {\(2\)}
\end{problem}

\begin{problem}
如果 \(\left|\cos \theta\right| = \frac{1}{5}\)，\(\frac{\pi}{2} < \theta < 3\pi\)，那么 \(\sin \frac{\theta}{2}\) 的值等于（\(\quad\)）

\choices
    {\(-\frac{\sqrt{10}}{5}\)}
    {\(\frac{\sqrt{10}}{5}\)}
    {\(-\frac{\sqrt{15}}{5}\)}
    {\(\frac{\sqrt{15}}{5}\)}
\end{problem}

\begin{problem}
直线 \(2x + 3y - 6 = 0\) 关于点 \((1,-1)\) 对称的直线是（\(\quad\)）

\choices
    {\(3x - 2y + 2 = 0\)}
    {\(2x + 3y + 7 = 0\)}
    {\(3x - 2y - 12 = 0\)}
    {\(2x + 3y + 8 = 0\)}
\end{problem}

\begin{problem}
已知球的两个平行截面的面积分别为 \(5\pi\) 和 \(8\pi\)，它们位于球心的同一侧，且相距为 \(1\)，那么这个球的半径是（\(\quad\)）

\choices
    {\(4\)}
    {\(3\)}
    {\(2\)}
    {\(5\)}
\end{problem}

\begin{problem}
由数字 \(1\)，\(2\)，\(3\)，\(4\)，\(5\) 组成没有重复数字的五位数，其中偶数共有（\(\quad\)）

\choices
    {\(60\) 个}
    {\(48\) 个}
    {\(36\) 个}
    {\(24\) 个}
\end{problem}

\begin{problem}
如果双曲线 \(\frac{x^2}{64} - \frac{y^2}{36} = 1\) 上一点 \(P\) 到它的右焦点的距离是 \(8\)，那么点 \(P\) 到它的右准线的距离是（\(\quad\)）

\choices
    {\(10\)}
    {\(\frac{32\sqrt{7}}{7}\)}
    {\(2\sqrt{7}\)}
    {\(\frac{32}{5}\)}
\end{problem}

\begin{problem}
如果 \(|x| \leqslant \frac{\pi}{4}\)，那么函数 \(f(x) = \cos^2 x + \sin x\) 的最小值是（\(\quad\)）

\choices
    {\(\frac{\sqrt{2} - 1}{2}\)}
    {\(-\frac{1 + \sqrt{2}}{2}\)}
    {\(-1\)}
    {\(\frac{1 - \sqrt{2}}{2}\)}
\end{problem}

\begin{problem}
已知 \(f(x) = 8 + 2x - x^2\)，如果 \(g(x) = f(2 - x^2)\)，那么 \(g(x)\)（\(\quad\)）

\choices
    {在区间 \((-1,0)\) 上是减函数}
    {在区间 \((0,1)\) 上是减函数}
    {在区间 \((-2,0)\) 上是增函数}
    {在区间 \((0,2)\) 上是增函数}
\end{problem}

\section{解答题}

\begin{problem}
设复数 \(z = \left(1 - \sqrt{3}\i\right)^5\)，求 \(z\) 的模和辐角的主值.
\end{problem}

\begin{problem}
证明：\(\tan \frac{3x}{2} - \tan \frac{x}{2} = \frac{2\sin x}{\cos x + \cos 2x}\).
\end{problem}

\begin{problem}
如图，在平行六面体 \(ABCD - A_1B_1C_1D_1\) 中，已知 \(AB = 5\)，\(AD = 4\)，\(AA_1 = 3\)，\(AB \perp AD\)，\(\angle A_1AB = \angle A_1AD = \frac{\pi}{3}\).

\begin{enumerate}
\item 求证：顶点 \(A_1\) 在底面 \(ABCD\) 的射影 \(O\) 在 \(\angle BAD\) 的平分线上；
\item 求这个平行六面体的体积.
\end{enumerate}

\bitmapfigure[max width=0.40\linewidth]{img/1989/national_liberal/q21_fig1.png}
\end{problem}

\begin{problem}
用数学归纳法证明：\((1 \cdot 2^2 - 2 \cdot 3^2) + (3 \cdot 4^2 - 4 \cdot 5^2) + \cdots + [(2n - 1)(2n)^2 - 2n(2n + 1)^2] = -n(n + 1)(4n + 3)\).
\end{problem}

\begin{problem}
已知 \(a > 0\)，\(a \neq 1\)，试求使方程 \(\log_a(x - ak) = \log_{a^2}(x^2 - a^2)\) 有解的 \(k\) 的取值范围.
\end{problem}

\begin{problem}
给定椭圆方程 \(\frac{x^2}{b^2} + \frac{y^2}{a^2} = 1\)（\(a > b > 0\)），求与这个椭圆有公共焦点的双曲线，使得以它们的交点为顶点的四边形面积最大，并求相应的四边形的顶点坐标.
\end{problem}
