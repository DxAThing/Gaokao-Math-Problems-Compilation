\section{解答题}

\begin{problem}

设 \(f(x) = a^{x - 1}\)，其中常数 \(a > 0\).

\begin{enumerate}
\item 如果 \(\{x_n\}\) 是等差数列，求证：\(\{f(x_n)\}\) 是等比数列；
\item 如果 \(a = 4\)，解方程 \(4f(x) + f(-x) - 1 = 0\).
\end{enumerate}

\end{problem}

\begin{solution}
（1）设等差数列 \(\{x_n\}\) 的公差为 \(d\). 因为 \(a > 0\)，所以
\[
f(x_n) = a^{x_n - 1} \neq 0.
\]
当 \(n \geqslant 2\) 时，
\[
\frac{f(x_n)}{f(x_{n-1})} = \frac{a^{x_n - 1}}{a^{x_{n-1} - 1}} = a^d.
\]
因为 \(a^d\) 是不等于 0 的常数，所以 \(\{f(x_n)\}\) 是等比数列. 

（2）当 \(a = 4\) 时，方程化为
\[
4^x + 4^{-x - 1} - 1 = 0.
\]
令 \(4^x = y\)，则原方程化为
\[
y + \frac{1}{4y} - 1 = 0.
\]
解得 \(y = \frac{1}{2}\)，所以 \(x = -\frac{1}{2}\).
\end{solution}

\begin{problem}

如图，在平面 \(\beta\) 内有 \(\triangle ABO\)，在平面 \(\beta\) 外有点 \(S\)，斜线 \(SA \perp AO\)，\(SB \perp BO\)，且斜线段 \(SA\) 和 \(SB\) 在 \(\beta\) 内的射影是等长的线段.

\bitmapfigure[max width=0.42\linewidth,max totalheight=4.2cm]{img/1989/guangdong_liberal/q2_fig1.png}

\begin{enumerate}
\item 求证：\(AO = BO\)；
\item 又设点 \(S\) 与平面 \(\beta\) 的距离为 \(3\text{厘米}\)，且 \(AO \perp BO\)，\(AB = 4\text{厘米}\)，求线段 \(SO\) 的长.
\end{enumerate}

\end{problem}

\begin{solution}
（1）如图，作 \(SD \perp \beta\) 于 \(D\)，连结 \(AD\)、\(BD\). 由已知 \(AD = BD\)，所以 \(SA = SB\). 因此 \(\mathrm{Rt}\triangle SAO \cong \mathrm{Rt}\triangle SBO\)，故 \(AO = BO\).

\bitmapfigure[max width=0.42\linewidth,max totalheight=4.2cm]{img/1989/guangdong_liberal/q2_solution_fig1.png}

（2）由三垂线定理的逆定理，可知 \(\angle DAO = \angle DBO = 90^\circ\). 又 \(\angle AOB = 90^\circ\)，且 \(AO = BO\)，所以四边形 \(AOBD\) 是正方形，因此 \(OD = AB = 4\mathrm{cm}\). 又 \(SD = 3\mathrm{cm}\)，所以
\[
SO = \sqrt{3^2 + 4^2} = 5\mathrm{cm}.
\]
\end{solution}

\begin{problem}

已知复数 \(z = 1 - 2\i\)，其中 \(\i\) 是虚数单位.

\begin{enumerate}
\item 求 \(\frac{2}{z + \i} - \overline{(z + 1)}\) 的值，并把该值表示成三角形式；
\item 求适合不等式 \(\log_{0.5}\left|\frac{az - \i}{\sqrt{a + 1}}\right| \leqslant \frac{1}{2}\) 的实数 \(a\) 的取值范围.
\end{enumerate}

\end{problem}

\begin{solution}
（1）因为 \(z = 1 - 2\i\)，所以
\[
\frac{2}{z + \i} - \overline{(z + 1)}
= \frac{2}{1 - \i} - \overline{(2 - 2\i)}
= 1 + \i - (2 + 2\i)
= -1 - \i.
\]
将其表示成三角形式，得
\[
-1 - \i = \sqrt{2}\left(\cos \frac{5\pi}{4} + \i \sin \frac{5\pi}{4}\right).
\]

（2）因为 \(z = 1 - 2\i\)，所以
\[
|az - \i| = \sqrt{5a^2 + 4a + 1}.
\]
由已知不等式可得
\[
\left\{
\begin{aligned}
\frac{\sqrt{5a^2 + 4a + 1}}{\sqrt{a + 1}} &\geqslant \sqrt{\frac{1}{2}},\\
a + 1 &> 0,\\
\frac{\sqrt{5a^2 + 4a + 1}}{\sqrt{a + 1}} &> 0.
\end{aligned}
\right.
\]
\end{solution}
