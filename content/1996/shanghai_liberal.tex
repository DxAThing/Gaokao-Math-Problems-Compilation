\begin{problem}
已知 \(\sin\left(\frac{\pi}{4}+\alpha\right)\sin\left(\frac{\pi}{4}-\alpha\right)=\frac{1}{6}\)，\(\alpha\in\left(\frac{\pi}{2},\pi\right)\)，求 \(\sin 4\alpha\) 的值.

\end{problem}

\begin{solution}
由 \(\sin\left(\frac{\pi}{4}+\alpha\right)\sin\left(\frac{\pi}{4}-\alpha\right)
=\sin\left(\frac{\pi}{4}+\alpha\right)\cos\left[\frac{\pi}{2}-\left(\frac{\pi}{4}-\alpha\right)\right]\)
得
\[
\sin\left(\frac{\pi}{4}+\alpha\right)\cos\left(\frac{\pi}{4}+\alpha\right)
=\frac{1}{2}\sin\left(\frac{\pi}{2}+2\alpha\right)
=\frac{1}{2}\cos 2\alpha=\frac{1}{6}
\]
故 \(\cos 2\alpha=\frac{1}{3}\). 又 \(2\alpha\in(\pi,2\pi)\)，得 \(\sin 2\alpha=-\frac{2\sqrt{2}}{3}\). 因此
由 \(\sin 4\alpha=2\sin 2\alpha\cos 2\alpha=2\cdot\left(-\frac{2\sqrt{2}}{3}\right)\cdot\frac{1}{3}=-\frac{4\sqrt{2}}{9}\).
\end{solution}

\begin{problem}
设 \(z=a+b\i\)（\(a\)，\(b\in\R\)，\(b\neq0\)，\(\i\) 为虚数单位），\(w=z+\frac{1}{z}\) 是实数，且 \(-1<w<2\).

\begin{enumerate}
\item 求 \(\left|z\right|\) 的值及 \(z\) 的实部的取值范围；
\item 设 \(u=\frac{1-z}{1+z}\). 求证：\(u\) 为纯虚数；
\item 求 \(w-u^2\) 的最小值.
\end{enumerate}

\end{problem}

\begin{solution}
（1）因为
\(w=a+b\i+\frac{1}{a+b\i}
=\left(a+\frac{a}{a^2+b^2}\right)+\left(b-\frac{b}{a^2+b^2}\right)\i\)，
而 \(w\) 是实数，且 \(b\neq0\)，所以 \(a^2+b^2=1\)，即 \(|z|=1\). 又 \(w=2a\)，由 \(-1<w<2\) 得 \(a\in\left(-\frac{1}{2},1\right)\). 故 \(z\) 的实部的取值范围是 \(\left(-\frac{1}{2},1\right)\).

（2）因为
\[
u=\frac{1-z}{1+z}
=\frac{1-a-b\i}{1+a+b\i}
=\frac{(1-a-b\i)(1+a-b\i)}{(1+a+b\i)(1+a-b\i)}
=\frac{1-a^2-b^2-2b\i}{(1+a)^2+b^2}
=-\frac{b}{a+1}\i
\]
所以 \(u\) 为纯虚数.

（3）由 \(a^2+b^2=1\)，得
\[
w-u^2=2a+\frac{b^2}{(a+1)^2}=2a+\frac{1-a^2}{(a+1)^2}
=2a-\frac{a-1}{a+1}=2\left[\left(a+1\right)+\frac{1}{a+1}\right]-3
\]
因为 \(a\in\left(-\frac{1}{2},1\right)\)，所以 \(a+1>0\). 从而 \(w-u^2\ge 2\times2-3=1\).
当 \(a+1=\frac{1}{a+1}\)，即 \(a=0\) 时取等号. 因此 \(w-u^2\) 的最小值是 \(1\).
\end{solution}

\begin{problem}
已知 \(A(-1,2)\) 为抛物线 \(C:y=2x^2\) 上的点，直线 \(l_1\) 过点 \(A\)，且与抛物线 \(C\) 相切. 直线 \(l_2:x=a\)（\(a\neq-1\)）交抛物线 \(C\) 于点 \(B\)，交直线 \(l_1\) 于点 \(D\).

\begin{enumerate}
\item 求直线 \(l_1\) 的方程；
\item 设 \(\triangle ABD\) 的面积为 \(S_1\)，求 \(|BD|\) 及 \(S_1\) 的值；
\item 设由抛物线 \(C\)，直线 \(l_1\)，\(l_2\) 所围成的图形的面积为 \(S_2\)，求证：\(S_1:S_2\) 的值为与 \(a\) 无关的常数.
\end{enumerate}

\end{problem}

\begin{solution}
（1）由 \(y=2x^2\)，得 \(y^\prime=4x\). 当 \(x=-1\) 时，\(y^\prime=-4\). 因此 \(l_1\) 的方程为 \(y-2=-4(x+1)\)，即 \(4x+y+2=0\).

（2）由 \(y=2x^2\) 及 \(x=a\)，解得点 \(B\) 的坐标为 \((a,2a^2)\). 由 \(4x+y+2=0\) 及 \(x=a\)，解得点 \(D\) 的坐标为 \((a,-4a-2)\). 又可求得点 \(A\) 到直线 \(BD\) 的距离为 \(\left|a+1\right|\)，且 \(\left|BD\right|=2a^2+4a+2=2\left(a+1\right)^2\). 因此 \(S_1=\left|a+1\right|^3\).

（3）当 \(a>-1\) 时，\(S_2=\int_{-1}^{a}\left(2x^2+4x+2\right)\,\mathrm{d}x=\frac{2}{3}\left(a+1\right)^3\). 当 \(a<-1\) 时，\(S_2=\int_{a}^{-1}\left(2x^2+4x+2\right)\,\mathrm{d}x=-\frac{2}{3}\left(a+1\right)^3\). 所以 \(S_1:S_2=\frac{3}{2}\). 即 \(S_1:S_2\) 的值为与 \(a\) 无关的常数.
\end{solution}

\begin{problem}
设 \(A_n\) 为数列 \(\{a_n\}\) 前 \(n\) 项的和，\(A_n=\frac{3}{2}(a_n-1)\)（\(n\in\N\)）. 数列 \(\{b_n\}\) 的通项公式为 \(b_n=4n+3\)（\(n\in\N\)）.

\begin{enumerate}
\item 求数列 \(\{a_n\}\) 的通项公式；
\item 若 \(d\in\{a_1,a_2,a_3,\cdots,a_n,\cdots\}\cap\{b_1,b_2,b_3,\cdots,b_n,\cdots\}\)，则称 \(d\) 为数列 \(\{a_n\}\) 与 \(\{b_n\}\) 的公共项. 将数列 \(\{a_n\}\) 与 \(\{b_n\}\) 的公共项，按它们在原数列中的先后顺序排成一个新的数列 \(\{d_n\}\)，证明数列 \(\{d_n\}\) 的通项公式为 \(d_n=3^{2n+1}\)（\(n\in\N\)）.
\end{enumerate}

\end{problem}

\begin{solution}
（1）由已知 \(A_n=\frac{3}{2}(a_n-1)\)（\(n\in\N\)）. 当 \(n=1\) 时，\(a_1=\frac{3}{2}(a_1-1)\)，解得 \(a_1=3\).

当 \(n\ge2\) 时，
\[
a_n=A_n-A_{n-1}=\frac{3}{2}(a_n-1)-\frac{3}{2}(a_{n-1}-1)=\frac{3}{2}(a_n-a_{n-1})
\]
由此解得 \(a_n=3a_{n-1}\). 即 \(\frac{a_n}{a_{n-1}}=3\)（\(n\ge2\)），所以数列 \(\{a_n\}\) 是首项为 \(3\)，公比为 \(3\) 的等比数列，故 \(a_n=3^n\)（\(n\in\N\)）.

（2）先证明形如 \(3^{2n+1}\)（\(n\in\N\)）的数都是数列 \(\{a_n\}\) 与 \(\{b_n\}\) 的公共项. 因为 \(a_{2n+1}=3^{2n+1}\)，所以 \(3^{2n+1}\in\{a_1,a_2,\cdots\}\).

又
\[
3^{2n+1}=3\cdot 9^n=3(8+1)^n
=3\left(8^n+\mathrm C_n^1 8^{n-1}+\cdots+\mathrm C_n^n\right)
=3(8t+1)=4(6t)+3
\]
（其中 \(t\in\N\)），所以 \(3^{2n+1}\in\{b_1,b_2,\cdots\}\). 因此 \(3^{2n+1}\in\{a_1,a_2,\cdots\}\cap\{b_1,b_2,\cdots\}\).

再证明数列 \(\{a_n\}\) 中除了形如 \(3^{2n+1}\)（\(n\in\N\)）的项外，其它项都不是 \(\{b_n\}\) 中的项. 显然 \(a_1=3\) 不是 \(\{b_n\}\) 中的项. 对任意偶数项，
\[
a_{2n}=3^{2n}=(8+1)^n=8t+1=4(2t)+1
\]
故 \(a_{2n}\) 不是 \(\{b_n\}\) 中的项.

于是数列 \(\{a_n\}\) 与 \(\{b_n\}\) 的公共项恰为 \(a_3\)，\(a_5\)，\(a_7\)，\(\cdots\)，\(a_{2n+1}\)，\(\cdots\)，所以 \(d_n=a_{2n+1}=3^{2n+1}\)（\(n\in\N\)）.
\end{solution}
