\section{单选题}
\begin{problem}
若圆 \(C\) 与圆 \( (x+2)^{2}+(y-1)^{2}=1 \) 关于原点对称，则圆 \(C\) 的方程为（\quad）
\choices{\( (x-2)^{2}+(y+1)^{2}=1 \)}{\( (x-2)^{2}+(y-1)^{2}=1 \)}{\( (x-1)^{2}+(y+2)^{2}=1 \)}{\( (x+1)^{2}+(y-2)^{2}=1 \)}
\begin{answer}
A
\end{answer}

\begin{solution}
已知圆的圆心为 \((-2,1)\)，半径为 \(1\). 关于原点对称后，圆心变为 \((2,-1)\)，半径不变，所以圆 \(C\) 的方程为 \((x-2)^{2}+(y+1)^{2}=1\)，故选 A.
\end{solution}
\end{problem}

\begin{problem}
抛物线 \(y=ax^{2}\) 的准线方程是 \(y=2\)，则 \(a\) 的值为（\quad）
\choices{\( \frac{1}{8} \)}{\( -\frac{1}{8} \)}{\(8\)}{\(-8\)}
\begin{answer}
B
\end{answer}

\begin{solution}
将抛物线写成标准形式 \(x^{2}=4py\)，则 \(y=ax^{2}\) 等价于 \(x^{2}=\frac{1}{a}y\)，所以 \(4p=\frac{1}{a}\). 抛物线的准线为 \(y=-p\)，由 \(-p=2\) 得 \(p=-2\)，从而 \(a=\frac{1}{4p}=-\frac{1}{8}\)，故选 B.
\end{solution}
\end{problem}

\begin{problem}
已知 \( x \in \left(-\frac{\pi}{2}, 0\right) \)， \( \cos x = \frac{4}{5} \)，则 \( \tan 2x = \)（\quad）
\choices{\( \frac{7}{24} \)}{\( -\frac{7}{24} \)}{\( \frac{24}{7} \)}{\( -\frac{24}{7} \)}
\begin{answer}
D
\end{answer}

\begin{solution}
由 \(x\in\left(-\frac{\pi}{2},0\right)\) 且 \(\cos x=\frac{4}{5}\)，得 \(\sin x=-\frac{3}{5}\)，所以 \(\tan x=-\frac{3}{4}\). 因此
\(\tan 2x=\dfrac{2\tan x}{1-\tan^{2}x}=\dfrac{-\frac{3}{2}}{1-\frac{9}{16}}=-\frac{24}{7}\)，
故选 D.
\end{solution}
\end{problem}

\begin{problem}
已知四边形 \(ABCD\) 是菱形，点 \(P\) 在对角线 \(AC\) 上（不包括端点 \(A\)，\(C\)），则 \( \overrightarrow{AP} = \)（\quad）
\choices{\( \lambda(\overrightarrow{AB} + \overrightarrow{AD}) \)（\( \lambda \in (0, 1) \)）}{\( \lambda(\overrightarrow{AB} + \overrightarrow{BC}) \)（\( \lambda \in (0, \frac{\sqrt{2}}{2}) \)）}{\( \lambda(\overrightarrow{AB} - \overrightarrow{AD}) \)（\( \lambda \in (0, 1) \)）}{\( \lambda(\overrightarrow{AB} - \overrightarrow{BC}) \)（\( \lambda \in (0, \frac{\sqrt{2}}{2}) \)）}
\begin{answer}
A
\end{answer}

\begin{solution}
菱形是平行四边形，所以 \(\overrightarrow{AC}=\overrightarrow{AB}+\overrightarrow{AD}\). 因为点 \(P\) 在对角线 \(AC\) 的内部，所以存在 \(\lambda\in(0,1)\)，使得 \(\overrightarrow{AP}=\lambda\overrightarrow{AC}=\lambda(\overrightarrow{AB}+\overrightarrow{AD})\)，故选 A.
\end{solution}
\end{problem}

\begin{problem}
设函数 \( f(x)=\begin{cases}
2^{-x}-1, & x\leq 0,\\
\frac{1}{x^{2}}, & x>0.
\end{cases} \)，若 \( f(x_{0})>1 \)，则 \( x_{0} \) 的取值范围是（\quad）
\choices{\( (-1,1) \)}{\( (-1,+\infty) \)}{\( (-\infty,-2)\cup(0,+\infty) \)}{\( (-\infty,-1)\cup(1,+\infty) \)}
\begin{answer}
原题无正确选项
\end{answer}

\begin{solution}
当 \(x_{0}\leq0\) 时，\(f(x_{0})>1\) 等价于 \(2^{-x_{0}}-1>1\)，即 \(2^{-x_{0}}>2\)，从而 \(x_{0}<-1\). 当 \(x_{0}>0\) 时，\(f(x_{0})>1\) 等价于 \(\frac{1}{x_{0}^{2}}>1\)，所以 \(0<x_{0}<1\). 因而按原始题面直接计算得到
\(x_{0}\in(-\infty,-1)\cup(0,1)\).
这个集合与题面给出的四个选项均不相同，因此原卷题面没有正确选项，现有答案记录不能与题干同时成立.
\end{solution}
\end{problem}

\begin{problem}
等差数列 \( \{a_{n}\} \) 中，已知 \( a_{1}=\frac{1}{3} \)， \( a_{2}+a_{5}=4 \)， \( a_{n}=33 \)，则 \(n\) 为（\quad）
\choices{\(48\)}{\(49\)}{\(50\)}{\(51\)}
\begin{answer}
C
\end{answer}

\begin{solution}
设等差数列的公差为 \(d\). 由 \(a_{2}+a_{5}=4\) 得 \(2a_{1}+5d=4\)，即 \(\frac{2}{3}+5d=4\)，所以 \(d=\frac{2}{3}\). 因而 \(a_{n}=a_{1}+(n-1)d=\frac{1}{3}+\frac{2(n-1)}{3}=\frac{2n-1}{3}\). 由 \(a_{n}=33\) 得 \(2n-1=99\)，所以 \(n=50\)，故选 C.
\end{solution}
\end{problem}

\begin{problem}
函数 \( y = \ln \frac{x + 1}{x - 1} \)， \( x \in (1, +\infty) \) 的反函数为（\quad）
\choices{\( y = \frac{\e^{x} - 1}{\e^{x} + 1} \)（\( x \in (0, +\infty) \)）}{\( y = \frac{\e^{x} + 1}{\e^{x} - 1} \)（\( x \in (0, +\infty) \)）}{\( y = \frac{\e^{x} - 1}{\e^{x} + 1} \)（\( x \in (-\infty, 0) \)）}{\( y = \frac{\e^{x} + 1}{\e^{x} - 1} \)（\( x \in (-\infty, 0) \)）}
\begin{answer}
B
\end{answer}

\begin{solution}
设原函数中自变量为 \(t\)，函数值为 \(u\)，则 \(u=\ln\frac{t+1}{t-1}\)，其中 \(t>1\). 因而 \(\e^{u}=\frac{t+1}{t-1}\)，解得
\(t=\frac{\e^{u}+1}{\e^{u}-1}\).
当 \(t>1\) 时，\(\frac{t+1}{t-1}>1\)，所以 \(u>0\). 交换自变量与函数值，反函数为 \(y=\frac{\e^{x}+1}{\e^{x}-1}\)，\(x\in(0,+\infty)\)，故选 B.
\end{solution}
\end{problem}

\begin{problem}
棱长为 \(a\) 的正方体中，连结相邻面的中心，以这些线段为棱的八面体的体积为（\quad）
\choices{\( \frac{a^{3}}{3} \)}{\( \frac{a^{3}}{4} \)}{\( \frac{a^{3}}{6} \)}{\( \frac{a^{3}}{12} \)}
\begin{answer}
C
\end{answer}

\begin{solution}
取正方体中心为原点，六个面的中心分别为 \(\left(\pm\frac{a}{2},0,0\right)\)、\(\left(0,\pm\frac{a}{2},0\right)\)、\(\left(0,0,\pm\frac{a}{2}\right)\)，它们构成一个正八面体. 该八面体可由中心与八个全等三角形面分成八个全等四面体.

每个三角形面的边长为 \(\frac{a}{\sqrt{2}}\)，面积为 \(\frac{\sqrt{3}}{8}a^{2}\). 中心到该三角形所在平面的距离为 \(\frac{a}{2\sqrt{3}}\)，所以一个小四面体的体积为
\(\frac{1}{3}\cdot\frac{\sqrt{3}}{8}a^{2}\cdot\frac{a}{2\sqrt{3}}=\frac{a^{3}}{48}\).
因此八面体的体积为 \(8\cdot\frac{a^{3}}{48}=\frac{a^{3}}{6}\)，故选 C.
\end{solution}
\end{problem}

\begin{problem}
设 \( a > 0 \)， \( f(x) = ax^{2} + bx + c \)，曲线 \( y = f(x) \) 在点 \( P(x_{0}, f(x_{0})) \) 处的切线的倾斜角的取值范围为 \( [0, \frac{\pi}{4}] \)，则 \(P\) 到曲线 \( y = f(x) \) 对称轴距离的取值范围为（\quad）
\choices{\( \left[0,\frac{1}{a}\right] \)}{\( \left[0,\frac{1}{2a}\right] \)}{\( \left[0,\left|\frac{b}{2a}\right|\right] \)}{\( \left[0,\left|\frac{b-1}{2a}\right|\right] \)}
\begin{answer}
B
\end{answer}

\begin{solution}
抛物线在点 \(P(x_{0},f(x_{0}))\) 处的切线斜率为 \(f'(x_{0})=2ax_{0}+b\). 切线倾斜角属于 \(\left[0,\frac{\pi}{4}\right]\)，所以
\(0\leq2ax_{0}+b\leq1\)，
即 \(-\frac{b}{2a}\leq x_{0}\leq\frac{1-b}{2a}\). 曲线的对称轴为 \(x=-\frac{b}{2a}\)，因此点 \(P\) 到对称轴的距离为 \(x_{0}+\frac{b}{2a}\)，其取值范围为 \(\left[0,\frac{1}{2a}\right]\)，故选 B.
\end{solution}
\end{problem}

\begin{problem}
已知双曲线中心在原点且一个焦点为 \(F(\sqrt{7}, 0)\)，直线 \(y = x - 1\) 与其相交于 \(M\)，\(N\) 两点，\(MN\) 中点的横坐标为 \(-\frac{2}{3}\)，则此双曲线的方程是（\quad）
\choices{\( \frac{x^{2}}{3}-\frac{y^{2}}{4}=1 \)}{\( \frac{x^{2}}{4}-\frac{y^{2}}{3}=1 \)}{\( \frac{x^{2}}{5}-\frac{y^{2}}{2}=1 \)}{\( \frac{x^{2}}{2}-\frac{y^{2}}{5}=1 \)}
\begin{answer}
D
\end{answer}

\begin{solution}
设双曲线方程为 \(\frac{x^{2}}{a^{2}}-\frac{y^{2}}{b^{2}}=1\)，其中焦半距为 \(\sqrt{a^{2}+b^{2}}=\sqrt{7}\)，所以 \(a^{2}+b^{2}=7\). 将直线 \(y=x-1\) 代入，得
\[
\frac{x^{2}}{a^{2}}-\frac{(x-1)^{2}}{b^{2}}=1
\]
设交点的横坐标为 \(x_{1}\)，\(x_{2}\). 整理为关于 \(x\) 的二次方程并由韦达定理得
\[
x_{1}+x_{2}=\frac{-2a^{2}}{b^{2}-a^{2}}
\]
由中点横坐标为 \(-\frac{2}{3}\)，得 \(x_{1}+x_{2}=-\frac{4}{3}\)，所以
\(\frac{-2a^{2}}{b^{2}-a^{2}}=-\frac{4}{3}\)，\(5a^{2}=2b^{2}\).
联立 \(a^{2}+b^{2}=7\)，得 \(a^{2}=2\)，\(b^{2}=5\). 因此双曲线方程为 \(\frac{x^{2}}{2}-\frac{y^{2}}{5}=1\)，故选 D.
\end{solution}
\end{problem}

\begin{problem}
已知长方形的四个顶点 \(A(0, 0)\)， \(B(2, 0)\)， \(C(2, 1)\) 和 \(D(0, 1)\). 一质点从 \(AB\) 的中点 \(P_0\) 沿与 \(AB\) 夹角为 \(\theta\) 的方向射到 \(BC\) 上的点 \(P_1\) 后，依次反射到 \(CD\)， \(DA\) 和 \(AB\) 上的点 \(P_2\)， \(P_3\) 和 \(P_4\)（入射角等于反射角）. 设 \(P_4\) 的坐标为 \( (x_4, 0) \)，若 \( 1 < x_4 < 2 \)，则 \( \tan \theta \) 的取值范围是（\quad）
\choices{\( \left(\frac{1}{3},1\right) \)}{\( \left(\frac{1}{3},\frac{2}{3}\right) \)}{\( \left(\frac{2}{5},\frac{1}{2}\right) \)}{\( \left(\frac{2}{5},\frac{2}{3}\right) \)}
\begin{answer}
C
\end{answer}

\begin{solution}
令 \(t=\tan\theta\). 由于质点先从 \(AB\) 射向 \(BC\)，取 \(t>0\). 将每次反射后的长方形关于反射边对称展开，反射轨迹就变为一条直线. 以原长方形左下角为坐标原点，在展开图中取 \(P_{0}=(1,0)\)，则直线轨迹为
\(Y=t(X-1)\).

第一次碰到 \(BC\) 时，展开图中的右边界为 \(X=2\)，故 \(P_{1}\) 对应于 \((2,t)\). 将长方形关于 \(BC\) 展开后，第二次碰到 \(CD\) 时仍在 \(2<X<4\) 的展开矩形内；由 \(Y=1\) 得
\[
X=1+\frac{1}{t}
\]
再关于 \(CD\) 展开，第三次碰到 \(DA\) 时对应于展开图中的边界 \(X=4\)，所以
\[
Y=3t
\]
再关于 \(DA\) 展开，第四次碰到 \(AB\) 时对应于 \(Y=2\)，因而
\[
X=1+\frac{2}{t}
\]

第四次碰到 \(AB\) 的展开矩形为 \(4<X<6\)，该矩形折回原长方形后横坐标为 \(x_{4}=X-4\)，所以
\[
x_{4}=\frac{2}{t}-3
\]
由 \(1<x_{4}<2\)，得
\(1<\frac{2}{t}-3<2\)，
从而 \(\frac{2}{5}<t<\frac{1}{2}\). 这个范围自动保证各次碰撞依次落在相应边界的内部，故 \(\tan\theta\in\left(\frac{2}{5},\frac{1}{2}\right)\)，选 C.
\end{solution}
\end{problem}

\begin{problem}
一个四面体的所有棱长都为 \( \sqrt{2} \)，四个顶点在同一球面上，则此球的表面积为（\quad）
\choices{\( 3\pi \)}{\( 4\pi \)}{\( 3\sqrt{3}\pi \)}{\( 6\pi \)}
\begin{answer}
A
\end{answer}

\begin{solution}
所有棱长相等的四面体是正四面体. 设其底面为边长 \(\sqrt{2}\) 的正三角形，底面中心到任一顶点的距离为 \(\frac{\sqrt{6}}{3}\). 正四面体的高为 \(\frac{2}{\sqrt{3}}\)，球心为正四面体的重心，因此球心到该底面的距离为高的 \(\frac{1}{4}\)，即 \(\frac{1}{2\sqrt{3}}\). 所以球半径 \(R\) 满足
\[
R^{2}=\left(\frac{\sqrt{6}}{3}\right)^{2}+\left(\frac{1}{2\sqrt{3}}\right)^{2}=\frac{2}{3}+\frac{1}{12}=\frac{3}{4}
\]
故球的表面积为 \(4\pi R^{2}=3\pi\)，选 A.
\end{solution}
\end{problem}

\section{填空题}
\begin{problem}
\( (x^{2}-\frac{1}{2x})^{9} \) 展开式中 \( x^{9} \) 的系数是\fillinblank{}.
\begin{answer}
\(-\frac{21}{2}\)
\end{answer}

\begin{solution}
展开式的通项为
\[
\mathrm C_{9}^{k}(x^{2})^{9-k}\left(-\frac{1}{2x}\right)^{k}=\mathrm C_{9}^{k}\frac{(-1)^{k}}{2^{k}}x^{18-3k}
\]
要得到 \(x^{9}\)，必须满足 \(18-3k=9\)，所以 \(k=3\). 因此所求系数为
\[
\mathrm C_{9}^{3}\frac{(-1)^{3}}{2^{3}}=-\frac{84}{8}=-\frac{21}{2}
\]
\end{solution}
\end{problem}

\begin{problem}
某公司生产三种型号的轿车，产量分别为 \(1200\text{ 辆}\)，\(6000\text{ 辆}\) 和 \(2000\text{ 辆}\). 为检验该公司的产品质量，现用分层抽样的方法抽取 \(46\text{ 辆}\) 进行检验，这三种型号的轿车依次应抽取\fillinblank{}，\fillinblank{}，\fillinblank{}\(\text{ 辆}\).
\begin{answer}
6，30，10
\end{answer}

\begin{solution}
三种型号轿车的总产量为 \(1200+6000+2000=9200\text{ 辆}\). 分层抽样要求各层的抽样比相同，抽样比为 \(\frac{46}{9200}=\frac{1}{200}\). 因而三种型号依次抽取 \(1200\cdot\frac{1}{200}=6\text{ 辆}\)，\(6000\cdot\frac{1}{200}=30\text{ 辆}\)，\(2000\cdot\frac{1}{200}=10\text{ 辆}\).
\end{solution}
\end{problem}

\begin{problem}
对于四面体 \(ABCD\)，给出下列四个命题：

\begin{circlelist}
\item 若 \(AB=AC\)，\(BD=CD\)，则 \(BC \perp AD\)；
\item 若 \(AB=CD\)，\(AC=BD\)，则 \(BC \perp AD\)；
\item 若 \(AB \perp AC\)，\(BD \perp CD\)，则 \(BC \perp AD\)；
\item 若 \(AB \perp CD\)，\(BD \perp AC\)，则 \(BC \perp AD\).
\end{circlelist}

其中真命题的序号是\fillinblank{}.
\begin{answer}
\(\circled{1}\)，\(\circled{4}\)
\end{answer}

\begin{solution}
命题\(\circled{1}\)中，满足 \(AB=AC\) 的点 \(A\) 和满足 \(BD=CD\) 的点 \(D\) 都在边 \(BC\) 的垂直平分面内. 因而直线 \(AD\) 在这个平面内，而垂直平分面的法向量平行于 \(BC\)，所以 \(AD\perp BC\)，命题\(\circled{1}\)正确.

对命题\(\circled{2}\)，取
\(A=(0,0,0)\)，\(B=(0,1,0)\)，\(C=(1,0,1)\)，\(D=(1,0,0)\). 则 \(AB=CD=1\)，\(AC=BD=\sqrt{2}\)，但 \(\overrightarrow{BC}\cdot\overrightarrow{AD}=(1,-1,1)\cdot(1,0,0)=1\neq0\)，故命题\(\circled{2}\)错误.

对命题\(\circled{3}\)，取
\(A=(0,0,0)\)，\(B=(1,0,0)\)，\(C=(0,1,0)\)，\(D=\left(0,\frac{1}{2},\frac{1}{2}\right)\). 此时 \(\overrightarrow{AB}\cdot\overrightarrow{AC}=0\)，且
\(\overrightarrow{BD}\cdot\overrightarrow{CD}=(-1,\tfrac{1}{2},\tfrac{1}{2})\cdot(0,-\tfrac{1}{2},\tfrac{1}{2})=0\)，
但 \(\overrightarrow{BC}\cdot\overrightarrow{AD}=(-1,1,0)\cdot\left(0,\frac{1}{2},\frac{1}{2}\right)=\frac{1}{2}\neq0\)，故命题\(\circled{3}\)错误.

对命题\(\circled{4}\)，设 \(A\) 为原点，记 \(\overrightarrow{AB}=\bs b\)，\(\overrightarrow{AC}=\bs c\)，\(\overrightarrow{AD}=\bs d\). 由 \(AB\perp CD\) 得 \(\bs b\cdot(\bs d-\bs c)=0\)，所以 \(\bs b\cdot\bs d=\bs b\cdot\bs c\). 由 \(BD\perp AC\) 得 \((\bs d-\bs b)\cdot\bs c=0\)，所以 \(\bs d\cdot\bs c=\bs b\cdot\bs c\). 因而
\(\overrightarrow{BC}\cdot\overrightarrow{AD}=(\bs c-\bs b)\cdot\bs d=\bs c\cdot\bs d-\bs b\cdot\bs d=0\)，
命题\(\circled{4}\)正确. 所以真命题的序号为 \(1\)，\(4\).
\end{solution}
\end{problem}

\begin{problem}
将三种作物种植在如图的 \(5\text{ 块}\) 试验田里，每块种植一种作物且相邻的试验田不能种植同一作物，不同的种植方法共有\fillinblank{}\(\text{ 种}\).

\begin{center}
\bitmapfigure[max width=0.35\linewidth]{img/2003/henan/q16_fig1.png}
\end{center}
\begin{answer}
42
\end{answer}

\begin{solution}
第一块试验田有 \(3\) 种选择. 以后每一块都不能与前一块种相同的作物，所以各有 \(2\) 种选择. 若不附加“三种作物均要使用”的条件，共有 \(3\cdot2^{4}=48\) 种.

其中只使用两种作物的种植方法需要先从三种作物中选出两种，有 \(\mathrm C_{3}^{2}\) 种选择；选定后沿五块相邻试验田只能交替种植，恰有 \(2\) 种方法. 因此只使用两种作物的方法有 \(\mathrm C_{3}^{2}\cdot2=6\) 种，故三种作物均使用时的方法数为 \(48-6=42\).
\end{solution}
\end{problem}

\begin{problem}
已知函数 \( f(x)=2\sin x(\sin x+\cos x) \).

\begin{enumerate}
\item 求函数 \(f(x)\) 的最小正周期和最大值；
\item 在给出的直角坐标系中，画出函数 \(y=f(x)\) 在区间 \(\left[-\frac{\pi}{2}, \frac{\pi}{2}\right]\) 上的图象.
\end{enumerate}

\begin{center}
\bitmapfigure[max width=0.60\linewidth]{img/2003/henan/q17_fig1.png}
\end{center}

\begin{solution}
\begin{enumerate}
\item \(f(x)=2\sin^{2}x+2\sin x\cos x=1-\cos 2x+\sin 2x=1+\sqrt{2}\sin\left(2x-\frac{\pi}{4}\right)\). 因此，函数的最小正周期为 \(\pi\)，最大值为 \(1+\sqrt{2}\).
\item \(f'(x)=2\left(\sin 2x+\cos 2x\right)=2\sqrt{2}\sin\left(2x+\frac{\pi}{4}\right)\). 在区间 \(\left[-\frac{\pi}{2},\frac{\pi}{2}\right]\) 上，函数在 \(\left[-\frac{\pi}{2},-\frac{\pi}{8}\right]\) 上递减，在 \(\left[-\frac{\pi}{8},\frac{3\pi}{8}\right]\) 上递增，在 \(\left[\frac{3\pi}{8},\frac{\pi}{2}\right]\) 上递减. 图象经过 \(\left(-\frac{\pi}{2},2\right)\)，\(\left(-\frac{\pi}{4},0\right)\)，\(\left(-\frac{\pi}{8},1-\sqrt{2}\right)\)，\((0,0)\)，\(\left(\frac{3\pi}{8},1+\sqrt{2}\right)\)，\(\left(\frac{\pi}{2},2\right)\) 这些点，在给出的坐标系中将各点按函数的变化趋势光滑连接即可.
\end{enumerate}
\end{solution}
\end{problem}

\begin{problem}
有三种产品，合格率分别是 \(0.90\)，\(0.95\) 和 \(0.95\)，各抽取一件进行检验.

\begin{enumerate}
\item 求恰有一件不合格的概率；
\item 求至少有两件不合格的概率.（精确到 \(0.001\)）
\end{enumerate}

\begin{solution}
\begin{enumerate}
\item 三种产品不合格的概率分别为 \(0.10\)，\(0.05\)，\(0.05\). 恰有一件不合格的概率为 \(0.10\times0.95\times0.95+0.90\times0.05\times0.95+0.90\times0.95\times0.05=0.17575\)，精确到 \(0.001\) 为 \(0.176\).

\begin{enumerate}
\item 建立空间直角坐标系，取 \(C\) 为原点，\(CA\)，\(CB\)，\(CC_{1}\) 分别为三个坐标轴正方向. 设 \(CA=CB=l\)，则

\begin{problem}
已知 \(c > 0\)，设 \(P\)：函数 \(y = c^{x}\) 在 \(\R\) 上单调递减，\(Q\)：不等式 \(x + |x - 2c| > 1\) 的解集为 \(\R\). 如果 \(P\) 和 \(Q\) 有且仅有一个正确，求 \(c\) 的取值范围.

\begin{solution}
函数 \(y=c^{x}\) 在 \(\R\) 上单调递减的充要条件是 \(0<c<1\)，所以命题 \(P\) 在此条件下正确.

对命题 \(Q\)，当 \(x<2c\) 时，\(x+|x-2c|=2c\)；当 \(x\geq2c\) 时，\(x+|x-2c|=2x-2c\geq2c\). 因而不等式对一切 \(x\in\R\) 成立的充要条件是 \(2c>1\)，即 \(c>\frac{1}{2}\)，所以命题 \(Q\) 在此条件下正确.

当 \(0<c\leq\frac{1}{2}\) 时，\(P\) 正确而 \(Q\) 错误；当 \(c\geq1\) 时，\(P\) 错误而 \(Q\) 正确. 因此 \(c\in\left(0,\frac{1}{2}\right]\cup[1,+\infty)\).
\end{solution}
\end{problem}

\begin{problem}
已知常数 \(a > 0\)，向量 \( \bs{c} = (0, a) \)， \( \bs{i} = (1, 0) \)，经过原点 \(O\) 以 \( \bs{c} + \lambda \bs{i} \) 为方向向量的直线与经过定点 \(A(0, a)\) 以 \( \bs{i} - 2\lambda \bs{c} \) 为方向向量的直线相交于点 \(P\)，其中 \(\lambda \in \R\). 试问：是否存在两个定点 \(E\)，\(F\)，使得 \(|PE| + |PF|\) 为定值. 若存在，求出 \(E\)，\(F\) 的坐标；若不存在，说明理由.

\begin{solution}
设交点 \(P\) 的坐标为 \((x,y)\). 两条直线的方向向量分别为 \((\lambda,a)\) 和 \((1,-2\lambda a)\)，所以它们的方程分别为 \(\lambda y=ax\) 和 \(y-a=-2\lambda ax\). 将第一式乘以 \(y\)，再利用第二式消去 \(\lambda\)，得
\[
y(y-a)=-2a^{2}x^{2}
\]
即
\[
2a^{2}x^{2}+\left(y-\frac{a}{2}\right)^{2}=\frac{a^{2}}{4}
\]
由于 \(a>0\)，点 \(P\) 在椭圆
\[
\frac{x^{2}}{1/8}+\frac{\left(y-\frac{a}{2}\right)^{2}}{\left(\frac{a}{2}\right)^{2}}=1
\]
上. 令 \(P=s(\lambda,a)=(s\lambda,sa)\)，同时由第二条直线可写成
\(P=(t,a-2\lambda at)\). 比较坐标得 \(t=s\lambda\) 以及
\(sa=a-2\lambda a(s\lambda)\)，从而 \(s=\frac{1}{1+2\lambda^{2}}\)，即
\[
P=\left(\frac{\lambda}{1+2\lambda^{2}},\frac{a}{1+2\lambda^{2}}\right)
\]
所以 \(y>0\). 反之，对上述椭圆上任意满足 \(y>0\) 的点 \((x,y)\)，由椭圆方程可得 \(2a^{2}x^{2}=y(a-y)\). 取 \(\lambda=\frac{ax}{y}\)，则 \(1+2\lambda^{2}=\frac{a}{y}\)，从而
\[
\left(\frac{\lambda}{1+2\lambda^{2}},\frac{a}{1+2\lambda^{2}}\right)=\left(\frac{ax/y}{a/y},\frac{a}{a/y}\right)=(x,y)
\]
因此，当 \(\lambda\) 取遍 \(\R\) 时，实际轨迹是上述椭圆去掉下端点 \((0,0)\)；原点只是 \(\lvert\lambda\rvert\to+\infty\) 时的极限点.

当 \(0<a<\frac{\sqrt{2}}{2}\) 时，横半轴 \(\frac{1}{2\sqrt{2}}\) 是长半轴，焦半距为
\[
\sqrt{\frac{1}{8}-\frac{a^{2}}{4}}=\frac{1}{2}\sqrt{\frac{1}{2}-a^{2}}
\]
取两个焦点 \(E\left(\frac{1}{2}\sqrt{\frac{1}{2}-a^{2}},\frac{a}{2}\right)\)，\(F\left(-\frac{1}{2}\sqrt{\frac{1}{2}-a^{2}},\frac{a}{2}\right)\).
椭圆上任一点到两个焦点的距离和为横轴长 \(\frac{1}{\sqrt{2}}\)，所以存在这样的两个定点.

当 \(a>\frac{\sqrt{2}}{2}\) 时，纵半轴 \(\frac{a}{2}\) 是长半轴，焦半距为
\[
\sqrt{\frac{a^{2}}{4}-\frac{1}{8}}=\frac{1}{2}\sqrt{a^{2}-\frac{1}{2}}
\]
取两个焦点 \(E\left(0,\frac{a+\sqrt{a^{2}-\frac{1}{2}}}{2}\right)\)，\(F\left(0,\frac{a-\sqrt{a^{2}-\frac{1}{2}}}{2}\right)\).
此时距离和为纵轴长 \(a\)，所以也存在这样的两个定点.

当 \(a=\frac{\sqrt{2}}{2}\) 时，椭圆变为圆，实际轨迹是该圆去掉下端点 \((0,0)\)，因而包含一个非退化圆弧. 若存在两个不同定点 \(E\)，\(F\) 使距离和在这段圆弧上恒定，则以 \(E\)，\(F\) 为焦点的非退化椭圆包含一段圆弧，这不可能，因为非退化椭圆不能与圆重合；只有两个焦点重合时，定和曲线才可能是圆. 因而圆情形不存在满足题意的两个不同定点.
\end{solution}
\end{problem}

\begin{problem}
已知 \(a > 0\)，\(n\) 为正整数.

\begin{enumerate}
\item 设 \(y=(x-a)^{n}\)，证明 \(y'=n(x-a)^{n-1}\)；
\item 设 \(f_{n}(x)=x^{n}-(x-a)^{n}\)，对任意 \(n \geq a\)，证明 \(f_{n+1}^{\prime}(n+1) > (n+1)f_{n}^{\prime}(n)\).
\end{enumerate}

\begin{solution}
\begin{enumerate}
\item 由二项式定理，\((x-a+h)^{n}-(x-a)^{n}=n(x-a)^{n-1}h+\sum_{k=2}^{n}\mathrm C_{n}^{k}(x-a)^{n-k}h^{k}\). 令 \(R(h)=\sum_{k=2}^{n}\mathrm C_{n}^{k}(x-a)^{n-k}h^{k-1}\). 当 \(n=1\) 时，\(R(h)=0\)；当 \(n\geq2\) 时，若 \(|h|\leq1\)，则 \(|R(h)|\leq M|h|\)，其中 \(M=\sum_{k=2}^{n}\left|\mathrm C_{n}^{k}(x-a)^{n-k}\right|\). 对任意 \(\varepsilon>0\)，取 \(\delta=\min\left\{1,\frac{\varepsilon}{M+1}\right\}\)，当 \(0<|h|<\delta\) 时有 \(|R(h)|<\varepsilon\)，所以 \(\lim_{h\to0}R(h)=0\). 因此
\[y'=\lim_{h\to0}\frac{(x-a+h)^{n}-(x-a)^{n}}{h}=n(x-a)^{n-1}+\lim_{h\to0}R(h)=n(x-a)^{n-1}\]
\item 由（1）得 \(f_{n}'(x)=n\left[x^{n-1}-(x-a)^{n-1}\right]\). 当 \(n=1\) 时，\(f_{2}'(2)=2a>0=2f_{1}'(1)\)，结论成立. 下设 \(n\geq2\)，令
\[\Delta(a)=(n+1)^{n}-(n+1-a)^{n}-n\left[n^{n-1}-(n-a)^{n-1}\right]\]
因为 \(0<a\leq n\)，且 \(\Delta(0)=0\)，对 \(0<a<n\) 有
\[\Delta'(a)=n\left[(n+1-a)^{n-1}-(n-1)(n-a)^{n-2}\right]\]
令 \(t=n-a>0\). 当 \(n=2\) 时，\((t+1)^{n-1}=t+1>1=(n-1)t^{n-2}\). 当 \(n\geq3\) 时，由二项式定理，
\[(t+1)^{n-1}=t^{n-1}+(n-1)t^{n-2}+\sum_{j=0}^{n-3}\mathrm C_{n-1}^{j}t^{j}>(n-1)t^{n-2}\]
所以 \(\Delta'(a)>0\)，即 \(\Delta\) 在 \((0,n)\) 上严格递增. 因而当 \(0<a<n\) 时，\(\Delta(a)>\Delta(0)=0\). 当 \(a=n\) 时，由连续性
\(\Delta(n)=\lim_{u\to n^-}\Delta(u)\geq\Delta\left(\frac{n}{2}\right)>0\).
最后
\[f_{n+1}'(n+1)-(n+1)f_{n}'(n)=(n+1)\Delta(a)>0\]
故不等式成立.
\end{enumerate}
\end{solution}
\end{problem}
