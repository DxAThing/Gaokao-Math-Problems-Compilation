\begin{problem}

\begin{problem}
已知 \(a > 0, n\) 为正整数.

\begin{enumerate}
\item 设 \( y = (x - a)^n \)，证明：\( y' = n(x - a)^{n-1} \);
\item 设 \( f_{n}(x) = x^{n} - (x - a)^{n} \)，对任意 \( n \geqslant a \)，证明：\( f_{n+1}'(n+1) > (n+1)f_{n}'(n) \).
\end{enumerate}
\end{problem}

\begin{problem}

\begin{enumerate}

\item 当 \(a = 1\) 时，证明：\(\sum_{k=1}^{n-1} (a_k - a_{k+1}) a_{k+2} < \frac{1}{3}\).
\end{enumerate}
