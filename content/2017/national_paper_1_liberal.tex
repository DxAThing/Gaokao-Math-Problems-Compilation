\section{单选题}

\begin{problem}
已知集合 \(A=\{x\mid x<2\}\)，\(B=\{x\mid 3-2x>0\}\)，则（\quad）

\choices
    {\(A\cap B=\{x\mid x<\dfrac{3}{2}\}\)}
    {\(A\cap B=\varnothing\)}
    {\(A\cup B=\{x\mid x<\dfrac{3}{2}\}\)}
    {\(A\cup B=\R\)}

\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
由 \(3-2x>0\) 得 \(B=\{x\mid x<\dfrac{3}{2}\}\)，所以 \(A\cap B=\{x\mid x<\dfrac{3}{2}\}\)，\(A\cup B=\{x\mid x<2\}\). 故选 \(A\).
\end{solution}

为评估一种农作物的种植效果，选了 \(n\) 块地作试验田. 这 \(n\) 块地的亩产量（单位：kg）分别是 \(x_1,x_2,\dots,x_n\)，下面给出的指标中可以用来评估这种农作物亩产量稳定程度的是（\quad）

    {\(x_1,x_2,\dots,x_n\) 的中位数}

\end{problem}

\begin{answer}

\begin{problem}
下列各式的运算结果为纯虚数的是（\quad）

\choices
    {\(\i(1+\i)^2\)}
    {\(\i^2(1-\i)\)}
    {\((1+\i)^2\)}
    {\(\i(1+\i)\)}

\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
\(\i(1+\i)^2=\i\cdot 2\i=-2\) 是实数，\(\i^2(1-\i)=-1+\i\) 不是纯虚数，\((1+\i)^2=2\i\) 是纯虚数，\(\i(1+\i)=\i-1\) 不是纯虚数，故选 \(C\).
\end{solution}

如图，正方形 \(ABCD\) 内的图形来自中国古代的太极图. 正方形内切圆中的黑色部分和白色部分关于正方形的中心成中心对称. 在正方形内随机取一点，则此点取自黑色部分的概率是（\quad）

\choices

    {\(\dfrac{\pi}{8}\)}
    {\(\dfrac{1}{2}\)}

\end{problem}

\begin{answer}
B
\end{answer}

\begin{problem}
\(\triangle ABC\) 的内角 \(A,B,C\) 的对边分别为 \(a,b,c\)，已知 \(\sin B+\sin A(\sin C-\cos C)=0\)，\(a=2\)，\(c=\sqrt2\)，则 \(C=\)（\quad）

\choices
    {\(\dfrac{\pi}{12}\)}
    {\(\dfrac{\pi}{6}\)}
    {\(\dfrac{\pi}{4}\)}
    {\(\dfrac{\pi}{3}\)}

\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
由 \(\sin B=\sin(A+C)=\sin A\cos C+\cos A\sin C\)，代入得 \(\cos A\sin C+\sin A\sin C=0\). 又 \(\sin C\ne 0\)，故 \(\cos A=-\sin A\)，从而 \(A=\dfrac{3\pi}{4}\). 由正弦定理 \(\dfrac{a}{\sin A}=\dfrac{c}{\sin C}\)，得 \(\dfrac{2}{\sqrt2/2}=\dfrac{\sqrt2}{\sin C}\)，\(\sin C=\dfrac12\). 结合 \(B+C=\dfrac{\pi}{4}\) 可得 \(C=\dfrac{\pi}{6}\). 故选 \(B\).
\end{solution}

\begin{problem}
已知向量 \(\bs a=(-1,2)\)，\(\bs b=(m,1)\)，若向量 \(\bs a+\bs b\) 与 \(\bs a\) 垂直，则 \(m=\fillinblank{}\).

\end{problem}

\begin{answer}
7
\end{answer}

\begin{solution}
由 \((\bs a+\bs b)\cdot \bs a=0\)，得 \((m-1,3)\cdot(-1,2)=0\)，即 \(-m+1+6=0\)，所以 \(m=7\).
\end{solution}

\begin{problem}
曲线 \(y=x^2+\dfrac{1}{x}\) 在点 \((1,2)\) 处的切线方程为\fillinblank{}.

\end{problem}

\begin{answer}
\(x-y+1=0\)
\end{answer}

\begin{solution}
函数导数为 \(y'=2x-\dfrac{1}{x^2}\)，故在 \(x=1\) 处切线斜率为 \(1\). 切线方程为 \(y-2=x-1\)，即 \(x-y+1=0\).
\end{solution}

\begin{problem}
已知 \(\alpha\in(0,\dfrac{\pi}{2})\)，\(\tan\alpha=2\)，则 \(\cos\left(\alpha-\dfrac{\pi}{4}\right)=\fillinblank{}\).

\end{problem}

\begin{answer}
\(\dfrac{3\sqrt{10}}{10}\)
\end{answer}

\begin{solution}
由 \(\tan\alpha=2\) 且 \(\alpha\in(0,\dfrac{\pi}{2})\)，得 \(\sin\alpha=\dfrac{2}{\sqrt5}\)，\(\cos\alpha=\dfrac{1}{\sqrt5}\). 因此 \(\cos\left(\alpha-\dfrac{\pi}{4}\right)=\cos\alpha\cos\dfrac{\pi}{4}+\sin\alpha\sin\dfrac{\pi}{4}=\dfrac{1+2}{\sqrt{10}}=\dfrac{3\sqrt{10}}{10}\).
\end{solution}

\begin{problem}
已知三棱锥 \(S-ABC\) 的所有顶点都在球 \(O\) 的球面上，\(SC\) 是球 \(O\) 的直径. 若平面 \(SCA\perp\) 平面 \(SCB\)，\(SA=AC\)，\(SB=BC\)，三棱锥 \(S-ABC\) 的体积为 \(9\)，则球 \(O\) 的表面积为\fillinblank{}.

\end{problem}

\begin{answer}
\(36\pi\)
\end{answer}

\begin{solution}
设球半径为 \(r\)，取 \(SC\) 为 \(z\) 轴，令 \(S(0,0,r)\)，\(C(0,0,-r)\). 由 \(SA=AC\)，\(SB=BC\) 知 \(A,B\) 在球的赤道平面上. 再由平面 \(SCA\perp\) 平面 \(SCB\) 可取 \(A(r,0,0)\)，\(B(0,r,0)\). 于是三棱锥 \(S-ABC\) 的体积为
\examdisplaymath{
V=\frac{1}{6}\left|\det\begin{pmatrix}
r&0&r\\
0&r&r\\
0&0&2r
\end{pmatrix}\right|=\frac{r^3}{3}.
}
由 \(V=9\) 得 \(r=3\)，所以球面面积为 \(4\pi r^2=36\pi\).
\end{solution}

\begin{problem}
记 \(S_n\) 为等比数列 \(\{a_n\}\) 的前 \(n\) 项和. 已知 \(S_2=2\)，\(S_3=-6\).

\begin{enumerate}
\item 求 \(\{a_n\}\) 的通项公式；
\item 求 \(S_n\)，并判断 \(S_{n+1},S_n,S_{n+2}\) 是否成等差数列.
\end{enumerate}

\end{problem}

\begin{solution}
设等比数列首项为 \(a_1\)，公比为 \(q\). 由 \(a_1+a_2=2\)，\(a_1+a_2+a_3=-6\) 得 \(a_3=S_3-S_2=-8\). 由 \(a_3=a_1q^2\)，\(a_2=a_1q\) 和 \(a_1+a_2=2\) 可得 \(q^2+4q+4=0\)，所以 \(q=-2\)，进而 \(a_1=-2\)，故 \(a_n=(-2)^n\). 于是 \(S_n=\sum_{k=1}^n(-2)^k=\dfrac{2\bigl((-2)^n-1\bigr)}{3}\). 再由 \(S_{n+1}=S_n+(-2)^{n+1}\)，\(S_{n+2}=S_n+(-2)^{n+1}+(-2)^{n+2}\)，可得 \(S_{n+1}+S_{n+2}=2S_n\)，因此 \(S_{n+1},S_n,S_{n+2}\) 成等差数列.
\end{solution}

\begin{problem}
如图，在四棱锥 \(P-ABCD\) 中，\(AB\parallel CD\)，且 \(\angle BAP=\angle CDP=90^\circ\).

\begin{enumerate}
\item 证明：平面 \(PAB\perp\) 平面 \(PAD\)；
\item 若 \(PA=PD=AB=DC\)，\(\angle APD=90^\circ\)，且四棱锥 \(P-ABCD\) 的体积为 \( \dfrac{8}{3}\)，求该四棱锥的侧面积.
\end{enumerate}

\bitmapfigure[max width=0.62\linewidth,max totalheight=4.8cm]{img/2017/national_paper_1_liberal/q18_fig1.png}

\end{problem}

\begin{solution}
（1）由 \(\angle BAP=\angle CDP=90^\circ\)，得 \(AB\perp PA\)，\(CD\perp PD\). 又 \(AB\parallel CD\)，所以 \(AB\perp PD\). 因为 \(PA,PD\subset\) 平面 \(PAD\)，且 \(PA\cap PD=P\)，所以 \(AB\perp\) 平面 \(PAD\). 又 \(AB\subset\) 平面 \(PAB\)，故平面 \(PAB\perp\) 平面 \(PAD\).

（2）设 \(PA=PD=AB=DC=a\)，取 \(AD\) 的中点 \(O\)，连结 \(PO\). 由 \(AB\parallel CD\) 且 \(AB=CD\)，知 \(ABCD\) 为平行四边形. 又由（1）知 \(AB\perp\) 平面 \(PAD\)，所以 \(AB\perp AD\)，从而 \(ABCD\) 为矩形，故 \(AD=BC\).

在 \(\triangle APD\) 中，\(PA=PD=a\)，且 \(\angle APD=90^\circ\)，所以
\(AD=a\sqrt2\).
又 \(O\) 是斜边 \(AD\) 的中点，因此
\(PO=OA=OD=\dfrac{a\sqrt2}{2}=\dfrac{a}{\sqrt2}\)，
且 \(PO\perp AD\).

过点 \(O\) 作 \(OE\parallel AB\)，交 \(BC\) 于点 \(E\). 因为 \(AB\perp\) 平面 \(PAD\)，而 \(PO\subset\) 平面 \(PAD\)，所以 \(PO\perp AB\)，从而 \(PO\perp OE\). 又 \(AD\) 与 \(OE\) 是底面 \(ABCD\) 内过点 \(O\) 的两条相交直线，所以 \(PO\perp\) 底面 \(ABCD\).

于是 \(V=\dfrac13\cdot S_{ABCD}\cdot PO=\dfrac13\cdot (AB\cdot AD)\cdot PO=\dfrac13\cdot a\cdot a\sqrt2\cdot \dfrac{a}{\sqrt2}=\dfrac{a^3}{3}\).
由 \(V=\dfrac{8}{3}\) 得 \(a=2\).

于是 \(PA=PD=AB=DC=2\)，\(AD=BC=2\sqrt2\)，\(PO=\sqrt2\).
又
\(PB=PC=\sqrt{PO^2+\left( \dfrac{AD}{2} \right)^2+AB^2}=\sqrt{2+2+4}=2\sqrt2\).
所以
\examdisplayaligned{
S_{\text{侧}}
&=S_{\triangle PAD}+S_{\triangle PAB}+S_{\triangle PDC}+S_{\triangle PBC}\\
&=\frac12\cdot PA\cdot PD+\frac12\cdot PA\cdot AB+\frac12\cdot PD\cdot DC+\frac12\cdot BC\cdot\sqrt{PB^2-\left(\frac{BC}{2}\right)^2}\\
&=2+2+2+2\sqrt3\\
&=6+2\sqrt3.
}
\end{solution}

\begin{problem}
为了监控某种零件的一条生产线的生产过程，检验员每隔 \(30\) min 从该生产线上随机抽取一个零件，并测量其尺寸（单位：cm）. 下面是检验员在一天内依次抽取的 \(16\) 个零件的尺寸：

经计算得 \(\bar x=\dfrac{1}{16}\sum_{i=1}^{16}x_i=9.97\)，\(s=\sqrt{\dfrac{1}{16}\sum_{i=1}^{16}(x_i-\bar x)^2}\approx0.212\)，\(\sqrt{\sum_{i=1}^{16}(i-8.5)^2}\approx18.439\)，\(\sum_{i=1}^{16}(x_i-\bar x)(i-8.5)=-2.78\)，其中 \(x_i\) 为抽取的第 \(i\) 个零件的尺寸，\(i=1,2,\dots,16\).

\begin{enumerate}
\item 求 \((x_i,i)\) \((i=1,2,\dots,16)\) 的相关系数 \(r\)，并回答是否可以认为这一天生产的零件尺寸不随生产过程的进行而系统地变大或变小（若 \(|r|<0.25\)，则可以认为零件的尺寸不随生产过程的进行而系统地变大或变小）；
\item 一天内抽检零件中，如果出现了尺寸在 \((\bar x-3s,\bar x+3s)\) 之外的零件，就认为这条生产线在这一天的生产过程可能出现了异常情况，需对当天的生产过程进行检查.

\begin{enumerate}
\item 从这一天抽检的结果看，是否需对当天的生产过程进行检查？
\item 在 \((\bar x-3s,\bar x+3s)\) 之外的数据称为离群值，试剔除离群值，估计这条生产线当天生产的零件尺寸的均值与标准差（精确到 \(0.01\)）.
\end{enumerate}
\end{enumerate}

\bitmapfigure[max width=0.82\linewidth,max totalheight=4.8cm]{img/2017/national_paper_1_liberal/q19_fig1.png}

\end{problem}

\begin{solution}
（1）相关系数 \(r=\dfrac{\sum_{i=1}^{16}(x_i-\bar x)(i-8.5)}{\sqrt{\sum_{i=1}^{16}(x_i-\bar x)^2}\sqrt{\sum_{i=1}^{16}(i-8.5)^2}}=\dfrac{-2.78}{0.212\times\sqrt{16}\times18.439}\approx-0.18\).
因为 \(|r|<0.25\)，所以可以认为这一天生产的零件尺寸不随生产过程的进行而系统地变大或变小.

（2）（1）由 \(\bar x=9.97\)，\(s=0.212\) 得合格范围为
\((\bar x-3s,\bar x+3s)=(9.334,10.606)\).
第 \(13\) 号零件尺寸为 \(9.22\)，不在此范围内，所以需要检查.

（2）剔除离群值后，剩下 \(15\) 个数据的平均值为
\(\bar x'=\dfrac{16\times 9.97-9.22}{15}=10.02\).
又
\(\sum_{i=1}^{16}x_i^2=16\times0.212^2+16\times9.97^2=1591.134\).
故剔除离群值后样本方差为
\(s'^2=\dfrac{1591.134-9.22^2-15\times10.02^2}{15}\approx0.008\).
所以标准差
\(s'=\sqrt{0.008}\approx0.09\).
\end{solution}

\begin{problem}
设 \(A,B\) 为曲线 \(C:y=\dfrac{x^2}{4}\) 上两点，\(A\) 与 \(B\) 的横坐标之和为 \(4\).

\begin{enumerate}
\item 求直线 \(AB\) 的斜率；
\item 设 \(M\) 为曲线 \(C\) 上一点，\(C\) 在 \(M\) 处的切线与直线 \(AB\) 平行，且 \(AM\perp BM\)，求直线 \(AB\) 的方程.
\end{enumerate}

\end{problem}

\begin{solution}
（1）设 \(A\bigl(x_1,\dfrac{x_1^2}{4}\bigr)\)，\(B\bigl(x_2,\dfrac{x_2^2}{4}\bigr)\)，则 \(k_{AB}=\dfrac{\frac{x_1^2}{4}-\frac{x_2^2}{4}}{x_1-x_2}=\dfrac{x_1+x_2}{4}=1\).

（2）设直线 \(AB\) 为 \(y=x+t\)，代入 \(y=\dfrac{x^2}{4}\) 得
\(x^2-4x-4t=0\)，
故 \(x_1+x_2=4\)，\(x_1x_2=-4t\).

设 \(M(m,\dfrac{m^2}{4})\)，则 \(y'=\dfrac{x}{2}\)，由切线平行得
\(\dfrac{m}{2}=1\)，
所以 \(m=2\)，即 \(M(2,1)\).

由 \(AM\perp BM\) 得
\(\dfrac{\frac{x_1^2}{4}-1}{x_1-2}\cdot \dfrac{\frac{x_2^2}{4}-1}{x_2-2}=-1\)，

\(x_1x_2+2(x_1+x_2)+20=0\).

\begin{problem}
已知函数 \(f(x)=\e^x(\e^x-a)-a^2x\).

\begin{enumerate}
\item 讨论 \(f(x)\) 的单调性；
\item 若 \(f(x)\ge 0\)，求 \(a\) 的取值范围.
\end{enumerate}

\end{problem}

\begin{solution}
先求导：\(f'(x)=2\e^{2x}-a\e^x-a^2=(2\e^x+a)(\e^x-a)\).

（1）当 \(a=0\) 时，\(f'(x)=2\e^{2x}>0\)，所以 \(f(x)\) 在 \(\R\) 上单调递增.

当 \(a>0\) 时，\(2\e^x+a>0\)，令 \(f'(x)=0\) 得 \(x=\ln a\). 于是 \(x<\ln a\) 时 \(f'(x)<0\)，\(x>\ln a\) 时 \(f'(x)>0\).

当 \(a<0\) 时，\(\e^x-a>0\)，令 \(f'(x)=0\) 得 \(x=\ln(-a/2)\). 于是 \(x<\ln(-a/2)\) 时 \(f'(x)<0\)，\(x>\ln(-a/2)\) 时 \(f'(x)>0\).

（2）当 \(a=0\) 时，\(f(x)=\e^{2x}>0\) 恒成立.

当 \(a>0\) 时，最小值在 \(x=\ln a\) 处取得：\(f_{\min}=-a^2\ln a\ge 0\)，
故 \(0<a\le 1\).

当 \(a<0\) 时，最小值在 \(x=\ln(-a/2)\) 处取得：
\(f_{\min}=\dfrac{3a^2}{4}-a^2\ln\!\left(-\dfrac{a}{2}\right)\ge 0\)，
即
\(\ln\!\left(-\dfrac{a}{2}\right)\le\dfrac34\)，
所以
\(-\dfrac{a}{2}\le \e^{3/4}\)，所以 \(a\ge -2\e^{3/4}\).
综上，\(a\in\bigl[-2\e^{3/4},1\bigr]\).
\end{solution}

\begin{problem}
在直角坐标系 \(xOy\) 中，曲线 \(C\) 的参数方程为 \(x=3\cos\theta\)，\(y=\sin\theta\ (\theta\text{ 为参数})\)，直线 \(l\) 的参数方程为 \(x=a+4t\)，\(y=1-t\ (t\text{ 为参数})\).

\begin{enumerate}
\item 若 \(a=-1\)，求 \(C\) 与 \(l\) 的交点坐标；
\item 若 \(C\) 上的点到 \(l\) 距离的最大值为 \(\sqrt{17}\)，求 \(a\).
\end{enumerate}

\end{problem}

\begin{solution}
（1）曲线 \(C\) 化为普通方程得 \(\dfrac{x^2}{9}+y^2=1\).
当 \(a=-1\) 时，直线 \(l\) 化为一般方程 \(x+4y-3=0\).
联立
\examdisplaycases{}{
\dfrac{x^2}{9}+y^2=1,\\
x+4y-3=0
}
可得交点为 \((3,0)\) 和 \(\left(-\dfrac{21}{25},\dfrac{24}{25}\right)\).

（2）直线 \(l\) 化为一般方程为
\(x+4y-a-4=0\).
曲线 \(C\) 上的点可设为 \(P(3\cos\theta,\sin\theta)\)，则点 \(P\) 到直线 \(l\) 的距离为
 \(d=\dfrac{|3\cos\theta+4\sin\theta-a-4|}{\sqrt{17}}\).
由于 \(3\cos\theta+4\sin\theta\in[-5,5]\)，所以距离的最大值为
 \(\dfrac{\max\{|1-a|,|a+9|\}}{\sqrt{17}}\).
题设给出最大值为 \(\sqrt{17}\)，于是
 \(\max\{|1-a|,|a+9|\}=17\).
解得 \(a=8\) 或 \(a=-16\).
\end{solution}

\begin{problem}
已知函数 \(f(x)=-x^2+ax+4\)，\(g(x)=|x+1|+|x-1|\).

\begin{enumerate}
\item 当 \(a=1\) 时，求不等式 \(f(x)\ge g(x)\) 的解集；
\item 若不等式 \(f(x)\ge g(x)\) 的解集包含 \([-1,1]\)，求 \(a\) 的取值范围.
\end{enumerate}

\end{problem}

\begin{solution}
（1）当 \(a=1\) 时，\(f(x)=-x^2+x+4\).
而 \(g(x)=|x+1|+|x-1|=\begin{cases}
2x,&x>1,\\
2,&-1\le x\le 1,\\
-2x,&x<-1.
\end{cases}\)
当 \(x>1\) 时，解 \(-x^2+x+4\ge 2x\)，得
\(x\le \dfrac{\sqrt{17}-1}{2}\)，
故此时解为 \(\left(1,\dfrac{\sqrt{17}-1}{2}\right]\).

当 \(-1\le x\le 1\) 时，\(g(x)=2\)，而
 \(-x^2+x+4\ge 2\)
在 \([-1,1]\) 上恒成立.

当 \(x<-1\) 时，\(g(x)=-2x\)，而 \(f(x)\) 单调递增，且 \(f(-1)=2=g(-1)\)，故此时无解.

综上，解集为 \(\left[-1,\dfrac{\sqrt{17}-1}{2}\right]\).

（2）要使解集包含 \([-1,1]\)，只需
 \(-x^2+ax+4\ge 2\)（\(x\in[-1,1]\)）
恒成立，即
 \(x^2-ax-2\le 0\)（\(x\in[-1,1]\)）
恒成立. 因为 \(x^2-ax-2\) 是开口向上的二次函数，所以它在区间 \([-1,1]\) 上的最大值在端点处取得，从而只需端点满足
 \(1-a-2\le 0\)，\(1+a-2\le 0\)，
即
 \(-1\le a\le 1\).
\end{solution}
