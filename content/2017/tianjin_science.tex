\section{单选题}

\begin{problem}
设集合 \(A=\{1,2,4,6\}\)，\(B=\{2,4\}\)，\(C=\{x\in\R\mid -1\le x\le 5\}\)，则 \((A\cup B)\cap C=\)（\quad）

\choices
    {\(\{2\}\)}
    {\(\{1,2,4\}\)}
    {\(\{1,2,4,5\}\)}
    {\(\{x\in\R\mid -1\le x\le 5\}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
因为 \(B=\{2,4\}\subset A\)，所以 \(A\cup B=A=\{1,2,4,6\}\). 集合 \(C\) 包含 \(1\)，\(2\)，\(4\)，但不包含 \(6\)，故
\((A\cup B)\cap C=\{1,2,4\}\).

故选 B.
\end{solution}

\begin{problem}
设变量 \(x\)，\(y\) 满足约束条件
\(\begin{cases}
2x+y\ge 0,\\
x+2y-2\ge 0,\\
x\le 0,\\
y\le 3,
\end{cases}\)
则目标函数 \(z=x+y\) 的最大值为（\quad）

\choices
    {\(\dfrac{2}{3}\)}
    {\(1\)}
    {\(\dfrac{3}{2}\)}
    {\(3\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
由约束条件 \(x\leq 0\) 和 \(y\leq 3\)，得目标函数
\(z=x+y\leq 0+3=3\). 点 \((0,3)\) 满足
\(2x+y=3\geq 0\)，\(x+2y-2=4\geq 0\)，\(x=0\leq 0\)，\(y=3\leq 3\)，且此时 \(z=3\).

因此 \(z\) 的最大值为 \(3\)，故选 D.
\end{solution}

\begin{problem}
设 \(\theta\in\R\)，则“\(\left|\theta-\dfrac{\pi}{12}\right|<\dfrac{\pi}{12}\)”是“\(\sin\theta<\dfrac12\)”的（\quad）

\choices
    {充分而不必要条件}
    {必要而不充分条件}
    {充要条件}
    {既不充分也不必要条件}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
令命题 \(p\) 为 \(\left|\theta-\dfrac{\pi}{12}\right|<\dfrac{\pi}{12}\)，命题 \(q\) 为 \(\sin\theta<\dfrac12\). 由 \(p\) 得
\(0<\theta<\dfrac{\pi}{6}\)，从而 \(0<\sin\theta<\sin\dfrac{\pi}{6}=\dfrac12\)，所以 \(p\) 能推出 \(q\).

但取 \(\theta=-\dfrac{\pi}{6}\)，有 \(\sin\theta=-\dfrac12<\dfrac12\)，而
\(\left|\theta-\dfrac{\pi}{12}\right|=\dfrac{\pi}{4}>\dfrac{\pi}{12}\)，所以 \(q\) 不能推出 \(p\).

因此，\(p\) 是 \(q\) 的充分而不必要条件，故选 A.
\end{solution}

\begin{problem}
已知双曲线 \(\dfrac{x^2}{a^2}-\dfrac{y^2}{b^2}=1\)（\(a>0\)，\(b>0\)）的左焦点为 \(F\)，离心率为 \(\sqrt{2}\). 若经过 \(F\) 和 \(P(0,4)\) 两点的直线平行于双曲线的一条渐近线，则双曲线的方程为（\quad）

\choices
    {\(\dfrac{x^2}{4}-\dfrac{y^2}{4}=1\)}
    {\(\dfrac{x^2}{8}-\dfrac{y^2}{8}=1\)}
    {\(\dfrac{x^2}{4}-\dfrac{y^2}{8}=1\)}
    {\(\dfrac{x^2}{8}-\dfrac{y^2}{4}=1\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
设双曲线的半焦距为 \(c\)，则 \(c^2=a^2+b^2\)，且由离心率得
\(\dfrac ca=\sqrt2\)，所以 \(c^2=2a^2\)，进而 \(b^2=a^2\). 左焦点为
\(F=(-c,0)=(-\sqrt2a,0)\)，故直线 \(FP\) 的斜率为
\(\dfrac{4}{\sqrt2a}=\dfrac{2\sqrt2}{a}\).

双曲线的渐近线斜率为 \(\pm\dfrac ba=\pm1\). 由于直线 \(FP\) 斜率为正，题设表明
\(\dfrac{2\sqrt2}{a}=1\)，所以 \(a=2\sqrt2\)，即 \(a^2=b^2=8\).

因此双曲线方程为 \(\dfrac{x^2}{8}-\dfrac{y^2}{8}=1\)，故选 B.
\end{solution}

\begin{problem}
已知奇函数 \(f(x)\) 在 \(\R\) 上是增函数，\(g(x)=xf(x)\). 若 \(a=g(-\log_2 5.1)\)，\(b=g(2^{0.8})\)，\(c=g(3)\)，则 \(a\)，\(b\)，\(c\) 的大小关系为（\quad）

\choices
    {\(a<b<c\)}
    {\(c<b<a\)}
    {\(b<a<c\)}
    {\(b<c<a\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
由 \(f(x)\) 为奇函数且在 \(\R\) 上递增，知 \(f(0)=0\)，并且当 \(x>0\) 时 \(f(x)>0\). 对 \(0\leq u<v\)，有
\(g(v)-g(u)=v[f(v)-f(u)]+(v-u)f(u)>0\)，所以 \(g\) 在 \([0,+\infty)\) 上递增.

又由奇偶性，\(g(-x)=(-x)f(-x)=xf(x)=g(x)\)，所以 \(g\) 是偶函数.由于
\(2^{0.8}<2<\log_2 5.1<3\)，故
\(g(2^{0.8})<g(\log_2 5.1)<g(3)\). 因此
\(b<a<c\)，故选 C.
\end{solution}

\begin{problem}
设函数 \(f(x)=2\sin(\omega x+\varphi)\)，\(x\in\R\)，其中 \(\omega>0\)，\(|\varphi|<\pi\). 若 \(f\left(\dfrac{5\pi}{8}\right)=2\)，\(f\left(\dfrac{11\pi}{8}\right)=0\)，且 \(f(x)\) 的最小正周期大于 \(2\pi\)，则（\quad）

\choices
    {\(\omega=\dfrac{2}{3}\)，\(\varphi=\dfrac{\pi}{12}\)}
    {\(\omega=\dfrac{2}{3}\)，\(\varphi=-\dfrac{11\pi}{12}\)}
    {\(\omega=\dfrac{1}{3}\)，\(\varphi=-\dfrac{11\pi}{24}\)}
    {\(\omega=\dfrac{1}{3}\)，\(\varphi=\dfrac{7\pi}{24}\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
由 \(f\left(\dfrac{5\pi}{8}\right)=2\)，得
\(\sin\left(\dfrac{5\pi\omega}{8}+\varphi\right)=1\)，所以存在 \(k\in\Z\)，使
\(\dfrac{5\pi\omega}{8}+\varphi=\dfrac\pi2+2k\pi\). 由
\(f\left(\dfrac{11\pi}{8}\right)=0\)，得存在 \(l\in\Z\)，使
\(\dfrac{11\pi\omega}{8}+\varphi=l\pi\). 两式相减，得到
\(\omega=\dfrac{2(2l-4k-1)}{3}\).

函数的最小正周期为 \(\dfrac{2\pi}{\omega}>2\pi\)，故 \(0<\omega<1\). 于是奇整数 \(2l-4k-1\) 只能取 \(1\)，从而 \(\omega=\dfrac23\). 代入第一式得
\(\varphi=\dfrac\pi{12}+2k\pi\). 结合 \(|\varphi|<\pi\)，可知 \(k=0\)，即 \(\varphi=\dfrac\pi{12}\).

故选 A.
\end{solution}

\begin{problem}
已知函数 \(f(x)=
\begin{cases}
x^2-x+3,&x\le 1,\\
x+\dfrac{2}{x},&x>1,
\end{cases}\)
设 \(a\in\R\)，若关于 \(x\) 的不等式 \(f(x)\ge \left|\dfrac{x}{2}+a\right|\) 在 \(\R\) 上恒成立，则 \(a\) 的取值范围是（\quad）

\choices
    {\(\left[-\dfrac{47}{16},2\right]\)}
    {\(\left[-\dfrac{47}{16},\dfrac{39}{16}\right]\)}
    {\(\left[-2\sqrt{3},2\right]\)}
    {\(\left[-2\sqrt{3},\dfrac{39}{16}\right]\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
不等式 \(f(x)\geq\left|\dfrac x2+a\right|\) 等价于
\(-f(x)-\dfrac x2\leq a\leq f(x)-\dfrac x2\). 当 \(x\leq1\) 时，
\(f(x)+\frac x2=\left(x-\frac14\right)^2+\frac{47}{16}\)，\(f(x)-\frac x2=\left(x-\frac34\right)^2+\frac{39}{16}\).
因此在此区间上，前者不小于 \(\dfrac{47}{16}\)，后者不小于 \(\dfrac{39}{16}\).

当 \(x>1\) 时，利用基本不等式得
\(f(x)+\dfrac x2=\dfrac{3x}{2}+\dfrac2x\geq2\sqrt3>\dfrac{47}{16}\)，以及
\(f(x)-\dfrac x2=\dfrac x2+\dfrac2x\geq2\). 所以对所有实数 \(x\)，都有
\(f(x)+\dfrac x2\geq\dfrac{47}{16}\)，\(f(x)-\dfrac x2\geq2\).

于是 \(-\dfrac{47}{16}\leq a\leq2\) 时原不等式恒成立.反之，取 \(x=\dfrac14\) 得
\(a\geq-\dfrac{47}{16}\)，取 \(x=2\) 得 \(a\leq2\). 故 \(a\) 的取值范围为
\(\left[-\dfrac{47}{16},2\right]\)，故选 A.
\end{solution}

\section{填空题}

\begin{problem}
已知 \(a\in\R\)，\(\i\) 为虚数单位，若 \(\dfrac{a-\i}{2+\i}\) 为实数，则 \(a\) 的值为\fillinblank{}.
\end{problem}

\begin{answer}
\(-2\)
\end{answer}

\begin{solution}
将分式分母有理化，得
\[
\frac{a-\i}{2+\i}=\frac{(a-\i)(2-\i)}{(2+\i)(2-\i)}
 =\frac{2a-1-(a+2)\i}{5}
\]
该复数为实数，当且仅当虚部为零，即 \(a+2=0\)，所以 \(a=-2\).
\end{solution}

\begin{problem}
已知一个正方体的所有顶点在一个球面上，若这个正方体的表面积为 \(18\)，则这个球的体积为\fillinblank{}.
\end{problem}

\begin{answer}
\(\dfrac{9\pi}{2}\)
\end{answer}

\begin{solution}
设正方体棱长为 \(s\). 由表面积得 \(6s^2=18\)，所以 \(s^2=3\). 正方体的外接球半径等于体对角线的一半，即
\(R=\dfrac{s\sqrt3}{2}=\dfrac32\). 因此球的体积为
\(V=\dfrac43\pi R^3=\dfrac43\pi\left(\dfrac32\right)^3=\dfrac{9\pi}{2}\).
\end{solution}

\begin{problem}
在极坐标系中，直线 \(4\rho\cos\left(\theta-\dfrac{\pi}{6}\right)+1=0\) 与圆 \(\rho=2\sin\theta\) 的公共点的个数为\fillinblank{}.
\end{problem}

\begin{answer}
\(2\)
\end{answer}

\begin{solution}
令直角坐标系中的点满足 \(x=\rho\cos\theta\)，\(y=\rho\sin\theta\). 直线方程化为
\(4\rho\left(\frac{\sqrt3}{2}\cos\theta+\frac12\sin\theta\right)+1=0\)，\(2\sqrt3x+2y+1=0\).
圆方程 \(\rho=2\sin\theta\) 化为 \(x^2+y^2=2y\)，即圆心为 \((0,1)\)、半径为 \(1\) 的圆.

圆心到直线的距离为 \(\dfrac{|2+1|}{\sqrt{(2\sqrt3)^2+2^2}}=\dfrac34<1\)，所以直线与圆相交于两个点.原点不在该直线上，极坐标表示不会造成交点数的重复或遗漏，故公共点的个数为 \(2\).
\end{solution}

\begin{problem}
若 \(a\)，\(b\in\R\)，\(ab>0\)，则 \(\dfrac{a^4+4b^4+1}{ab}\) 的最小值为\fillinblank{}.
\end{problem}

\begin{answer}
\(4\)
\end{answer}

\begin{solution}
因为 \(ab>0\)，令 \(u=|a|\)，\(v=|b|\)，则 \(u\)，\(v>0\)，且 \(uv=ab\). 记 \(t=uv\)，则
\[
u^4+4v^4+1=(u^2-2v^2)^2+4t^2+1\geq4t^2+1
\]
所以
\[
\frac{a^4+4b^4+1}{ab}\geq4t+\frac1t\geq4
\]
其中最后一个不等式由基本不等式得到.取 \(t=\dfrac12\) 且 \(u^2=2v^2\) 时等号可以同时成立，因此最小值为 \(4\).
\end{solution}

\begin{problem}
在 \(\triangle ABC\) 中，\(\angle A=60^\circ\)，\(AB=3\)，\(AC=2\). 若 \(\overrightarrow{BD}=2\overrightarrow{DC}\)，\(\overrightarrow{AE}=\lambda\overrightarrow{AC}-\overrightarrow{AB}\)（\(\lambda\in\R\)），且 \(\overrightarrow{AD}\cdot\overrightarrow{AE}=-4\)，则 \(\lambda\) 的值为\fillinblank{}.
\end{problem}

\begin{answer}
\(\dfrac{3}{11}\)
\end{answer}

\begin{solution}
取 \(A\) 为原点，记 \(\overrightarrow{AB}=\bs u\)，\(\overrightarrow{AC}=\bs v\). 由题意
\(\bs u\cdot\bs u=9\)，\(\bs v\cdot\bs v=4\)，\(\bs u\cdot\bs v=3\times2\cos60^\circ=3\).

由 \(\overrightarrow{BD}=2\overrightarrow{DC}\)，得
\(\overrightarrow{AD}=\dfrac13\bs u+\dfrac23\bs v\). 又
\(\overrightarrow{AE}=\lambda\bs v-\bs u\)，所以
\[
\begin{aligned}
\overrightarrow{AD}\cdot\overrightarrow{AE}
&=\frac13(\bs u+2\bs v)\cdot(\lambda\bs v-\bs u)\\
&=\frac13(3\lambda-9+8\lambda-6)=\frac{11\lambda-15}{3}.
\end{aligned}
\]
由题设该数量积等于 \(-4\)，故 \(\dfrac{11\lambda-15}{3}=-4\)，解得
\(\lambda=\dfrac3{11}\).
\end{solution}

\begin{problem}
用数字 \(1\)，\(2\)，\(3\)，\(4\)，\(5\)，\(6\)，\(7\)，\(8\)，\(9\) 组成没有重复数字，且至多有一个数字是偶数的四位数，这样的四位数一共有\fillinblank{} 个. （用数字作答）
\end{problem}

\begin{answer}
\(1080\)
\end{answer}

\begin{solution}
所组成的四位数首位不为零，而给出的数字中没有零，因此只需按偶数个数分类.

若四位数中没有偶数，则四个数字都从 \(1\)，\(3\)，\(5\)，\(7\)，\(9\) 中选取并排列，共有
\(\mathrm A_5^4=5\times4\times3\times2=120\) 个.

若恰有一个偶数，先选偶数的位置、偶数和三个奇数，分别有 \(4\)，\(4\)，\(\mathrm A_5^3=5\times4\times3=60\) 种，因此共有
\(4\times4\times60=960\) 个.

总数为 \(120+960=1080\) 个.
\end{solution}

\begin{problem}
在 \(\triangle ABC\) 中，内角 \(A\)，\(B\)，\(C\) 所对的边分别为 \(a\)，\(b\)，\(c\). 已知 \(a>b\)，\(a=5\)，\(c=6\)，\(\sin B=\dfrac{3}{5}\).

\begin{enumerate}
\item 求 \(b\) 和 \(\sin A\) 的值；
\item 求 \(\sin\left( 2A+\dfrac{\pi}{4} \right)\) 的值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 若角 \(B\) 为钝角，则 \(A<B\)，这与 \(a>b\) 矛盾，故 \(B\) 为锐角，从而 \(\cos B=\dfrac45\). 由余弦定理，
\(b^2=a^2+c^2-2ac\cos B=25+36-2\times5\times6\times\dfrac45=13\)，所以 \(b=\sqrt{13}\).
由正弦定理，\(\dfrac{b}{\sin B}=\dfrac{a}{\sin A}\)，得
\(\sin A=\dfrac{a\sin B}{b}=\dfrac{5\times\frac35}{\sqrt{13}}=\dfrac{3\sqrt{13}}{13}\).

\item 由余弦定理，
\(\cos A=\dfrac{b^2+c^2-a^2}{2bc}=\dfrac{13+36-25}{12\sqrt{13}}=\dfrac{2\sqrt{13}}{13}\). 因此
\(\sin2A=2\sin A\cos A=\dfrac{12}{13}\)，\(\cos2A=1-2\sin^2A=1-\dfrac{18}{13}=-\dfrac5{13}\).
于是
\[
\sin\left(2A+\frac\pi4\right)=\sin2A\cos\frac\pi4+\cos2A\sin\frac\pi4
=\frac{1}{\sqrt2}\left(\frac{12}{13}-\frac5{13}\right)=\frac{7\sqrt2}{26}
\]
\end{enumerate}
\end{solution}

\begin{problem}
从甲地到乙地要经过 \(3\) 个十字路口，设各路口信号灯工作相互独立，且在各路口遇到红灯的概率分别为 \(\dfrac{1}{2}\)，\(\dfrac{1}{3}\)，\(\dfrac{1}{4}\).

\begin{enumerate}
\item 设 \(X\) 表示一辆车从甲地到乙地遇到红灯的个数，求随机变量 \(X\) 的分布列和数学期望；
\item 若有 \(2\) 辆车独立地从甲地到乙地，求这 \(2\) 辆车共遇到 \(1\) 个红灯的概率.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 记三个路口遇到红灯的事件分别为 \(R_1\)，\(R_2\)，\(R_3\). 由独立性，\(X\) 的可能取值为 \(0\)，\(1\)，\(2\)，\(3\)，且
\(P(X=0)=\dfrac12\times\dfrac23\times\dfrac34=\dfrac14\)，
\(P(X=3)=\dfrac12\times\dfrac13\times\dfrac14=\dfrac1{24}\).

恰遇到一个红灯时，分别在第一个、第二个、第三个路口遇到红灯，所以
\[
P(X=1)=\frac12\times\frac23\times\frac34
+\frac12\times\frac13\times\frac34
+\frac12\times\frac23\times\frac14=\frac{11}{24}
\]
恰遇到两个红灯时，分别漏过第三个、第二个、第一个路口，所以
\[
P(X=2)=\frac12\times\frac13\times\frac34
+\frac12\times\frac23\times\frac14
+\frac12\times\frac13\times\frac14=\frac14
\]
因此分布列为
\(P(X=0)=\dfrac14\)，\(P(X=1)=\dfrac{11}{24}\)，\(P(X=2)=\dfrac14\)，\(P(X=3)=\dfrac1{24}\).
数学期望为
\[
E(X)=0\times\frac14+1\times\frac{11}{24}+2\times\frac14+3\times\frac1{24}=\frac{13}{12}
\]

\item 一辆车遇到一个红灯的概率为 \(\dfrac{11}{24}\)，遇到零个红灯的概率为 \(\dfrac14\). 两辆车共遇到一个红灯，只有一辆车遇到一个红灯、另一辆车遇到零个红灯，且两种次序互斥，故所求概率为
\(2\times\dfrac{11}{24}\times\dfrac14=\dfrac{11}{48}\).
\end{enumerate}
\end{solution}

\begin{problem}
如图，在三棱锥 \(P-ABC\) 中，\(PA\perp\) 底面 \(ABC\)，\(\angle BAC=90^\circ\). 点 \(D\)，\(E\)，\(N\) 分别为棱 \(PA\)，\(PC\)，\(BC\) 的中点，\(M\) 是线段 \(AD\) 的中点，\(PA=AC=4\)，\(AB=2\).

\begin{enumerate}
\item 求证：\(MN\parallel\) 平面 \(BDE\)；
\item 求二面角 \(C-EM-N\) 的正弦值；
\item 已知点 \(H\) 在棱 \(PA\) 上，且直线 \(NH\) 与直线 \(BE\) 所成角的余弦值为 \(\dfrac{\sqrt{7}}{21}\)，求线段 \(AH\) 的长.
\end{enumerate}

\bitmapfigure[max width=6.2cm]{img/2017/tianjin_science/q17_fig1.png}
\end{problem}

\begin{solution}
建立空间直角坐标系，取 \(A\) 为原点，\(AB\) 为 \(x\) 轴，\(AC\) 为 \(y\) 轴，\(AP\) 为 \(z\) 轴.则
\(B=(2,0,0)\)，\(C=(0,4,0)\)，\(P=(0,0,4)\)，
\(D=(0,0,2)\)，\(E=(0,2,2)\)，\(N=(1,2,0)\)，\(M=(0,0,1)\).

由 \(B\)，\(D\)，\(E\) 三点可得平面 \(BDE\) 的方程为 \(x+z=2\)，而 \(M=(0,0,1)\) 不在该平面内.

\begin{enumerate}
\item 有 \(\overrightarrow{MN}=(1,2,-1)\)，\(\overrightarrow{BD}=(-2,0,2)\)，\(\overrightarrow{BE}=(-2,2,2)\)，并且
\(\overrightarrow{MN}=-\dfrac32\overrightarrow{BD}+\overrightarrow{BE}\). 由于 \(\overrightarrow{BD}\) 与 \(\overrightarrow{BE}\) 是平面 \(BDE\) 内两条相交直线的方向向量，故 \(MN\) 的方向向量平行于平面 \(BDE\)，从而 \(MN\parallel\) 平面 \(BDE\).

\item 平面 \(CEM\) 的方程为 \(x=0\)，可取法向量 \(\bs n_1=(1,0,0)\). 平面 \(EMN\) 内有
\(\overrightarrow{EM}=(0,-2,-1)\)，\(\overrightarrow{EN}=(1,0,-2)\)，故可取法向量
\(\bs n_2=\overrightarrow{EM}\times\overrightarrow{EN}=(4,-1,2)\). 设二面角 \(C-EM-N\) 的平面角为 \(\alpha\)，则
\(\cos\alpha=\dfrac{|\bs n_1\cdot\bs n_2|}{|\bs n_1||\bs n_2|}=\dfrac4{\sqrt{21}}\). 因此
\(\sin\alpha=\sqrt{1-\dfrac{16}{21}}=\dfrac{\sqrt{105}}{21}\).

\item 设 \(H=(0,0,h)\)，其中 \(0\leq h\leq4\)，则 \(AH=h\). 取
\(\overrightarrow{NH}=(-1,-2,h)\)，\(\overrightarrow{BE}=(-2,2,2)\)，有
\(\overrightarrow{NH}\cdot\overrightarrow{BE}=2h-2\)，
\(|\overrightarrow{NH}|=\sqrt{h^2+5}\)，\(|\overrightarrow{BE}|=2\sqrt3\). 直线所成的角取锐角，故
\[
\frac{|h-1|}{\sqrt{3(h^2+5)}}=\frac{\sqrt7}{21}
\]
两边平方并化简，得 \(21(h-1)^2=h^2+5\)，即
\(10h^2-21h+8=0\)，解得 \(h=\dfrac85\) 或 \(h=\dfrac12\). 二者均在 \([0,4]\) 内，故 \(AH=\dfrac85\) 或 \(\dfrac12\).
\end{enumerate}
\end{solution}

\begin{problem}
已知 \(\{a_n\}\) 为等差数列，前 \(n\) 项和为 \(S_n\)（\(n\in\N^*\)），\(\{b_n\}\) 是首项为 \(2\) 的等比数列，且公比大于 \(0\)，\(b_2+b_3=12\)，\(b_3=a_4-2a_1\)，\(S_{11}=11b_4\).

\begin{enumerate}
\item 求 \(\{a_n\}\) 和 \(\{b_n\}\) 的通项公式；
\item 求数列 \(\{a_{2n}b_{2n-1}\}\) 的前 \(n\) 项和（\(n\in\N^*\)）.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设等比数列 \(\{b_n\}\) 的公比为 \(q\)，则 \(q>0\)，且
\(b_n=2q^{n-1}\). 由 \(b_2+b_3=12\)，得 \(2q+2q^2=12\)，即
\(q^2+q-6=0\). 因为 \(q>0\)，所以 \(q=2\)，从而 \(b_n=2^n\).

设等差数列 \(\{a_n\}\) 的首项为 \(a_1\)，公差为 \(d\). 由 \(b_3=a_4-2a_1\) 得
\(8=(a_1+3d)-2a_1=3d-a_1\)，即 \(a_1=3d-8\). 又
\(S_{11}=\dfrac{11}{2}(2a_1+10d)=11(a_1+5d)=11b_4=176\)，所以 \(a_1+5d=16\). 联立解得 \(d=3\)，\(a_1=1\)，故 \(a_n=3n-2\).

\item 记所求前 \(n\) 项和为 \(T_n\). 由通项公式，
\(a_{2k}b_{2k-1}=(6k-2)2^{2k-1}=4(3k-1)4^{k-1}\)，所以
\[
T_n=4\sum_{k=1}^n(3k-1)4^{k-1}
\]
利用
\(\displaystyle\sum_{k=1}^n4^{k-1}=\dfrac{4^n-1}{3}\) 及
\(\displaystyle\sum_{k=1}^nk4^{k-1}=\dfrac{1-(n+1)4^n+n4^{n+1}}{9}\)，得
\[
T_n=\frac{3n-2}{3}\,4^{n+1}+\frac83
\]
\end{enumerate}
\end{solution}

\begin{problem}
设椭圆 \(\dfrac{x^2}{a^2}+\dfrac{y^2}{b^2}=1\)（\(a>b>0\)）的左焦点为 \(F\)，右顶点为 \(A\)，离心率为 \(\dfrac{1}{2}\). 已知 \(A\) 是抛物线 \(y^2=2px\)（\(p>0\)）的焦点，\(F\) 到抛物线的准线 \(l\) 的距离为 \(\dfrac{1}{2}\).

\begin{enumerate}
\item 求椭圆的方程和抛物线的方程；
\item 设 \(l\) 上两点 \(P\)，\(Q\) 关于 \(x\) 轴对称，直线 \(AP\) 与椭圆相交于点 \(B\)（\(B\) 异于点 \(A\)），直线 \(BQ\) 与 \(x\) 轴相交于点 \(D\). 若 \(\triangle APD\) 的面积为 \(\dfrac{\sqrt{6}}{2}\)，求直线 \(AP\) 的方程.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设椭圆的半焦距为 \(c\). 由离心率得 \(c=\dfrac a2\)，所以
\(b^2=a^2-c^2=\dfrac{3a^2}{4}\). 椭圆的右顶点为 \(A=(a,0)\). 抛物线
\(y^2=2px\) 的焦点为 \(\left(\dfrac p2,0\right)\)，准线为 \(x=-\dfrac p2\).
因 \(A\) 是抛物线的焦点，得 \(\dfrac p2=a\). 又左焦点
\(F=\left(-\dfrac a2,0\right)\) 到准线的距离为 \(\dfrac a2=\dfrac12\)，故 \(a=1\).

于是 \(c=\dfrac12\)，\(b^2=\dfrac34\)，椭圆方程为
\(x^2+\dfrac{4y^2}{3}=1\)，且 \(p=2\)，抛物线方程为 \(y^2=4x\).

\item 设 \(P=(-1,t)\)，\(Q=(-1,-t)\)，其中 \(t\neq0\). 直线 \(AP\) 的方程为
\(y=\dfrac t2(1-x)\). 将其代入椭圆方程，得
\(3x^2+t^2(1-x)^2=3\). 已知 \(x=1\) 对应于点 \(A\)，另一个交点 \(B\) 的横坐标为
\(x_B=\dfrac{t^2-3}{t^2+3}\)，从而 \(y_B=\dfrac{3t}{t^2+3}\).

直线 \(BQ\) 与 \(x\) 轴交于 \(D\). 由两点式求得
\[
x_D=x_B-\frac{y_B(x_B+1)}{y_B+t}=\frac{t^2-6}{t^2+6}
\]
所以 \(AD=1-x_D=\dfrac{12}{t^2+6}\). 因此
\[
S_{\triangle APD}=\frac12\,AD\,|t|=\frac{6|t|}{t^2+6}=\frac{\sqrt6}{2}
\]
令 \(s=|t|>0\)，则 \(12s=\sqrt6(s^2+6)\)，即
\(s^2-2\sqrt6s+6=0\)，所以 \(s=\sqrt6\). 于是直线 \(AP\) 的斜率为
\(-\dfrac t2=\pm\dfrac{\sqrt6}{2}\)，故直线方程为
\(3x+\sqrt6y-3=0\) 或 \(3x-\sqrt6y-3=0\).
\end{enumerate}
\end{solution}

\begin{problem}
设 \(a\in\Z\). 已知定义在 \(\R\) 上的函数 \(f(x)=2x^4+3x^3-3x^2-6x+a\) 在区间 \((1,2)\) 内有一个零点 \(x_0\)，\(g(x)\) 为 \(f(x)\) 的导函数.

\begin{enumerate}
\item 求 \(g(x)\) 的单调区间；
\item 设 \(m\in[1,x_0)\cup(x_0,2]\)，函数 \(h(x)=g(x)(m-x_0)-f(m)\)，求证：\(h(m)h(x_0)<0\)；
\item 求证：存在大于 \(0\) 的常数 \(A\)，使得对于任意的正整数 \(p\)，\(q\)，且 \(\dfrac{p}{q}\in[1,x_0)\cup(x_0,2]\)，满足 \(\left|\dfrac{p}{q}-x_0\right|\ge \dfrac{1}{Aq^4}\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由 \(g(x)=f'(x)\)，得 \(g(x)=8x^3+9x^2-6x-6\)，从而
\(g'(x)=24x^2+18x-6=6(4x-1)(x+1)\). 因此 \(g'(x)>0\) 在
\((-\infty,-1)\) 和 \(\left(\dfrac14,+\infty\right)\) 上成立，\(g'(x)<0\) 在
\(\left(-1,\dfrac14\right)\) 上成立.所以 \(g(x)\) 在
\((-\infty,-1)\)，\(\left(\dfrac14,+\infty\right)\) 上单调递增，在
\(\left(-1,\dfrac14\right)\) 上单调递减.

\item 当 \(1\leq x\leq2\) 时，\(g'(x)>0\)，所以 \(g\) 在 \([1,2]\) 上严格递增.由 \(f(x_0)=0\) 及拉格朗日中值定理，存在严格介于 \(m\) 与 \(x_0\) 之间的 \(\xi\)，使
\(f(m)=g(\xi)(m-x_0)\). 于是
\(h(m)=[g(m)-g(\xi)](m-x_0)\)，
\(h(x_0)=[g(x_0)-g(\xi)](m-x_0)\).

若 \(m>x_0\)，则 \(x_0<\xi<m\)，所以 \(g(x_0)<g(\xi)<g(m)\)，从而
\(h(m)>0\)，\(h(x_0)<0\). 若 \(m<x_0\)，则 \(m<\xi<x_0\)，所以
\(g(m)<g(\xi)<g(x_0)\)，而 \(m-x_0<0\)，从而 \(h(m)>0\)，\(h(x_0)<0\). 两种情形均有
\(h(m)h(x_0)<0\).

\item 先注意到 \(g(1)=5>0\)，且 \(g\) 在 \([1,2]\) 上递增，所以 \(g(x)>0\) 在该区间上成立，进而 \(f\) 在 \([1,2]\) 上严格递增.故除了 \(x_0\) 外，区间 \([1,2]\) 内没有其他零点.

令 \(r=\dfrac pq\). 由题设，\(r\in[1,x_0)\cup(x_0,2]\)，所以 \(r\neq x_0\)，从而 \(f(r)\neq0\). 又
\[
q^4f\left(\frac pq\right)=2p^4+3p^3q-3p^2q^2-6pq^3+aq^4
\]
是非零整数，因此 \(\left|f\left(\dfrac pq\right)\right|\geq\dfrac1{q^4}\).

再次由中值定理，存在 \(\eta\) 介于 \(r\) 与 \(x_0\) 之间，使
\(\left|f(r)\right|=g(\eta)|r-x_0|\). 因为 \(g\) 在 \([1,2]\) 上递增，且
\(g(2)=64+36-12-6=82\)，所以
\[
\frac1{q^4}\leq\left|f(r)\right|\leq82\left|r-x_0\right|
\]
因此取 \(A=82\)，便有
\(\left|\dfrac pq-x_0\right|\geq\dfrac1{82q^4}\)，命题得证.
\end{enumerate}
\end{solution}
