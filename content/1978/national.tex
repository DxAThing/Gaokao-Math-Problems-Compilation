\section{解答题}

\begin{problem}
\begin{enumerate}
\item 分解因式： \(x^{2} - 4xy + 4y^{2} - 4z^{2}\).
\item 已知正方形的边长为 \(a\)，求侧面积等于这个正方形的面积，高等于这个正方形边长的直圆柱体的体积.
\item 求函数 \(y = \sqrt{\lg(2 + x)}\) 的定义域.
\item 不查表求 \(\cos80^{\circ}\cos35^{\circ}+\cos10^{\circ}\cos55^{\circ}\) 的值.
\item 化简： \(\left(\frac{1}{4}\right)^{-\frac{1}{2}} \cdot \frac{\left(\sqrt{4ab^{-1}}\right)^{3}}{(0.1)^{-2}\left(a^{3}b^{-4}\right)^{\frac{1}{2}}}\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item \((x-2y-2z)(x-2y+2z)\).
\item \(\dfrac{a^{3}}{4\pi}\).
\item \(x\geqslant-1\).
\item \(\dfrac{\sqrt{2}}{2}\).
\item \(\dfrac{4}{25}b^{\frac{1}{2}}\).
\end{enumerate}
\end{solution}

\begin{problem}
已知方程 \(kx^{2} + y^{2} = 4\)，其中 \(k\) 为实数. 对于不同范围的 \(k\) 值，分别指出方程所代表图形的类型，并画出显示其数量特征的草图.
\end{problem}

\begin{solution}
\begin{enumerate}
\item 当 \(k>1\) 时，是以原点为中心、长轴在 \(y\) 轴上的椭圆，半长轴为 \(2\)，半短轴为 \(\dfrac{2}{\sqrt{k}}\)；当 \(k=1\) 时，是半径为 \(2\) 的圆；当 \(0<k<1\) 时，是以原点为中心、长轴在 \(x\) 轴上的椭圆，半长轴为 \(\dfrac{2}{\sqrt{k}}\)，半短轴为 \(2\).
\item 当 \(k=0\) 时，是两条平行于 \(x\) 轴的直线 \(y=2\) 和 \(y=-2\).
\item 当 \(k<0\) 时，是以原点为中心、实轴在 \(y\) 轴上的双曲线，方程可写为 \(\dfrac{y^{2}}{4}-\dfrac{x^{2}}{4/\lvert k\rvert}=1\).
\end{enumerate}
\end{solution}

\begin{problem}
已知 \(\log_{18}9 = a\)，\(18^{b} = 5\)，求 \(\log_{36}45\).
\end{problem}

\begin{solution}
\(\log_{36}45=\dfrac{a+b}{2-a}\).
\end{solution}

\section{选做题}

\begin{problem}
已知 \(\triangle ABC\) 的三个内角的大小成等差数列，\(\tan A \tan C = 2 + \sqrt{3}\)，求角 \(A\)，\(B\)，\(C\) 的大小. 又已知顶点 \(C\) 的对边 \(c\) 上的高等于 \(4\sqrt{3}\)，求三角形各边 \(a\)，\(b\)，\(c\) 的长. （提示：必要时可验证 \((1 + \sqrt{3})^2 = 4 + 2\sqrt{3}\)）
\end{problem}

\begin{solution}
若约定 \(A<C\)，则 \(A=45^{\circ}\)，\(B=60^{\circ}\)，\(C=75^{\circ}\)；\(a=8\)，\(b=4\sqrt{6}\)，\(c=4\sqrt{3}+4\). 原题未规定 \(A<C\)，若不作此约定，则 \(A\)、\(C\) 以及 \(a\)、\(c\) 可同时互换.
\end{solution}

\begin{problem}
已知 \(\alpha\)，\(\beta\) 为锐角，\(3\sin^2\alpha + 2\sin^2\beta = 1\)，\(3\sin 2\alpha - 2\sin 2\beta = 0\). 求证： \(\alpha + 2\beta = \frac{\pi}{2}\).
\end{problem}

\begin{solution}
结论为 \(\alpha+2\beta=\dfrac{\pi}{2}\).
\end{solution}

\section{解答题}

\begin{problem}
已知函数 \(y = x^{2} + (2m + 1)x + m^{2} - 1\)（\(m\) 为实数）.

\begin{enumerate}
\item \(m\) 是什么数值时，\(y\) 的极值是 0？
\item 求证：不论 \(m\) 是什么数值，函数图象（即抛物线）的顶点都在同一条直线 \(L_{1}\) 上，画出 \(m=-1\)，\(m=0\)，\(m=1\) 时抛物线的草图，来检验这个结论；
\item 平行于 \(L_{1}\) 的直线中，哪些与抛物线相交，哪些不相交？ 求证：任一条平行于 \(L_{1}\) 而与抛物线相交的直线，被各抛物线截出的线段都相等.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item \(m=-\dfrac{5}{4}\).
\item 各抛物线的顶点都在直线 \(L_{1}\) 上，且直线 \(L_{1}\) 的方程为 \(x-y=\dfrac{3}{4}\).
\item 设直线为 \(x-y=a\). 当 \(a\leqslant1\) 时相交，当 \(a>1\) 时不相交；相交时，被抛物线截出的线段长为 \(2\sqrt{2(1-a)}\)，与 \(m\) 无关.
\end{enumerate}
\end{solution}
