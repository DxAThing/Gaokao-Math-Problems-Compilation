\section{单选题}

\begin{problem}
设集合 \(M = \{m \in \Z \mid -3 < m < 2\}\)，\(N = \{n \in \Z \mid -1 \leqslant n \leqslant 3\}\)，则 \(M \cap N =\)（\quad）

\choices
    {\(\{0,1\}\)}
    {\(\{-1, 0, 1\}\)}
    {\(\{0, 1, 2\}\)}
    {\(\{-1, 0, 1, 2\}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
由 \(-3<m<2\) 且 \(m\in\Z\)，得 \(M=\{-2,-1,0,1\}\)；由 \(-1\leqslant n\leqslant3\) 且 \(n\in\Z\)，得 \(N=\{-1,0,1,2,3\}\). 因此 \(M\cap N=\{-1,0,1\}\)，故选 B.
\end{solution}

\begin{problem}
设 \(a\)，\(b \in \R\) 且 \(b \neq 0\)，若复数 \((a + b\i)^3\) 是实数，则（\quad）

\choices
    {\( b^{2}=3a^{2} \)}
    {\(a^2 = 3b^2\)}
    {\( b^{2}=9a^{2} \)}
    {\(a^2 = 9b^2\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
展开得 \((a+b\i)^3=a^3-3ab^2+(3a^2b-b^3)\i\). 该复数为实数，所以虚部为零，即 \(3a^2b-b^3=b(3a^2-b^2)=0\). 由 \(b\neq0\)，得 \(b^2=3a^2\)，故选 A.
\end{solution}

\begin{problem}
函数 \( f(x) = \frac{1}{x} - x \) 的图象关于（\quad）

\choices
    {\(y\) 轴对称}
    {直线 \( y = -x \) 对称}
    {坐标原点对称}
    {直线 \( y = x \) 对称}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
函数定义域为 \(\{x\in\R\mid x\ne0\}\)，且 \(f(-x)=-\frac{1}{x}+x=-f(x)\). 因此 \(f\) 是奇函数，其图象关于坐标原点对称，故选 C.
\end{solution}

\begin{problem}
若 \(x \in (e^{-1}, 1)\)，\(a = \ln x\)，\(b = 2 \ln x\)，\(c = \ln^3 x\)，则（\quad）

\choices
    {\(a < b < c\)}
    {\(c < a < b\)}
    {\(b < a < c\)}
    {\(b < c < a\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
由 \(e^{-1}<x<1\)，得 \(-1<\ln x<0\). 令 \(t=-\ln x\)，则 \(0<t<1\)，并且 \(a=-t\)，\(b=-2t\)，\(c=-t^3\). 因为 \(-2t<-t\)，且 \(0<t<1\) 时 \(-t<-t^3\)，所以 \(b<a<c\)，故选 C.
\end{solution}

\begin{problem}
设变量 \(x\)，\(y\) 满足约束条件：\(\begin{cases}
y \geqslant x,\\
x + 2y \leqslant 2,\\
x \geqslant -2,
\end{cases}\) 则 \(z = x - 3y\) 的最小值（\quad）

\choices
    {\(-2\)}
    {\(-4\)}
    {\(-6\)}
    {\(-8\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
可行域由直线 \(y=x\)、\(x+2y=2\) 和 \(x=-2\) 围成，其顶点分别为 \((-2,-2)\)、\((-2,2)\) 和 \(\left(\frac{2}{3},\frac{2}{3}\right)\). 线性函数 \(z=x-3y\) 在可行域上的最小值必在顶点取得. 逐点计算得 \(z(-2,-2)=4\)，\(z(-2,2)=-8\)，\(z\left(\frac{2}{3},\frac{2}{3}\right)=-\frac{4}{3}\)，故最小值为 \(-8\)，选 D.
\end{solution}

\begin{problem}
从 \(20\text{ 名}\) 男同学，\(10\text{ 名}\) 女同学中任选 \(3\text{ 名}\) 参加体能测试，则选到的 \(3\text{ 名}\) 同学中既有男同学又有女同学的概率为（\quad）

\choices
    {\(\frac{9}{29}\)}
    {\(\frac{10}{29}\)}
    {\(\frac{19}{29}\)}
    {\(\frac{20}{29}\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
从 \(30\) 名同学中任选 \(3\) 名的总数为 \(\mathrm C_{30}^{3}\). 选出的同学性别单一的情况包括全为男生和全为女生，共有 \(\mathrm C_{20}^{3}+\mathrm C_{10}^{3}\) 种. 因此所求概率为 \(\frac{\mathrm C_{30}^{3}-\mathrm C_{20}^{3}-\mathrm C_{10}^{3}}{\mathrm C_{30}^{3}}=\frac{4060-1140-120}{4060}=\frac{20}{29}\)，故选 D.
\end{solution}

\begin{problem}
\((1 - \sqrt{x})^{6}(1 + \sqrt{x})^{4}\) 的展开式中 \(x\) 的系数是（\quad）

\choices
    {\(-4\)}
    {\(-3\)}
    {\(3\)}
    {\(4\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
令 \(t=\sqrt{x}\)，则原式 \((1-t)^6(1+t)^4=(1-t)^2(1-t^2)^4\). 其中 \(t^2\) 项只来自 \((1-t)^2\) 的常数项与 \((1-t^2)^4\) 的 \(t^2\) 项，以及 \((1-t)^2\) 的 \(t^2\) 项与后者的常数项，所以 \(t^2\) 的系数为 \(-4+1=-3\). 由于 \(t^2=x\)，故 \(x\) 的系数为 \(-3\)，选 B.
\end{solution}

\begin{problem}
若动直线 \(x = a\) 与函数 \(f(x) = \sin x\) 和 \(g(x) = \cos x\) 的图象分别交于 \(M\)，\(N\) 两点，则 \(|MN|\) 的最大值为（\quad）

\choices
    {\(1\)}
    {\(\sqrt{2}\)}
    {\(\sqrt{3}\)}
    {\(2\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
两交点的横坐标均为 \(a\)，故 \(M=(a,\sin a)\)，\(N=(a,\cos a)\)，从而 \(|MN|=|\sin a-\cos a|=\sqrt{2}|\sin\left(a-\frac{\pi}{4}\right)|\leqslant\sqrt{2}\). 当 \(a=\frac{3\pi}{4}+k\pi\) 时等号成立，故最大值为 \(\sqrt{2}\)，选 B.
\end{solution}

\begin{problem}
设 \(a > 1\)，则双曲线 \(\frac{x^2}{a^2} - \frac{y^2}{(a + 1)^2} = 1\) 的离心率 \(e\) 的取值范围是（\quad）

\choices
    {\((\sqrt{2}, 2)\)}
    {\((\sqrt{2}, \sqrt{5})\)}
    {\((2,5)\)}
    {\((2, \sqrt{5})\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
双曲线的半实轴长为 \(a\)，半虚轴长为 \(a+1\)，所以焦半距满足 \(c^2=a^2+(a+1)^2\)，离心率为 \(e=\frac{c}{a}=\sqrt{2+\frac{2}{a}+\frac{1}{a^2}}\). 当 \(a>1\) 时，\(0<\frac{2}{a}+\frac{1}{a^2}<3\)，并且该表达式的下界和上界均不能取到，故 \(\sqrt{2}<e<\sqrt{5}\)，选 B.
\end{solution}

\begin{problem}
已知正四棱锥 \(S - ABCD\) 的侧棱长与底面边长都相等，\(E\) 是 \(SB\) 的中点，则 \(AE\)，\(SD\) 所成的角的余弦值为（\quad）

\choices
    {\(\frac{1}{3}\)}
    {\(\frac{\sqrt{2}}{3}\)}
    {\(\frac{\sqrt{3}}{3}\)}
    {\(\frac{2}{3}\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
设底面边长为 \(l\)，建立直角坐标系，使正方形底面中心为原点，取 \(A=\left(-\frac l2,-\frac l2,0\right)\)，\(B=\left(\frac l2,-\frac l2,0\right)\)，\(D=\left(-\frac l2,\frac l2,0\right)\). 设 \(S=(0,0,h)\)，由 \(SA=l\) 得 \(h^2+\frac{l^2}{2}=l^2\)，即 \(h=\frac{l}{\sqrt{2}}\). 于是 \(E=\left(\frac l4,-\frac l4,\frac h2\right)\)，从而 \(\overrightarrow{AE}=\frac l4(3,1,\sqrt{2})\)，\(\overrightarrow{SD}=\frac l2(-1,1,-\sqrt{2})\). 两向量的数量积为 \(-\frac{l^2}{2}\)，长度分别为 \(\frac{l\sqrt{3}}{2}\) 和 \(l\). 直线所成的角取锐角，所以其余弦为 \(\frac{|\overrightarrow{AE}\cdot\overrightarrow{SD}|}{|AE||SD|}=\frac{\sqrt{3}}{3}\)，故选 C.
\end{solution}

\begin{problem}
等腰三角形两腰所在直线的方程分别为 \(x + y - 2 = 0\) 与 \(x - 7y - 4 = 0\)，原点在等腰三角形的底边上，则底边所在直线的斜率为（\quad）

\choices
    {\(3\)}
    {\(2\)}
    {\(-\frac{1}{3}\)}
    {\(-\frac{1}{2}\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
设底边所在直线为 \(y=kx\)，它与两腰所在直线分别交于 \(P\)，\(Q\). 两腰交点为 \(V=\left(\frac94,-\frac14\right)\). 由两直线方程得 \(P=\left(\frac{2}{1+k},\frac{2k}{1+k}\right)\)，\(Q=\left(\frac{4}{1-7k},\frac{4k}{1-7k}\right)\). 因为 \(P\) 在直线 \(x+y-2=0\) 上，\(Q\) 在直线 \(x-7y-4=0\) 上，所以
\(|VP|^2=\frac{(1+9k)^2}{8(1+k)^2}\)，\(|VQ|^2=\frac{25(1+9k)^2}{8(1-7k)^2}\).
等腰条件 \(|VP|=|VQ|\) 且非退化，给出 \((1-7k)^2=25(1+k)^2\)，故 \(k=-\frac13\) 或 \(k=3\). 当 \(k=-\frac13\) 时，\(P\) 与 \(Q\) 的横坐标均为正，原点不在线段 \(PQ\) 上；当 \(k=3\) 时，\(P=\left(\frac12,\frac32\right)\)，\(Q=\left(-\frac15,-\frac35\right)\)，原点在线段 \(PQ\) 上. 因此底边斜率为 \(3\)，故选 A.
\end{solution}

\begin{problem}
已知球的半径为 2，相互垂直的两个平面分别截球面得两个圆. 若两圆的公共弦长为 2，则两圆的圆心距等于（\quad）

\choices
    {\(1\)}
    {\(\sqrt{2}\)}
    {\(\sqrt{3}\)}
    {\(2\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
设球心为 \(O\)，取相互垂直的两个平面为 \(x=d_1\) 和 \(y=d_2\)，则两平面的交线为 \(l\colon x=d_1\)，\(y=d_2\). 两个截面圆的圆心分别为 \(O_1=(d_1,0,0)\) 和 \(O_2=(0,d_2,0)\)，所以 \(|O_1O_2|^2=d_1^2+d_2^2\). 公共弦在直线 \(l\) 上，弦长为 \(2\)，其半弦长为 \(1\)，故球心到 \(l\) 的距离满足 \(d_1^2+d_2^2=2^2-1^2=3\). 因此两圆的圆心距为 \(\sqrt{3}\)，故选 C.
\end{solution}

\section{填空题}

\begin{problem}
设向量 \( \bs{a}=(1,2) \)，\( \bs{b}=(2,3) \). 若向量 \( \lambda\bs{a}+\bs{b} \) 与向量 \( \bs{c}=(-4,-7) \) 共线，则 \(\lambda=\)\fillinblank{}.
\end{problem}

\begin{answer}
2
\end{answer}

\begin{solution}
有 \(\lambda\bs{a}+\bs{b}=(\lambda+2,2\lambda+3)\). 与 \(\bs{c}=(-4,-7)\) 共线的充要条件是行列式为零，即 \((\lambda+2)(-7)-(2\lambda+3)(-4)=0\). 化简得 \(\lambda-2=0\)，所以 \(\lambda=2\).
\end{solution}

\begin{problem}
设曲线 \( y = \e^{ax} \) 在点 \( (0,1) \) 处的切线与直线 \( x + 2y + 1 = 0 \) 垂直，则 \(a=\)\fillinblank{}.
\end{problem}

\begin{answer}
2
\end{answer}

\begin{solution}
函数 \(y=\e^{ax}\) 的导数为 \(y'=a\e^{ax}\)，所以在 \((0,1)\) 处的切线斜率为 \(a\). 直线 \(x+2y+1=0\) 的斜率为 \(-\frac12\)，两直线垂直，所以 \(a\left(-\frac12\right)=-1\)，解得 \(a=2\).
\end{solution}

\begin{problem}
已知 \(F\) 是抛物线 \(C\)：\( y^{2} = 4x \) 的焦点，过 \(F\) 且斜率为 1 的直线交 \(C\) 于 \(A\)，\(B\) 两点，设 \( |FA| > |FB| \)，则 \( |FA| \) 与 \( |FB| \) 的比值等于\fillinblank{}.
\end{problem}

\begin{answer}
\(3+2\sqrt{2}\)
\end{answer}

\begin{solution}
抛物线 \(y^2=4x\) 的焦点为 \(F=(1,0)\)，所给直线为 \(y=x-1\). 代入抛物线得 \((x-1)^2=4x\)，即 \(x^2-6x+1=0\)，所以两个交点的横坐标为 \(x_1=3+2\sqrt{2}\) 和 \(x_2=3-2\sqrt{2}\). 直线斜率为 \(1\)，故点到焦点的距离为 \(\sqrt{2}|x-1|\). 于是
\[
\frac{|FA|}{|FB|}=\frac{\sqrt{2}(x_1-1)}{\sqrt{2}|x_2-1|}=\frac{2+2\sqrt{2}}{2\sqrt{2}-2}=\frac{1+\sqrt{2}}{\sqrt{2}-1}=3+2\sqrt{2}
\]
\end{solution}

\begin{problem}
平面内的一个四边形为平行四边形的充要条件有多个，如两组对边分别平行，类似地，写出空间中的一个四棱柱为平行六面体的两个充要条件：

\begin{circlelist}
\item\fillinblank{}；
\item\fillinblank{}.
\end{circlelist}

（写出你认为正确的两个充要条件）
\end{problem}

\begin{answer}
三组对面分别平行；平行六面体的四条体对角线交于一点，并且在交点处互相平分.
\end{answer}

\begin{solution}
一个平行六面体的三组对面分别平行，因此“三组对面分别平行”是必要条件. 反过来，四棱柱的两个底面已经平行；若两组侧面对面也分别平行，则它们与底面的交线分别满足 \(AB\parallel CD\) 和 \(AD\parallel BC\)，底面是平行四边形，故该四棱柱是平行六面体，所以该条件也是充分的.

平行六面体的四条体对角线交于同一点，且该点是每条体对角线的中点，这是必要性. 反过来，设四棱柱的底面顶点按顺序为 \(A\)，\(B\)，\(C\)，\(D\)，上底面由同一平移向量得到. 取 \(A\) 为原点，记 \(\overrightarrow{AB}=\bs{u}\)，\(\overrightarrow{AD}=\bs{v}\)，\(\overrightarrow{AC}=\bs{w}\). 两条体对角线的中点重合意味着 \(\frac{\bs{w}+\bs{t}}{2}=\frac{\bs{u}+\bs{v}+\bs{t}}{2}\)，其中 \(\bs{t}\) 是棱柱的平移向量，于是 \(\bs{w}=\bs{u}+\bs{v}\)，即底面为平行四边形. 因此该四棱柱是平行六面体，故“体对角线交于一点且在交点处互相平分”也是充分条件.
\end{solution}

\begin{problem}
在 \(\triangle ABC\) 中，\(\cos B = -\frac{5}{13}\)，\(\cos C = \frac{4}{5}\).

\begin{enumerate}
\item 求 \(\sin A\) 的值；
\item 设 \(\triangle ABC\) 的面积 \(S_{\triangle ABC} = \frac{33}{2}\)，求 \(BC\) 的长.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 因为 \(B\)，\(C\) 是三角形内角，所以 \(\sin B=\sqrt{1-\cos^2B}=\frac{12}{13}\)，\(\sin C=\sqrt{1-\cos^2C}=\frac35\). 又 \(A=\pi-(B+C)\)，所以 \(\sin A=\sin(B+C)=\sin B\cos C+\cos B\sin C=\frac{12}{13}\cdot\frac45-\frac5{13}\cdot\frac35=\frac{33}{65}\).
\item 设 \(BC=a_0\). 由正弦定理，\(AB=\frac{a_0\sin C}{\sin A}\)，\(AC=\frac{a_0\sin B}{\sin A}\)，故
\[
S_{\triangle ABC}=\frac12 AB\cdot AC\sin A=\frac{a_0^2\sin B\sin C}{2\sin A}=\frac{a_0^2}{2}\cdot\frac{\frac{12}{13}\cdot\frac35}{\frac{33}{65}}=\frac{6a_0^2}{11}
\]
由 \(\frac{6a_0^2}{11}=\frac{33}{2}\)，得 \(a_0^2=\frac{121}{4}\). 因为边长为正，所以 \(BC=a_0=\frac{11}{2}\).
\end{enumerate}
\end{solution}

\begin{problem}
购买某种保险，每个投保人每年度向保险公司交纳保费 \(a\text{ 元}\)，若投保人在购买保险的一年度内出险，则可以获得 \(10000\text{ 元}\) 的赔偿金. 假定在一年度内有 \(10000\text{ 人}\) 购买了这种保险，且各投保人是否出险相互独立. 已知保险公司在一年度内至少支付赔偿金 \(10000\text{ 元}\) 的概率为 \(1 - 0.999^{10^{4}}\).

\begin{enumerate}
\item 求一投保人在一年度内出险的概率 \(p\)；
\item 设保险公司开办该项险种业务除赔偿金外的成本为 \(50000\text{ 元}\)，为保证盈利的期望不小于 0，求每位投保人应交的最低保费. （单位：元）
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 保险公司至少支付一笔赔偿金等价于至少有一名投保人出险，其对立事件是无人出险. 因此 \(1-(1-p)^{10000}=1-0.999^{10000}\)，从而 \((1-p)^{10000}=0.999^{10000}\). 由于 \(p\) 是概率，\(0\leqslant1-p\leqslant1\)，而 \(0.999\in(0,1)\)，故 \(1-p\in(0,1)\). 函数 \(u\mapsto u^{10000}\) 在 \([0,1]\) 上严格递增，所以 \(1-p=0.999\)，即 \(p=0.001\).
\item 设一年内出险人数为 \(X\)，则 \(E(X)=10000p=10\). 保险公司的期望收入为 \(10000a\) 元，期望赔偿金为 \(10000E(X)=100000\) 元，所以期望利润为 \(10000a-100000-50000=10000a-150000\) 元. 要求期望利润不小于 \(0\)，得 \(a\geqslant15\). 因此每位投保人的最低保费为 \(15\) 元.
\end{enumerate}
\end{solution}

\begin{problem}
如图，正四棱柱 \(ABCD - A_{1}B_{1}C_{1}D_{1}\) 中，\(AA_{1} = 2AB = 4\)，点 \(E\) 在 \(CC_{1}\) 上且 \(C_{1}E = 3EC\).

\begin{enumerate}
\item 证明： \(A_{1}C \perp\) 平面 \(BED\)；
\item 求二面角 \(A_{1} - DE - B\) 的大小.
\end{enumerate}

\bitmapfigure[width=0.34\linewidth]{img/2008/syllabus_paper_2_science/q19_fig1.png}
\end{problem}

\begin{solution}
取空间直角坐标系，使 \(A=(0,0,0)\)，\(B=(2,0,0)\)，\(D=(0,2,0)\)，\(A_1=(0,0,4)\). 于是 \(C=(2,2,0)\)，\(C_1=(2,2,4)\). 由 \(C_1E=3EC\) 得 \(E=(2,2,1)\).

\begin{enumerate}
\item 有 \(\overrightarrow{A_1C}=(2,2,-4)\)，\(\overrightarrow{BD}=(-2,2,0)\)，\(\overrightarrow{BE}=(0,2,1)\). 于是 \(\overrightarrow{A_1C}\cdot\overrightarrow{BD}=0\)，且 \(\overrightarrow{A_1C}\cdot\overrightarrow{BE}=0\). 由于 \(BD\) 与 \(BE\) 是平面 \(BED\) 内相交的两条直线，所以 \(A_1C\perp\) 平面 \(BED\).
\item 取平面 \(A_1DE\) 的法向量 \(\bs{n}_1=(1,-4,-2)\)，平面 \(BDE\) 的法向量 \(\bs{n}_2=(1,1,-2)\). 例如，\(\bs{n}_1\) 与 \(\overrightarrow{DE}=(2,0,1)\)、\(\overrightarrow{DA_1}=(0,-2,4)\) 均垂直，\(\bs{n}_2\) 与 \(\overrightarrow{DE}=(2,0,1)\)、\(\overrightarrow{DB}=(2,-2,0)\) 均垂直. 故二面角的余弦为
\[
\cos\theta=\frac{|\bs{n}_1\cdot\bs{n}_2|}{|\bs{n}_1||\bs{n}_2|}=\frac{1}{\sqrt{21}\sqrt{6}}=\frac{1}{3\sqrt{14}}
\]
于是 \(\tan^2\theta=\frac{1}{\cos^2\theta}-1=126-1=125\). 二面角取锐角，故 \(\tan\theta=5\sqrt{5}\)，从而二面角的大小为 \(\arctan(5\sqrt{5})\).
\end{enumerate}
\end{solution}

\begin{problem}
设数列 \(\{a_n\}\) 的前 \(n\) 项和为 \(S_n\). 已知 \(a_1=a\)，\(a_{n+1}=S_n+3^n\)，\(n\in\N^*\).

\begin{enumerate}
\item 设 \(b_n=S_n-3^n\)，求数列 \(\{b_n\}\) 的通项公式；
\item 若 \(a_{n+1}\geqslant a_n\)，\(n\in\N^*\)，求 \(a\) 的取值范围.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 因为 \(S_{n+1}=S_n+a_{n+1}\)，所以由题设 \(S_{n+1}=2S_n+3^n\). 于是 \(b_{n+1}=S_{n+1}-3^{n+1}=2S_n+3^n-3\cdot3^n=2(S_n-3^n)=2b_n\). 又 \(b_1=S_1-3=a-3\)，所以 \(\{b_n\}\) 是首项为 \(a-3\)、公比为 \(2\) 的等比数列，故 \(b_n=(a-3)2^{n-1}\).
\item 由上式得 \(S_n=3^n+(a-3)2^{n-1}\). 当 \(n\geqslant2\) 时，\(a_n=S_n-S_{n-1}=2\cdot3^{n-1}+(a-3)2^{n-2}\)，从而
\[
a_{n+1}-a_n=4\cdot3^{n-1}+(a-3)2^{n-2}=2^{n-2}\left(8\left(\frac32\right)^{n-1}+a-3\right)
\]
当 \(n=1\) 时，\(a_2=S_1+3=a+3>a=a_1\)，条件自动成立. 当 \(n\geqslant2\) 时，要求上式非负，等价于 \(a\geqslant3-8\left(\frac32\right)^{n-1}\). 右端随 \(n\) 增大而减小，其最大值在 \(n=2\) 时为 \(-9\)，所以对所有正整数 \(n\) 成立的充要条件是 \(a\geqslant-9\).
\end{enumerate}
\end{solution}

\begin{problem}
设椭圆中心在坐标原点，\(A(2,0)\)，\(B(0,1)\) 是它的两个顶点，直线 \(y=kx\)（\(k>0\)）与 \(AB\) 相交于点 \(D\)，与椭圆相交于 \(E\)，\(F\) 两点.

\begin{enumerate}
\item 若 \(\overrightarrow{ED}=6\overrightarrow{DF}\)，求 \(k\) 的值；
\item 求四边形 \(AEBF\) 面积的最大值.
\end{enumerate}
\end{problem}

\begin{solution}
椭圆方程为 \(\frac{x^2}{4}+y^2=1\)，直线 \(AB\) 的方程为 \(y=1-\frac{x}{2}\). 直线 \(y=kx\) 与 \(AB\) 的交点为 \(D=\left(\frac{2}{2k+1},\frac{2k}{2k+1}\right)\)，与椭圆的两个交点横坐标为 \(\pm\frac{2}{\sqrt{1+4k^2}}\).

\begin{enumerate}
\item 由于 \(D\) 在线段 \(AB\) 内，且 \(\overrightarrow{ED}\) 与 \(\overrightarrow{DF}\) 同向，故取 \(E\) 为直线上的负向交点，\(F\) 为正向交点. 记 \(t=\frac{2}{\sqrt{1+4k^2}}\)，则沿该直线用横坐标表示向量条件为 \(\frac{2}{2k+1}+t=6\left(t-\frac{2}{2k+1}\right)\)，即 \(\frac{2}{2k+1}=\frac{5t}{7}\). 因此 \(7\sqrt{1+4k^2}=5(2k+1)\)，平方并化简得 \(24k^2-25k+6=0\)，所以 \(k=\frac23\) 或 \(k=\frac38\).
\item 仍记 \(t=\frac{2}{\sqrt{1+4k^2}}\)，则 \(E=(-t,-kt)\)，\(F=(t,kt)\). 由鞋带公式，四边形 \(AEBF\) 的面积为 \(t(2k+1)=\frac{2(2k+1)}{\sqrt{1+4k^2}}\). 对任意 \(k>0\)，有
\[
\frac{(2k+1)^2}{1+4k^2}\leqslant2
\]
因为不等式等价于 \((2k-1)^2\geqslant0\). 所以四边形面积不超过 \(2\sqrt{2}\)，且当 \(k=\frac12\) 时等号成立. 因此面积的最大值为 \(2\sqrt{2}\).
\end{enumerate}
\end{solution}

\begin{problem}
设函数 \(f(x)=\frac{\sin x}{2+\cos x}\).

\begin{enumerate}
\item 求 \(f(x)\) 的单调区间；
\item 如果对任何 \(x\geqslant0\)，都有 \(f(x)\leqslant ax\)，求 \(a\) 的取值范围.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 因为 \(2+\cos x\geqslant1>0\)，所以函数定义域为 \(\R\)，且
\[
f'(x)=\frac{(2+\cos x)\cos x+\sin^2x}{(2+\cos x)^2}=\frac{2\cos x+1}{(2+\cos x)^2}
\]
因此当 \(2k\pi-\frac{2\pi}{3}<x<2k\pi+\frac{2\pi}{3}\) 时，\(f'(x)>0\)；当 \(2k\pi+\frac{2\pi}{3}<x<2k\pi+\frac{4\pi}{3}\) 时，\(f'(x)<0\)，其中 \(k\in\Z\). 所以 \(f(x)\) 在区间 \(\left(2k\pi-\frac{2\pi}{3},2k\pi+\frac{2\pi}{3}\right)\) 上单调递增，在区间 \(\left(2k\pi+\frac{2\pi}{3},2k\pi+\frac{4\pi}{3}\right)\) 上单调递减.
\item 先证充分性. 令 \(g(x)=ax-f(x)\)，则
\[
g'(x)=a-\frac{2\cos x+1}{(2+\cos x)^2}=a-\frac13+\frac{(1-\cos x)^2}{3(2+\cos x)^2}\geqslant a-\frac13
\]
当 \(a\geqslant\frac13\) 时，\(g'(x)\geqslant0\)，且 \(g(0)=0\)，所以对所有 \(x\geqslant0\)，有 \(g(x)\geqslant0\)，即 \(f(x)\leqslant ax\).

再证必要性. 当 \(a\leqslant0\) 时，取 \(x=\frac\pi2\)，有 \(f\left(\frac\pi2\right)=\frac12>0\geqslant a\frac\pi2\)，不满足条件. 当 \(0<a<\frac13\) 时，取 \(0<x<\arccos(3a)\)，令 \(h(x)=\sin x-3ax\)，则 \(h(0)=0\)，且 \(h'(x)=\cos x-3a>0\)，所以 \(\sin x>3ax\). 又此时 \(\sin x>0\) 且 \(2+\cos x<3\)，故 \(f(x)=\frac{\sin x}{2+\cos x}>\frac{\sin x}{3}>ax\)，同样不满足条件. 因此 \(a\) 的取值范围为 \(\left[\frac13,+\infty\right)\).
\end{enumerate}
\end{solution}
