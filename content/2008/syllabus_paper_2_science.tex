\section{选择题}

\begin{problem}
设集合 \(M = \{m \in \Z \mid -3 < m < 2\}\)，\(N = \{n \in \Z \mid -1 \leqslant n \leqslant 3\}\)，则 \(M \cap N =\)（\quad）

\choices
    {\(\{0,1\}\)}
    {\(\{-1, 0, 1\}\)}
    {\(\{0, 1, 2\}\)}
    {\(\{-1, 0, 1, 2\}\)}
\end{problem}

\begin{problem}
设 \(a\)，\(b \in \R\) 且 \(b \neq 0\)，若复数 \((a + bi)^3\) 是实数，则（\quad）

\choices
    {\( b^{2}=3a^{2} \)}
    {\(a^2 = 3b^2\)}
    {\( b^{2}=9a^{2} \)}
    {\(a^2 = 9b^2\)}
\end{problem}

\begin{problem}
函数 \( f(x) = \frac{1}{x} - x \) 的图象关于（\quad）

\choices
    {\(y\) 轴对称}
    {直线 \( y = -x \) 对称}
    {坐标原点对称}
    {直线 \( y = x \) 对称}
\end{problem}

\begin{problem}
若 \(x \in (e^{-1}, 1)\)，\(a = \ln x\)，\(b = 2 \ln x\)，\(c = \ln^3 x\)，则（\quad）

\choices
    {\(a < b < c\)}
    {\(c < a < b\)}
    {\(b < a < c\)}
    {\(b < c < a\)}
\end{problem}

\begin{problem}
设变量 \(x\)，\(y\) 满足约束条件：\(\left\{ \begin{array}{l} y \geqslant x, \\ x + 2y \leqslant 2, \\ x \geqslant -2, \end{array} \right.\) 则 \(z = x - 3y\) 的最小值（\quad）

\choices
    {\(-2\)}
    {\(-4\)}
    {\(-6\)}
    {\(-8\)}
\end{problem}

\begin{problem}
从 20 名男同学，10 名女同学中任选 3 名参加体能测试，则选到的 3 名同学中既有男同学又有女同学的概率为（\quad）

\choices
    {\(\frac{9}{29}\)}
    {\(\frac{10}{29}\)}
    {\(\frac{19}{29}\)}
    {\(\frac{20}{29}\)}
\end{problem}

\begin{problem}
\((1 - \sqrt{x})^{6}(1 + \sqrt{x})^{4}\) 的展开式中 \(x\) 的系数是（\quad）

\choices
    {\(-4\)}
    {\(-3\)}
    {\(3\)}
    {\(4\)}
\end{problem}

\begin{problem}
若动直线 \(x = a\) 与函数 \(f(x) = \sin x\) 和 \(g(x) = \cos x\) 的图象分别交于 \(M\)，\(N\) 两点，则 \(|MN|\) 的最大值为（\quad）

\choices
    {\(1\)}
    {\(\sqrt{2}\)}
    {\(\sqrt{3}\)}
    {\(2\)}
\end{problem}

\begin{problem}
设 \(a > 1\)，则双曲线 \(\frac{x^2}{a^2} - \frac{y^2}{(a + 1)^2} = 1\) 的离心率 \(e\) 的取值范围是（\quad）

\choices
    {\((\sqrt{2}, 2)\)}
    {\((\sqrt{2}, \sqrt{5})\)}
    {\((2,5)\)}
    {\((2, \sqrt{5})\)}
\end{problem}

\begin{problem}
已知正四棱锥 \(S - ABCD\) 的侧棱长与底面边长都相等，\(E\) 是 \(SB\) 的中点，则 \(AE\)，\(SD\) 所成的角的余弦值为（\quad）

\choices
    {\(\frac{1}{3}\)}
    {\(\frac{\sqrt{2}}{3}\)}
    {\(\frac{\sqrt{3}}{3}\)}
    {\(\frac{2}{3}\)}
\end{problem}

\begin{problem}
等腰三角形两腰所在直线的方程分别为 \(x + y - 2 = 0\) 与 \(x - 7y - 4 = 0\)，原点在等腰三角形的底边上，则底边所在直线的斜率为（\quad）

\choices
    {\(3\)}
    {\(2\)}
    {\(-\frac{1}{3}\)}
    {\(-\frac{1}{2}\)}
\end{problem}

\begin{problem}
已知球的半径为 2，相互垂直的两个平面分别截球面得两个圆. 若两圆的公共弦长为 2，则两圆的圆心距等于（\quad）

    {\(2\)}

\section{解答题}

\begin{problem}
在 \(\triangle ABC\) 中，\(\cos B = -\frac{5}{13}\)，\(\cos C = \frac{4}{5}\).

\begin{enumerate}
\item 求 \(\sin A\) 的值；
\item 设 \(\triangle ABC\) 的面积 \(S_{\triangle ABC} = \frac{33}{2}\)，求 \(BC\) 的长.
\end{enumerate}
\end{problem}

\begin{problem}
购买某种保险，每个投保人每年度向保险公司交纳保费 \(a\) 元，若投保人在购买保险的一年度内出险，则可以获得 10000 元的赔偿金.假定在一年度内有 10000 人购买了这种保险，且各投保人是否出险相互独立.已知保险公司在一年度内至少支付赔偿金 10000 元的概率为 \(1 - 0.999^{10^{4}}\).

\begin{enumerate}
\item 求一投保人在一年度内出险的概率 \(p\)；
\item 设保险公司开办该项险种业务除赔偿金外的成本为 50000 元，为保证盈利的期望不小于 0，求每位投保人应交的最低保费.（单位：元）
\end{enumerate}
\end{problem}

\begin{problem}
设数列 \(\{a_{n}\}\) 的前 \(n\) 项和为 \(S_{n}\).已知 \(a_1 = a\)，\(a_{n+1} = S_n + 3^n\)，\(n \in \mathbf{N}^*\).

\begin{enumerate}
\item 设 \( b_{n} = S_{n} - 3^{n} \)，求数列 \(\{b_{n}\}\) 的通项公式；
\item 若 \(a_{n+1} \geqslant a_n\)，\(n \in \mathbf{N}^*\)，求 \(a\) 的取值范围.
\end{enumerate}
\end{problem}

\begin{problem}
设椭圆中心在坐标原点，\(A(2,0)\)，\(B(0,1)\) 是它的两个顶点，直线 \(y = kx\)（\(k > 0\)）与 \(AB\) 相交于点 \(D\)，与椭圆相交于 \(E\)，\(F\) 两点.

\begin{enumerate}
\item 若 \(\overline{ED} = 6\overline{DF}\)，求 \(k\) 的值；
\item 求四边形 \(AEBF\) 面积的最大值.
\end{enumerate}
\end{problem}

\begin{problem}
设函数 \( f(x) = \frac{\sin x}{2 + \cos x} \).

\begin{enumerate}
\item 求 \( f(x) \) 的单调区间；
\item 如果对任何 \(x \geqslant 0\)，都有 \(f(x) \leqslant a x\)，求 \(a\) 的取值范围.
\end{enumerate}
\end{problem}
