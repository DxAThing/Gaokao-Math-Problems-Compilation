\begin{problem}
已知 \(\sin 2\alpha = -\sin \alpha\)（\(\alpha \in \left(\frac{\pi}{2}, \pi\right)\)），则 \(\cot \alpha =\)\fillinblank{}.
\end{problem}

\begin{answer}
\(-\frac{\sqrt{3}}{3}\)
\end{answer}

\begin{solution}
由 \(\alpha \in \left(\frac{\pi}{2},\pi\right)\) 可知 \(\sin\alpha>0\). 由二倍角公式，原条件化为
\[
2\sin\alpha\cos\alpha=-\sin\alpha.
\]
两边同除以 \(\sin\alpha\)，得 \(2\cos\alpha=-1\)，所以 \(\cos\alpha=-\frac{1}{2}\). 又因为 \(\alpha\) 在第二象限，\(\sin\alpha=\frac{\sqrt{3}}{2}\)，从而
\[
\cot\alpha=\frac{\cos\alpha}{\sin\alpha}=-\frac{1}{\sqrt{3}}=-\frac{\sqrt{3}}{3}.
\]
\end{solution}

\section{解答题}
\begin{problem}
在等比数列 \(\{a_{n}\}\) 中，已知 \(a_6 - a_4 = 24\)，\(a_3a_5 = 64\)，求 \(\{a_{n}\}\) 前 \(8\) 项的和 \(S_{8}\).
\end{problem}

\begin{solution}
设等比数列的公比为 \(q\). 因为 \(a_3a_5=a_4^2\)，所以 \(a_4^2=64\)，即 \(a_4=8\) 或 \(a_4=-8\). 又由等比数列的性质，\(a_6=a_4q^2\)，因此
\[
a_4(q^2-1)=24.
\]
当 \(a_4=8\) 时，\(q^2-1=3\)，得 \(q=2\) 或 \(q=-2\)；当 \(a_4=-8\) 时，\(q^2-1=-3\)，这不可能，因为 \(q^2\geqslant0\).

当 \(q=2\) 时，\(a_1=a_4/q^3=1\)，故前八项为 \(1,2,4,8,16,32,64,128\)，从而 \(S_8=255\). 当 \(q=-2\) 时，\(a_1=a_4/q^3=-1\)，从而
\[
S_8=a_1\frac{1-q^8}{1-q}=-1\cdot\frac{1-256}{3}=85.
\]
所以 \(S_8=255\) 或 \(S_8=85\).
\end{solution}

\begin{problem}
已知 \(\sin^2 2\alpha + \sin 2\alpha \cos \alpha - \cos 2\alpha = 1\)（\(\alpha \in \left(0, \frac{\pi}{2}\right)\)），求 \(\sin \alpha\)，\(\tan \alpha\) 的值.
\end{problem}

\begin{solution}
令 \(s=\sin\alpha\)，\(c=\cos\alpha\). 由 \(0<\alpha<\frac{\pi}{2}\) 知 \(0<s<1\)，且 \(c^2=1-s^2\). 把 \(\sin2\alpha=2sc\)、\(\cos2\alpha=c^2-s^2\) 代入原式，得
\[
4s^2c^2+2sc^2-(c^2-s^2)=1.
\]
再代入 \(c^2=1-s^2\)，并移项化简，得到
\[
2s^4+s^3-3s^2-s+1=0.
\]
因式分解得
\[
(2s-1)(s-1)(s+1)^2=0.
\]
由于 \(0<s<1\)，只能取 \(s=\frac{1}{2}\)，于是 \(c=\frac{\sqrt{3}}{2}\). 因此 \(\sin\alpha=\frac{1}{2}\)，\(\tan\alpha=\frac{\sin\alpha}{\cos\alpha}=\frac{\sqrt{3}}{3}\).
\end{solution}

\begin{problem}[甲]
如图，正三棱柱 \(ABC - A_{1}B_{1}C_{1}\) 的底面边长为 \(a\)，侧棱长为 \(\sqrt{2}a\).

\begin{enumerate}
\item 建立适当的坐标系，并写出点 \(A\)，\(B\)，\(A_{1}\)，\(C_{1}\) 的坐标；
\item 求 \(AC_{1}\) 与侧面 \(ABB_{1}A_{1}\) 所成的角.
\end{enumerate}

\bitmapfigure[max width=0.36\linewidth]{img/2002/new_curriculum_liberal/q19_fig1.png}
\FloatBarrier
\end{problem}

\begin{solution}
\begin{enumerate}
\item 取 \(A\) 为原点，\(AB\) 所在方向为 \(y\) 轴正向，\(AA_1\) 所在方向为 \(z\) 轴正向，并取 \(x\) 轴垂直于侧面 \(ABB_1A_1\). 于是可设 \(A(0,0,0)\)，\(B(0,a,0)\)，\(A_1(0,0,\sqrt{2}a)\)，\(C_1\left(-\frac{\sqrt{3}}{2}a,\frac{a}{2},\sqrt{2}a\right)\).
\item 侧面 \(ABB_1A_1\) 的方程为 \(x=0\)，而
\(\overrightarrow{AC_1}=\left(-\frac{\sqrt{3}}{2}a,\frac{a}{2},\sqrt{2}a\right)\)，且 \(|AC_1|=\sqrt{3}a\).
设直线 \(AC_1\) 与平面 \(ABB_1A_1\) 所成的角为 \(\varphi\). 直线方向向量在该平面法向方向上的分量的绝对值为 \(\frac{\sqrt{3}}{2}a\)，所以
\[
\sin\varphi=\frac{\frac{\sqrt{3}}{2}a}{\sqrt{3}a}=\frac{1}{2}.
\]
由于所成角取锐角，得 \(\varphi=30^{\circ}\).
\end{enumerate}
\end{solution}

\begin{problem}[乙]
如图，正方形 \(ABCD\)，\(ABEF\) 的边长都是 \(1\)，而且平面 \(ABCD\)，\(ABEF\) 互相垂直. 点 \(M\) 在 \(AC\) 上移动，点 \(N\) 在 \(BF\) 上移动，若 \(CM = BN = a\)（\(0 < a < \sqrt{2}\)）.

\begin{enumerate}
\item 求 \(MN\) 的长；
\item \(a\) 为何值时，\(MN\) 的长最小；
\item 当 \(MN\) 的长最小时，求面 \(MNA\) 与面 \(MNB\) 所成二面角 \(\alpha\) 的大小.
\end{enumerate}

\bitmapfigure[max width=0.36\linewidth]{img/2002/new_curriculum_liberal/q20_fig1.png}
\FloatBarrier
\end{problem}

\begin{solution}
\begin{enumerate}
\item 建立直角坐标系，取 \(A(0,0,0)\)，\(B(1,0,0)\)，\(C(1,1,0)\)，\(D(0,1,0)\)，\(E(1,0,1)\)，\(F(0,0,1)\).
此时平面 \(ABCD\) 为 \(z=0\)，平面 \(ABEF\) 为 \(y=0\)，二者互相垂直. 因为 \(|AC|=|BF|=\sqrt{2}\)，且 \(CM=BN=a\)，所以
\[
M=C+\frac{a}{\sqrt{2}}\overrightarrow{CA}=\left(1-\frac{a}{\sqrt{2}},1-\frac{a}{\sqrt{2}},0\right),
\]
\[
N=B+\frac{a}{\sqrt{2}}\overrightarrow{BF}=\left(1-\frac{a}{\sqrt{2}},0,\frac{a}{\sqrt{2}}\right).
\]
因此
\[
MN^2=\left(1-\frac{a}{\sqrt{2}}\right)^2+\left(\frac{a}{\sqrt{2}}\right)^2=\left(a-\frac{\sqrt{2}}{2}\right)^2+\frac{1}{2},
\]
故 \(MN=\sqrt{\left(a-\frac{\sqrt{2}}{2}\right)^2+\frac{1}{2}}\).
\item 由上式可知，当 \(a=\frac{\sqrt{2}}{2}\) 时，\(MN\) 取得最小值 \(\frac{\sqrt{2}}{2}\).
\item 当 \(a=\frac{\sqrt{2}}{2}\) 时，\(M\left(\frac{1}{2},\frac{1}{2},0\right)\)，\(N\left(\frac{1}{2},0,\frac{1}{2}\right)\). 取平面 \(MNA\)、\(MNB\) 的法向量分别为 \(\bs{n}_1=\overrightarrow{AM}\times\overrightarrow{AN}=\frac{1}{4}(1,-1,-1)\)，\(\bs{n}_2=\overrightarrow{BM}\times\overrightarrow{BN}=\frac{1}{4}(1,1,1)\).
设二面角为 \(\alpha\)，则
\[
\cos\alpha=\frac{\bs{n}_1\cdot\bs{n}_2}{|\bs{n}_1||\bs{n}_2|}=-\frac{1}{3}.
\]
所以 \(\alpha=\arccos\left(-\frac{1}{3}\right)\).
\end{enumerate}
\end{solution}

\begin{problem}
某单位 \(6\) 个员工借助互联网开展工作，每个员工上网的概率都是 \(0.5\)（相互独立）.

\begin{enumerate}
\item 求至少 \(3\) 人同时上网的概率；
\item 至少几人同时上网的概率小于 \(0.3\)？
\end{enumerate}
\end{problem}

\begin{solution}
设同时上网的员工人数为随机变量 \(X\). 由于每个员工上网与否相互独立，且上网概率均为 \(0.5\)，所以恰有 \(k\) 人同时上网的概率为 \(\frac{\mathrm C_6^k}{2^6}\).
\begin{enumerate}
\item 至少 \(3\) 人同时上网的概率为
\[
P(X\geqslant3)=\frac{\mathrm C_6^3+\mathrm C_6^4+\mathrm C_6^5+\mathrm C_6^6}{2^6}=1-\frac{\mathrm C_6^0+\mathrm C_6^1+\mathrm C_6^2}{2^6}=1-\frac{1+6+15}{64}=\frac{21}{32}.
\]
\item 至少 \(4\) 人同时上网的概率为
\[
P(X\geqslant4)=\frac{\mathrm C_6^4+\mathrm C_6^5+\mathrm C_6^6}{64}=\frac{15+6+1}{64}=\frac{22}{64}>0.3,
\]
而至少 \(5\) 人同时上网的概率为
\[
P(X\geqslant5)=\frac{\mathrm C_6^5+\mathrm C_6^6}{64}=\frac{6+1}{64}=\frac{7}{64}<0.3.
\]
因此，至少 \(5\) 人同时上网时，该概率小于 \(0.3\).
\end{enumerate}
\end{solution}

\begin{problem}
已知 \(a > 0\)，函数 \(f(x) = x^3 - a\)（\(x \in (0, +\infty)\)）. 设 \(x_1 > 0\)，记曲线 \(y = f(x)\) 在点 \((x_1, f(x_1))\) 处的切线为 \(l\).

\begin{enumerate}
\item 求 \(l\) 的方程；
\item 设 \(l\) 与 \(x\) 轴交点为 \((x_{2}, 0)\). 证明：
\begin{enumerate}
\item \(x_{2} \geqslant a^{\frac{1}{3}}\)；
\item 若 \(x_{2} > a^{\frac{1}{3}}\)，则 \(a^{\frac{1}{3}} < x_{2} < x_{1}\).
\end{enumerate}
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 因为 \(f'(x)=3x^2\)，所以曲线在 \((x_1,f(x_1))\) 处的切线斜率为 \(3x_1^2\). 由点斜式，切线方程为
\[
y-(x_1^3-a)=3x_1^2(x-x_1),
\]
即 \(l\) 的方程为 \(y=3x_1^2x-2x_1^3-a\).
\item 令 \(y=0\)，得切线与 \(x\) 轴交点的横坐标
\[
x_2=\frac{2x_1^3+a}{3x_1^2}.
\]
\begin{enumerate}
\item 令 \(t=a^{\frac{1}{3}}>0\)，则
\[
x_2-t=\frac{2x_1^3+t^3-3tx_1^2}{3x_1^2}=\frac{(x_1-t)^2(2x_1+t)}{3x_1^2}\geqslant0.
\]
所以 \(x_2\geqslant a^{\frac{1}{3}}\).
\item 原题给出的条件 \(x_2>a^{\frac{1}{3}}\) 只能说明 \(x_1\neq a^{\frac{1}{3}}\)，并不能推出 \(x_2<x_1\). 事实上
\[
x_1-x_2=\frac{x_1^3-a}{3x_1^2}=\frac{(x_1-t)(x_1^2+tx_1+t^2)}{3x_1^2}.
\]
当 \(x_1>t\) 时，确有 \(t<x_2<x_1\)；但当 \(0<x_1<t\) 时，反而有 \(x_2>x_1>0\). 例如取 \(a=1\)、\(x_1=\frac{1}{2}\)，则 \(x_2=\frac{5}{3}>1=a^{\frac{1}{3}}\)，但 \(x_2>x_1\). 因此，第（2）小问按原题字面不成立，不能给出其所要求的证明；若补充条件 \(x_1>a^{\frac{1}{3}}\)，则上式立即给出题中结论.
\end{enumerate}
\end{enumerate}
\end{solution}

\begin{problem}
已知两点 \(M(-1,0)\)，\(N(1,0)\)，且点 \(P\) 使 \(\overrightarrow{MP} \cdot \overrightarrow{MN}\)，\(\overrightarrow{PM} \cdot \overrightarrow{PN}\)，\(\overrightarrow{NM} \cdot \overrightarrow{NP}\) 成公差小于零的等差数列.

\begin{enumerate}
\item 点 \(P\) 的轨迹是什么曲线？
\item 若点 \(P\) 坐标为 \((x_0, y_0)\)，记 \(\theta\) 为 \(\overrightarrow{PM}\) 与 \(\overrightarrow{PN}\) 的夹角，求 \(\tan \theta\).
\end{enumerate}
\end{problem}

\begin{solution}
设 \(P(x,y)\). 则
\[
\overrightarrow{MP}\cdot\overrightarrow{MN}=2(x+1),
\]
\[
\overrightarrow{PM}\cdot\overrightarrow{PN}=x^2+y^2-1,
\]
\[
\overrightarrow{NM}\cdot\overrightarrow{NP}=2(1-x).
\]
由于这三个数按所列顺序成等差数列，故中间项是首、末两项的平均数，即
\[
2(x^2+y^2-1)=2(x+1)+2(1-x)=4.
\]
从而 \(x^2+y^2=3\). 等差数列的公差为
\[
(x^2+y^2-1)-2(x+1)=-2x.
\]
题设公差小于零，所以 \(x>0\).
\begin{enumerate}
\item 点 \(P\) 的轨迹是圆 \(x^2+y^2=3\) 的右半圆，即 \(x^2+y^2=3\)，\(x>0\).
\item 在轨迹上，
\[
\overrightarrow{PM}\cdot\overrightarrow{PN}=2,
\]
且 \(|PM|^2=(x+1)^2+y^2=4+2x\)，\(|PN|^2=(x-1)^2+y^2=4-2x\).
因此 \(\theta\) 为锐角. 由两向量夹角的正切公式，
\[
\tan\theta=\frac{\left|(-x-1)(-y)-(-y)(1-x)\right|}{\overrightarrow{PM}\cdot\overrightarrow{PN}}=\frac{2|y|}{2}=|y|.
\]
若 \(P=(x_0,y_0)\)，则
\[
\tan\theta=|y_0|=\sqrt{3-x_0^2}.
\]
\end{enumerate}
\end{solution}
