\chapter{2002年新课标卷（文）}
\section{单选题}
\begin{problem}
直线 \((1 + a)x + y + 1 = 0\) 与圆 \(x^{2} + y^{2} - 2x = 0\) 相切，则 \(a\) 的值为（\quad）

\choices
    {\(\pm 1\)}
    {\(\pm 2\)}
    {\(1\)}
    {\(-1\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
圆的方程可化为 \((x-1)^2+y^2=1\)，所以圆心为 \(C(1,0)\)，半径为 \(1\). 直线到圆心的距离为
\[
\frac{|(1+a)\cdot1+0+1|}{\sqrt{(1+a)^2+1}}=\frac{|a+2|}{\sqrt{(1+a)^2+1}}
\]
相切时该距离等于半径，故 \((a+2)^2=(a+1)^2+1\)，即 \(2a+2=0\)，得 \(a=-1\). 因此选 D.
\end{solution}

\begin{problem}
已知 \(m\)，\(n\) 为异面直线，\(m \subset\) 平面 \(\alpha\)，\(n \subset\) 平面 \(\beta\)，\(\alpha \cap \beta = l\)，则 \(l\)（\quad）

\choices
    {与 \(m\)，\(n\) 都相交}
    {与 \(m\)，\(n\) 中至少一条相交}
    {与 \(m\)，\(n\) 都不相交}
    {至多与 \(m\)，\(n\) 中的一条相交}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
若交线 \(l\) 不与直线 \(m\) 相交，由于 \(l\) 与 \(m\) 同在平面 \(\alpha\) 内，只能有 \(l\parallel m\)；同理，若 \(l\) 不与直线 \(n\) 相交，则 \(l\parallel n\). 若 \(l\) 同时不与 \(m\)，\(n\) 相交，就有 \(m\parallel l\parallel n\)，这与 \(m\)，\(n\) 为异面直线矛盾. 所以 \(l\) 至少与 \(m\)，\(n\) 中的一条相交，选 B.
\end{solution}

\begin{problem}
函数 \( y = a^x \) 在 \([0,1]\) 上的最大值与最小值之和为 \(3\)，则 \( a = \)（\quad）

\choices
    {\(\frac{1}{2}\)}
    {\(2\)}
    {\(4\)}
    {\(\frac{1}{4}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
当 \(a>1\) 时，函数 \(y=a^x\) 在 \([0,1]\) 上递增，最小值为 \(1\)，最大值为 \(a\)，所以 \(1+a=3\)，得 \(a=2\). 当 \(0<a<1\) 时，函数递减，最大值为 \(1\)，最小值为 \(a\)，仍需 \(1+a=3\)，这与 \(0<a<1\) 矛盾；当 \(a=1\) 时最大值与最小值之和为 \(2\). 因此 \(a=2\)，选 B.
\end{solution}

\begin{problem}
不等式 \((1 + x)(1 - |x|) > 0\) 的解集是（\quad）

\choices
    {\(\{x\mid 0\leqslant x < 1\}\)}
    {\(\{x\mid x < 0 \text{ 且 } x\neq -1\}\)}
    {\(\{x\mid -1 < x < 1\}\)}
    {\(\{x\mid x < 1 \text{ 且 } x\neq -1\}\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
当 \(x<0\) 时，\(|x|=-x\)，原不等式化为 \((1+x)^2>0\)，所以解为 \(x<0\) 且 \(x\neq-1\). 当 \(x\geqslant0\) 时，原不等式化为 \((1+x)(1-x)>0\)；因 \(1+x>0\)，所以 \(0\leqslant x<1\). 合并两种情况，解集为 \(\{x\mid x<1\text{ 且 }x\neq-1\}\)，选 D.
\end{solution}

\begin{problem}
在 \((0,2\pi)\) 内，使 \(\sin x > \cos x\) 成立的 \(x\) 的取值范围是（\quad）

\choices
    {\(\left(\frac{\pi}{4},\frac{\pi}{2}\right)\cup \left(\pi ,\frac{5\pi}{4}\right)\)}
    {\(\left(\frac{\pi}{4},\pi\right)\)}
    {\(\left(\frac{\pi}{4},\frac{5\pi}{4}\right)\)}
    {\(\left(\frac{\pi}{4},\pi\right)\cup \left(\frac{5\pi}{4},\frac{3\pi}{2}\right)\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
由 \(\sin x>\cos x\) 得 \(\sin x-\cos x>0\)，即 \(\sqrt{2}\sin\left(x-\frac{\pi}{4}\right)>0\). 在 \(0<x<2\pi\) 内，正弦值为正的部分满足 \(0<x-\frac{\pi}{4}<\pi\)，从而 \(\frac{\pi}{4}<x<\frac{5\pi}{4}\). 因此选 C.
\end{solution}

\begin{problem}
设集合 \(M = \left\{x \mid x = \frac{k}{2} + \frac{1}{4}, k \in \Z\right\}\)，\(N = \left\{x \mid x = \frac{k}{4} + \frac{1}{2}, k \in \Z\right\}\)，则（\quad）

\choices
    {\(M = N\)}
    {\(M \subseteq N\)}
    {\(M \supseteq N\)}
    {\(M \cap N = \varnothing\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
集合 \(M\) 中的任意元素可写为 \(x=\frac{2k+1}{4}\). 取整数 \(j=2k-1\)，则 \(x=\frac{j}{4}+\frac{1}{2}\)，所以 \(M\subseteq N\). 又 \(\frac{1}{2}\in N\)，而 \(\frac{1}{2}\notin M\)，故 \(M\subseteq N\) 且 \(M\neq N\)，选 B.
\end{solution}

\begin{problem}
椭圆 \(5x^{2} + ky^{2} = 5\) 的一个焦点是 \((0,2)\)，那么 \(k =\)（\quad）

\choices
    {\(-1\)}
    {\(1\)}
    {\(\sqrt{5}\)}
    {\(-\sqrt{5}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
将椭圆方程除以 \(5\)，得 \(x^2+\frac{k}{5}y^2=1\). 由于焦点在 \(y\) 轴上，椭圆的长半轴沿 \(y\) 轴，横向半轴为 \(1\)，纵向半轴的平方为 \(\frac{5}{k}\). 设焦距的一半为 \(c\)，则 \(c=2\)，由椭圆关系 \(c^2=b^2-1\) 得 \(b^2=5\). 因而 \(\frac{5}{k}=5\)，所以 \(k=1\)，选 B.
\end{solution}

\begin{problem}
正六棱柱 \(ABCDEF - A_{1}B_{1}C_{1}D_{1}E_{1}F_{1}\) 的底面边长为 \(1\)，侧棱长为 \(\sqrt{2}\)，则这个棱柱侧面对角线 \(E_{1}D\) 与 \(BC_{1}\) 所成的角是（\quad）

\choices
    {\(90^{\circ}\)}
    {\(60^{\circ}\)}
    {\(45^{\circ}\)}
    {\(30^{\circ}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
在底面内取坐标 \(A(1,0,0)\)、\(B\left(\frac12,\frac{\sqrt3}{2},0\right)\)、\(C\left(-\frac12,\frac{\sqrt3}{2},0\right)\)、\(D(-1,0,0)\)、\(E\left(-\frac12,-\frac{\sqrt3}{2},0\right)\)，并令上底面各点的 \(z\) 坐标为 \(\sqrt2\). 则
\(\overrightarrow{DE_1}=\left(\frac12,-\frac{\sqrt3}{2},\sqrt2\right)\)，\(\overrightarrow{BC_1}=(-1,0,\sqrt2)\). 因此
\(\overrightarrow{DE_1}\cdot\overrightarrow{BC_1}=\frac32\)，且 \(|DE_1|=|BC_1|=\sqrt3\).
设两直线所成的锐角为 \(\theta\)，则 \(\cos\theta=\frac{3/2}{\sqrt3\cdot\sqrt3}=\frac12\)，所以 \(\theta=60^{\circ}\)，选 B.
\end{solution}

\begin{problem}
函数 \( y = x^{2} + bx + c \)（\(x \in [0, +\infty)\)）是单调函数的充要条件是（\quad）

\choices
    {\(b \geqslant 0\)}
    {\(b \leqslant 0\)}
    {\(b > 0\)}
    {\(b < 0\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
任取 \(0\leqslant x<y\)，有
\(f(y)-f(x)=(y-x)(x+y+b)\).
若 \(b\geqslant0\)，则 \(x+y+b>0\)，所以 \(f(y)>f(x)\)，函数在 \([0,+\infty)\) 上单调递增. 若 \(b<0\)，取 \(x=0\)，\(y=-\frac b2\)，则 \(f(y)-f(0)<0\)；又取 \(x=-b\)，\(y=-b+1\)，则 \(f(y)-f(x)>0\). 此时函数既不单调递增也不单调递减. 所以充要条件是 \(b\geqslant0\)，选 A.
\end{solution}

\begin{problem}
已知 \(0 < x < y < a < 1\)，则有（\quad）

\choices
    {\(\log_a(xy) < 0\)}
    {\(0 < \log_a(xy) < 1\)}
    {\(1 < \log_a(xy) < 2\)}
    {\(\log_a(xy) > 2\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
由 \(0<x<a\)，\(0<y<a\)，得 \(0<xy<a^2\). 因为 \(0<a<1\)，函数 \(\log_a t\) 在正数范围内递减，所以
\(\log_a(xy)>\log_a(a^2)=2\).
因此选 D.
\end{solution}

\begin{problem}
从正方体的 \(6\) 个面中选取 \(3\) 个面，其中有 \(2\) 个面不相邻的选法共有（\quad）

\choices
    {\(8\) 种}
    {\(12\) 种}
    {\(16\) 种}
    {\(20\) 种}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
正方体的面可分成 \(3\) 对相对的面. 若所选的 \(3\) 个面中有一对不相邻的面，它们必须是一对相对面；先选这对面有 \(3\) 种选法，再从剩下的 \(4\) 个面中选第 \(3\) 个面. 三个面中不可能同时出现两对相对面，所以选法总数为 \(3\times4=12\). 因此选 B.
\end{solution}

\begin{problem}
平面直角坐标系中，\(O\) 为坐标原点，已知两点 \(A(3,1)\)，\(B(-1,3)\)，若点 \(C\) 满足 \(\overrightarrow{OC} = \alpha \overrightarrow{OA} + \beta \overrightarrow{OB}\)，其中 \(\alpha\)，\(\beta \in \R\)，且 \(\alpha + \beta = 1\)，则点 \(C\) 的轨迹方程为（\quad）

\choices
    {\( 3x + 2y - 11 = 0 \)}
    {\((x - 1)^{2} + (y - 2)^{2} = 5\)}
    {\(2x - y = 0\)}
    {\(x + 2y - 5 = 0\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
由 \(\alpha+\beta=1\)，有
\[
\overrightarrow{OC}=\alpha\overrightarrow{OA}+(1-\alpha)\overrightarrow{OB}
\]
所以 \(C\) 在线段 \(AB\) 所在直线上. 直线 \(AB\) 的斜率为 \(\frac{3-1}{-1-3}=-\frac12\)，故其方程为
\(y-1=-\frac12(x-3)\)，
即 \(x+2y-5=0\). 反过来，直线上的任意点都可写成 \(\alpha A+(1-\alpha)B\) 的形式，因此轨迹方程为 \(x+2y-5=0\)，选 D.
\end{solution}

\begin{problem}
据新华社 2002 年 3 月 12 日电，1985 年到 2000 年间，我国农村人均居住面积如图所示，

其中，从\fillinblank{}年\fillinblank{}年的五年间增长最快.

\bitmapfigure[max width=0.58\linewidth]{img/2002/new_curriculum_liberal/q13_fig1.png}
\end{problem}

\begin{answer}
\(1995\)，\(2000\)
\end{answer}

\begin{solution}
从图中读得农村人均居住面积依次为 \(14.7\)、\(17.8\)、\(21.0\)、\(24.8\). 相邻五年间的增长量分别为 \(17.8-14.7=3.1\)、\(21.0-17.8=3.2\)、\(24.8-21.0=3.8\). 最大增长量是 \(3.8\)，对应从 1995 年到 2000 年，所以填 \(1995\)，\(2000\).
\end{solution}

\begin{problem}
已知 \(\sin 2\alpha = -\sin \alpha\)（\(\alpha \in \left(\frac{\pi}{2}, \pi\right)\)），则 \(\cot \alpha =\)\fillinblank{}.
\end{problem}

\begin{answer}
\(-\frac{\sqrt{3}}{3}\)
\end{answer}

\begin{solution}
由 \(\alpha \in \left(\frac{\pi}{2},\pi\right)\) 可知 \(\sin\alpha>0\). 由二倍角公式，原条件化为
\(2\sin\alpha\cos\alpha=-\sin\alpha\).
两边同除以 \(\sin\alpha\)，得 \(2\cos\alpha=-1\)，所以 \(\cos\alpha=-\frac{1}{2}\). 又因为 \(\alpha\) 在第二象限，\(\sin\alpha=\frac{\sqrt{3}}{2}\)，从而
\(\cot\alpha=\frac{\cos\alpha}{\sin\alpha}=-\frac{1}{\sqrt{3}}=-\frac{\sqrt{3}}{3}\).
\end{solution}

\begin{problem}
甲，乙两种冬小麦试验品种连续 \(5\) 年的平均单位面积产量如下（单位：\(\mathrm{t}/\mathrm{km}^2\)）：

\begin{center}
\begin{tabular}{|c|c|c|c|c|c|}
\hline
品种 & 第\(1\)年 & 第\(2\)年 & 第\(3\)年 & 第\(4\)年 & 第\(5\)年 \\
\hline
甲 & \(9.8\) & \(9.9\) & \(10.1\) & \(10\) & \(10.2\) \\
\hline
乙 & \(9.4\) & \(10.3\) & \(10.8\) & \(9.7\) & \(9.8\) \\
\hline
\end{tabular}
\end{center}

其中产量比较稳定的小麦品种是\fillinblank{}.
\end{problem}

\begin{answer}
甲
\end{answer}

\begin{solution}
两品种五年的平均产量均为 \(10\). 以平均数为基准，甲的偏差为 \(-0.2\)、\(-0.1\)、\(0.1\)、\(0\)、\(0.2\)，偏差平方和为 \(0.10\)，方差为 \(0.10/5=0.02\). 乙的偏差为 \(-0.6\)、\(0.3\)、\(0.8\)、\(-0.3\)、\(-0.2\)，偏差平方和为 \(1.22\)，方差为 \(1.22/5=0.244\). 甲的方差和标准差都较小，故甲的产量比较稳定.
\end{solution}

\begin{problem}
设函数 \(f(x)\) 在 \((-\infty, +\infty)\) 内有定义，下列函数

\begin{circlelist}
\item \(y = -|f(x)|\)；
\item \(y = x f(x^{2})\)；
\item \(y = -f(-x)\)；
\item \(y = f(x) - f(-x)\).
\end{circlelist}
其中必为奇函数的有\fillinblank{}.（要求填写正确答案的序号）
\end{problem}

\begin{answer}
\(\circled{2}\)，\(\circled{4}\)
\end{answer}

\begin{solution}
对第（1）个函数，取恒正函数 \(f(x)=1\)，则 \(-|f(x)|=-1\) 不是奇函数，所以第（1）个函数不一定为奇函数. 对第（2）个函数，令 \(g(x)=xf(x^2)\)，则 \(g(-x)=-xf((-x)^2)=-xf(x^2)=-g(x)\)，所以它必为奇函数. 对第（3）个函数，仍取 \(f(x)=1\)，则 \(-f(-x)=-1\) 不是奇函数，所以它不一定为奇函数. 对第（4）个函数，令 \(h(x)=f(x)-f(-x)\)，则 \(h(-x)=f(-x)-f(x)=-h(x)\)，所以它必为奇函数. 因此应填 \(\circled{2}\)，\(\circled{4}\).
\end{solution}

\section{解答题}
\begin{problem}
在等比数列 \(\{a_{n}\}\) 中，已知 \(a_6 - a_4 = 24\)，\(a_3a_5 = 64\)，求 \(\{a_{n}\}\) 前 \(8\) 项的和 \(S_{8}\).
\end{problem}

\begin{solution}
设等比数列的公比为 \(q\). 因为 \(a_3a_5=a_4^2\)，所以 \(a_4^2=64\)，即 \(a_4=8\) 或 \(a_4=-8\). 又由等比数列的性质，\(a_6=a_4q^2\)，因此
\(a_4(q^2-1)=24\).
当 \(a_4=8\) 时，\(q^2-1=3\)，得 \(q=2\) 或 \(q=-2\)；当 \(a_4=-8\) 时，\(q^2-1=-3\)，这不可能，因为 \(q^2\geqslant0\).

当 \(q=2\) 时，\(a_1=a_4/q^3=1\)，故前八项为 \(1\)，\(2\)，\(4\)，\(8\)，\(16\)，\(32\)，\(64\)，\(128\)，从而 \(S_8=255\). 当 \(q=-2\) 时，\(a_1=a_4/q^3=-1\)，从而
\(S_8=a_1\frac{1-q^8}{1-q}=-1\cdot\frac{1-256}{3}=85\).
所以 \(S_8=255\) 或 \(S_8=85\).
\end{solution}

\begin{problem}
已知 \(\sin^2 2\alpha + \sin 2\alpha \cos \alpha - \cos 2\alpha = 1\)（\(\alpha \in \left(0, \frac{\pi}{2}\right)\)），求 \(\sin \alpha\)，\(\tan \alpha\) 的值.
\end{problem}

\begin{solution}
令 \(s=\sin\alpha\)，\(c=\cos\alpha\). 由 \(0<\alpha<\frac{\pi}{2}\) 知 \(0<s<1\)，且 \(c^2=1-s^2\). 把 \(\sin2\alpha=2sc\)、\(\cos2\alpha=c^2-s^2\) 代入原式，得
\[
4s^2c^2+2sc^2-(c^2-s^2)=1
\]
再代入 \(c^2=1-s^2\)，并移项化简，得到
\[
2s^4+s^3-3s^2-s+1=0
\]
因式分解得
\[
(2s-1)(s-1)(s+1)^2=0
\]
由于 \(0<s<1\)，只能取 \(s=\frac{1}{2}\)，于是 \(c=\frac{\sqrt{3}}{2}\). 因此 \(\sin\alpha=\frac{1}{2}\)，\(\tan\alpha=\frac{\sin\alpha}{\cos\alpha}=\frac{\sqrt{3}}{3}\).
\end{solution}

\section{选做题：解答题}
\begin{problem}[甲]
如图，正三棱柱 \(ABC - A_{1}B_{1}C_{1}\) 的底面边长为 \(a\)，侧棱长为 \(\sqrt{2}a\).

\begin{enumerate}
\item 建立适当的坐标系，并写出点 \(A\)，\(B\)，\(A_{1}\)，\(C_{1}\) 的坐标；
\item 求 \(AC_{1}\) 与侧面 \(ABB_{1}A_{1}\) 所成的角.
\end{enumerate}

\bitmapfigure[max width=0.36\linewidth]{img/2002/new_curriculum_liberal/q19_fig1.png}
\FloatBarrier
\end{problem}

\begin{solution}
\begin{enumerate}
\item 取 \(A\) 为原点，\(AB\) 所在方向为 \(y\) 轴正向，\(AA_1\) 所在方向为 \(z\) 轴正向，并取 \(x\) 轴垂直于侧面 \(ABB_1A_1\). 于是可设 \(A(0,0,0)\)，\(B(0,a,0)\)，\(A_1(0,0,\sqrt{2}a)\)，\(C_1\left(-\frac{\sqrt{3}}{2}a,\frac{a}{2},\sqrt{2}a\right)\).
\item 侧面 \(ABB_1A_1\) 的方程为 \(x=0\)，而
\(\overrightarrow{AC_1}=\left(-\frac{\sqrt{3}}{2}a,\frac{a}{2},\sqrt{2}a\right)\)，且 \(|AC_1|=\sqrt{3}a\).
设直线 \(AC_1\) 与平面 \(ABB_1A_1\) 所成的角为 \(\varphi\). 直线方向向量在该平面法向方向上的分量的绝对值为 \(\frac{\sqrt{3}}{2}a\)，所以
\(\sin\varphi=\frac{\frac{\sqrt{3}}{2}a}{\sqrt{3}a}=\frac{1}{2}\).
由于所成角取锐角，得 \(\varphi=30^{\circ}\).
\end{enumerate}
\end{solution}

\begin{problem}[乙]
如图，正方形 \(ABCD\)，\(ABEF\) 的边长都是 \(1\)，而且平面 \(ABCD\)，\(ABEF\) 互相垂直. 点 \(M\) 在 \(AC\) 上移动，点 \(N\) 在 \(BF\) 上移动，若 \(CM = BN = a\)（\(0 < a < \sqrt{2}\)）.

\begin{enumerate}
\item 求 \(MN\) 的长；
\item \(a\) 为何值时，\(MN\) 的长最小；
\item 当 \(MN\) 的长最小时，求面 \(MNA\) 与面 \(MNB\) 所成二面角 \(\alpha\) 的大小.
\end{enumerate}

\bitmapfigure[max width=0.36\linewidth]{img/2002/new_curriculum_liberal/q20_fig1.png}
\FloatBarrier
\end{problem}

\begin{solution}
\begin{enumerate}
\item 建立直角坐标系，取 \(A(0,0,0)\)，\(B(1,0,0)\)，\(C(1,1,0)\)，\(D(0,1,0)\)，\(E(1,0,1)\)，\(F(0,0,1)\).
此时平面 \(ABCD\) 为 \(z=0\)，平面 \(ABEF\) 为 \(y=0\)，二者互相垂直. 因为 \(|AC|=|BF|=\sqrt{2}\)，且 \(CM=BN=a\)，所以
\[
M=C+\frac{a}{\sqrt{2}}\overrightarrow{CA}=\left(1-\frac{a}{\sqrt{2}},1-\frac{a}{\sqrt{2}},0\right)
\]
\[
N=B+\frac{a}{\sqrt{2}}\overrightarrow{BF}=\left(1-\frac{a}{\sqrt{2}},0,\frac{a}{\sqrt{2}}\right)
\]
因此
\[
MN^2=\left(1-\frac{a}{\sqrt{2}}\right)^2+\left(\frac{a}{\sqrt{2}}\right)^2=\left(a-\frac{\sqrt{2}}{2}\right)^2+\frac{1}{2}
\]
故 \(MN=\sqrt{\left(a-\frac{\sqrt{2}}{2}\right)^2+\frac{1}{2}}\).
\item 由上式可知，当 \(a=\frac{\sqrt{2}}{2}\) 时，\(MN\) 取得最小值 \(\frac{\sqrt{2}}{2}\).
\item 当 \(a=\frac{\sqrt{2}}{2}\) 时，\(M\left(\frac{1}{2},\frac{1}{2},0\right)\)，\(N\left(\frac{1}{2},0,\frac{1}{2}\right)\). 取 \(MN\) 的中点 \(H\). 由中点公式得 \(H\left(\frac{1}{2},\frac{1}{4},\frac{1}{4}\right)\)，并且
\(\overrightarrow{HA}=\left(-\frac{1}{2},-\frac{1}{4},-\frac{1}{4}\right)\)，
\(\overrightarrow{HB}=\left(\frac{1}{2},-\frac{1}{4},-\frac{1}{4}\right)\).
由前述坐标可得 \(AM=AN=MN=BM=BN=\frac{\sqrt{2}}{2}\)，所以 \(\triangle AMN\) 与 \(\triangle BMN\) 都是正三角形，从而 \(HA\perp MN\)，\(HB\perp MN\). 因此 \(\angle AHB\) 是所求二面角的平面角. 于是
\[
\cos\alpha=\frac{\overrightarrow{HA}\cdot\overrightarrow{HB}}{|HA||HB|}
=\frac{-\frac{1}{8}}{\frac{3}{8}}=-\frac{1}{3}
\]
所以 \(\alpha=\arccos\left(-\frac{1}{3}\right)\).
\end{enumerate}
\end{solution}

\section{解答题}
\begin{problem}
某单位 \(6\) 个员工借助互联网开展工作，每个员工上网的概率都是 \(0.5\)（相互独立）.

\begin{enumerate}
\item 求至少 \(3\) 人同时上网的概率；
\item 至少几人同时上网的概率小于 \(0.3\)？
\end{enumerate}
\end{problem}

\begin{solution}
设同时上网的员工人数为随机变量 \(X\). 由于每个员工上网与否相互独立，且上网概率均为 \(0.5\)，所以恰有 \(k\) 人同时上网的概率为 \(\frac{\mathrm C_6^k}{2^6}\).
\begin{enumerate}
\item 至少 \(3\) 人同时上网的概率为
\[
P(X\geqslant3)=\frac{\mathrm C_6^3+\mathrm C_6^4+\mathrm C_6^5+\mathrm C_6^6}{2^6}=1-\frac{\mathrm C_6^0+\mathrm C_6^1+\mathrm C_6^2}{2^6}=1-\frac{1+6+15}{64}=\frac{21}{32}
\]
\item 至少 \(4\) 人同时上网的概率为
\[
P(X\geqslant4)=\frac{\mathrm C_6^4+\mathrm C_6^5+\mathrm C_6^6}{64}=\frac{15+6+1}{64}=\frac{22}{64}>0.3
\]
而至少 \(5\) 人同时上网的概率为
\[
P(X\geqslant5)=\frac{\mathrm C_6^5+\mathrm C_6^6}{64}=\frac{6+1}{64}=\frac{7}{64}<0.3
\]
因此，至少 \(5\) 人同时上网时，该概率小于 \(0.3\).
\end{enumerate}
\end{solution}

\begin{problem}
已知 \(a > 0\)，函数 \(f(x) = x^3 - a\)（\(x \in (0, +\infty)\)）. 设 \(x_1 > 0\)，记曲线 \(y = f(x)\) 在点 \((x_1, f(x_1))\) 处的切线为 \(l\).

\begin{enumerate}
\item 求 \(l\) 的方程；
\item 设 \(l\) 与 \(x\) 轴交点为 \((x_{2}, 0)\). 证明：
\begin{enumerate}
\item \(x_{2} \geqslant a^{\frac{1}{3}}\)；
\item 若 \(x_{2} > a^{\frac{1}{3}}\)，则 \(a^{\frac{1}{3}} < x_{2} < x_{1}\).
\end{enumerate}
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 因为 \(f'(x)=3x^2\)，所以曲线在 \((x_1,f(x_1))\) 处的切线斜率为 \(3x_1^2\). 由点斜式，切线方程为
\[
y-(x_1^3-a)=3x_1^2(x-x_1)
\]
即 \(l\) 的方程为 \(y=3x_1^2x-2x_1^3-a\).
\item 令 \(y=0\)，得切线与 \(x\) 轴交点的横坐标
\[
x_2=\frac{2x_1^3+a}{3x_1^2}
\]
\begin{enumerate}
\item 令 \(t=a^{\frac{1}{3}}>0\)，则
\[
x_2-t=\frac{2x_1^3+t^3-3tx_1^2}{3x_1^2}=\frac{(x_1-t)^2(2x_1+t)}{3x_1^2}\geqslant0
\]
所以 \(x_2\geqslant a^{\frac{1}{3}}\).
\item 原题给出的条件 \(x_2>a^{\frac{1}{3}}\) 只能说明 \(x_1\neq a^{\frac{1}{3}}\)，并不能推出 \(x_2<x_1\). 事实上
\[
x_1-x_2=\frac{x_1^3-a}{3x_1^2}=\frac{(x_1-t)(x_1^2+tx_1+t^2)}{3x_1^2}
\]
当 \(x_1>t\) 时，确有 \(t<x_2<x_1\)；但当 \(0<x_1<t\) 时，反而有 \(x_2>x_1>0\). 例如取 \(a=1\)、\(x_1=\frac{1}{2}\)，则 \(x_2=\frac{5}{3}>1=a^{\frac{1}{3}}\)，但 \(x_2>x_1\). 因此，第（2）小问按原题字面不成立，不能给出其所要求的证明；若补充条件 \(x_1>a^{\frac{1}{3}}\)，则上式立即给出题中结论.
\end{enumerate}
\end{enumerate}
\end{solution}

\begin{problem}
已知两点 \(M(-1,0)\)，\(N(1,0)\)，且点 \(P\) 使 \(\overrightarrow{MP} \cdot \overrightarrow{MN}\)，\(\overrightarrow{PM} \cdot \overrightarrow{PN}\)，\(\overrightarrow{NM} \cdot \overrightarrow{NP}\) 成公差小于零的等差数列.

\begin{enumerate}
\item 点 \(P\) 的轨迹是什么曲线？
\item 若点 \(P\) 坐标为 \((x_0, y_0)\)，记 \(\theta\) 为 \(\overrightarrow{PM}\) 与 \(\overrightarrow{PN}\) 的夹角，求 \(\tan \theta\).
\end{enumerate}
\end{problem}

\begin{solution}
设 \(P(x,y)\). 则
\(\overrightarrow{MP}\cdot\overrightarrow{MN}=2(x+1)\)，
\(\overrightarrow{PM}\cdot\overrightarrow{PN}=x^2+y^2-1\)，
\(\overrightarrow{NM}\cdot\overrightarrow{NP}=2(1-x)\).
由于这三个数按所列顺序成等差数列，故中间项是首、末两项的平均数，即
\(2(x^2+y^2-1)=2(x+1)+2(1-x)=4\)，
从而 \(x^2+y^2=3\). 等差数列的公差为
\((x^2+y^2-1)-2(x+1)=-2x\).
题设公差小于零，所以 \(x>0\).
\begin{enumerate}
\item 点 \(P\) 的轨迹是圆 \(x^2+y^2=3\) 的右半圆，即 \(x^2+y^2=3\)，\(x>0\).
\item 在轨迹上，
\(\overrightarrow{PM}\cdot\overrightarrow{PN}=2\)，
且 \(|PM|^2=(x+1)^2+y^2=4+2x\)，\(|PN|^2=(x-1)^2+y^2=4-2x\).
由于 \(\overrightarrow{PM}\cdot\overrightarrow{PN}=2>0\)，所以 \(\theta\in\left[0,\frac{\pi}{2}\right)\). 由两向量夹角的正切公式，
\[
\tan\theta=\frac{\left|(-x-1)(-y)-(-y)(1-x)\right|}{\overrightarrow{PM}\cdot\overrightarrow{PN}}=\frac{2|y|}{2}=|y|
\]
若 \(P=(x_0,y_0)\)，则
\(\tan\theta=|y_0|=\sqrt{3-x_0^2}\).
\end{enumerate}
\end{solution}
