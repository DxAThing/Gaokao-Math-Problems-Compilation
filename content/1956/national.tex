\chapter{1956年全国卷}

\begin{problem}
利用对数性质，计算 \(\lg^{2}5+\lg2\cdot\lg50\).

\end{problem}

\begin{solution}
因为 \(\lg2=1-\lg5\)，\(\lg50=1+\lg5\)，所以
\[
\begin{aligned}
\text{原式}&=\lg^{2}5+\lg2\cdot\lg50=\lg^{2}5+\left( 1-\lg5 \right)\left( 1+\lg5 \right)\\
&=\lg^{2}5+1-\lg^{2}5=1
\end{aligned}
\]
\end{solution}

\begin{problem}
设 \(m\) 是实数，求证方程 \(2x^{2}-\left( 4m-1 \right)x-m^{2}-m=0\) 的两个根必定都是实数.

\end{problem}

\begin{solution}
方程的判别式 \(\Delta=\left( 4m-1 \right)^{2}+8\left( m^{2}+m \right)=24m^{2}+1\). 因为 \(m^{2}\ge0\)，所以 \(\Delta>0\)，所以两个根都是实数.
\end{solution}

\begin{problem}
设 \(M\) 是 \(\triangle ABC\) 的边 \(AC\) 的中点，过 \(M\) 作直线交 \(AB\) 直线于 \(E\)，过 \(B\) 作直线平行于 \(ME\) 交 \(AC\) 直线于 \(F\). 求证 \(\triangle AEF\) 的面积等于 \(\triangle ABC\) 的面积的一半.

\end{problem}

\begin{solution}
连结 \(BM\).

\bitmapfigure[max width=4.4cm,max totalheight=3.2cm]{img/1956/national/q3_solution_fig1.png}

因为 \(\triangle AEF\) 的面积等于 \(\triangle ABF\) 的面积加 \(\triangle BEF\) 的面积，且 \(BF\parallel ME\)，所以 \(\triangle BEF\) 的面积等于 \(\triangle BFM\) 的面积. 因此 \(\triangle AEF\) 的面积等于 \(\triangle BAM\) 的面积.

因为 \(\triangle BAM\) 的面积等于 \(\frac{1}{2}\triangle ABC\) 的面积，所以 \(\triangle AEF\) 的面积等于 \(\frac{1}{2}\triangle ABC\) 的面积.
\end{solution}

\begin{problem}
一个三角形三边的长分别是 \(3\text{尺}\)，\(4\text{尺}\) 及 \(\sqrt{37}\text{尺}\). 求这个三角形的最大角的度数.

\end{problem}

\begin{solution}
最大角对最大边，由余弦定律得
\[
\begin{aligned}
37&=3^{2}+4^{2}-2\cdot3\cdot4\cos\theta\\
\cos\theta&=\frac{9+16-37}{24}=-\frac{1}{2}
\end{aligned}
\]
所以 \(\theta=120^{\circ}\).
\end{solution}

\begin{problem}
设 \(\tan\alpha\) 与 \(\tan\beta\) 是方程 \(x^{2}+6x+7=0\) 的两个根，求证 \(\sin\left( \alpha+\beta \right)=\cos\left( \alpha+\beta \right)\).

\end{problem}

\begin{solution}
\(\tan\alpha+\tan\beta=-6\)，\(\tan\alpha\cdot\tan\beta=7\). 所以
\[
\frac{\sin\left( \alpha+\beta \right)}{\cos\left( \alpha+\beta \right)}=\tan\left( \alpha+\beta \right)=\frac{\tan\alpha+\tan\beta}{1-\tan\alpha\tan\beta}=\frac{-6}{-6}=1
\]
所以 \(\sin\left( \alpha+\beta \right)=\cos\left( \alpha+\beta \right)\).
\end{solution}

\begin{problem}
解联立方程 \(\begin{cases}
7\sqrt{x+y}-x-y=12,\\
x^{2}+y^{2}=136.
\end{cases}\)

\end{problem}

\begin{solution}
\[
\begin{cases}
7\sqrt{x+y}-x-y=12 \qquad (1)\\
x^{2}+y^{2}=136 \qquad (2)
\end{cases}
\]

由（1）移项，得 \(\left( x+y \right)-7\sqrt{x+y}+12=0\)，即 \(\left( \sqrt{x+y}-3 \right)\left( \sqrt{x+y}-4 \right)=0\). 所以 \(\sqrt{x+y}=3\) 或 \(\sqrt{x+y}=4\). 两边分别平方，得 \(x+y=9\) 或 \(x+y=16\).

（1）当 \(x+y=9\) 时，
\[
\left( x-y \right)^{2}=2\left( x^{2}+y^{2} \right)-\left( x+y \right)^{2}=272-81=191
\]
所以 \(x-y=\pm\sqrt{191}\). 因而两组解为 \(x_{1}=\frac{9+\sqrt{191}}{2}\)，\(y_{1}=\frac{9-\sqrt{191}}{2}\)，以及 \(x_{2}=\frac{9-\sqrt{191}}{2}\)，\(y_{2}=\frac{9+\sqrt{191}}{2}\).

（2）当 \(x+y=16\) 时，
\[
\left( x-y \right)^{2}=2\left( x^{2}+y^{2} \right)-\left( x+y \right)^{2}=272-256=16
\]
所以 \(x-y=\pm4\). 因而两组解为 \(x_{3}=10\)，\(y_{3}=6\)，以及 \(x_{4}=6\)，\(y_{4}=10\).

经检验，以上四组解均为原方程的解.
\end{solution}

\begin{problem}
设 \(P\) 是等边三角形 \(ABC\) 外接圆 \(\myarc{BC}\) 上的一点，求证 \(PA^{2}=AB^{2}+PB\cdot PC\).

\end{problem}

\begin{solution}
令 \(AP\) 与 \(BC\) 的交点是 \(M\).

\bitmapfigure[max width=4.0cm,max totalheight=4.2cm]{img/1956/national/q7_solution_fig1.png}

因为 \(\angle APB=\angle ABM\)，所以 \(\triangle ABP\sim\triangle AMB\)，所以 \(AP\cdot AM=AB^{2}\).

因为 \(\angle APC=\angle BPM\)，\(\angle PAC=\angle PBM\)，所以 \(\triangle ACP\sim\triangle BMP\)，所以 \(AP\cdot MP=PB\cdot PC\).

两式相加，得 \(AP\cdot AM+AP\cdot MP=AB^{2}+PB\cdot PC\)，即 \(AP^{2}=AB^{2}+PB\cdot PC\).
\end{solution}

\begin{problem}
有一四棱柱体，底面 \(ABCD\) 为菱形，\(\angle A'AB=\angle A'AD\)（如图），求证平面 \(A'ACC'\) 垂直于底面 \(ABCD\).

\bitmapfigure[max width=4.8cm,max totalheight=3.2cm]{img/1956/national/q8_fig1.png}

\end{problem}

\begin{solution}
设 \(AC\) 与 \(BD\) 交于 \(O\). 因为 \(ABCD\) 是菱形，所以 \(AC\perp BD\)，从而 \(AO\perp BD\).

\bitmapfigure[max width=4.4cm,max totalheight=3.2cm]{img/1956/national/q8_solution_fig1.png}

因为 \(AB=AD\)，\(A'A\) 是公共边，且 \(\angle A'AB=\angle A'AD\)，所以 \(\triangle A'AB \cong \triangle A'AD\). 因此 \(A'B=A'D\).

又因为菱形的对角线互相平分，所以 \(O\) 是 \(BD\) 的中点. 在等腰三角形 \(A'BD\) 中，\(A'O\perp BD\).

因为 \(AO\) 和 \(A'O\) 是平面 \(A'ACC'\) 内两条相交直线，且都垂直于 \(BD\)，所以 \(BD\) 垂直于平面 \(A'ACC'\).

又因为 \(BD\) 在平面 \(ABCD\) 内，所以平面 \(A'ACC'\) 垂直于平面 \(ABCD\).
\end{solution}

\begin{problem}
若三角形的三个角成等差级数，则其中一定有一个角是 \(60^{\circ}\)；若这样的三角形的三边又成等比级数，则三个角都是 \(60^{\circ}\)，试证明之.

\end{problem}

\begin{solution}
设三角形的三个角为 \(A\)，\(B\)，\(C\). 因为 \(A-B=B-C\)，且 \(A+B+C=180^{\circ}\)，所以 \(B=60^{\circ}\).

再设三边的长为 \(a\)，\(aq\)，\(aq^{2}\)，则长为 \(aq\) 的边所对的角是 \(60^{\circ}\). 根据余弦定理，
\[
\left( aq \right)^{2}=a^{2}+\left( aq^{2} \right)^{2}-2a\cdot aq^{2}\cos60^{\circ}
\]
即 \(a^{2}q^{2}=a^{2}+a^{2}q^{4}-a^{2}q^{2}\)，所以 \(q^{4}-2q^{2}+1=0\)，即 \(\left( q^{2}-1 \right)^{2}=0\). 因此 \(q=\pm1\). 但 \(q=-1\) 不合题意，所以 \(q=1\). 因而三边相等，所以三个角都是 \(60^{\circ}\).
\end{solution}
