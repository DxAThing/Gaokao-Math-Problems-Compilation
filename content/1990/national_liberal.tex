\begin{problem}
有四个数，其中前三个数成等差数列，后三个数成等比数列，并且第一个数与第四个数的和是 \(16\)，第二个数与第三个数的和是 \(12\). 求这四个数.
\end{problem}

\begin{solution}
设四个数依次为 \(x\)，\(y\)，\(z\)，\(w\). 由题意，\(x\)，\(y\)，\(z\) 成等差数列，\(y\)，\(z\)，\(w\) 成等比数列，所以
\[
2y=x+z,\qquad z^2=yw.
\]
又由 \(x+w=16\)，\(y+z=12\)，得 \(z=12-y\)，\(x=2y-z=3y-12\)，\(w=16-x=28-3y\). 代入等比数列的关系式，得
\[
(12-y)^2=y(28-3y),
\]
即
\[
4y^2-52y+144=0,\qquad y^2-13y+36=0.
\]
因此 \(y=4\) 或 \(y=9\). 当 \(y=4\) 时，\((x,z,w)=(0,8,16)\)；当 \(y=9\) 时，\((x,z,w)=(15,3,1)\). 两组数分别满足公差为 \(4\)、公比为 \(2\)，以及公差为 \(-6\)、公比为 \(\dfrac{1}{3}\)，故所求四个数为
\[
0,\ 4,\ 8,\ 16\quad\text{或}\quad 15,\ 9,\ 3,\ 1.
\]
\end{solution}

\begin{problem}
已知 \(\sin \alpha+\sin \beta=\frac{1}{4}\)，\(\cos \alpha+\cos \beta=\frac{1}{3}\)，求 \(\tan(\alpha+\beta)\) 的值.
\end{problem}

\begin{solution}
由和差化积公式，
\[
\sin\alpha+\sin\beta=2\sin\frac{\alpha+\beta}{2}\cos\frac{\alpha-\beta}{2}=\frac{1}{4},
\]
\[
\cos\alpha+\cos\beta=2\cos\frac{\alpha+\beta}{2}\cos\frac{\alpha-\beta}{2}=\frac{1}{3}.
\]
第二式的右端不为零，故 \(\cos\dfrac{\alpha-\beta}{2}\neq0\)，并且 \(\cos\dfrac{\alpha+\beta}{2}\neq0\). 两式相除，得
\[
\tan\frac{\alpha+\beta}{2}=\frac{1/4}{1/3}=\frac{3}{4}.
\]
于是
\[
\tan(\alpha+\beta)=\frac{2\tan\dfrac{\alpha+\beta}{2}}{1-\tan^2\dfrac{\alpha+\beta}{2}}
=\frac{2\cdot\dfrac{3}{4}}{1-\left(\dfrac{3}{4}\right)^2}
=\frac{24}{7}.
\]
\end{solution}

\begin{problem}
如图，在三棱锥 \(S-ABC\) 中，\(SA\perp\) 底面 \(ABC\)，\(AB\perp BC\). \(DE\) 垂直平分 \(SC\)，且分别交 \(AC\)，\(SC\) 于 \(D\)，\(E\). 又 \(SA=AB\)，\(SB=BC\). 求以 \(BD\) 为棱，以 \(BDE\) 与 \(BDC\) 为面的二面角的度数.

\bitmapinlinefigure[\linewidth][max width=0.28\linewidth]{img/1990/national_liberal/q23_fig1.png}
\end{problem}

\begin{solution}
取长度单位使 \(AB=SA=1\). 在直角坐标系中令底面 \(ABC\) 为 \(z=0\)，取
\[
B=(0,0,0),\qquad A=(1,0,0),\qquad C=(0,\sqrt{2},0),\qquad S=(1,0,1).
\]
此时 \(AB\perp BC\)，且 \(SB=\sqrt{1^2+1^2}=\sqrt{2}=BC\)，满足题设条件. 因为 \(DE\) 是 \(SC\) 的垂直平分线，\(E\) 为 \(SC\) 的中点，所以
\[
E=\left(\frac{1}{2},\frac{\sqrt{2}}{2},\frac{1}{2}\right).
\]
设 \(D\) 在 \(AC\) 上，写成 \(D=(1-t,t\sqrt{2},0)\). 由 \(DS=DC\)，有
\[
\left(1-t-1\right)^2+(t\sqrt{2})^2+(-1)^2
=\left(1-t\right)^2+(t\sqrt{2}-\sqrt{2})^2,
\]
即
\[
3t^2+1=3(1-t)^2=3-6t+3t^2,
\]
所以 \(t=\dfrac{1}{3}\)，从而
\[
D=\left(\frac{2}{3},\frac{\sqrt{2}}{3},0\right).
\]
平面 \(BDC\) 就是底面，其法向量可取 \(\bs{n}_2=(0,0,1)\). 又由于
\[
\overrightarrow{BD}\parallel(2,\sqrt{2},0),\qquad \overrightarrow{BE}\parallel(1,\sqrt{2},1),
\]
平面 \(BDE\) 的一个法向量为
\[
\bs{n}_1=(2,\sqrt{2},0)\times(1,\sqrt{2},1)=(\sqrt{2},-2,\sqrt{2}).
\]
设所求二面角为 \(\theta\)，则
\[
\cos\theta=\frac{|\bs{n}_1\cdot\bs{n}_2|}{|\bs{n}_1||\bs{n}_2|}
=\frac{\sqrt{2}}{\sqrt{2+4+2}}=\frac{1}{2}.
\]
因此 \(\theta=60^\circ\).
\end{solution}

\begin{problem}
已知 \(a>0\)，\(a\neq 1\)，解不等式 \(\log_a\left( 4+3x-x^2 \right)-\log_a\left( 2x-1 \right)>\log_a 2\).
\end{problem}

\begin{solution}
首先由对数真数为正，得
\[
4+3x-x^2>0,\qquad 2x-1>0.
\]
因为 \(4+3x-x^2=-(x-4)(x+1)\)，所以不等式的定义域为 \(\dfrac{1}{2}<x<4\). 在此定义域内，原不等式可化为
\[
\log_a\frac{4+3x-x^2}{2x-1}>\log_a2.
\]
\begin{enumerate}
\item 当 \(a>1\) 时，对数函数单调递增，故
\[
\frac{4+3x-x^2}{2x-1}>2.
\]
由于 \(2x-1>0\)，上式等价于
\[
6-x-x^2>0,
\]
即 \(-(x+3)(x-2)>0\). 结合定义域，得 \(\dfrac{1}{2}<x<2\).

\item 当 \(0<a<1\) 时，对数函数单调递减，故
\[
\frac{4+3x-x^2}{2x-1}<2.
\]
同理，\(-(x+3)(x-2)<0\)，结合定义域，得 \(2<x<4\).
\end{enumerate}
因此，当 \(a>1\) 时，解集为 \(\left(\dfrac{1}{2},2\right)\)；当 \(0<a<1\) 时，解集为 \((2,4)\).
\end{solution}

\begin{problem}
设 \(a\geqslant 0\)，在复数集 \(C\) 中解方程：\(z^2+2\left| z \right|=a\).
\end{problem}

\begin{solution}
设 \(z=x+y\i\)，其中 \(x,y\in\R\)，则 \(|z|=\sqrt{x^2+y^2}\)，且
\[
z^2=x^2-y^2+2xy\i.
\]
原方程右端为实数，所以虚部必须为零，即 \(2xy=0\). 分两种情况讨论.
\begin{enumerate}
\item 若 \(y=0\)，则 \(z=x\)，方程化为
\[
x^2+2|x|=a.
\]
令 \(t=|x|\geqslant0\)，则 \(t^2+2t-a=0\)，所以
\[
t=-1+\sqrt{a+1}.
\]
因此实数解为 \(z=\pm\left(\sqrt{a+1}-1\right)\).

\item 若 \(x=0\)，则 \(z=y\i\). 令 \(t=|y|\geqslant0\)，则
\[
-t^2+2t=a,\qquad t^2-2t+a=0.
\]
故
\[
t=1\pm\sqrt{1-a}.
\]
要使 \(t\) 为实数，必须有 \(0\leqslant a\leqslant1\)，且这时两个根均非负. 因而还得到
\[
z=\pm\left(1+\sqrt{1-a}\right)\i,\qquad z=\pm\left(1-\sqrt{1-a}\right)\i.
\]
\end{enumerate}
综上，任意 \(a\geqslant0\) 时都有实数解 \(z=\pm\left(\sqrt{a+1}-1\right)\)；当且仅当 \(0\leqslant a\leqslant1\) 时，还要加上上述两组纯虚数解，重合的根只计一次.
\end{solution}

\begin{problem}
设椭圆的中心是坐标原点，长轴在 \(x\) 轴上，离心率 \(e=\frac{\sqrt{3}}{2}\)，已知点 \(P\left( 0,\frac{3}{2} \right)\) 到这个椭圆上的点的最远距离是 \(\sqrt{7}\). 求这个椭圆的方程，并求椭圆上到点 \(P\) 的距离等于 \(\sqrt{7}\) 的点的坐标.
\end{problem}

\begin{solution}
设椭圆的半长轴、半短轴分别为 \(p\)，\(q\)，其中 \(p>q>0\)，焦距之半为 \(c\). 由离心率条件
\[
\frac{c}{p}=\frac{\sqrt{3}}{2},\qquad q^2=p^2-c^2,
\]
得 \(q^2=\dfrac{p^2}{4}\)，即 \(p=2q\). 因而椭圆可写成
\[
\frac{x^2}{4q^2}+\frac{y^2}{q^2}=1,
\]
且椭圆上的点满足 \(-q\leqslant y\leqslant q\) 和 \(x^2=4q^2-4y^2\).

设椭圆上一点为 \((x,y)\)，则它到 \(P\left(0,\dfrac{3}{2}\right)\) 的距离平方为
\[
d^2=x^2+\left(y-\frac{3}{2}\right)^2
=4q^2-3y^2-3y+\frac{9}{4}
=4q^2+3-3\left(y+\frac{1}{2}\right)^2.
\]
因此只需在区间 \([-q,q]\) 上求右端的最大值.
\begin{enumerate}
\item 若 \(q\geqslant\dfrac{1}{2}\)，则 \(-\dfrac{1}{2}\in[-q,q]\)，最大值在 \(y=-\dfrac{1}{2}\) 处取得. 由最远距离为 \(\sqrt{7}\)，得
\[
4q^2+3=7,\qquad q=1.
\]
此时 \(p=2\)，并且由 \(x^2=4-4\cdot\dfrac{1}{4}=3\)，得
\[
(x,y)=\left(\sqrt{3},-\frac{1}{2}\right)\quad\text{或}\quad\left(-\sqrt{3},-\frac{1}{2}\right).
\]

\item 若 \(0<q<\dfrac{1}{2}\)，则区间 \([-q,q]\) 中离 \(-\dfrac{1}{2}\) 最近的端点为 \(-q\)，故最大值在 \(y=-q\) 处取得. 这时
\[
d_{\max}^2=q^2+3q+\frac{9}{4}=\left(q+\frac{3}{2}\right)^2.
\]
令其等于 \(7\)，得 \(q=\sqrt{7}-\dfrac{3}{2}>\dfrac{1}{2}\)，与本情形矛盾.
\end{enumerate}
所以只能有 \(q=1\)，椭圆方程为
\[
\frac{x^2}{4}+y^2=1.
\]
所求点的坐标为 \(\left(\pm\sqrt{3},-\dfrac{1}{2}\right)\).
\end{solution}
