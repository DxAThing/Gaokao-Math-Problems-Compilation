\chapter{1990年上海卷}
\section{填空题}

\begin{problem}
函数 \(y=\frac{\sqrt{x+4}}{x+2}\) 的定义域是\fillinblank{}.
\end{problem}

\begin{answer}
\([-4,-2)\cup(-2,+\infty)\)
\end{answer}

\begin{solution}
根式有意义要求 \(x+4\geq0\)，分母不为零要求 \(x+2\ne0\)，所以 \(x\geq-4\) 且 \(x\ne-2\). 因此定义域为 \([-4,-2)\cup(-2,+\infty)\).
\end{solution}

\begin{problem}
函数 \(y=\arcsin x\)（\(x\in[-1,1]\)）的反函数是\fillinblank{}.
\end{problem}

\begin{answer}
\(y=\sin x\)，\(x\in[-\frac{\pi}{2},\frac{\pi}{2}]\)
\end{answer}

\begin{solution}
令 \(y=\arcsin x\)，则 \(x=\sin y\)，且 \(y\in[-\frac{\pi}{2},\frac{\pi}{2}]\). 交换自变量与因变量，得到反函数 \(y=\sin x\)，其定义域为 \([-\frac{\pi}{2},\frac{\pi}{2}]\).
\end{solution}

\begin{problem}
过点 \((1,2)\) 且与直线 \(2x+y-1=0\) 平行的直线方程是\fillinblank{}.
\end{problem}

\begin{answer}
\(2x+y-4=0\)
\end{answer}

\begin{solution}
与 \(2x+y-1=0\) 平行的直线可写为 \(2x+y+c=0\). 代入点 \((1,2)\)，得 \(2+2+c=0\)，所以 \(c=-4\). 因此所求直线为 \(2x+y-4=0\).
\end{solution}

\begin{problem}
已知圆柱的轴截面是正方形，它的面积是 \(4\mathrm{~cm}^2\)，
那么这个圆柱的体积是\fillinblank{}\(\mathrm{cm}^3\)（结果中保留 \(\pi\)）.
\end{problem}

\begin{answer}
\(2\pi\)
\end{answer}

\begin{solution}
设圆柱底面半径为 \(r\)，高为 \(h\). 轴截面是边长分别为 \(2r\) 与 \(h\) 的矩形，且为正方形，所以 \(h=2r\). 由轴截面面积得 \(2r\cdot h=4\)，从而 \(4r^2=4\)，故 \(r=1\)，\(h=2\). 因此圆柱体积为 \(\pi r^2h=2\pi\ \mathrm{cm}^3\).
\end{solution}

\begin{problem}
在 \(\triangle ABC\) 中，已知 \(\cos A=-\frac{3}{5}\)，则 \(\sin\frac{A}{2}=\)\fillinblank{}.
\end{problem}

\begin{answer}
\(\frac{2\sqrt5}{5}\)
\end{answer}

\begin{solution}
因为 \(A\) 是三角形内角，所以 \(0<A<\pi\)，从而 \(\sin\frac{A}{2}>0\). 利用半角公式，
\(\sin\frac{A}{2}=\sqrt{\frac{1-\cos A}{2}}=\sqrt{\frac{1+\frac35}{2}}=\sqrt{\frac45}=\frac{2\sqrt5}{5}\).
\end{solution}

\begin{problem}
设复数 \(\omega=\cos\frac{2\pi}{5}+\i\sin\frac{2\pi}{5}\)，则 \(\omega+\omega^2+\omega^3+\omega^4+\omega^5\) 的值是\fillinblank{}.
\end{problem}

\begin{answer}
\(0\)
\end{answer}

\begin{solution}
由定义，\(\omega\) 是五次单位根，满足 \(\omega^5=1\) 且 \(\omega\ne1\). 因此 \(1+\omega+\omega^2+\omega^3+\omega^4=0\). 加上 \(\omega^5=1\)，原式仍为 \(0\).
\end{solution}

\begin{problem}
已知圆锥的中截面周长为 \(a\)，母线长为 \(l\)，则它的侧面积等于\fillinblank{}.
\end{problem}

\begin{answer}
\(al\)
\end{answer}

\begin{solution}
设圆锥底面半径为 \(R\). 中截面平行于底面且经过母线中点，所以中截面半径为 \(\frac{R}{2}\)，其周长为 \(\pi R=a\). 圆锥侧面积为 \(\pi Rl\)，故侧面积为 \(al\).
\end{solution}

\begin{problem}
已知 \((x+a)^7\) 的展开式中，\(x^4\) 的系数是 \(-280\)，则实数 \(a=\)\fillinblank{}.
\end{problem}

\begin{answer}
\(-2\)
\end{answer}

\begin{solution}
展开式中 \(x^4\) 项来自取四个 \(x\) 和三个 \(a\)，其系数为 \(\mathrm C_7^4a^3=35a^3\). 由题意 \(35a^3=-280\)，得 \(a^3=-8\)，所以 \(a=-2\).
\end{solution}

\begin{problem}
双曲线 \(2mx^2-my^2=2\) 的一条准线是 \(y=1\)，则 \(m=\)\fillinblank{}.
\end{problem}

\begin{answer}
\(-\frac43\)
\end{answer}

\begin{solution}
若 \(m>0\)，方程化为横轴双曲线，准线为竖直直线，不可能有准线 \(y=1\)，所以 \(m<0\). 此时方程化为
\(\frac{y^2}{-2/m}-\frac{x^2}{-1/m}=1\).
取 \(a^2=-\frac2m\)，\(b^2=-\frac1m\)，则离心率满足 \(e^2=1+\frac{b^2}{a^2}=\frac32\). 纵轴双曲线的准线为 \(y=\pm\frac ae\)，所以
\(\frac ae=\sqrt{\frac{-2/m}{3/2}}=\sqrt{-\frac4{3m}}=1\). 因而 \(m=-\frac43\).
\end{solution}

\begin{problem}
平面上，四条平行直线与另外五条平行直线互相垂直，则它的矩形共有\fillinblank{} 个（结果用数值表示）.
\end{problem}

\begin{answer}
\(60\)
\end{answer}

\begin{solution}
一个矩形由第一组中任取两条直线和第二组中任取两条直线确定. 因此矩形共有 \(\mathrm C_4^2\mathrm C_5^2=6\times10=60\) 个.
\end{solution}

\section{单选题}

\begin{problem}
圆的半径是 \(1\)，圆心的极坐标是 \((1,0)\)，则这个圆的极坐标方程是（\quad）

\choices
    {\(\rho=\cos\theta\)}
    {\(\rho=\sin\theta\)}
    {\(\rho=2\cos\theta\)}
    {\(\rho=2\sin\theta\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
圆心的直角坐标为 \((1,0)\)，半径为 \(1\)，所以圆的直角坐标方程为 \((x-1)^2+y^2=1\)，即 \(x^2+y^2=2x\). 代入 \(x=\rho\cos\theta\)，\(x^2+y^2=\rho^2\)，得 \(\rho^2=2\rho\cos\theta\)，故圆的极坐标方程为 \(\rho=2\cos\theta\)，选 C.
\end{solution}

\begin{problem}
函数 \(f(x)\) 和 \(g(x)\) 的定义域均为 \(\R\)，“\(f(x)\)，\(g(x)\) 都是奇函数”是“\(f(x)\) 与 \(g(x)\) 的积是偶函数”的（\quad）

\choices
    {必要条件但非充分条件}
    {充分条件但非必要条件}
    {充分必要条件}
    {非充分条件也非必要条件}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
若 \(f\)、\(g\) 都是奇函数，则 \(f(-x)=-f(x)\)，\(g(-x)=-g(x)\)，从而 \((fg)(-x)=f(x)g(x)=(fg)(x)\)，所以这是积函数为偶函数的充分条件. 但它不是必要条件，例如取 \(f(x)=g(x)=1\)，积函数是偶函数，而 \(f\)、\(g\) 都不是奇函数. 因此选 B.
\end{solution}

\begin{problem}
设点 \(P\) 在有向线段 \(\overrightarrow{AB}\) 的延长线上，\(P\) 分 \(\overrightarrow{AB}\) 所成的比为 \(\lambda\)，则（\quad）

\choices
    {\(\lambda<-1\)}
    {\(-1<\lambda<0\)}
    {\(0<\lambda<1\)}
    {\(\lambda>1\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
在有向直线上取坐标 \(A=0\)，\(B=1\)，则 \(P\) 在 \(B\) 的外侧时可设其坐标为 \(p>1\). 按有向线段的定比分点定义，
\(\lambda=\frac{\overline{AP}}{\overline{PB}}=\frac{p}{1-p}\). 因为 \(p>1\)，所以 \(\lambda<0\)，且 \(|p|>|1-p|\)，从而 \(\lambda<-1\). 因此选 A.
\end{solution}

\begin{problem}
设 \(2^a=3\)，\(2^b=6\)，\(2^c=12\)，则数列 \(a\)，\(b\)，\(c\)（\quad）

\choices
    {是等差数列但不是等比数列}
    {是等比数列但不是等差数列}
    {既是等差数列又是等比数列}
    {既不是等差数列又不是等比数列}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
由已知，
\(b-a=\log_2\frac{2^b}{2^a}=\log_2 2=1\)，同理 \(c-b=\log_2\frac{2^c}{2^b}=1\)，所以 \(a,b,c\) 成等差数列. 又有 \(b^2-ac=1\ne0\)，故不成等比数列. 因此选 A.
\end{solution}

\begin{problem}
设 \(\alpha\) 角属于第二象限，且 \(\left|\cos\frac{\alpha}{2}\right|=-\cos\frac{\alpha}{2}\)，则 \(\frac{\alpha}{2}\) 角属于（\quad）

\choices
    {第一象限}
    {第二象限}
    {第三象限}
    {第四象限}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
由绝对值条件知 \(\cos\frac{\alpha}{2}<0\). 又因为 \(\alpha\) 属于第二象限，所以 \(\sin\alpha>0\). 利用倍角公式，
\(\sin\alpha=2\sin\frac{\alpha}{2}\cos\frac{\alpha}{2}>0\). 在 \(\cos\frac{\alpha}{2}<0\) 的条件下，必须有 \(\sin\frac{\alpha}{2}<0\)，故 \(\frac{\alpha}{2}\) 属于第三象限，选 C.
\end{solution}

\begin{problem}
设过长方体同一个顶点的三个面的对角线长分别是 \(a\)，\(b\)，\(c\)，那么这个长方体的对角线长是（\quad）

\choices
    {\(\sqrt{a^2+b^2+c^2}\)}
    {\(\sqrt{\frac{a^2+b^2+c^2}{2}}\)}
    {\(\sqrt{\frac{a^2+b^2+c^2}{3}}\)}
    {\(\frac{\sqrt{a^2+b^2+c^2}}{2}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
设长方体从同一顶点出发的三条棱长为 \(u,v,w\)，空间对角线长为 \(d\). 则三个面的对角线满足
\(a^2=u^2+v^2\)，\(b^2=v^2+w^2\)，\(c^2=w^2+u^2\). 三式相加得 \(a^2+b^2+c^2=2(u^2+v^2+w^2)=2d^2\)，所以 \(d=\sqrt{\frac{a^2+b^2+c^2}{2}}\)，选 B.
\end{solution}

\begin{problem}
函数 \(f(x)=a\tan\frac{x}{a}\) 的最小正周期是（\quad）

\choices
    {\(\pi a\)}
    {\(\pi|a|\)}
    {\(\frac{\pi}{a}\)}
    {\(\frac{\pi}{|a|}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
这里 \(a\ne0\). 由于正切函数的最小正周期为 \(\pi\)，要求 \(\frac{x+T}{a}-\frac{x}{a}=\pm\pi\)，所以正周期应满足 \(T=\pi|a|\). 乘在函数值外的常数 \(a\) 不改变周期，故选 B.
\end{solution}

\begin{problem}
已知 \(1<x<d\)，令 \(a=\left(\log_d x\right)^2\)，\(b=\log_d\left(x^2\right)\)，\(c=\log_d\left(\log_d x\right)\)，则（\quad）

\choices
    {\(a<b<c\)}
    {\(a<c<b\)}
    {\(c<b<a\)}
    {\(c<a<b\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
由 \(1<x<d\) 可知 \(d>1\). 令 \(t=\log_d x\)，则 \(0<t<1\)，并且 \(a=t^2>0\)，\(b=2t>0\)，\(c=\log_d t<0\). 又因为 \(0<t<1\)，有 \(t^2<2t\)，所以 \(c<a<b\)，选 D.
\end{solution}

\begin{problem}
设 \(a\)，\(b\) 是两条异面直线，那么下列四个命题中的假命题是（\quad）

\choices
    {经过直线 \(a\) 有且只有一个平面平行于直线 \(b\)}
    {经过直线 \(a\) 有且只有一个平面垂直于直线 \(b\)}
    {存在分别经过直线 \(a\) 和 \(b\) 的两个互相平行的平面}
    {存在分别经过直线 \(a\) 和 \(b\) 的两个互相垂直的平面}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
过直线 \(a\) 作一条与 \(b\) 平行的直线，它与 \(a\) 确定唯一平面，所以第一项正确. 该平面平行于直线 \(b\)，再作一个经过 \(b\) 且与它平行的平面，得到第三项. 通过分别选择包含 \(a\)、\(b\) 的适当平面，可以使两平面互相垂直，故第四项也可成立.

但平面若垂直于直线 \(b\)，则其中每条直线的方向都必须与 \(b\) 的方向垂直；异面直线 \(a\)、\(b\) 的方向不一定垂直，所以不一定存在经过 \(a\) 且垂直于 \(b\) 的平面. 因此第二项是假命题.
\end{solution}

\begin{problem}
下列四个函数中，在定义域内不具有单调性的函数是（\quad）

\choices
    {\(y=\cot\left(\arccos x\right)\)}
    {\(y=\tan\left(\arcsin x\right)\)}
    {\(y=\sin\left(\arctan x\right)\)}
    {\(y=\cos\left(\arctan x\right)\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
在相应定义域内，
\(\cot(\arccos x)=\frac{x}{\sqrt{1-x^2}}\)，其在 \((-1,1)\) 上递增；
\(\tan(\arcsin x)=\frac{x}{\sqrt{1-x^2}}\)，也在 \((-1,1)\) 上递增；
\(\sin(\arctan x)=\frac{x}{\sqrt{1+x^2}}\)，在 \(\R\) 上递增. 而 \(\cos(\arctan x)=\frac{1}{\sqrt{1+x^2}}\)，它在 \((-\infty,0)\) 上递增、在 \((0,+\infty)\) 上递减，所以在定义域 \(\R\) 内不具有单调性. 因此选 D.
\end{solution}

\section{解答题}

\begin{problem}
已知 \(\log_5(x^2+2x-2)=0\)，\(2\log_5(x+2)-\log_5 y+\frac{1}{2}=0\)，求 \(y\) 的值.
\end{problem}

\begin{solution}
由第一式得 \(x^2+2x-2=1\)，即 \(x^2+2x-3=0\)，所以 \(x=1\) 或 \(x=-3\). 第二式中的 \(\log_5(x+2)\) 要求 \(x>-2\)，故只能取 \(x=1\). 代入第二式，得
\(2\log_5 3-\log_5 y+\frac12=0\)，从而 \(\log_5 y=\log_5 9+\log_5\sqrt5=\log_5(9\sqrt5)\). 因此 \(y=9\sqrt5\).
\end{solution}

\begin{problem}
求方程 \(\sqrt{5}\cos x+\cos 2x+\sin x=0\) 在 \([0,2\pi)\) 上的解.
\end{problem}

\begin{solution}
令 \(s=\sin x\)，\(c=\cos x\). 原方程化为 \(2c^2-1+\sqrt5c+s=0\). 利用 \(c^2=1-s^2\)，得
\(\sqrt5c=2s^2-s-1\). 再由 \(s^2+c^2=1\)，有
\[
\left(2s^2-s-1\right)^2=5\left(1-s^2\right).
\]
整理并因式分解，得
\[
(1-s)\left((1-s)(2s+1)^2-5(1+s)\right)=0.
\]

当 \(s=1\) 时，\(c=0\)，得到 \(x=\frac{\pi}{2}\). 当 \(s\ne1\) 时，继续化简得
\(2s^3+s+2=0\). 设该方程在 \((-1,0)\) 内的根为 \(s_0\). 因为其导数 \(6s^2+1>0\)，且在 \(-1\)、\(0\) 处的值分别为 \(-1\)、\(2\)，所以 \(s_0\) 是唯一实根；又方程在 \(-\frac12\) 处的值为正，故 \(s_0<-\frac12\). 于是 \(2s_0^2-s_0-1>0\)，由 \(c=\frac{2s_0^2-s_0-1}{\sqrt5}\) 知 \(c>0\)，所以相应的角在第四象限. 由前面的平方关系还可知 \(c^2=1-s_0^2\)，故该 \(s_0\) 与上述 \(c\) 确实给出原方程的解，没有增根.

用卡尔丹公式可写出
\[
s_0=\sqrt[3]{\frac{-18+\sqrt{330}}{36}}+\sqrt[3]{\frac{-18-\sqrt{330}}{36}}.
\]
因此第二个解为 \(x=2\pi+\arcsin s_0\). 所以在给定区间内的全部解为
\(x=\frac{\pi}{2}\) 以及 \(x=2\pi+\arcsin s_0\)，其中 \(s_0\) 如上所述.
\end{solution}

\begin{problem}
已知点 \(P\) 在直线 \(x=2\) 上移动，直线 \(l\) 通过原点且与 \(OP\) 垂直，通过点 \(A(1,0)\) 及点 \(P\) 的直线 \(m\) 和直线 \(l\) 交于点 \(Q\). 求点 \(Q\) 的轨迹方程，并指出该轨迹的名称和它的焦点坐标.
\end{problem}

\begin{solution}
设 \(P=(2,t)\)，其中 \(t\in\R\). 因为 \(OP\) 的方向向量为 \((2,t)\)，所以直线 \(l\) 的方程为 \(2x+ty=0\). 直线 \(m\) 经过 \((1,0)\) 和 \((2,t)\)，方程为 \(y=t(x-1)\). 联立两式，得
\(Q=(X,Y)=\left(\frac{t^2}{t^2+2},-\frac{2t}{t^2+2}\right)\).

由 \(1-X=\frac{2}{t^2+2}\)，可得
\(Y^2=\frac{4t^2}{(t^2+2)^2}=2X(1-X)\). 因而轨迹方程为
\[
4\left(X-\frac12\right)^2+2Y^2=1.
\]
这是一条中心为 \((\frac12,0)\) 的椭圆，半长轴为 \(\frac1{\sqrt2}\)，半短轴为 \(\frac12\)，长轴在 \(y\) 轴方向. 因此焦半距为
\(c=\sqrt{\frac12-\frac14}=\frac12\)，焦点坐标为 \((\frac12,\frac12)\) 和 \((\frac12,-\frac12)\). 当 \(t\) 取有限实数时，点 \((1,0)\) 不取到；若只写轨迹方程，则对应上述椭圆.
\end{solution}

\begin{problem}
已知直线 \(l:x-ny=0\)（\(n\in\N\)）；圆 \(M:(x+1)^2+(y+1)^2=1\)；抛物线 \(\Phi:y=(x-1)^2\). 又 \(l\) 与 \(M\) 交于点 \(A\)，\(B\)；\(l\) 与 \(\Phi\) 交于点 \(C\)，\(D\). 求 \(\displaystyle\lim_{n\to\infty}\frac{\left|AB\right|^2}{\left|CD\right|^2}\).
\end{problem}

\begin{solution}
直线 \(l\) 写成 \(y=\frac{x}{n}\). 圆心 \((-1,-1)\) 到该直线的距离为
\(d=\frac{|n-1|}{\sqrt{n^2+1}}\). 因此圆的弦 \(AB\) 满足
\[
\left|AB\right|^2=4(1-d^2)=\frac{8n}{n^2+1}.
\]

直线与抛物线的交点横坐标是方程 \(\frac{x}{n}=(x-1)^2\) 的两个根，亦即
\(x^2-\left(2+\frac1n\right)x+1=0\). 设两根为 \(x_1,x_2\)，则
\(\left|x_1-x_2\right|^2=\frac{4n+1}{n^2}\). 因为弦 \(CD\) 在直线 \(y=x/n\) 上，
\[
\left|CD\right|^2=\left(1+\frac1{n^2}\right)\left|x_1-x_2\right|^2=\frac{(n^2+1)(4n+1)}{n^4}.
\]
所以
\[
\frac{\left|AB\right|^2}{\left|CD\right|^2}=\frac{8n^5}{(n^2+1)^2(4n+1)}\longrightarrow2.
\]
\end{solution}

\begin{problem}
关于实数 \(x\) 的不等式 \(\left|x-\frac{(a+1)^2}{2}\right|\le\frac{(a-1)^2}{2}\) 与 \(x^2-3(a+1)x+2(3a+1)\le0\)（其中 \(a\in\R\)）的解集依次记为 \(A\) 与 \(B\). 求使 \(A\subseteq B\) 的 \(a\) 的取值范围.
\end{problem}

\begin{solution}
第一个不等式给出
\[
A=\left[\frac{(a+1)^2-(a-1)^2}{2},\frac{(a+1)^2+(a-1)^2}{2}\right]=[2a,a^2+1].
\]
第二个不等式中的二次式可因式分解为
\(x^2-3(a+1)x+2(3a+1)=(x-2)(x-3a-1)\). 因而
\(B=[3a+1,2]\)（当 \(a\le\frac13\)），或 \(B=[2,3a+1]\)（当 \(a\ge\frac13\)).

当 \(a\le\frac13\) 时，包含关系要求 \(3a+1\le2a\) 且 \(a^2+1\le2\)，即 \(a\le-1\) 且 \(-1\le a\le1\)，所以只有 \(a=-1\). 当 \(a\ge\frac13\) 时，包含关系要求 \(2\le2a\) 且 \(a^2+1\le3a+1\)，即 \(a\ge1\) 且 \(0\le a\le3\)，所以 \(1\le a\le3\). 因此所求范围为 \(\{-1\}\cup[1,3]\).
\end{solution}

\begin{problem}
如图，平面 \(\alpha\)，\(\beta\) 相交于直线 \(MN\)，点 \(A\) 在平面 \(\alpha\) 上，点 \(B\) 在平面 \(\beta\) 上，点 \(C\) 在直线 \(MN\) 上，\(\angle ACM=\angle BCN=45^\circ\)，二面角 \(A-MN-B\) 是 \(60^\circ\)，\(AC=1\). 求：

\begin{enumerate}
\item 点 \(A\) 到平面 \(\beta\) 的距离；
\item 二面角 \(A-BC-M\) 的大小（用反三角函数表示）.
\end{enumerate}

\end{problem}

\begin{solution}
\begin{enumerate}
\item
取 \(C\) 为原点，取 \(MN\) 为 \(x\) 轴，正方向为 \(CN\) 的方向，令平面 \(\alpha\) 为 \(z=0\). 取平面 \(\alpha\) 内垂直于 \(MN\) 且指向 \(A\) 的方向为正 \(y\) 方向，则
\(A=(-\frac{\sqrt2}{2},\frac{\sqrt2}{2},0)\).

由二面角 \(A-MN-B=60^\circ\)，可取平面 \(\beta\) 内垂直于 \(MN\) 且指向 \(B\) 所在半平面的单位方向为 \((0,\frac12,\frac{\sqrt3}{2})\)，所以平面 \(\beta\) 的方程可写为 \(z=\sqrt3y\). 因而点 \(A\) 到平面 \(\beta\) 的距离为
\[
d=\frac{|\sqrt3y_A-z_A|}{\sqrt{1+3}}=\frac{\sqrt6}{4}.
\]

\item
再设 \(B\) 的位置向量方向去掉正比例因子后为 \(\bs b=(1,\frac12,\frac{\sqrt3}{2})\)，而 \(A\) 的位置向量去掉正比例因子后为 \(\bs a=(-1,1,0)\). 平面 \(MBC\) 的一个法向量和平面 \(ABC\) 的一个法向量分别为
\[
\bs n_1=(1,0,0)\times\bs b=\left(0,-\frac{\sqrt3}{2},\frac12\right),
\quad
\bs n_2=\bs a\times\bs b=\left(\frac{\sqrt3}{2},\frac{\sqrt3}{2},-\frac32\right).
\]
于是二面角 \(A-BC-M\) 的余弦为
\[
\cos\theta=\frac{|\bs n_1\cdot\bs n_2|}{|\bs n_1||\bs n_2|}=\frac{\frac32}{1\cdot\frac{\sqrt{15}}2}=\sqrt{\frac35}.
\]
该二面角取锐角，所以
\(\theta=\arccos\sqrt{\frac35}=\arctan\sqrt{\frac23}\).
\end{enumerate}
\end{solution}

\begin{problem}
复平面上点 \(A\)，\(B\) 对应的复数分别为 \(z_1=2\)，\(z_2=-3\)，点 \(P\) 对应的复数为 \(z\)，\(\frac{z-z_1}{z-z_2}\) 的辐角主值为 \(\varphi\). 当点 \(P\) 在以原点为圆心，\(1\) 为半径的上半圆周（不包括两个端点）上运动时，求 \(\varphi\) 的最小值.
\end{problem}

\begin{solution}
设 \(P=(\cos t,\sin t)\)，其中 \(0<t<\pi\). 因为 \(P\) 在上半平面，向量 \(\overrightarrow{AP}=(\cos t-2,\sin t)\) 在第二象限，向量 \(\overrightarrow{BP}=(\cos t+3,\sin t)\) 在第一象限，所以
\(\varphi=\arg\overrightarrow{AP}-\arg\overrightarrow{BP}\) 是三角形 \(APB\) 在 \(P\) 处的内角，且 \(\frac\pi2<\varphi<\pi\).

两向量的内积和行列式分别为
\[
\overrightarrow{AP}\cdot\overrightarrow{BP}=(\cos t-2)(\cos t+3)+\sin^2t=\cos t-5,
\]
\[
\left|\det(\overrightarrow{AP},\overrightarrow{BP})\right|=5\sin t.
\]
令 \(\delta=\pi-\varphi\)，则 \(\delta\) 为锐角，且
\[
\tan\delta=\frac{5\sin t}{5-\cos t}.
\]
因此使 \(\varphi\) 最小，等价于使 \(\frac{\sin t}{5-\cos t}\) 最大. 对其求导，
\[
\left(\frac{\sin t}{5-\cos t}\right)'=\frac{5\cos t-1}{(5-\cos t)^2}.
\]
所以最大值在 \(\cos t=\frac15\) 处取得，此时 \(\sin t=\frac{2\sqrt6}{5}\)，从而
\(\tan\delta=\frac{5\sqrt6}{12}\). 于是
\[
\varphi_{\min}=\pi-\arctan\frac{5\sqrt6}{12}.
\]
\end{solution}
