\begin{problem}
设全集 \(U=\{1,2,3,4,5\}\)，集合 \(M=\{1,4\}\)，\(N=\{2,5\}\)，则 \(N\cup\complement_U M=\)（\quad）

\choices
    {\(\{2,3,5\}\)}
    {\(\{1,3,4\}\)}
    {\(\{1,2,4,5\}\)}
    {\(\{2,3,4,5\}\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
由 \(M=\{1,4\}\) 得 \(\complement_U M=\{2,3,5\}\). 又 \(N=\{2,5\}\)，故 \(N\cup\complement_U M=\{2,5\}\cup\{2,3,5\}=\{2,3,5\}\). 故选 A.
\end{solution}

\begin{problem}
\(\dfrac{5\left(1+\i^3\right)}{\left(2+\i\right)\left(2-\i\right)}=\)（\quad）

\choices
    {\(-1\)}
    {\(1\)}
    {\(1-\i\)}
    {\(1+\i\)}

\begin{answer}
C
\end{answer}

\begin{solution}
由 \(\i^3=-\i\) 得分子为 \(5(1-\i)\)，分母为 \((2+\i)(2-\i)=5\)，故原式等于 \(1-\i\). 故选 C.
\end{solution}

已知向量 \(\bs{a}=\left(3,1\right)\)，\(\bs{b}=\left(2,2\right)\)，则 \(\cos\left(\bs{a}+\bs{b},\bs{a}-\bs{b}\right)=\)（\quad）

    {\(\dfrac{\sqrt{5}}{5}\)}

\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
\(\bs{a}+\bs{b}=(5,3)\)，\(\bs{a}-\bs{b}=(1,-1)\). 于是 \(\cos\left(\bs{a}+\bs{b},\bs{a}-\bs{b}\right)=\dfrac{5-3}{\sqrt{34}\sqrt2}=\dfrac{2}{\sqrt{68}}=\dfrac{\sqrt{17}}{17}\). 故选 B.
\end{solution}

某校文艺部有 4 名学生，其中高一、高二年级各 2 名. 从这 4 名学生中随机选 2 名组织校文艺汇演，则这 2 名学生来自不同年级的概率为（\quad）

\choices

    {\(\dfrac{1}{3}\)}
    {\(\dfrac{1}{2}\)}
    {\(\dfrac{2}{3}\)}
\end{problem}

\begin{answer}
D
\end{answer}

\end{solution}

记 \(S_n\) 为等差数列 \(\{a_n\}\) 的前 \(n\) 项和. 若 \(a_2+a_6=10\)，\(a_4a_8=45\)，则 \(S_5=\)（\quad）

\end{answer}

\begin{problem}
设 \(F_1\)，\(F_2\) 为椭圆 \(C:\dfrac{x^2}{5}+y^2=1\) 的两个焦点，点 \(P\) 在 \(C\) 上，若 \(\overrightarrow{PF_1}\cdot\overrightarrow{PF_2}=0\)，则 \(\left|PF_1\right|\cdot\left|PF_2\right|=\)（\quad）

\choices
    {\(1\)}
    {\(2\)}
    {\(4\)}
    {\(5\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
由椭圆定义，\(\left|PF_1\right|+\left|PF_2\right|=2\sqrt5\). 又 \(\overrightarrow{PF_1}\cdot\overrightarrow{PF_2}=0\)，故 \(\left|PF_1\right|^2+\left|PF_2\right|^2=\left|F_1F_2\right|^2=16\). 两式相减得 \(2\left|PF_1\right|\cdot\left|PF_2\right|=20-16=4\)，即 \(\left|PF_1\right|\cdot\left|PF_2\right|=2\). 故选 B.

\begin{problem}
曲线 \(y=\dfrac{\e^x}{x+1}\) 在点 \(\left(1,\dfrac{\e}{2}\right)\) 处的切线方程为（\quad）

\choices
    {\(y=\dfrac{\e}{4}x\)}
    {\(y=\dfrac{\e}{2}x\)}
    {\(y=\dfrac{\e}{4}x+\dfrac{\e}{4}\)}
    {\(y=\dfrac{\e}{2}x+\dfrac{3\e}{4}\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
\(y'=\dfrac{\e^x(x+1)-\e^x}{(x+1)^2}=\dfrac{x\e^x}{(x+1)^2}\)，在 \(x=1\) 处斜率为 \(\dfrac{\e}{4}\). 故切线方程为 \(y-\dfrac{\e}{2}=\dfrac{\e}{4}(x-1)\)，即 \(y=\dfrac{\e}{4}x+\dfrac{\e}{4}\). 故选 C.
\end{solution}

在三棱锥 \(P-ABC\) 中，\(\triangle ABC\) 是边长为 \(2\) 的等边三角形，\(PA=PB=2\)，\(PC=\sqrt{6}\)，则该棱锥的体积为（\quad）

\begin{problem}
函数 \(y=f(x)\) 的图象由 \(y=\cos\left(2x+\dfrac{\pi}{6}\right)\) 的图象向左平移 \(\dfrac{\pi}{6}\) 个单位长度得到，则 \(y=f(x)\) 的图象与直线 \(y=\dfrac{1}{2}x-\dfrac{1}{2}\) 的交点个数为（\quad）

\choices
    {\(1\)}
    {\(2\)}
    {\(3\)}
    {\(4\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
平移得 \(f(x)=\cos\left(2x+\dfrac{\pi}{2}\right)=-\sin2x\). 记 \(d(x)=-\sin2x-\dfrac{x-1}{2}\)，交点个数即 \(d(x)\) 的零点个数. 当 \(x<-1\) 时 \(\dfrac{x-1}{2}<-1\leqslant-\sin2x\)，当 \(x>3\) 时 \(\dfrac{x-1}{2}>1\geqslant-\sin2x\)，均无零点；当 \(-1\leqslant x<0\) 时 \(-\sin2x>0>\dfrac{x-1}{2}\)，也无零点. 在 \([0,3]\) 上，\(d'(x)=-2\cos2x-\dfrac12\) 恰有两个零点 \(x_1=\dfrac12\arccos\left(-\dfrac14\right)\)，\(x_2=\pi-x_1\)，且 \(d\) 在 \((0,x_1)\)、\((x_2,3)\) 上严格递减，在 \((x_1,x_2)\) 上严格递增. 又 \(d(0)=\dfrac12>0\)，\(d(x_1)=-\dfrac{\sqrt{15}}{4}-\dfrac{x_1-1}{2}<0\)，\(d(x_2)=\dfrac{\sqrt{15}}{4}-\dfrac{x_2-1}{2}>0\)，\(d(3)=-\sin6-1<0\)，故三段区间内各恰有一个零点. 因此交点个数为 \(3\). 故选 C.

\begin{problem}
记 \(S_n\) 为等比数列 \(\{a_n\}\) 的前 \(n\) 项和. 若 \(8S_6=7S_3\)，则 \(\{a_n\}\) 的公比为\fillinblank{}.
\end{problem}

\begin{answer}
\(-\dfrac{1}{2}\)
\end{answer}

\begin{solution}
第 \(4\) 至 \(6\) 项是首三项的 \(q^3\) 倍，故 \(S_6=S_3+q^3S_3\). 条件化为 \(8(1+q^3)S_3=7S_3\). 若 \(S_3=0\)，则 \(a_1(1+q+q^2)=0\)，而 \(1+q+q^2>0\) 恒成立，得 \(a_1=0\)，此时为零数列而公比不唯一，舍去. 故 \(S_3\ne0\)，得 \(1+q^3=\dfrac78\)，即 \(q=-\dfrac12\).
\end{solution}

\begin{problem}
若 \(f(x)=\left(x-1\right)^2+ax+\sin\left(x+\dfrac{\pi}{2}\right)\) 为偶函数，则 \(a=\fillinblank{}\).
\end{problem}

\begin{answer}
\(2\)
\end{answer}

\begin{solution}
\(\sin\left(x+\dfrac{\pi}{2}\right)=\cos x\)，故 \(f(x)=(x-1)^2+ax+\cos x\). 由偶函数得 \(f(-x)=f(x)\) 恒成立，即 \((x+1)^2-ax=(x-1)^2+ax\)，亦即 \(4x-2ax=0\) 对一切 \(x\) 成立，故 \(a=2\).
\end{solution}

若 \(x\)，\(y\) 满足约束条件
\(\begin{cases}
3x-2y\le3,\\
-2x+3y\le3,\\
x+y\ge1,
\end{cases}\)，
则 \(z=3x+2y\) 的最大值为\fillinblank{}.
\end{problem}

\begin{answer}
\(15\)
\end{answer}

三条边界直线两两交于 \((3,3)\)，\((1,0)\)，\((0,1)\)，可行域为该三角形及其内部. 目标函数在顶点处的值分别为 \(15\)，\(3\)，\(2\)，故最大值为 \(15\).
\end{solution}

\begin{problem}
在正方体 \(ABCD-A_1B_1C_1D_1\) 中，\(AB=4\)，\(O\) 为 \(AC_1\) 的中点，若该正方体的棱与球 \(O\) 的球面有公共点，则球 \(O\) 的半径的取值范围是\fillinblank{}.
\end{problem}

\begin{answer}
\(\left[2\sqrt{2},2\sqrt{3}\right]\)
\end{answer}

\begin{solution}
\(O\) 为正方体中心. 中心到每条棱所在直线的距离均为 \(2\sqrt2\)，且垂足为棱的中点，故球面与棱有公共点当且仅当半径不小于 \(2\sqrt2\). 中心到顶点的距离为 \(2\sqrt3\)，半径超过该值时整个正方体含于球内，棱与球面无公共点. 故半径的取值范围是 \(\left[2\sqrt{2},2\sqrt{3}\right]\).
\end{solution}

\begin{problem}
记 \(\triangle ABC\) 的内角 \(A\)，\(B\)，\(C\) 的对边分别为 \(a\)，\(b\)，\(c\)，已知 \(\dfrac{b^2+c^2-a^2}{\cos A}=2\).
\begin{enumerate}
\item 求 \(bc\)；
\item 若 \(\dfrac{a\cos B-b\cos A}{a\cos B+b\cos A}-\dfrac{b}{c}=1\)，求 \(\triangle ABC\) 面积.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由余弦定理，\(b^2+c^2-a^2=2bc\cos A\). 因分式有意义，\(\cos A\ne0\)，代入条件得 \(2bc=2\)，即 \(bc=1\).
\item 由正弦定理，\(\dfrac{a\cos B-b\cos A}{a\cos B+b\cos A}=\dfrac{\sin A\cos B-\sin B\cos A}{\sin A\cos B+\sin B\cos A}=\dfrac{\sin(A-B)}{\sin(A+B)}=\dfrac{\sin(A-B)}{\sin C}\). 又 \(\dfrac{b}{c}=\dfrac{\sin B}{\sin C}\)，条件化为 \(\sin(A-B)-\sin B=\sin C=\sin(A+B)\)，即 \(-2\cos A\sin B=\sin B\). 因 \(\sin B>0\)，得 \(\cos A=-\dfrac12\). 结合 \(A\) 为三角形内角，得 \(\angle A=\dfrac{2\pi}{3}\). 故面积为 \(\dfrac12bc\sin A=\dfrac12\cdot1\cdot\dfrac{\sqrt3}{2}=\dfrac{\sqrt3}{4}\).
\end{enumerate}
\end{solution}

如图，在三棱柱 \(ABC-A_1B_1C_1\) 中，\(A_1C\perp\) 平面 \(ABC\)，\(\angle ACB=90^\circ\).

\begin{enumerate}
\item 证明：平面 \(ACC_1A_1\perp\) 平面 \(BCC_1B_1\)；
\item 设 \(AB=A_1B\)，\(AA_1=2\)，求四棱锥 \(A_1-BB_1C_1C\) 的高.
\end{enumerate}

\item 取坐标 \(C=(0,0,0)\)，\(A=(x,0,0)\)，\(B=(0,y,0)\)（\(x>0\)，\(y>0\)），则 \(A_1=(0,0,h)\)（\(h>0\)）. 由 \(AA_1=2\) 得 \(x^2+h^2=4\)；由 \(A_1B^2=y^2+h^2\) 及 \(AB=A_1B\) 得 \(x^2=y^2+h^2-y^2\)，即 \(x=h\). 故 \(x=h=\sqrt2\). 平面 \(BCC_1B_1\) 由 \(BC=(0,y,0)\) 与 \(CC_1=(-x,0,h)\) 张成，其法向量可取 \((yh,0,xy)\)，点 \(A_1\) 到该平面的距离为 \(\dfrac{hxy}{y\sqrt{h^2+x^2}}=\dfrac{x}{\sqrt2}=1\). 因此四棱锥 \(A_1-BB_1C_1C\) 的高为 \(1\).
\end{enumerate}
\end{solution}

\begin{problem}
已知函数 \(f(x)=ax-\dfrac{\sin x}{\cos^2x}\)，\(x\in\left(0,\dfrac{\pi}{2}\right)\).
\begin{enumerate}
\item 当 \(a=1\) 时，讨论 \(f(x)\) 的单调性；
\item 若 \(f(x)+\sin x<0\)，求 \(a\) 的取值范围.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 当 \(a=1\) 时，\(f'(x)=1-\dfrac{\cos^3x+2\sin^2x\cos x}{\cos^4x}=1-\dfrac{1+\sin^2x}{\cos^3x}\). 记 \(t=\cos x\in(0,1)\)，则 \(\cos^3x-1-\sin^2x=t^3+t^2-2=(t-1)(t^2+2t+2)<0\)，故 \(f'(x)<0\) 在 \(\left(0,\dfrac{\pi}{2}\right)\) 上恒成立. 因此 \(f(x)\) 在 \(\left(0,\dfrac{\pi}{2}\right)\) 上单调递减.
\item 条件等价于 \(ax<\dfrac{\sin^3x}{\cos^2x}\) 对一切 \(x\in\left(0,\dfrac{\pi}{2}\right)\) 成立，即 \(a\) 小于 \(h(x)=\dfrac{\sin^3x}{x\cos^2x}\) 的下确界. 因 \(0<\sin x<x\)，有 \(0<h(x)<\dfrac{x^2}{\cos^2x}\to0\)（\(x\to0^+\)），故下确界为 \(0\)，必要性给出 \(a\leqslant0\). 当 \(a=0\) 时，\(f(x)+\sin x=-\dfrac{\sin^3x}{\cos^2x}<0\) 恒成立；若 \(a<0\)，则 \(ax<0<\dfrac{\sin^3x}{\cos^2x}\)，同样恒成立. 因此 \(a\leqslant0\).
\end{enumerate}
\end{solution}

\begin{problem}
已知直线 \(x-2y+1=0\) 与抛物线 \(C:y^2=2px\)（\(p>0\)） 交于 \(A\)，\(B\) 两点，\(|AB|=4\sqrt{15}\).
\begin{enumerate}
\item 求 \(p\)；
\item 设 \(F\) 为 \(C\) 的焦点，\(M\)，\(N\) 为 \(C\) 上两点，且 \(\overrightarrow{FM}\cdot\overrightarrow{FN}=0\)，求 \(\triangle MFN\) 面积的最小值.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由 \(x=2y-1\) 代入 \(y^2=2px\) 得 \(y^2-4py+2p=0\). 弦长 \(|AB|=\sqrt5\cdot\sqrt{16p^2-8p}=4\sqrt{15}\)，即 \(16p^2-8p-48=0\)，解得 \(p=2\)（舍去负根 \(p=-\dfrac32\)）.
\item 此时 \(F=(1,0)\). 设 \(M=(s^2,2s)\)，\(N=(t^2,2t)\)（\(s\ne t\)），则 \(\overrightarrow{FM}\cdot\overrightarrow{FN}=(s^2-1)(t^2-1)+4st=0\)，即 \((st)^2+6st+1=(s+t)^2\). 记 \(u=st\)，则 \((u+3)^2\geqslant8\). 面积 \(S=\dfrac12(s^2+1)(t^2+1)=\dfrac12\left(u^2+(s+t)^2-2u+1\right)=(u+1)^2\). 在 \(u\leqslant-3-2\sqrt2\) 或 \(u\geqslant-3+2\sqrt2\) 上，\((u+1)^2\) 的最小值在 \(u=-3+2\sqrt2\) 处取得，为 \((2\sqrt2-2)^2=12-8\sqrt2\). 故 \(\triangle MFN\) 面积的最小值为 \(12-8\sqrt{2}\).
\end{enumerate}
\end{solution}

\section{选考题：解答题}

\begin{problem}
已知点 \(P(2,1)\)，直线 \(l\) 的参数方程为
\(\begin{cases}
x=2+t\cos\alpha,\\
y=1+t\sin\alpha,
\end{cases}\)（\(t\) 为参数），\(\alpha\) 为 \(l\) 的倾斜角，\(l\) 与 \(x\) 轴正半轴，\(y\) 轴正半轴分别交于 \(A\)，\(B\)，且 \(|PA|\cdot|PB|=4\).
\begin{enumerate}
\item 求 \(\alpha\)；
\item 以坐标原点为极点，\(x\) 轴正半轴为极轴建立极坐标系，求 \(l\) 的极坐标方程.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设交点对应的参数为 \(t_A\)，\(t_B\)，则 \(|PA|=|t_A|\)，\(|PB|=|t_B|\). 由 \(A\) 在 \(x\) 轴上得 \(t_A=-\dfrac1{\sin\alpha}\)，由 \(B\) 在 \(y\) 轴上得 \(t_B=-\dfrac2{\cos\alpha}\)，故 \(|PA|\cdot|PB|=\dfrac{2}{|\sin\alpha\cos\alpha|}=\dfrac{4}{|\sin2\alpha|}=4\)，即 \(|\sin2\alpha|=1\). 直线与两正半轴相交则斜率为负，故 \(\alpha\in\left(\dfrac{\pi}{2},\pi\right)\)，得 \(\alpha=\dfrac{3\pi}{4}\).
\item 此时直线斜率为 \(-1\)，过 \((2,1)\)，即 \(x+y=3\). 化为极坐标方程得 \(\rho\cos\theta+\rho\sin\theta-3=0\).
\end{enumerate}
\end{solution}

\begin{problem}
已知 \(f(x)=2|x-a|-a\)，\(a>0\).
\begin{enumerate}
\item 求不等式 \(f(x)<x\) 的解集；
\item 若曲线 \(y=f(x)\) 与 \(x\) 轴所围成的图形的面积为 \(2\)，求 \(a\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 不等式化为 \(2|x-a|<x+a\). 右端须为正，且两边平方得 \(4(x-a)^2<(x+a)^2\)，即 \(3x^2-10ax+3a^2=(3x-a)(x-3a)<0\). 又 \(a>0\)，得 \(\dfrac{a}{3}<x<3a\)，此时 \(x+a>0\) 自动成立. 故解集为 \(\left(\dfrac{a}{3},3a\right)\).
\item \(f(x)=0\) 得 \(|x-a|=\dfrac{a}{2}\)，即 \(x=\dfrac{a}{2}\) 或 \(x=\dfrac{3a}{2}\). 图像是以 \((a,-a)\) 为顶点的 V 形，与 \(x\) 轴围成底为 \(a\)、高为 \(a\) 的三角形，面积为 \(\dfrac{a^2}{2}\). 由 \(\dfrac{a^2}{2}=2\) 及 \(a>0\) 得 \(a=2\).
\end{enumerate}
\end{solution}
