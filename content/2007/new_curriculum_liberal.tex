\section{单选题}

\begin{problem}
设集合 \(A = \{x \mid x > -1\}\)，\(B = \{x \mid -2 < x < 2\}\)，则 \(A \cup B =\)（\quad）

\choices
    {\(\{x \mid x > -2\}\)}
    {\(\{x \mid x > -1\}\)}
    {\(\{x \mid -2 < x < -1\}\)}
    {\(\{x \mid -1 < x < 2\}\)}
\end{problem}

\begin{answer}
A.
\end{answer}

\begin{solution}
由 \(A=\{x\mid x>-1\}\)，\(B=\{x\mid -2<x<2\}\). 当 \(-2<x\leqslant -1\) 时，\(x\in B\)；当 \(x>-1\) 时，\(x\in A\). 因此 \(A\cup B=\{x\mid x>-2\}\)，故选 A.
\end{solution}

\begin{problem}
已知命题 \(p\)：\(\forall x \in \R\)，\(\sin x \leqslant 1\)，则（\quad）

\choices
    {\(\neg p\)：\(\exists x \in \R\)，\(\sin x \geqslant 1\)}
    {\(\neg p\)：\(\forall x \in \R\)，\(\sin x \geqslant 1\)}
    {\(\neg p\)：\(\exists x \in \R\)，\(\sin x > 1\)}
    {\(\neg p\)：\(\forall x \in \R\)，\(\sin x > 1\)}
\end{problem}

\begin{answer}
C.
\end{answer}

\begin{solution}
命题 \(p\) 是“对任意 \(x\in\R\)，都有 \(\sin x\leqslant1\)”. 全称命题的否定是存在一个 \(x\in\R\) 使原不等式不成立，即 \(\sin x>1\). 因而 \(\neg p\) 应为“\(\exists x\in\R\)，\(\sin x>1\)”，故选 C.
\end{solution}

\begin{problem}
函数 \( y = \sin \left(2x - \frac{\pi}{3}\right) \) 在区间 \(\left[-\frac{\pi}{2}, \pi\right]\) 的简图是（\quad）

\choices
    {\choicebitmap[width=0.15\paperwidth]{img/2007/new_curriculum_liberal/q3_optA.png}}
    {\choicebitmap[width=0.15\paperwidth]{img/2007/new_curriculum_liberal/q3_optB.png}}
    {\choicebitmap[width=0.15\paperwidth]{img/2007/new_curriculum_liberal/q3_optC.png}}
    {\choicebitmap[width=0.15\paperwidth]{img/2007/new_curriculum_liberal/q3_optD.png}}
\end{problem}

\begin{answer}
A.
\end{answer}

\begin{solution}
令 \(u=2x-\dfrac{\pi}{3}\). 在 \(x\in\left[-\dfrac{\pi}{2},\pi\right]\) 上，\(u\) 从 \(-\dfrac{4\pi}{3}\) 变化到 \(\dfrac{5\pi}{3}\). 图象在 \(x=-\dfrac{\pi}{2}\) 处的函数值为 \(\dfrac{\sqrt{3}}{2}\)，且此处函数递减；在 \(x=-\dfrac{\pi}{12}\) 处取得极小值 \(-1\)，在 \(x=\dfrac{5\pi}{12}\) 处取得极大值 \(1\)，在 \(x=\dfrac{11\pi}{12}\) 处再次取得极小值 \(-1\). 零点为 \(x=-\dfrac{\pi}{3}\)、\(\dfrac{\pi}{6}\)、\(\dfrac{2\pi}{3}\). 这些端点值、极值点和零点的变化顺序与选项 A 的图象一致，故选 A.
\end{solution}

\begin{problem}
已知平面向量 \( \bs{a} = (1, 1) \)，\( \bs{b} = (1, -1) \)，则向量 \( \frac{1}{2}\bs{a} - \frac{3}{2}\bs{b} = \)（\quad）

\choices
    {\((-2, - 1)\)}
    {\((-2, 1)\)}
    {\((-1, 0)\)}
    {\((-1, 2)\)}
\end{problem}

\begin{answer}
D.
\end{answer}

\begin{solution}
逐坐标计算得
\[
\frac{1}{2}\bs a-\frac{3}{2}\bs b
=\left(\frac12,\frac12\right)-\left(\frac32,-\frac32\right)
=\left(-1,2\right)
\]
故选 D.
\end{solution}

如果执行下面的程序框图，那么输出的 \(S =\)（\quad）

\begin{problem}
已知 \(a\)，\(b\)，\(c\)，\(d\) 成等比数列，且曲线 \(y = x^2 - 2x + 3\) 的顶点是 \((b, c)\)，则 \(ad\) 等于（\quad）

\choices
    {\(3\)}
    {\(2\)}
    {\(1\)}
    {\(-2\)}
\end{problem}

\begin{answer}
B.
\end{answer}

\begin{solution}
将抛物线方程配方为 \(y=(x-1)^2+2\)，所以顶点 \((b,c)=(1,2)\). 等比数列的公比为 \(q=\dfrac{c}{b}=2\)，从而 \(a=\dfrac{b}{q}=\dfrac12\)，\(d=cq=4\). 因此 \(ad=\dfrac12\cdot4=2\)，故选 B.
\end{solution}

\begin{problem}
已知抛物线 \( y^{2} = 2px \)（\( p > 0 \)）的焦点为 \(F\)，点 \( P_{1}(x_{1}, y_{1}) \)，\( P_{2}(x_{2}, y_{2}) \)，\( P_{3}(x_{3}, y_{3}) \) 在抛物线上，且 \( 2x_{2} = x_{1} + x_{3} \)，则有（\quad）

\choices
    {\( |FP_{1}| + |FP_{2}| = |FP_{3}| \)}
    {\( |FP_{1}|^{2} + |FP_{2}|^{2} = |FP_{3}|^{2} \)}
    {\(2|FP_{2}| = |FP_{1}| + |FP_{3}|\)}
    {\( |FP_{2}|^{2} = |FP_{1}| \cdot |FP_{3}| \)}
\end{problem}

\begin{answer}
C.
\end{answer}

\begin{solution}
抛物线 \(y^2=2px\) 的焦点为 \(F\left(\dfrac p2,0\right)\). 对任意抛物线上点 \(P_i(x_i,y_i)\)，由 \(y_i^2=2px_i\) 得
\[
|FP_i|^2=\left(x_i-\frac p2\right)^2+y_i^2
=\left(x_i+\frac p2\right)^2
\]
因 \(x_i\geqslant0\)，故 \(|FP_i|=x_i+\dfrac p2\). 由 \(2x_2=x_1+x_3\)，可得
\[
2|FP_2|=2x_2+p=x_1+x_3+p=|FP_1|+|FP_3|
\]
故选 C.
\end{solution}

\begin{problem}
已知某个几何体的三视图如下，根据图中标出的尺寸（单位：\(\mathrm{cm}\)），可得这个几何体的体积是（\quad）

\bitmapfigure[width=0.45\linewidth]{img/2007/new_curriculum_liberal/q8_fig1.png}

\choices
    {\(\frac{4000}{3} \mathrm{~cm}^{3}\)}
    {\(\frac{8000}{3} \mathrm{~cm}^{3}\)}
    {\(2000 \mathrm{~cm}^{3}\)}
    {\(4000 \mathrm{~cm}^{3}\)}
\end{problem}

\begin{answer}
B.
\end{answer}

\begin{solution}
由俯视图可知该几何体的底面为边长 \(20\,\mathrm{cm}\) 的正方形，底面积为 \(20\times20=400\,\mathrm{cm}^2\). 由正视图可知几何体的高为 \(20\,\mathrm{cm}\). 因而它的体积为
\[
V=\frac13\times400\times20=\frac{8000}{3}\,\mathrm{cm}^3
\]
故选 B.
\end{solution}

\begin{problem}
若 \(\frac{\cos 2\alpha}{\sin \left(\alpha - \frac{\pi}{4}\right)} = -\frac{\sqrt{2}}{2}\)，则 \(\cos \alpha + \sin \alpha\) 的值为（\quad）

\choices
    {\(-\frac{\sqrt{7}}{2}\)}
    {\(-\frac{1}{2}\)}
    {\(\frac{1}{2}\)}
    {\(\frac{\sqrt{7}}{2}\)}
\end{problem}

\begin{answer}
C.
\end{answer}

\begin{solution}
由分母不为零知 \(\sin\left(\alpha-\dfrac\pi4\right)\ne0\)，即 \(\cos\alpha-\sin\alpha\ne0\). 利用恒等式
\(\cos2\alpha=(\cos\alpha+\sin\alpha)(\cos\alpha-\sin\alpha)\)，\(\sin\left(\alpha-\frac\pi4\right)=\frac{\sin\alpha-\cos\alpha}{\sqrt2}\)
原等式左边化为 \(-\sqrt2(\cos\alpha+\sin\alpha)\). 所以
\[
-\sqrt2(\cos\alpha+\sin\alpha)=-\frac{\sqrt2}{2}
\]
从而 \(\cos\alpha+\sin\alpha=\dfrac12\)，故选 C.
\end{solution}

\begin{problem}
曲线 \( y = \e^{x} \) 在点 \((2, \e^{2})\) 处的切线与坐标轴所围三角形的面积为（\quad）

\choices
    {\(\frac{9}{4}\e^2\)}
    {\(2\e^2\)}
    {\(\e^2\)}
    {\(\frac{\e^2}{2}\)}
\end{problem}

\begin{answer}
D.
\end{answer}

\begin{solution}
函数 \(y=\e^x\) 的导数为 \(y'=\e^x\)，所以在 \((2,\e^2)\) 处的切线方程为
\[
y-\e^2=\e^2(x-2),\qquad\text{即}\qquad y=\e^2x-\e^2
\]
它与 \(x\) 轴交于 \((1,0)\)，与 \(y\) 轴交于 \((0,-\e^2)\). 所围三角形的两条直角边长分别为 \(1\) 和 \(\e^2\)，故面积为 \(\dfrac{\e^2}{2}\)，选 D.
\end{solution}

\begin{problem}
已知三棱锥 \(S - ABC\) 的各顶点都在一个半径为 \(r\) 的球面上，球心 \(O\) 在 \(AB\) 上，\(SO \perp\) 底面 \(ABC\)，\(AC = \sqrt{2} r\)，则球的体积与三棱锥体积之比是（\quad）

\choices
    {\(\pi\)}
    {\(2\pi\)}
    {\(3\pi\)}
    {\(4\pi\)}
\end{problem}

\begin{answer}
D.
\end{answer}

\begin{solution}
由于 \(O\) 是球心且 \(O\in AB\)，\(AB\) 是球的直径，所以 \(AB=2r\). 又 \(SO\perp\) 平面 \(ABC\)，且 \(SO=r\)，故三棱锥的高为 \(r\). 由直径所对的圆周角为直角，\(\angle ACB=90^\circ\). 在直角三角形 \(ABC\) 中，
\[
BC=\sqrt{AB^2-AC^2}=\sqrt{(2r)^2-(\sqrt2r)^2}=\sqrt2r
\]
所以底面面积为
\[
S_{\triangle ABC}=\frac12\,AC\cdot BC=\frac12\cdot\sqrt2r\cdot\sqrt2r=r^2
\]
三棱锥体积为 \(\dfrac13r^3\). 球的体积为 \(\dfrac43\pi r^3\)，因此二者之比为
\[
\frac{\frac43\pi r^3}{\frac13r^3}=4\pi
\]
故选 D.
\end{solution}

\begin{problem}
甲，乙，丙三名射箭运动员在某次测试中各射箭 \(20\) 次，三人的测试成绩如下表

\begin{center}
\begin{tabular}{ccccc}
\multicolumn{5}{c}{甲的成绩} \\
环数 & \(7\) & \(8\) & \(9\) & \(10\) \\
频数 & \(5\) & \(5\) & \(5\) & \(5\)
\end{tabular}
\qquad
\begin{tabular}{ccccc}
\multicolumn{5}{c}{乙的成绩} \\
环数 & \(7\) & \(8\) & \(9\) & \(10\) \\
频数 & \(6\) & \(4\) & \(4\) & \(6\)
\end{tabular}
\qquad
\begin{tabular}{ccccc}
\multicolumn{5}{c}{丙的成绩} \\
环数 & \(7\) & \(8\) & \(9\) & \(10\) \\
频数 & \(4\) & \(6\) & \(6\) & \(4\)
\end{tabular}
\end{center}

\(s_{1}\)，\(s_{2}\)，\(s_{3}\) 分别表示甲，乙，丙三名运动员这次测试成绩的标准差，则有（\quad）

\choices
    {\( s_{3} > s_{1} > s_{2} \)}
    {\( s_{2} > s_{1} > s_{3} \)}
    {\( s_{1} > s_{2} > s_{3} \)}
    {\( s_{2} > s_{3} > s_{1} \)}
\end{problem}

\begin{answer}
B.
\end{answer}

\begin{solution}
三人的平均成绩都为 \(8.5\)，所以可直接比较方差. 对甲、乙、丙分别计算得
\[
s_1^2=\frac{5(1.5^2+0.5^2+0.5^2+1.5^2)}{20}=\frac54
\]
\[
s_2^2=\frac{6\cdot1.5^2+4\cdot0.5^2+4\cdot0.5^2+6\cdot1.5^2}{20}=\frac{29}{20}
\]
\[
s_3^2=\frac{4\cdot1.5^2+6\cdot0.5^2+6\cdot0.5^2+4\cdot1.5^2}{20}=\frac{21}{20}
\]
因标准差均为非负数，\(s_2^2>s_1^2>s_3^2\) 等价于 \(s_2>s_1>s_3\)，故选 B.
\end{solution}

\section{填空题}

\begin{problem}
已知双曲线的顶点到渐近线的距离为 \(2\)，焦点到渐近线的距离为 \(6\)，则该双曲线的离心率为\fillinblank{}.
\end{problem}

\begin{answer}
\(3\).
\end{answer}

\begin{solution}
设双曲线为 \(\dfrac{x^2}{a^2}-\dfrac{y^2}{b^2}=1\)，焦距参数满足 \(c^2=a^2+b^2\). 其渐近线为 \(bx\pm ay=0\). 顶点 \((a,0)\) 到渐近线的距离为
\[
\frac{ab}{\sqrt{a^2+b^2}}=\frac{ab}{c}=2
\]
焦点 \((c,0)\) 到渐近线的距离为
\[
\frac{bc}{\sqrt{a^2+b^2}}=b=6
\]
于是
\[
\frac ca=\frac{b}{ab/c}=\frac{6}{2}=3
\]
故离心率为 \(3\).
\end{solution}

\begin{problem}
设函数 \( f(x) = (x + 1)(x + a) \) 为偶函数，则 \( a = \)\fillinblank{}.
\end{problem}

\begin{answer}
\(-1\).
\end{answer}

\begin{solution}
展开得 \(f(x)=x^2+(a+1)x+a\). 二次函数为偶函数，当且仅当一次项系数为零，因此 \(a+1=0\)，即 \(a=-1\).
\end{solution}

\begin{problem}
\(\i\) 是虚数单位，\(\i + 2\i^2 + 3\i^3 + \cdots + 8\i^8 = \)\fillinblank{}. （用 \(a + b\i\) 的形式表示，\(a\)，\(b \in \R\)）
\end{problem}

\begin{answer}
\(4 - 4\i\).
\end{answer}

\begin{solution}
利用 \(\i^2=-1\)，并按四次幂循环计算，得
\[
\i+2\i^2+3\i^3+4\i^4+5\i^5+6\i^6+7\i^7+8\i^8
=(\i-2-3\i+4+5\i-6-7\i+8)=4-4\i
\]
故填 \(4-4\i\).
\end{solution}

\begin{problem}
已知 \(\{a_{n}\}\) 是等差数列，\(a_{4} + a_{6} = 6\)，其前 5 项和 \(S_{5} = 10\)，则其公差 \(d = \)\fillinblank{}.
\end{problem}

\begin{answer}
\(\frac{1}{2}\).
\end{answer}

\begin{solution}
设等差数列首项为 \(a_1\)，公差为 \(d\). 由 \(a_4+a_6=6\)，得
\[
(a_1+3d)+(a_1+5d)=6
\]
即 \(a_1+4d=3\). 又
\[
S_5=\frac52(2a_1+4d)=5(a_1+2d)=10
\]
所以 \(a_1+2d=2\). 两式相减得 \(2d=1\)，故 \(d=\dfrac12\).
\end{solution}

\section{解答题}

\begin{problem}
如图，测量河对岸的塔高 \(AB\) 时，可以选与塔底 \(B\) 在同一水平面内的两个测点 \(C\) 与 \(D\). 现测得 \(\angle BCD = \alpha\)，\(\angle BDC = \beta\)，\(CD = s\)，并在点 \(C\) 测得塔顶 \(A\) 的仰角为 \(\theta\)，求塔高 \(AB\).
\bitmapfigure[width=0.34\linewidth]{img/2007/new_curriculum_liberal/q17_fig1.png}
\end{problem}

\begin{solution}
在 \(\triangle BCD\) 中，\(\angle CBD=\pi-\alpha-\beta\). 由正弦定理，得
\(\dfrac{BC}{\sin\beta}=\dfrac{CD}{\sin(\pi-\alpha-\beta)}=\dfrac{s}{\sin(\alpha+\beta)}\)，所以 \(BC=\dfrac{s\sin\beta}{\sin(\alpha+\beta)}\).

由于塔 \(AB\) 垂直于地面，且 \(\theta\) 是在点 \(C\) 测得的仰角，在直角三角形 \(ABC\) 中有 \(\tan\theta=\dfrac{AB}{BC}\). 因此塔高为
\[
AB=BC\tan\theta=\frac{s\sin\beta\tan\theta}{\sin(\alpha+\beta)}
\]
\end{solution}

\begin{problem}
如图，\(A\)，\(B\)，\(C\)，\(D\) 为空间四点. 在 \(\triangle ABC\) 中，\(AB = 2\)，\(AC = BC = \sqrt{2}\)，等边三角形 \(ADB\) 以 \(AB\) 为轴运动.
\begin{enumerate}
\item 当平面 \(ADB \perp\) 平面 \(ABC\) 时，求 \(CD\).
\item 当 \(\triangle ADB\) 转动时，是否总有 \(AB \perp CD\)？ 证明你的结论.
\end{enumerate}

\bitmapfigure[width=0.42\linewidth]{img/2007/new_curriculum_liberal/q18_fig1.png}
\end{problem}

\begin{solution}
设 \(O\) 为 \(AB\) 的中点. 由 \(AC=BC=\sqrt{2}\)，\(AB=2\)，可得 \(AB^{2}=AC^{2}+BC^{2}\)，所以 \(\angle ACB=\dfrac{\pi}{2}\). 因而 \(CO\) 是直角三角形 \(ABC\) 斜边上的中线，且 \(CO=AO=1\)，\(CO\perp AB\).

又因为 \(\triangle ADB\) 是边长为 \(2\) 的等边三角形，\(DO\) 是其中线和高，所以 \(DO\perp AB\)，且 \(DO=\sqrt{2^{2}-1^{2}}=\sqrt{3}\).

\begin{enumerate}
\item 当平面 \(ADB\perp\) 平面 \(ABC\) 时，两平面的交线为 \(AB\). 由于 \(CO\) 在平面 \(ABC\) 内且 \(CO\perp AB\)，由二面角为直二面角可知 \(CO\perp\) 平面 \(ADB\)，从而 \(CO\perp DO\). 在直角三角形 \(COD\) 中，\(CD^{2}=CO^{2}+DO^{2}=1+3=4\)，故 \(CD=2\).
\item 在 \(\triangle ABC\) 中，\(\overrightarrow{OC}\perp\overrightarrow{AB}\)；在等边三角形 \(ADB\) 中，\(\overrightarrow{OD}\perp\overrightarrow{AB}\). 因此
\(\overrightarrow{CD}=\overrightarrow{OD}-\overrightarrow{OC}\)，从而 \(\overrightarrow{CD}\cdot\overrightarrow{AB}=0\). 所以无论等边三角形 \(ADB\) 如何绕 \(AB\) 转动，总有 \(AB\perp CD\).
\end{enumerate}
\end{solution}

\begin{problem}
设函数 \( f(x) = \ln (2x + 3) + x^2 \).
\begin{enumerate}
\item 讨论 \( f(x) \) 的单调性.
\item 求 \( f(x) \) 在区间 \( \left[-\frac{3}{4}, \frac{1}{4}\right] \) 的最大值和最小值.
\end{enumerate}
\end{problem}

\begin{solution}
函数的定义域为 \(\left(-\dfrac{3}{2},+\infty\right)\)，且
\(f'(x)=\dfrac{2}{2x+3}+2x=\dfrac{2(2x+1)(x+1)}{2x+3}\).

\begin{enumerate}
\item 在定义域内分点 \(-1\)、\(-\dfrac{1}{2}\) 讨论导数符号. 因为分母 \(2x+3>0\)，所以 \(f'(x)>0\) 当 \(x\in\left(-\dfrac{3}{2},-1\right)\) 或 \(x\in\left(-\dfrac{1}{2},+\infty\right)\)，\(f'(x)<0\) 当 \(x\in\left(-1,-\dfrac{1}{2}\right)\). 因此 \(f(x)\) 在 \(\left(-\dfrac{3}{2},-1\right)\) 和 \(\left(-\dfrac{1}{2},+\infty\right)\) 上单调递增，在 \(\left(-1,-\dfrac{1}{2}\right)\) 上单调递减.
\item 在区间 \(\left[-\dfrac{3}{4},\dfrac{1}{4}\right]\) 内，函数先减后增，因此最小值在 \(x=-\dfrac{1}{2}\) 处取得，且
\(f\left(-\dfrac{1}{2}\right)=\ln2+\dfrac{1}{4}\).

区间两个端点处的函数值为 \(f\left(-\dfrac{3}{4}\right)=\ln\dfrac{3}{2}+\dfrac{9}{16}\) 和 \(f\left(\dfrac{1}{4}\right)=\ln\dfrac{7}{2}+\dfrac{1}{16}\). 又因为
\(\ln\dfrac{7}{3}=\displaystyle\int_{1}^{7/3}\dfrac{1}{t}\,\mathrm{d}t>\dfrac{4}{7}>\dfrac{1}{2}\)，所以 \(f\left(\dfrac{1}{4}\right)-f\left(-\dfrac{3}{4}\right)=\ln\dfrac{7}{3}-\dfrac{1}{2}>0\). 故最大值在 \(x=\dfrac{1}{4}\) 处取得，为 \(\ln\dfrac{7}{2}+\dfrac{1}{16}\).
\end{enumerate}
\end{solution}

\begin{problem}
设有关于 \(x\) 的一元二次方程 \(x^{2} + 2ax + b^{2} = 0\).
\begin{enumerate}
\item 若 \(a\) 是从 \(0\)，\(1\)，\(2\)，\(3\) 四个数中任取的一个数，\(b\) 是从 \(0\)，\(1\)，\(2\) 三个数中任取的一个数，求上述方程有实根的概率.
\item 若 \(a\) 是从区间 \([0,3]\) 任取的一个数，\(b\) 是从区间 \([0,2]\) 任取的一个数，求上述方程有实根的概率.
\end{enumerate}
\end{problem}

\begin{solution}
关于 \(x\) 的一元二次方程有实根的充要条件是判别式 \(\Delta\geqslant0\)，即
\(\Delta=(2a)^{2}-4b^{2}\geqslant0\)，从而 \(a^{2}\geqslant b^{2}\). 由于本题中 \(a\geqslant0\)，\(b\geqslant0\)，该条件等价于 \(a\geqslant b\).

\begin{enumerate}
\item 基本事件总数为 \(4\times3=12\). 当 \(b=0\) 时有 \(4\) 个满足条件的 \(a\)；当 \(b=1\) 时有 \(3\) 个；当 \(b=2\) 时有 \(2\) 个. 因而有实根的基本事件数为 \(4+3+2=9\)，所求概率为 \(\dfrac{9}{12}=\dfrac{3}{4}\).
\item \((a,b)\) 在矩形 \(0\leqslant a\leqslant3\)，\(0\leqslant b\leqslant2\) 内均匀取值，全部区域面积为 \(3\times2=6\). 满足 \(a\geqslant b\) 的区域面积为
\(\displaystyle\int_{0}^{2}a\,\mathrm{d}a+\int_{2}^{3}2\,\mathrm{d}a=2+2=4\). 因而所求概率为 \(\dfrac{4}{6}=\dfrac{2}{3}\).
\end{enumerate}
\end{solution}

\begin{problem}
在平面直角坐标系 \(xOy\) 中，已知圆 \(x^{2} + y^{2} - 12x + 32 = 0\) 的圆心为 \(Q\)，过点 \(P(0,2)\) 且斜率为 \(k\) 的直线与圆 \(Q\) 相交于不同的交点 \(A\)，\(B\).
\begin{enumerate}
\item 求 \(k\) 的取值范围.
\item 是否存在常数 \(k\)，使得向量 \(\overrightarrow{OA} + \overrightarrow{OB}\) 与 \(\overrightarrow{PQ}\) 共线？如果存在，求 \(k\) 值；如果不存在，请说明理由.
\end{enumerate}
\end{problem}

\begin{solution}
圆的方程可化为 \((x-6)^{2}+y^{2}=4\)，所以圆心为 \(Q(6,0)\)，半径为 \(2\). 过点 \(P(0,2)\) 且斜率为 \(k\) 的直线为 \(kx-y+2=0\).

\begin{enumerate}
\item 直线与圆相交于不同的两点，当且仅当圆心到直线的距离小于半径，即
\(\dfrac{|6k+2|}{\sqrt{k^{2}+1}}<2\). 两边平方并化简，得 \((6k+2)^{2}<4(k^{2}+1)\)，即 \(8k(4k+3)<0\)，所以 \(-\dfrac{3}{4}<k<0\).
\item 设弦 \(AB\) 的中点为 \(M\). 因为 \(QM\perp AB\)，所以 \(M\) 是点 \(Q\) 到直线 \(kx-y+2=0\) 的垂足. 由垂足公式，得
\(M\left(\dfrac{6-2k}{k^{2}+1},\dfrac{6k+2}{k^{2}+1}\right)\).

由于 \(M\) 是弦 \(AB\) 的中点，\(\overrightarrow{OA}+\overrightarrow{OB}=2\overrightarrow{OM}\). 又 \(\overrightarrow{PQ}=(6,-2)\)，故题设共线条件等价于 \(\overrightarrow{OM}\) 与 \((6,-2)\) 共线，即 \(x_M+3y_M=0\). 代入 \(M\) 的坐标，得
\(\dfrac{6-2k+3(6k+2)}{k^{2}+1}=0\)，从而 \(k=-\dfrac{3}{4}\). 但这不满足第（1）问的严格不等式，表示此时直线为圆的切线，不能得到两个不同的交点. 所以不存在满足条件的常数 \(k\).
\end{enumerate}
\end{solution}

\begin{problem}
\(\odot O_{1}\) 和 \(\odot O_{2}\) 的极坐标方程分别为 \(\rho = 4\cos \theta\)，\(\rho = -4\sin \theta\).
\begin{enumerate}
\item 把 \(\odot O_{1}\) 和 \(\odot O_{2}\) 的极坐标方程化为直角坐标方程.
\item 求经过 \(\odot O_{1}\)，\(\odot O_{2}\) 交点的直线的直角坐标方程.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由极坐标与直角坐标的关系 \(x=\rho\cos\theta\)，\(y=\rho\sin\theta\)，\(\rho^{2}=x^{2}+y^{2}\)，对 \(\rho=4\cos\theta\) 两边同乘 \(\rho\)，得 \(\rho^{2}=4\rho\cos\theta=4x\)，所以 \(\odot O_{1}\) 的直角坐标方程为 \(x^{2}+y^{2}-4x=0\)，即 \((x-2)^{2}+y^{2}=4\).

对 \(\rho=-4\sin\theta\) 两边同乘 \(\rho\)，得 \(\rho^{2}=-4\rho\sin\theta=-4y\)，所以 \(\odot O_{2}\) 的直角坐标方程为 \(x^{2}+y^{2}+4y=0\)，即 \(x^{2}+(y+2)^{2}=4\).
\item 两圆交点满足
\(x^{2}+y^{2}-4x=0\)，\(x^{2}+y^{2}+4y=0\). 两式相减得 \(x+y=0\)，即 \(y=-x\). 代入第一式，得 \(2x^{2}-4x=0\)，所以 \(x=0\) 或 \(x=2\)，相应交点为 \((0,0)\) 和 \((2,-2)\). 过这两点的直线斜率为 \(-1\)，故所求直角坐标方程为 \(y=-x\)，即 \(x+y=0\).
\end{enumerate}
\end{solution}

\begin{problem}
设函数 \( f(x) = |2x + 1| - |x - 4| \).
\begin{enumerate}
\item 解不等式 \( f(x) > 2 \).
\item 求函数 \( y = f(x) \) 的最小值.
\end{enumerate}
\end{problem}

\begin{solution}
函数的分界点为 \(x=-\dfrac{1}{2}\) 和 \(x=4\). 分三段讨论：
\[
f(x)=
\begin{cases}
-x-5, & x< -\dfrac{1}{2},\\
3x-3, & -\dfrac{1}{2}\leqslant x<4,\\
x+5, & x\geqslant4.
\end{cases}
\]

\begin{enumerate}
\item 当 \(x< -\dfrac{1}{2}\) 时，\(-x-5>2\) 等价于 \(x<-7\)；当 \(-\dfrac{1}{2}\leqslant x<4\) 时，\(3x-3>2\) 等价于 \(x>\dfrac{5}{3}\)；当 \(x\geqslant4\) 时，\(x+5>2\) 恒成立. 合并得不等式的解集为 \(\left(-\infty,-7\right)\cup\left(\dfrac{5}{3},+\infty\right)\).
\item 在第一段，\(f(x)=-x-5\) 单调递减；在后两段，\(f(x)\) 分别单调递增，且在分界点处连续. 因此函数在 \(x=-\dfrac{1}{2}\) 处取得最小值，最小值为 \(f\left(-\dfrac{1}{2}\right)=-\dfrac{9}{2}\).
\end{enumerate}
\end{solution}
