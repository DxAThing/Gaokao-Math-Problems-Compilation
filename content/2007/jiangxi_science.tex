\section{单选题}

\begin{problem}
化简 \(\frac{2 + 4\i}{(1 + \i)^2}\) 的结果是（\quad）

\choices
    {\(2 + \i\)}
    {\(-2 + \i\)}
    {\(2 - \i\)}
    {\(-2 - \i\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
因为 \((1+\i)^2=2\i\)，所以
\[
\frac{2+4\i}{(1+\i)^2}=\frac{2+4\i}{2\i}=\frac{1+2\i}{\i}=2-\i
\]
故选 C.
\end{solution}

\begin{problem}
\(\displaystyle \lim\limits_{x\to 1}\frac{x^3 - x^2}{x - 1}\)（\quad）

\choices
    {等于 \(0\)}
    {等于 \(1\)}
    {等于 \(3\)}
    {不存在}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
当 \(x\ne1\) 时，
\[
\frac{x^3-x^2}{x-1}=\frac{x^2(x-1)}{x-1}=x^2
\]
因此
\[
\displaystyle\lim\limits_{x\to1}\frac{x^3-x^2}{x-1}=\lim\limits_{x\to1}x^2=1
\]
故选 B.
\end{solution}

\begin{problem}
若 \(\tan \left(\frac{\pi}{4} - \alpha\right) = 3\)，则 \(\cot \alpha\) 等于（\quad）

\choices
    {\(-2\)}
    {\(-\frac{1}{2}\)}
    {\(\frac{1}{2}\)}
    {\(2\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
由正切差角公式，
\[
\tan\left(\frac{\pi}{4}-\alpha\right)=\frac{1-\tan\alpha}{1+\tan\alpha}=3
\]
解得 \(1-\tan\alpha=3+3\tan\alpha\)，所以 \(\tan\alpha=-\frac12\)，从而 \(\cot\alpha=-2\).
故选 A.
\end{solution}

\begin{problem}
已知 \(\left(\sqrt{x} + \frac{3}{\sqrt[3]{x}}\right)^n\) 展开式中，各项系数的和与其各项二项式系数的和之比为 \(64\)，则 \(n\) 等于（\quad）

\choices
    {\(4\)}
    {\(5\)}
    {\(6\)}
    {\(7\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
令各项系数之和为 \(S_1\)，各项二项式系数之和为 \(S_2\). 令展开式中的 \(x=1\)，得
\[
S_1=\left(1+\frac{3}{1}\right)^n=4^n
\]
而二项式系数之和为 \(S_2=2^n\)，所以
\[
\frac{S_1}{S_2}=\frac{4^n}{2^n}=2^n=64=2^6
\]
故 \(n=6\)，选 C.
\end{solution}

\begin{problem}
若 \(0 < x < \frac{\pi}{2}\)，则下列命题中正确的是（\quad）

\choices
    {\(\sin x < \frac{3}{\pi}x\)}
    {\(\sin x > \frac{3}{\pi}x\)}
    {\(\sin x < \frac{4}{\pi^{2}}x^{2}\)}
    {\(\sin x > \frac{4}{\pi^{2}}x^{2}\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
因为 \(y''=-\sin x<0\)（\(0<x<\frac{\pi}{2}\)），所以 \(y=\sin x\) 在 \(\left[0,\frac{\pi}{2}\right]\) 上的图象位于端点弦 \(y=\frac{2}{\pi}x\) 的上方，即
\[
\sin x>\frac{2}{\pi}x
\]
又因为 \(0<x<\frac{\pi}{2}\)，有 \(\frac{2}{\pi}x>\frac{4}{\pi^2}x^2\)，故
\[
\sin x>\frac{4}{\pi^2}x^2
\]
另外，\(\displaystyle\lim_{x\to0^+}\frac{\sin x}{x}=1>\frac{3}{\pi}\)，所以第一个命题不恒成立；在 \(x\) 充分接近 \(\frac{\pi}{2}\) 时，\(\frac{\sin x}{x}\) 接近 \(\frac{2}{\pi}<\frac{3}{\pi}\)，所以第二个命题不恒成立；在 \(x\to0^+\) 时，\(\sin x/x^2\to+\infty\)，所以第三个命题不恒成立.
故选 D.
\end{solution}

\begin{problem}
若集合 \(M = \{0, 1, 2\}\)，\(N = \{(x, y) \mid x - 2y + 1 \geqslant 0 \text{ 且 } x - 2y - 1 \leqslant 0,\ x, y \in M\}\)，则 \(N\) 中元素的个数为（\quad）

\choices
    {\(9\)}
    {\(6\)}
    {\(4\)}
    {\(2\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
两个不等式等价于 \(-1\leqslant x-2y\leqslant1\). 当 \(y=0\) 时，符合条件的 \(x\) 为 \(0,1\)；当 \(y=1\) 时，符合条件的 \(x\) 为 \(1,2\)；当 \(y=2\) 时没有符合条件的 \(x\). 因而
\[
N=\{(0,0),(1,0),(1,1),(2,1)\}
\]
共有 \(4\) 个元素，故选 C.
\end{solution}

\begin{problem}
如图，正方体 \(ABCD - A_{1}B_{1}C_{1}D_{1}\) 的棱长为 \(1\)，过点 \(A\) 作平面 \(A_{1}BD\) 的垂线，垂足为点 \(H\). 则以下命题中，错误的是（\quad）

\bitmapfigure[max width=0.38\linewidth]{img/2007/jiangxi_science/q7_fig1.png}

\choices
    {点 \(H\) 是 \(\triangle A_{1}BD\) 的垂心}
    {\(AH\) 垂直平面 \(CB_{1}D_{1}\)}
    {\(AH\) 的延长线经过点 \(C_{1}\)}
    {直线 \(AH\) 和 \(BB_{1}\) 所成的角为 \(45^{\circ}\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
建立直角坐标系，取
\(A=(0,0,1)\)，\(B=(1,0,1)\)，\(D=(0,1,1)\)，\(A_1=(0,0,0)\).
平面 \(A_1BD\) 的方程为 \(z=x+y\)，其法向量可取 \((-1,-1,1)\). 由点 \(A\) 沿法向量作垂线，得到垂足
\[
H=\left(\frac13,\frac13,\frac23\right)
\]
有 \(\overrightarrow{BH}=\left(-\frac23,\frac13,-\frac13\right)\)，\(\overrightarrow{A_1D}=(0,1,1)\)，所以 \(\overrightarrow{BH}\cdot\overrightarrow{A_1D}=0\). 另外 \(\overrightarrow{DH}=\left(\frac13,-\frac23,-\frac13\right)\)，\(\overrightarrow{A_1B}=(1,0,1)\)，从而 \(\overrightarrow{DH}\cdot\overrightarrow{A_1B}=0\)，故 A 正确.

平面 \(CB_1D_1\) 的一个法向量为 \((1,1,-1)\)，而 \(\overrightarrow{AH}=\left(\frac13,\frac13,-\frac13\right)\) 与之平行，故 B 正确. 又 \(C_1-A=(1,1,-1)=3\overrightarrow{AH}\)，故 C 正确.

直线 \(AH\) 与 \(BB_1\) 的方向向量可分别取 \((1,1,-1)\) 和 \((0,0,1)\)，两直线所成角 \(\varphi\) 满足
\[
\cos\varphi=\frac{1}{\sqrt3}
\]
故 \(\varphi\ne45^{\circ}\). 所以错误的是 D.
\end{solution}

\begin{problem}
四位好朋友在一次聚会上，他们按照各自的爱好选择了形状不同，内空高度相等，杯口半径相等的圆口酒杯，如图所示，盛满酒后他们约定：先各自饮杯中酒的一半. 设剩余酒的高度从左到右依次为 \(h_1\)，\(h_2\)，\(h_3\)，\(h_4\)，则它们的大小关系正确的是（\quad）

\FigureLayoutDeclare{img/2007/jiangxi_science/q8_fig1.png}{12.0cm}{144bp 12bp 117bp 52bp}
\bitmapfigure[max width=0.82\linewidth]{img/2007/jiangxi_science/q8_fig1.png}

\choices
    {\(h_2 > h_1 > h_4\)}
    {\(h_1 > h_2 > h_3\)}
    {\(h_3 > h_2 > h_4\)}
    {\(h_2 > h_4 > h_1\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
设四个酒杯的内空高度均为 \(H\)，杯口半径均为 \(R\). 第二个杯子为尖底圆锥形，设液面高度为 \(h\)，由相似三角形得
\[
\frac{V_2(h)}{V_2(H)}=\left(\frac{h}{H}\right)^3
\]
所以饮去一半后 \(h_2=2^{-1/3}H\). 第四个杯子为圆柱形，所以 \(h_4=\frac12H\).

记 \(V_i(t)\) 为第 \(i\) 个杯子从杯底到高度 \(t\) 的容积. 由截面图形可知，在相同的相对高度 \(0<t<H\) 下，第一杯的累计容积占比介于第二杯和第四杯之间，即：
\[
\frac{V_2(t)}{V_2(H)}
<
\frac{V_1(t)}{V_1(H)}
<
\frac{V_4(t)}{V_4(H)}
\]
第二杯、第四杯达到半容积的高度分别为 \(h_2\)、\(h_4\)，故
\(\frac{V_2(h_2)}{V_2(H)}=\frac12\)，\(\frac{V_4(h_4)}{V_4(H)}=\frac12\).
代入上面的相对容积比较，得到
\(\frac{V_1(h_2)}{V_1(H)}>\frac12\)，\(\frac{V_1(h_4)}{V_1(H)}<\frac12\).
由于 \(V_1(t)\) 随 \(t\) 严格递增，得
\[
h_2>h_1>h_4
\]
故选 A.
\end{solution}

\begin{problem}
设椭圆 \(\frac{x^{2}}{a^{2}} + \frac{y^{2}}{b^{2}} = 1\)（\(a > b > 0\)）的离心率为 \(e = \frac{1}{2}\)，右焦点为 \(F(c, 0)\). 方程 \(ax^{2} + bx - c = 0\) 的两个实根分别为 \(x_{1}\) 和 \(x_{2}\)，则点 \(P(x_{1}, x_{2})\)（\quad）

\choices
    {必在圆 \(x^{2} + y^{2} = 2\) 内}
    {必在圆 \(x^{2} + y^{2} = 2\) 上}
    {必在圆 \(x^{2} + y^{2} = 2\) 外}
    {以上三种情形都有可能}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
由椭圆的离心率，\(c/a=1/2\)，所以
\[
\frac{b^2}{a^2}=1-\frac{c^2}{a^2}=\frac34
\]
由韦达定理，
\(x_1+x_2=-\frac ba\)，\(x_1x_2=-\frac ca=-\frac12\).
因此
\[
x_1^2+x_2^2=(x_1+x_2)^2-2x_1x_2=\frac{b^2}{a^2}+1=\frac74<2
\]
所以点 \(P(x_1,x_2)\) 必在圆 \(x^2+y^2=2\) 内，故选 A.
\end{solution}

\begin{problem}
将一个骰子连续抛掷三次，它落地时向上的点数依次成等差数列的概率为（\quad）

\choices
    {\(\frac{1}{9}\)}
    {\(\frac{1}{12}\)}
    {\(\frac{1}{15}\)}
    {\(\frac{1}{18}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
设三次点数为 \(a\)，\(b\)，\(c\)，则成为等差数列的条件是 \(a+c=2b\). 对固定的中项 \(b\)，首项和末项可写成 \(b-d\)，\(b+d\)，其中 \(d\) 为整数且三项都在 \(\{1,2,3,4,5,6\}\) 中. 当 \(d=0\) 时有 \(6\) 种；当 \(d=\pm1\) 时各有 \(4\) 种；当 \(d=\pm2\) 时各有 \(2\) 种. 因而有利结果数为
\[
6+2\times4+2\times2=18
\]
总结果数为 \(6^3=216\)，所以所求概率为 \(18/216=1/12\)，故选 B.
\end{solution}

\begin{problem}
设函数 \(f(x)\) 是 \(\R\) 上以 \(5\) 为周期的可导偶函数，则曲线 \(y=f(x)\) 在 \(x=5\) 处的切线的斜率为（\quad）

\choices
    {\(-\frac{1}{5}\)}
    {\(0\)}
    {\(\frac{1}{5}\)}
    {\(5\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
由周期性，\(f(x+5)=f(x)\)，对其求导得 \(f'(x+5)=f'(x)\)，所以 \(f'(5)=f'(0)\). 又因为 \(f\) 是偶函数，\(f'\) 是奇函数，从而 \(f'(0)=0\). 因此切线斜率为 \(0\)，故选 B.
\end{solution}

\begin{problem}
设 \(p\)：\(f(x) = \e^{x} + \ln x + 2x^{2} + mx + 1\) 在 \((0, +\infty)\) 内单调递增，\(q\)：\(m \geqslant -5\)，则 \(p\) 是 \(q\) 的（\quad）

\choices
    {充分不必要条件}
    {必要不充分条件}
    {充分必要条件}
    {既不充分也不必要条件}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
函数的导数为
\[
f'(x)=\e^x+\frac1x+4x+m
\]
由 \(\e^x>1\) 以及均值不等式 \(4x+\frac1x\geqslant4\)，有
\[
\e^x+\frac1x+4x>5
\]
因此当 \(m\geqslant-5\) 时，\(f'(x)>0\)，命题 \(q\) 是命题 \(p\) 的充分条件.

另一方面，取 \(m=-\frac{27}{5}<-5\). 由 \(\e^x>1+x\) 和 \(5x+\frac1x\geqslant2\sqrt5\)，有
\[
f'(x)>1+5x+\frac1x-\frac{27}{5}\geqslant2\sqrt5-\frac{22}{5}>0
\]
所以此时命题 \(p\) 成立而命题 \(q\) 不成立，说明 \(p\) 不是 \(q\) 的充分条件. 综合两点，\(q\) 是 \(p\) 的充分而不必要条件，等价地，\(p\) 是 \(q\) 的必要不充分条件，选 B.
\end{solution}

\section{填空题}

\begin{problem}
设函数 \(y = 4 + \log_{2}(x - 1)\)（\(x \geqslant 3\)），则其反函数的定义域为\fillinblank{}.
\end{problem}

\begin{answer}
\([5,+\infty)\)
\end{answer}

\begin{solution}
原函数的定义域为 \([3,+\infty)\). 在该区间上，\(\log_2(x-1)\) 单调递增，故原函数的值域从最小值
\[
4+\log_2(3-1)=5
\]
开始取到，并且没有上界，即值域为 \([5,+\infty)\). 反函数的定义域等于原函数的值域，所以填 \([5,+\infty)\).
\end{solution}

\begin{problem}
已知数列 \(\{a_n\}\) 对于任意 \(p\)，\(q \in \N^*\)，有 \(a_p + a_q = a_{p+q}\)，若 \(a_1 = \frac{1}{9}\)，则 \(a_{36} =\)\fillinblank{}.
\end{problem}

\begin{answer}
4
\end{answer}

\begin{solution}
令 \(q=1\)，由题设得 \(a_p+a_1=a_{p+1}\). 依次取 \(p=1\)，\(2\)，\(\ldots\)，\(n-1\)，可得
\[
a_n=na_1
\]
因此
\[
a_{36}=36a_1=36\times\frac19=4
\]
所以填 \(4\).
\end{solution}

\begin{problem}
如图，在 \(\triangle ABC\) 中，点 \(O\) 是 \(BC\) 的中点，过点 \(O\) 的直线分别交直线 \(AB\)，\(AC\) 于不同的两点 \(M\)，\(N\)，若 \(\overrightarrow{AB} = m\overrightarrow{AM}\)，\(\overrightarrow{AC} = n\overrightarrow{AN}\)，则 \(m + n\) 的值为\fillinblank{}.

\bitmapfigure[max width=0.38\linewidth]{img/2007/jiangxi_science/q15_fig1.png}
\end{problem}

\begin{answer}
2
\end{answer}

\begin{solution}
取点 \(O\) 为原点，记 \(\overrightarrow{OA}=\bs a\)，\(\overrightarrow{OB}=\bs b\). 因为 \(O\) 是 \(BC\) 的中点，所以 \(\overrightarrow{OC}=-\bs b\). 由题设
\(\overrightarrow{OM}=\left(1-\frac1m\right)\bs a+\frac1m\bs b\)，\(\overrightarrow{ON}=\left(1-\frac1n\right)\bs a-\frac1n\bs b\).
点 \(M\)，\(O\)，\(N\) 共线，且 \(\bs a\)，\(\bs b\) 不共线，因此上述两个向量的系数行列式为零：
\[
\left(1-\frac1m\right)\left(-\frac1n\right)-\frac1m\left(1-\frac1n\right)=0
\]
乘以 \(mn\) 得 \(-(n-1)-(m-1)=0\)，所以 \(m+n=2\).
\end{solution}

\begin{problem}
设有一组圆 \(C_k: (x - k + 1)^{2} + (y - 3k)^{2} = 2k^{4}\)（\(k \in \N^*\)）. 下面四个命题：

\choices
    {存在一条定直线与所有的圆均相切}
    {存在一条定直线与所有的圆均相交}
    {存在一条定直线与所有的圆均不相交}
    {所有的圆均不经过原点}

其中真命题的代号是\fillinblank{}.（写出所有真命题的代号）
\end{problem}

\begin{answer}
BD
\end{answer}

\begin{solution}
圆 \(C_k\) 的圆心和半径分别为
\(O_k=(k-1,3k)\)，\(r_k=\sqrt2 k^2\).
这些圆心都在定直线 \(y=3x+3\) 上，所以该直线经过每个圆心，必与所有的圆相交，命题 B 正确.

若定直线为 \(ux+vy+w=0\)，其到 \(O_k\) 的距离的平方是关于 \(k\) 的二次式，而 \(r_k^2=2k^4\) 是四次式，因此不可能对所有正整数 \(k\) 都相等，命题 A 错误. 对任意定直线，圆心到直线的距离至多随 \(k\) 线性增长，而半径按 \(k^2\) 增长，所以当 \(k\) 足够大时该直线必与圆 \(C_k\) 相交，命题 C 错误.

原点在圆 \(C_k\) 上的条件为
\[
(k-1)^2+9k^2=2k^4
\]
当 \(k=1\)，\(2\) 时等式左边分别为 \(9,37\)，右边分别为 \(2,32\)，均不相等；当 \(k\geqslant3\) 时
\[
2k^4-(k-1)^2-9k^2=2k^4-10k^2+2k-1>0
\]
因为 \(2k^4-10k^2+2k-1=2k^2(k^2-5)+2k-1\)，且 \(k\geqslant3\) 时右侧大于 \(0\)，也不可能相等. 所以第四个命题正确；结合前面的结论，真命题为第二、第四个命题的代号.
\end{solution}

\section{解答题}

\begin{problem}
已知函数 \(f(x)=\begin{cases}
cx + 1, & 0 < x < c,\\
2^{-\frac{x}{c^{2}}} + k, & c \leqslant x < 1,
\end{cases}\) 在区间 \((0,1)\) 内连续，且 \(f(c^{2}) = \frac{9}{8}\).

\begin{enumerate}
\item 求实数 \(k\) 和 \(c\) 的值；
\item 解不等式 \(f(x) > \frac{\sqrt{2}}{8} + 1\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 因为 \(f(c^2)\) 有意义，所以 \(c^2\in(0,1)\)，从而 \(-1<c<1\) 且 \(c\ne0\). 先按本题通常约定的分界点 \(0<c<1\) 讨论，此时 \(0<c^2<c\)，从而
\[
f(c^2)=c\cdot c^2+1=c^3+1=\frac98
\]
因此 \(c^3=\frac18\)，结合 \(c>0\) 得 \(c=\frac12\). 连续性要求在 \(x=c\) 处左右函数值相等，即
\[
c^2+1=2^{-1/c}+k
\]
代入 \(c=\frac12\)，得
\[
k=\frac54-2^{-2}=1
\]

\item 将 \(c=\frac12\)，\(k=1\) 代入，得
\[
f(x)=\begin{cases}
\frac{x}{2}+1, & 0<x<\frac12,\\
2^{-4x}+1, & \frac12\leqslant x<1.
\end{cases}
\]
当 \(0<x<\frac12\) 时，
\[
\frac{x}{2}+1>1+\frac{\sqrt2}{8}\quad\Longleftrightarrow\quad x>\frac{\sqrt2}{4}
\]
当 \(\frac12\leqslant x<1\) 时，
\[
2^{-4x}+1>1+\frac{\sqrt2}{8}
\quad\Longleftrightarrow\quad
2^{-4x}>2^{-5/2}
\quad\Longleftrightarrow\quad
x<\frac58
\]
合并两段并注意 \(x=\frac12\) 时不等式成立，解集为
\[
\left(\frac{\sqrt2}{4},\frac58\right)
\]
\end{enumerate}

严格按原题字面还需讨论 \(-1<c<0\). 此时第一段在 \((0,1)\) 内为空，第二段在该区间内给出函数值，且 \(c^2\in(0,1)\). 因而
\[
f(c^2)=2^{-c^2/c^2}+k=\frac12+k=\frac98
\]
从而 \(k=\frac58\). 此时不等式在 \((0,1)\) 内等价于
\[
2^{-x/c^2}>\frac{3+\sqrt2}{8}
\quad\Longleftrightarrow\quad
0<x<c^2\log_2\frac{8}{3+\sqrt2}
\]
所以原题若不隐含 \(0<c<1\)，则 \(c\)，\(k\) 并不唯一；上面的 \(c=\frac12\)，\(\ k=1\) 及解集是按高考题的标准约定 \(0<c<1\) 得到的.
\end{solution}

\begin{problem}
如图，函数 \(y = 2\cos(\omega x + \theta)\)（\(x \in \R\)，\(0 \leqslant \theta \leqslant \frac{\pi}{2}\)）的图象与 \(y\) 轴交于点 \((0, \sqrt{3})\)，且在该点处切线的斜率为 \(-2\).

\begin{enumerate}
\item 求 \(\theta\) 和 \(\omega\) 的值；
\item 已知点 \(A\left(\frac{\pi}{2}, 0\right)\)，点 \(P\) 是该函数图象上的一点，点 \(Q(x_0, y_0)\) 是 \(PA\) 的中点，当 \(y_0 = \frac{\sqrt{3}}{2}\)，\(x_0 \in \left[\frac{\pi}{2}, \pi\right]\) 时，求 \(x_0\) 的值.
\end{enumerate}

\FigureLayoutDeclare{img/2007/jiangxi_science/q18_fig1.png}{9.0cm}{39bp 47bp 23bp 27bp}
\bitmapfigure[max width=0.55\linewidth]{img/2007/jiangxi_science/q18_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 由图象经过 \((0,\sqrt3)\)，得
\[
2\cos\theta=\sqrt3
\]
因为 \(0\leqslant\theta\leqslant\frac\pi2\)，所以 \(\theta=\frac\pi6\). 函数的导数为 \(y'=-2\omega\sin(\omega x+\theta)\)，由切线斜率为 \(-2\) 得
\[
-2\omega\sin\theta=-2
\]
代入 \(\theta=\frac\pi6\)，得 \(\omega=2\).

\item 设 \(P=(x_P,y_P)\). 因为 \(A=\left(\frac\pi2,0\right)\)，且 \(Q\) 是 \(PA\) 的中点，所以
\[
y_0=\frac{y_P}{2}=\frac{\sqrt3}{2}
\]
从而 \(y_P=\sqrt3\). 又 \(P\) 在函数图象上，故
\[
2\cos\left(2x_P+\frac\pi6\right)=\sqrt3
\]
即 \(2x_P+\frac\pi6=2k\pi\pm\frac\pi6\). 由 \(x_0\in\left[\frac\pi2,\pi\right]\) 及中点公式，得 \(x_P=2x_0-\frac\pi2\in\left[\frac\pi2,\frac{3\pi}{2}\right]\). 因此在相应范围内
\[
2x_P+\frac\pi6=\frac{11\pi}{6}\quad\text{或}\quad\frac{13\pi}{6}
\]
从而 \(x_P=\frac{5\pi}{6}\) 或 \(x_P=\pi\). 最后
\[
x_0=\frac{x_P+\frac\pi2}{2}=\frac{2\pi}{3}\quad\text{或}\quad\frac{3\pi}{4}
\]
\end{enumerate}
\end{solution}

\begin{problem}
某陶瓷厂准备烧制甲，乙，丙三件不同的工艺品，制作过程中必须先后经过两次烧制，当第一次烧制合格后可进入第二次烧制，两次烧制过程相互独立. 根据该厂现有的技术水平，经过第一次烧制后，甲，乙，丙三件产品合格的概率依次为 \(0.5\)，\(0.6\)，\(0.4\). 经过第二次烧制后，甲，乙，丙三件产品合格的概率依次为 \(0.6\)，\(0.5\)，\(0.75\).

\begin{enumerate}
\item 求第一次烧制后恰有一件产品合格的概率；
\item 经过前后两次烧制后，合格工艺品的个数为 \(\xi\)，求随机变量 \(\xi\) 的期望.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设第一次烧制后甲、乙、丙合格分别为事件 \(A\)，\(B\)，\(C\). 由相互独立性，第一次烧制后恰有一件合格的概率为
\[
P(A\overline B\,\overline C)+P(\overline A B\overline C)+P(\overline A\,\overline B C)
=0.5\times0.4\times0.6+0.5\times0.6\times0.6+0.5\times0.4\times0.4
=0.38
\]

\item 设 \(I_A\)，\(I_B\)，\(I_C\) 分别表示甲、乙、丙经过前后两次烧制后合格的示性变量，则
\(E(I_A)=0.5\times0.6=0.3\)，\(E(I_B)=0.6\times0.5=0.3\)，\(E(I_C)=0.4\times0.75=0.3\).
由于 \(\xi=I_A+I_B+I_C\)，由期望的线性性质得
\[
E(\xi)=E(I_A)+E(I_B)+E(I_C)=0.9
\]
\end{enumerate}
\end{solution}

\begin{problem}
如图是一个直三棱柱（以 \(A_{1}B_{1}C_{1}\) 为底面）被一平面所截得到的几何体，截面为 \(ABC\). 已知 \(A_{1}B_{1} = B_{1}C_{1} = 1\)，\(\angle A_{1}B_{1}C_{1} = 90^{\circ}\)，\(AA_{1} = 4\)，\(BB_{1} = 2\)，\(CC_{1} = 3\).

\begin{enumerate}
\item 设点 \(O\) 是 \(AB\) 的中点，证明：\(OC \parallel\) 平面 \(A_{1}B_{1}C_{1}\)；
\item 求二面角 \(B - AC - A_{1}\) 的大小；
\item 求此几何体的体积.
\end{enumerate}

\bitmapfigure[max width=0.34\linewidth]{img/2007/jiangxi_science/q20_fig1.png}
\end{problem}

\begin{solution}
建立直角坐标系，令
\(A_1=(0,0,0)\)，\(B_1=(1,0,0)\)，\(C_1=(1,1,0)\).
由直三棱柱的性质，
\(A=(0,0,4)\)，\(B=(1,0,2)\)，\(C=(1,1,3)\).
\begin{enumerate}
\item 点 \(O\) 是 \(AB\) 的中点，所以 \(O=\left(\frac12,0,3\right)\). 因而
\[
\overrightarrow{OC}=\left(\frac12,1,0\right)
\]
其竖直坐标为零，说明 \(OC\) 的方向平行于底面平面；又 \(O\) 的竖坐标为 \(3\)，不在底面平面 \(z=0\) 内，故 \(OC\parallel\) 平面 \(A_1B_1C_1\).

\item 由三点 \(A\)，\(B\)，\(C\) 可得截面平面方程
\[
z=4-2x+y
\]
平面 \(ABC\) 的两个方向向量可取 \(\overrightarrow{AB}=(1,0,-2)\) 和 \(\overrightarrow{AC}=(1,1,-1)\)，故其法向量为
\[
\overrightarrow{AB}\times\overrightarrow{AC}=(2,-1,1)
\]
平面 \(A_1AC\) 的法向量可取 \((1,-1,0)\). 因此两平面的夹角 \(\varphi\) 满足
\[
\cos\varphi=\frac{|(2,-1,1)\cdot(1,-1,0)|}{\sqrt6\sqrt2}=\frac{3}{\sqrt{12}}=\frac{\sqrt3}{2}
\]
所以二面角 \(B-AC-A_1\) 的大小为 \(\varphi=\frac\pi6\).

\item 底面三角形 \(A_1B_1C_1\) 在坐标平面上的区域为 \(0\leqslant y\leqslant x\leqslant1\)，截面平面给出的高度为 \(z=4-2x+y\). 故几何体体积为
\[
V=\int_0^1\int_0^x(4-2x+y)\,\mathrm{d}y\,\mathrm{d}x
=\int_0^1\left(4x-\frac32x^2\right)\,\mathrm{d}x
=\frac32
\]
\end{enumerate}
\end{solution}

\begin{problem}
设动点 \(P\) 到点 \(A(-1, 0)\) 和 \(B(1, 0)\) 的距离分别为 \(d_{1}\) 和 \(d_{2}\)，\(\angle APB = 2\theta\). 且存在常数 \(\lambda\)（\(0 < \lambda < 1\)），使得 \(d_{1}d_{2}\sin^{2}\theta = \lambda\).

\begin{enumerate}
\item 证明：动点 \(P\) 的轨迹 \(C\) 为双曲线，并求出 \(C\) 的方程；
\item 过点 \(B\) 作直线交双曲线 \(C\) 的右支于 \(M\)，\(N\) 两点，试确定 \(\lambda\) 的范围，使 \(\overrightarrow{OM}\cdot\overrightarrow{ON}=0\)，其中点 \(O\) 为坐标原点.
\end{enumerate}

\bitmapfigure[max width=0.36\linewidth]{img/2007/jiangxi_science/q21_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 在三角形 \(APB\) 中，利用余弦定理得
\[
4=d_1^2+d_2^2-2d_1d_2\cos2\theta
\]
于是
\[
d_1d_2\sin^2\theta
=\frac{d_1d_2(1-\cos2\theta)}{2}
=\frac{4-(d_1-d_2)^2}{4}
\]
由题设该式等于 \(\lambda\)，故
\((d_1-d_2)^2=4(1-\lambda)\)，\(|d_1-d_2|=2\sqrt{1-\lambda}\).
焦点 \(A\)，\(B\) 间的距离为 \(2\)，所以动点轨迹是以 \(A\)，\(B\) 为焦点、实轴长 \(2\sqrt{1-\lambda}\) 的双曲线. 令
\(a=\sqrt{1-\lambda}\)，\(c=1\)，\(b^2=c^2-a^2=\lambda\)
则其方程为
\[
\frac{x^2}{1-\lambda}-\frac{y^2}{\lambda}=1
\]

\item 先考虑非竖直直线，设其方程为 \(y=m(x-1)\)，并令 \(u=m^2\). 该直线与双曲线的交点横坐标 \(x_1\)，\(x_2\) 满足
\[
\bigl(\lambda-(1-\lambda)u\bigr)x^2+2(1-\lambda)ux-(1-\lambda)u-(1-\lambda)\lambda=0
\]
由韦达定理，记 \(D=\lambda-(1-\lambda)u\)，则
\(x_1+x_2=-\frac{2(1-\lambda)u}{D}\)，\(x_1x_2=-\frac{(1-\lambda)(u+\lambda)}{D}\).
又因为 \(M\)，\(N\) 的位置向量分别为 \((x_1,m(x_1-1))\) 和 \((x_2,m(x_2-1))\)，条件 \(\overrightarrow{OM}\cdot\overrightarrow{ON}=0\) 等价于
\[
x_1x_2+u(x_1-1)(x_2-1)=0
\]
代入韦达定理并化简，得到
\[
u\bigl(\lambda^2-(1-\lambda)\bigr)=\lambda(1-\lambda)
\]
因此
\[
u=\frac{\lambda(1-\lambda)}{\lambda^2+\lambda-1}
\]
要使 \(u\geqslant0\)，必须有 \(\lambda>\frac{\sqrt5-1}{2}\). 此时进一步有
\(x_1+x_2=-\frac{2(1-\lambda)^2}{3\lambda-2}\)，\(x_1x_2=-\frac{(1-\lambda)\lambda^2}{3\lambda-2}\).
若 \(\lambda>\frac23\)，则 \(x_1x_2<0\)，两个交点不可能都在双曲线右支上. 若 \(\frac{\sqrt5-1}{2}<\lambda<\frac23\)，记右支顶点的横坐标为 \(a=\sqrt{1-\lambda}\)，并令上式左端的二次多项式为 \(F(x)\). 记 \(\alpha=\frac{\sqrt5-1}{2}\)，则此时二次项系数 \(D<0\)，且：
\(F(a)=-(1-\lambda)u(a-1)^2<0\)，\(F(1)=\lambda^2>0\).
由于二次项系数 \(D<0\)，二次函数在两根之间为正；结合 \(F(a)<0<F(1)\) 且 \(a<1\)，可知一个根在 \((a,1)\) 内，另一个根在 \((1,+\infty)\) 内，故两交点均在右支上. 当 \(\lambda=\frac23\) 时二次项系数为零，不会得到两个交点.

还需考虑竖直直线 \(x=1\). 它与双曲线的交点为
\(\left(1,\frac{\lambda}{\sqrt{1-\lambda}}\right)\)，\(\left(1,-\frac{\lambda}{\sqrt{1-\lambda}}\right)\).
两交点的位置向量垂直当且仅当
\[
1-\frac{\lambda^2}{1-\lambda}=0
\]
即 \(\lambda^2+\lambda-1=0\)，在 \(0<\lambda<1\) 中得到 \(\lambda=\frac{\sqrt5-1}{2}\). 综上，
\[
\lambda\in\left[\frac{\sqrt5-1}{2},\frac23\right)
\]
\end{enumerate}
\end{solution}

\begin{problem}
设正整数数列 \(\{a_n\}\) 满足：\(a_2 = 4\)，且对于任何 \(n \in \N^*\)，有 \(2 + \frac{1}{a_{n+1}} < \frac{\frac{1}{a_n} + \frac{1}{a_{n+1}}}{\frac{1}{n} - \frac{1}{n+1}} < 2 + \frac{1}{a_n}\).

\begin{enumerate}
\item 求 \(a_1\)，\(a_3\)；
\item 求数列 \(\{a_n\}\) 的通项 \(a_n\).
\end{enumerate}
\end{problem}

\begin{solution}
令 \(b_n=1/a_n\). 当 \(n=1\) 时，题设不等式为
\[
2+\frac14<2\left(b_1+\frac14\right)<2+b_1
\]
左侧不等式给出 \(b_1>\frac78\). 由于 \(a_1\) 是正整数，故 \(a_1=1\).

当 \(n=2\) 时，
\[
2+b_3<6\left(\frac14+b_3\right)<2+\frac14
\]
从而
\[
\frac1{10}<b_3<\frac18
\]
所以 \(8<a_3<10\)，即 \(a_3=9\).

下面证明通项公式. 设 \(A=a_n\)，\(B=a_{n+1}\)，\(K=n(n+1)\). 因为
\[
\frac1n-\frac1{n+1}=\frac1{n(n+1)}=\frac1K
\]
原不等式等价于
\[
2+\frac1B<K\left(\frac1A+\frac1B\right)<2+\frac1A
\]
乘以正数 \(AB\)，得
\[
A(2B+1)<K(A+B)<B(2A+1)
\]
在下面的归纳步骤中 \(n\geqslant2\)，且将取 \(A=n^2\)，此时
\(2A-K=n(n-1)>0\)，所以两个分母均为正，可以整理为
\[
\frac{AK}{2A-K+1}<B<\frac{A(K-1)}{2A-K}
\]

已知 \(a_2=2^2\). 假设对某个 \(n\geqslant2\) 有 \(a_n=n^2\)，则 \(A=n^2\)，\(K=n(n+1)\)，上式化为
\[
\frac{n^3(n+1)}{n^2-n+1}<a_{n+1}<\frac{n(n^2+n-1)}{n-1}
\]
注意到
\[
\frac{n^3(n+1)}{n^2-n+1}= (n+1)^2-1+\frac{n(n-2)}{n^2-n+1}\geqslant(n+1)^2-1
\]
以及
\[
\frac{n(n^2+n-1)}{n-1}=(n+1)^2+\frac1{n-1}\leqslant(n+1)^2+1
\]
由于 \(a_{n+1}\) 是正整数，必有 \(a_{n+1}=(n+1)^2\). 由数学归纳法，
\(a_n=n^2\)，\((n\geqslant2)\).
结合 \(a_1=1=1^2\)，数列通项为
\(a_n=n^2\)，\((n\in\N^*)\).
\end{solution}
