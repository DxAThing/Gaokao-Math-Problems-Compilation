\section{单选题}

\begin{problem}
设集合 \(U = \{1, 2, 3, 4, 5\}\)，\(A = \{1, 3\}\)，\(B = \{2, 3, 4\}\)，则 \((\complement_U A) \cap (\complement_U B) =\)（\quad）

\choices
    {\(\{1\}\)}
    {\(\{5\}\)}
    {\(\{2,4\}\)}
    {\(\{1, 2, 4, 5\}\)}
\end{problem}

\begin{answer}
B.
\end{answer}

\begin{solution}
在全集 \(U\) 中，
\(\complement_U A=\{2,4,5\}\)，\(\complement_U B=\{1,5\}\).
因此
\[
(\complement_U A)\cap(\complement_U B)=\{5\}
\]
所以选 B.
\end{solution}

\begin{problem}
若函数 \( y = f(x) \) 的反函数图象过点 \((1,5)\)，则函数 \( y = f(x) \) 的图象必过点（\quad）

\choices
    {\((1,1)\)}
    {\((1,5)\)}
    {\((5,1)\)}
    {\((5,5)\)}
\end{problem}

\begin{answer}
C.
\end{answer}

\begin{solution}
反函数的图象与原函数的图象关于直线 \(y=x\) 对称. 因为反函数图象过点 \((1,5)\)，所以原函数图象必过关于 \(y=x\) 对称的点 \((5,1)\)，故选 C.
\end{solution}


\begin{problem}
若向量 \(\bs{a}\) 与 \(\bs{b}\) 不共线，\(\bs{a} \cdot \bs{b} \neq 0\)，且 \(\bs{c} = \bs{a} - \left(\frac{\bs{a} \cdot \bs{a}}{\bs{a} \cdot \bs{b}}\right)\bs{b}\)，则向量 \(\bs{a}\) 与 \(\bs{c}\) 的夹角为（\quad）

\choices
    {\(0\)}
    {\(\frac{\pi}{6}\)}
    {\(\frac{\pi}{3}\)}
    {\(\frac{\pi}{2}\)}
\end{problem}

\begin{answer}
D.
\end{answer}

\begin{solution}
由向量数量积的运算律，
\[
\bs{a}\cdot\bs{c}
=\bs{a}\cdot\bs{a}-\frac{\bs{a}\cdot\bs{a}}{\bs{a}\cdot\bs{b}}(\bs{a}\cdot\bs{b})=0
\]
又因为 \(\bs{a}\) 与 \(\bs{b}\) 不共线，所以 \(\bs{c}\ne\bs{0}\). 因而 \(\bs{a}\) 与 \(\bs{c}\) 垂直，夹角为 \(\frac{\pi}{2}\)，选 D.
\end{solution}


\begin{problem}
设等差数列 \(\{a_{n}\}\) 的前 \(n\) 项和为 \(S_{n}\)，若 \(S_{3} = 9\)，\(S_{6} = 36\)，则 \(a_{7} + a_{8} + a_{9} =\)（\quad）

\choices
    {\(63\)}
    {\(45\)}
    {\(36\)}
    {\(27\)}
\end{problem}

\begin{answer}
B.
\end{answer}

\begin{solution}
设等差数列的首项为 \(a_1\)，公差为 \(d\). 由前 \(n\) 项和公式，
\(S_3=3a_1+3d=9\)，\(S_6=6a_1+15d=36\).
由此得到
\(a_1+d=3\)，\(2a_1+5d=12\).
两式相减得 \(3d=6\)，所以 \(d=2\)，\(a_1=1\). 因而
\[
a_7+a_8+a_9=(1+6d)+(1+7d)+(1+8d)=3+21d=45
\]
故选 B.
\end{solution}


\begin{problem}
若 \(\theta \in \left(\frac{3\pi}{4}, \frac{5\pi}{4}\right)\)，则复数 \((\cos \theta + \sin \theta) + (\sin \theta - \cos \theta)\i\) 在复平面内所对应的点在（\quad）

\choices
    {第一象限}
    {第二象限}
    {第三象限}
    {第四象限}
\end{problem}

\begin{answer}
B.
\end{answer}

\begin{solution}
复数对应点的横坐标和纵坐标分别为
\[
\cos\theta+\sin\theta=\sqrt{2}\sin\left(\theta+\frac{\pi}{4}\right)
\]
\[
\sin\theta-\cos\theta=\sqrt{2}\sin\left(\theta-\frac{\pi}{4}\right)
\]
当 \(\theta\in\left(\frac{3\pi}{4},\frac{5\pi}{4}\right)\) 时，
\(\theta+\frac{\pi}{4}\in\left(\pi,\frac{3\pi}{2}\right)\)，\(\theta-\frac{\pi}{4}\in\left(\frac{\pi}{2},\pi\right)\).
所以横坐标为负、纵坐标为正，对应点在第二象限，选 B.
\end{solution}


\begin{problem}
若函数 \(y = f(x)\) 的图象按向量 \(\bs{a}\) 平移后，得到函数 \(y = f(x + 1) - 2\) 的图象，则向量 \(\bs{a} =\)（\quad）

\choices
    {\((-1, -2)\)}
    {\((1, -2)\)}
    {\((-1, 2)\)}
    {\((1, 2)\)}
\end{problem}

\begin{answer}
A.
\end{answer}

\begin{solution}
图象 \(y=f(x)\) 按向量 \((h,k)\) 平移后，方程变为 \(y=f(x-h)+k\). 题中
\[
f(x+1)-2=f\bigl(x-(-1)\bigr)+(-2)
\]
所以平移向量为 \((-1,-2)\)，选 A.
\end{solution}


\begin{problem}
若 \(m\)，\(n\) 是两条不同的直线，\(\alpha\)，\(\beta\)，\(\gamma\) 是三个不同的平面，则下列命题中的真命题是（\quad）

\choices
    {若 \(m \subset \beta\)，\(\alpha \perp \beta\)，则 \(m \perp \alpha\)}
    {若 \(\alpha \cap \gamma = m\)，\(\beta \cap \gamma = n\)，\(m \parallel n\)，则 \(\alpha \parallel \beta\)}
    {若 \(m \perp \beta\)，\(m \parallel \alpha\)，则 \(\alpha \perp \beta\)}
    {若 \(\alpha \perp \gamma\)，\(\alpha \perp \beta\)，则 \(\beta \perp \gamma\)}
\end{problem}

\begin{answer}
C.
\end{answer}

\begin{solution}
选项 C 正确：若 \(m\perp\beta\)，则 \(m\) 的方向垂直于平面 \(\beta\)；又 \(m\parallel\alpha\)，说明平面 \(\alpha\) 内有一条直线的方向垂直于平面 \(\beta\)，因此 \(\alpha\perp\beta\).

其余命题不成立. 例如取 \(\beta:z=0\)，\(\alpha:x=0\)，直线 \(m:y=x\)，\(z=0\) 满足 \(m\subset\beta\) 且 \(\alpha\perp\beta\)，但 \(m\not\perp\alpha\)，故 A 错. 取 \(\gamma:z=0\)，\(\alpha:y=0\)，\(\beta:y=z+1\)，则 \(\alpha\cap\gamma\) 与 \(\beta\cap\gamma\) 是两条平行直线，而 \(\alpha\) 与 \(\beta\) 相交，故 B 错. 取 \(\alpha:z=0\)，\(\beta:y=0\)，\(\gamma:y=x\)，则 \(\alpha\perp\beta\)，\(\alpha\perp\gamma\)，但 \(\beta\) 与 \(\gamma\) 不垂直，故 D 错.
\end{solution}


\begin{problem}
已知变量 \(x\)，\(y\) 满足约束条件 \(\begin{cases}
x - y + 2 \leqslant 0,\\
x \geqslant 1,\\
x + y - 7 \leqslant 0,
\end{cases}\) 则 \(\frac{y}{x}\) 的取值范围是（\quad）

\choices
    {\(\left[\frac{9}{5}, 6\right]\)}
    {\(\left(-\infty, \frac{9}{5}\right] \cup [6, +\infty)\)}
    {\((-\infty, 3] \cup [6, +\infty)\)}
    {\([3, 6]\)}
\end{problem}

\begin{answer}
A.
\end{answer}

\begin{solution}
由约束条件得
\(y\geqslant x+2\)，\(y\leqslant 7-x\)，\(x\geqslant1\).
要使上下界有交集，必须有 \(x+2\leqslant7-x\)，所以 \(1\leqslant x\leqslant\frac{5}{2}\). 在这个三角形区域内 \(x>0\)，令 \(r=\frac{y}{x}\)，则
\[
1+\frac{2}{x}\leqslant r\leqslant\frac{7}{x}-1
\]
函数 \(1+\frac{2}{x}\) 和 \(\frac{7}{x}-1\) 在 \(\left[1,\frac{5}{2}\right]\) 上都递减，因此
\(r_{\min}=1+\frac{2}{5/2}=\frac{9}{5}\)，\(r_{\max}=\frac{7}{1}-1=6\).
该可行域连通且 \(x>0\)，连续函数 \(\frac{y}{x}\) 的取值包含两端点之间的全部数值，故取值范围为 \(\left[\frac{9}{5},6\right]\)，选 A.
\end{solution}


\begin{problem}
一个坛子里有编号为 \(1\)，\(2\)，\(\ldots\)，\(12\) 的 \(12\) 个大小相同的球，其中 \(1\) 到 \(6\) 号球是红球，其余的是黑球.若从中任取 \(2\) 个球，则取到的都是红球，且至少有 \(1\) 个球的号码是偶数的概率为（\quad）

\choices
    {\(\frac{1}{22}\)}
    {\(\frac{1}{11}\)}
    {\(\frac{3}{22}\)}
    {\(\frac{2}{11}\)}
\end{problem}

\begin{answer}
D.
\end{answer}

\begin{solution}
任取 \(2\) 个球的基本事件总数为
\[
\binom{12}{2}=66
\]
两球都是红球的取法有 \(\binom{6}{2}\) 种，其中两个号码都是奇数的取法有 \(\binom{3}{2}\) 种，因为红球中的奇数号码为 \(1\)，\(3\)，\(5\). 所以满足“都是红球且至少有一个偶数号码”的取法数为
\[
\binom{6}{2}-\binom{3}{2}=15-3=12
\]
所求概率为
\[
\frac{12}{66}=\frac{2}{11}
\]
故选 D.
\end{solution}


\begin{problem}
设 \(p\)，\(q\) 是两个命题，\(p: \log_{\frac{1}{2}}(|x| - 3) > 0\)，\(q: x^2 - \frac{5}{6}x + \frac{1}{6} > 0\)，则 \(p\) 是 \(q\) 的（\quad）

\choices
    {充分而不必要条件}
    {必要而不充分条件}
    {充分必要条件}
    {既不充分也不必要条件}
\end{problem}

\begin{answer}
A.
\end{answer}

\begin{solution}
因为底数 \(\frac{1}{2}\) 在 \(0\) 与 \(1\) 之间，故
\[
p\Longleftrightarrow 0<|x|-3<1
\Longleftrightarrow 3<|x|<4
\]
于是 \(p\) 的解集为 \((-4,-3)\cup(3,4)\). 另一方面，
\[
x^2-\frac{5}{6}x+\frac{1}{6}>0
\Longleftrightarrow (3x-1)(2x-1)>0
\Longleftrightarrow x<\frac{1}{3}\text{ 或 }x>\frac{1}{2}
\]
显然 \(p\) 的解集包含于 \(q\) 的解集，所以 \(p\) 是 \(q\) 的充分条件. 例如 \(x=0\) 满足 \(q\) 但不满足 \(p\)，故不是必要条件，选 A.
\end{solution}


\begin{problem}
设 \(P\) 为双曲线 \(x^2 - \frac{y^2}{12} = 1\) 上的一点，\(F_1\)，\(F_2\) 是该双曲线的两个焦点.若 \(|PF_1| : |PF_2| = 3 : 2\)，则 \(\triangle PF_1F_2\) 的面积为（\quad）

\choices
    {\(6\sqrt{3}\)}
    {\(12\)}
    {\(12\sqrt{3}\)}
    {\(24\)}
\end{problem}

\begin{answer}
B.
\end{answer}

\begin{solution}
双曲线可写为
\[
\frac{x^2}{1^2}-\frac{y^2}{(2\sqrt{3})^2}=1
\]
所以 \(a=1\)，\(c=\sqrt{a^2+b^2}=\sqrt{13}\)，两焦点间距离为 \(2\sqrt{13}\). 双曲线上一点到两焦点距离之差的绝对值为 \(2a=2\). 由距离比 \(3:2\)，可设两距离为 \(3k\)，\(2k\)，于是 \(k=2\)，两距离为 \(6,4\).

不妨设 \(F_1=(-\sqrt{13},0)\)，\(F_2=(\sqrt{13},0)\)，并令 \(PF_1=6\)，\(PF_2=4\). 由两距离的平方相减，
\[
PF_1^2-PF_2^2=4\sqrt{13}x=36-16=20
\]
得 \(x=\frac{5}{\sqrt{13}}\). 代入双曲线方程，
\[
y^2=12\left(x^2-1\right)=12\left(\frac{25}{13}-1\right)=\frac{144}{13}
\]
所以 \(|y|=\frac{12}{\sqrt{13}}\). 因此
\[
S_{\triangle PF_1F_2}=\frac{1}{2}\cdot2\sqrt{13}\cdot\frac{12}{\sqrt{13}}=12
\]
若两焦点的标号相反，只改变 \(x\) 的符号，不影响面积，故选 B.
\end{solution}


\begin{problem}
已知 \(f(x)\) 与 \(g(x)\) 是定义在 \(\R\) 上的连续函数，如果 \(f(x)\) 与 \(g(x)\) 仅当 \(x = 0\) 时的函数值为 \(0\)，且 \(f(x) \geqslant g(x)\)，那么下列情形不可能出现的是（\quad）

\choices
    {\(0\) 是 \(f(x)\) 的极大值，也是 \(g(x)\) 的极大值}
    {\(0\) 是 \(f(x)\) 的极小值，也是 \(g(x)\) 的极小值}
    {\(0\) 是 \(f(x)\) 的极大值，但不是 \(g(x)\) 的极值}
    {\(0\) 是 \(f(x)\) 的极小值，但不是 \(g(x)\) 的极值}
\end{problem}

\begin{answer}
C.
\end{answer}

\begin{solution}
由题意 \(f(0)=g(0)=0\)，且当 \(x\ne0\) 时，\(f(x)\ne0\)，\(g(x)\ne0\). 假设选项 C 的情形成立，则 \(0\) 是 \(f(x)\) 的极大值. 因而在 \(0\) 的邻域内，\(f(x)\leqslant f(0)=0\). 又 \(f(x)\geqslant g(x)\)，所以
\[
g(x)\leqslant f(x)\leqslant0
\]
当 \(x\ne0\) 时 \(g(x)\ne0\)，故 \(g(x)<0=g(0)\). 这说明 \(0\) 也是 \(g(x)\) 的极大值，与选项中“不是 \(g(x)\) 的极值”矛盾.

因此不可能出现的是 C.
\end{solution}


\section{填空题}

\begin{problem}
已知函数 \(f(x) = \begin{cases}
a\cos x, & x \geqslant 0,\\
x^{2} - 1, & x < 0,
\end{cases}\) 在点 \(x = 0\) 处连续，则 \(a =\fillinblank{}\).
\end{problem}

\begin{answer}
\(-1\).
\end{answer}

\begin{solution}
当 \(x\to0^-\) 时，\(f(x)=x^2-1\to-1\)，而 \(f(0)=a\cos0=a\). 连续性要求左极限等于函数在 \(0\) 处的值，因此
\[
a=-1
\]
\end{solution}


\begin{problem}
设椭圆 \(\frac{x^{2}}{25} + \frac{y^{2}}{16} = 1\) 上一点 \(P\) 到左准线的距离为 \(10\)，\(F\) 是该椭圆的左焦点.若点 \(M\) 满足 \(\overrightarrow{OM} = \frac{1}{2}\left(\overrightarrow{OP} + \overrightarrow{OF}\right)\)，则 \(\left|\overrightarrow{OM}\right| =\fillinblank{}\).
\end{problem}

\begin{answer}
2.
\end{answer}

\begin{solution}
椭圆的长半轴、短半轴分别为 \(a=5\)、\(b=4\)，所以焦半距
\[
c=\sqrt{a^2-b^2}=3
\]
离心率 \(e=\frac{3}{5}\)，左准线为 \(x=-\frac{a}{e}=-\frac{25}{3}\). 设 \(P=(x,y)\)，由于椭圆上点的横坐标不小于 \(-5\)，点 \(P\) 到左准线的距离为 \(x+\frac{25}{3}\). 由题设
\(x+\frac{25}{3}=10\)，\(x=\frac{5}{3}\).
又 \(P\) 在椭圆上，所以
\(\frac{y^2}{16}=1-\frac{x^2}{25}=1-\frac{1}{9}\)，\(y^2=\frac{128}{9}\).
左焦点为 \(F=(-3,0)\)，故
\[
\left|\overrightarrow{OP}+\overrightarrow{OF}\right|^2=(x-3)^2+y^2
=\left(-\frac{4}{3}\right)^2+\frac{128}{9}=16
\]
由 \(\overrightarrow{OM}=\frac{1}{2}(\overrightarrow{OP}+\overrightarrow{OF})\)，得
\[
\left|\overrightarrow{OM}\right|=\frac{1}{2}\cdot4=2
\]
\end{solution}


\begin{problem}
若一个底面边长为 \(\frac{\sqrt{6}}{2}\)，侧棱长为 \(\sqrt{6}\) 的正六棱柱的所有顶点都在一个球的面上，则此球的体积为\fillinblank{}.
\end{problem}

\begin{answer}
\(4\sqrt{3}\pi\).
\end{answer}

\begin{solution}
正六边形的外接圆半径等于其边长，因此底面中心到任一顶点的距离为
\[
r=\frac{\sqrt{6}}{2}
\]
正六棱柱的高为侧棱长 \(h=\sqrt{6}\). 球心位于两底面中心连线的中点，球半径 \(R\) 满足
\(R^2=r^2+\left(\frac{h}{2}\right)^2 =\frac{6}{4}+\frac{6}{4}=3\)，\(R=\sqrt{3}\).
所以球的体积为
\[
V=\frac{4}{3}\pi R^3=\frac{4}{3}\pi(\sqrt{3})^3=4\sqrt{3}\pi
\]
\end{solution}


\begin{problem}
将数字 \(1\)，\(2\)，\(3\)，\(4\)，\(5\)，\(6\) 拼成一列，记第 \(i\) 个数为 \(a_i\) \((i = 1, 2, \ldots, 6)\). 若 \(a_1 \neq 1\)，\(a_3 \neq 3\)，\(a_5 \neq 5\)，\(a_1 < a_3 < a_5\)，则不同的排列方法有\fillinblank{}种.（用数字作答）
\end{problem}

\begin{answer}
30.
\end{answer}

\begin{solution}
由于 \(a_1<a_3<a_5\)，先选择并按递增顺序确定奇数位置上的三个数. 从 \(1\)，\(2\)，\(3\)，\(4\)，\(5\)，\(6\) 中选三个数共有 \(\binom{6}{3}=20\) 种.

设 \(E_1\)、\(E_2\)、\(E_3\) 分别表示 \(a_1=1\)、\(a_3=3\)、\(a_5=5\). 则
\(|E_1|=\binom{5}{2}=10\)，\(|E_2|=2\cdot3=6\)，\(|E_3|=\binom{4}{2}=6\)
\(|E_1\cap E_2|=3\)，\(|E_1\cap E_3|=3\)，\(|E_2\cap E_3|=2\)，\(|E_1\cap E_2\cap E_3|=1\).
由容斥原理，符合三个不等式条件的三元组数为
\[
20-(10+6+6)+(3+3+2)-1=5
\]
奇数位置确定后，剩下的三个数在 \(a_2\)，\(a_4\)，\(a_6\) 中任意排列，有 \(3!=6\) 种. 因而排列总数为
\[
5\times3!=30
\]
\end{solution}


\section{解答题}

\begin{problem}
已知函数 \(f(x) = \sin \left(\omega x + \frac{\pi}{6}\right) + \sin \left(\omega x - \frac{\pi}{6}\right) - 2 \cos^2 \frac{\omega x}{2}\)，\(x \in \R\)（其中 \(\omega > 0\)）.

\begin{enumerate}
\item 求函数 \(f(x)\) 的值域.
\item 若对任意的 \(a \in \R\)，函数 \(y = f(x)\)，\(x \in (a, a + \pi]\) 的图象与直线 \(y = -1\) 有且仅有两个不同的交点，试确定 \(\omega\) 的值（不必证明），并求函数 \(y = f(x)\)，\(x \in \R\) 的单调增区间.
\end{enumerate}
\end{problem}

\begin{solution}
由和差化积公式及二倍角公式，
\[
f(x)=\sqrt{3}\sin(\omega x)-1-\cos(\omega x)=2\sin\left(\omega x-\frac{\pi}{6}\right)-1
\]

\begin{enumerate}
\item 因为 \(-1\leqslant\sin\left(\omega x-\frac{\pi}{6}\right)\leqslant1\)，所以函数 \(f(x)\) 的值域为 \([-3,1]\).
\item 由 \(f(x)=-1\)，得
\(\sin\left(\omega x-\frac{\pi}{6}\right)=0\)，\(\omega x-\frac{\pi}{6}=k\pi\)，\((k\in\Z)\).
因此相邻两个交点的横坐标之差为 \(\dfrac{\pi}{\omega}\). 任意长度为 \(\pi\) 的半开区间 \((a,a+\pi]\) 内都有且仅有两个交点，所以必须有 \(\pi=2\cdot\dfrac{\pi}{\omega}\)，从而 \(\omega=2\).

此时 \(f(x)=2\sin\left(2x-\dfrac{\pi}{6}\right)-1\)，且
\[
f'(x)=4\cos\left(2x-\frac{\pi}{6}\right)
\]
当 \(-\dfrac{\pi}{2}+2k\pi<2x-\dfrac{\pi}{6}<\dfrac{\pi}{2}+2k\pi\) 时，\(f'(x)>0\)，即
\(k\pi-\frac{\pi}{6}<x<k\pi+\frac{\pi}{3}\)，\((k\in\Z)\).
所以函数在每个区间 \(\left[k\pi-\dfrac{\pi}{6},k\pi+\dfrac{\pi}{3}\right]\) 上单调递增.
\end{enumerate}
\end{solution}

\begin{problem}
如图，在直三棱柱 \(ABC - A_{1}B_{1}C_{1}\) 中，\(\angle ACB = 90^{\circ}\)，\(AC = BC = a\)，\(D\)，\(E\) 分别为棱 \(AB\)，\(BC\) 的中点，\(M\) 为棱 \(AA_{1}\) 上的点，二面角 \(M - DE - A\) 为 \(30^{\circ}\).

\begin{enumerate}
\item 证明： \(A_{1}B_{1} \perp C_{1}D\).
\item 求 \(MA\) 的长，并求点 \(C\) 到平面 \(MDE\) 的距离.
\end{enumerate}

\end{problem}

\begin{solution}
建立空间直角坐标系，取
\(C(0,0,0)\)，\(A(a,0,0)\)，\(B(0,a,0)\)，\(A_{1}(a,0,h)\)，\(B_{1}(0,a,h)\)，\(C_{1}(0,0,h)\)
其中 \(h=AA_{1}>0\). 于是
\(D\left(\frac{a}{2},\frac{a}{2},0\right)\)，\(E\left(0,\frac{a}{2},0\right)\)，\(M(a,0,m)\)
其中 \(m=MA\).

平面 \(MDE\) 的一个法向量为
\[
\overrightarrow{MD}\times\overrightarrow{ME}=\left(0,\frac{am}{2},\frac{a^{2}}{4}\right)
\]
所以可取法向量 \(\bs{n}=(0,2m,a)\). 平面 \(ADE\) 是底面 \(z=0\)，其法向量为 \((0,0,1)\). 由二面角 \(M-DE-A\) 为 \(30^{\circ}\)，得
\[
\cos30^{\circ}=\frac{a}{\sqrt{4m^{2}+a^{2}}}
\]
两边平方并整理，得
\(\frac{3}{4}=\frac{a^{2}}{4m^{2}+a^{2}}\)，\(m^{2}=\frac{a^{2}}{12}\).

\begin{enumerate}
\item \(\overrightarrow{A_{1}B_{1}}=(-a,a,0)\)，\(\overrightarrow{C_{1}D}=\left(\dfrac{a}{2},\dfrac{a}{2},-h\right)\)，从而
\[
\overrightarrow{A_{1}B_{1}}\cdot\overrightarrow{C_{1}D}=-a\cdot\frac{a}{2}+a\cdot\frac{a}{2}=0
\]
因此 \(A_{1}B_{1}\perp C_{1}D\).
\item 因为 \(m>0\)，所以
\[
MA=m=\frac{\sqrt{3}}{6}a
\]

由法向量 \((0,2m,a)\) 及点 \(M(a,0,m)\)，平面 \(MDE\) 的方程为
\[
2my+az-am=0
\]
所以点 \(C\) 到平面 \(MDE\) 的距离为
\[
\frac{am}{\sqrt{4m^{2}+a^{2}}}=\frac{a}{4}
\]
\end{enumerate}
\end{solution}

\begin{problem}
某企业准备投产一批特殊型号的产品，已知该种产品的成本 \(C\) 与产量 \(q\) 的函数关系式为 \(C = \frac{q^{3}}{3} - 3q^{2} + 20q + 10\)（\(q > 0\)）. 该种产品的市场前景无法确定，有三种可能出现的情形，各种情形发生的概率及产品价格 \(p\) 与产量 \(q\) 的函数关系如下表所示：

\begin{center}\small
\begin{tabular}{|c|c|c|}
\hline
市场情形 & 概率 & 价格 \(p\) 与产量 \(q\) 的函数关系式 \\
\hline
好 & \(0.4\) & \(p = 164 - 3q\) \\
\hline
中 & \(0.4\) & \(p = 101 - 3q\) \\
\hline
差 & \(0.2\) & \(p = 70 - 3q\) \\
\hline
\end{tabular}
\end{center}

设 \(L_1\)，\(L_2\)，\(L_3\) 分别表示市场情形好，中，差时的利润，随机变量 \(\xi_q\) 表示当产量 \(q\) 而市场前景无法确定时的利润.

\begin{enumerate}
\item 分别求利润 \(L_1\)，\(L_2\)，\(L_3\) 与产量 \(q\) 的函数关系式.
\item 当产量 \(q\) 确定时，求期望 \(E(\xi_q)\).
\item 试问产量 \(q\) 取何值时，\(E(\xi_q)\) 取得最大值.
\end{enumerate}
\end{problem}

\begin{solution}
利润等于收入减去成本，而收入等于价格与产量的乘积.

\begin{enumerate}
\item 在三种市场情形下分别有
\[
L_{1}=(164-3q)q-C=-\frac{q^{3}}{3}+144q-10
\]
\[
L_{2}=(101-3q)q-C=-\frac{q^{3}}{3}+81q-10
\]
\[
L_{3}=(70-3q)q-C=-\frac{q^{3}}{3}+50q-10
\]
\item 由全概率公式，
\[
E(\xi_q)=0.4L_{1}+0.4L_{2}+0.2L_{3}=-\frac{q^{3}}{3}+100q-10
\]
\item 当 \(q>0\) 时，
\[
\frac{\mathrm{d}}{\mathrm{d}q}E(\xi_q)=-q^{2}+100
\]
因此当 \(0<q<10\) 时，\(E(\xi_q)\) 单调递增；当 \(q>10\) 时，\(E(\xi_q)\) 单调递减. 所以当 \(q=10\) 时，\(E(\xi_q)\) 取得最大值，最大值为
\[
E(\xi_{10})=-\frac{10^{3}}{3}+100\times10-10=\frac{1970}{3}
\]
\end{enumerate}
\end{solution}

\begin{problem}
已知正三角形 \(OAB\) 的三个顶点都在抛物线 \(y^{2} = 2x\) 上，其中 \(O\) 为坐标原点，设圆 \(C\) 是 \(\triangle OAB\) 的外接圆（点 \(C\) 为圆心）.

\begin{enumerate}
\item 求圆 \(C\) 的方程.
\item 设圆 \(M\) 的方程为 \((x - 4 - 7\cos \theta)^2 + (y - 7\sin \theta)^2 = 1\)，过圆 \(M\) 上任意一点 \(P\) 分别作圆 \(C\) 的两条切线 \(PE\)，\(PF\)，切点为 \(E\)，\(F\)，求 \(\overrightarrow{CE} \cdot \overrightarrow{CF}\) 的最大值和最小值.
\end{enumerate}
\end{problem}

\begin{solution}
抛物线 \(y^{2}=2x\) 上的点可表示为 \(\left(\dfrac{s^{2}}{2},s\right)\). 设
\(A\left(\frac{u^{2}}{2},u\right)\)，\(B\left(\frac{v^{2}}{2},v\right)\).

\begin{enumerate}
\item 因为 \(\triangle OAB\) 为正三角形，所以 \(OA=OB\). 于是
\[
\frac{u^{4}}{4}+u^{2}=\frac{v^{4}}{4}+v^{2}
\]
令 \(r=u^{2}\)，则函数 \(\dfrac{r^{2}}{4}+r\) 在 \(r\geqslant0\) 上严格递增，故 \(u^{2}=v^{2}\). 又因 \(A\ne B\)，所以 \(v=-u\).

此时 \(A\)，\(B\) 关于 \(x\) 轴对称，且 \(AB=2|u|\). 由 \(OA=AB\)，得
\(\frac{u^{4}}{4}+u^{2}=4u^{2}\)，\(u^{2}=12\).
故可取 \(A=(6,2\sqrt{3})\)，\(B=(6,-2\sqrt{3})\). 设外接圆圆心为 \(C=(c,0)\)，由 \(CO=CA\) 得
\(c^{2}=(c-6)^{2}+12\)，\(c=4\).
所以圆 \(C\) 的方程为
\[
(x-4)^{2}+y^{2}=16
\]
\item 圆 \(M\) 的圆心为 \(M_{0}=(4+7\cos\theta,7\sin\theta)\)，半径为 \(1\). 设圆 \(M\) 上的点 \(P\) 对应的参数为 \(\varphi\)，则
\[
P=(4+7\cos\theta+\cos\varphi,7\sin\theta+\sin\varphi)
\]
从而
\[
CP^{2}=(7\cos\theta+\cos\varphi)^{2}+(7\sin\theta+\sin\varphi)^{2}=50+14\cos(\theta-\varphi)
\]
所以 \(36\leqslant CP^{2}\leqslant64\).

设 \(\alpha=\angle ECP=\angle PCF\). 在直角三角形 \(CEP\) 中，\(CE=4\)，故
\[
\cos\alpha=\frac{4}{CP}
\]
因此 \(\angle ECF=2\alpha\)，于是
\[
\overrightarrow{CE}\cdot\overrightarrow{CF}=16\cos2\alpha=16\left(2\cos^{2}\alpha-1\right)=\frac{512}{CP^{2}}-16
\]
右端随 \(CP^{2}\) 增大而减小，所以
\(\max\left(\overrightarrow{CE}\cdot\overrightarrow{CF}\right)=\frac{512}{36}-16=-\frac{16}{9}\)，\(\min\left(\overrightarrow{CE}\cdot\overrightarrow{CF}\right)=\frac{512}{64}-16=-8\).
\end{enumerate}
\end{solution}

\begin{problem}
已知数列 \(\{a_{n}\}\)，\(\{b_{n}\}\) 与函数 \(f(x)\)，\(g(x)\)，\(x \in \R\) 满足条件：\(b_{1} = b\)，\(a_{n} = f(b_{n}) = g(b_{n+1})\)（\(n \in \N^{*}\)）.

\begin{enumerate}
\item 若 \(f(x) = tx + 1\)（\(t \neq 0\)，\(t \neq 2\)），\(g(x) = 2x\)，\(f(b) \neq g(b)\)，且 \(\displaystyle\lim_{n \to \infty} a_n\) 存在，求 \(t\) 的取值范围，并求 \(\displaystyle\lim_{n \to \infty} a_n\)（用 \(t\) 表示）.
\item 若函数 \(y = f(x)\) 为 \(\R\) 上的增函数，\(g(x) = f^{-1}(x)\)，\(b = 1\)，\(f(1) < 1\)，证明对任意的 \(n \in \N^{*}\)，\(a_{n+1} < a_n\).
\end{enumerate}
\end{problem}

\begin{solution}
由 \(a_n=f(b_n)=g(b_{n+1})\) 及 \(g(x)=2x\)，有
\(tb_n+1=2b_{n+1}\)，\(b_{n+1}=\frac{t}{2}b_n+\frac{1}{2}\).

\begin{enumerate}
\item 由于 \(t\ne2\)，上述递推式的唯一不动点为
\[
b_{*}=\frac{1}{2-t}
\]
又 \(f(b)\ne g(b)\) 等价于 \(tb+1\ne2b\)，即 \(b\ne b_{*}\). 因而
\[
b_n-b_{*}=\left(\frac{t}{2}\right)^{n-1}(b-b_{*})
\]
其中 \(b-b_{*}\ne0\).

因为 \(a_n=2b_{n+1}\)，所以 \(\displaystyle\lim_{n\to\infty}a_n\) 存在时，数列 \(\{b_n\}\) 也收敛. 由上式及 \(b-b_{*}\ne0\)，这要求 \(\left|\dfrac{t}{2}\right|<1\). 结合 \(t\ne0\)，得到
\[
t\in(-2,0)\cup(0,2)
\]
此时 \(b_n\to b_{*}\)，故
\[
\displaystyle\lim_{n\to\infty}a_n=2b_{*}=\frac{2}{2-t}
\]
\item 由 \(g=f^{-1}\)，对关系 \(a_n=g(b_{n+1})\) 两边施加函数 \(f\)，得
\[
b_{n+1}=f(a_n)=f(f(b_n))
\]
记 \(F(x)=f(f(x))\)，则 \(F\) 在 \(\R\) 上为增函数. 因为 \(b_1=1\)，且 \(a_1=f(1)<1=b_1\)，所以
\[
b_2=f(a_1)<f(b_1)=a_1<b_1
\]
若 \(b_n<b_{n-1}\)，则由 \(F\) 的增性，
\[
b_{n+1}=F(b_n)<F(b_{n-1})=b_n
\]
归纳可得 \(b_{n+1}<b_n\) 对任意 \(n\in\N^{*}\) 成立. 再由 \(a_n=f(b_n)\) 及 \(f\) 的增性，得到
\[
a_{n+1}=f(b_{n+1})<f(b_n)=a_n
\]
\end{enumerate}
\end{solution}

\begin{problem}
已知函数 \(f(x) = \e^{2x} - 2t(\e^{x} + x) + x^{2} + 2t^{2} + 1\)，\(g(x) = \frac{1}{2} f'(x)\).

\begin{enumerate}
\item 证明：当 \(t < 2\sqrt{2}\) 时，\(g(x)\) 在 \(\R\) 上是增函数.
\item 对于给定的闭区间 \([a, b]\)，试说明存在实数 \(k\)，当 \(t > k\) 时，\(g(x)\) 在闭区间 \([a, b]\) 上是减函数.
\item 证明：\(f(x) \geqslant \frac{3}{2}\).
\end{enumerate}
\end{problem}

\begin{solution}
由定义直接求导，得
\(g(x)=\e^{2x}-t\e^{x}+x-t\)，\(g'(x)=2\e^{2x}-t\e^{x}+1\).

\begin{enumerate}
\item 令 \(u=\e^{x}>0\)，则 \(g'(x)=2u^{2}-tu+1\).

当 \(t\leqslant0\) 时，\(2u^{2}-tu+1>0\). 当 \(0<t<2\sqrt{2}\) 时，
\[
2u^{2}-tu+1=2\left(u-\frac{t}{4}\right)^{2}+1-\frac{t^{2}}{8}>0
\]
所以对任意 \(x\in\R\)，都有 \(g'(x)>0\)，即 \(g(x)\) 在 \(\R\) 上是增函数.
\item 对任意 \(x\in[a,b]\)，有
\[
g'(x)<0\quad\Longleftrightarrow\quad t>2\e^{x}+\e^{-x}
\]
定义 \(k\) 为
\[
k=\max_{x\in[a,b]}\left(2\e^{x}+\e^{-x}\right)
\]
这是一个实数. 因为函数 \(2\e^{x}+\e^{-x}\) 的二阶导数为 \(2\e^{x}+\e^{-x}>0\)，其在闭区间上的最大值在端点取得，所以也可取
\[
k=\max\left\{2\e^{a}+\e^{-a},2\e^{b}+\e^{-b}\right\}
\]
当 \(t>k\) 时，\(g'(x)<0\) 对一切 \(x\in[a,b]\) 成立，故 \(g(x)\) 在 \([a,b]\) 上是减函数.
\item 配方得
\[
f(x)=(\e^{x}-t)^{2}+(x-t)^{2}+1
\]
\[
f(x)=\frac{1}{2}(\e^{x}-x)^{2}+2\left(t-\frac{\e^{x}+x}{2}\right)^{2}+1
\]
又由 \(\e^{x}\geqslant x+1\)，得 \((\e^{x}-x)^{2}\geqslant1\). 因此
\[
f(x)\geqslant\frac{1}{2}+1=\frac{3}{2}
\]
当且仅当 \(x=0\)，\(t=\dfrac{1}{2}\) 时取等号.
\end{enumerate}
\end{solution}
