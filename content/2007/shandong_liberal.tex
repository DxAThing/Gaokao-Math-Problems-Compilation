\begin{problem}
在 \(\triangle ABC\) 中，角 \(A, B, C\) 的对边分别为 \(a, b, c\)，\(\tan C = 3\sqrt{7}\).

\begin{enumerate}
\item 求 \( \cos C \).
\item 若 \( \overrightarrow{CB} \cdot \overrightarrow{CA} = \frac{5}{2} \)，且 \( a + b = 9 \)，求 \(c\).
\end{enumerate}
\end{problem}

\begin{problem}
设 \(\{a_{n}\}\) 是公比大于1的等比数列，\(S_{n}\) 为数列 \(\{a_{n}\}\) 的前 \(n\) 项和. 已知 \(S_{3} = 7\)，且 \(a_{1} + 3, 3a_{2}, a_{3} + 4\) 构成等差数列.

\begin{enumerate}
\item 求数列 \( \{a_{n}\} \) 的通项；
\item 令 \( b_{n} = \ln a_{3n + 1}, n = 1, 2, \cdots \)，求数列 \( \{b_{n}\} \) 的前 \( n \) 项和 \( T_{n} \).
\end{enumerate}
\end{problem}

\begin{problem}
