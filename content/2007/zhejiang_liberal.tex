\chapter{2007年浙江卷（文）}

\section{填空题}

\begin{problem}
函数 \( y = \frac{x^2}{x^2 + 1} (x \in \R) \) 的值域是 \fillinblank{} \fillinblank{}.
\end{problem}

\begin{problem}
若 \(\sin \theta + \cos \theta = \frac{1}{5}\)，则 \(\sin 2\theta\) 的值是 \fillinblank{} \fillinblank{}.
\end{problem}

\begin{problem}
某校有学生 2000 人，其中高三学生 500 人. 为了解学生的身体素质情况，采用按年级分层抽样的方法，从该校学生中抽取一个 200 人的样本. 则样本中高三学生的人数为 \fillinblank{} \fillinblank{}.
\end{problem}

\begin{problem}
\(z = 2x + y\) 中的 \(x, y\) 满足约束条件 \(\left\{ \begin{array}{l} x - 2y + 5 \geqslant 0, \\ 3 - x \geqslant 0, \\ x + y \geqslant 0, \end{array} \right.\) 则 \(z\) 的最小值是 \fillinblank{} \fillinblank{}.
\end{problem}

\begin{problem}
曲线 \( y = x^3 - 2x^2 - 4x + 2 \) 在点 (1，\(-3\)) 处的切线方程是 \fillinblank{} \fillinblank{}.
\end{problem}

\begin{problem}
某书店有 11 种杂志，2 元 1 本的 8 种，1 元 1 本的 3 种. 小张用 10 元钱买杂志 (每种至多买一本，10 元钱刚好用完)，则不同买法的种数是 \fillinblank{}. (用数字作答)
\end{problem}

\begin{problem}
已知点 O 在二面角 \( \alpha - AB - \beta \) 的棱上，点 P 在 \( \alpha \) 内，且 \( \angle POB = 45^{\circ} \) . 若对于 \( \beta \) 内异于 O 的任意一点 Q，都有 \( \angle POQ \geqslant 45^{\circ} \) ，则二面角 \( \alpha - AB - \beta \) 的大小是 \fillinblank{} \fillinblank{}.
\end{problem}

\section{解答题}

\begin{problem}
已知 \(\triangle ABC\) 的周长为 \(\sqrt{2} + 1\)，且 \(\sin A + \sin B = \sqrt{2} \sin C\). (1) 求边 AB 的长; (2) 若 \(\triangle ABC\) 的面积为 \(\frac{1}{6} \sin C\)，求角 \(C\) 的度数.
\end{problem}

\begin{problem}
已知数列 \(\{n_n\}\) 中的相邻两项 \(a_{2k-1}, a_{2k}\) 是关于 \(x\) 的方程 \(x^2 - (3k + 2^k)x + 3k \cdot 2^k = 0\) 的两个根，且 \(a_{2k-1} \leqslant a_{2k}\) (\(k = 1, 2, 3, \cdots\)). (1) 求 \(a_{1}, a_{3}, a_{5}, a_{7}\) 及 \(a_{2k} (n \geqslant 4)\) (不必证明); (2) 求数列 \( \{a_{n}\} \) 的前 2n 项和 \( S_{2n} \) .
\end{problem}

\begin{problem}
在如图所示的几何体中，\(EA \perp\) 平面 \(ABC\)，\(DB \perp\) 平面 \(ABC\)，\(AC \perp BC\)，且 \(AC = BC = BD = 2AE\)，\(M\) 是 \(AB\) 的中点. (1) 求证: \( CM \perp EM; \) (2) 求 DE 与平面 EMC 所成角的正切值.
\end{problem}

\begin{problem}
如图，直线 \( y = kx + b \) 与椭圆 \( \frac{x^2}{4} + y^2 = 1 \) 交于 \( A, B \) 两点，记 \( \triangle AOB \) 的面积为 \( S \). (1) 求在 \(k = 0, 0 < b < 1\) 的条件下，\(S\) 的最大值； (2) 当 \( \left|AB\right|=2, S=1 \) 时，求直线 AB 的方程.
\end{problem}

\begin{problem}
已知 \( f(x) = |x^2 - 1| + x^2 + kx \). (1) 若 \(k = 2\)，求方程 \(f(x) = 0\) 的解; (2) 若关于 \(x\) 的方程 \(f(x) = 0\) 在 \((0,2)\) 上有两个解 \(x_{1}, x_{2}\)，求 \(k\) 的取值范围. 并证明 \(\frac{1}{x_1} + \frac{1}{x_2} < 4\).
\end{problem}
