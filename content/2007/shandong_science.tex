\begin{problem}
设数列 \(\{a_{n}\}\) 满足 \(a_1 + 3a_2 + 3^2 a_3 + \dots + 3^{n-1} a_n = \frac{n}{3}\)，\(n \in \N^*\).

\begin{enumerate}
\item 求数列 \(\{a_{n}\}\) 的通项；
\item 设 \( b_{n}=\frac{n}{a_{n}} \)，求数列 \(\{b_{n}\}\) 的前 \(n\) 项和 \( S_{n} \).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 记左端为 \(S_n\). 当 \(n=1\) 时，\(a_1=\frac{1}{3}\). 当 \(n\geqslant2\) 时，将条件分别写成 \(S_n=\frac{n}{3}\) 和 \(S_{n-1}=\frac{n-1}{3}\)，两式相减得
\[
3^{n-1}a_n=S_n-S_{n-1}=\frac{n}{3}-\frac{n-1}{3}=\frac{1}{3}
\]
所以 \(a_n=\frac{1}{3^n}\). 结合 \(n=1\) 时的结果，数列的通项为 \(a_n=\frac{1}{3^n}\).
\item 由 \(a_n=\frac{1}{3^n}\)，得 \(b_n=\frac{n}{a_n}=n3^n\). 记数列 \(\{b_n\}\) 的前 \(n\) 项和为 \(T_n\)，则
\[
T_n=3+2\cdot3^2+\dots+n3^n.
\]
两边同乘 \(3\) 后相减，得
\[
2T_n=-3-(3^2+3^3+\dots+3^n)+n3^{n+1}=-3-\frac{3^{n+1}-9}{2}+n3^{n+1}=\frac{3+(2n-1)3^{n+1}}{2}.
\]
因此 \(S_n=T_n=\frac{3+(2n-1)3^{n+1}}{4}\).
\end{enumerate}
\end{solution}

\begin{problem}
设 \(b\) 和 \(c\) 分别是先后抛掷一枚骰子得到的点数. 用随机变量 \(\xi\) 表示方程 \(x^{2} + bx + c = 0\) 实根的个数（重根按一个计）.

\begin{enumerate}
\item 求方程 \( x^{2} + bx + c = 0 \) 有实根的概率；
\item 求 \( \xi \) 的分布列和数学期望；
\item 求在先后两次出现的点数中有 \(5\) 的条件下，方程 \(x^{2} + bx + c = 0\) 有实根的概率.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 方程有实根的充要条件是判别式 \(\Delta=b^2-4c\geqslant0\). 当 \(b=1,2,3,4,5,6\) 时，满足条件的 \(c\) 的个数分别为 \(0,1,2,4,6,6\)，所以有实根的基本事件数为 \(0+1+2+4+6+6=19\). 因此所求概率为 \(\frac{19}{36}\).
\item 当 \(\Delta=0\) 时为重根. 在 \(1\leqslant b,c\leqslant6\) 中，\(b^2=4c\) 的情形为 \((b,c)=(2,1)\) 和 \((4,4)\)，共有 \(2\) 个. 因而 \(\xi=1\) 有 \(2\) 个基本事件，\(\xi=2\) 有 \(19-2=17\) 个基本事件，\(\xi=0\) 有 \(36-19=17\) 个基本事件. 所以
\[
P(\xi=0)=\frac{17}{36},\qquad P(\xi=1)=\frac{1}{18},\qquad P(\xi=2)=\frac{17}{36}.
\]
数学期望为 \(E(\xi)=0\cdot\frac{17}{36}+1\cdot\frac{1}{18}+2\cdot\frac{17}{36}=1\).
\item 先后两次出现的点数中有 \(5\) 时，基本事件数为 \(6+6-1=11\). 当 \(b=5\) 时，不论 \(c\) 取何值都有 \(\Delta=25-4c\geqslant1>0\)，共有 \(6\) 个有实根的基本事件. 当 \(c=5\) 且 \(b\neq5\) 时，只有 \(b=6\) 满足 \(\Delta=36-20>0\)，另有 \(1\) 个有实根的基本事件. 故所求条件概率为 \(\frac{6+1}{11}=\frac{7}{11}\).
\end{enumerate}
\end{solution}

如图，在直四棱柱 \(ABCD - A_{1}B_{1}C_{1}D_{1}\) 中，已知 \(DC = DD_{1} = 2AD = 2AB\)，\(AD \perp DC\)，\(AB \parallel DC\).

\begin{enumerate}

\bitmapfigure[max width=0.38\linewidth]{img/2007/shandong_science/q19_fig1.png}

\begin{enumerate}
\item 由已知条件，取 \(AD=AB=1\)，建立直角坐标系，使 \(A(0,0,0)\)，\(B(1,0,0)\)，\(D(0,1,0)\)，\(C(2,1,0)\)，\(A_1(0,0,2)\)，\(C_1(2,1,2)\)，\(D_1(0,1,2)\). 由于 \(E\) 是 \(DC\) 的中点，所以 \(E(1,1,0)\). 在平面 \(A_1BD\) 中，
\[
\overrightarrow{DA_1}=(0,-1,2),\qquad \overrightarrow{DB}=(1,-1,0).

\[
\overrightarrow{D_1E}=(1,0,-2)=-\overrightarrow{DA_1}+\overrightarrow{DB},
\]
所以直线 \(D_1E\) 的方向向量平行于平面 \(A_1BD\). 该平面的一个法向量为
\[
\mathbf n_1=\overrightarrow{DA_1}\times\overrightarrow{DB}=(2,2,1),
\]
其方程为 \(2x+2y+z-2=0\)，而 \(D_1(0,1,2)\) 不在此平面上. 因此 \(D_1E\parallel\) 平面 \(A_1BD\).

\mathbf n_2=\overrightarrow{DB}\times\overrightarrow{DC_1}=(-2,-2,2).

所以二面角 \(A_1-BD-C_1\) 的余弦值为
\[
\frac{|\mathbf n_1\cdot\mathbf n_2|}{|\mathbf n_1||\mathbf n_2|}=\frac{6}{3\cdot2\sqrt{3}}=\frac{\sqrt{3}}{3}.
\]
\end{enumerate}
\end{solution}

\begin{problem}
如图，甲船以每小时 \(30 \sqrt{2}\) 海里的速度向正北方向航行，乙船按固定方向匀速直线航行. 当甲船位于 \(A_{1}\) 处时，乙船位于甲船的北偏西 \(105^{\circ}\) 方向的 \(B_{1}\) 处，此时两船相距 \(20\) 海里. 当甲船航行 \(20\) 分钟到达 \(A_{2}\) 处时，乙船航行到甲船的北偏西 \(120^{\circ}\) 方向的 \(B_{2}\) 处，此时两船相距 \(10 \sqrt{2}\) 海里. 问乙船每小时航行多少海里.

\bitmapfigure[max width=0.32\linewidth]{img/2007/shandong_science/q20_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 取正东方向为 \(x\) 轴正向，正北方向为 \(y\) 轴正向，并令 \(A_1(0,0)\). 甲船 \(20\) 分钟航行的距离为
\[
30\sqrt{2}\times\frac{1}{3}=10\sqrt{2},
\]
所以 \(A_2(0,10\sqrt{2})\). 由北偏西角的方向分量可得
\[
\overrightarrow{A_1B_1}=20(-\sin105^{\circ},\cos105^{\circ})=(-5\sqrt{6}-5\sqrt{2},5\sqrt{2}-5\sqrt{6}),
\]
从而 \(B_1(-5\sqrt{6}-5\sqrt{2},5\sqrt{2}-5\sqrt{6})\). 同理，
\[
\overrightarrow{A_2B_2}=10\sqrt{2}(-\sin120^{\circ},\cos120^{\circ})=(-5\sqrt{6},-5\sqrt{2}),
\]
从而 \(B_2(-5\sqrt{6},5\sqrt{2})\). 因此
\[
\overrightarrow{B_1B_2}=(5\sqrt{2},5\sqrt{6}),\qquad B_1B_2=\sqrt{(5\sqrt{2})^2+(5\sqrt{6})^2}=10\sqrt{2}.
\]
乙船同样航行了 \(20\) 分钟，所以其速度为
\[
\frac{10\sqrt{2}}{1/3}=30\sqrt{2}\text{ 海里/小时}.
\]
\end{enumerate}
\end{solution}

\begin{problem}
已知椭圆 \(C\) 的中心在坐标原点，焦点在 \(x\) 轴上，椭圆 \(C\) 上的点到焦点距离的最大值为 \(3\)，最小值为 \(1\).

\begin{enumerate}
\item 求椭圆 \(C\) 的标准方程；
\item 若直线 \(l: y = kx + m\) 与椭圆 \(C\) 相交于 \(A\)，\(B\) 两点（\(A\)，\(B\) 不是左右顶点），且以 \(AB\) 为直径的圆过椭圆 \(C\) 的右顶点，求证：直线 \(l\) 过定点，并求出该定点的坐标.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 设椭圆方程为 \(\frac{x^2}{a^2}+\frac{y^2}{b^2}=1\)，其中 \(a>b>0\)，焦半距为 \(c\). 对同一个焦点，椭圆上点到该焦点的最大、最小距离分别为 \(a+c\)、\(a-c\)，所以
\[
a+c=3,\qquad a-c=1.
\]
解得 \(a=2\)，\(c=1\)，再由 \(b^2=a^2-c^2\) 得 \(b^2=3\). 因此椭圆 \(C\) 的标准方程为 \(\frac{x^2}{4}+\frac{y^2}{3}=1\).
\item 设 \(A(x_1,kx_1+m)\)，\(B(x_2,kx_2+m)\). 将直线 \(l\) 代入椭圆方程，得交点横坐标满足
\[
(3+4k^2)x^2+8kmx+4m^2-12=0.
\]
由韦达定理，
\[
x_1+x_2=-\frac{8km}{3+4k^2},\qquad x_1x_2=\frac{4m^2-12}{3+4k^2}.
\]
设椭圆的右顶点为 \(R(2,0)\). 以 \(AB\) 为直径的圆过 \(R\)，等价于
\[
\overrightarrow{RA}\cdot\overrightarrow{RB}=0,
\]
即
\[
(x_1-2)(x_2-2)+(kx_1+m)(kx_2+m)=0.
\]
代入上面的两式并化简，得
\[
7m^2+16km+4k^2=(7m+2k)(m+2k)=0.
\]
若 \(m=-2k\)，则直线 \(l\) 为 \(y=k(x-2)\)，经过右顶点 \(R\)，这会使 \(R\) 成为交点 \(A\) 或 \(B\)，与 \(A\)，\(B\) 不是左右顶点矛盾. 因此只能有 \(m=-\frac{2}{7}k\)，于是
\[
y=kx-\frac{2}{7}k=k\left(x-\frac{2}{7}\right).
\]
所以直线 \(l\) 恒过定点 \(\left(\frac{2}{7},0\right)\).
\end{enumerate}
\end{solution}

\begin{problem}
设函数 \( f(x) = x^{2} + b \ln (x + 1) \)，其中 \( b \neq 0 \).

\begin{enumerate}
\item 当 \(b > \frac{1}{2}\) 时，判断函数 \(f(x)\) 在定义域上的单调性；
\item 求函数 \( f(x) \) 的极值点；
\item 证明对任意的正整数 \(n\)，不等式 \(\ln \left(\frac{1}{n} + 1\right) > \frac{1}{n^2} - \frac{1}{n^3}\) 都成立.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 函数定义域为 \((-1,+\infty)\)，且
\[
f'(x)=2x+\frac{b}{x+1}=\frac{2x^2+2x+b}{x+1}.
\]
当 \(b>\frac{1}{2}\) 时，\(2x^2+2x+b=2\left(x+\frac{1}{2}\right)^2+b-\frac{1}{2}>0\)，又 \(x+1>0\)，所以 \(f'(x)>0\). 因此函数 \(f(x)\) 在 \((-1,+\infty)\) 上单调递增.
\item 令 \(f'(x)=0\)，得 \(2x^2+2x+b=0\). 下面分情况讨论.

当 \(b>\frac{1}{2}\) 时无实根；当 \(b=\frac{1}{2}\) 时只有根 \(x=-\frac{1}{2}\)，但 \(f'(x)\geqslant0\) 且在该点两侧不变号，所以均无极值点.

当 \(0<b<\frac{1}{2}\) 时，令 \(d=\sqrt{1-2b}\)，则 \(0<d<1\)，两个根为 \(x_1=\frac{-1-d}{2}\)，\(x_2=\frac{-1+d}{2}\). 导数符号依次为正、负、正，所以极大点和极小点分别为
\[
\left(\frac{-1-d}{2},\frac{(1+d)^2}{4}+b\ln\left(\frac{1-d}{2}\right)\right)
\]
和
\[
\left(\frac{-1+d}{2},\frac{(1-d)^2}{4}+b\ln\left(\frac{1+d}{2}\right)\right).
\]

当 \(b<0\) 时，令 \(d=\sqrt{1-2b}>1\)，根 \(\frac{-1-d}{2}<-1\) 不在定义域内，另一个根 \(x_0=\frac{-1+d}{2}\) 在定义域内，且导数在 \(x_0\) 左侧为负、右侧为正，所以极小点为
\[
\left(\frac{-1+d}{2},\frac{(d-1)^2}{4}+b\ln\left(\frac{d+1}{2}\right)\right),
\]
无极大点.
\item 令 \(g(x)=\ln(1+x)-x^2+x^3\)，则 \(g(0)=0\)，且在 \(0<x\leqslant1\) 时
\[
g'(x)=\frac{1}{1+x}-2x+3x^2=\frac{(1-x)^2+3x^3}{1+x}>0.
\]
所以 \(g(x)>g(0)=0\). 对任意正整数 \(n\)，取 \(x=\frac{1}{n}\in(0,1]\)，即得
\[
\ln\left(\frac{1}{n}+1\right)>\frac{1}{n^2}-\frac{1}{n^3}.
\]
\end{enumerate}
\end{solution}
