\section{解答题}

\begin{problem}
解方程 \(3^{x + 2} - 3^{2 - x} = 80\).
\end{problem}

\begin{solution}
设 \(a=3^x>0\)，则原方程化为
\[
9a-\frac{9}{a}=80
\]
即 \(9a^2-80a-9=0\). 其判别式为 \(\Delta=(-80)^2-4\cdot9\cdot(-9)=82^2\)，所以
\[
a=\frac{80\pm82}{18}=9\quad\text{或}\quad-\frac{1}{9}
\]
由于 \(a>0\)，只能取 \(a=9\). 因而 \(3^x=9=3^2\)，解得 \(x=2\).
\end{solution}

\begin{problem}
设复数 \(z = \cos \theta + \i\sin \theta\)，\(\theta \in (\pi, 2\pi)\)，求复数 \(z^2 + z\) 的模和辐角.
\end{problem}

\begin{solution}
由复数的三角形式，\(z=\cos\theta+\i\sin\theta\)，故
\[
z^2+z=\e^{2\i\theta}+\e^{\i\theta}=2\cos\frac{\theta}{2}\,\e^{\frac{3\theta}{2}\i}
\]
其中 \(\theta\in(\pi,2\pi)\)，所以 \(\theta/2\in(\pi/2,\pi)\)，从而 \(\cos(\theta/2)<0\). 因此
\[
\left|z^2+z\right|=-2\cos\frac{\theta}{2}
\]
且负实数因子使辐角增加 \(\pi\)，所以 \(z^2+z\) 的辐角为
\[
\frac{3\theta}{2}+(2k-1)\pi\quad(k\in\Z)
\]
\end{solution}

\begin{problem}
设 \(\{a_{n}\} \) 是由正数组成的等比数列， \(S_{n} \) 是其前 \(n\) 项和，证明： \(\frac{\log_{0.5}S_{n}+\log_{0.5}S_{n+2}}{2}>\log_{0.5}S_{n+1}. \)
\end{problem}

\begin{solution}
设等比数列的公比为 \(q>0\). 因为 \(S_n\)，\(S_{n+1}\)，\(S_{n+2}\) 都是正数，且对数函数 \(\log_{0.5}u\) 在正数范围内单调递减，所以
\[
\frac{\log_{0.5}S_n+\log_{0.5}S_{n+2}}{2}
=\log_{0.5}\sqrt{S_nS_{n+2}}>\log_{0.5}S_{n+1}
\]
等价于 \(\sqrt{S_nS_{n+2}}<S_{n+1}\)，也就是证明 \(S_{n+1}^2-S_nS_{n+2}>0\). 由 \(S_{n+1}=S_n+a_{n+1}\) 及 \(S_{n+2}=S_n+a_{n+1}+a_{n+2}\)，得
\[
S_{n+1}^2-S_nS_{n+2}=a_{n+1}S_{n+1}-a_{n+2}S_n=a_{n+1}\left(S_{n+1}-qS_n\right)
\]
又因 \(a_{j+1}=qa_j\)，有
\[
S_{n+1}-qS_n=(a_1+a_2+\cdots+a_{n+1})-(a_2+a_3+\cdots+a_{n+1})=a_1
\]
所以 \(S_{n+1}^2-S_nS_{n+2}=a_1a_{n+1}>0\)，原不等式成立.
\end{solution}

\begin{problem}
如图，圆柱的轴截面 \(ABCD\) 是正方形，点 \(E\) 在底面的圆周上，\(AF \perp DE \)，\(F\) 是垂足.

\begin{enumerate}
\item 求证：\(AF \perp DB\)；
\item 如果圆柱与三棱锥 \(D-ABE\) 的体积的比等于 \(3\pi \)，求点 \(E\) 到截面 \(ABCD\) 的距离.
\end{enumerate}

\bitmapfigure[max width=0.34\linewidth]{img/1995/national_liberal/q24_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item
设圆柱的底面半径为 \(r\)，则由轴截面 \(ABCD\) 为正方形知圆柱的高为 \(2r\). 建立直角坐标系，使底面圆心为原点，截面 \(ABCD\) 为平面 \(y=0\)，取 \(A=(-r,0,0)\)，\(B=(r,0,0)\)，\(D=(-r,0,2r)\)，并令 \(E=(r\cos\varphi,r\sin\varphi,0)\).
其中 \(\varphi\) 为实数. 设 \(F=D+\lambda(E-D)\)，则
\[
\overrightarrow{AF}=r\left(\lambda(1+\cos\varphi),\lambda\sin\varphi,2-2\lambda\right)
\]
且
\[
\overrightarrow{DE}=r(1+\cos\varphi,\sin\varphi,-2)
\]
由 \(AF\perp DE\)，得
\[
\overrightarrow{AF}\cdot\overrightarrow{DE}=r^2\left(2\lambda(3+\cos\varphi)-4\right)=0
\]
另一方面，\(\overrightarrow{DB}=r(2,0,-2)\)，所以
\[
\overrightarrow{AF}\cdot\overrightarrow{DB}=r^2\left(2\lambda(3+\cos\varphi)-4\right)=0
\]
从而 \(AF\perp DB\).

\item
设点 \(E\) 到平面 \(ABCD\) 的距离为 \(d\). 因为平面 \(ABCD\) 为 \(y=0\)，所以 \(d=|r\sin\varphi|\)，且三角形 \(ABE\) 的底边 \(AB=2r\)，对应高为 \(d\)，故其面积为 \(rd\). 因此圆柱的体积为 \(V_{\text{圆柱}}=\pi r^2\cdot2r=2\pi r^3\)，三棱锥 \(D-ABE\) 的体积为 \(V_{D-ABE}=\frac13\cdot rd\cdot2r=\frac23r^2d\).
于是
\[
\frac{V_{\text{圆柱}}}{V_{D-ABE}}=\frac{2\pi r^3}{\frac23r^2d}=\frac{3\pi r}{d}=3\pi
\]
解得 \(d=r\). 所以点 \(E\) 到截面 \(ABCD\) 的距离等于圆柱的底面半径.
\end{enumerate}
\end{solution}

\begin{problem}
某地为促进淡水鱼养殖业的发展，将价格控制在适当范围内，决定对淡水鱼养殖提供政府补贴. 设淡水鱼的市场价格为 \(x\) 元/千克，政府补贴为 \(t\) 元/千克. 根据市场调查，当 \(8 \leqslant x \leqslant 14\) 时，淡水鱼的市场日供应量 \(P\) 千克与市场日需求量 \(Q\) 千克近似满足关系：\(P = 1000(x + t - 8)\)（\(x \geqslant 8\)，\(t \geqslant 0\)），\(Q = 500\sqrt{40 - (x - 8)^2}\)（\(8 \leqslant x \leqslant 14\)）. 当 \(P = Q\) 时市场价格称为市场平衡价格.

\begin{enumerate}
\item 将市场平衡价格表示为政府补贴的函数，并求出函数的定义域；
\item 为使市场平衡价格不高于每千克 \(10\) 元，政府补贴至少为每千克多少元？
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 市场平衡时 \(P=Q\)，所以
\[
1000(x+t-8)=500\sqrt{40-(x-8)^2}
\]
即
\[
2(x+t-8)=\sqrt{40-(x-8)^2}
\]
令 \(u=x-8\)，则 \(0\leqslant u\leqslant6\)，且
\[
2(u+t)=\sqrt{40-u^2}
\]
两边均非负，可以平方得 \(4(u+t)^2=40-u^2\)，整理为
\[
5u^2+8tu+4t^2-40=0
\]
解得
\[
u=\frac{-4t\pm2\sqrt{50-t^2}}{5}
\]
由于 \(u\geqslant0\) 且 \(t\geqslant0\)，只能取正号，故
\[
x(t)=8+u=8-\frac45t+\frac25\sqrt{50-t^2}
\]
要使根式有意义，需 \(0\leqslant t\leqslant\sqrt{50}\)；要使正号所得的 \(u\geqslant0\)，还需
\[
2\sqrt{50-t^2}\geqslant4t
\]
即 \(t^2\leqslant10\). 当 \(0\leqslant t\leqslant\sqrt{10}\) 时，所得 \(u\) 从 \(2\sqrt{2}\) 减至 \(0\)，始终满足 \(0\leqslant u\leqslant6\). 所以函数的定义域为 \([0,\sqrt{10}]\).

\item 在定义域内，\(x(t)\) 随 \(t\) 增大而减小，因为
\[
x'(t)=-\frac45-\frac{2t}{5\sqrt{50-t^2}}<0
\]
对 \(x=10\)，即 \(u=2\)，代入平衡方程得
\[
2(2+t)=\sqrt{40-2^2}=6
\]
所以 \(t=1\). 因为平衡价格随补贴增加而降低，要使市场平衡价格不高于每千克 \(10\) 元，政府补贴至少为每千克 \(1\) 元.
\end{enumerate}
\end{solution}

\begin{problem}
已知椭圆 \(\frac{x^2}{24} + \frac{y^2}{16} = 1\)，直线 \(l: x = 12\). \(P\) 是 \(l\) 上一点，射线 \(OP\) 交椭圆于点 \(R\)，又点 \(Q\) 在 \(OP\) 上且满足 \(|OQ| \cdot |OP| = |OR|^2\)，当点 \(P\) 在 \(l\) 上移动时，求点 \(Q\) 的轨迹方程，并说明轨迹是什么曲线.
\end{problem}

\begin{solution}
设 \(P=(12,p)\)，其中 \(p\in\R\). 设射线 \(OP\) 与椭圆的交点为 \(R=\lambda P\)，则 \(\lambda>0\)，且由椭圆方程得
\[
\frac{(12\lambda)^2}{24}+\frac{(p\lambda)^2}{16}=1
\]
即
\[
\lambda^2\left(6+\frac{p^2}{16}\right)=1
\]
所以 \(\lambda^2=\dfrac{1}{6+\frac{p^2}{16}}\). 又因 \(Q\) 在射线 \(OP\) 上，设 \(Q=\mu P\)，其中 \(\mu\geqslant0\). 由题设
\[
|OQ|\cdot|OP|=|OR|^2
\]
以及 \(|OQ|=\mu|OP|\)，\(|OR|=\lambda|OP|\)，可得 \(\mu=\lambda^2\). 因而若 \(Q=(x,y)\)，则 \(x=\frac{12}{6+\frac{p^2}{16}}\)，\(y=\frac{p}{6+\frac{p^2}{16}}\).
特别地，\(x>0\)，且由 \(y/x=p/12\) 得 \(p=12y/x\). 代入 \(12/x=6+p^2/16\)，得到
\[
\frac{12}{x}=6+\frac{9y^2}{x^2}
\]
即
\[
6x^2+9y^2=12x
\]
从而
\[
(x-1)^2+\frac{y^2}{\frac23}=1,\qquad x>0
\]
反过来，椭圆上任一点满足 \(x>0\) 时，取 \(p=12y/x\) 即可按上述关系得到相应的点 \(P\)，所以这就是点 \(Q\) 的完整轨迹. 该轨迹是以 \((1,0)\) 为中心、长半轴为 \(1\)、短半轴为 \(\sqrt{6}/3\) 的椭圆，去掉左端点 \((0,0)\).
\end{solution}
