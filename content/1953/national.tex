\begin{problem}
解 \(\frac{x + 1}{x - 1} + \frac{x - 1}{x + 1} = \frac{10}{3}\).

\end{problem}

\begin{solution}
将原方程整化得 \(6\left( x^{2} + 1 \right) = 10\left( x^{2} - 1 \right)\)，故 \(4x^{2} = 16\)，\(x = \pm 2\).
\end{solution}

\begin{problem}
若 \(3x^{2} + kx + 12 = 0\) 之二根相等，求 \(k\).

\end{problem}

\begin{solution}
原方程二根相等之条件为 \(k^2 - 4 \times 3 \times 12 = 0\)，即 \(k^{2} = 12^{2}\)，\(k = \pm 12\).
\end{solution}

\begin{problem}
求 \(\left| \begin{array}{ccc}3 & -1 & 1\\ 2 & 4 & 6\\ 7 & 0 & 5 \end{array} \right|\) 之值.

\end{problem}

\begin{solution}
\[
\begin{aligned}
\text{原行列式}&=3 \times 4 \times 5 - 6 \times 7 - 4 \times 7 + 2 \times 5\\
&=60 - 42 - 28 + 10=0
\end{aligned}
\]
\end{solution}

\begin{problem}
求 \(\log_{10}\frac{300}{7} + \log_{10}\frac{700}{3} + \log_{10}1\) 之值.

\end{problem}

\begin{solution}
\[
\begin{aligned}
\text{原式}&=\log_{10}\left( \frac{300}{7}\times \frac{700}{3}\times 1 \right)\\
&=\log_{10}10000=4
\end{aligned}
\]
\end{solution}

\begin{problem}
求 \(\tan \left( 870^{\circ} \right)\).

\end{problem}

\begin{solution}
\[
\begin{aligned}
\tan 870^{\circ}&=\tan \left( 900^{\circ} - 30^{\circ} \right)=\tan \left( -30^{\circ} \right)=-\tan 30^{\circ}=-\frac{1}{\sqrt{3}}
\end{aligned}
\]
\end{solution}

\begin{problem}
若 \(\cos 2x = \frac{1}{2}\)，求 \(x\) 之通值.

\end{problem}

\begin{solution}
因 \(\cos 2x = \frac{1}{2}\)，故 \(2x = 2n\pi \pm \frac{\pi}{3}\)，\(x = n\pi \pm \frac{\pi}{6}\).
\end{solution}

\begin{problem}
两三角形相似之条件为何？

\end{problem}

\begin{solution}
（i）\(\angle A = \angle A'\)，\(\angle B = \angle B'\)；（ii）\(\angle A = \angle A'\)，\(AB:A'B' = AC:A'C'\)；（iii）\(AB:A'B' = AC:A'C' = BC:B'C'\)；三者各为 \(\triangle ABC \sim \triangle A'B'C'\) 之条件.
\end{solution}

\begin{problem}
长方体之长、宽、高为 \(12\) 寸，\(3\) 寸，\(4\) 寸，求对角线之长.

\end{problem}

\begin{solution}
对角线为 \(\sqrt{12^2 + 3^2 + 4^2} = \sqrt{169} = 13\) 寸.
\end{solution}

\begin{problem}
垂直三棱柱之高为 \(6\) 寸，底面三边之长为 \(3\) 寸，\(4\) 寸，\(5\) 寸，求体积.

\end{problem}

\begin{solution}
因 \(3^{2} + 4^{2} = 5^{2}\)，故底面为直角三角形，其面积为 \(\frac{1}{2} \times 3 \times 4\) 方寸 \(= 6\) 方寸. （或用“底面积 \(= \frac{1}{4}\sqrt{\left( 3 + 4 + 5 \right)\left( 3 + 4 - 5 \right)\left( 3 - 4 + 5 \right)\left( -3 + 4 + 5 \right)}\) 方寸 \(= 6\) 方寸”亦可）棱柱体积 \(= 6\times 6\) 立方寸 \(= 36\) 立方寸.
\end{solution}

\begin{problem}
球之表面积为 \(36\pi\) 方寸，求体积.

\end{problem}

\begin{solution}
设球的半径为 \(R\) 寸，则 \(4\pi R^{2}=36\pi\)，因此 \(R=3\). 球的体积为 \(\frac{4}{3}\pi R^3 = 36\pi\)（立方寸）.
\end{solution}

\begin{problem}
解 \(\left\{ \begin{array}{l} x^{2} - 2xy + 3y^{2} = 9, \\ 4x^{2} - 5xy + 6y^{2} = 30 \end{array} \right.\)

\end{problem}

\begin{solution}
解：原方程组消去常数项，得
\[
2x^{2} + 5xy - 12y^{2} = 0
\]
将此方程左边分解因式，得
\[
\left( x + 4y \right)\left( 2x - 3y \right) = 0
\]
即 \(x + 4y = 0\)，\(2x - 3y = 0\).

由此有

\begin{center}
\begin{tabular}{@{}c@{\hspace{2.5em}}c@{}}
（I）\ \(\left\{ \begin{array}{l} x^{2} - 2xy + 3y^{2} = 9, \\ x + 4y = 0 \end{array} \right.\);%
&
（II）\ \(\left\{ \begin{array}{l} x^{2} - 2xy + 3y^{2} = 9, \\ 2x - 3y = 0 \end{array} \right.\).%
\end{tabular}
\end{center}

解方程组（I），得

\begin{center}
\begin{tabular}{@{}c@{\hspace{2.5em}}c@{}}
\(\begin{array}{l} x_{1} = \frac{4}{3}\sqrt{3}, \\ y_{1} = -\frac{1}{3}\sqrt{3} \end{array}\);%
&
\(\begin{array}{l} x_{2} = -\frac{4}{3}\sqrt{3}, \\ y_{2} = \frac{1}{3}\sqrt{3} \end{array}\);%
\end{tabular}
\end{center}

解方程组（II），得

\begin{center}
\begin{tabular}{@{}c@{\hspace{2.5em}}c@{}}
\(\begin{array}{l} x_{3} = 3, \\ y_{3} = 2 \end{array}\);%
&
\(\begin{array}{l} x_{4} = -3, \\ y_{4} = -2 \end{array}\).%
\end{tabular}
\end{center}
\end{solution}

\begin{problem}
\begin{enumerate}
    \item 化简 \(\sqrt{\frac{12}{25}} + \sqrt[4]{90000} + \sqrt[6]{\frac{64}{27}}\).
    \item 求 \(\left( 2x^{3} + \frac{1}{x} \right)^{12}\) 之展开式中之常数项.
\end{enumerate}

\end{problem}

\begin{solution}
解：（1）
\[
\begin{aligned}
\text{原式}&=\sqrt{\frac{3\times 2^{2}}{5^{2}}} + \sqrt[4]{3^{2}\times 10^{4}} + \sqrt[6]{\frac{2^{6}}{3^{3}}}\\
&=\frac{2}{5}\sqrt{3} + 10\sqrt{3} + \frac{2}{3}\sqrt{3}=\frac{166}{15}\sqrt{3}
\end{aligned}
\]

（2）由二项展开式的通项公式，
\[
\begin{aligned}
T_{r+1} &= \mathrm C_{12}^{r}\left( 2x^{3} \right)^{12-r}\left( \frac{1}{x} \right)^{r} \\
&= \mathrm C_{12}^{r}\cdot 2^{12-r}\cdot x^{36-4r}
\end{aligned}
\]
令 \(36 - 4r = 0\)，因此 \(r = 9\). 故常数项为 \(\mathrm C_{12}^{9} \cdot 2^{12-9} = \mathrm C_{12}^{3} \cdot 2^{3} = 1760\).
\end{solution}

\begin{problem}
锐角三角形 \(ABC\) 之三高线为 \(AD, BE, CF\)，垂心为 \(H\)，求证 \(HD\) 平分 \(\angle EDF\).

\end{problem}

\begin{solution}
证明：由于 \(AD\perp BC\)，\(BE\perp CA\)，因此点 \(A, B, D, E\) 共圆. 故 \(\angle ADE = \angle ABE\). 又因点 \(F, B, C, E\) 共圆，因此 \(\angle FBE = \angle FCE\). 又因点 \(C, A, F, D\) 共圆，因此 \(\angle FCA = \angle FDA\). 综上可得 \(\angle ADE = \angle FDA\)，即 \(AD\) 平分 \(\angle EDF\).

\end{solution}

\begin{problem}
已知三角形的两个角为 \(45^{\circ}\) 及 \(60^{\circ}\)，而其夹边长 \(1\) 尺；求最小边之长及面积.

\end{problem}

\begin{solution}
解：已知 \(\angle B=45^{\circ}\)，\(\angle C=60^{\circ}\)，于是 \(\angle A=75^{\circ}\). 由正弦定理得
\[
\frac{AC}{\sin45^{\circ}}=\frac{1}{\sin75^{\circ}}
\]
所以
\[
\begin{aligned}
AC&=\frac{\sin45^{\circ}}{\sin75^{\circ}}=\frac{\frac{\sqrt{2}}{2}}{\frac{\sqrt{2}}{2}\left( \frac{\sqrt{3}}{2}+\frac{1}{2} \right)}=\sqrt{3}-1\text{（尺）}
\end{aligned}
\]
\(\triangle ABC\) 的面积为
\[
\begin{aligned}
S&=\frac{1}{2}\times 1\times \left( \sqrt{3}-1 \right)\cdot \sin60^{\circ}=\frac{1}{4}\left( 3-\sqrt{3} \right)\text{（平方尺）}
\end{aligned}
\]
\end{solution}
