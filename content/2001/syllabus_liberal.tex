\begin{problem}

\begin{enumerate}
\item 求 \(f\left(\frac{1}{2}\right)\)，\(f\left(\frac{1}{4}\right)\)；
\item 证明：\(f(x)\) 是周期函数.
\end{enumerate}
\end{problem}
