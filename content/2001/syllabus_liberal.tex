\section{单选题}
\begin{problem}
\(\tan 300^{\circ} + \cot 405^{\circ}\) 的值为（\quad）

\choices
    {\(1 + \sqrt{3}\)}
    {\(1 - \sqrt{3}\)}
    {\(-1 - \sqrt{3}\)}
    {\(-1 + \sqrt{3}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
因为 \(\tan 300^{\circ}=-\tan 60^{\circ}=-\sqrt{3}\)，且 \(\cot 405^{\circ}=\cot 45^{\circ}=1\)，所以原式为 \(1-\sqrt{3}\)，故选 B.
\end{solution}

\begin{problem}
过点 \(A(1, -1)\)，\(B(-1, 1)\) 且圆心在直线 \(x + y - 2 = 0\) 上的圆的方程是（\quad）

\choices
    {\((x - 3)^2 + (y + 1)^2 = 4\)}
    {\((x + 3)^2 + (y - 1)^2 = 4\)}
    {\((x - 1)^2 + (y - 1)^2 = 4\)}
    {\((x + 1)^2 + (y + 1)^2 = 4\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
设圆心为 \(O(u,v)\). 由 \(OA=OB\) 得 \((u-1)^2+(v+1)^2=(u+1)^2+(v-1)^2\)，化简得 \(u=v\). 又因为圆心在直线 \(x+y-2=0\) 上，所以 \(u+v=2\)，从而 \(O(1,1)\). 半径 \(r=OA=2\)，故圆的方程为 \((x-1)^2+(y-1)^2=4\)，选 C.
\end{solution}

\begin{problem}
若一个圆锥的轴截面积是等边三角形，其面积为 \(\sqrt{3}\)，则这个圆锥的全面积是（\quad）

\choices
    {\(3\pi\)}
    {\(3\sqrt{3}\pi\)}
    {\(6\pi\)}
    {\(9\pi\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
设圆锥的底面半径为 \(r\)，母线长为 \(l\). 轴截面是等边三角形时，其底边为圆锥底面的直径，所以 \(l=2r\). 由轴截面面积为 \(\sqrt{3}\)，得 \(\frac{\sqrt{3}}{4}(2r)^2=\sqrt{3}\)，从而 \(r=1\)，\(l=2\). 因此圆锥的全面积为 \(\pi r^2+\pi rl=\pi+2\pi=3\pi\)，故选 A.
\end{solution}

\begin{problem}
若定义在区间 \((-1,0)\) 内的函数 \(f(x) = \log_{2a}(x + 1)\) 满足 \(f(x) > 0\)，则 \(a\) 的取值范围是（\quad）

\choices
    {\(\left(0, \frac{1}{2}\right)\)}
    {\(\left(0, \frac{1}{2}\right]\)}
    {\(\left(\frac{1}{2},+\infty\right)\)}
    {\((0, +\infty)\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
当 \(x\in(-1,0)\) 时，\(0<x+1<1\). 对于底数 \(2a\) 合法的对数，\(\log_{2a}(x+1)>0\) 对所有 \(x+1\in(0,1)\) 成立，当且仅当 \(0<2a<1\). 因而 \(0<a<\frac12\)，故选 A.
\end{solution}

\begin{problem}
已知复数 \(z = \sqrt{2} + \sqrt{6}\i\)，则 \(\arg \frac{1}{z}\) 是（\quad）

\choices
    {\(\frac{\pi}{3}\)}
    {\(\frac{5\pi}{3}\)}
    {\(\frac{\pi}{6}\)}
    {\(\frac{11\pi}{6}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
复数 \(z=\sqrt{2}+\sqrt{6}\i\) 位于第一象限，且 \(\tan\arg z=\frac{\sqrt{6}}{\sqrt{2}}=\sqrt{3}\)，所以 \(\arg z=\frac{\pi}{3}\). 因此 \(\arg\frac1z=-\frac{\pi}{3}\pmod{2\pi}\)，在选项给出的范围内为 \(\frac{5\pi}{3}\)，故选 B.
\end{solution}

\begin{problem}
函数 \(y = 2^{-x} + 1\)（\(x > 0\)）的反函数是（\quad）

\choices
    {\( y = \log_2\frac{1}{x - 1} \)（\(x \in (1,2)\)）}
    {\(y = -\log_2\frac{1}{x - 1}\)（\(x \in (1,2)\)）}
    {\( y = \log_2\frac{1}{x - 1} \)（\(x \in (1,2]\)）}
    {\( y = -\log_2\frac{1}{x - 1} \)（\(x \in (1,2]\)）}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
由 \(x>0\) 得 \(1<2^{-x}+1<2\)，所以原函数的值域为 \((1,2)\). 交换自变量与因变量，得 \(x=2^{-y}+1\)，即 \(x-1=2^{-y}\)，从而 \(y=-\log_2(x-1)=\log_2\frac{1}{x-1}\)，且 \(x\in(1,2)\). 故选 A.
\end{solution}

\begin{problem}
若椭圆经过原点，且焦点为 \(F_{1}(1,0)\)，\(F_{2}(3,0)\)，则其离心率为（\quad）

\choices
    {\(\frac{3}{4}\)}
    {\(\frac{2}{3}\)}
    {\(\frac{1}{2}\)}
    {\(\frac{1}{4}\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
椭圆中心为两个焦点的中点 \((2,0)\)，焦半距为 \(c=1\). 原点在椭圆上，故原点到两个焦点的距离之和为 \(2a\)，即 \(1+3=2a\)，所以 \(a=2\). 因而离心率 \(e=\frac{c}{a}=\frac12\)，故选 C.
\end{solution}

\begin{problem}
若 \(0 < \alpha < \beta < \frac{\pi}{4}\)，\(\sin \alpha + \cos \alpha = a\)，\(\sin \beta + \cos \beta = b\)，则（\quad）

\choices
    {\(a > b\)}
    {\(a < b\)}
    {\(ab < 1\)}
    {\(ab > 2\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
设 \(h(t)=\sin t+\cos t\)，则 \(h'(t)=\cos t-\sin t\). 当 \(0<t<\frac{\pi}{4}\) 时，\(\cos t>\sin t\)，所以 \(h(t)\) 在该区间上单调递增. 由 \(\alpha<\beta\) 得 \(a=h(\alpha)<h(\beta)=b\)，故选 B.
\end{solution}

\begin{problem}
在正三棱柱 \(ABC - A_{1}B_{1}C_{1}\) 中，若 \(AB = \sqrt{2} BB_{1}\)，则 \(AB_{1}\) 与 \(C_{1}B\) 所成的角的大小为（\quad）

\choices
    {\(60^{\circ}\)}
    {\(90^{\circ}\)}
    {\(45^{\circ}\)}
    {\(120^{\circ}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
设 \(AB=s\)，则由 \(AB=\sqrt{2}BB_1\) 得 \(BB_1=\frac{s}{\sqrt{2}}\). 建立直角坐标系，取 \(A=(0,0,0)\)，\(B=(s,0,0)\)，\(C=\left(\frac{s}{2},\frac{\sqrt{3}s}{2},0\right)\)，并令棱柱高为 \(h=\frac{s}{\sqrt{2}}\). 则 \(\overrightarrow{AB_1}=(s,0,h)\)，\(\overrightarrow{BC_1}=\left(-\frac{s}{2},\frac{\sqrt{3}s}{2},h\right)\). 两向量的内积为 \(-\frac{s^2}{2}+h^2=0\)，所以两直线垂直，所成角为 \(90^{\circ}\)，故选 B.
\end{solution}

\begin{problem}
设 \(f(x)\)，\(g(x)\) 都是单调函数，有如下四个命题中，正确的命题是

\begin{circlelist}
\item 若 \(f(x)\) 单调递增，\(g(x)\) 单调递增，则 \(f(x) - g(x)\) 单调递增；
\item 若 \(f(x)\) 单调递增，\(g(x)\) 单调递减，则 \(f(x) - g(x)\) 单调递增；
\item 若 \(f(x)\) 单调递减，\(g(x)\) 单调递增，则 \(f(x) - g(x)\) 单调递减；
\item 若 \(f(x)\) 单调递减，\(g(x)\) 单调递减，则 \(f(x) - g(x)\) 单调递减.
\end{circlelist}

（\quad）

\choices
    {\circled{1}\circled{3}}
    {\circled{1}\circled{4}}
    {\circled{2}\circled{3}}
    {\circled{2}\circled{4}}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
若 \(x_1<x_2\)，且 \(f\) 单调递增、\(g\) 单调递减，则 \(f(x_1)<f(x_2)\)，\(g(x_1)>g(x_2)\)，从而 \(f(x_1)-g(x_1)<f(x_2)-g(x_2)\)，第二个命题正确. 若 \(f\) 单调递减、\(g\) 单调递增，则 \(f(x_1)>f(x_2)\)，\(g(x_1)<g(x_2)\)，从而 \(f(x_1)-g(x_1)>f(x_2)-g(x_2)\)，第三个命题正确. 两个函数同向单调时，差的单调性不能确定，所以第一个、第四个命题不一定正确，故选 C.
\end{solution}

\begin{problem}
一间平房的屋顶有如图三种不同的盖法，分别是：

\begin{circlelist}
\item 单向倾斜；
\item 双向倾斜；
\item 四向倾斜.
\end{circlelist}

记三种盖法屋顶面积分别为 \(P_{1}\)，\(P_{2}\)，\(P_{3}\). 若屋顶斜面与水平面所成的角都是 \(\alpha\)，则（\quad）

\problemresetlayout
\bitmapfigure[max width=0.74\linewidth]{img/2001/syllabus_liberal/q11_fig1.png}

\choices
    {\(P_{3} > P_{2} > P_{1}\)}
    {\(P_{3} > P_{2} = P_{1}\)}
    {\(P_{3} = P_{2} > P_{1}\)}
    {\(P_{3} = P_{2} = P_{1}\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
设屋顶水平投影的面积为 \(S_0\). 每个斜面与水平面所成的角均为 \(\alpha\)，故斜面面积与其水平投影面积的关系均为 \(S_0=P_i\cos\alpha\). 因为三种盖法的水平投影相同，且 \(\cos\alpha>0\)，所以 \(P_1=P_2=P_3\)，故选 D.
\end{solution}

\begin{problem}
如图，小圆圈表示网络的结点，结点之间的连线表示它们有网线相联，连线标注的数字表示该段网线单位时间内可以通过的最大信息量，现从结点 \(A\) 向结点 \(B\) 传递信息，信息可以分开沿不同的路线同时传递，则单位时间内传递的最大信息量为（\quad）

\bitmapfigure[max width=0.42\linewidth]{img/2001/syllabus_liberal/q12_fig1.png}

\choices
    {\(26\)}
    {\(24\)}
    {\(20\)}
    {\(19\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
把图中上方两条路线的汇合结点记为 \(U\)，下方两条路线的汇合结点记为 \(V\). 经过 \(U\) 的信息流量至多为从 \(B\) 出发的两条上方网线容量之和 \(3+4=7\)，故上方路线至多传递 \(7\) 个单位的信息. 经过 \(V\) 的信息最终都要通过容量为 \(12\) 的网线到达 \(A\)，故下方路线至多传递 \(12\) 个单位的信息. 因此总流量不超过 \(7+12=19\).

取上方两条路线分别传递 \(3\) 和 \(4\) 个单位，下方两条路线分别传递 \(6\) 和 \(6\) 个单位，则各段网线的容量均不超限，且到达 \(A\) 的总量为 \(3+4+6+6=19\). 所以最大信息量为 \(19\)，故选 D.
\end{solution}

\begin{problem}
\(\left(\frac{1}{2} x + 1\right)^{10}\) 的二项展开式中 \(x^{3}\) 的系数为\fillinblank{}.
\end{problem}

\begin{answer}
15
\end{answer}

\begin{solution}
由二项式定理，\(x^3\) 项必须从十个因式中取出三个 \(\frac12x\)，其余七个因式取 \(1\). 因此 \(x^3\) 的系数为 \(\mathrm C_{10}^{3}\left(\frac12\right)^3=120\cdot\frac18=15\).
\end{solution}

\begin{problem}
双曲线 \( \frac{x^{2}}{9}-\frac{y^{2}}{16}=1 \) 的两个焦点为 \(F_{1}\)，\(F_{2}\)，点 \(P\) 在双曲线上，若 \( PF_{1} \perp PF_{2} \)，则点 \(P\) 到 \(x\) 轴的距离为\fillinblank{}.
\end{problem}

\begin{answer}
\(\frac{16}{5}\)
\end{answer}

\begin{solution}
双曲线的焦半距满足 \(c=\sqrt{9+16}=5\)，所以两个焦点为 \((-5,0)\) 和 \((5,0)\). 设 \(P=(x,y)\)，由 \(PF_1\perp PF_2\) 得 \(\overrightarrow{PF_1}\cdot\overrightarrow{PF_2}=0\)，即 \((-5-x,-y)\cdot(5-x,-y)=0\)，从而 \(x^2+y^2=25\). 又因 \(P\) 在双曲线上，\(\frac{x^2}{9}-\frac{y^2}{16}=1\). 代入 \(y^2=25-x^2\)，得 \(16x^2-9(25-x^2)=144\)，所以 \(x^2=\frac{369}{25}\)，进而 \(y^2=25-\frac{369}{25}=\frac{256}{25}\). 因而点 \(P\) 到 \(x\) 轴的距离为 \(|y|=\frac{16}{5}\).
\end{solution}

\begin{problem}
设 \(\{a_{n}\}\) 是公比为 \(q\) 的等比数列，\(S_{n}\) 是它的前 \(n\) 项和. 若 \(\{S_{n}\}\) 是等差数列，则 \(q = \)\fillinblank{}.
\end{problem}

\begin{answer}
1
\end{answer}

\begin{solution}
因为 \(\{S_n\}\) 是等差数列，所以 \(S_2-S_1=S_3-S_2\). 又 \(S_2-S_1=a_2=a_1q\)，\(S_3-S_2=a_3=a_1q^2\)，故 \(a_1q=a_1q^2\). 等比数列的首项及公比均非零，所以 \(a_1q\ne0\)，从而 \(q=1\).
\end{solution}

\begin{problem}
圆周上有 \(2n\) 个等分点（\(n > 1\)），以其中三个点为顶点的直角三角形的个数为\fillinblank{}.
\end{problem}

\begin{answer}
\(2n(n-1)\)
\end{answer}

\begin{solution}
圆周上任取三个等分点，若它们构成直角三角形，则由圆周角定理，直角所对的边必为直径. 反过来，每取一个直径，再从其余的 \(2n-2\) 个等分点中任选一个作为第三个顶点，就得到一个直角三角形. 圆周上的 \(2n\) 个点两两相对，共有 \(n\) 条直径，故直角三角形的个数为 \(n(2n-2)=2n(n-1)\).
\end{solution}

\begin{problem}
已知等差数列前三项为 \(a\)，\(4\)，\(3a\)，前 \(n\) 项的和为 \(S_{n}\)，\(S_{k} = 2550\).

\begin{enumerate}
\item 求 \(a\) 及 \(k\) 的值；
\item 求 \(\displaystyle\lim_{n\to\infty}\left(\frac{1}{S_{1}}+\frac{1}{S_{2}}+\cdots+\frac{1}{S_{n}}\right)\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 等差数列满足中间项是首、末两项的平均数，所以 \(a+3a=2\times4\)，解得 \(a=2\). 此时首项为 \(2\)，公差为 \(4-2=2\)，从而 \(S_k=2k+\frac{k(k-1)}{2}\times2=k(k+1)\). 由 \(S_k=2550\)，得 \(k^2+k-2550=0\)，即 \((k-50)(k+51)=0\). 因为 \(k\) 为正整数，所以 \(k=50\).
\item 对任意正整数 \(n\)，有 \(S_n=2n+\frac{n(n-1)}{2}\times2=n(n+1)\). 因而
\[
\frac{1}{S_1}+\frac{1}{S_2}+\cdots+\frac{1}{S_n}=\sum_{k=1}^{n}\frac{1}{k(k+1)}=\sum_{k=1}^{n}\left(\frac{1}{k}-\frac{1}{k+1}\right)=1-\frac{1}{n+1}.
\]
所以所求极限为 \(\displaystyle\lim_{n\to\infty}\left(1-\frac{1}{n+1}\right)=1\).
\end{enumerate}
\end{solution}

\begin{problem}
如图，在底面是直角梯形的四棱锥 \(S - ABCD\) 中，\(\angle ABC = 90^{\circ}\)，\(SA \perp\) 面 \(ABCD\)，\(SA = AB = BC = 1\)，\(AD = \frac{1}{2}\).

\begin{enumerate}
\item 求四棱锥 \(S - ABCD\) 的体积；
\item 求面 \(SCD\) 与面 \(SBA\) 所成的二面角的正切值.
\end{enumerate}

\bitmapfigure[max width=0.34\linewidth]{img/2001/syllabus_liberal/q18_fig1.png}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 因为底面是直角梯形，且 \(AD\parallel BC\)，所以 \(AB\) 是梯形的高. 底面面积为 \(S_{ABCD}=\frac{AD+BC}{2}\cdot AB=\frac{\frac12+1}{2}\cdot1=\frac34\). 又因为 \(SA\perp\) 面 \(ABCD\)，四棱锥的高为 \(SA=1\)，所以体积为 \(V=\frac13S_{ABCD}\cdot SA=\frac13\cdot\frac34\cdot1=\frac14\).
\item 建立空间直角坐标系，取 \(A=(0,0,0)\)，\(B=(1,0,0)\)，\(C=(1,1,0)\)，\(D=\left(0,\frac12,0\right)\)，\(S=(0,0,1)\). 则平面 \(SBA\) 的一个法向量为 \(\bs{n}_1=(0,1,0)\). 又 \(\overrightarrow{SC}=(1,1,-1)\)，\(\overrightarrow{SD}=\left(0,\frac12,-1\right)\)，所以平面 \(SCD\) 的一个法向量可取 \(\bs{n}_2=\overrightarrow{SC}\times\overrightarrow{SD}=\left(-\frac12,1,\frac12\right)\)，即也可取 \(\bs{n}_2=(-1,2,1)\). 设两平面所成的锐二面角为 \(\theta\)，则
\[
\cos\theta=\frac{|\bs{n}_1\cdot\bs{n}_2|}{|\bs{n}_1||\bs{n}_2|}=\frac{2}{\sqrt{6}}.
\]
因此 \(\tan\theta=\frac{\sqrt{1-\cos^2\theta}}{\cos\theta}=\frac{1}{\sqrt{2}}\).
\end{enumerate}
\end{solution}

\begin{problem}
已知圆内接四边形 \(ABCD\) 的边长分别为 \(AB = 2\)，\(BC = 6\)，\(CD = DA = 4\)，求四边形 \(ABCD\) 的面积.
\end{problem}

\begin{solution}
连接 \(BD\)，设 \(\angle BAD=A\)，\(\angle BCD=C\). 因为四边形 \(ABCD\) 为圆内接四边形，所以 \(A+C=180^{\circ}\)，从而 \(\sin A=\sin C\). 因此四边形面积为
\[
S=\frac12AB\cdot AD\sin A+\frac12BC\cdot CD\sin C=16\sin A.
\]
在三角形 \(ABD\) 和 \(CBD\) 中分别使用余弦定理，得 \(BD^2=AB^2+AD^2-2AB\cdot AD\cos A=20-16\cos A\)，以及 \(BD^2=BC^2+CD^2-2BC\cdot CD\cos C=52-48\cos C\). 又 \(\cos C=-\cos A\)，所以 \(20-16\cos A=52+48\cos A\)，从而 \(\cos A=-\frac12\)，即 \(\sin A=\frac{\sqrt3}{2}\). 因此 \(S=16\cdot\frac{\sqrt3}{2}=8\sqrt3\).
\end{solution}

\begin{problem}
设抛物线 \(y^{2} = 2px\)（\(p > 0\)）的焦点为 \(F\)，经过点 \(F\) 的直线交抛物线于 \(A\)，\(B\) 两点. 点 \(C\) 在抛物线的准线上，且 \(BC \parallel x\) 轴. 证明直线 \(AC\) 经过原点 \(O\).
\end{problem}

\begin{solution}
抛物线 \(y^2=2px\) 的焦点为 \(F\left(\frac p2,0\right)\)，准线为 \(x=-\frac p2\). 设抛物线上点 \(A\)、\(B\) 的参数分别为 \(t_1\)、\(t_2\)，则 \(A=\left(\frac p2t_1^2,pt_1\right)\)，\(B=\left(\frac p2t_2^2,pt_2\right)\).
连接参数为 \(t_1,t_2\) 的两点所得弦的方程为 \(2x-(t_1+t_2)y+pt_1t_2=0\). 因为该直线经过焦点 \(F\)，代入得 \(p+pt_1t_2=0\)，所以 \(t_1t_2=-1\).

由 \(BC\parallel x\) 轴且 \(C\) 在准线上，得 \(C=\left(-\frac p2,pt_2\right)\). 直线 \(OA\) 的斜率为 \(\frac{pt_1}{\frac p2t_1^2}=\frac{2}{t_1}\)，而直线 \(OC\) 的斜率为 \(\frac{pt_2}{-\frac p2}=-2t_2=\frac{2}{t_1}\)，其中用到了 \(t_1t_2=-1\). 因此 \(O\)、\(A\)、\(C\) 三点共线，直线 \(AC\) 经过原点 \(O\).
\end{solution}

\begin{problem}
设计一幅宣传画，要求画面面积为 \(4840 \mathrm{~cm}^{2}\)，画面的宽与高的比为 \(\lambda\)（\(\lambda < 1\)），画面的上、下各留 \(8 \mathrm{~cm}\) 空白，左、右各留 \(5 \mathrm{~cm}\) 空白. 怎样确定画面的高与宽尺寸，能使宣传画所用纸张面积最小？
\end{problem}

\begin{solution}
设画面高为 \(h\mathrm{~cm}\)，宽为 \(w\mathrm{~cm}\)，则 \(wh=4840\)，且 \(\lambda=\frac{w}{h}<1\). 纸张的高为 \(h+16\)，宽为 \(w+10\)，所以纸张面积为
\[
S=(h+16)(w+10)=4840+16w+10h+160=5000+16w+10h.
\]

\begin{problem}
设 \(f(x)\) 是定义在 \(\R\) 上的偶函数，其图象关于直线 \(x = 1\) 对称，对任意 \(x_{1}, x_{2} \in \left[0, \frac{1}{2}\right]\)，都有 \(f(x_{1} + x_{2}) = f(x_{1}) \cdot f(x_{2})\).

\begin{enumerate}
\item 设 \(f(1)=2\)，求 \(f\left(\frac{1}{2}\right)\)，\(f\left(\frac{1}{4}\right)\)；
\item 证明：\(f(x)\) 是周期函数.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 对任意 \(x\in[0,1]\)，取 \(x_{1}=x_{2}=\frac{x}{2}\)，则 \(x_{1},x_{2}\in\left[0,\frac{1}{2}\right]\)，从而 \(f(x)=f\left(\frac{x}{2}\right)f\left(\frac{x}{2}\right)=f\left(\frac{x}{2}\right)^{2}\geqslant0\). 取 \(x_{1}=x_{2}=\frac{1}{2}\)，由 \(f(1)=2\) 得 \(f\left(\frac{1}{2}\right)^{2}=f(1)=2\). 结合 \(f\left(\frac{1}{2}\right)\geqslant0\)，有 \(f\left(\frac{1}{2}\right)=\sqrt{2}=2^{\frac{1}{2}}\). 再取 \(x_{1}=x_{2}=\frac{1}{4}\)，得 \(f\left(\frac{1}{4}\right)^{2}=f\left(\frac{1}{2}\right)=2^{\frac{1}{2}}\). 由于同样有 \(f\left(\frac{1}{4}\right)\geqslant0\)，所以 \(f\left(\frac{1}{4}\right)=2^{\frac{1}{4}}\).
\item 由函数图象关于直线 \(x=1\) 对称，对任意 \(x\in\R\) 有 \(f(x)=f(2-x)\). 又因为 \(f\) 是偶函数，所以 \(f(-x)=f(x)\). 在 \(f(x)=f(2-x)\) 中以 \(-x\) 代替 \(x\)，得到 \(f(-x)=f(2+x)\). 因而 \(f(x)=f(x+2)\)，对任意 \(x\in\R\) 成立，所以 \(f(x)\) 是周期函数，且 \(2\) 是它的一个周期.
\end{enumerate}
\end{solution}
