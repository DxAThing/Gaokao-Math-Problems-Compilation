\section{解答题}

\begin{problem}
设正数 \(a,b,c\) 满足 \(a^{2}+b^{2}=c^{2}\).
\begin{enumerate}
\item 求证：\(\log_{2}\left(1+\frac{b+c}{a}\right)+\log_{2}\left(1+\frac{a-c}{b}\right)=1\).
\item 又设 \(\log_{4}\left(1+\frac{b+c}{a}\right)=1\)，\(\log_{8}(a+b-c)=\frac23\). 求 \(a\)，\(b\)，和 \(c\) 的值.
\end{enumerate}
\end{problem}
\begin{solution}
由 \(a^{2}+b^{2}=c^{2}\)，有
\[
\log_{2}\left(1+\frac{b+c}{a}\right)+\log_{2}\left(1+\frac{a-c}{b}\right)
=\log_{2}\frac{(a+b)^{2}-c^{2}}{ab}=\log_{2}2=1.
\]

又由第二问条件，\(a+b+c=4a\)，\(a+b-c=4\)，且上式给出 \(\log_{2}\frac{a+b-c}{b}=-1\)，故 \(a+b-c=\frac b2\)，从而 \(b=8\)，联立得 \(a=6,c=10\).
\end{solution}

\begin{problem}
在三棱台 \(A_{1}B_{1}C_{1}-ABC\) 中，侧棱 \(B_{1}B\perp\) 底面 \(ABC\)，且 \(\angle ABC=\angle AA_{1}C=\frac\pi2\)，\(AB=2A_{1}B_{1}=2\mathrm{~cm}\).
\begin{enumerate}
\item 求证：\(BC\perp A_{1}B\)，\(BC\perp A_{1}A\)，\(A_{1}A\perp A_{1}B\).
\item 求异面直线 \(A_{1}A\) 和 \(BC\) 的距离.
\end{enumerate}
\bitmapfigure[max width=0.46\linewidth,max totalheight=5.2cm]{img/1987/guangdong_liberal/q7_fig1.png}
\end{problem}
\begin{solution}
因为 \(B_{1}B\perp\) 平面 \(ABC\)，且 \(BC\perp AB\)，所以 \(BC\perp\) 平面 \(BB_{1}A_{1}A\)，从而 \(BC\perp A_{1}B,BC\perp A_{1}A\)，又 \(AA_{1}\perp A_{1}C\)，且 \(AA_{1}\perp BC\)，故 \(AA_{1}\perp\) 平面 \(A_{1}BC\)，于是 \(AA_{1}\perp A_{1}B\).

由棱台性质 \(AB\parallel A_{1}B_{1}\)，且 \(B_{1}B\perp A_{1}B_{1}\)，\(A_{1}B\perp AA_{1}\)，得 \(\triangle A_{1}B_{1}B\sim\triangle BA_{1}A\)，故
\[
A_{1}B^{2}=A_{1}B_{1}\cdot AB=1\cdot2=2
\]
因此 \(A_{1}B\) 是两异面直线的公垂线，距离为 \(\sqrt2\mathrm{~cm}\).
\end{solution}

\begin{problem}
直线 \(l\) 通过抛物线 \(y^{2}=2px\ (p\neq0)\) 的焦点，并且与这抛物线相交于 \(A(x_{1},y_{1})\) 和 \(B(x_{2},y_{2})\) 两点.
\begin{enumerate}
\item 求证：\(y_{1}y_{2}=-p^{2}\).
\item 点 \(C\) 在这抛物线的准线上，且 \(AC\) 平行于 \(x\) 轴，求证：\(B,C\) 和这抛物线的顶点三点共线.
\end{enumerate}
\end{problem}
\begin{solution}
抛物线焦点为 \(F(\frac p2,0)\)，若 \(l\perp x\) 轴，则 \(x_{1}=x_{2}=\frac p2\)，显然 \(y_{1}y_{2}=-p^{2}\)，否则设 \(l:y=k(x-\frac p2)\)，代入抛物线得关于 \(y\) 的二次方程，按韦达定理得 \(y_{1}y_{2}=-p^{2}\).

准线为 \(x=-\frac p2\)，故由 \(AC\parallel x\) 轴，\(C=(-\frac p2,y_{1})\)，又 \(x_{2}=\frac{y_{2}^{2}}{2p}\)，于是
\[
\frac{y_{2}}{x_{2}}=\frac{2p}{y_{2}}=\frac{2py_{1}}{-p^{2}}=\frac{y_{1}}{-p/2},
\]
所以 \(B(x_{2},y_{2})\)、\(C(-\frac p2,y_{1})\) 与顶点 \(O\) 共线.
\end{solution}

\begin{problem}
设 \(|\theta|<\frac\pi2\)，数列 \(\{a_n\}\) 的通项是 \(a_n=\sin\frac{n\pi}{2}\cdot\tan^{n}\theta\)，记 \(S_{2n}=a_{1}+a_{2}+\dots+a_{2n-1}+a_{2n}\).
\begin{enumerate}
\item 求证：对任何自然数 \(n\)，\(S_{2n}=\frac12(\sin2\theta)[1+(-1)^{n+1}\tan^{2n}\theta]\).
\item 求证：当 \(n\geqslant2\) 时，\(a_n^{2}+a_{n-1}a_{n+1}+(-1)^n(a_{1}^{n})^{2}=0\).
\end{enumerate}
\end{problem}
\begin{solution}
偶数项为零，且 \(a_{2k-1}=(-1)^{k+1}\tan^{2k-1}\theta\)，故
\[S_{2n}=\tan\theta-\tan^{3}\theta+\cdots+(-1)^{n+1}\tan^{2n-1}\theta=\frac{\tan\theta[1+(-1)^{n+1}\tan^{2n}\theta]}{1+\tan^{2}\theta}=\frac12\sin2\theta[1+(-1)^{n+1}\tan^{2n}\theta]\].

当 \(n=2k\) 时，\(a_n=0\)，且 \(a_{n-1}a_{n+1}=(-1)^{2k+1}\tan^{4k}\theta=-\tan^{4k}\theta\)，与 \((-1)^n(a_1^n)^2=\tan^{4k}\theta\) 相消；当 \(n=2k-1\) 时，\(a_{n-1}a_{n+1}=0\)，而 \(a_n^2=\tan^{4k-2}\theta\)，\((-1)^n(a_1^n)^2=-\tan^{4k-2}\theta\)，故等式均成立.
\end{solution}
