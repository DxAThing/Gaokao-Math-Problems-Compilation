    {充要条件}

\begin{solution}
由正弦差公式，
\[
\sin(\alpha+\beta)-\sin\alpha=2\cos\left(\alpha+\frac{\beta}{2}\right)\sin\frac{\beta}{2}
\]
由于 \(\beta\) 为锐角，\(\sin\frac{\beta}{2}>0\)，且 \(0<\alpha+\frac{\beta}{2}<\frac{3\pi}{4}\)，所以
\[
p\Longleftrightarrow \alpha+\frac{\beta}{2}<\frac{\pi}{2}
\]
若 \(q\) 成立，则 \(\alpha+\frac{\beta}{2}<\alpha+\beta<\frac{\pi}{2}\)，故 \(q\Rightarrow p\). 取 \(\alpha=\frac{\pi}{3}\)，\(\beta=\frac{\pi}{6}\)，则 \(p\) 成立而 \(q\) 不成立. 因此 \(p\) 是 \(q\) 的必要而不充分条件，选 B.
\end{solution}

\begin{problem}
若函数 \( f(x) = \frac{1 + \cos 2x}{4\sin\left(\frac{\pi}{2} + x\right)} - a\sin \frac{x}{2}\cos \left(\pi - \frac{x}{2}\right) \) 的最大值为\(2\)，试确定常数 \( a \) 的值.
\end{problem}

\begin{solution}
原函数的定义域满足 \(\cos x\ne0\). 在此定义域内，
\[
\frac{1+\cos2x}{4\sin\left(\frac{\pi}{2}+x\right)}=\frac{2\cos^{2}x}{4\cos x}=\frac{1}{2}\cos x,
\]
又 \(\cos\left(\pi-\frac{x}{2}\right)=-\cos\frac{x}{2}\)，所以
\[
f(x)=\frac{1}{2}\cos x+a\sin\frac{x}{2}\cos\frac{x}{2}=\frac{\cos x+a\sin x}{2}
\]
由柯西不等式，
\[
f(x)=\frac{1}{2}(1,a)\cdot(\cos x,\sin x)\leqslant\frac{1}{2}\sqrt{1+a^{2}}
\]
当 \(\cos x=\frac{1}{\sqrt{1+a^{2}}}\)，\(\sin x=\frac{a}{\sqrt{1+a^{2}}}\) 时等号成立，且此时 \(\cos x\ne0\)，故上界就是最大值. 因此
\[
\frac{1}{2}\sqrt{1+a^{2}}=2\Longrightarrow a^{2}=15,
\]
从而 \(a=\pm\sqrt{15}\).
\end{solution}

\begin{problem}
在一次购物抽奖活动中，假设某 \(10\) 张券中有一等奖券 \(1\) 张，可获价值 \(50\text{ 元}\) 的奖品；有二等奖券 \(3\) 张，每张可获价值 \(10\text{ 元}\) 的奖品；其余 \(6\) 张没有奖，某顾客从此 \(10\) 张券中任抽 \(2\) 张，求：

\begin{enumerate}
\item 该顾客中奖的概率；
\item 该顾客获得的奖品总价值 \(\xi\text{（元）}\) 的概率分布列和期望 \(E(\xi)\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item
从 \(10\) 张券中任取 \(2\) 张共有 \(\mathrm C_{10}^{2}=45\) 种等可能结果. 没有中奖的结果有 \(\mathrm C_{6}^{2}=15\) 种，所以该顾客中奖的概率为
\[
1-\frac{\mathrm C_{6}^{2}}{\mathrm C_{10}^{2}}=1-\frac{15}{45}=\frac{2}{3}
\]

\item
随机变量 \(\xi\) 的可能取值为 \(0\)，\(10\)，\(20\)，\(50\)，\(60\). 分别计算得 \(P(\xi=0)=\frac{\mathrm C_{6}^{2}}{\mathrm C_{10}^{2}}=\frac{15}{45}\)，\(P(\xi=10)=\frac{\mathrm C_{3}^{1}\mathrm C_{6}^{1}}{\mathrm C_{10}^{2}}=\frac{18}{45}\)，\(P(\xi=20)=\frac{\mathrm C_{3}^{2}}{\mathrm C_{10}^{2}}=\frac{3}{45}\)，\(P(\xi=50)=\frac{\mathrm C_{6}^{1}}{\mathrm C_{10}^{2}}=\frac{6}{45}\)，\(P(\xi=60)=\frac{\mathrm C_{3}^{1}}{\mathrm C_{10}^{2}}=\frac{3}{45}\).

故 \(\xi\) 的分布列为
\begin{center}
\begin{tabular}{c|ccccc}
\(\xi\) & \(0\) & \(10\) & \(20\) & \(50\) & \(60\) \\
\hline
\(P\) & \(\dfrac{15}{45}\) & \(\dfrac{18}{45}\) & \(\dfrac{3}{45}\) & \(\dfrac{6}{45}\) & \(\dfrac{3}{45}\)
\end{tabular}
\end{center}
于是
\[
E(\xi)=0\cdot\frac{15}{45}+10\cdot\frac{18}{45}+20\cdot\frac{3}{45}+50\cdot\frac{6}{45}+60\cdot\frac{3}{45}=16
\]
\end{enumerate}
\end{solution}

\begin{problem}
已知 \(a \in \R\)，讨论函数 \(f(x) = \e^{x}(x^{2} + ax + a + 1)\) 的极值点的个数.
\end{problem}

\begin{solution}
令 \(g(x)=x^{2}+ax+a+1\)，则
\[
f'(x)=\e^{x}\bigl(g(x)+g'(x)\bigr)=\e^{x}\left[x^{2}+(a+2)x+2a+1\right]
\]
由于 \(\e^{x}>0\)，\(f'(x)\) 的符号由
\[
h(x)=x^{2}+(a+2)x+2a+1
\]
决定. 其判别式为
\[
\Delta=(a+2)^{2}-4(2a+1)=a(a-4)
\]
当 \(a<0\) 或 \(a>4\) 时，\(\Delta>0\)，\(h(x)\) 有两个不等实根，且开口向上，因此 \(f'(x)\) 的符号依次为正、负、正，函数有一个极大值点和一个极小值点，共 \(2\) 个极值点.

当 \(0<a<4\) 时，\(\Delta<0\)，又二次项系数为正，所以 \(h(x)>0\)，函数没有极值点. 当 \(a=0\) 或 \(a=4\) 时，分别有
\[
h(x)=(x+1)^{2}\quad\text{或}\quad h(x)=(x+3)^{2},
\]
\(f'(x)\) 不变号，也没有极值点.

所以极值点的个数为
\[
\begin{cases}
2, & a<0\ \text{或}\ a>4,\\
0, & 0\leqslant a\leqslant4
\end{cases}
\]
\end{solution}

\begin{problem}
如图，在三棱柱 \(ABC - A_{1}B_{1}C_{1}\) 中，\(AB \perp\) 侧面 \(BB_{1}C_{1}C\)，\(E\) 为棱 \(CC_{1}\) 上异于 \(C\)，\(C_{1}\) 的一点，\(EA \perp EB_{1}\)，已知 \(AB = \sqrt{2}\)，\(BB_{1} = 2\)，\(BC = 1\)，\(\angle BCC_{1} = \frac{\pi}{3}\)，求：

\begin{enumerate}
\item 异面直线 \(AB\) 与 \(EB_{1}\) 的距离；
\item 二面角 \(A - EB_{1} - A_{1}\) 的平面角的正切值.
\end{enumerate}

\bitmapfigure[max width=0.42\linewidth]{img/2005/chongqing_science/q20_fig1.png}
\end{problem}

\begin{solution}
建立直角坐标系，令 \(B=(0,0,0)\)，\(C=(1,0,0)\)，并令 \(z\) 轴垂直于侧面 \(BB_{1}C_{1}C\)，则
\[
A=(0,0,\sqrt{2})
\]
设棱柱的平移向量为 \(\overrightarrow{BB_{1}}=\overrightarrow{CC_{1}}=(u,v,0)\). 由 \(|\overrightarrow{CC_{1}}|=2\) 以及 \(\angle BCC_{1}=\frac{\pi}{3}\)，有
\[
(-1,0,0)\cdot(u,v,0)=2\cos\frac{\pi}{3}=1,\qquad u^{2}+v^{2}=4
\]
取 \(v>0\)，得 \((u,v)=(-1,\sqrt{3})\)，从而
\[
B_{1}=(-1,\sqrt{3},0),\quad C_{1}=(0,\sqrt{3},0),\quad A_{1}=(-1,\sqrt{3},\sqrt{2})
\]
设 \(E=(1-t)C+tC_{1}=(1-t,\sqrt{3}t,0)\)，其中 \(0<t<1\). 由 \(EA\perp EB_{1}\)，
\[
\overrightarrow{EA}\cdot\overrightarrow{EB_{1}}=(t-1)(t-2)-3t(1-t)=4t^{2}-6t+2=0
\]
所以 \((2t-1)(t-1)=0\). 因 \(E\ne C_{1}\)，故 \(t=\frac12\)，即
\[
E=\left(\frac12,\frac{\sqrt{3}}2,0\right)
\]

\begin{enumerate}
\item 取 \(AB\) 和 \(EB_{1}\) 的方向向量
\[
\bs{u}=(0,0,\sqrt{2}),\qquad \bs{v}=\left(-\frac32,\frac{\sqrt{3}}2,0\right)
\]
则
\[
\bs{u}\times\bs{v}=\left(-\frac{\sqrt{6}}2,-\frac{3\sqrt{2}}2,0\right),\qquad \left|\bs{u}\times\bs{v}\right|=\sqrt{6}
\]
因此两异面直线的距离为
\[
d=\frac{\left|\overrightarrow{BE}\cdot(\bs{u}\times\bs{v})\right|}{\left|\bs{u}\times\bs{v}\right|}
=\frac{\left|\left(\frac12,\frac{\sqrt{3}}2,0\right)\cdot\left(-\frac{\sqrt{6}}2,-\frac{3\sqrt{2}}2,0\right)\right|}{\sqrt{6}}=1
\]

\item 设平面 \(AEB_{1}\)，\(A_{1}EB_{1}\) 的法向量分别为
\[
\bs{n}_{1}=\overrightarrow{EA}\times\overrightarrow{EB_{1}}=\left(-\frac{\sqrt{6}}2,-\frac{3\sqrt{2}}2,-\sqrt{3}\right),
\]
\[
\bs{n}_{2}=\overrightarrow{EA_{1}}\times\overrightarrow{EB_{1}}=\left(-\frac{\sqrt{6}}2,-\frac{3\sqrt{2}}2,0\right)
\]
若二面角的平面角为 \(\theta\)，则
\[
\tan\theta=\frac{|\bs{n}_{1}\times\bs{n}_{2}|}{|\bs{n}_{1}\cdot\bs{n}_{2}|}=\frac{3\sqrt{2}}{6}=\frac{\sqrt{2}}2
\]
\end{enumerate}
\end{solution}

\begin{problem}
已知椭圆 \(C_1\) 的方程为 \(\frac{x^2}{4} + y^2 = 1\)，双曲线 \(C_2\) 的左、右焦点分别为 \(C_1\) 的左、右顶点，而 \(C_2\) 的左、右顶点分别是 \(C_1\) 的左、右焦点.

\begin{enumerate}
\item 求双曲线 \(C_2\) 的方程.
\item 若直线 \(l: y = kx + \sqrt{2}\) 与椭圆 \(C_1\) 及双曲线 \(C_2\) 都恒有两个不同的交点，且 \(l\) 与 \(C_2\) 的两个交点 \(A\) 和 \(B\) 满足 \(\overrightarrow{OA} \cdot \overrightarrow{OB} < 6\) （其中 \(O\) 为原点），求 \(k\) 的取值范围.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 椭圆 \(C_{1}\) 的长半轴为 \(2\)，焦半距为 \(\sqrt{3}\). 因此双曲线 \(C_{2}\) 的焦半距为 \(2\)，实半轴为 \(\sqrt{3}\)，虚半轴满足
\[
b^{2}=2^{2}-(\sqrt{3})^{2}=1
\]
故
\[
C_{2}:\frac{x^{2}}{3}-y^{2}=1
\]

\item 直线 \(l\) 与椭圆的交点横坐标满足
\[
\left(k^{2}+\frac14\right)x^{2}+2\sqrt{2}kx+1=0,
\]
其判别式为 \(4k^{2}-1\)，所以与椭圆有两个不同交点的条件是 \(|k|>\frac12\).

直线 \(l\) 与双曲线的交点横坐标满足
\[
\left(\frac13-k^{2}\right)x^{2}-2\sqrt{2}kx-3=0
\]
其判别式为 \(4(1-k^{2})\)，且二次项系数不能为零. 因此与双曲线有两个不同交点的条件是
\[
|k|<1,\qquad k^{2}\ne\frac13
\]

设 \(A\)，\(B\) 的横坐标分别为 \(x_{1}\)，\(x_{2}\)，记 \(D=1-3k^{2}\). 由上面的双曲线交点方程及韦达定理，
\[
x_{1}+x_{2}=\frac{6\sqrt{2}k}{D},\qquad x_{1}x_{2}=-\frac{9}{D}
\]
又 \(y_i=kx_i+\sqrt{2}\)，所以
\[
\overrightarrow{OA}\cdot\overrightarrow{OB}
=x_{1}x_{2}+y_{1}y_{2}
=(1+k^{2})x_{1}x_{2}+\sqrt{2}k(x_{1}+x_{2})+2
=-\frac{7+3k^{2}}{1-3k^{2}}
\]
令 \(t=k^{2}\). 结合交点条件，有 \(\frac14<t<1\)，且 \(t\ne\frac13\). 当 \(\frac14<t<\frac13\) 时，上式小于 \(0\)，自然小于 \(6\)；当 \(\frac13<t<1\) 时，条件
\[
\frac{7+3t}{3t-1}<6
\]
等价于 \(t>\frac{13}{15}\). 因此
\[
k^{2}\in\left(\frac14,\frac13\right)\cup\left(\frac{13}{15},1\right),
\]
即
\[
k\in\left(-1,-\sqrt{\frac{13}{15}}\right)\cup\left(-\frac{1}{\sqrt{3}},-\frac12\right)\cup\left(\frac12,\frac{1}{\sqrt{3}}\right)\cup\left(\sqrt{\frac{13}{15}},1\right)
\]
\end{enumerate}
\end{solution}

\begin{problem}
数列 \(\{a_{n}\}\) 满足 \(a_1 = 1\) 且 \(a_{n + 1} = \left(1 + \frac{1}{n^2 + n}\right)a_n + \frac{1}{2^n}\) （\(n \geqslant 1\)）.

\begin{enumerate}
\item 用数学归纳法证明： \( a_{n} \geqslant 2 \) （\(n \geqslant 2\)）；
\item 已知不等式 \(\ln(1+x) < x\) 对 \(x > 0\) 成立，证明 \( a_{n}<\e^{2} \) （\(n\geqslant1\)），其中无理数 \(\e = 2.71828\cdots\).
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 当 \(n=2\) 时，由递推式得
\[
a_{2}=\left(1+\frac{1}{1^{2}+1}\right)a_{1}+\frac12=\frac32+\frac12=2
\]
假设 \(n\geqslant2\) 时 \(a_{n}\geqslant2\)，则
\[
a_{n+1}=\left(1+\frac{1}{n^{2}+n}\right)a_{n}+\frac{1}{2^{n}}
\geqslant2\left(1+\frac{1}{n^{2}+n}\right)+\frac{1}{2^{n}}>2
\]
所以由数学归纳法，\(a_{n}\geqslant2\) 对一切 \(n\geqslant2\) 成立.

\item 令
\[
p_{1}=1,\qquad p_{n}=\prod_{j=1}^{n-1}\left(1+\frac{1}{j^{2}+j}\right)\quad(n\geqslant2)
\]
将递推式除以 \(p_{n+1}\)，得
\[
\frac{a_{n+1}}{p_{n+1}}=\frac{a_{n}}{p_{n}}+\frac{1}{2^{n}p_{n+1}}
\]
逐项相加，得到
\[
\frac{a_{n}}{p_{n}}=1+\sum_{j=1}^{n-1}\frac{1}{2^{j}p_{j+1}}<1+\sum_{j=1}^{n-1}\frac{1}{2^{j}}<2,
\]
其中 \(p_{j+1}>1\). 另一方面，由题设不等式，
\[
\ln p_{n}
=\sum_{j=1}^{n-1}\ln\left(1+\frac{1}{j^{2}+j}\right)
<\sum_{j=1}^{n-1}\frac{1}{j^{2}+j}
\]

又
\[
\sum_{j=1}^{n-1}\frac{1}{j^{2}+j}
=\sum_{j=1}^{n-1}\left(\frac1j-\frac{1}{j+1}\right)=1-\frac1n<1
\]
因此 \(p_{n}<\e\)，从而
\[
a_{n}<2p_{n}<2\e<\e^{2},
\]
因为 \(\e>2\). 故 \(a_{n}<\e^{2}\) 对一切 \(n\geqslant1\) 成立.
\end{enumerate}
\end{solution}
