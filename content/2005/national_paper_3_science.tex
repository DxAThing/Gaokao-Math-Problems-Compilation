\section{单选题}
\begin{problem}
已知 \(\alpha\) 为第三象限角，则 \(\frac{\alpha}{2}\) 所在的象限是（\(\quad\)）

\choices
    {第一或第二象限}
    {第二或第三象限}
    {第一或第三象限}
    {第二或第四象限}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
第三象限角加上 \(2k\pi\) 后除以 \(2\)，所得角在第二或第四象限，故选 D.
\end{solution}

\begin{problem}
已知过点 \(A(-2, m)\) 和 \(B(m, 4)\) 的直线与直线 \(2x + y - 1 = 0\) 平行，则 \(m\) 的值为（\(\quad\)）

\choices
    {\(0\)}
    {\(-8\)}
    {\(2\)}
    {\(10\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
由平行条件 \(\frac{4-m}{m+2}=-2\)，解得 \(m=-8\)，故选 B.
\end{solution}

\begin{problem}
在 \((x - 1)(x + 1)^8\) 的展开式中 \(x^5\) 的系数是（\(\quad\)）

\choices
    {\(-14\)}
    {\(14\)}
    {\(-28\)}
    {\(28\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
\(x^5\) 的系数为 \(\mathrm C_8^4-\mathrm C_8^5=70-56=14\)，故选 B.
\end{solution}

\begin{problem}
设三棱柱 \(ABC - A_{1}B_{1}C_{1}\) 的体积为 \(V\)，\(P\)，\(Q\) 分别是侧棱 \(AA_{1}\)，\(CC_{1}\) 上的点，且 \(PA = QC_{1}\)，则四棱锥 \(B - APQC\) 的体积为（\(\quad\)）

\choices
    {\(\frac{1}{6} V\)}
    {\(\frac{1}{4} V\)}
    {\(\frac{1}{3} V\)}
    {\(\frac{1}{2} V\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
在侧面 \(ACC_{1}A_{1}\) 内，\(AP+CQ=AA_{1}\)，故四边形 \(APQC\) 面积为侧面面积的一半；四棱锥高等于棱柱高方向的相应距离，体积为棱柱体积的 \(\frac13\)，故选 C.
\end{solution}

\begin{problem}
\(\lim\limits_{x\to 1}\left(\frac{1}{x^2 - 3x + 2} -\frac{2}{x^2 - 4x + 3}\right) =\)（\(\quad\)）

\choices
    {\(-\frac{1}{2}\)}
    {\(\frac{1}{2}\)}
    {\(-\frac{1}{6}\)}
    {\(\frac{1}{6}\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
通分并约去 \(x-1\)：\(\frac{1}{(x-1)(x-2)}-\frac{2}{(x-1)(x-3)}=-\frac1{(x-2)(x-3)}\)，故极限为 \(-\frac12\)，选 A.
\end{solution}

\begin{problem}
若 \(a = \frac{\ln 2}{2}\)，\(b = \frac{\ln 3}{3}\)，\(c = \frac{\ln 5}{5}\)，则（\(\quad\)）

\choices
    {\(a < b < c\)}
    {\(c < b < a\)}
    {\(c < a < b\)}
    {\(b < a < c\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
由 \(3^2>2^3\)、\(2^5>5^2\)，得 \(\frac{\ln5}{5}<\frac{\ln2}{2}<\frac{\ln3}{3}\)，故选 C.
\end{solution}

\begin{problem}
设 \(0 \leqslant x \leqslant 2\pi\)，且 \(\sqrt{1 - \sin 2x} = \sin x - \cos x\)，则（\(\quad\)）

\choices
    {\(0 \leqslant x \leqslant \pi\)}
    {\(\frac{\pi}{4} \leqslant x \leqslant \frac{7\pi}{4}\)}
    {\(\frac{\pi}{4} \leqslant x \leqslant \frac{5\pi}{4}\)}
    {\(\frac{\pi}{2} \leqslant x \leqslant \frac{3\pi}{2}\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
\(\sqrt{1-\sin2x}=|\sin x-\cos x|\)，题设要求 \(\sin x\geqslant\cos x\)，故 \(x\in[\frac\pi4,\frac{5\pi}4]\)，选 C.
\end{solution}

\begin{problem}
\(\frac{2\sin 2\alpha}{1 + \cos 2\alpha} \cdot \frac{\cos^2\alpha}{\cos 2\alpha} =\)（\(\quad\)）

\choices
    {\(\tan \alpha\)}
    {\(\tan 2\alpha\)}
    {\(1\)}
    {\(\frac{1}{2}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
利用 \(1+\cos2\alpha=2\cos^2\alpha\)、\(\sin2\alpha=2\sin\alpha\cos\alpha\)，原式化为 \(\tan2\alpha\)，故选 B.
\end{solution}

\begin{problem}
已知双曲线 \( x^{2}-\frac{y^{2}}{2}=1 \) 的焦点为 \(F_{1}\)，\(F_{2}\)，点 \(M\) 在双曲线上且 \( \overrightarrow{MF_{1}}\cdot\overrightarrow{MF_{2}}=0 \)，则点 \(M\) 到 \(x\) 轴的距离为（\(\quad\)）

\choices
    {\(\frac{4}{3}\)}
    {\(\frac{5}{3}\)}
    {\(\frac{2\sqrt{3}}{3}\)}
    {\(\sqrt{3}\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
设焦半距为 \(\sqrt3\). 由向量垂直得 \(x^2+y^2=3\)，与双曲线方程联立得 \(y^2=\frac43\)，故距离为 \(\frac{2\sqrt3}{3}\)，选 C.
\end{solution}

\begin{problem}
设椭圆的两个焦点分别为 \(F_{1}\)，\(F_{2}\)，过 \(F_{2}\) 作椭圆长轴的垂线交椭圆于点 \(P\)，若 \(\triangle F_{1}PF_{2}\) 为等腰直角三角形，则椭圆的离心率是（\(\quad\)）

\choices
    {\( \frac{\sqrt{2}}{2} \)}
    {\(\frac{\sqrt{2} - 1}{2}\)}
    {\(2 - \sqrt{2}\)}
    {\(\sqrt{2} - 1\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
设椭圆长半轴为 \(a\)，离心率为 \(e\)，则 \(F_1F_2=2ae\)，且 \(F_2P=a(1-e^2)\). 由等腰直角三角形的两直角边相等，得 \(2ae=a(1-e^2)\)，从而 \(e=\sqrt2-1\)，选 D.
\end{solution}

\begin{problem}
不共面的四个定点到平面 \(\alpha\) 的距离都相等，这样的平面 \(\alpha\) 共有（\(\quad\)）

\choices
    {\(3\text{ 个}\)}
    {\(4\text{ 个}\)}
    {\(6\text{ 个}\)}
    {\(7\text{ 个}\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
对四个点到平面的有向距离分别取符号，除整体换号外共有 \(2^3-1=7\) 个非平凡符号型，每型确定一个平面，故选 D.
\end{solution}

\begin{problem}
计算机中常用十六进制是逢 \(16\) 进 \(1\) 的计数制，采用数字 \(0\sim 9\) 和字母 \(A\sim F\) 共 \(16\) 个计数符号，这些符号与十进制的数的对应关系如下表：

\begin{center}
\small
\begin{tabular}{|c|c|c|c|c|c|c|c|c|}
\hline
十六进制 & 0 & 1 & 2 & 3 & 4 & 5 & 6 & 7 \\
\hline
十进制 & 0 & 1 & 2 & 3 & 4 & 5 & 6 & 7 \\
\hline
十六进制 & 8 & 9 & A & B & C & D & E & F \\
\hline
十进制 & 8 & 9 & 10 & 11 & 12 & 13 & 14 & 15 \\
\hline
\end{tabular}
\end{center}

例如，用十六进制表示：\(\mathrm{E} + \mathrm{D} = 1\mathrm{B}\)，则 \(\mathrm{A}\times \mathrm{B} =\)（\(\quad\)）

\choices
    {\(6\mathrm{E}\)}
    {\(72\)}
    {\(5\mathrm{F}\)}
    {\(\mathrm{B}0\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
十六进制中 \(A=10\)、\(B=11\)，故 \(A\times B=110=6E\)，选 A.
\end{solution}

\section{填空题}
\begin{problem}
已知复数 \(z_0 = 3 + 2\i\)，复数 \(z\) 满足 \(z \cdot z_0 = 3z + z_0\)，则复数 \(z = \)\fillinblank{}.
\end{problem}

\begin{answer}
\(1-\frac32\i\)
\end{answer}

\begin{solution}
由 \(2\i z=3+2\i\)，得 \(z=1-\frac32\i\).
\end{solution}

\begin{problem}
已知向量 \( \overrightarrow{OA} = (k, 12) \)，\( \overrightarrow{OB} = (4, 5) \)，\( \overrightarrow{OC} = (-k, 10) \)，且 \(A\)，\(B\)，\(C\) 三点共线，则 \(k=\)\fillinblank{}.
\end{problem}

\begin{answer}
\(-\frac23\)
\end{answer}

\begin{solution}
共线条件给出 \((4-k)(-2)-(-7)(-2k)=0\)，故 \(k=-\frac23\).
\end{solution}

\begin{problem}
设 \(l\) 为平面上过点 \((0,1)\) 的直线，\(l\) 的斜率等可能地取
\(-2\sqrt{2}\)，\(-\sqrt{3}\)，\(-\frac{\sqrt{5}}{2}\)，\(0\)，\(\frac{\sqrt{5}}{2}\)，\(\sqrt{3}\)，\(2\sqrt{2}\)，用 \(\xi\) 表示坐标原点到 \(l\) 的距离，则随机变量 \(\xi\) 的数学期望 \(E(\xi) =\)\fillinblank{}.
\end{problem}

\begin{answer}
\(\frac47\)
\end{answer}

\begin{solution}
斜率为 \(m\) 的直线到原点的距离为 \(1/\sqrt{1+m^2}\). 七个斜率对应的距离之和为 \(1+2\left(\frac23+\frac12+\frac13\right)=4\)，故 \(E(\xi)=\frac47\).
\end{solution}

\begin{problem}
已知在 \(\triangle ABC\) 中，\(\angle ACB = 90^{\circ}\)，\(BC = 3\)，\(AC = 4\)，\(P\) 是 \(AB\) 上的点，则点 \(P\) 到 \(AC\)，\(BC\) 的距离乘积的最大值是\fillinblank{}.
\end{problem}

\begin{answer}
\(3\)
\end{answer}

\begin{solution}
设 \(P\) 将 \(AB\) 按比例 \(t:(1-t)\) 分割，则两距离为 \(3t\) 与 \(4(1-t)\)，乘积 \(12t(1-t)\leqslant3\)，最大值为 \(3\).
\end{solution}

\section{解答题}
\begin{problem}
设甲，乙，丙三台机器是否需要照顾相互之间没有影响，已知在某一小时内，甲，乙都需要照顾的概率为 \(0.05\)，甲，丙都需要照顾的概率为 \(0.1\)，乙，丙都需要照顾的概率为 \(0.125\).

\begin{enumerate}
\item 求甲，乙，丙每台机器在这个小时内需要照顾的概率分别是多少；
\item 计算这个小时内至少有一台需要照顾的概率.
\end{enumerate}
\end{problem}

\begin{solution}
（1）设三台机器需要照顾的概率为 \(p_1\)，\(p_2\)，\(p_3\). 独立性给出 \(p_1p_2=0.05\)、\(p_1p_3=0.1\)、\(p_2p_3=0.125\)，解得 \(p_1=0.2\)，\(p_2=0.25\)，\(p_3=0.5\).

（2）至少一台需要照顾的概率为 \(1-(0.8)(0.75)(0.5)=0.7\).
\end{solution}

\begin{enumerate}
\item 证明：\(AB \perp\) 平面 \(VAD\)；

\item 求面 \(VAD\) 与面 \(VDB\) 所成的二面角的大小.

（1）由 \(AB\perp AD\) 且两平面 \(VAD\) 与 \(ABCD\) 垂直，得 \(AB\perp\) 平面 \(VAD\).

\begin{problem}
\(\triangle ABC\) 中，内角 \(A\)，\(B\)，\(C\) 的对边分别为 \(a\)，\(b\)，\(c\)，已知 \(a\)，\(b\)，\(c\) 成等比数列，\(\cos B = \frac{3}{4}\).

\begin{enumerate}
\item 求 \( \cot A + \cot C \) 的值；
\item 设 \(\overrightarrow{BA} \cdot \overrightarrow{BC} = \frac{3}{2}\)，求 \(a + c\) 的值.
\end{enumerate}
\end{problem}

\begin{solution}
（1）由 \(\cot A+\cot C=\frac{b^2}{2\Delta}\)，且 \(b^2=ac\)、\(\Delta=\frac12ac\sin B\)，得 \(\cot A+\cot C=\frac1{\sin B}=\frac4{\sqrt7}\).

（2）由 \(\overrightarrow{BA}\cdot\overrightarrow{BC}=ac\cos B\)，得 \(ac=2\). 由等比条件和 \(\cos B=3/4\) 得 \(a:c=2:1\) 或 \(1:2\)，故 \(a+c=3\).
\end{solution}

\begin{problem}
在等差数列 \(\{a_{n}\}\) 中，公差 \(d \neq 0\)，\(a_{2}\) 是 \(a_{1}\) 与 \(a_{4}\) 的等差中项. 已知数列 \(a_{1}\)，\(a_{3}\)，\(a_{k_1}\)，\(a_{k_2}\)，\(\cdots\)，\(a_{k_n}\)，\(\cdots\) 成等比数列，求数列 \(\{k_{n}\}\) 的通项 \(k_{n}\).
\end{problem}

\begin{solution}
等差数列给出 \(a_2=a_1+d\)、\(a_4=a_1+3d\). 若 \(a_2\) 是 \(a_1\)，\(a_4\) 的等差中项，则 \(2(a_1+d)=a_1+a_1+3d\)，从而 \(d=0\)，与题设 \(d\ne0\) 矛盾. 因此题设条件不相容，不存在这样的数列.
\end{solution}

\begin{problem}
设 \(A(x_{1}, y_{1})\)，\(B(x_{2}, y_{2})\) 两点在抛物线 \(y = 2x^{2}\) 上，\(l\) 是 \(AB\) 的垂直平分线.

\begin{enumerate}
\item 当且仅当 \(x_{1} + x_{2}\) 取何值时，直线 \(l\) 经过抛物线的焦点 \(F\)？证明你的结论；
\item 当直线 \(l\) 的斜率为 2 时，求 \(l\) 在 \(y\) 轴上截距的取值范围.
\end{enumerate}
\end{problem}

\begin{solution}
（1）抛物线焦点为 \(F=(0,\frac18)\). 由 \(FA=FB\) 化简得 \((x_1^2-x_2^2)\{4(x_1^2+x_2^2)+\frac12\}=0\)，故且仅当 \(x_1+x_2=0\).

（2）当 \(k_l=2\) 时弦 \(AB\) 斜率为 \(-\frac12=2(x_1+x_2)\)，故 \(x_1+x_2=-\frac14\). 设 \(p=x_1x_2<\frac1{64}\)，则截距 \(b=\frac5{16}-2p>\frac9{32}\)，取值范围为 \((\frac9{32},+\infty)\).
\end{solution}

\begin{problem}
已知函数 \(f(x) = \frac{4x^2 - 7}{2 - x}\)，\(x \in [0,1]\).

\begin{enumerate}
\item 求 \( f(x) \) 的单调区间和值域.
\item 设 \(a \geqslant 1\)，函数 \(g(x) = x^3 - 3a^2x - 2a\)，\(x \in [0,1]\)，若对于任意 \(x_1 \in [0,1]\)，总存在 \(x_0 \in [0,1]\)，使得 \(g(x_0) = f(x_1)\) 成立，求 \(a\) 的取值范围.
\end{enumerate}
\end{problem}

\begin{solution}
（1）\(f'(x)=-4+\frac9{(2-x)^2}\)，故在 \([0,\frac12]\) 递减、\([\frac12,1]\) 递增，且 \(f(\frac12)=-4\)，\(f(1)=-3\)，值域为 \([-4,-3]\).

（2）\(g'(x)=3x^2-3a^2\leqslant0\)，故 \(g([0,1])=[1-3a^2-2a,-2a]\). 要覆盖 \([-4,-3]\)，需 \(1-3a^2-2a\leqslant-4\) 且 \(-2a\geqslant-3\)，结合 \(a\geqslant1\) 得 \(1\leqslant a\leqslant\frac32\).
\end{solution}
