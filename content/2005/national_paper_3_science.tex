\section{单选题}
\begin{problem}
已知 \(\alpha\) 为第三象限角，则 \(\frac{\alpha}{2}\) 所在的象限是（\(\quad\)）

\choices
    {第一或第二象限}
    {第二或第三象限}
    {第一或第三象限}
    {第二或第四象限}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
第三象限角满足 \(\pi+2k\pi<\alpha<\dfrac{3\pi}{2}+2k\pi\)，其中 \(k\in\Z\). 两边同除以 \(2\)，得 \(\dfrac{\pi}{2}+k\pi<\dfrac{\alpha}{2}<\dfrac{3\pi}{4}+k\pi\). 当 \(k\) 为偶数时，\(\dfrac{\alpha}{2}\) 在第二象限；当 \(k\) 为奇数时，\(\dfrac{\alpha}{2}\) 在第四象限. 故选 D.
\end{solution}

\begin{problem}
已知过点 \(A(-2, m)\) 和 \(B(m, 4)\) 的直线与直线 \(2x + y - 1 = 0\) 平行，则 \(m\) 的值为（\(\quad\)）

\choices
    {\(0\)}
    {\(-8\)}
    {\(2\)}
    {\(10\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
直线 \(2x+y-1=0\) 的斜率为 \(-2\). 当 \(m\neq-2\) 时，直线 \(AB\) 的斜率为 \(\dfrac{4-m}{m+2}\)，所以 \(\dfrac{4-m}{m+2}=-2\)，即 \(4-m=-2m-4\)，解得 \(m=-8\). 当 \(m=-2\) 时，点 \(A\)，\(B\) 的横坐标相同，直线 \(AB\) 为竖直直线，不可能与斜率为 \(-2\) 的直线平行. 因此 \(m=-8\)，故选 B.
\end{solution}

\begin{problem}
在 \((x - 1)(x + 1)^8\) 的展开式中 \(x^5\) 的系数是（\(\quad\)）

\choices
    {\(-14\)}
    {\(14\)}
    {\(-28\)}
    {\(28\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
由 \((x-1)(x+1)^8=x(x+1)^8-(x+1)^8\)，展开式中 \(x^5\) 的系数为 \(\mathrm C_8^4-\mathrm C_8^5=70-56=14\). 故选 B.
\end{solution}

\begin{problem}
设三棱柱 \(ABC - A_{1}B_{1}C_{1}\) 的体积为 \(V\)，\(P\)，\(Q\) 分别是侧棱 \(AA_{1}\)，\(CC_{1}\) 上的点，且 \(PA = QC_{1}\)，则四棱锥 \(B - APQC\) 的体积为（\(\quad\)）

\choices
    {\(\frac{1}{6} V\)}
    {\(\frac{1}{4} V\)}
    {\(\frac{1}{3} V\)}
    {\(\frac{1}{2} V\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
设 \(AA_1=CC_1=h\)，并令 \(t=\dfrac{PA}{h}=\dfrac{QC_1}{h}\)，则 \(0\leqslant t\leqslant1\). 取 \(A\) 为原点，记 \(\bs u=\overrightarrow{AB}\)，\(\bs v=\overrightarrow{AC}\)，\(\bs w=\overrightarrow{AA_1}\)，则 \(B=\bs u\)，\(C=\bs v\)，\(P=t\bs w\)，\(Q=\bs v+(1-t)\bs w\). 将该四棱锥分成三棱锥 \(B-APQ\) 和 \(B-AQC\)，其体积分别为
\[
V_1=\frac{1}{6}\left|\det(\bs u,t\bs w,\bs v+(1-t)\bs w)\right|=\frac{t}{6}\left|\det(\bs u,\bs w,\bs v)\right|
\]
\[
V_2=\frac{1}{6}\left|\det(\bs u,\bs v+(1-t)\bs w,\bs v)\right|=\frac{1-t}{6}\left|\det(\bs u,\bs w,\bs v)\right|
\]
因此四棱锥的体积为 \(V_1+V_2=\dfrac{1}{6}|\det(\bs u,\bs v,\bs w)|\). 而三棱柱的体积为 \(V=\dfrac{1}{2}|\det(\bs u,\bs v,\bs w)|\)，所以四棱锥的体积为 \(\dfrac{1}{3}V\)，故选 C.
\end{solution}

\begin{problem}
\(\displaystyle\lim_{x\to 1}\left(\frac{1}{x^2 - 3x + 2} -\frac{2}{x^2 - 4x + 3}\right) =\)（\(\quad\)）

\choices
    {\(-\frac{1}{2}\)}
    {\(\frac{1}{2}\)}
    {\(-\frac{1}{6}\)}
    {\(\frac{1}{6}\)}
\end{problem}

\begin{answer}
A
\end{answer}

\begin{solution}
因式分解得 \(x^2-3x+2=(x-1)(x-2)\)，\(x^2-4x+3=(x-1)(x-3)\). 因而
\[
\frac{1}{(x-1)(x-2)}-\frac{2}{(x-1)(x-3)}
=\frac{1}{x-1}\left(\frac{1}{x-2}-\frac{2}{x-3}\right)
\]
通分并约去 \(x-1\)，得
\[
\frac{1-x}{(x-1)(x-2)(x-3)}
=-\frac{1}{(x-2)(x-3)}
\]
所以原极限为 \(-\dfrac{1}{(1-2)(1-3)}=-\dfrac{1}{2}\)，故选 A.
\end{solution}

\begin{problem}
若 \(a = \frac{\ln 2}{2}\)，\(b = \frac{\ln 3}{3}\)，\(c = \frac{\ln 5}{5}\)，则（\(\quad\)）

\choices
    {\(a < b < c\)}
    {\(c < b < a\)}
    {\(c < a < b\)}
    {\(b < a < c\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
比较三个数可直接利用对数函数的单调性. 因为 \(\ln25<\ln32\)，所以 \(2\ln5<5\ln2\)，即 \(\dfrac{\ln5}{5}<\dfrac{\ln2}{2}\). 又因为 \(\ln8<\ln9\)，所以 \(3\ln2<2\ln3\)，即 \(\dfrac{\ln2}{2}<\dfrac{\ln3}{3}\). 因此 \(c<a<b\)，故选 C.
\end{solution}

\begin{problem}
设 \(0 \leqslant x \leqslant 2\pi\)，且 \(\sqrt{1 - \sin 2x} = \sin x - \cos x\)，则（\(\quad\)）

\choices
    {\(0 \leqslant x \leqslant \pi\)}
    {\(\frac{\pi}{4} \leqslant x \leqslant \frac{7\pi}{4}\)}
    {\(\frac{\pi}{4} \leqslant x \leqslant \frac{5\pi}{4}\)}
    {\(\frac{\pi}{2} \leqslant x \leqslant \frac{3\pi}{2}\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
由 \(1-\sin2x=1-2\sin x\cos x=(\sin x-\cos x)^2\)，原方程化为 \(|\sin x-\cos x|=\sin x-\cos x\)，这等价于 \(\sin x-\cos x\geqslant0\). 又 \(\sin x-\cos x=\sqrt{2}\sin\left(x-\dfrac{\pi}{4}\right)\)，在 \(0\leqslant x\leqslant2\pi\) 上解得 \(\dfrac{\pi}{4}\leqslant x\leqslant\dfrac{5\pi}{4}\). 故选 C.
\end{solution}

\begin{problem}
\(\frac{2\sin 2\alpha}{1 + \cos 2\alpha} \cdot \frac{\cos^2\alpha}{\cos 2\alpha} =\)（\(\quad\)）

\choices
    {\(\tan \alpha\)}
    {\(\tan 2\alpha\)}
    {\(1\)}
    {\(\frac{1}{2}\)}
\end{problem}

\begin{answer}
B
\end{answer}

\begin{solution}
在原式有意义的条件下，利用 \(1+\cos2\alpha=2\cos^2\alpha\) 和 \(\sin2\alpha=2\sin\alpha\cos\alpha\)，有
\[
\frac{2\sin2\alpha}{1+\cos2\alpha}\cdot\frac{\cos^2\alpha}{\cos2\alpha}
=2\tan\alpha\cdot\frac{\cos^2\alpha}{\cos2\alpha}
=\frac{\sin2\alpha}{\cos2\alpha}
=\tan2\alpha
\]
故选 B.
\end{solution}

\begin{problem}
已知双曲线 \(x^{2} - \frac{y^{2}}{2} = 1\) 的焦点为 \(F_{1}\)，\(F_{2}\)，点 \(M\) 在双曲线上且 \(\overrightarrow{MF_{1}} \cdot \overrightarrow{MF_{2}} = 0\)，则点 \(M\) 到 \(x\) 轴的距离为（\(\quad\)）

\choices
    {\(\frac{4}{3}\)}
    {\(\frac{5}{3}\)}
    {\(\frac{2\sqrt{3}}{3}\)}
    {\(\sqrt{3}\)}
\end{problem}

\begin{answer}
C
\end{answer}

\begin{solution}
双曲线可写为 \(\dfrac{x^2}{1^2}-\dfrac{y^2}{(\sqrt{2})^2}=1\)，所以焦点为 \(F_1(-\sqrt{3},0)\)，\(F_2(\sqrt{3},0)\). 设 \(M(x,y)\)，则
\[
\overrightarrow{MF_1}\cdot\overrightarrow{MF_2}=(-\sqrt{3}-x,-y)\cdot(\sqrt{3}-x,-y)=x^2+y^2-3=0
\]
结合双曲线方程 \(x^2-\dfrac{y^2}{2}=1\)，得 \(y^2=3-x^2\)，从而 \(x^2-\dfrac{3-x^2}{2}=1\)，即 \(x^2=\dfrac{5}{3}\)，\(y^2=\dfrac{4}{3}\). 因此 \(M\) 到 \(x\) 轴的距离为 \(|y|=\dfrac{2\sqrt{3}}{3}\)，故选 C.
\end{solution}

\begin{problem}
设椭圆的两个焦点分别为 \(F_{1}\)，\(F_{2}\)，过 \(F_{2}\) 作椭圆长轴的垂线交椭圆于点 \(P\)，若 \(\triangle F_{1}PF_{2}\) 为等腰直角三角形，则椭圆的离心率是（\(\quad\)）

\choices
    {\(\frac{\sqrt{2}}{2}\)}
    {\(\frac{\sqrt{2} - 1}{2}\)}
    {\(2 - \sqrt{2}\)}
    {\(\sqrt{2} - 1\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
设椭圆的半长轴为 \(a\)，半焦距为 \(c\)，离心率为 \(e=\dfrac{c}{a}\). 取长轴为 \(x\) 轴，令 \(F_1(-c,0)\)，\(F_2(c,0)\). 过 \(F_2\) 的长轴垂线与椭圆交于 \(P\)，所以 \(P=(c,y)\). 由椭圆方程 \(\dfrac{x^2}{a^2}+\dfrac{y^2}{a^2-c^2}=1\)，得 \(y^2=a^2(1-e^2)^2\)，故 \(F_2P=a(1-e^2)\). 又 \(F_1F_2=2c=2ae\)，且三角形 \(F_1PF_2\) 在 \(F_2\) 处为直角；它是等腰直角三角形，所以 \(a(1-e^2)=2ae\). 因为 \(a>0\)，得 \(e^2+2e-1=0\). 离心率为正，故 \(e=\sqrt{2}-1\)，故选 D.
\end{solution}

\begin{problem}
不共面的四个定点到平面 \(\alpha\) 的距离都相等，这样的平面 \(\alpha\) 共有（\(\quad\)）

\choices
    {\(3\text{ 个}\)}
    {\(4\text{ 个}\)}
    {\(6\text{ 个}\)}
    {\(7\text{ 个}\)}
\end{problem}

\begin{answer}
D
\end{answer}

\begin{solution}
设四个定点为 \(P_1\)，\(P_2\)，\(P_3\)，\(P_4\)，其位矢分别为 \(\bs p_1\)，\(\bs p_2\)，\(\bs p_3\)，\(\bs p_4\). 平面 \(\alpha\) 不可能经过这四点，否则四点共面. 设平面 \(\alpha\) 的单位法向量为 \(\bs n\)，四点到它的有向距离分别为 \(d_i\). 由题意 \(|d_1|=|d_2|=|d_3|=|d_4|=r>0\). 四个有向距离不能同号，否则四点都在同一个与 \(\alpha\) 平行的平面上而共面，因此它们的正负号把四个点分成两个非空部分.

反过来，对四个点作任意一个非平凡二分，令仿射函数 \(L(\bs x)=\bs n_0\cdot\bs x+c_0\) 在一部分点上的值为 \(1\)，在另一部分点上的值为 \(-1\). 因为四点不共面，它们仿射独立，所以 \(L\) 唯一；又因函数值不全相同，\(\bs n_0\neq\bs0\)，其零点集是唯一的平面，且该平面到四个点的距离均为 \(1/\lVert\bs n_0\rVert\). 把一个二分及其整体反号视为同一个平面，二分的个数为 \(\dfrac{2^4-2}{2}=7\). 故选 D.
\end{solution}

\section{填空题}
\begin{problem}
已知复数 \(z_0 = 3 + 2\i\)，复数 \(z\) 满足 \(z \cdot z_0 = 3z + z_0\)，则复数 \(z = \)\fillinblank{}.
\end{problem}

\begin{answer}
\(1-\frac{3}{2}\i\)
\end{answer}

\begin{solution}
由 \(z\cdot z_0=3z+z_0\) 得 \(z(z_0-3)=z_0\). 代入 \(z_0=3+2\i\)，有 \(2\i z=3+2\i\)，所以 \(z=\dfrac{3+2\i}{2\i}=1-\dfrac{3}{2}\i\).
\end{solution}

\begin{problem}
已知向量 \( \overrightarrow{OA} = (k, 12) \)，\( \overrightarrow{OB} = (4, 5) \)，\( \overrightarrow{OC} = (-k, 10) \)，且 \(A\)，\(B\)，\(C\) 三点共线，则 \(k=\)\fillinblank{}.
\end{problem}

\begin{answer}
\(-\frac{2}{3}\)
\end{answer}

\begin{solution}
由 \(\overrightarrow{AB}=(4-k,-7)\)，\(\overrightarrow{AC}=(-2k,-2)\)，且 \(A\)，\(B\)，\(C\) 三点共线，得两向量的行列式为零，即 \((4-k)(-2)-(-7)(-2k)=0\). 化简得 \(-8-12k=0\)，所以 \(k=-\dfrac{2}{3}\).
\end{solution}

\begin{problem}
设 \(l\) 为平面上过点 \((0,1)\) 的直线，\(l\) 的斜率等可能地取
\(-2\sqrt{2}\)，\(-\sqrt{3}\)，\(-\frac{\sqrt{5}}{2}\)，\(0\)，\(\frac{\sqrt{5}}{2}\)，\(\sqrt{3}\)，\(2\sqrt{2}\)，用 \(\xi\) 表示坐标原点到 \(l\) 的距离，则随机变量 \(\xi\) 的数学期望 \(E(\xi) =\)\fillinblank{}.
\end{problem}

\begin{answer}
\(\frac{4}{7}\)
\end{answer}

\begin{solution}
设直线 \(l\) 的斜率为 \(m\)，因其经过 \((0,1)\)，方程为 \(mx-y+1=0\). 原点到该直线的距离为 \(\xi=\dfrac{1}{\sqrt{m^2+1}}\). 当 \(m=\pm2\sqrt{2}\) 时，\(\xi=\dfrac13\)；当 \(m=\pm\sqrt{3}\) 时，\(\xi=\dfrac12\)；当 \(m=\pm\dfrac{\sqrt{5}}2\) 时，\(\xi=\dfrac23\)；当 \(m=0\) 时，\(\xi=1\). 七个斜率等可能，所以
\[
E(\xi)=\frac{2\cdot\frac13+2\cdot\frac12+2\cdot\frac23+1}{7}=\frac47
\]
\end{solution}

\begin{problem}
已知在 \(\triangle ABC\) 中，\(\angle ACB = 90^{\circ}\)，\(BC = 3\)，\(AC = 4\)，\(P\) 是 \(AB\) 上的点，则点 \(P\) 到 \(AC\)，\(BC\) 的距离乘积的最大值是\fillinblank{}.
\end{problem}

\begin{answer}
3
\end{answer}

\begin{solution}
以 \(C\) 为原点，令 \(AC\) 在 \(x\) 轴正方向、\(BC\) 在 \(y\) 轴正方向，则 \(A=(4,0)\)，\(B=(0,3)\). 设 \(P=(x,y)\)，其中 \(0\leqslant x\leqslant4\)，\(0\leqslant y\leqslant3\). 直线 \(AB\) 的方程为 \(3x+4y=12\)，且点 \(P\) 到 \(AC\)，\(BC\) 的距离分别为 \(y\)，\(x\)，所以距离乘积为 \(xy=\dfrac34x(4-x)\). 由于 \((x-2)^2\geqslant0\)，有 \(x(4-x)=4-(x-2)^2\leqslant4\)，等号在 \(x=2\)，即 \(y=\dfrac32\) 时成立. 因此距离乘积的最大值为 \(\dfrac34\times4=3\).
\end{solution}

\section{解答题}
\begin{problem}
设甲，乙，丙三台机器是否需要照顾相互之间没有影响. 已知在某一小时内，甲，乙都需要照顾的概率为 \(0.05\)，甲，丙都需要照顾的概率为 \(0.1\)，乙，丙都需要照顾的概率为 \(0.125\).

\begin{enumerate}
\item 求甲，乙，丙每台机器在这个小时内需要照顾的概率分别是多少；
\item 计算这个小时内至少有一台需要照顾的概率.
\end{enumerate}
\end{problem}

\begin{solution}
设甲、乙、丙在这一小时内需要照顾的概率分别为 \(p_{1}\)，\(p_{2}\)，\(p_{3}\).

\begin{enumerate}
\item 由相互独立及题设条件，得
\[
\begin{cases}
p_{1}p_{2}=0.05,\\
p_{1}p_{3}=0.1,\\
p_{2}p_{3}=0.125
\end{cases}
\]
将前两个等式相乘，再除以第三个等式，得
\[
p_{1}^{2}=\frac{0.05\times0.1}{0.125}=0.04
\]
因为概率非负，所以 \(p_{1}=0.2\). 于是 \(p_{2}=0.05\div0.2=0.25\)，\(p_{3}=0.1\div0.2=0.5\).
因此，甲、乙、丙每台机器需要照顾的概率分别为 \(0.2\)，\(0.25\)，\(0.5\).

\item 三台机器都不需要照顾的概率为
\[
(1-p_{1})(1-p_{2})(1-p_{3})=0.8\times0.75\times0.5=0.3
\]
所以这个小时内至少有一台需要照顾的概率为
\[
1-0.3=0.7
\]
\end{enumerate}
\end{solution}

V=\left(0,\dfrac{s}{2},\dfrac{\sqrt{3}s}{2}\right)

\cos\theta=\frac{|\bs n_{1}\cdot\bs n_{2}|}{|\bs n_{1}|\,|\bs n_{2}|}=\frac{\sqrt{3}}{\sqrt{7}}=\frac{\sqrt{21}}{7}

\begin{problem}
\(\triangle ABC\) 中，内角 \(A\)，\(B\)，\(C\) 的对边分别为 \(a\)，\(b\)，\(c\)，已知 \(a\)，\(b\)，\(c\) 成等比数列，\(\cos B = \frac{3}{4}\).

\begin{enumerate}
\item 求 \( \cot A + \cot C \) 的值；
\item 设 \(\overrightarrow{BA} \cdot \overrightarrow{BC} = \frac{3}{2}\)，求 \(a + c\) 的值.
\end{enumerate}
\end{problem}

\begin{solution}
因为 \(a\)，\(b\)，\(c\) 成等比数列，所以 \(b^{2}=ac\). 又 \(B\) 是三角形内角，故
\[
\sin B=\sqrt{1-\cos^{2}B}=\dfrac{\sqrt{7}}{4}
\]

\begin{enumerate}
\item 利用 \(A+B+C=\pi\)，有
\[
\cot A+\cot C=\frac{\cos A\sin C+\sin A\cos C}{\sin A\sin C}=\frac{\sin B}{\sin A\sin C}
\]
设三角形外接圆半径为 \(R\). 由正弦定理和 \(ac=b^{2}\)，得
\[
\frac{\sin B}{\sin A\sin C}
=\frac{b/(2R)}{ac/(2R)^{2}}
=\frac{2R}{b}
=\frac{1}{\sin B}
=\frac{4\sqrt{7}}{7}
\]
所以 \(\cot A+\cot C=\dfrac{4\sqrt{7}}{7}\).

\item 由向量 \(\overrightarrow{BA}\) 与 \(\overrightarrow{BC}\) 的夹角为 \(B\)，得
\[
\overrightarrow{BA}\cdot\overrightarrow{BC}=ac\cos B
\]
由题设 \(ac\cos B=\dfrac32\)，且 \(\cos B=\dfrac34\)，所以 \(ac=2\). 结合 \(b^{2}=ac\)，得 \(b^{2}=2\). 再由余弦定理，
\[
a^{2}+c^{2}=b^{2}+2ac\cos B=2+3=5
\]
于是
\[
(a+c)^{2}=a^{2}+c^{2}+2ac=5+4=9
\]
因为边长为正数，所以 \(a+c=3\).
\end{enumerate}
\end{solution}

\begin{problem}
在等差数列 \(\{a_{n}\}\) 中，公差 \(d \neq 0\)，\(a_{2}\) 是 \(a_{1}\) 与 \(a_{4}\) 的等差中项. 已知数列 \(a_{1}\)，\(a_{3}\)，\(a_{k_1}\)，\(a_{k_2}\)，\(\cdots\)，\(a_{k_n}\)，\(\cdots\) 成等比数列，求数列 \(\{k_{n}\}\) 的通项 \(k_{n}\).
\end{problem}

\begin{solution}
设 \(a_{1}=u\). 由等差数列的通项公式，\(a_{2}=u+d\)，\(a_{4}=u+3d\). 由 \(a_{2}\) 是 \(a_{1}\) 与 \(a_{4}\) 的等差中项，得
\[
u+d=\frac{a_{1}+a_{4}}{2}=u+\frac{3d}{2}
\]
即 \(d=0\)，这与题设 \(d\neq0\) 矛盾. 因此，不存在满足题设条件的等差数列，后续所述等比数列及数列 \(\{k_{n}\}\) 也就不存在，故按原题字面无法得到 \(k_{n}\) 的通项.
\end{solution}

\begin{problem}
设 \(A(x_{1},y_{1})\)，\(B(x_{2},y_{2})\) 两点在抛物线 \(y=2x^{2}\) 上，\(l\) 是 \(AB\) 的垂直平分线.

\begin{enumerate}
\item 当且仅当 \(x_{1} + x_{2}\) 取何值时，直线 \(l\) 经过抛物线的焦点 \(F\)？证明你的结论；
\item 当直线 \(l\) 的斜率为 2 时，求 \(l\) 在 \(y\) 轴上截距的取值范围.
\end{enumerate}
\end{problem}

\begin{solution}
由抛物线方程可知，\(A=(x_{1},2x_{1}^{2})\)，\(B=(x_{2},2x_{2}^{2})\). 抛物线 \(y=2x^{2}\) 可写为 \(x^{2}=\dfrac{1}{2}y\)，所以其焦点为 \(F\left(0,\dfrac{1}{8}\right)\).

\begin{enumerate}
\item 直线 \(l\) 是线段 \(AB\) 的垂直平分线，因此 \(l\) 经过 \(F\) 当且仅当 \(FA=FB\). 计算得
\[
FA^{2}-FB^{2}=x_{1}^{2}-x_{2}^{2}+\left(2x_{1}^{2}-\dfrac{1}{8}\right)^{2}-\left(2x_{2}^{2}-\dfrac{1}{8}\right)^{2}=(x_{1}^{2}-x_{2}^{2})\left(4x_{1}^{2}+4x_{2}^{2}+\dfrac{1}{2}\right)
\]
由于 \(4x_{1}^{2}+4x_{2}^{2}+\dfrac{1}{2}>0\)，故 \(FA=FB\) 当且仅当 \(x_{1}^{2}=x_{2}^{2}\). 点 \(A\)，\(B\) 不重合，所以 \(x_{1}\neq x_{2}\)，于是 \(x_{1}=-x_{2}\)，即
\[
x_{1}+x_{2}=0
\]
反之，当 \(x_{1}+x_{2}=0\) 时，必有 \(FA=FB\)，故 \(F\) 在 \(AB\) 的垂直平分线 \(l\) 上. 因此，当且仅当 \(x_{1}+x_{2}=0\) 时，直线 \(l\) 经过焦点 \(F\).

\item 设 \(s=x_{1}+x_{2}\). 当 \(x_{1}\neq x_{2}\) 时，直线 \(AB\) 的斜率为 \(2s\)，所以直线 \(l\) 的斜率为 \(-\dfrac{1}{2s}\). 由题设斜率为 \(2\)，得 \(s=-\dfrac{1}{4}\).

线段 \(AB\) 的中点为 \(M\left(\dfrac{s}{2},x_{1}^{2}+x_{2}^{2}\right)\). 设直线 \(l\) 在 \(y\) 轴上的截距为 \(t\)，则
\[
t=x_{1}^{2}+x_{2}^{2}-2\cdot\dfrac{s}{2}=s^{2}-2x_{1}x_{2}-s=\dfrac{5}{16}-2x_{1}x_{2}
\]
又因为 \(x_{1}\neq x_{2}\)，有
\[
x_{1}x_{2}<\dfrac{(x_{1}+x_{2})^{2}}{4}=\dfrac{1}{64}
\]
所以 \(t>\dfrac{9}{32}\). 反之，任意给定 \(t>\dfrac{9}{32}\)，令 \(x_{1}+x_{2}=-\dfrac14\)，\(x_{1}x_{2}=\dfrac{5}{32}-\dfrac{t}{2}\)，则关于 \(x\) 的二次方程有两个不相等的实根，且由上式得到该截距为 \(t\). 因此截距的取值范围为
\[
\left(\dfrac{9}{32},+\infty\right)
\]
\end{enumerate}
\end{solution}

\begin{problem}
已知函数 \(f(x) = \frac{4x^2 - 7}{2 - x}\)，\(x \in [0,1]\).

\begin{enumerate}
\item 求 \( f(x) \) 的单调区间和值域.
\item 设 \(a \geqslant 1\)，函数 \(g(x) = x^3 - 3a^2x - 2a\)，\(x \in [0,1]\)，若对于任意 \(x_1 \in [0,1]\)，总存在 \(x_0 \in [0,1]\)，使得 \(g(x_0) = f(x_1)\) 成立，求 \(a\) 的取值范围.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item 在区间 \([0,1]\) 上，\(2-x>0\)，且
\[
f'(x)=\frac{-4x^{2}+16x-7}{(2-x)^{2}}=-\frac{4\left(x-\dfrac12\right)\left(x-\dfrac72\right)}{(2-x)^{2}}
\]
因此，\(f(x)\) 在 \(\left[0,\dfrac12\right]\) 上单调递减，在 \(\left[\dfrac12,1\right]\) 上单调递增. 又
\(f(0)=-\dfrac72\)，\(f\left(\dfrac12\right)=-4\)，\(f(1)=-3\)，
故 \(f(x)\) 的值域为 \([-4,-3]\).

\item 因为 \(a\geqslant1\) 且 \(x\in[0,1]\)，所以
\[
g'(x)=3x^{2}-3a^{2}\leqslant0
\]
于是 \(g(x)\) 在 \([0,1]\) 上单调递减，值域为
\[
\left[g(1),g(0)\right]=\left[1-3a^{2}-2a,-2a\right]
\]
题设“对任意 \(x_{1}\in[0,1]\)，总存在 \(x_{0}\in[0,1]\)，使得 \(g(x_{0})=f(x_{1})\)”等价于
\[
[-4,-3]\subseteq\left[1-3a^{2}-2a,-2a\right]
\]
于是需要同时满足
\(1-3a^{2}-2a\leqslant-4\) 和 \(-2a\geqslant-3\)
前一个不等式化为 \((a-1)(3a+5)\geqslant0\)，在 \(a\geqslant1\) 时恒成立；后一个不等式给出 \(a\leqslant\dfrac32\). 因此
\[
a\in\left[1,\dfrac32\right]
\]
\end{enumerate}
\end{solution}
