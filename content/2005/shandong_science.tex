\section{单选题}
\begin{problem}
\(\dfrac{1-\i}{(1+\i)^2}+\dfrac{1+\i}{(1-\i)^2}=\)（\quad）

\choices
    {\(\i\)}
    {\(-\i\)}
    {\(1\)}
    {\(-1\)}
\end{problem}
\answer{D}

\begin{solution}
因为 \((1+\i)^2=2\i\)，\((1-\i)^2=-2\i\)，所以
\(\dfrac{1-\i}{(1+\i)^2}+\dfrac{1+\i}{(1-\i)^2}=\dfrac{1-\i}{2\i}+\dfrac{1+\i}{-2\i}=-1\). 故选 D.
\end{solution}

\begin{problem}
函数 \(y=\dfrac{1-x}{x}\)（\(x\neq 0\)）的反函数图象大致是（\quad）

\choices
    {\choicebitmap[width=0.15\paperwidth]{img/2005/shandong_science/q2_optA_nolabel.png}}
    {\choicebitmap[width=0.15\paperwidth]{img/2005/shandong_science/q2_optB_nolabel.png}}
    {\choicebitmap[width=0.15\paperwidth]{img/2005/shandong_science/q2_optC_nolabel.png}}
    {\choicebitmap[width=0.15\paperwidth]{img/2005/shandong_science/q2_optD_nolabel.png}}
\end{problem}
\answer{B}

\begin{solution}
由 \(y=\dfrac{1-x}{x}=\dfrac{1}{x}-1\)，交换 \(x\)，\(y\) 得 \(x=\dfrac{1}{y}-1\)，从而反函数为 \(y=\dfrac{1}{x+1}\)，定义域为 \(x\neq-1\). 其图象的渐近线为 \(x=-1\) 和 \(y=0\)，且当 \(x>-1\) 时 \(y>0\)，当 \(x<-1\) 时 \(y<0\)，与选项 B 的图象一致. 故选 B.
\end{solution}

\begin{problem}
已知函数 \(y=\sin\left(x-\dfrac{\pi}{12}\right)\cos\left(x-\dfrac{\pi}{12}\right)\)，则下列判断正确的是（\quad）

\choices
    {此函数的最小周期为 \(2\pi\)，其图象的一个对称中心是 \(\left(\dfrac{\pi}{12},0\right)\)}
    {此函数的最小周期为 \(\pi\)，其图象的一个对称中心是 \(\left(\dfrac{\pi}{12},0\right)\)}
    {此函数的最小周期为 \(2\pi\)，其图象的一个对称中心是 \(\left(\dfrac{\pi}{6},0\right)\)}
    {此函数的最小周期为 \(\pi\)，其图象的一个对称中心是 \(\left(\dfrac{\pi}{6},0\right)\)}
\end{problem}
\answer{B}

\begin{solution}
令 \(u=x-\dfrac{\pi}{12}\)，则
\(y=\sin u\cos u=\dfrac{1}{2}\sin 2u=\dfrac{1}{2}\sin\left(2x-\dfrac{\pi}{6}\right)\). 因此最小正周期为 \(\pi\). 当 \(2x-\dfrac{\pi}{6}=k\pi\) 时函数值为零，取 \(k=0\) 得图象经过 \(\left(\dfrac{\pi}{12},0\right)\)，且正弦图象在该点中心对称. 故选 B.
\end{solution}

\begin{problem}
下列函数既是奇函数，又在区间 \([-1,1]\) 上单调递减的是（\quad）

\choices
    {\(f(x)=\sin x\)}
    {\(f(x)=-|x+1|\)}
    {\(f(x)=\dfrac{1}{2}(a^x+a^{-x})\)}
    {\(f(x)=\ln\dfrac{2-x}{2+x}\)}
\end{problem}
\answer{D}

\begin{solution}
选项 A 中，\(f(x)=\sin x\) 在 \([-1,1]\) 上的导数 \(f'(x)=\cos x>0\)，这是因为 \([-1,1]\subset(-\frac{\pi}{2},\frac{\pi}{2})\)，故单调递增，不合题意. 选项 B 中 \(f(0)=-1\)，而 \(f(0)\) 不满足奇函数必须有的 \(f(0)=0\)，故不是奇函数. 选项 C 满足 \(f(-x)=f(x)\)，是偶函数而不是奇函数.

选项 D 的定义域包含 \([-1,1]\)，并且
\(f(-x)=\ln\dfrac{2+x}{2-x}=-\ln\dfrac{2-x}{2+x}=-f(x)\)，所以它是奇函数；又
\(f'(x)=-\dfrac{1}{2-x}-\dfrac{1}{2+x}=-\dfrac{4}{4-x^2}<0\)，故在 \([-1,1]\) 上单调递减. 故选 D.
\end{solution}

\begin{problem}
如果 \(\left(3x-\dfrac{1}{\sqrt[3]{x^2}}\right)^n\) 的展开式中各项系数之和为 128，则展开式中 \(\dfrac{1}{x^3}\) 的系数是（\quad）

\choices
    {\(7\)}
    {\(-7\)}
    {\(21\)}
    {\(-21\)}
\end{problem}
\answer{C}

\begin{solution}
令 \(x=1\)，展开式各项系数之和为 \((3-1)^n=2^n=128=2^7\)，所以 \(n=7\). 展开式通项中取第二项 \(k\) 次为
\(\mathrm C_7^k(3x)^{7-k}\left(-x^{-\frac{2}{3}}\right)^k\)，其中 \(x\) 的指数为 \(7-k-\dfrac{2k}{3}=7-\dfrac{5k}{3}\). 令其等于 \(-3\)，得 \(k=6\). 因此 \(\dfrac{1}{x^3}\) 的系数为 \(\mathrm C_7^6\cdot3\cdot(-1)^6=21\). 故选 C.
\end{solution}

\begin{problem}
函数 \(f(x)=\begin{cases}
\sin(\pi x^2), & -1<x<0,\\
\e^{x-1}, & x\geqslant 0.
\end{cases}\)，若 \(f(1)+f(a)=2\)，则 \(a\) 的所有可能值为（\quad）

\choices
    {\(1\)}
    {\(-\dfrac{\sqrt{2}}{2}\)}
    {\(1\)，\(-\dfrac{\sqrt{2}}{2}\)}
    {\(1\)，\(\dfrac{\sqrt{2}}{2}\)}
\end{problem}
\answer{C}

\begin{solution}
因为 \(f(1)=\e^{0}=1\)，所以 \(f(a)=1\). 当 \(a\geqslant0\) 时，\(f(a)=\e^{a-1}=1\)，解得 \(a=1\). 当 \(-1<a<0\) 时，\(f(a)=\sin(\pi a^2)=1\)，故 \(\pi a^2=\dfrac{\pi}{2}+2k\pi\). 由 \(0<a^2<1\) 可知只有 \(k=0\)，于是 \(a^2=\dfrac{1}{2}\)，从而 \(a=-\dfrac{\sqrt{2}}{2}\). 因此所有可能值为 \(1\) 和 \(-\dfrac{\sqrt{2}}{2}\). 故选 C.
\end{solution}

\begin{problem}
已知向量 \(\bs{a}\)，\(\bs{b}\)，且 \(\overrightarrow{AB}=\bs{a}+2\bs{b}\)，\(\overrightarrow{BC}=-5\bs{a}+6\bs{b}\)，\(\overrightarrow{CD}=7\bs{a}-2\bs{b}\)，则一定共线的三点是（\quad）

\choices
    {\(A,B,D\)}
    {\(A\)，\(B\)，\(C\)}
    {\(B\)，\(C\)，\(D\)}
    {\(A\)，\(C\)，\(D\)}
\end{problem}
\answer{A}

\begin{solution}
由向量相加，
\(\overrightarrow{AD}=\overrightarrow{AB}+\overrightarrow{BC}+\overrightarrow{CD}=(\bs a+2\bs b)+(-5\bs a+6\bs b)+(7\bs a-2\bs b)=3\bs a+6\bs b=3\overrightarrow{AB}\).
所以 \(\overrightarrow{AD}\) 与 \(\overrightarrow{AB}\) 平行，且二者有公共点 \(A\)，故 \(A\)，\(B\)，\(D\) 三点共线. 故选 A.
\end{solution}

\begin{problem}
设地球的半径为 \(R\)，若甲地位于北纬 \(45^{\circ}\) 东经 \(120^{\circ}\)，乙地位于南纬 \(75^{\circ}\) 东经 \(120^{\circ}\)，则甲，乙两地的球面距离为（\quad）

\choices
    {\(\sqrt{3}R\)}
    {\(\dfrac{\pi}{6}R\)}
    {\(\dfrac{5\pi}{6}R\)}
    {\(\dfrac{2\pi}{3}R\)}
\end{problem}
\answer{D}

\begin{solution}
甲、乙两地的经度相同，因此两地在同一条经线上. 它们对应的球心角等于纬度的差，即 \(45^{\circ}+75^{\circ}=120^{\circ}=\dfrac{2\pi}{3}\). 球面距离为半径乘以所对的圆心角，所以为 \(R\cdot\dfrac{2\pi}{3}=\dfrac{2\pi}{3}R\). 故选 D.
\end{solution}

\begin{problem}
\(10\) 张奖券中只有 \(3\) 张有奖，\(5\) 个人购买，每人 \(1\) 张，至少有 \(1\) 人中奖的概率是（\quad）

\choices
    {\(\dfrac{3}{10}\)}
    {\(\dfrac{1}{12}\)}
    {\(\dfrac{1}{2}\)}
    {\(\dfrac{11}{12}\)}
\end{problem}
\answer{D}

\begin{solution}
至少有 1 人中奖的对立事件是 5 人买到的奖券全部没有奖. 总的取法数为 \(\mathrm C_{10}^{5}\)，全部为无奖券的取法数为 \(\mathrm C_{7}^{5}\)，所以所求概率为
\(1-\dfrac{\mathrm C_{7}^{5}}{\mathrm C_{10}^{5}}=1-\dfrac{21}{252}=\dfrac{11}{12}\). 故选 D.
\end{solution}

\begin{problem}
设集合 \(A\)，\(B\) 是全集 \(U\) 的两个子集，则 \(A \subsetneq B\) 是 \((\complement_U A) \cup B = U\) 的（\quad）

\choices
    {充分不必要条件}
    {必要不充分条件}
    {充要条件}
    {既不充分也不必要条件}
\end{problem}
\answer{A}

\begin{solution}
\((\complement_U A)\cup B=U\) 等价于 \(A\subseteq B\)：确实，若 \(x\in A\)，要使 \(x\) 属于并集 \(U\)，只能有 \(x\in B\)；反过来 \(A\subseteq B\) 时，\(A\) 中元素属于 \(B\)，其余元素属于 \(\complement_U A\)，并集就是 \(U\).

因此 \(A\subsetneq B\) 能推出该等式，是充分条件；但该等式只推出 \(A\subseteq B\)，允许 \(A=B\)，所以不是必要条件. 故选 A.
\end{solution}

\begin{problem}
\(0<a<1\)，下列不等式一定成立的是（\quad）

\choices
    {\(\left|\log_{1+a}(1-a)\right|+\left|\log_{1-a}(1+a)\right|>2\)}
    {\(\left|\log_{1+a}(1-a)\right|<\left|\log_{1-a}(1+a)\right|\)}
    {\(\left|\log_{1+a}(1-a)+\log_{1-a}(1+a)\right|<\left|\log_{1+a}(1-a)\right|+\left|\log_{1-a}(1+a)\right|\)}
    {\(\left|\log_{1+a}(1-a)-\log_{1-a}(1+a)\right|<\left|\log_{1+a}(1-a)\right|-\left|\log_{1-a}(1+a)\right|\)}
\end{problem}
\answer{A}

\begin{solution}
令 \(t=\dfrac{\ln(1+a)}{-\ln(1-a)}\). 由于 \(0<a<1\)，有
\(\dfrac{1}{1-a}>1+a\)，所以 \(-\ln(1-a)>\ln(1+a)>0\)，从而 \(0<t<1\). 于是
\(\left|\log_{1+a}(1-a)\right|=\dfrac{-\ln(1-a)}{\ln(1+a)}=\dfrac{1}{t}\)，
\(\left|\log_{1-a}(1+a)\right|=\dfrac{\ln(1+a)}{-\ln(1-a)}=t\).

由 \(t+\dfrac{1}{t}>2\)，选项 A 恒成立. 选项 B 与 \(t<1\) 给出的 \(\dfrac{1}{t}>t\) 相反；选项 C 中两项对数均为负，左边为 \(\dfrac{1}{t}+t\)，与右边相等；选项 D 左边为 \(\dfrac{1}{t}-t\)，右边也为 \(\dfrac{1}{t}-t\)，同样是等号，均不成立. 故选 A.
\end{solution}

\begin{problem}
设直线 \(l\colon 2x+y+2=0\) 关于原点对称的直线为 \(l'\)，若 \(l'\) 与椭圆 \(x^2+\dfrac{y^2}{4}=1\) 的交点为 \(A\)，\(B\)，点 \(P\) 为椭圆上的动点，则使 \(\triangle PAB\) 的面积为 \(\dfrac{1}{2}\) 的点 \(P\) 的个数为（\quad）

\choices
    {\(1\)}
    {\(2\)}
    {\(3\)}
    {\(4\)}
\end{problem}
\answer{B}

\begin{solution}
关于原点对称后，直线 \(l\) 变为 \(l'\colon 2x+y-2=0\). 与椭圆联立，令 \(y=2-2x\)，得
\(x^2+\dfrac{(2-2x)^2}{4}=1\)，即 \(2x(x-1)=0\)，所以交点为 \(A=(0,2)\)，\(B=(1,0)\).

直线 \(AB\) 的方程为 \(2x+y-2=0\)，且 \(|AB|=\sqrt{5}\). 点 \(P=(x,y)\) 到该直线的距离为 \(\dfrac{|2x+y-2|}{\sqrt{5}}\)，故
\(S_{\triangle PAB}=\dfrac{1}{2}\sqrt{5}\cdot\dfrac{|2x+y-2|}{\sqrt{5}}=\dfrac{|2x+y-2|}{2}\).

令面积为 \(\dfrac{1}{2}\)，得到 \(|2x+y-2|=1\)，即需考察两条直线 \(2x+y=3\) 与 \(2x+y=1\) 和椭圆的交点. 代入椭圆，前者得到 \(8x^2-12x+5=0\)，判别式为 \(-16<0\)，没有交点；后者得到 \(8x^2-4x-3=0\)，判别式为 \(112>0\)，有两个交点. 因此满足条件的点 \(P\) 有 2 个. 故选 B.
\end{solution}

\section{填空题}
\begin{problem}
\(\displaystyle\lim_{n\to \infty}\frac{\mathrm C_n^2+2\mathrm C_n^{n-2}}{(n+1)^2}=\)\fillinblank{}.
\end{problem}
\answer{\(\dfrac{3}{2}\)}

\begin{solution}
因为 \(\mathrm C_n^{n-2}=\mathrm C_n^2=\dfrac{n(n-1)}{2}\)，所以
\(\displaystyle\frac{\mathrm C_n^2+2\mathrm C_n^{n-2}}{(n+1)^2}=\frac{3n(n-1)}{2(n+1)^2}\). 令 \(n\to\infty\)，所得极限为 \(\dfrac{3}{2}\).
\end{solution}

\begin{problem}
设双曲线 \(\dfrac{x^{2}}{a^{2}}-\dfrac{y^{2}}{b^{2}}=1\)（\(a>0\)，\(b>0\)）的右焦点为 \(F\)，右准线 \(l\) 与两条渐近线交于 \(P\)，\(Q\) 两点，如果 \(\triangle PQF\) 是直角三角形，则双曲线的离心率 \(e=\)\fillinblank{}.
\end{problem}
\answer{\(\sqrt{2}\)}

\begin{solution}
设双曲线的焦距参数为 \(c=\sqrt{a^2+b^2}\)，则右焦点为 \(F=(c,0)\)，右准线为 \(x=\dfrac{a^2}{c}\). 两条渐近线为 \(y=\pm\dfrac{b}{a}x\)，故
\(P=\left(\dfrac{a^2}{c},\dfrac{ab}{c}\right)\)，\(Q=\left(\dfrac{a^2}{c},-\dfrac{ab}{c}\right)\).

点 \(P\)，\(Q\) 关于 \(x\) 轴对称，且点 \(F\) 在 \(x\) 轴上，所以 \(FP=FQ\)，三角形 \(PQF\) 为等腰三角形. 若它是直角三角形，直角只能在 \(F\) 处，因此
\(\overrightarrow{FP}\cdot\overrightarrow{FQ}=\left(\dfrac{a^2}{c}-c\right)^2-\dfrac{a^2b^2}{c^2}=0\). 由 \(c^2=a^2+b^2\)，有 \(\dfrac{a^2}{c}-c=-\dfrac{b^2}{c}\)，从而 \(b^4=a^2b^2\). 因 \(a\)，\(b>0\)，得 \(a=b\). 所以离心率
\(e=\dfrac{c}{a}=\sqrt{1+\dfrac{b^2}{a^2}}=\sqrt{2}\).
\end{solution}

\begin{problem}
设 \(x\)，\(y\) 满足约束条件 \(\begin{cases}
x+y\leqslant 5,\\
3x+2y\leqslant 12,\\
0\leqslant x\leqslant 3,\\
0\leqslant y\leqslant 4,
\end{cases}\) 则使得目标函数 \(z=6x+5y\) 的最大的点 \((x,y)\) 是\fillinblank{}.
\end{problem}
\answer{\((2,3)\)}

\begin{solution}
由约束条件 \(x+y\leqslant5\) 和 \(3x+2y\leqslant12\)，有
\(z=6x+5y=3(x+y)+(3x+2y)\leqslant3\cdot5+12=27\).

当且仅当 \(x+y=5\) 且 \(3x+2y=12\) 时取等号. 解方程组得 \(x=2\)，\(y=3\)，且该点满足其余约束条件. 因此目标函数取得最大值的点为 \((2,3)\).
\end{solution}

\begin{problem}
已知 \(m\)，\(n\) 是不同的直线，\(\alpha\)，\(\beta\) 是不重合的平面，给出下列命题：
\begin{circlelist}
\item 若 \(\alpha\parallel\beta\)，\(m\subset\alpha\)，\(n\subset\beta\)，则 \(m\parallel n\)；
\item 若 \(m\)，\(n\subset\alpha\)，\(m\parallel\beta\)，\(n\parallel\beta\)，则 \(\alpha\parallel\beta\)；
\item 若 \(m\perp\alpha\)，\(n\perp\beta\)，\(m\parallel n\)，则 \(\alpha\parallel\beta\)；
\item \(m\)，\(n\) 是两条异面直线，若 \(m\parallel\alpha\)，\(m\parallel\beta\)，\(n\parallel\alpha\)，\(n\parallel\beta\)，则 \(\alpha\parallel\beta\).
\end{circlelist}
上面的命题中，真命题的序号是\fillinblank{}.（写出所有真命题的序号）
\end{problem}

\answer{\(\circled{3}\)、\(\circled{4}\)}

\begin{solution}
命题\(\circled{1}\)不成立. 两平面平行时，分别位于两平面内的直线方向可以不同，因此不一定平行.

命题\(\circled{2}\)不成立. 例如取两相交平面 \(\alpha\) 和 \(\beta\)，在 \(\alpha\) 内取两条互不重合且都平行于交线的直线，则它们都平行于平面 \(\beta\)，但 \(\alpha\) 与 \(\beta\) 不平行.

命题\(\circled{3}\)成立. 直线 \(m\parallel n\)，且 \(m\perp\alpha\)，\(n\perp\beta\)，所以平面 \(\alpha\) 与 \(\beta\) 的法线方向平行. 两个不重合平面的法线方向平行，说明两平面平行.

命题\(\circled{4}\)成立. 若 \(\alpha\) 与 \(\beta\) 不平行，则它们相交于一条直线. 凡同时平行于 \(\alpha\) 和 \(\beta\) 的直线，其方向都必须平行于这条交线. 因此 \(m\) 与 \(n\) 将相互平行，这与它们是异面直线矛盾. 所以 \(\alpha\parallel\beta\). 故真命题的序号为 \(\circled{3}\)、\(\circled{4}\).
\end{solution}

\section{解答题}
\begin{problem}
已知向量 \(\bs{m}=(\cos\theta,\sin\theta)\) 和 \(\bs{n}=(\sqrt{2}-\sin\theta,\cos\theta)\)，\(\theta\in(\pi,2\pi)\)，且 \(\left|\bs{m}+\bs{n}\right|=\dfrac{8\sqrt{2}}{5}\)，求 \(\cos\left(\dfrac{\theta}{2}+\dfrac{\pi}{8}\right)\) 的值.
\end{problem}

\begin{solution}
由
\(\bs m+\bs n=(\sqrt{2}+\cos\theta-\sin\theta,\sin\theta+\cos\theta)\)，得
\(\left|\bs m+\bs n\right|^2=(\sqrt{2}+\cos\theta-\sin\theta)^2+(\sin\theta+\cos\theta)^2=4+2\sqrt{2}(\cos\theta-\sin\theta)\).

由已知 \(\left|\bs m+\bs n\right|^2=\dfrac{128}{25}\)，所以
\(\cos\theta-\sin\theta=\dfrac{7\sqrt{2}}{25}\). 令 \(\varphi=\dfrac{\theta}{2}+\dfrac{\pi}{8}\)，则 \(2\varphi=\theta+\dfrac{\pi}{4}\)，从而
\(\cos 2\varphi=\dfrac{\cos\theta-\sin\theta}{\sqrt{2}}=\dfrac{7}{25}\).

因为 \(\theta\in(\pi,2\pi)\)，所以 \(\varphi\in\left(\dfrac{5\pi}{8},\dfrac{9\pi}{8}\right)\)，在此区间内 \(\cos\varphi<0\). 于是
\(\cos^2\varphi=\dfrac{1+\cos2\varphi}{2}=\dfrac{16}{25}\)，故
\(\cos\left(\dfrac{\theta}{2}+\dfrac{\pi}{8}\right)=\cos\varphi=-\dfrac{4}{5}\).
\end{solution}

\begin{problem}
袋中装有黑球和白球共 \(7\) 个，从中任取 \(2\) 个球都是白球的概率为 \(\dfrac{1}{7}\)，现有甲，乙两人从袋中轮流摸取 \(1\) 球，甲先取，乙后取，然后甲再取 \(\cdots\)，取后不放回，直到两人中有一人取到白球时既终止. 每个球在每一次被取出的机会是等可能的，用 \(\xi\) 表示取球终止所需要的取球次数.

\begin{enumerate}
\item 求袋中所有的白球的个数；
\item 求随机变量 \(\xi\) 的概率分布；
\item 求甲取到白球的概率.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item
设白球有 \(w\) 个，则
\(\dfrac{\mathrm C_w^2}{\mathrm C_7^2}=\dfrac{1}{7}\)，即 \(\dfrac{w(w-1)}{42}=\dfrac{1}{7}\). 因此 \(w(w-1)=6\)，结合 \(0\leqslant w\leqslant7\)，得 \(w=3\)，黑球有 \(4\) 个.

\item
取球直到第一次取到白球为止，所以 \(\xi\) 的可能值为 \(1\)，\(2\)，\(3\)，\(4\)，\(5\). 逐次按“不放回”计算，有
\(P(\xi=1)=\dfrac{3}{7}\)，
\(P(\xi=2)=\dfrac{4}{7}\cdot\dfrac{3}{6}=\dfrac{2}{7}\)，
\(P(\xi=3)=\dfrac{4}{7}\cdot\dfrac{3}{6}\cdot\dfrac{3}{5}=\dfrac{6}{35}\)，
\(P(\xi=4)=\dfrac{4}{7}\cdot\dfrac{3}{6}\cdot\dfrac{2}{5}\cdot\dfrac{3}{4}=\dfrac{3}{35}\)，
\(P(\xi=5)=\dfrac{4}{7}\cdot\dfrac{3}{6}\cdot\dfrac{2}{5}\cdot\dfrac{1}{4}\cdot\dfrac{3}{3}=\dfrac{1}{35}\). 这些概率之和为 \(1\)，所以这就是 \(\xi\) 的概率分布.

\item
甲只可能在第 \(1\)，第 \(3\) 或第 \(5\) 次取到白球，且这三个事件互斥. 因此
\(P(\text{甲取到白球})=P(\xi=1)+P(\xi=3)+P(\xi=5)=\dfrac{3}{7}+\dfrac{6}{35}+\dfrac{1}{35}=\dfrac{22}{35}\).
\end{enumerate}
\end{solution}

\begin{problem}
已知 \(x=1\) 是函数 \(f(x)=mx^3-3(m+1)x^2+nx+1\) 的一个极值点，其中 \(m\)，\(n\in\R\)，\(m<0\).

\begin{enumerate}
\item 求 \(m\) 与 \(n\) 的关系式；
\item 求 \(f(x)\) 的单调区间；
\item 当 \(x\in[-1,1]\) 时，函数 \(y=f(x)\) 的图象上任意一点的切线斜率恒大于 \(3m\)，求 \(m\) 的取值范围.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item
由 \(x=1\) 是极值点，必有 \(f'(1)=0\). 又
\(f'(x)=3mx^2-6(m+1)x+n\)，所以 \(3m-6(m+1)+n=0\)，即
\(n=3m+6\). 此时 \(f''(1)=6m-6(m+1)=-6\neq0\)，确为极值点.

\item
代入 \(n=3m+6\)，得
\(f'(x)=3mx^2-6(m+1)x+3m+6=3m(x-1)\left(x-1-\dfrac{2}{m}\right)\). 由于 \(m<0\)，令 \(r=1+\dfrac{2}{m}\)，则 \(r<1\). 因 \(3m<0\)，当 \(x<r\) 时，\(x-1<0\)，\(x-r<0\)，故 \(f'(x)<0\)；当 \(r<x<1\) 时，\(x-1<0\)，\(x-r>0\)，故 \(f'(x)>0\)；当 \(x>1\) 时，\(x-1>0\)，\(x-r>0\)，故 \(f'(x)<0\). 因此 \(f(x)\) 在 \(\left(-\infty,r\right)\) 和 \((1,+\infty)\) 上单调递减，在 \((r,1)\) 上单调递增.

\item
当 \(x\in[-1,1]\) 时，切线斜率为 \(f'(x)\). 由
\(f'(x)-3m=3\left(mx^2-2(m+1)x+2\right)\)，记 \(g(x)=mx^2-2(m+1)x+2\). 由于 \(m<0\)，\(g(x)\) 是开口向下的二次函数，其在闭区间 \([-1,1]\) 上的最小值取在端点. 又
\(g(-1)=3m+4\)，\(g(1)=-m>0\). 因此对所有 \(x\in[-1,1]\) 有 \(f'(x)>3m\) 的充要条件是 \(3m+4>0\)，结合 \(m<0\)，得
\(-\dfrac{4}{3}<m<0\).
\end{enumerate}
\end{solution}

\begin{problem}
如图，已知长方体 \(ABCD-A_{1}B_{1}C_{1}D_{1}\)，\(AB=2\)，\(AA_{1}=1\)，直线 \(BD\) 与平面 \(AA_{1}B_{1}B\) 所成的角为 \(30^{\circ}\)，\(AE\) 垂直 \(BD\) 于 \(E\)，\(F\) 为 \(A_{1}B_{1}\) 的中点.

\begin{enumerate}
\item 求异面直线 \(AE\) 与 \(BF\) 所成的角；
\item 求平面 \(BDF\) 与平面 \(AA_{1}B\) 所成的二面角（锐角）的大小；
\item 求点 \(A\) 到平面 \(BDF\) 的距离.
\end{enumerate}

\bitmapfigure[max width=0.46\linewidth]{img/2005/shandong_science/q20_fig1.png}
\end{problem}

\begin{solution}
建立空间直角坐标系，取
\(B=(0,0,0)\)，\(A=(2,0,0)\)，\(C=(0,d,0)\)，\(D=(2,d,0)\)，\(B_1=(0,0,1)\)，\(A_1=(2,0,1)\). 平面 \(AA_1B_1B\) 为 \(y=0\)，而 \(\overrightarrow{BD}=(2,d,0)\). 由直线与平面所成角的定义，
\(\sin30^{\circ}=\dfrac{d}{\sqrt{4+d^2}}\)，解得 \(d=\dfrac{2}{\sqrt{3}}\).

点 \(E\) 是 \(A\) 在直线 \(BD\) 上的投影. 设 \(E=t\overrightarrow{BD}\)，则
\(t=\dfrac{\overrightarrow{BA}\cdot\overrightarrow{BD}}{|\overrightarrow{BD}|^2}=\dfrac{4}{4+d^2}=\dfrac{3}{4}\)，所以 \(E=\left(\dfrac{3}{2},\dfrac{\sqrt{3}}{2},0\right)\). 又 \(F\) 为 \(A_1B_1\) 的中点，故 \(F=(1,0,1)\).

\begin{enumerate}
\item
取两直线的方向向量 \(\overrightarrow{EA}=\left(\dfrac{1}{2},-\dfrac{\sqrt{3}}{2},0\right)\) 和 \(\overrightarrow{BF}=(1,0,1)\)，则
\(\cos\angle(AE,BF)=\dfrac{\overrightarrow{EA}\cdot\overrightarrow{BF}}{|\overrightarrow{EA}|\,|\overrightarrow{BF}|}=\dfrac{1/2}{1\cdot\sqrt{2}}=\dfrac{\sqrt{2}}{4}\).
所以异面直线 \(AE\) 与 \(BF\) 所成的角为 \(\arccos\dfrac{\sqrt{2}}{4}\).

\item
取 \(\overrightarrow{BD}=(2,d,0)\)，\(\overrightarrow{BF}=(1,0,1)\)，则平面 \(BDF\) 的一个法向量为
\(\bs n_1=\overrightarrow{BD}\times\overrightarrow{BF}=(d,-2,-d)\). 平面 \(AA_1B\) 为 \(y=0\)，其法向量可取 \(\bs n_2=(0,1,0)\). 因此两平面所成锐角 \(\varphi\) 满足
\(\cos\varphi=\dfrac{|\bs n_1\cdot\bs n_2|}{|\bs n_1|\,|\bs n_2|}=\dfrac{2}{\sqrt{4+2d^2}}=\sqrt{\dfrac{3}{5}}=\dfrac{\sqrt{15}}{5}\).
所以二面角的大小为 \(\arccos\dfrac{\sqrt{15}}{5}\).

\item
平面 \(BDF\) 过原点 \(B\)，法向量为 \(\bs n_1\)，所以点 \(A\) 到该平面的距离为
\(\dfrac{|\bs n_1\cdot\overrightarrow{BA}|}{|\bs n_1|}=\dfrac{2d}{\sqrt{4+2d^2}}=\dfrac{2\sqrt{5}}{5}\).
\end{enumerate}
\end{solution}

\begin{problem}
已知数列 \(\{a_n\}\) 的首项 \(a_1=5\)，前 \(n\) 项和为 \(S_n\)，且 \(S_{n+1}=2S_n+n+5\)（\(n\in\mathbf{N}^{*}\)）.

\begin{enumerate}
\item 证明数列 \(\{a_n+1\}\) 是等比数列；
\item 令 \(f(x)=a_1x+a_2x^2+\cdots+a_nx^n\)，求函数 \(f(x)\) 在点 \(x=1\) 处的导数 \(f'(1)\) 并比较 \(2f'(1)\) 与 \(23n^2-13n\) 的大小.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item
由 \(S_1=a_1=5\) 及递推式得 \(S_2=2S_1+1+5=16\)，所以 \(a_2=S_2-S_1=11\). 当 \(n\geqslant2\) 时，
\(a_{n+1}=S_{n+1}-S_n=S_n+n+5\)，
\(a_n=S_n-S_{n-1}=S_{n-1}+n+4\)，且 \(S_n=2S_{n-1}+n+4\). 因此
\(a_{n+1}+1=S_n+n+6=2S_{n-1}+2n+10=2(a_n+1)\). 当 \(n=1\) 时也有 \(a_2+1=12=2(a_1+1)\)，故 \(\{a_n+1\}\) 是首项为 \(6\)、公比为 \(2\) 的等比数列.

从而 \(a_n+1=6\cdot2^{n-1}=3\cdot2^n\)，即
\(a_n=3\cdot2^n-1\).

\item
由 \(f(x)=\sum_{k=1}^{n}a_kx^k\)，得
\(f'(1)=\sum_{k=1}^{n}ka_k=3\sum_{k=1}^{n}k2^k-\sum_{k=1}^{n}k\). 记 \(T=\sum_{k=1}^{n}k2^k\)，则
\(2T-T=n2^{n+1}-\sum_{k=2}^{n}2^k-2=(n-1)2^{n+1}+2\)，所以
\(f'(1)=3(n-1)2^{n+1}+6-\dfrac{n(n+1)}{2}\).

令
\(H=2f'(1)-(23n^2-13n)\)，则
\(H=12\left((n-1)2^n-2n^2+n+1\right)\). 记括号内的式子为 \(h(n)\)，则 \(h(1)=0\)，\(h(2)=-1\)，\(h(3)=2\)，且
\(h(n+1)-h(n)=(n+1)2^n-4n-1\). 当 \(n\geqslant3\) 时，\((n+1)2^n-4n-1\geqslant8(n+1)-4n-1=4n+7>0\)，所以 \(h(n)\) 在 \(n\geqslant3\) 时递增. 因此，当 \(n=1\) 时两者相等；当 \(n=2\) 时 \(2f'(1)<23n^2-13n\)；当 \(n\geqslant3\) 时 \(2f'(1)>23n^2-13n\).
\end{enumerate}
\end{solution}

\begin{problem}
已知动圆过定点 \(\left(\dfrac{p}{2},0\right)\)，且与直线 \(x=-\dfrac{p}{2}\) 相切，其中 \(p>0\).

\begin{enumerate}
\item 求动圆圆心 \(C\) 的轨迹的方程；
\item 设 \(A\)，\(B\) 是轨迹 \(C\) 上异于原点 \(O\) 的两个不同点，直线 \(OA\) 和 \(OB\) 的倾斜角分别为 \(\alpha\) 和 \(\beta\)，当 \(\alpha\)，\(\beta\) 变化且 \(\alpha+\beta\) 为定值 \(\theta\)（\(0<\theta<\pi\)）时，证明直线 \(AB\) 恒过定点，并求出该定点的坐标.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item
设动圆圆心为 \(C=(x,y)\). 因为动圆过点 \(\left(\dfrac{p}{2},0\right)\)，其半径平方为
\(\left(x-\dfrac{p}{2}\right)^2+y^2\)；又动圆与直线 \(x=-\dfrac{p}{2}\) 相切，半径平方为 \(\left(x+\dfrac{p}{2}\right)^2\). 两者相等，得
\(\left(x-\dfrac{p}{2}\right)^2+y^2=\left(x+\dfrac{p}{2}\right)^2\)，即圆心轨迹为 \(y^2=2px\).

\item
由于 \(A\)，\(B\) 是抛物线 \(y^2=2px\) 上异于原点的点，故它们的横坐标均为正，直线 \(OA\)，\(OB\) 的斜率存在. 记其斜率分别为 \(\tan\alpha\)，\(\tan\beta\)，再令 \(u=\cot\alpha\)，\(v=\cot\beta\). 将直线 \(y=(\tan\alpha)x\) 与抛物线联立，得到
\(A=(2pu^2,2pu)\). 将直线 \(y=(\tan\beta)x\) 与抛物线联立，得到 \(B=(2pv^2,2pv)\). 因此直线 \(AB\) 的方程为
\(x-(u+v)y+2puv=0\).

由和角公式，
\(\cot\theta=\cot(\alpha+\beta)=\dfrac{uv-1}{u+v}\)，所以 \(uv=(u+v)\cot\theta+1\). 代入直线方程得
\(x+2p+(u+v)(2p\cot\theta-y)=0\). 因此点
\(\left(-2p,2p\cot\theta\right)\) 恒在直线 \(AB\) 上.

当 \(\theta=\dfrac{\pi}{2}\) 时，该点为 \((-2p,0)\)；当 \(\theta\neq\dfrac{\pi}{2}\) 时，也可写为 \(\left(-2p,\dfrac{2p}{\tan\theta}\right)\).
\end{enumerate}
\end{solution}
