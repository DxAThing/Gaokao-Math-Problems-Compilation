\begin{problem}
已知函数 \(f(x) = \frac{x^2}{ax + b}\)（\(a, b\) 为常数），且方程 \(f(x) - x + 12 = 0\) 有两个实根为 \(x_1 = 3, x_2 = 4\).

\begin{enumerate}[label=（\arabic*）]
\item 求函数 \(f(x)\) 的解析式；
\item 设 \(k > 1\)，解关于 \(x\) 的不等式：\(f(x) < \frac{(k + 1)x - k}{2 - x}\).
\end{enumerate}
\end{problem}

\begin{problem}

\end{problem}
