\section{单选题}
\begin{problem}
设集合 \(A = \{x \mid |4x - 1| \geqslant 9,\ x \in \R\}\)，\(B = \left\{x \mid \frac{x}{x + 3} \geqslant 0,\ x \in \R\right\}\)，则 \(A \cap B =\)（\(\quad\)）

\choices
    {\((-3,-2]\)}
    {\((-3,-2] \cup \left[0,\frac{5}{2}\right]\)}
    {\((-\infty,-3] \cup \left[\frac{5}{2},+\infty\right)\)}
    {\((-\infty,-3) \cup \left[\frac{5}{2},+\infty\right)\)}
\end{problem}

\begin{problem}
若复数 \(\frac{a + 3\i}{1 + 2\i}\)（\(a \in \R\)，\(\ \i\) 为虚数单位）是纯虚数，则实数 \(a\) 的值为（\(\quad\)）

\choices
    {\(-2\)}
    {\(4\)}
    {\(-6\)}
    {\(6\)}
\end{problem}

\begin{problem}
给出下列三个命题：

\begin{circlelist}
\item 若 \(a \geqslant b > -1\)，则 \(\frac{a}{1 + a} \geqslant \frac{b}{1 + b}\)；
\item 若正整数 \(m\) 和 \(n\) 满足 \(m \leqslant n\)，则 \(\sqrt{m(n - m)} \leqslant \frac{n}{2}\)；
\item 设 \(P(x_{1}, y_{1})\) 为圆 \(O_{1}: x^{2} + y^{2} = 9\) 上任一点，圆 \(O_{2}\) 以 \(Q(a, b)\) 为圆心且半径为 \(1\). 当 \((a - x_{1})^{2} + (b - y_{1})^{2} = 1\) 时，圆 \(O_{1}\) 与圆 \(O_{2}\) 相切.
\end{circlelist}

其中假命题的个数为（\(\quad\)）

\choices
    {\(0\)}
    {\(1\)}
    {\(2\)}
    {\(3\)}
\end{problem}

\begin{problem}
设 \(\alpha\)，\(\beta\)，\(\gamma\) 为平面，\(m\)，\(n\)，\(l\) 为直线，则 \(m \perp \beta\) 的一个充分条件是（\(\quad\)）

\choices
    {\(\alpha \perp \beta,\ \alpha \cap \beta = l,\ m \perp l\)}
    {\(\alpha \cap \gamma = m\)，\(\ \alpha \perp \gamma\)，\(\ \beta \perp \gamma\)}
    {\(\alpha \perp \gamma\)，\(\ \beta \perp \gamma\)，\(\ m \perp \alpha\)}
    {\(n \perp \alpha\)，\(\ n \perp \beta\)，\(\ m \perp \alpha\)}
\end{problem}

\begin{problem}
设双曲线以椭圆 \(\frac{x^2}{9} + \frac{y^2}{4} = 1\) 长轴的两个端点为焦点，其准线过椭圆的焦点，则双曲线的渐近线的斜率为（\(\quad\)）

\choices
    {\(\pm 2\)}
    {\(\pm \frac{4}{3}\)}
    {\(\pm \frac{1}{2}\)}
    {\(\pm \frac{3}{4}\)}
\end{problem}

\begin{problem}
从集合 \(\{1, 2, 3, \cdots, 11\}\) 中任选两个元素作为椭圆方程 \(\frac{x^2}{m^2} + \frac{y^2}{n^2} = 1\) 中的 \(m\) 和 \(n\)，则能组成落在矩形区域 \(B = \{(x, y) \mid |x| < 11\text{ 且 }|y| < 9\}\) 内的椭圆个数为（\(\quad\)）

\choices
    {\(43\)}
    {\(72\)}
    {\(86\)}
    {\(90\)}
\end{problem}

\begin{problem}
某人射击一次击中的概率为 \(0.6\)，经过 \(3\) 次射击，此人至少有两次击中目标的概率为（\(\quad\)）

\choices
    {\(\frac{81}{125}\)}
    {\(\frac{54}{125}\)}
    {\(\frac{36}{125}\)}
    {\(\frac{27}{125}\)}
\end{problem}

\begin{problem}
要得到函数 \(y = \sqrt{2} \cos x\) 的图象，只需将函数 \(y = \sqrt{2} \sin \left(2x + \frac{\pi}{4}\right)\) 的图象上所有的点的（\(\quad\)）

\choices
    {横坐标缩短到原来的 \(\frac{1}{2}\) 倍（纵坐标不变），再向左平行移动 \(\frac{\pi}{8}\) 个单位长度}
    {横坐标缩短到原来的 \(\frac{1}{2}\) 倍（纵坐标不变），再向右平行移动 \(\frac{\pi}{4}\) 个单位长度}
    {横坐标伸长到原来的 \(2\) 倍（纵坐标不变），再向左平行移动 \(\frac{\pi}{4}\) 个单位长度}
    {横坐标伸长到原来的 \(2\) 倍（纵坐标不变），再向右平行移动 \(\frac{\pi}{8}\) 个单位长度}
\end{problem}

\begin{problem}
设 \(f^{-1}(x)\) 是函数 \(f(x)=\frac{1}{2}(a^{x}-a^{-x})\)（\(a > 1\)）的反函数，则使 \(f^{-1}(x)>1\) 成立的 \(x\) 的取值范围为（\(\quad\)）

\choices
    {\(\left(\frac{a^2 - 1}{2a}, +\infty\right)\)}
    {\(\left(-\infty, \frac{a^2 - 1}{2a}\right)\)}
    {\(\left(\frac{a^2 - 1}{2a}, a\right)\)}
    {\(\left[a, +\infty\right)\)}
\end{problem}

\begin{problem}
若函数 \(f(x) = \log_a(x^3 - ax)\)（\(a > 0\)，\(a \neq 1\)）在区间 \(\left(-\frac{1}{2}, 0\right)\) 内单调递增，则 \(a\) 的取值范围是（\(\quad\)）

\choices
    {\(\left[\frac{1}{4}, 1\right)\)}
    {\(\left[\frac{3}{4}, 1\right)\)}
    {\(\left(\frac{9}{4}, +\infty\right)\)}
    {\(\left(1, \frac{9}{4}\right)\)}
\end{problem}

\end{problem}

\begin{problem}

\begin{problem}
在数列 \(\{a_{n}\}\) 中，\(a_{1} = 1\)，\(a_{2} = 2\)，且 \(a_{n+2} - a_{n} = 1 + (-1)^{n}\)（\(n \in \mathbf{N}^{*}\)）. 则 \(S_{100} =\)\fillinblank{}.

\begin{problem}
在直角坐标系 \(xOy\) 中，已知点 \(A(0,1)\) 和点 \(B(-3,4)\)，若点 \(C\) 在 \(\angle AOB\) 的平分线上且 \(\left|\overrightarrow{OC}\right| = 2\)，则 \(\overrightarrow{OC} =\fillinblank{}.\)
\end{problem}

\begin{problem}
某公司有 \(5\) 万元资金用于投资开发项目，如果成功，一年后可获利 \(12\%\)，一旦失败，一年后将丧失全部资金的 \(50\%\). 下表是过去 \(200\) 例类似项目开发的实施结果：

\begin{center}
\begin{tabular}{|c|c|}
\hline
投资成功 & 投资失败 \\
\hline
\(192\)次 & \(8\)次 \\
\hline
\end{tabular}
\end{center}

则该公司一年后估计可获收益的期望是（元）\fillinblank{}.
\end{problem}

\begin{problem}
设 \(f(x)\) 是定义在 \(\R\) 上的奇函数，且 \(y = f(x)\) 的图象关于直线 \(x = \frac{1}{2}\) 对称，则 \(f(1) + f(2) + f(3) + f(4) + f(5) =\)\fillinblank{}.
\end{problem}

\begin{problem}
在 \(\triangle ABC\) 中，\(\angle A\)，\(\angle B\)，\(\angle C\) 所对的边长分别为 \(a\)，\(b\)，\(c\). 设 \(a\)，\(b\)，\(c\) 满足条件 \(b^2 + c^2 - bc = a^2\) 和 \(\frac{c}{b} = \frac{1}{2} + \sqrt{3}\)，求 \(\angle A\) 和 \(\tan B\) 的值.
\end{problem}

\begin{problem}
已知 \(u_{n} = a^{n} + a^{n - 1}b + a^{n - 2}b^{2} + \cdots +ab^{n - 1} + b^{n}\)（\(n\in \mathbf{N}^{*}\)，\(a > 0\)，\(b > 0\)）.

\begin{enumerate}
\item 当 \(a = b\) 时，求数列 \(\{u_n\}\) 的前 \(n\) 项和 \(S_n\)；
\item 求 \(\lim\limits_{n\to\infty}\frac{u_n}{u_{n-1}}\).
\end{enumerate}
\end{problem}

\begin{problem}
抛物线 \(C\) 的方程为 \(y = ax^2\)（\(a < 0\)），过抛物线 \(C\) 上一点 \(P(x_{0}, y_{0})\)（\(x_{0} \neq 0\)）作斜率为 \(k_1\)，\(k_2\) 的两条直线分别交抛物线 \(C\) 于 \(A(x_{1}, y_{1})\)，\(B(x_{2}, y_{2})\) 两点（\(P\)，\(A\)，\(B\) 三点互不相同），且满足 \(k_2 + \lambda k_1 = 0\)（\(\lambda \neq 0\)，\(\lambda \neq -1\)）.

\begin{enumerate}
\item 求抛物线 \(C\) 的焦点坐标和准线方程；
\item 设直线 \(AB\) 上一点 \(M\)，满足 \(\overrightarrow{BM} = \lambda \overrightarrow{MA}\)，证明线段 \(PM\) 的中点在 \(y\) 轴上；
\item 当 \(\lambda = 1\) 时，若点 \(P\) 的坐标为 \((1, -1)\)，求 \(\angle PAB\) 为钝角时点 A 的纵坐标 \(y_{1}\) 的取值范围.
\end{enumerate}
\end{problem}

\begin{problem}
设函数 \(f(x) = x \sin x\)（\(x \in \R\)）.

\begin{enumerate}
\item 证明 \(f(x + 2k\pi) - f(x) = 2k\pi \sin x\)，其中 \(k\) 为整数；
\item 设 \(x_0\) 为 \(f(x)\) 的一个极值点，证明 \(\left[f(x_0)\right]^2 = \frac{x_0^4}{1 + x_0^2}\)；
\item 设 \(f(x)\) 在 \((0, +\infty)\) 内的全部极值点按从小到大的顺序排列 \(a_1\)，\(a_2\)，\(\cdots\)，\(a_n\)，\(\cdots\)，证明 \(\frac{2}{9} < a_{n+1} - a_n < \pi\)（\(n = 1, 2, \cdots\)）.
\end{enumerate}
\end{problem}
