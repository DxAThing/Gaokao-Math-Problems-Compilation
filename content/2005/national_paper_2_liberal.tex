\section{单选题}

\begin{problem}
函数 \( f(x) = |\sin x + \cos x| \) 的最小正周期是（\quad）

\choices
    {\(\frac{\pi}{4}\)}
    {\(\frac{\pi}{2}\)}
    {\(\pi\)}
    {\(2\pi\)}
\end{problem}

\begin{answer}C\end{answer}

\begin{solution}
因为 \(\sin x+\cos x=\sqrt{2}\sin\left(x+\frac{\pi}{4}\right)\)，所以函数值为 \(\sqrt{2}\left|\sin\left(x+\frac{\pi}{4}\right)\right|\)，其周期为正弦函数绝对值的最小正周期 \(\pi\). 例如 \(f\left(\frac{\pi}{4}\right)=\sqrt{2}\)，而 \(f\left(\frac{3\pi}{4}\right)=0\)，故 \(\frac{\pi}{2}\) 不是周期，最小正周期为 \(\pi\)，故选 C.
\end{solution}

\begin{problem}
正方体 \(ABCD - A_{1}B_{1}C_{1}D_{1}\) 中，\(P\)，\(Q\)，\(R\) 分别是 \(AB\)，\(AD\)，\(B_{1}C_{1}\) 的中点. 那么，正方体的过 \(P\)，\(Q\)，\(R\) 的截面图形是（\quad）

\choices
    {三角形}
    {四边形}
    {五边形}
    {六边形}
\end{problem}

\begin{answer}D\end{answer}

\begin{solution}
取正方体边长为 \(1\)，设 \(A=(0,0,0)\)，\(B=(1,0,0)\)，\(D=(0,1,0)\)，\(A_{1}=(0,0,1)\)，则 \(P=\left(\frac{1}{2},0,0\right)\)，\(Q=\left(0,\frac{1}{2},0\right)\)，\(R=\left(1,\frac{1}{2},1\right)\). 过三点的平面方程为 \(x+y-z=\frac{1}{2}\). 它与正方体的边界相交于六点：底面上的 \(P,Q\)，棱 \(BB_{1}\)、\(DD_{1}\) 上各一点，棱 \(B_{1}C_{1}\) 上的点 \(R\)，以及顶面上除 \(R\) 外的另一点. 因此截面为六边形，故选 D.
\end{solution}

\begin{problem}
函数 \( y = x^{2} - 1 \)（\(x \leqslant 0\)）的反函数是（\quad）

\choices
    {\( y = \sqrt{x + 1} \)（\(x \geqslant -1\)）}
    {\( y = -\sqrt{x + 1} \)（\(x \geqslant -1\)）}
    {\( y = \sqrt{x + 1} \)（\(x \geqslant 0\)）}
    {\( y = -\sqrt{x + 1} \)（\(x \geqslant 0\)）}
\end{problem}

\begin{answer}B\end{answer}

\begin{solution}
由 \(y=x^{2}-1\) 且 \(x\leqslant0\)，得 \(x=-\sqrt{y+1}\)，并且 \(y\geqslant-1\). 交换 \(x,y\)，反函数为 \(y=-\sqrt{x+1}\)，定义域为 \(x\geqslant-1\)，故选 B.
\end{solution}

\begin{problem}
已知函数 \( y = \tan \omega x \) 在 \(\left(-\frac{\pi}{2}, \frac{\pi}{2}\right)\) 内是减函数，则（\quad）

\choices
    {\(0 < \omega \leqslant 1\)}
    {\(-1 \leqslant \omega < 0\)}
    {\(\omega \geqslant 1\)}
    {\(\omega \leqslant -1\)}
\end{problem}

\begin{answer}B\end{answer}

\begin{solution}
函数在给定开区间内有定义，需满足 \(\left|\omega x\right|<\frac{\pi}{2}\) 对该区间内所有 \(x\) 成立，因此 \(\left|\omega\right|\leqslant1\). 又 \(y'=\omega\sec^{2}\omega x\)，而 \(\sec^{2}\omega x>0\)，函数为减函数等价于 \(\omega<0\). 所以 \(-1\leqslant\omega<0\)，故选 B.
\end{solution}

\begin{problem}
抛物线 \(x^{2} = 4y\) 上一点 \(A\) 的纵坐标为\(4\)，则点 \(A\) 与抛物线焦点的距离为（\quad）

\choices
    {\(2\)}
    {\(3\)}
    {\(4\)}
    {\(5\)}
\end{problem}

\begin{answer}D\end{answer}

\begin{solution}
抛物线 \(x^{2}=4y\) 的参数 \(p=1\)，焦点为 \(F(0,1)\). 点 \(A\) 的纵坐标为 \(4\)，代入得 \(x^{2}=16\)，所以 \(A=(4,4)\) 或 \((-4,4)\). 于是 \(AF=\sqrt{(\pm4)^{2}+3^{2}}=5\)，故选 D.
\end{solution}

\begin{problem}
双曲线 \(\frac{x^2}{4} - \frac{y^2}{9} = 1\) 的渐近线方程是（\quad）

\choices
    {\( y = \pm \frac{2}{3} x \)}
    {\( y = \pm \frac{4}{9} x \)}
    {\( y = \pm \frac{3}{2} x \)}
    {\(\sqrt{3}\)}
\end{problem}

\begin{answer}C\end{answer}

\begin{solution}
双曲线 \(\frac{x^{2}}{a^{2}}-\frac{y^{2}}{b^{2}}=1\) 的渐近线为 \(y=\pm\frac{b}{a}x\). 本题中 \(a=2\)，\(b=3\)，所以渐近线方程为 \(y=\pm\frac{3}{2}x\)，故选 C.
\end{solution}

\begin{problem}
如果数列 \(\{a_{n}\}\) 是等差数列，则（\quad）

\choices
    {\(a_1 + a_8 < a_4 + a_5\)}
    {\(a_1 + a_8 = a_4 + a_5\)}
    {\(a_{1} + a_{8} > a_{4} + a_{5}\)}
    {\(a_{1}a_{8} = a_{4}a_{5}\)}
\end{problem}

\begin{answer}B\end{answer}

\begin{solution}
设等差数列首项为 \(a_{1}\)，公差为 \(d\)，则 \(a_{n}=a_{1}+(n-1)d\). 因此 \(a_{1}+a_{8}=2a_{1}+7d\)，且 \(a_{4}+a_{5}=\left(a_{1}+3d\right)+\left(a_{1}+4d\right)=2a_{1}+7d\)，所以 \(a_{1}+a_{8}=a_{4}+a_{5}\)，故选 B.
\end{solution}

\begin{problem}
\((x - \sqrt{2} y)^{10}\) 的展开式中 \(x^{6}y^{4}\) 项的系数是（\quad）

\choices
    {\(840\)}
    {\(-840\)}
    {\(210\)}
    {\(-210\)}
\end{problem}

\begin{answer}A\end{answer}

\begin{solution}
展开式的通项为 \(\mathrm C_{10}^{k}x^{10-k}\left(-\sqrt{2}y\right)^{k}\). 要得到 \(x^{6}y^{4}\)，必须取 \(k=4\)，故该项系数为 \(\mathrm C_{10}^{4}\left(-\sqrt{2}\right)^{4}=210\times4=840\)，故选 A.
\end{solution}

\begin{problem}
已知点 \(A(\sqrt{3}, 1)\)，\(B(0, 0)\)，\(C(\sqrt{3}, 0)\). 设 \(\angle BAC\) 的平分线 \(AE\) 与 \(BC\) 相交于 \(E\)，那么有 \(\overrightarrow{BC} = \lambda \overrightarrow{CE}\)，其中 \(\lambda\) 等于（\quad）

\choices
    {\(2\)}
    {\(\frac{1}{2}\)}
    {\(-3\)}
    {\(-\frac{1}{3}\)}
\end{problem}

\begin{answer}C\end{answer}

\begin{solution}
由 \(AB=2\)，\(AC=1\)，且 \(BC=\sqrt{3}\)，角平分线定理给出 \(\frac{BE}{EC}=\frac{AB}{AC}=2\). 又 \(BC=\sqrt{3}\)，所以 \(EC=\frac{\sqrt{3}}{3}\). 向量 \(\overrightarrow{CE}\) 与 \(\overrightarrow{BC}\) 方向相反，且 \(\left|\overrightarrow{BC}\right|=3\left|\overrightarrow{CE}\right|\)，故 \(\overrightarrow{BC}=-3\overrightarrow{CE}\)，即 \(\lambda=-3\)，故选 C.
\end{solution}

\begin{problem}
已知集合 \(M = \{x \mid x^2 - 3x - 28 \leqslant 0\}\)，\(N = \{x \mid x^2 - x - 6 > 0\}\)，则 \(M \cap N\) 为（\quad）

\choices
    {\(\left\{x\mid -4\leqslant x < -2\text{\ 或\ }3 < x\leqslant 7\right\}\)}
    {\(\left\{x\mid -4 < x\leqslant -2\text{\ 或\ }3\leqslant x < 7\right\}\)}
    {\(\left\{x\mid x\leqslant -2\text{\ 或\ }x > 3\right\}\)}
    {\(\left\{x\mid x < -2\text{\ 或\ }x\geqslant 3\right\}\)}
\end{problem}

\begin{answer}A\end{answer}

\begin{solution}
由 \(x^{2}-3x-28=(x-7)(x+4)\)，得 \(M=[-4,7]\). 又 \(x^{2}-x-6=(x-3)(x+2)>0\)，得 \(N=(-\infty,-2)\cup(3,+\infty)\). 所以 \(M\cap N=[-4,-2)\cup(3,7]\)，故选 A.
\end{solution}

\begin{problem}
点 \(P\) 在平面上作匀速直线运动. 速度向量 \(\bs{v} = (4, -3)\)（即点 \(P\) 的运动方向与 \(\bs{v}\) 相同，且每秒移动的距离为 \(|\bs{v}|\text{ 个单位}\)）. 设开始时点 \(P\) 的坐标为 \((-10, 10)\)，则 \(5\text{ 秒}\) 后点 \(P\) 的坐标为（\quad）

\choices
    {\((-2, 4)\)}
    {\((-30, 25)\)}
    {\((10, -5)\)}
    {\((5, -10)\)}
\end{problem}

\begin{answer}C\end{answer}

\begin{solution}
因为 \(\left|\bs{v}\right|=\sqrt{4^{2}+(-3)^{2}}=5\)，且每秒移动的距离正好为 \(5\)，所以每秒位移向量就是 \(\bs{v}=(4,-3)\). 经过 \(5\) 秒，位移为 \(5\bs{v}=(20,-15)\)，故点 \(P\) 的坐标为 \((-10,10)+(20,-15)=(10,-5)\)，故选 C.
\end{solution}

\begin{problem}
\(\triangle ABC\) 的顶点 \(B\) 在平面 \(\alpha\) 内，\(A\)，\(C\) 在 \(\alpha\) 的同一侧，\(AB\)，\(BC\) 与 \(\alpha\) 所成的角分别是 \(30^{\circ}\) 和 \(45^{\circ}\). 若 \(AB = 3\)，\(BC = 4\sqrt{2}\)，\(AC = 5\)，则 \(AC\) 与 \(\alpha\) 所成的角为（\quad）

\choices
    {\(60^{\circ}\)}
    {\(45^{\circ}\)}
    {\(30^{\circ}\)}
    {\(15^{\circ}\)}
\end{problem}

\begin{answer}C\end{answer}

\begin{solution}
取 \(B\) 为参考点，记点 \(A,C\) 到平面 \(\alpha\) 的距离分别为 \(h_A,h_C\). 由线面角定义，\(h_A=AB\sin30^{\circ}=\frac{3}{2}\)，\(h_C=BC\sin45^{\circ}=4\). 因 \(A,C\) 在平面 \(\alpha\) 同侧，\(AC\) 的垂直分量为 \(h_C-h_A=\frac{5}{2}\). 设 \(AC\) 与平面 \(\alpha\) 所成的角为 \(\theta\)，则 \(AC\sin\theta=\frac{5}{2}\). 代入 \(AC=5\)，得 \(\sin\theta=\frac{1}{2}\). 由于线面角取锐角或直角，故 \(\theta=30^{\circ}\)，故选 C.
\end{solution}

\section{填空题}

\begin{problem}
在 \(\frac{8}{3}\) 和 \(\frac{27}{2}\) 之间插入三个数，使这五个数成等比数列，则插入的三个数的乘积为\fillinblank{}.
\end{problem}

\begin{answer}\(216\)\end{answer}

\begin{solution}
设五个数依次为 \(a_{1},a_{2},a_{3},a_{4},a_{5}\)，则 \(a_{1}=\frac{8}{3}\)，\(a_{5}=\frac{27}{2}\). 等比数列中关于中项对称的两项乘积相等，所以 \(a_{3}^{2}=a_{1}a_{5}=\frac{8}{3}\cdot\frac{27}{2}=36\). 又 \(a_{3}=a_{1}q^{2}>0\)，故 \(a_{3}=6\). 插入的三个数为 \(a_{2},a_{3},a_{4}\)，其乘积为 \(a_{2}a_{3}a_{4}=a_{3}^{3}=6^{3}=216\).
\end{solution}

\begin{problem}
圆心为 \((1,2)\) 且与直线 \(5x - 12y - 7 = 0\) 相切的圆的方程为\fillinblank{}.
\end{problem}

\begin{answer}\(\left(x-1\right)^{2}+\left(y-2\right)^{2}=4\)\end{answer}

\begin{solution}
圆心 \(O(1,2)\) 到直线 \(5x-12y-7=0\) 的距离为 \(r=\frac{|5\times1-12\times2-7|}{\sqrt{5^{2}+(-12)^{2}}}=\frac{26}{13}=2\). 所以所求圆的方程为 \(\left(x-1\right)^{2}+\left(y-2\right)^{2}=r^{2}=4\).
\end{solution}

\begin{problem}
在由数字 \(0\)，\(1\)，\(2\)，\(3\)，\(4\)，\(5\) 所组成的没有重复数字的四位数中，不能被 \(5\) 整除的数共有\fillinblank{}\(\text{ 个}\).
\end{problem}

\begin{answer}\(192\)\end{answer}

\begin{solution}
四位数不能被 \(5\) 整除，末位不能为 \(0\) 或 \(5\)，所以末位有 \(4\) 种选法. 确定末位后，首位从剩下的五个数字中不能取 \(0\)，有 \(4\) 种选法；中间两位从剩下的四个数字中依次选取，有 \(4\times3\) 种选法. 因此共有 \(4\times4\times4\times3=192\) 个.
\end{solution}

\begin{problem}
下面是关于三棱锥的四个命题：

\begin{circlelist}
\item 底面是等边三角形，侧面与底面所成的二面角都相等的三棱锥是正三棱锥；
\item 底面是等边三角形，侧面都是等腰三角形的三棱锥是正三棱锥；
\item 底面是等边三角形，侧面的面积都相等的三棱锥是正三棱锥；
\item 侧棱与底面所成的角都相等，且侧面与底面所成的二面角都相等的三棱锥是正三棱锥.
\end{circlelist}

其中，真命题的编号是\fillinblank{}. （写出所有真命题的编号）
\end{problem}

\begin{answer}\(\circled{1}\circled{4}\)\end{answer}

\begin{solution}
设三棱锥顶点为 \(S\)，底面为等边三角形 \(ABC\)，\(H\) 为 \(S\) 在底面上的射影.

对于命题 \(\circled{1}\)，在分别垂直于 \(AB\)、\(BC\)、\(CA\) 的截面内，二面角的正切等于 \(SH\) 除以 \(H\) 到相应底边所在直线的有向距离. 三个二面角相等，故 \(H\) 到三条边所在直线的有向距离相等；这说明 \(H\) 在三角形 \(ABC\) 的三条内角平分线上，即 \(H\) 为内心. 等边三角形的内心也是外心，所以 \(SA=SB=SC\)，三棱锥为正三棱锥，命题 \(\circled{1}\) 正确.

命题 \(\circled{2}\) 不正确. 例如取底面 \(A=(-1,0,0)\)，\(B=(1,0,0)\)，\(C=(0,\sqrt{3},0)\)，顶点 \(S=(0,0,1)\). 底面三边均为 \(2\)，且 \(SA=SB=\sqrt{2}\)，\(SC=2\)，所以三个侧面分别因两边相等而为等腰三角形，但 \(SA\ne SC\)，不是正三棱锥.

命题 \(\circled{3}\) 不正确. 侧面面积相等只说明顶点到底面三边所在直线的距离相等；射影 \(H\) 既可能是内心，也可能是旁心. 例如仍取上述底面，令 \(H=(2,\sqrt{3},0)\) 为一个旁心，取 \(S=(2,\sqrt{3},1)\). 点 \(H\) 到三边所在直线的距离都为 \(\sqrt{3}\)，因而 \(S\) 到三边所在直线的距离都为 \(2\)，三个侧面的面积相等，但 \(H\) 不是底面中心，三棱锥不是正三棱锥.

对于命题 \(\circled{4}\)，若三条侧棱与底面所成的角相等，则 \(SH/SA=SH/SB=SH/SC\)，从而 \(SA=SB=SC\). 又底面为等边三角形，顶点的射影为底面外心，亦即底面中心，故三棱锥为正三棱锥；命题 \(\circled{4}\) 正确. 因此真命题的编号为 \(\circled{1}\)、\(\circled{4}\).
\end{solution}

\section{解答题}

\begin{problem}
已知 \(\alpha\) 为第二象限的角，\(\sin \alpha = \frac{3}{5}\)，\(\beta\) 为第一象限的角，\(\cos \beta = \frac{5}{13}\). 求 \(\tan (2\alpha - \beta)\) 的值.
\end{problem}

\begin{solution}
由象限知 \(\cos\alpha=-\frac{4}{5}\)，所以 \(\tan\alpha=-\frac{3}{4}\)，从而 \(\tan2\alpha=\frac{2\tan\alpha}{1-\tan^{2}\alpha}=\frac{-\frac{3}{2}}{\frac{7}{16}}=-\frac{24}{7}\). 又因 \(\beta\) 为第一象限角，\(\sin\beta=\frac{12}{13}\)，故 \(\tan\beta=\frac{12}{5}\). 于是 \(\tan(2\alpha-\beta)=\frac{\tan2\alpha-\tan\beta}{1+\tan2\alpha\tan\beta}=\frac{-\frac{24}{7}-\frac{12}{5}}{1-\frac{24}{7}\cdot\frac{12}{5}}=\frac{204}{253}\).
\end{solution}

\begin{problem}
甲，乙两队进行一场排球比赛，根据以往经验，单局比赛甲队胜乙队的概率为 \(0.6\)，本场比赛采用五局三胜制，即先胜三局的队获胜，比赛结束，设各局比赛相互间没有影响. 求：

\begin{enumerate}
\item 前三局比赛甲队领先的概率；
\item 本场比赛乙队以 \(3:2\) 取胜的概率. （精确到 \(0.001\)）
\end{enumerate}
\end{problem}

\begin{solution}
设每局甲队获胜的概率为 \(p=0.6\)，乙队获胜的概率为 \(q=0.4\).

\begin{enumerate}
\item 前三局甲队领先，表示甲队前三局至少胜两局. 因此所求概率为 \(\mathrm C_{3}^{2}p^{2}q+p^{3}=3\times0.6^{2}\times0.4+0.6^{3}=0.648\).
\item 乙队以 \(3:2\) 取胜时，前四局乙队恰胜两局且第五局乙队获胜. 因此所求概率为 \(\mathrm C_{4}^{2}q^{2}p^{2}\times q=6\times0.4^{2}\times0.6^{2}\times0.4=0.13824\)，精确到 \(0.001\) 为 \(0.138\).
\end{enumerate}
\end{solution}

\begin{problem}
已知 \(\left\{a_{n}\right\}\) 是各项均为正数的等差数列，\(\lg a_{1}\)，\(\lg a_{2}\)，\(\lg a_{4}\) 成等差数列. 又 \(b_{n} = \frac{1}{a_{2^{n}}}\)，\(n = 1\)，\(2\)，\(3\)，\(\cdots\).

\begin{enumerate}
\item 证明 \(\left\{b_{n}\right\}\) 为等比数列；
\item 如果数列 \(\{b_n\}\) 前 \(3\) 项的和等于 \(\frac{7}{24}\)，求数列 \(\{a_n\}\) 的首项 \(a_1\) 和公差 \(d\).
\end{enumerate}
\end{problem}

\begin{solution}
设等差数列 \(\{a_n\}\) 的公差为 \(d\)，则 \(a_n=a_1+(n-1)d\). 由 \(\lg a_1\)，\(\lg a_2\)，\(\lg a_4\) 成等差数列，得 \(2\lg a_2=\lg a_1+\lg a_4\)，从而 \(a_2^2=a_1a_4\). 代入等差数列的通项式，得 \((a_1+d)^2=a_1(a_1+3d)\)，即 \(d(d-a_1)=0\).

\begin{enumerate}
\item 当 \(d=0\) 时，\(a_n=a_1\)，所以 \(b_n=\frac{1}{a_{2^n}}=\frac{1}{a_1}\)，这是公比为 \(1\) 的等比数列. 当 \(d=a_1\) 时，\(a_n=na_1\)，所以 \(a_{2^n}=2^na_1\)，从而 \(b_n=\frac{1}{a_1}\left(\frac{1}{2}\right)^n\)，这是公比为 \(\frac{1}{2}\) 的等比数列.
\item 当 \(d=0\) 时，\(b_1+b_2+b_3=\frac{3}{a_1}=\frac{7}{24}\)，解得 \(a_1=\frac{72}{7}\)，此时 \(d=0\). 当 \(d=a_1\) 时，\(b_1+b_2+b_3=\frac{1}{2a_1}+\frac{1}{4a_1}+\frac{1}{8a_1}=\frac{7}{8a_1}=\frac{7}{24}\)，解得 \(a_1=3\)，此时 \(d=3\). 题干未另行限定公差非零，故两组解均应保留.
\end{enumerate}
\end{solution}

\begin{problem}
如图，四棱锥 \(P - ABCD\) 中，底面 \(ABCD\) 为矩形，\(PD \perp\) 底面 \(ABCD\)，\(AD = PD\)，\(E\)，\(F\) 分别为 \(CD\)，\(PB\) 的中点.

\begin{enumerate}
\item 求证：\(EF \perp\) 平面 \(PAB\)；
\item 设 \(AB = \sqrt{2}BC\)，求 \(AC\) 与平面 \(AEF\) 所成的角的大小.
\end{enumerate}

\begin{solution}
设 \(BC=AD=PD=b\)，则 \(AB=\sqrt{2}b\). 建立直角坐标系，取 \(A=(0,0,0)\)，\(B=(\sqrt{2}b,0,0)\)，\(D=(0,b,0)\)，\(C=(\sqrt{2}b,b,0)\)，\(P=(0,b,b)\). 由中点定义，\(E=\left(\frac{\sqrt{2}b}{2},b,0\right)\)，\(F=\left(\frac{\sqrt{2}b}{2},\frac{b}{2},\frac{b}{2}\right)\).

\begin{enumerate}
\item \(\overrightarrow{EF}=\left(0,-\frac{b}{2},\frac{b}{2}\right)\)，而 \(\overrightarrow{AB}=(\sqrt{2}b,0,0)\)，\(\overrightarrow{AP}=(0,b,b)\). 于是 \(\overrightarrow{EF}\cdot\overrightarrow{AB}=0\)，且 \(\overrightarrow{EF}\cdot\overrightarrow{AP}=0\). 由于 \(\overrightarrow{AB}\) 与 \(\overrightarrow{AP}\) 是平面 \(PAB\) 内的两条相交直线的方向向量，所以 \(EF\perp\) 平面 \(PAB\).
\item 由 \(\overrightarrow{AE}=\left(\frac{\sqrt{2}b}{2},b,0\right)\)，\(\overrightarrow{AF}=\left(\frac{\sqrt{2}b}{2},\frac{b}{2},\frac{b}{2}\right)\)，可取平面 \(AEF\) 的法向量 \(\bs{n}=(2,-\sqrt{2},-\sqrt{2})\). 又 \(\overrightarrow{AC}=b(\sqrt{2},1,0)\)，所以 \(\left|\overrightarrow{AC}\cdot\bs{n}\right|=\sqrt{2}b\)，\(\left|\overrightarrow{AC}\right|=\sqrt{3}b\)，\(\left|\bs{n}\right|=2\sqrt{2}\). 设所成的角为 \(\theta\)，则 \(\sin\theta=\frac{\left|\overrightarrow{AC}\cdot\bs{n}\right|}{\left|\overrightarrow{AC}\right|\left|\bs{n}\right|}=\frac{1}{2\sqrt{3}}=\frac{\sqrt{3}}{6}\)，故 \(\theta=\arcsin\frac{\sqrt{3}}{6}\).
\end{enumerate}
\end{solution}

\begin{problem}
设 \(a\) 为实数，函数 \(f(x) = x^3 - x^2 - x + a\).

\begin{enumerate}
\item 求 \(f(x)\) 的极值；
\item 当 \(a\) 在什么范围内取值时，曲线 \(y = f(x)\) 与 \(x\) 轴仅有一个交点.
\end{enumerate}
\end{problem}

\begin{solution}
\begin{enumerate}
\item \(f'(x)=3x^{2}-2x-1=(3x+1)(x-1)\). 因此当 \(x\) 取值依次经过 \(-\frac{1}{3}\) 和 \(1\) 时，\(f'(x)\) 的符号依次为正、负、正，所以 \(x=-\frac{1}{3}\) 处为极大值点，\(x=1\) 处为极小值点. 相应极大值为 \(f\left(-\frac{1}{3}\right)=a+\frac{5}{27}\)，极小值为 \(f(1)=a-1\).
\item 当 \(a+\frac{5}{27}<0\)，即 \(a<-\frac{5}{27}\) 时，极大值小于零，曲线只在右侧上升段与 \(x\) 轴相交一次；当 \(a-1>0\)，即 \(a>1\) 时，极小值大于零，曲线只在左侧下降到极小值以前与 \(x\) 轴相交一次. 当 \(-\frac{5}{27}<a<1\) 时，极大值大于零且极小值小于零，有三个交点；当 \(a=-\frac{5}{27}\) 或 \(a=1\) 时，一个极值点在 \(x\) 轴上且另有一次穿轴相交，也有两个不同交点. 因此所求范围为 \(a<-\frac{5}{27}\) 或 \(a>1\).
\end{enumerate}
\end{solution}

\begin{problem}
\(P\)，\(Q\)，\(M\)，\(N\) 四点都在椭圆 \(x^{2} + \frac{y^{2}}{2} = 1\) 上，\(F\) 为椭圆在 \(y\) 轴正半轴上的焦点，已知 \(\overrightarrow{PF}\) 与 \(\overrightarrow{FQ}\) 共线，\(\overrightarrow{MF}\) 与 \(\overrightarrow{FN}\) 共线，且 \(\overrightarrow{PF} \cdot \overrightarrow{MF} = 0\). 求四边形 \(PMQN\) 的面积的最小值和最大值.
\end{problem}

\begin{solution}
椭圆的长半轴为 \(\sqrt{2}\)，短半轴为 \(1\)，故焦距的一半为 \(1\)，且焦点 \(F=(0,1)\). 由题意，\(PQ\) 与 \(MN\) 是过 \(F\) 的两条弦；又 \(\overrightarrow{PF}\cdot\overrightarrow{MF}=0\)，所以两条弦互相垂直，且它们是四边形 \(PMQN\) 的两条对角线.

设其中一条弦的方向单位向量为 \((\cos\theta,\sin\theta)\)，则该弦上的点可写为 \((t\cos\theta,1+t\sin\theta)\). 代入椭圆方程，得 \(\left(\cos^{2}\theta+\frac{1}{2}\sin^{2}\theta\right)t^{2}+\sin\theta\,t-\frac{1}{2}=0\). 设两根为 \(t_{1},t_{2}\)，由根的差值公式，弦长为 \(L_{1}=|t_{1}-t_{2}|=\frac{\sqrt{\sin^{2}\theta+2\cos^{2}\theta+\sin^{2}\theta}}{\cos^{2}\theta+\frac{1}{2}\sin^{2}\theta}=\frac{2\sqrt{2}}{1+\cos^{2}\theta}\).

与它垂直的弦方向可取 \((-\sin\theta,\cos\theta)\)，同理弦长为 \(L_{2}=\frac{2\sqrt{2}}{1+\sin^{2}\theta}\). 因两条对角线垂直，四边形面积为 \(S=\frac{1}{2}L_{1}L_{2}=\frac{4}{(1+\cos^{2}\theta)(1+\sin^{2}\theta)}=\frac{4}{2+\sin^{2}\theta\cos^{2}\theta}\).

由 \(0\leqslant\sin^{2}\theta\cos^{2}\theta\leqslant\frac{1}{4}\)，得 \(\frac{16}{9}\leqslant S\leqslant2\). 当 \(|\sin\theta|=|\cos\theta|\) 时取最小值 \(\frac{16}{9}\)，当 \(\sin\theta\cos\theta=0\) 时取最大值 \(2\).
\end{solution}
