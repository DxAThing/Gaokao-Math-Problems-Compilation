% Shared statistical-chart templates.  Data and ranges stay source-local;
% typography, axis-title anchoring, and chart-type defaults stay uniform.
\pgfplotsset{
  exam statistical common/.style={
    scaled ticks=false,
    every tick label/.append style={font=\normalsize},
    every axis label/.append style={font=\normalsize},
    every axis title/.append style={font=\normalsize},
    every node near coord/.append style={font=\normalsize},
  },
  exam statistical histogram/.style={exam statistical common},
  exam statistical bar/.style={exam statistical common},
  exam statistical line/.style={exam statistical common},
  exam statistical scatter/.style={exam statistical common},
  % Fixed plot-area size; labels do not change the data rectangle's aspect.
  exam statistical template axis/.style={
    exam statistical common,
    scale only axis, width=80mm, height=50mm,
    axis lines=left, axis line style={line width=0.6pt,-{Stealth}},
    tick style={line width=0.6pt}, tick align=outside,
    yticklabel style={/pgf/number format/fixed,/pgf/number format/precision=4},
    clip=false, enlargelimits=false,
    every axis plot/.append style={black,line width=0.6pt},
    legend style={font=\normalsize,draw=none},
    title style={yshift=14pt},
  },
  exam statistical template bar/.style={
    exam statistical template axis,ybar,bar width=8mm,
  },
  exam statistical template histogram/.style={
    exam statistical template axis,ybar interval,
    x tick label as interval=false,grid=none,
  },
  exam statistical template line/.style={
    exam statistical template axis,sharp plot,
  },
  exam statistical template scatter/.style={
    exam statistical template axis,only marks,mark=*,mark size=1.5pt,
  },
  exam statistical template wide/.style={width=110mm,height=68.75mm},
  exam statistical template paired/.style={width=52mm,height=50mm},
  % Narrow categorical columns retain visible gaps in a paired 52mm axis.
  exam statistical template paired bar/.style={exam statistical template paired,bar width=4mm},
  % Twelve monthly categories need a visible gap in the paired plot area.
  exam statistical template paired monthly bar/.style={exam statistical template paired,bar width=3mm},
  exam statistical template dense/.style={width=136mm,height=85mm},
  % Dense paired histograms keep vertical tick clearance without excess width.
  exam statistical template paired dense/.style={width=56mm,height=85mm},
  exam statistical template paired dense compact/.style={width=50mm,height=85mm},
  exam statistical template stacked/.style={ybar stacked,bar width=14mm},
  exam statistical template option/.style={width=24mm,height=15mm},
}
% Dense tick labels retain their true tick positions; leaders disambiguate shifts.
\newcommand{\examstatyticklabel}[4]{%
  \ifdim #4pt=0pt\relax\else
    \draw[line width=0.6pt]
      ([xshift=-1.5mm]axis cs:#1,#2) --
      ([xshift=-4mm,yshift=#4pt]axis cs:#1,#2);%
  \fi
  \node[anchor=east,inner sep=0pt,font=\normalsize]
    at ([xshift=-4mm,yshift=#4pt]axis cs:#1,#2) {#3};%
}

% pgf-pie is loaded by the chart wrapper only when needed. Shared radius is 25mm.

    style={exam statistical pie slice},#1]{#2}%
}

\newcommand{\examstatxlabel}[2][]{%
  \node[exam statistical xlabel,#1] at (current axis.right of origin) {#2};%

% Fixed-size horizontal-axis discontinuity.  Its physical dimensions do not
% grow with the data range, so every statistical chart gets the same short,
% shallow mark.  #2/#3 are the centre in axis coordinates.
\newcommand{\examstatxbreak}[3][]{%
  \draw[white,line width=3.2pt]
    ([xshift=-10pt]axis cs:#2,#3) -- ([xshift=10pt]axis cs:#2,#3);%
  \draw[line width=0.65pt,line cap=round,line join=round,#1]
    ([xshift=-9pt]axis cs:#2,#3)
    -- ++(3pt,0pt)
    -- ++(3pt,1.4pt)
    -- ++(3pt,-2.8pt)
    -- ++(3pt,2.8pt)
    -- ++(3pt,-1.4pt)
    -- ++(3pt,0pt);%
}

% Natural-size panels: horizontal when both fit the 170mm body, else vertical.
\newsavebox{\examstatleftpanel}
\newsavebox{\examstatrightpanel}
\newlength{\examstatpanelwidth}
\newcommand{\examstatpanels}[2]{%
  \begingroup
  \sbox{\examstatleftpanel}{#1}%
  \sbox{\examstatrightpanel}{#2}%
  \setlength{\examstatpanelwidth}{\linewidth}%
  \ifdim\examstatpanelwidth>170mm\setlength{\examstatpanelwidth}{170mm}\fi
  \ifdim\dimexpr\wd\examstatleftpanel+\wd\examstatrightpanel+6mm\relax>\examstatpanelwidth
    \begin{tabular}{@{}c@{}}\usebox{\examstatleftpanel}\\[6mm]\usebox{\examstatrightpanel}\end{tabular}%
  \else
    \begin{tabular}{@{}c@{\hspace{6mm}}c@{}}\usebox{\examstatleftpanel}&\usebox{\examstatrightpanel}\end{tabular}%
  \fi
  \endgroup
}
